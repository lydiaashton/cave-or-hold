import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['did_interaction'] = df['switched_branch'] * df['after_switch']

# Model 1: Simple DiD (no controls)
model1 = ols('service_score ~ switched_branch + after_switch + did_interaction', data=df).fit()
print("=== MODEL 1: Simple DiD (no controls) ===")
print(model1.summary())

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + did_interaction + staffing_index + visits_index', data=df).fit()
print("\n=== MODEL 2: DiD with controls ===")
print(model2.summary())

# Model 3: With branch fixed effects and cluster-robust SEs
# First, let's create branch dummies
df = df.copy()
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df, branch_dummies], axis=1)

# Build formula with branch FE
fe_cols = [col for col in df_with_fe.columns if col.startswith('branch_')]
fe_formula = 'service_score ~ after_switch + did_interaction + staffing_index + visits_index + ' + ' + '.join(fe_cols)
model3 = ols(fe_formula, data=df_with_fe).fit()
print("\n=== MODEL 3: With Branch Fixed Effects ===")
# Just show the key coefficients
print(f"after_switch coef: {model3.params['after_switch']:.4f}")
print(f"did_interaction coef: {model3.params['did_interaction']:.4f}")
print(f"did_interaction std err: {model3.bse['did_interaction']:.4f}")
print(f"did_interaction p-value: {model3.pvalues['did_interaction']:.4f}")

# Cluster-robust standard errors at branch level
from statsmodels.stats.sandwich_covariance import cov_cluster
cov_clustered = cov_cluster(model3, df_with_fe['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))
print(f"\nClustered SE for did_interaction: {se_clustered[list(model3.params).index('did_interaction')]:.4f}")

# Calculate t-stat and p-value with clustered SE
did_coef = model3.params['did_interaction']
did_se_clustered = se_clustered[list(model3.params).index('did_interaction')]
did_t = did_coef / did_se_clustered
from scipy import stats
did_p = 2 * (1 - stats.t.cdf(abs(did_t), df=len(df_with_fe) - len(model3.params)))
print(f"Clustered t-stat: {did_t:.3f}")
print(f"Clustered p-value: {did_p:.4f}")

# Set up panel data
df_panel = df.copy()
df_panel = df_panel.set_index(['branch_id', 'week'])

# PanelOLS with entity (branch) fixed effects and clustered standard errors
# This is the proper way to do DiD with panel data
model_panel = PanelOLS(
    df_panel['service_score'],
    df_panel[['after_switch', 'did_interaction', 'staffing_index', 'visits_index']],
    entity_effects=True,
    time_effects=False
)
result_panel = model_panel.fit(cov_type='clustered', cluster_entity=True)

print("=== PANEL DiD MODEL (Branch Fixed Effects, Clustered SEs) ===")
print(result_panel)

# Also run without controls for comparison
model_panel_simple = PanelOLS(
    df_panel['service_score'],
    df_panel[['after_switch', 'did_interaction']],
    entity_effects=True,
    time_effects=False
)
result_panel_simple = model_panel_simple.fit(cov_type='clustered', cluster_entity=True)

print("\n=== PANEL DiD MODEL (Simple, Branch FE, Clustered SEs) ===")
