import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster

# Load the data
df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']

# Run DiD with branch FE
model_fe = ols('service_score ~ after_switch + did + staffing_index + visits_index + C(branch_id)', data=df).fit()

# Get cluster-robust standard errors
cov_clustered = cov_cluster(model_fe, df['branch_id'].astype(int))
se_clustered = np.sqrt(np.diag(cov_clustered))

# Create a summary table with clustered SEs
params = model_fe.params
print("Coefficients with Cluster-Robust Standard Errors:")
print("-" * 60)
print(f"{'Variable':<20} {'Coef':>10} {'Cluster SE':>12} {'t-stat':>10} {'p-value':>10}")
print("-" * 60)

key_vars = ['after_switch', 'did', 'staffing_index', 'visits_index']
for var in key_vars:
    coef = params[var]
    se = se_clustered[model_fe.params.index.get_loc(var)]
    t_stat = coef / se
    # Approximate p-value using normal distribution
    from scipy import stats
    p_val = 2 * (1 - stats.norm.cdf(abs(t_stat)))
    print(f"{var:<20} {coef:>10.4f} {se:>12.4f} {t_stat:>10.3f} {p_val:>10.4f}")

import pandas as pd

import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')
