print("\n\nDifference-in-Differences calculation:")
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()

print(f"Treated group (switched): Before={treated_before:.2f}, After={treated_after:.2f}, Change={treated_after-treated_before:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run DiD with branch fixed effects (using dummy variables for branches)
# This is more robust as it controls for time-invariant branch characteristics
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch')]
formula = 'service_score ~ switched_branch * after_switch + staffing_index + visits_index + ' + ' + '.join(branch_cols)

model_fe = ols(formula, data=df_with_dummies).fit(cov_type='cluster', cov_kwds={'groups': df_with_dummies['branch_id']})
print("DiD with Branch Fixed Effects (clustered SEs):")
print(model_fe.summary().tables[1].iloc[:5])  # Just show first few rows

# Extract the DiD coefficient
did_coef = model_fe.params['switched_branch:after_switch']
did_se = model_fe.bse['switched_branch:after_switch']
did_pvalue = model_fe.pvalues['switched_branch:after_switch']
print(f"\nDiD coefficient: {did_coef:.3f}")
print(f"Standard error: {did_se:.3f}")
print(f"P-value: {did_pvalue:.4f}")
print(f"95% CI: [{did_coef - 1.96*did_se:.3f}, {did_coef + 1.96*did_se:.3f}]")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# With branch FEs, switched_branch is absorbed (time-invariant)
# So we only include after_switch and the interaction
model_fe = ols('service_score ~ after_switch + switched_branch:after_switch + staffing_index + visits_index + C(branch_id)', 
               data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract key coefficients
print("Key coefficients from DiD with Branch Fixed Effects:")
print("="*60)
key_vars = ['after_switch', 'switched_branch:after_switch', 'staffing_index', 'visits_index']
for var in key_vars:
    coef = model_fe.params[var]
    se = model_fe.bse[var]
    pval = model_fe.pvalues[var]
    ci_low = coef - 1.96*se
    ci_high = coef + 1.96*se
    print(f"{var:35s}: {coef:7.3f} (SE={se:.3f}, p={pval:.4f}, 95% CI [{ci_low:.3f}, {ci_high:.3f}])")

print("\n" + "="*60)
print("INTERPRETATION:")
print(f"The new procedure increased service scores by {model_fe.params['switched_branch:after_switch']:.2f} points")
print(f"on average, relative to branches that did not switch (p<0.001).")
