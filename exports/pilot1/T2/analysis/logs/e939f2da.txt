import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch fixed effects
# Model: service_score = β0 + β1*treatment + β2*switched_branch + β3*after_switch + ε
# With branch FE, we control for time-invariant branch characteristics

print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES REGRESSION ANALYSIS")
print("=" * 60)

# Model 1: Basic DiD (no fixed effects)
model1 = ols('service_score ~ treatment + switched_branch + after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n--- Model 1: Basic DiD (clustered SEs) ---")
print(model1.summary().tables[1])

# Model 2: With branch fixed effects
model2 = ols('service_score ~ treatment + after_switch + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n--- Model 2: DiD with Branch Fixed Effects ---")
print(model2.summary().tables[1])

# Model 3: With controls (staffing and visits)
model3 = ols('service_score ~ treatment + after_switch + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n--- Model 3: DiD with Branch FE + Controls ---")
print(model3.summary().tables[1])

# Extract key coefficient
did_effect = model3.params['treatment']
did_se = model3.bse['treatment']
did_pvalue = model3.pvalues['treatment']
did_ci_low = did_effect - 1.96 * did_se
did_ci_high = did_effect + 1.96 * did_se

print(f"\n=== KEY RESULT ===")
print(f"Treatment Effect (DiD): {did_effect:.3f} points")
print(f"95% CI: [{did_ci_low:.3f}, {did_ci_high:.3f}]")
print(f"P-value: {did_pvalue:.4f}")
print(f"Statistically significant: {'Yes' if did_pvalue < 0.05 else 'No'}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Event study design: estimate effect for each week relative to switch
# Create relative week variable (week 14 is the last pre-switch week, so it's the reference)
df['relative_week'] = df['week'] - 14  # week 14 = 0, week 15 = 1, etc.

# Create indicators for each relative week
for rw in range(-8, 9):  # weeks -8 to +8 relative to week 14
    df[f'rel_week_{rw}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (omitting week -1 as reference)
event_study_vars = [f'rel_week_{rw}' for rw in range(-8, 9) if rw != -1]
formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(branch_id) + C(week)'

event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== EVENT STUDY RESULTS ===")
print("Coefficients for each relative week (baseline: week -1, i.e., week 13)")
print("(Negative weeks = pre-switch, Positive weeks = post-switch)")
print()

    if rw == -1:
        results.append((rw, 0.0, 0.0, 1.0, "REF"))

    coef = event_model.params[var_name]
    se = event_model.bse[var_name]
    pval = event_model.pvalues[var_name]

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['relative_week'] = df['week'] - 14  # week 14 = 0, week 15 = 1, etc.

# Create indicators for each relative week with cleaner names (all start with letter)
for rw in range(-8, 9):
    if rw < 0:
        name = f'w{abs(rw)}'  # w8, w7, ... w1 for pre-period
    elif rw == 0:
        name = 'w0'  # week 14 (switch week reference)
    else:
        name = f'p{rw}'  # p1, p2, ... p8 for post-period
    df[name] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (omitting w1 as reference - week 13)
event_study_vars = ['w8', 'w7', 'w6', 'w5', 'w4', 'w3', 'w2', 'w0', 'p1', 'p2', 'p3', 'p4', 'p5', 'p6', 'p7', 'p8']
formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(branch_id) + C(week)'

event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== EVENT STUDY RESULTS ===")
print("Coefficients for treatment branches by week (baseline: week 13 = w1)")
print()

# Pre-period coefficients
print("PRE-SWITCH (testing parallel trends):")
pre_coefs = []
for i in range(8, 1, -1):  # w8 to w2
    var_name = f'w{i}'
    week = 14 - i
    coef = event_model.params[var_name]
    se = event_model.bse[var_name]
    pval = event_model.pvalues[var_name]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    pre_coefs.append(coef)
    print(f"  Week {week} ({var_name}): {coef:6.3f} (SE={se:.3f}, p={pval:.3f}) {sig}")

print("\nSWITCH WEEK:")
coef = event_model.params['w0']

print(f"  Week 14 (w0): {coef:6.3f} (SE={se:.3f}, p={pval:.3f}) {sig}")

post_coefs = []

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main DiD model with controls
model = ols('service_score ~ treatment + after_switch + staffing_index + visits_index + C(branch_id)', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get key statistics
did_effect = model.params['treatment']
did_se = model.bse['treatment']
did_pvalue = model.pvalues['treatment']
did_ci_low = did_effect - 1.96 * did_se
did_ci_high = did_effect + 1.96 * did_se

# Calculate effect size relative to baseline
baseline_score = df['service_score'].mean()
effect_pct = (did_effect / baseline_score) * 100

print("=" * 70)
print("FINAL ANALYSIS SUMMARY")
print("=" * 70)

print(f"   - Total observations: {len(df):,}")

print(f"   - Control branches (no switch): {df[df['switched_branch']==0]['branch_id'].nunique()}")
print(f"   - Study period: Weeks 6-22 (17 weeks)")
print(f"   - Switch occurred: Week 15")

print(f"\n📈 MAIN FINDING:")
print(f"   Treatment Effect: +{did_effect:.2f} points on service score")
print(f"   95% Confidence Interval: [{did_ci_low:.2f}, {did_ci_high:.2f}]")
print(f"   P-value: {did_pvalue:.6f}")
print(f"   Relative to average score ({baseline_score:.1f}): +{effect_pct:.1f}%")

print(f"\n⚠️  IMPORTANT CAVEATS:")
