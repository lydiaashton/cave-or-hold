import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Check treatment/control balance
print("Treatment vs Control branches:")
treatment_branches = df[df['switched_branch'] == 1]['branch_id'].nunique()
control_branches = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"Treatment branches: {treatment_branches}")
print(f"Control branches: {control_branches}")
print(f"Total branches: {treatment_branches + control_branches}")

# Check observations per group
print("\nObservations by group:")
print(df.groupby('switched_branch').size())

# Check service score by group and period
print("\nService score by group and period:")
df['period'] = df['after_switch'].map({0: 'Before', 1: 'After'})
print(df.groupby(['switched_branch', 'period'])['service_score'].agg(['mean', 'std', 'count']))

# Calculate simple DiD estimate
print("\n--- Simple Difference-in-Differences ---")
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"Treatment - Before: {before_treatment:.2f}")
print(f"Treatment - After: {after_treatment:.2f}")
print(f"Control - Before: {before_control:.2f}")
print(f"Control - After: {after_control:.2f}")

treatment_change = after_treatment - before_treatment
control_change = after_control - before_control
did_estimate = treatment_change - control_change

print(f"\nTreatment change: {treatment_change:.2f}")
print(f"Control change: {control_change:.2f}")
print(f"DiD Estimate (Treatment effect): {did_estimate:.2f}")

# Run DiD regression with robust standard errors
# Basic DiD model
model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='HC3')
print("=== Basic DiD Regression ===")
print(model1.summary())

# DiD with controls
model2 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='HC3')
print("\n=== DiD Regression with Controls ===")
print(model2.summary())

# Branch fixed effects model (more rigorous)
# This controls for time-invariant branch characteristics
df['switched_x_after'] = df['switched_branch'] * df['after_switch']
model3 = ols('service_score ~ switched_x_after + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['switched_x_after'] = df['switched_branch'] * df['after_switch']

# Final preferred model: Two-way FE (branch + week)
final_model = ols('service_score ~ switched_x_after + C(week) + staffing_index + visits_index + C(branch_id)', 
                  data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Also get the later period effect
df['post_late'] = (df['week'] >= 17).astype(int)
df['switched_x_post_late'] = df['switched_branch'] * df['post_late']
late_model = ols('service_score ~ switched_x_post_late + C(week) + staffing_index + visits_index + C(branch_id)', 
                 data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Create the visualization
plt.figure(figsize=(12, 6))

# Weekly averages by group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

plt.plot(weekly_avg.index, weekly_avg[0], 'o-', label='Control (Old Procedure)', linewidth=2.5, markersize=6, color='#E63946')
plt.plot(weekly_avg.index, weekly_avg[1], 's-', label='Treatment (New Procedure)', linewidth=2.5, markersize=6, color='#457B9D')

plt.axvline(x=14.5, color='#1D3557', linestyle='--', linewidth=2, label='Procedure Switch (Week 15)')
plt.axvspan(15, 16.5, alpha=0.2, color='orange', label='Adjustment Period')

plt.xlabel('Week', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Library Service Scores: New vs. Old Shelving Procedure', fontsize=14, fontweight='bold')
plt.legend(loc='upper left', fontsize=10)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.ylim(44, 57)

# Add annotations
plt.annotate('New procedure\nbegins', xy=(15, 52), xytext=(16, 54),
             arrowprops=dict(arrowstyle='->', color='#1D3557'), fontsize=10)
