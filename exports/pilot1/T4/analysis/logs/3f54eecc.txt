import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# DiD with branch fixed effects using dummy variables
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df.reset_index(drop=True), branch_dummies.reset_index(drop=True)], axis=1)

# Run regression with branch FEs
fe_cols = [c for c in df_with_fe.columns if c.startswith('branch_')]
formula = 'service_score ~ switched_branch + after_switch + treatment_interaction + staffing_index + visits_index + ' + ' + '.join(fe_cols)
print("Running branch FE model (this may take a moment)...")
model_fe = ols(formula, data=df_with_fe).fit(cov_type='cluster', cov_kwds={'groups': df_with_fe['branch_id']})
print("\n=== DiD with Branch Fixed Effects ===")
# Just print the key coefficients
print("Key coefficients:")
print(f"  treatment_interaction: {model_fe.params['treatment_interaction']:.4f} (SE: {model_fe.bse['treatment_interaction']:.4f}, p={model_fe.pvalues['treatment_interaction']:.4f})")
print(f"  after_switch: {model_fe.params['after_switch']:.4f} (SE: {model_fe.bse['after_switch']:.4f})")

# Event study - estimate effect for each week relative to switch
print("\n=== Event Study Analysis ===")
df['week_relative'] = df['week'] - 15  # week 15 is first post-switch week

# Create week dummies interacted with treatment
event_study_results = []
for w in range(-9, 9):  # weeks -9 to +8 relative to switch
    if w == -1:  # reference category (week 14, last pre-switch week)
        continue
    df[f'week_{w}'] = (df['week_relative'] == w).astype(int)
    df[f'treat_week_{w}'] = df['switched_branch'] * df[f'week_{w}']

# Run event study regression (omitting week -1 as reference)
treat_week_cols = [f'treat_week_{w}' for w in range(-9, 9) if w != -1]
week_cols = [f'week_{w}' for w in range(-9, 9) if w != -1]
event_formula = 'service_score ~ ' + ' + '.join(treat_week_cols) + ' + ' + ' + '.join(week_cols) + ' + staffing_index + visits_index'
event_model = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for treatment effects
event_coefs = []
event_se = []
event_weeks = []
for w in range(-9, 9):
    if w == -1:
        event_coefs.append(0)
        event_se.append(0)
        event_weeks.append(w)
    else:
        coef_name = f'treat_week_{w}'
        if coef_name in event_model.params:
            event_coefs.append(event_model.params[coef_name])
            event_se.append(event_model.bse[coef_name])
            event_weeks.append(w)

# Plot event study
fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(event_weeks, event_coefs, yerr=[1.96*s for s in event_se], fmt='o', capsize=5, label='Treatment Effect')
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='--', label='Switch Point')
ax.set_xlabel('Weeks Relative to Switch (0 = Week 15)')
ax.set_ylabel('Coefficient (Effect on Service Score)')
ax.set_title('Event Study: Treatment Effect by Week')
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("Event study plot saved as event_study.png")

# Print event study coefficients
print("\nEvent Study Coefficients:")
for w, coef, se in zip(event_weeks, event_coefs, event_se):
    marker = "***" if w != -1 and abs(coef/se) > 2.58 else "**" if w != -1 and abs(coef/se) > 1.96 else "*" if w != -1 and abs(coef/se) > 1.64 else ""
    if w == -1:
        print(f"  Week {w}: {coef:.3f} (reference)")
    else:
        print(f"  Week {w}: {coef:.3f} (SE: {se:.3f}) {marker}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create interaction term
df['treatment_interaction'] = df['switched_branch'] * df['after_switch']

# Main DiD model with controls
print("=== Main DiD Results ===")
model = ols('service_score ~ switched_branch + after_switch + treatment_interaction + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model.summary())

# Extract key results
did_coef = model.params['treatment_interaction']
did_se = model.bse['treatment_interaction']
did_pvalue = model.pvalues['treatment_interaction']
did_ci_low = model.conf_int().loc['treatment_interaction'][0]
did_ci_high = model.conf_int().loc['treatment_interaction'][1]

print(f"\n=== KEY FINDINGS ===")
print(f"DiD Estimate: {did_coef:.2f} points")
print(f"95% CI: [{did_ci_low:.2f}, {did_ci_high:.2f}]")
print(f"P-value: {did_pvalue:.4f}")
print(f"Statistical significance: {'Yes' if did_pvalue < 0.05 else 'No'}")

# Effect size relative to baseline
baseline = df[df['after_switch']==0]['service_score'].mean()
print(f"\nBaseline average service score: {baseline:.1f}")
print(f"Effect as % of baseline: {(did_coef/baseline)*100:.1f}%")

# Event study - simpler approach
print("\n=== Event Study (Simplified) ===")
df['week_relative'] = df['week'] - 15

# Calculate average treatment effect by week relative to switch
event_data = []
for w in range(-9, 9):
    week_mask = df['week_relative'] == w
    if week_mask.sum() == 0:
        continue
    week_df = df[week_mask]
    treat_mean = week_df[week_df['switched_branch']==1]['service_score'].mean()
    ctrl_mean = week_df[week_df['switched_branch']==0]['service_score'].mean()
    diff = treat_mean - ctrl_mean
    treat_n = len(week_df[week_df['switched_branch']==1])
    ctrl_n = len(week_df[week_df['switched_branch']==0])
    # Simple SE for difference in means
    treat_var = week_df[week_df['switched_branch']==1]['service_score'].var()
    ctrl_var = week_df[week_df['switched_branch']==0]['service_score'].var()
    se = np.sqrt(treat_var/treat_n + ctrl_var/ctrl_n)
    event_data.append({'week_relative': w, 'diff': diff, 'se': se, 'n_treat': treat_n, 'n_ctrl': ctrl_n})

event_df = pd.DataFrame(event_data)
print(event_df.to_string())

# Plot event study
fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(event_df['week_relative'], event_df['diff'], 
            yerr=[1.96*event_df['se']], fmt='o', capsize=5, 
            label='Treatment - Control Difference')
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='--', label='Switch Point (Week 15)')
ax.set_xlabel('Weeks Relative to Switch (0 = Week 15)')
ax.set_ylabel('Service Score Difference (Treatment - Control)')
ax.set_title('Event Study: Weekly Differences Between Groups')
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study_simple.png', dpi=150)
print("\nEvent study plot saved as event_study_simple.png")
