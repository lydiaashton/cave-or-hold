import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['DiD'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + DiD', data=df).fit()
print("=== Basic Difference-in-Differences Regression ===")
print(model.summary())

# Calculate group means for the 2x2 table
print("\n=== Group Means (2x2 Table) ===")
group_means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean()
print(group_means)

# Calculate the DiD estimate manually
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treated_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treated_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"\nControl (non-switched) before: {control_before:.2f}")
print(f"Control (non-switched) after: {control_after:.2f}")
print(f"Treatment (switched) before: {treated_before:.2f}")
print(f"Treatment (switched) after: {treated_after:.2f}")

control_change = control_after - control_before
treated_change = treated_after - treated_before
did_estimate = treated_change - control_change

print(f"\nControl change: {control_change:.2f}")
print(f"Treatment change: {treated_change:.2f}")
print(f"DiD estimate (treatment effect): {did_estimate:.2f}")

# Now add controls for staffing and visits
print("\n=== DiD Regression with Controls ===")
model_controls = ols('service_score ~ switched_branch + after_switch + DiD + staffing_index + visits_index', data=df).fit()
print(model_controls.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['DiD'] = df['switched_branch'] * df['after_switch']

# Branch fixed effects model (absorbs switched_branch since it's time-invariant)
print("=== Branch Fixed Effects Model ===")
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_fe = pd.concat([df, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [c for c in df_fe.columns if c.startswith('branch')]
formula = 'service_score ~ after_switch + DiD + staffing_index + visits_index + ' + ' + '.join(branch_cols)
model_fe = ols(formula, data=df_fe).fit(cov_type='cluster', cov_kwds={'groups': df_fe['branch_id']})
print(model_fe.summary())

# Also try with week fixed effects
print("\n=== Branch + Week Fixed Effects Model ===")
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df_fefw = pd.concat([df, branch_dummies, week_dummies], axis=1)

week_cols = [c for c in df_fefw.columns if c.startswith('week')]
# Remove after_switch since it's collinear with week FEs
formula_fefw = 'service_score ~ DiD + staffing_index + visits_index + ' + ' + '.join(branch_cols) + ' + ' + ' + '.join(week_cols)
model_fefw = ols(formula_fefw, data=df_fefw).fit(cov_type='cluster', cov_kwds={'groups': df_fefw['branch_id']})
print(model_fefw.summary())

# Event study - effects by week relative to treatment
print("\n=== Event Study Analysis ===")
df['weeks_relative'] = df['week'] - 15  # week 15 is period 0 (first post-treatment week)
df['weeks_relative'] = df['weeks_relative'].apply(lambda x: max(x, -9) if x < 0 else min(x, 7))  # cap for readability

# Create relative week dummies interacting with treatment
for wr in range(-9, 8):
    df[f'rel_week_{wr}'] = ((df['weeks_relative'] == wr) & (df['switched_branch'] == 1)).astype(int)

# Use week -1 as reference (last pre-treatment week)
rel_week_cols = [f'rel_week_{wr}' for wr in range(-9, 8) if wr != -1]
formula_event = 'service_score ~ ' + ' + '.join(rel_week_cols) + ' + staffing_index + visits_index + ' + ' + '.join(branch_cols) + ' + ' + ' + '.join(week_cols)

# This might be too many variables, let's simplify
print("Relative week coefficients (treatment effects by week):")
event_data = df.copy()
event_data = pd.concat([event_data, branch_dummies], axis=1)
week_dummies_simple = pd.get_dummies(df['week'], prefix='w', drop_first=False)
event_data = pd.concat([event_data, week_dummies_simple], axis=1)
week_cols_simple = [c for c in event_data.columns if c.startswith('w')]
branch_cols_simple = [c for c in event_data.columns if c.startswith('branch')]

# Simpler approach: just show the DiD by post-treatment week
post_data = df[df['week'] >= 15].copy()
post_data = pd.concat([post_data, branch_dummies], axis=1)
branch_cols_post = [c for c in post_data.columns if c.startswith('branch')]

# Average treatment effect in post period
post_formula = 'service_score ~ switched_branch + staffing_index + visits_index + ' + ' + '.join(branch_cols_post)
model_post = ols(post_formula, data=post_data).fit(cov_type='cluster', cov_kwds={'groups': post_data['branch_id']})
print("\nPost-treatment only model (weeks 15-22):")
print(model_post.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['DiD'] = df['switched_branch'] * df['after_switch']

# Best model: Branch + Week FE with controls
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df_fe = pd.concat([df, branch_dummies, week_dummies], axis=1)

branch_cols = [c for c in df_fe.columns if c.startswith('branch')]
week_cols = [c for c in df_fe.columns if c.startswith('week')]

# Simplified model without perfect collinearity issues
formula = 'service_score ~ DiD + staffing_index + visits_index + ' + ' + '.join(branch_cols) + ' + ' + ' + '.join(week_cols)
model = ols(formula, data=df_fe).fit(cov_type='cluster', cov_kwds={'groups': df_fe['branch_id']})

print("=== Preferred Model (Branch + Week FE + Controls) ===")
print(f"DiD coefficient: {model.params['DiD']:.2f}")
print(f"Standard error: {model.bse['DiD']:.2f}")
print(f"95% CI: [{model.conf_int().loc['DiD', 0]:.2f}, {model.conf_int().loc['DiD', 1]:.2f}]")
print(f"P-value: {model.pvalues['DiD']:.3f}")

# Calculate effect size relative to mean
mean_score = df['service_score'].mean()
effect_size_pct = (model.params['DiD'] / mean_score) * 100
print(f"\nMean service score: {mean_score:.2f}")
print(f"Effect as % of mean: {effect_size_pct:.1f}%")

# Power analysis - what effect could we detect?
n_branches = df['branch_id'].nunique()
n_treated = df['switched_branch'].sum() // 17  # approx branches
n_control = (len(df) - df['switched_branch'].sum()) // 17
print(f"\nNumber of branches: {n_branches}")
print(f"Treated branches: ~{n_treated}")
print(f"Control branches: ~{n_control}")

# Standard deviation of service score
sd_score = df['service_score'].std()
print(f"SD of service score: {sd_score:.2f}")

# Minimum detectable effect (rough approximation)
# With 80% power, alpha=0.05, two-sided
# MDE ≈ 2.8 * SE
se_did = model.bse['DiD']
mde_80 = 2.8 * se_did
print(f"\nMinimum detectable effect (80% power, approx): {mde_80:.2f} points")
print(f"As % of mean: {(mde_80/mean_score)*100:.1f}%")

# Create visualization
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
