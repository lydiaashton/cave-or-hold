import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks relative to switch at week 15)
# Week 14 is -1, week 15 is 0, etc.
df['relative_week'] = df['week'] - 15

# Create event study dummies for each week (using week 14 as reference)
# We'll interact these with switched_branch
weeks = sorted(df['week'].unique())
print("Weeks:", weeks)

# Create dummy variables for each week except the reference (week 14)
for w in weeks:
    if w != 14:  # reference period
        df[f'week_{w}'] = (df['week'] == w).astype(int)

# Event study regression - interact each week dummy with treatment
# Reference is week 14 (just before switch)
event_study_vars = [f'switched_branch:week_{w}' for w in weeks if w != 14]
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'week_{w}' for w in weeks if w != 14]) + ' + ' + ' + '.join(event_study_vars)

model_es = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results ===")
print("Coefficients for treatment interaction terms (relative to week 14):")
for w in weeks:

        se = model_es.bse[f'switched_branch:week_{w}']
        pval = model_es.pvalues[f'switched_branch:week_{w}']
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        period = "Pre" if w < 14 else "Post"
        print(f"Week {w} ({period}): coef={coef:.2f}, SE={se:.2f}, 95% CI=[{ci_low:.2f}, {ci_high:.2f}], p={pval:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run the main DiD model again to get the confidence interval
model = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

did_coef = model.params['switched_branch:after_switch']
did_se = model.bse['switched_branch:after_switch']
did_ci_low = did_coef - 1.96 * did_se
did_ci_high = did_coef + 1.96 * did_se
did_pval = model.pvalues['switched_branch:after_switch']

print("=== Final DiD Results ===")
print(f"Point estimate: {did_coef:.2f} points")
print(f"Standard error: {did_se:.2f}")
print(f"95% Confidence interval: [{did_ci_low:.2f}, {did_ci_high:.2f}]")
print(f"p-value: {did_pval:.3f}")
print()

# Minimum detectable effect (MDE) at 80% power
# Using the formula: MDE ≈ 2.8 * SE (for 80% power, two-tailed test at 5% significance)
# Actually more precisely: MDE = (z_alpha + z_beta) * SE
# z_0.025 = 1.96, z_0.20 = 0.84, so MDE ≈ 2.8 * SE
mde_80 = 2.8 * did_se
print(f"Minimum detectable effect (80% power): ~{mde_80:.2f} points")
print()

# What does this mean in practical terms?
print("=== Practical Interpretation ===")
print(f"Average service score (all branches): {df['service_score'].mean():.1f}")
print(f"Service score SD: {df['service_score'].std():.1f}")
print(f"DiD effect as % of mean: {did_coef/df['service_score'].mean()*100:.1f}%")
print(f"DiD effect as fraction of SD: {did_coef/df['service_score'].std():.2f} SD")
print()

# Power analysis - what's the probability of detecting various effect sizes?
from scipy import stats
print("=== Power for different effect sizes ===")
for effect in [1.0, 1.5, 2.0, 2.5, 3.0]:
    # z-statistic for this effect
    z = effect / did_se
    # Power = P(|Z| > 1.96 | true effect = effect)
    # = P(Z > 1.96 - z) + P(Z < -1.96 - z)
    power = 1 - stats.norm.cdf(1.96 - z) + stats.norm.cdf(-1.96 - z)
    print(f"Effect size {effect:.1f} points: Power = {power:.1%}")
