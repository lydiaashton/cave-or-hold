import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression (no controls)
model1 = ols('service_score ~ switched_branch + after_switch + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Difference-in-Differences Regression (No Controls) ===")
print(model1.summary().tables[1])

# DiD regression with controls
model2 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Difference-in-Differences Regression (With Controls) ===")
print(model2.summary().tables[1])

# Get the DiD coefficient and confidence interval
did_coef = model2.params['did']
did_ci = model2.conf_int().loc['did']
print(f"\nDiD Estimate: {did_coef:.3f}")
print(f"95% CI: [{did_ci[0]:.3f}, {did_ci[1]:.3f}]")
print(f"P-value: {model2.pvalues['did']:.4f}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Calculate average service score by week and group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_score']

# Create a plot
plt.figure(figsize=(10, 6))

# Plot for each group
for switched in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == switched]
    label = 'Switched to New Procedure' if switched == 1 else 'Old Procedure (Control)'
    plt.plot(subset['week'], subset['avg_score'], marker='o', linewidth=2, markersize=6, label=label)

# Add vertical line at switch point
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')

plt.xlabel('Week', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Scores Over Time: Treatment vs Control Groups', fontsize=14)
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_scores_over_time.png', dpi=150)
print("Plot saved as 'service_scores_over_time.png'")

# Check parallel trends - compare pre-switch trends
pre_switch = df[df['week'] < 15]
pre_trends = pre_switch.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print("\n=== Pre-Switch Weekly Averages (Parallel Trends Check) ===")
print(pre_trends)

# Calculate pre-switch trend difference
print("\n=== Pre-Switch Trend Analysis ===")
for switched in [0, 1]:
    subset = pre_switch[pre_switch['switched_branch'] == switched]
    weekly = subset.groupby('week')['service_score'].mean()
    # Simple linear trend
    weeks = weekly.index.values
    scores = weekly.values
    slope = np.polyfit(weeks, scores, 1)[0]
    print(f"Group {switched} (switched_branch={switched}): Pre-switch trend slope = {slope:.3f} points/week")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Run DiD with branch fixed effects (more robust)
# This controls for time-invariant branch characteristics
df['after_switch'] = df['after_switch'].astype(int)
df['switched_branch'] = df['switched_branch'].astype(int)

# Create dummy variables for branch fixed effects
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df, branch_dummies], axis=1)

# Build formula with branch dummies
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_')]
formula = 'service_score ~ switched_branch + after_switch + did + staffing_index + visits_index + ' + ' + '.join(branch_cols)

model_fe = ols(formula, data=df_with_dummies).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== DiD with Branch Fixed Effects ===")
print(f"DiD coefficient: {model_fe.params['did']:.3f}")
print(f"Std error: {model_fe.bse['did']:.3f}")
print(f"P-value: {model_fe.pvalues['did']:.4f}")
print(f"95% CI: [{model_fe.conf_int().loc['did'][0]:.3f}, {model_fe.conf_int().loc['did'][1]:.3f}]")

# Also check event study / dynamic effects (optional but useful)
# Create week dummies interacted with treatment
df['relative_week'] = df['week'] - 15  # weeks relative to switch (week 15 = 0)

# Check post-switch effects by week
post_switch = df[df['week'] >= 15]
print("\n=== Post-Switch Weekly Effects (Treatment vs Control) ===")
weekly_diff = post_switch.groupby('week').apply(
    lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()
)
print(weekly_diff)

print(f"Weeks analyzed: {df['week'].min()} to {df['week'].max()}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Re-run the main DiD analysis to confirm
df['did'] = df['switched_branch'] * df['after_switch']

# DiD regression with controls
model = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== CONFIRMATION: Main DiD Results ===")
print(f"DiD Coefficient: {model.params['did']:.3f}")
print(f"Standard Error: {model.bse['did']:.3f}")
print(f"95% CI: [{model.conf_int().loc['did'][0]:.3f}, {model.conf_int().loc['did'][1]:.3f}]")
print(f"P-value: {model.pvalues['did']:.6f}")

# Check pre-trends more carefully
pre_switch = df[df['week'] < 15]
print("\n=== PRE-TREND CHECK ===")
print("Weekly averages by group (before switch):")
pre_weekly = pre_switch.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(pre_weekly)

# Test for differential pre-trends
print("\n=== Pre-trend statistical test ===")
pre_switch_copy = pre_switch.copy()
pre_switch_copy['week_centered'] = pre_switch_copy['week'] - 9  # center around middle of pre-period
pre_trend_model = ols('service_score ~ switched_branch * week_centered', data=pre_switch_copy).fit(cov_type='cluster', cov_kwds={'groups': pre_switch_copy['branch_id']})
print(f"Interaction term (differential pre-trend): {pre_trend_model.params['switched_branch:week_centered']:.3f}")
print(f"P-value: {pre_trend_model.pvalues['switched_branch:week_centered']:.4f}")

# Effect size relative to baseline
baseline = df['service_score'].mean()
effect_size_pct = (model.params['did'] / baseline) * 100
print(f"\n=== EFFECT SIZE ===")
print(f"Baseline average service score: {baseline:.2f}")
print(f"DiD effect: {model.params['did']:.2f} points")
print(f"Relative improvement: {effect_size_pct:.1f}%")

# Standardized effect size (in standard deviations)
sd = df['service_score'].std()
effect_size_sd = model.params['did'] / sd
print(f"Standard deviation of service score: {sd:.2f}")
print(f"Effect size in SD units: {effect_size_sd:.2f} SD")
