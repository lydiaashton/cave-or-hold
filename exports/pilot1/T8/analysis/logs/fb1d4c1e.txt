df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary by group and period

df['period'] = df['after_switch'].map({0: 'Pre-Switch', 1: 'Post-Switch'})
df['group'] = df['switched_branch'].map({0: 'Control (Old)', 1: 'Treatment (New)'})

pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()

import pandas as pd

# Create a more detailed trend plot with confidence intervals
fig, ax = plt.subplots(figsize=(12, 6))

# Calculate weekly means and standard errors by group
weekly_stats = df.groupby(['week', 'switched_branch']).agg({
    'service_score': ['mean', 'std', 'count']
})
weekly_stats.columns = ['mean', 'std', 'count']
weekly_stats['se'] = weekly_stats['std'] / np.sqrt(weekly_stats['count'])

weeks = range(6, 23)

# Control group
control_means = weekly_stats.loc[(slice(None), 0), 'mean'].reset_index(drop=True)
control_se = weekly_stats.loc[(slice(None), 0), 'se'].reset_index(drop=True)

# Treatment group
treatment_means = weekly_stats.loc[(slice(None), 1), 'mean'].reset_index(drop=True)
treatment_se = weekly_stats.loc[(slice(None), 1), 'se'].reset_index(drop=True)

ax.plot(weeks, control_means, 'o-', label='Control (Old Procedure)', linewidth=2.5, markersize=8, color='#2E86AB')
ax.fill_between(weeks, control_means - 1.96*control_se, control_means + 1.96*control_se, alpha=0.2, color='#2E86AB')

ax.plot(weeks, treatment_means, 's-', label='Treatment (New Procedure)', linewidth=2.5, markersize=8, color='#A23B72')
ax.fill_between(weeks, treatment_means - 1.96*treatment_se, treatment_means + 1.96*treatment_se, alpha=0.2, color='#A23B72')

ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2.5, label='Switch Implementation (Week 15)')
ax.axvspan(15, 22, alpha=0.1, color='green', label='Post-Switch Period')

ax.set_xlabel('Week of Year', fontsize=13, fontweight='bold')
ax.set_ylabel('Average Service Score', fontsize=13, fontweight='bold')
