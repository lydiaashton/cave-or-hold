import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Service score summary
print("Service score summary:")
print(df['service_score'].describe())

# Average service score by group and period
print("\nAverage service score by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the DiD manually
# Before switch (after_switch=0): switched=0 vs switched=1
# After switch (after_switch=1): switched=0 vs switched=1

before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
before_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
after_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"  Treatment (switched): {before_treatment:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Basic DiD Regression:")
print(model1.summary())

# DiD with branch fixed effects
model2 = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit()
print("\n\nDiD with Branch Fixed Effects:")
print(model2.summary())

# DiD with controls and branch fixed effects
model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index + C(branch_id)', data=df).fit()
print("\n\nDiD with Branch Fixed Effects and Controls:")
print(model3.summary())

# Clustered standard errors by branch
from statsmodels.stats.sandwich_covariance import cov_cluster
model4 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index + C(branch_id)', data=df).fit()
cov_clust = cov_cluster(model4, df['branch_id'])
se_clust = np.sqrt(np.diag(cov_clust))
print("\n\nClustered SE by branch:")
for i, name in enumerate(model4.params.index):
    print(f"{name}: coef={model4.params[i]:.3f}, cluster_se={se_clust[i]:.3f}, t={model4.params[i]/se_clust[i]:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Pre-trend test (weeks 6-14)
pre_df = df[df['week'] < 15]
pre_model = ols('service_score ~ week * switched_branch', data=pre_df).fit()
print("Pre-trend test (weeks 6-14):")
print(pre_model.summary())

# Event study - effect by week relative to switch
df['weeks_relative'] = df['week'] - 14  # week 15 is week 1 relative to switch

# Create dummies for each week relative to switch, interacted with treatment
# Use simpler names without special characters
for w in range(-8, 9):  # weeks -8 to 8 relative to switch
    if w != 0:  # skip reference week
        col_name = f'rel_week{abs(w)}_{"neg" if w < 0 else "pos"}'
        df[col_name] = ((df['weeks_relative'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression with branch FE
event_vars = [f'rel_week{abs(w)}_{"neg" if w < 0 else "pos"}' for w in range(-8, 9) if w != 0]
formula = 'service_score ~ ' + ' + '.join(event_vars) + ' + C(branch_id)'
event_model = ols(formula, data=df).fit()

print("\n\nEvent Study Results (relative to week 14):")
# Extract only the relative week coefficients
rel_coefs = event_model.params[[c for c in event_model.params.index if 'rel_week' in c]]
rel_se = event_model.bse[[c for c in event_model.bse.index if 'rel_week' in c]]
rel_results = pd.DataFrame({'coef': rel_coefs, 'se': rel_se})
rel_results['week_rel'] = rel_results.index.str.extract(r'rel_week(\d+)_(neg|pos)').apply(
    lambda x: -int(x[0]) if x[1]=='neg' else int(x[0]), axis=1)
rel_results['t'] = rel_results['coef'] / rel_results['se']
rel_results['p'] = event_model.pvalues[[c for c in event_model.pvalues.index if 'rel_week' in c]]
rel_results['ci_low'] = rel_results['coef'] - 1.96 * rel_results['se']
rel_results['ci_high'] = rel_results['coef'] + 1.96 * rel_results['se']
rel_results = rel_results.sort_values('week_rel')
print(rel_results[['week_rel', 'coef', 'se', 'p', 'ci_low', 'ci_high']].round(3))

# Plot event study
plt.figure(figsize=(12, 6))
plt.errorbar(rel_results['week_rel'], rel_results['coef'], 
             yerr=1.96*rel_results['se'], fmt='o', capsize=5, 
             label='Treatment Effect', markersize=8, linewidth=2)
plt.axhline(y=0, color='red', linestyle='--', linewidth=2)
plt.axvline(x=0.5, color='gray', linestyle=':', linewidth=2, label='Switch Point')
plt.xlabel('Weeks Relative to Switch (0 = Week 14)', fontsize=12)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main DiD model with PanelOLS and clustered SE
from linearmodels.panel import PanelOLS

df_panel = df.copy()
df_panel = df_panel.set_index(['branch_id', 'week'])

mod = PanelOLS(df_panel['service_score'], 
               df_panel[['after_switch', 'treatment', 'staffing_index', 'visits_index']],
               entity_effects=True)
res = mod.fit(cov_type='clustered', cluster_entity=True)

print("=== MAIN RESULTS ===")
print(f"Treatment effect (DiD estimate): {res.params['treatment']:.2f}")
print(f"Standard error (clustered): {res.std_errors['treatment']:.2f}")
print(f"t-statistic: {res.tstats['treatment']:.2f}")
print(f"p-value: {res.pvalues['treatment']:.3f}")
print(f"95% CI: [{res.conf_int().loc['treatment'][0]:.2f}, {res.conf_int().loc['treatment'][1]:.2f}]")

# Pre-trend test
pre_df = df[df['week'] < 15]
pre_model = ols('service_score ~ week * switched_branch', data=pre_df).fit()
print(f"\nPre-trend test (week:switched_branch interaction): p={pre_model.pvalues['week:switched_branch']:.3f}")

# Event study - simpler approach
df['weeks_relative'] = df['week'] - 14

# Create cleaner variable names
event_vars_dict = {}
for w in range(-8, 9):
    if w != 0:
        sign = 'm' if w < 0 else 'p'
        col_name = f'w{sign}{abs(w)}'
        df[col_name] = ((df['weeks_relative'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_vars_dict[w] = col_name

event_vars = list(event_vars_dict.values())
formula = 'service_score ~ ' + ' + '.join(event_vars) + ' + C(branch_id)'
event_model = ols(formula, data=df).fit()

# Extract results
rel_results = []
for w, col in event_vars_dict.items():
    rel_results.append({
        'week_rel': w,
        'coef': event_model.params[col],
        'se': event_model.bse[col],
        'p': event_model.pvalues[col],
        'ci_low': event_model.params[col] - 1.96 * event_model.bse[col],
        'ci_high': event_model.params[col] + 1.96 * event_model.bse[col]
    })
rel_results = pd.DataFrame(rel_results).sort_values('week_rel')

print("\n\nEvent Study Coefficients:")
print(rel_results.round(3).to_string(index=False))

# Plot trends
plt.figure(figsize=(12, 6))
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly.columns = ['Control', 'Treatment']

plt.plot(weekly.index, weekly['Control'], 'o-', label='Control (No Switch)', linewidth=2, markersize=8)
plt.plot(weekly.index, weekly['Treatment'], 's-', label='Treatment (Switched)', linewidth=2, markersize=8)
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (Week 15)')
plt.xlabel('Week', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Score Trends: Treatment vs Control Groups', fontsize=14)
plt.legend(fontsize=11)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('trends.png', dpi=150)

# Plot event study
plt.figure(figsize=(12, 6))
plt.errorbar(rel_results['week_rel'], rel_results['coef'], 
             yerr=1.96*rel_results['se'], fmt='o', capsize=5, 
             label='Treatment Effect', markersize=8, linewidth=2)
plt.axhline(y=0, color='red', linestyle='--', linewidth=2)
plt.axvline(x=0.5, color='gray', linestyle=':', linewidth=2, label='Switch Point')
plt.xlabel('Weeks Relative to Switch (0 = Week 14)', fontsize=12)
plt.ylabel('Coefficient (vs Control)', fontsize=12)
plt.title('Event Study: Treatment Effect by Week', fontsize=14)
plt.legend(fontsize=11)
plt.grid(True, alpha=0.3)
plt.xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)

print("\n\nPlots saved: trends.png, event_study.png")
