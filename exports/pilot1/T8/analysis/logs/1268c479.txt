import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# First, let's look at the 2x2 table of means
print("=== SERVICE SCORE MEANS ===")
print("\nBefore switch (weeks 6-14):")
before = df[df['after_switch'] == 0]
print(f"  Control branches (didn't switch): {before[before['switched_branch']==0]['service_score'].mean():.2f}")
print(f"  Treatment branches (switched): {before[before['switched_branch']==1]['service_score'].mean():.2f}")

print("\nAfter switch (weeks 15-22):")
after = df[df['after_switch'] == 1]
print(f"  Control branches (didn't switch): {after[after['switched_branch']==0]['service_score'].mean():.2f}")
print(f"  Treatment branches (switched): {after[after['switched_branch']==1]['service_score'].mean():.2f}")

# Calculate the differences
control_before = before[before['switched_branch']==0]['service_score'].mean()
control_after = after[after['switched_branch']==0]['service_score'].mean()
treatment_before = before[before['switched_branch']==1]['service_score'].mean()
treatment_after = after[after['switched_branch']==1]['service_score'].mean()

control_diff = control_after - control_before
treatment_diff = treatment_after - treatment_before
did_estimate = treatment_diff - control_diff

print(f"\n=== DIFFERENCE-IN-DIFFERENCES ===")
print(f"Control change (after - before): {control_diff:.2f}")
print(f"Treatment change (after - before): {treatment_diff:.2f}")
print(f"DiD estimate (treatment change - control change): {did_estimate:.2f}")

# Now run the regression
print("\n=== REGRESSION ANALYSIS ===")

# Model 1: Basic DiD
model1 = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Model 3: With branch fixed effects
model3 = ols('service_score ~ after_switch + did + staffing_index + visits_index + C(branch_id)', data=df).fit()
print("\nModel 3: With branch fixed effects")
print(model3.summary().tables[1])
