import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# For fixed effects, we should not include treatment (time-invariant) with branch FEs
# Let's use a proper two-way fixed effects model
df['post'] = df['after_switch']
df['did'] = df['switched_branch'] * df['after_switch']

# Model with branch FEs only (treatment is absorbed)
df_with_dummies = pd.get_dummies(df, columns=['branch_id'], drop_first=True)
fe_cols = [c for c in df_with_dummies.columns if c.startswith('branch_id')]
fe_formula = 'service_score ~ post + did + ' + ' + '.join(fe_cols)
model_fe = ols(fe_formula, data=df_with_dummies).fit()
print("=== DiD with Branch Fixed Effects (treatment absorbed) ===")
print(model_fe.summary().tables[1])

# Also add week fixed effects
print("\n=== DiD with Branch and Week Fixed Effects ===")
df_week_dummies = pd.get_dummies(df, columns=['branch_id', 'week'], drop_first=True)
fe_cols = [c for c in df_week_dummies.columns if c.startswith('branch_id') or c.startswith('week')]
fe_formula2 = 'service_score ~ post + did + ' + ' + '.join(fe_cols)
model_fe2 = ols(fe_formula2, data=df_week_dummies).fit()
print(model_fe2.summary().tables[1])

# Cluster-robust standard errors at branch level
print("\n=== DiD with Branch FEs and Cluster-Robust SEs ===")
from statsmodels.stats.sandwich_covariance import cov_cluster
model_fe_robust = ols(fe_formula, data=df_with_dummies).fit(cov_type='cluster', cov_kwds={'groups': df_with_dummies['branch_id']})
