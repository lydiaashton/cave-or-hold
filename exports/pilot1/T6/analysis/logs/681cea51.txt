import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['did_interaction'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + did_interaction', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Basic Difference-in-Differences Regression ===")
print(model1.summary().tables[1])

# DiD with control variables
model2 = ols('service_score ~ switched_branch + after_switch + did_interaction + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Regression with Controls ===")
print(model2.summary().tables[1])

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study - create week dummies relative to switch (week 15)
df['week_relative'] = df['week'] - 15  # week 14 = -1, week 15 = 0, etc.

# Create dummies for each week (using week 14 as reference, the week before switch)
week_dummies = pd.get_dummies(df['week_relative'], prefix='week_rel', drop_first=False)
df_event = df.join(week_dummies)

# Drop the reference week (week_rel_-1, which is week 14)
# Also need to handle the interaction with switched_branch
event_vars = [col for col in df_event.columns if col.startswith('week_rel_')]
event_vars = [v for v in event_vars if v != 'week_rel_-1']  # reference category

# Create interaction terms
for var in event_vars:
    df_event[f'{var}_treat'] = df_event[var] * df_event['switched_branch']

# Build formula for event study
treat_interactions = [f'{v}_treat' for v in event_vars]
formula = 'service_score ~ switched_branch + ' + ' + '.join(event_vars) + ' + ' + ' + '.join(treat_interactions) + ' + staffing_index + visits_index'

model_event = ols(formula, data=df_event).fit(cov_type='cluster', cov_kwds={'groups': df_event['branch_id']})

# Extract coefficients for the interaction terms (these are the event study estimates)
print("=== Event Study Results ===")
print("Week relative to switch | Coefficient | Std Error | P-value")
print("-" * 60)

event_coefs = []
for var in sorted(event_vars):
    interact_var = f'{var}_treat'
    coef = model_event.params[interact_var]
    se = model_event.bse[interact_var]
    pval = model_event.pvalues[interact_var]
    week_num = int(var.replace('week_rel_', ''))
    event_coefs.append((week_num, coef, se, pval))
    print(f"Week {week_num:+d} (actual week {week_num+15}) | {coef:.3f} | {se:.3f} | {pval:.4f}")

# Plot event study
event_coefs_df = pd.DataFrame(event_coefs, columns=['week_relative', 'coef', 'se', 'pval'])
event_coefs_df = event_coefs_df.sort_values('week_relative')

fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(event_coefs_df['week_relative'], event_coefs_df['coef'], 
            yerr=1.96*event_coefs_df['se'], fmt='o', capsize=5, label='Treatment Effect')
ax.axhline(y=0, color='red', linestyle='--', alpha=0.5)
ax.axvline(x=-0.5, color='green', linestyle='--', alpha=0.5, label='Switch point')
ax.set_xlabel('Weeks Relative to Switch (Week 15 = 0)')
ax.set_ylabel('Effect on Service Score')
ax.set_title('Event Study: Effect of New Shelving Procedure on Service Score')
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved to event_study.png")

import pandas as pd
import numpy as np
import statsmodels.api as sm
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create week relative to switch
df['week_relative'] = df['week'] - 15  # week 14 = -1, week 15 = 0, etc.

# Create dummies for each week relative, excluding week -1 (reference)
weeks = sorted(df['week_relative'].unique())
weeks = [w for w in weeks if w != -1]  # exclude reference week

# Build design matrix manually
X = pd.DataFrame()
X['intercept'] = 1
X['switched_branch'] = df['switched_branch']

# Week dummies (main effects)
for w in weeks:
    X[f'week_{w}'] = (df['week_relative'] == w).astype(int)

# Interaction terms (treatment effect for each week)
for w in weeks:
    X[f'treat_week_{w}'] = ((df['week_relative'] == w) & (df['switched_branch'] == 1)).astype(int)

# Controls
X['staffing_index'] = df['staffing_index']
X['visits_index'] = df['visits_index']

y = df['service_score']

# Fit model
model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract treatment effect coefficients
print("=== Event Study Results ===")
print("Week relative to switch | Coefficient | Std Error | P-value | 95% CI")
print("-" * 75)

event_coefs = []
for w in sorted(weeks):
    coef = model.params[f'treat_week_{w}']
    se = model.bse[f'treat_week_{w}']
    pval = model.pvalues[f'treat_week_{w}']
    ci_low = coef - 1.96*se
    ci_high = coef + 1.96*se
    event_coefs.append((w, coef, se, pval, ci_low, ci_high))
    print(f"Week {w:+d} (actual week {w+15:2d}) | {coef:7.3f} | {se:7.3f} | {pval:7.4f} | [{ci_low:6.3f}, {ci_high:6.3f}]")

# Plot event study
event_coefs_df = pd.DataFrame(event_coefs, columns=['week_relative', 'coef', 'se', 'pval', 'ci_low', 'ci_high'])
event_coefs_df = event_coefs_df.sort_values('week_relative')

fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(event_coefs_df['week_relative'], event_coefs_df['coef'], 
            yerr=1.96*event_coefs_df['se'], fmt='o', capsize=5, label='Treatment Effect')
ax.axhline(y=0, color='red', linestyle='--', alpha=0.5)
ax.axvline(x=-0.5, color='green', linestyle='--', alpha=0.5, label='Switch point (week 15)')
ax.set_xlabel('Weeks Relative to Switch (Week 14 = reference, Week 15 = 0)')
ax.set_ylabel('Effect on Service Score')
ax.set_title('Event Study: Effect of New Shelving Procedure on Service Score')
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved to event_study.png")

# Check pre-trends more formally
print("\n=== Pre-Trend Test (weeks -8 to -1) ===")
pre_treat = event_coefs_df[event_coefs_df['week_relative'] < 0]
print(f"Number of pre-treatment weeks: {len(pre_treat)}")
print(f"Average pre-treatment coefficient: {pre_treat['coef'].mean():.3f}")
print(f"Are pre-treatment effects jointly significant?")
# Simple test: are any pre-treatment coefficients significantly different from 0?
pre_significant = pre_treat[pre_treat['pval'] < 0.05]
print(f"Number of significantly different from zero (p<0.05): {len(pre_significant)}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Basic DiD with branch fixed effects
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)

# Build X matrix
X = pd.DataFrame(index=df.index)
X['switched_branch'] = df['switched_branch']
X['after_switch'] = df['after_switch']
X['did'] = df['switched_branch'] * df['after_switch']
X['staffing_index'] = df['staffing_index']
X['visits_index'] = df['visits_index']

# Add branch fixed effects
for col in branch_dummies.columns:
    X[col] = branch_dummies[col]

y = df['service_score']

# Fit model with cluster-robust SEs
model_fe = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== DiD with Branch Fixed Effects ===")
print("Variable            | Coefficient | Std Error | P-value | 95% CI")
print("-" * 70)
for var in ['switched_branch', 'after_switch', 'did', 'staffing_index', 'visits_index']:
    coef = model_fe.params[var]
    se = model_fe.bse[var]
    pval = model_fe.pvalues[var]
    ci_low = coef - 1.96*se
    ci_high = coef + 1.96*se
    print(f"{var:20s} | {coef:11.3f} | {se:9.3f} | {pval:7.4f} | [{ci_low:6.3f}, {ci_high:6.3f}]")

# Also run without fixed effects for comparison
X_simple = pd.DataFrame()
X_simple['intercept'] = 1
X_simple['switched_branch'] = df['switched_branch']
X_simple['after_switch'] = df['after_switch']
X_simple['did'] = df['switched_branch'] * df['after_switch']
X_simple['staffing_index'] = df['staffing_index']
X_simple['visits_index'] = df['visits_index']

model_simple = sm.OLS(y, X_simple).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n=== DiD without Branch Fixed Effects ===")
print("Variable            | Coefficient | Std Error | P-value | 95% CI")
print("-" * 70)
for var in ['intercept', 'switched_branch', 'after_switch', 'did', 'staffing_index', 'visits_index']:
    coef = model_simple.params[var]
    se = model_simple.bse[var]
    pval = model_simple.pvalues[var]
    ci_low = coef - 1.96*se
    ci_high = coef + 1.96*se
    print(f"{var:20s} | {coef:11.3f} | {se:9.3f} | {pval:7.4f} | [{ci_low:6.3f}, {ci_high:6.3f}]")
