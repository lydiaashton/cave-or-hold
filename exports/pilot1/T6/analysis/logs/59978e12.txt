# Event study specification - week fixed effects interacted with treatment

# Create week dummies interacted with treatment

for w in range(6, 23):
    df[f'week_{w}'] = (df['week'] == w).astype(int)
    df[f'week_{w}_treated'] = df[f'week_{w}'] * df['switched_branch']

# Use week 14 as reference (last pre-switch week)

week_dummies = [f'week_{w}' for w in range(6, 15)] + [f'week_{w}' for w in range(15, 23)]

week_fe = [f'C(week)' for w in range(6, 23)]
formula = 'service_score ~ treated + treated_post + ' + ' + '.join([f'C(week)' for w in range(6, 23)]) + ' + staffing_index + visits_index'

formula = 'service_score ~ treated + treated_post + C(week) + staffing_index + visits_index'
model_fe = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print(model_fe.bse[['treated', 'treated_post']])
print("\nP-values:")
print(model_fe.pvalues[['treated', 'treated_post']])

branch_pre.columns = ['branch_id', 'switched_branch', 'score_pre']

branch_post.columns = ['branch_id', 'switched_branch', 'score_post']

branch_panel['score_change'] = branch_panel['score_post'] - branch_panel['score_pre']

pre_data['treated_trend'] = pre_data['treated'] * pre_data['week_centered']

pre_model = ols('service_score ~ treated + week_centered + treated_trend', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})

print("=== PARALLEL TRENDS TEST (Pre-switch period only) ===")
print(pre_model.summary().tables[1])
print(f"\nTreated x Time interaction p-value: {pre_model.pvalues['treated_trend']:.6f}")
print(f"Interpretation: If p < 0.05, parallel trends assumption is VIOLATED")

# Show the weekly gap evolution
print("\n=== Weekly Gap (Treated - Control) ===")
weekly_gap = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_gap['gap'] = weekly_gap[1] - weekly_gap[0]
print(weekly_gap[['gap']])

# Calculate pre-switch trend in the gap
pre_gap = weekly_gap[weekly_gap.index < 15]['gap']
from scipy import stats
slope, intercept, r_value, p_value, std_err = stats.linregress(pre_gap.index, pre_gap.values)
print(f"\nPre-switch gap trend: {slope:.3f} points per week (p={p_value:.4f})")

# What would we expect post-switch if pre-trend continued?
expected_week_22 = intercept + slope * 22
actual_week_22 = weekly_gap.loc[22, 'gap']
print(f"Expected gap at week 22 (if pre-trend continued): {expected_week_22:.2f}")
print(f"Actual gap at week 22: {actual_week_22:.2f}")

# Alternative: synthetic control style comparison
print("\n=== Alternative Interpretation ===")
print("The treated group was already improving relative to control BEFORE the switch.")
print("This violates the parallel trends assumption required for valid DiD.")
print("The estimated 'effect' may reflect pre-existing differences, not the procedure.")
