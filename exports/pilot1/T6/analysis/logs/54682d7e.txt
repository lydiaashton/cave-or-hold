                       index='switched_branch', 
                       columns='after_switch',

# Run the basic DiD regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is our DiD estimate

model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("Basic DiD Regression:")
print(model1.summary())

model2 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit()

print("\n\nDiD Regression with controls and cluster-robust SE (by branch):")

ax1.set_ylabel('Average Service Score')
ax1.set_title('Service Scores Over Time by Group')
ax1.legend()
ax1.grid(True, alpha=0.3)

# Panel 2: Distribution of service scores before and after
ax2 = axes[0, 1]
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score']

before_treated = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score']

ax2.set_title('Distribution of Service Scores')

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Calculate week-by-week treated-control gap
weekly_diff = df.groupby('week').apply(
    lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()
).reset_index()
weekly_diff.columns = ['week', 'gap']

# Split into pre and post periods
pre_period = weekly_diff[weekly_diff['week'] < 15]
post_period = weekly_diff[weekly_diff['week'] >= 15]

print("Week-by-week Treated-Control Gap:")
print(weekly_diff.to_string(index=False))

print("\n" + "="*60)
print("PRE-TREND ANALYSIS")
print("="*60)

# Fit a linear trend to pre-period gap
pre_period_with_trend = pre_period.copy()
pre_period_with_trend['week_centered'] = pre_period_with_trend['week'] - pre_period_with_trend['week'].mean()
pre_trend_model = ols('gap ~ week_centered', data=pre_period_with_trend).fit()
print("\nPre-period trend (weeks 6-14):")
print(f"  Slope coefficient: {pre_trend_model.params['week_centered']:.3f} points per week")
print(f"  P-value: {pre_trend_model.pvalues['week_centered']:.4f}")
print(f"  R-squared: {pre_trend_model.rsquared:.3f}")

# Calculate average pre-period gap vs post-period gap
avg_pre_gap = pre_period['gap'].mean()
avg_post_gap = post_period['gap'].mean()
print(f"\nAverage gap - Pre period (weeks 6-14): {avg_pre_gap:.2f}")
print(f"Average gap - Post period (weeks 15-22): {avg_post_gap:.2f}")
print(f"Raw difference: {avg_post_gap - avg_pre_gap:.2f}")

# Alternative: event study regression
print("\n" + "="*60)
print("EVENT STUDY REGRESSION")
print("="*60)

# Create week dummies interacted with treated
df['week_cat'] = df['week'].astype(str)
df['treated'] = df['switched_branch']

# Use week 6 as reference
event_study_formula = 'service_score ~ ' + ' + '.join([f'C(week_cat)[\'{w}\']:treated' for w in range(7, 23)]) + ' + C(week_cat)'
event_model = ols(event_study_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract the interaction coefficients (these are the week-by-week treatment effects relative to week 6)
coef_df = pd.DataFrame({
    'coef': event_model.params,
    'se': event_model.bse,
    'pvalue': event_model.pvalues
})
interaction_coefs = coef_df[coef_df.index.str.contains('treated')]
print("\nWeek-by-week treatment effects (relative to week 6 baseline):")
print(interaction_coefs)

# Visualize pre-trends
fig, ax = plt.subplots(figsize=(10, 6))
ax.plot(weekly_diff['week'], weekly_diff['gap'], 'o-', linewidth=2, markersize=8)
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')
ax.axhline(y=0, color='black', linewidth=1)
ax.set_xlabel('Week')
ax.set_ylabel('Treated - Control Gap (Service Score)')
ax.set_title('Weekly Gap Between Treated and Control Branches')
ax.legend()
ax.grid(True, alpha=0.3)
plt.savefig('pretrends_check.png', dpi=150, bbox_inches='tight')
print("\nPre-trends visualization saved to pretrends_check.png")
