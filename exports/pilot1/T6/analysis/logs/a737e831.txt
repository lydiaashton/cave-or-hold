import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Post-treatment differences
post_data = df[df['week'] >= 15]
post_weekly = post_data.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
post_diff = post_weekly[1] - post_weekly[0]
print("=== Post-Treatment Weekly Differences (Treatment - Control) ===")
print(post_diff)
print(f"\nAverage post-treatment difference: {post_diff.mean():.3f}")

# The DiD estimate should be: post_diff_avg - pre_diff_avg
pre_diff_avg = 1.059  # from previous calculation
post_diff_avg = post_diff.mean()
did_check = post_diff_avg - pre_diff_avg
print(f"\nDiD estimate (post - pre difference): {did_check:.3f}")

# Run regression with branch fixed effects (using a sample approach since 709 FE is a lot)
# Use within-branch demeaning approach
df['service_score_demeaned'] = df.groupby('branch_id')['service_score'].transform(lambda x: x - x.mean())
df['treatment_demeaned'] = df.groupby('branch_id')['treatment'].transform(lambda x: x - x.mean())
df['after_switch_demeaned'] = df.groupby('branch_id')['after_switch'].transform(lambda x: x - x.mean())
df['staffing_index_demeaned'] = df.groupby('branch_id')['staffing_index'].transform(lambda x: x - x.mean())
df['visits_index_demeaned'] = df.groupby('branch_id')['visits_index'].transform(lambda x: x - x.mean())

# Fixed effects regression (within estimator)
model_fe = ols('service_score_demeaned ~ treatment_demeaned + after_switch_demeaned + staffing_index_demeaned + visits_index_demeaned', data=df).fit()
print("\n=== Fixed Effects Regression (Within Estimator) ===")
print(model_fe.summary())

# Cluster-robust standard errors at branch level
from statsmodels.stats.sandwich_covariance import cov_cluster
cov_clustered = cov_cluster(model_fe, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))
print("\n=== Cluster-Robust Standard Errors (at branch level) ===")
print(f"Treatment coefficient: {model_fe.params['treatment_demeaned']:.4f}")
print(f"Cluster-robust SE: {se_clustered['treatment_demeaned']:.4f}")
print(f"t-statistic: {model_fe.params['treatment_demeaned'] / se_clustered['treatment_demeaned']:.3f}")
print(f"p-value: {2 * (1 - sm.stats.norm.cdf(abs(model_fe.params['treatment_demeaned'] / se_clustered['treatment_demeaned']))):.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Fixed effects regression (within estimator)
df['service_score_demeaned'] = df.groupby('branch_id')['service_score'].transform(lambda x: x - x.mean())
df['treatment_demeaned'] = df.groupby('branch_id')['treatment'].transform(lambda x: x - x.mean())
df['after_switch_demeaned'] = df.groupby('branch_id')['after_switch'].transform(lambda x: x - x.mean())
df['staffing_index_demeaned'] = df.groupby('branch_id')['staffing_index'].transform(lambda x: x - x.mean())
df['visits_index_demeaned'] = df.groupby('branch_id')['visits_index'].transform(lambda x: x - x.mean())

model_fe = ols('service_score_demeaned ~ treatment_demeaned + after_switch_demeaned + staffing_index_demeaned + visits_index_demeaned', data=df).fit()

# Cluster-robust standard errors at branch level
cov_clustered = cov_cluster(model_fe, groups=df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))
coef_names = model_fe.params.index.tolist()

print("=== Fixed Effects with Cluster-Robust SE ===")
print(f"\n{'Variable':<30} {'Coef':>10} {'SE':>10} {'t':>8} {'p-value':>10}")
print("-" * 70)
for name in coef_names:
    if name != 'Intercept':
        coef = model_fe.params[name]
        se = se_clustered[coef_names.index(name)]
        t_stat = coef / se
        p_val = 2 * (1 - sm.stats.norm.cdf(abs(t_stat)))
        print(f"{name:<30} {coef:>10.4f} {se:>10.4f} {t_stat:>8.3f} {p_val:>10.4f}")
