import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create week dummies for event study (relative to week 14 as baseline)
df['relative_week'] = df['week'] - 14  # Week 14 is the last pre-switch week

# Event study regression with week 14 as baseline
# We'll create dummies for each week relative to baseline
event_study_formula = 'service_score ~ switched_branch + ' + ' + '.join([f'C(week)' for week in range(6, 23) if week != 14]) + ' + ' + ' + '.join([f'C(week):switched_branch' for week in range(6, 23) if week != 14])

# Simpler approach: create interaction terms manually
for week in range(6, 23):
    if week != 14:  # baseline
        df[f'week_{week}'] = (df['week'] == week).astype(int)
        df[f'week_{week}_treated'] = ((df['week'] == week) & (df['switched_branch'] == 1)).astype(int)

# Build formula
interaction_terms = ' + '.join([f'week_{week}_treated' for week in range(6, 23) if week != 14])
week_terms = ' + '.join([f'week_{week}' for week in range(6, 23) if week != 14])
formula = f'service_score ~ switched_branch + {week_terms} + {interaction_terms}'

model_es = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results (week 14 as baseline) ===")
print("\nTreatment x Week interactions (DiD estimates for each week):")
for week in range(6, 23):
    if week != 14:
        coef = model_es.params[f'week_{week}_treated']
        se = model_es.bse[f'week_{week}_treated']
        pval = model_es.pvalues[f'week_{week}_treated']
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        marker = " <-- SWITCH" if week == 15 else ""
        sig = "*" if pval < 0.1 else ""
        print(f"Week {week}: coef={coef:.2f}, SE={se:.2f}, 95% CI=[{ci_low:.2f}, {ci_high:.2f}], p={pval:.3f}{sig}{marker}")

# Also run a simpler model with just post-period interaction
print("\n\n=== Simple DiD with robust SEs ===")
df['treatment_x_post'] = df['switched_branch'] * df['after_switch']
model_simple = ols('service_score ~ switched_branch + after_switch + treatment_x_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_simple.summary().tables[1])

# Calculate effect size relative to baseline
print(f"\n\nEffect size: {model_simple.params['treatment_x_post']:.2f} points")
print(f"Baseline average service score: ~50")
print(f"Effect as % of baseline: {model_simple.params['treatment_x_post']/50*100:.1f}%")
