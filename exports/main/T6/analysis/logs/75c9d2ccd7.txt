import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
switched_counts = df.groupby('branch_id')['switched_branch'].first().value_counts()
print("Branches by switch status:")
print(switched_counts)
print(f"\nTotal branches: {df['branch_id'].nunique()}")
print(f"Switched: {switched_counts.get(1, 0)}")
print(f"Did not switch: {switched_counts.get(0, 0)}")

# Check balance - do all branches have all weeks?
weeks_per_branch = df.groupby('branch_id')['week'].nunique()
print(f"\nWeeks per branch - min: {weeks_per_branch.min()}, max: {weeks_per_branch.max()}")

# Mean service scores by group and period
print("\n\n=== Mean Service Scores ===")
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='after_switch', 
                       aggfunc='mean')
print(pivot)

# Calculate the differences
print("\n=== Difference-in-Differences Calculation ===")
# Before period (after_switch=0)
before_treated = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
# After period (after_switch=1)
after_treated = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print(f"Treated - Before: {before_treated:.2f}")
print(f"Treated - After: {after_treated:.2f}")
print(f"Control - Before: {before_control:.2f}")
print(f"Control - After: {after_control:.2f}")

# Differences
diff_treated = after_treated - before_treated
diff_control = after_control - before_control
did_effect = diff_treated - diff_control

print(f"\nChange in Treated: {diff_treated:.2f}")
print(f"Change in Control: {diff_control:.2f}")
print(f"\nDiD Effect (Treated change - Control change): {did_effect:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treated'] * df['post']

# Run DiD regression with branch and week fixed effects
# First, a simple DiD without fixed effects
model_simple = ols('service_score ~ treated + post + did', data=df).fit()
print("=== Simple DiD Regression ===")
print(model_simple.summary().tables[1])

# Now with branch fixed effects
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ treated + post + did + C(branch_id_str)', data=df).fit()
print("\n=== DiD with Branch Fixed Effects ===")
print(model_fe.summary().tables[1])

# With branch and week fixed effects
df['week_str'] = df['week'].astype(str)
model_full = ols('service_score ~ treated + post + did + C(branch_id_str) + C(week_str)', data=df).fit()
print("\n=== DiD with Branch and Week Fixed Effects ===")
print(model_full.summary().tables[1])

# Check standard errors - let's cluster by branch
from statsmodels.stats.api import sandwich_covariance
cluster_se = model_full.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print("\n=== DiD with Clustered SE (by branch) ===")
print(cluster_se.summary().tables[1])

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch']

# Pre-trends: test if the difference between groups was stable before the switch
pre_data = df[df['after_switch'] == 0].copy()

# Calculate week-by-week differences
pre_pivot = pre_data.pivot_table(values='service_score', 
                                  index='week', 
                                  columns='switched_branch', 
                                  aggfunc='mean')
pre_pivot['diff'] = pre_pivot[1] - pre_pivot[0]

# Test for trend in the differences (is the gap changing over time?)
pre_data['week_centered'] = pre_data['week'] - 10  # center around middle of pre-period

# Regress service score on week, treated, and their interaction
pre_model = ols('service_score ~ week_centered + treated + week_centered:treated', data=pre_data).fit()
print("=== Pre-Trends Test ===")
print("Testing if the gap between treated and control was changing before the switch...")
print("\nPre-period regression coefficients:")
print(pre_model.summary().tables[1])

# The interaction term tells us if the gap was widening/narrowing
interaction_coef = pre_model.params['week_centered:treated']
interaction_pval = pre_model.pvalues['week_centered:treated']
print(f"\nInteraction coefficient (trend difference): {interaction_coef:.3f}")
print(f"P-value: {interaction_pval:.3f}")
