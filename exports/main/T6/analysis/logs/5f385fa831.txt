import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Verify branch counts
branches_switched = df[df['switched_branch'] == 1]['branch_id'].nunique()
branches_not_switched = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"Branches that switched: {branches_switched}")
print(f"Branches that did not switch: {branches_not_switched}")

# Create treatment interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD: average scores by group and period
print("\n=== Average Service Scores ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
pivot.columns = ['Before (weeks 6-14)', 'After (weeks 15-22)']
pivot.index = ['Control (no switch)', 'Treatment (switched)']
print(pivot)

# Calculate the DiD estimate manually
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

control_change = control_after - control_before
treatment_change = treatment_after - treatment_before
did_estimate = treatment_change - control_change

print(f"\nControl group change: {control_change:.3f}")
print(f"Treatment group change: {treatment_change:.3f}")
print(f"DiD estimate (treatment effect): {did_estimate:.3f}")

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Formal pre-trend test: interact treatment group with week (before period only)
pre_data = df[df['week'] <= 14].copy()
pre_data['week_centered'] = pre_data['week'] - 10  # center around middle of pre-period

# Test if pre-trends differ between groups
pre_trend_model = smf.ols('service_score ~ switched_branch * week_centered', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print("=== Pre-Trend Test (weeks 6-14) ===")
print(pre_trend_model.summary().tables[1])
print("\nInteraction p-value tests if trends differ between groups before treatment")

# Post-treatment trends
post_data = df[df['week'] >= 15].copy()
post_data['week_centered'] = post_data['week'] - 18  # center around middle of post-period

post_trend_model = smf.ols('service_score ~ switched_branch * week_centered', data=post_data).fit(cov_type='cluster', cov_kwds={'groups': post_data['branch_id']})
print("\n=== Post-Treatment Trend Test (weeks 15-22) ===")
print(post_trend_model.summary().tables[1])

# Create visualization data
viz_data = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
viz_data['group'] = viz_data['switched_branch'].map({0: 'Control (old procedure)', 1: 'Treatment (new procedure)'})

# Plot
plt.figure(figsize=(10, 6))
for group in viz_data['group'].unique():
    subset = viz_data[viz_data['group'] == group]
    plt.plot(subset['week'], subset['service_score'], marker='o', label=group, linewidth=2, markersize=6)

plt.axvline(x=14.5, color='gray', linestyle='--', label='Switch implemented')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Scores by Branch Group Over Time')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('service_scores_over_time.png', dpi=150)
print("\nPlot saved as 'service_scores_over_time.png'")

# Calculate effect size relative to baseline
baseline = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
effect_size_pct = (1.98 / baseline) * 100
print(f"\nEffect size: +1.98 points ({effect_size_pct:.1f}% improvement relative to control baseline)")
