import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Create relative week indicators (weeks relative to switch at week 15)
df['relative_week'] = df['week'] - 15

# Create indicators for each week relative to switch
# We'll use week 14 (one week before switch) as the reference
event_cols = []
for w in range(-9, 8):  # weeks -9 to +7 relative to week 15
    if w != -1:  # skip reference week (week 14 = relative_week -1)
        col = f'week_rel_{w}'
        df[col] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_cols.append(col)

# Run regression with event study indicators
formula = 'service_score ~ C(branch_id) + C(week) + ' + ' + '.join(event_cols)
event_model = ols(formula, data=df).fit()

print("Event Study Coefficients (treatment group relative to control, by week):")
print("="*80)
results = []
for col in event_cols:
    week_num = int(col.split('_')[-1])
    coef = event_model.params[col]
    se = event_model.bse[col]
    p = event_model.pvalues[col]
    ci_low, ci_high = event_model.conf_int().loc[col]
    results.append({
        'week': week_num,
        'coef': coef,
        'se': se,
        'p': p,
        'ci_low': ci_low,
        'ci_high': ci_high
    })
    print(f"Week {week_num:3d} (relative to switch): coef={coef:6.2f}, SE={se:.2f}, p={p:.3f}, 95% CI=[{ci_low:.2f}, {ci_high:.2f}]")

# Create event study plot
results_df = pd.DataFrame(results)
results_df = results_df.sort_values('week')

fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(results_df['week'], results_df['coef'], 
            yerr=[results_df['coef'] - results_df['ci_low'], 
                  results_df['ci_high'] - results_df['coef']],

ax.set_xlabel('Weeks Relative to Switch (Week 15 = 0)')
ax.set_ylabel('Service Score Difference (Treatment vs Control)')
ax.set_title('Event Study: Treatment Effect by Week')
ax.set_xticks(range(-9, 8))
plt.tight_layout()
