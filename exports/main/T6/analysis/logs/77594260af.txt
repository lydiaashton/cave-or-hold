)

print("\n--- Difference-in-Differences Calculation ---")

# Before switch (after_switch=0)
before_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()

# After switch (after_switch=1)
after_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"Control branches - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")

# Difference in differences
diff_switched = after_switched - before_switched
diff_control = after_control - before_control

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression
# service_score = beta0 + beta1*switched + beta2*after + beta3*(switched*after) + error
# beta3 is the DiD estimate
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Difference-in-Differences Regression Results:")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check parallel trends - compare weekly averages before the switch (weeks 6-14)
print("=== Parallel Trends Check (Pre-Switch Weeks 6-14) ===")
pre_switch = df[df['week'] < 15].copy()

weekly_avg = pre_switch.pivot_table(
    values='service_score',
    index='week',
    columns='switched_branch',
    aggfunc='mean'
)
print("Weekly averages before switch:")
print(weekly_avg)

# Calculate week-by-week differences
weekly_avg['diff'] = weekly_avg[1] - weekly_avg[0]
print("\nWeek-by-week difference (switched - control):")
print(weekly_avg['diff'])

# Run regression with branch fixed effects
print("\n\n=== Regression with Branch Fixed Effects ===")
df_fe = df.copy()
# Create branch dummies
branch_dummies = pd.get_dummies(df_fe['branch_id'], prefix='branch', drop_first=True)
df_fe = pd.concat([df_fe, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [col for col in df_fe.columns if col.startswith('branch_')]
formula = f"service_score ~ switched_branch + after_switch + treatment + {' + '.join(branch_cols)}"

model_fe = ols(formula, data=df_fe).fit(cov_type='cluster', cov_kwds={'groups': df_fe['branch_id']})
print(model_fe.summary().tables[1])
