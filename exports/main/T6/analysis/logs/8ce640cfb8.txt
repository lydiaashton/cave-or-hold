import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES ANALYSIS")
print("=" * 60)

# Model 1: Basic DiD (no controls)
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 1: Basic DiD (clustered SE by branch)")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nModel 2: DiD with controls (clustered SE by branch)")
print(model2.summary().tables[1])

# Model 3: With branch fixed effects
print("\n\nModel 3: DiD with branch fixed effects (clustered SE by branch)")
df['branch_id_str'] = df['branch_id'].astype(str)
model3 = ols('service_score ~ after_switch + treatment + staffing_index + visits_index + C(branch_id_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model3.summary().tables[1])

# Get the key coefficient
did_coef = model3.params['treatment']
did_se = model3.bse['treatment']
did_pvalue = model3.pvalues['treatment']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se

print(f"\n\nKEY RESULT:")
print(f"DiD Estimate: {did_coef:.4f}")
print(f"95% CI: [{ci_lower:.4f}, {ci_upper:.4f}]")
print(f"P-value: {did_pvalue:.4f}")
print(f"Statistically significant at 5%: {did_pvalue < 0.05}")
