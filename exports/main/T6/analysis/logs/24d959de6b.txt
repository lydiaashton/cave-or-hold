import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create DiD variables
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# DiD with covariates
model_cov = ols('service_score ~ treated + post + DiD + staffing_index + visits_index', data=df).fit()
cluster_se_cov = model_cov.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("DiD Regression with Covariates (Cluster-Robust SEs):")
print(cluster_se_cov.summary())

# Calculate means by group and week for visualization
weekly_means = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
print("\n\nWeekly means by group:")
print(weekly_means)

# Create a proper event study with relative weeks
df['relative_week'] = df['week'] - 14  # week 14 is last pre period (reference)

# Event study regression
event_df = df.copy()
event_dummies = pd.get_dummies(event_df['relative_week'], prefix='rw', drop_first=False)
event_df = pd.concat([event_df, event_dummies], axis=1)

# Drop the reference category (rw_0 which is week 14)
event_cols = [c for c in event_dummies.columns if c != 'rw_0']

# Create interaction terms
for col in event_cols:
    event_df[f'{col}_treated'] = event_df[col] * event_df['treated']

# Build formula
formula = 'service_score ~ ' + ' + '.join(event_cols) + ' + ' + ' + '.join([f'{c}_treated' for c in event_cols])
event_model = ols(formula, data=event_df).fit()
cluster_event = event_model.get_robustcov_results(cov_type='cluster', groups=event_df['branch_id'])

print("\n\nEvent Study Coefficients (Treatment x Relative Week):")
coef_df = pd.DataFrame({
    'coef': cluster_event.params[[c for c in cluster_event.params.index if '_treated' in c]],
    'se': cluster_event.bse[[c for c in cluster_event.params.index if '_treated' in c]],
    'pvalue': cluster_event.pvalues[[c for c in cluster_event.params.index if '_treated' in c]]
})
coef_df['relative_week'] = coef_df.index.str.replace('rw', '').str.replace('_treated', '').astype(int)
coef_df = coef_df.sort_values('relative_week')
print(coef_df)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create DiD variables
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Simple DiD with cluster-robust SEs
model = ols('service_score ~ treated + post + DiD', data=df).fit()
cluster_se = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("=== FINAL DiD RESULTS ===")
print(f"DiD Coefficient: {cluster_se.params['DiD']:.2f}")
print(f"Standard Error: {cluster_se.bse['DiD']:.2f}")
print(f"P-value: {cluster_se.pvalues['DiD']:.3f}")
print(f"95% CI: [{cluster_se.conf_int().loc['DiD'][0]:.2f}, {cluster_se.conf_int().loc['DiD'][1]:.2f}]")

# Create visualization
weekly_means = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
weekly_means['group'] = weekly_means['switched_branch'].map({0: 'Control (Old Procedure)', 1: 'Treatment (New Procedure)'})

fig, ax = plt.subplots(figsize=(10, 6))

for group in [0, 1]:
    subset = weekly_means[weekly_means['switched_branch'] == group]
    label = 'Treatment (New Procedure)' if group == 1 else 'Control (Old Procedure)'
    ax.plot(subset['week'], subset['service_score'], marker='o', linewidth=2, label=label, 
            color='green' if group == 1 else 'blue')

ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (Week 15)')
ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Mean Service Score', fontsize=12)
ax.set_title('Service Scores by Group Over Time', fontsize=14)
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("\nPlot saved as service_scores_trend.png")

# Pre-trend test: compare trends in pre-period
pre_df = df[df['week'] < 15].copy()
# Test if treatment and control had different slopes pre-treatment
pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()
pre_df['treated_slope'] = pre_df['week_centered'] * pre_df['treated']
pre_trend_model = ols('service_score ~ week_centered + treated + treated_slope', data=pre_df).fit()
pre_trend_cluster = pre_trend_model.get_robustcov_results(cov_type='cluster', groups=pre_df['branch_id'])

print("\n=== PRE-TREND TEST ===")
print(f"Interaction coefficient (different pre-trends): {pre_trend_cluster.params['treated_slope']:.3f}")
print(f"P-value: {pre_trend_cluster.pvalues['treated_slope']:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create DiD variables
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Simple DiD with cluster-robust SEs
model = ols('service_score ~ treated + post + DiD', data=df).fit()
cluster_se = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Extract results (params are in order: Intercept, treated, post, DiD)
did_coef = cluster_se.params[3]
did_se = cluster_se.bse[3]
did_pval = 2 * (1 - sm.stats.norm.cdf(abs(did_coef / did_se)))  # approximate p-value
conf_int = cluster_se.conf_int()[3]

print("=== FINAL DiD RESULTS ===")
print(f"DiD Coefficient: {did_coef:.2f}")
print(f"Standard Error: {did_se:.2f}")
print(f"t-statistic: {did_coef/did_se:.2f}")
print(f"P-value: {did_pval:.3f}")
print(f"95% CI: [{conf_int[0]:.2f}, {conf_int[1]:.2f}]")

# Create visualization
weekly_means = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

fig, ax = plt.subplots(figsize=(10, 6))

for group in [0, 1]:
    subset = weekly_means[weekly_means['switched_branch'] == group]
    label = 'Treatment (New Procedure)' if group == 1 else 'Control (Old Procedure)'

ax.set_xlabel('Week', fontsize=12)

ax.set_title('Service Scores by Group Over Time', fontsize=14)
ax.legend()

cluster_se = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Extract results
did_coef = cluster_se.params[3]
did_se = cluster_se.bse[3]

# Effect size relative to baseline
baseline_mean = df[df['after_switch']==0]['service_score'].mean()
print(f"\nBaseline mean service score: {baseline_mean:.1f}")
