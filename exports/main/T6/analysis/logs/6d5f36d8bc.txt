import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD table: average service scores by group and period
print("=== Difference-in-Differences Summary ===\n")

# Before period (after_switch=0)
before_switched = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()

# After period (after_switch=1)
after_switched = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print("Average Service Scores:")
print(f"  Switched branches - Before: {before_switched:.2f}, After: {after_switched:.2f}, Change: {after_switched - before_switched:.2f}")
print(f"  Control branches  - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")
print(f"\n  Difference-in-Differences: {(after_switched - before_switched) - (after_control - before_control):.2f}")

# Run DiD regression
print("\n=== DiD Regression (service_score ~ treatment + switched_branch + after_switch) ===")
model = ols('service_score ~ treatment + switched_branch + after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model.summary())

# Also run with controls
print("\n=== DiD Regression with Controls ===")
model_ctrl = ols('service_score ~ treatment + switched_branch + after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_ctrl.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create event study: interaction of switched_branch with each week indicator
# This shows the effect relative to a baseline week

# Create week dummies interacted with treatment
for w in range(6, 23):
    df[f'week_{w}'] = (df['week'] == w).astype(int)
    df[f'treat_week_{w}'] = df['switched_branch'] * df[f'week_{w}']

# Run event study regression (omit week 14 as baseline)
treat_cols = [f'treat_week_{w}' for w in range(6, 23) if w != 14]
week_cols = [f'week_{w}' for w in range(6, 23) if w != 14]

formula = 'service_score ~ ' + ' + '.join(treat_cols + week_cols + ['switched_branch'])
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract treatment effects by week
print("=== Event Study Results (relative to week 14) ===")
results = []
for w in range(6, 23):
    if w == 14:
        results.append({'week': w, 'coef': 0, 'std_err': 0, 'pvalue': np.nan, 'ci_low': 0, 'ci_high': 0})
    else:
        coef_name = f'treat_week_{w}'
        coef = model_event.params[coef_name]
        std_err = model_event.bse[coef_name]
        pval = model_event.pvalues[coef_name]
        ci_low = coef - 1.96 * std_err
        ci_high = coef + 1.96 * std_err
        results.append({'week': w, 'coef': coef, 'std_err': std_err, 'pvalue': pval, 'ci_low': ci_low, 'ci_high': ci_high})

event_df = pd.DataFrame(results)
print(event_df.to_string(index=False))

# Save the plot
plt.figure(figsize=(10, 6))
plt.errorbar(event_df['week'], event_df['coef'], yerr=1.96*event_df['std_err'], 
             fmt='o', capsize=5, label='Treatment Effect')
plt.axvline(x=14.5, color='red', linestyle='--', label='Switch Week (15)')
plt.axhline(y=0, color='gray', linestyle='-')
plt.xlabel('Week')
plt.ylabel('Service Score Effect (vs. Control)')
plt.title('Event Study: Effect of New Procedure on Service Score\n(Week 14 is baseline)')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved to event_study.png")
