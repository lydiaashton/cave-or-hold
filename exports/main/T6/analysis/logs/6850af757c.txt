ax1.set_xticks(range(6, 23))

# Plot 2: Difference between treatment and control over time
ax2 = axes[1]
diff_over_time = []
for week in range(6, 23):

    control_avg = week_data[week_data['switched_branch']==0]['service_score'].mean()

ax2.grid(True, alpha=0.3)
