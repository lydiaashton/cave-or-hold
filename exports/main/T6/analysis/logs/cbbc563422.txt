import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the treatment effect

df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Basic Difference-in-Differences Regression:")
print(model.summary())

# Add branch fixed effects (this is more robust)
# We'll use entity fixed effects
model_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit()
print("\n\nDiD Regression with Branch Fixed Effects:")
print(model_fe.summary())

# Let's also check with robust standard errors clustered by branch
from statsmodels.stats.sandwich_covariance import cov_cluster_robust

# Get the model with fixed effects and cluster by branch
model_fe_robust = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with Branch Fixed Effects (Clustered SE):")
print(model_fe_robust.summary())
