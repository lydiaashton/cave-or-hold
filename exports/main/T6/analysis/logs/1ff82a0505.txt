import pandas as pd
import statsmodels.formula.api as smf
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# Difference-in-differences regression
# The key coefficient is the interaction: switched_branch * after_switch
model = smf.ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("Difference-in-Differences Regression Results:")
print("=" * 60)
print(model.summary())

# Cluster-robust standard errors by branch
model_clustered = smf.ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nWith Cluster-Robust Standard Errors (by branch):")
print("=" * 60)
print(model_clustered.summary())

# Extract the DiD coefficient
did_coef = model_clustered.params['switched_branch:after_switch']
did_se = model_clustered.bse['switched_branch:after_switch']
did_pvalue = model_clustered.pvalues['switched_branch:after_switch']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se

print(f"\n\nKey DiD Estimate:")
print(f"Coefficient: {did_coef:.3f}")
print(f"Std Error: {did_se:.3f}")
print(f"95% CI: [{ci_lower:.3f}, {ci_upper:.3f}]")
print(f"P-value: {did_pvalue:.4f}")

import pandas as pd
import statsmodels.formula.api as smf
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create relative week variable (week 14 is the last pre-switch week = -1, week 15 = 0)
df['relative_week'] = df['week'] - 15

# Event study regression with week fixed effects
# Use week 14 (relative_week = -1) as reference
df['post_14'] = (df['relative_week'] >= 0).astype(int)

# Create dummies for each relative week
event_study_df = df.copy()
for rw in sorted(event_study_df['relative_week'].unique()):
    if rw != -1:  # -1 is reference (week 14)
        event_study_df[f'rel_week_{rw}'] = (event_study_df['relative_week'] == rw).astype(int)

# Run event study with interaction terms
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'switched_branch:rel_week_{rw}' for rw in sorted(df['relative_week'].unique()) if rw != -1])
event_model = smf.ols(formula, data=event_study_df).fit(cov_type='cluster', cov_kwds={'groups': event_study_df['branch_id']})

# Extract coefficients for the interaction terms
coef_dict = {}
se_dict = {}
for rw in sorted(df['relative_week'].unique()):
    if rw == -1:
        coef_dict[rw] = 0
        se_dict[rw] = 0
    else:
        coef_name = f'switched_branch:rel_week_{rw}'
        coef_dict[rw] = event_model.params.get(coef_name, 0)
        se_dict[rw] = event_model.bse.get(coef_name, 0)

# Plot event study
plt.figure(figsize=(10, 6))
weeks = sorted(coef_dict.keys())
coefs = [coef_dict[w] for w in weeks]
ses = [se_dict[w] for w in weeks]

plt.errorbar(weeks, coefs, yerr=[1.96*s for s in ses], fmt='o', capsize=5, 
             color='#A23B72', ecolor='#A23B72', markersize=8, linewidth=2)
plt.axhline(y=0, color='gray', linestyle='--', alpha=0.5)
plt.axvline(x=-0.5, color='gray', linestyle='-', alpha=0.7, label='Switch Point')
plt.xlabel('Relative Week (0 = First Week After Switch)', fontsize=12)
plt.ylabel('Effect on Service Score (vs Control)', fontsize=12)
plt.title('Event Study: Treatment Effect Over Time', fontsize=14, fontweight='bold')
plt.grid(True, alpha=0.3, linestyle='--')
plt.xticks(weeks)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150, bbox_inches='tight')
print("Saved event study plot")

# Print event study coefficients
print("\nEvent Study Coefficients (relative to week 14):")
for rw in weeks:
    ci_low = coef_dict[rw] - 1.96 * se_dict[rw]
    ci_high = coef_dict[rw] + 1.96 * se_dict[rw]
    sig = "***" if se_dict[rw] > 0 and abs(coef_dict[rw]/se_dict[rw]) > 2.58 else "**" if se_dict[rw] > 0 and abs(coef_dict[rw]/se_dict[rw]) > 1.96 else ""
    print(f"Week {rw+15}: {coef_dict[rw]:+.2f} (95% CI: [{ci_low:+.2f}, {ci_high:+.2f}]) {sig}")

import pandas as pd
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# The key issue: parallel trends assumption
# Let's quantify the pre-trend more clearly

pre_data = df[df['week'] < 15].copy()

# Fit a model with week as continuous to get the trend difference
pre_model = smf.ols('service_score ~ switched_branch * week', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})

print("Pre-Trend Analysis (Weeks 6-14):")
print("=" * 60)
print(f"Intercept (control branch at week 0): {pre_model.params['Intercept']:.2f}")
print(f"Control branch weekly trend: {pre_model.params['week']:.3f} points/week")
print(f"Treatment branch baseline difference: {pre_model.params['switched_branch']:.2f} points")
print(f"Treatment branch DIFFERENTIAL trend: {pre_model.params['switched_branch:week']:.3f} points/week")
print(f"  p-value for differential trend: {pre_model.pvalues['switched_branch:week']:.4f}")

# What would we expect if the pre-trend continued?
# Average pre-switch gap was 0.92
# If treatment was gaining 0.315 points/week relative to control...
# Over 8 post-switch weeks, that's 0.315 * 8 = 2.52 points of additional gap

print("\n" + "=" * 60)
print("Implications:")
print(f"Pre-trend differential: +0.315 points/week (treatment gaining on control)")
print(f"If this trend continued for 8 weeks: +{0.315*8:.2f} points expected gap increase")
print(f"Actual observed DiD estimate: +2.09 points")
print(f"Residual effect after accounting for pre-trend: {2.09 - 0.315*8:.2f} points")

# Let's also look at the weekly gaps more carefully
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
gaps = weekly_avg[1] - weekly_avg[0]

print("\n" + "=" * 60)
print("Weekly Gaps (Treatment - Control):")
for w, g in gaps.items():
    period = "PRE" if w < 15 else "POST"
    print(f"Week {w}: {g:+.2f} ({period})")

# Calculate the trend in the pre-period
import numpy as np
pre_weeks = gaps[gaps.index < 15].index.values
pre_gap_values = gaps[gaps.index < 15].values

# Linear fit to pre-trend
pre_trend_slope = np.polyfit(pre_weeks, pre_gap_values, 1)[0]
print(f"\nLinear pre-trend slope: {pre_trend_slope:.3f} points/week")

# Extrapolate to post period
post_weeks = gaps[gaps.index >= 15].index.values
expected_gaps = pre_gap_values[-1] + pre_trend_slope * (post_weeks - 14)
actual_gaps = gaps[gaps.index >= 15].values

print(f"Expected post gaps (if pre-trend continued): {expected_gaps.mean():.2f}")
print(f"Actual post gaps: {actual_gaps.mean():.2f}")
print(f"Difference: {actual_gaps.mean() - expected_gaps.mean():.2f}")
