import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Proper DiD with branch fixed effects (dropping switched_branch since it's collinear with branch FE)
# Also add week fixed effects to control for time trends
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Model with branch and week fixed effects
model_fe = ols('service_score ~ treatment_post + C(branch_id) + C(week)', data=df).fit()
print("--- DiD with Branch and Week Fixed Effects ---")
print(model_fe.summary())

# Now with robust standard errors clustered at branch level
model_fe_robust = ols('service_score ~ treatment_post + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n--- DiD with Branch and Week FE (Clustered SE at Branch Level) ---")
print(model_fe_robust.summary())

# Also check if staffing and visits are correlated with treatment
print("\n--- Covariate Balance Check ---")
print("Staffing index by group:")
print(df.groupby('switched_branch')['staffing_index'].mean())
print("\nVisits index by group:")
print(df.groupby('switched_branch')['visits_index'].mean())

# Add covariates to the model
model_with_covariates = ols('service_score ~ treatment_post + staffing_index + visits_index + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n--- DiD with Covariates, Branch and Week FE (Clustered SE) ---")
print(model_with_covariates.summary())
