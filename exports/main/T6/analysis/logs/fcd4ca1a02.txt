import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Run a simple DiD regression
# service_score = beta0 + beta1*treatment + beta2*post + beta3*did + error
# beta3 is the DiD estimator - the effect of the new procedure

model = ols('service_score ~ treatment + post + did', data=df).fit()
print("Simple DiD Regression Results:")
print(model.summary())

# Also run with branch fixed effects to control for time-invariant branch characteristics
# This is a more robust specification
df['branch_id'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ treatment + post + did + C(branch_id)', data=df).fit()
print("\n\nDiD Regression with Branch Fixed Effects:")
print(model_fe.summary())

# Calculate the DiD estimate manually to verify
print("\n\nManual DiD Calculation:")
# Treatment group: before and after
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
# Control group: before and after
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print(f"Treatment before: {treat_before:.2f}, after: {treat_after:.2f}, change: {treat_after - treat_before:.2f}")

df['post'] = df['after_switch']
df['did'] = df['switched_branch'] * df['after_switch']

# Main specification with branch and week fixed effects
model = ols('service_score ~ post + did + C(branch_id) + C(week)', data=df).fit()

print("=== VERIFICATION OF KEY NUMBERS ===\n")

# DiD coefficient
did_coef = model.params['did']
did_pvalue = model.pvalues['did']
did_ci = model.conf_int().loc['did']

print(f"DiD Coefficient: {did_coef:.2f}")
print(f"P-value: {did_pvalue:.3f}")
print(f"95% CI: [{did_ci.iloc[0]:.2f}, {did_ci.iloc[1]:.2f}]")

# Check if CI includes zero
if did_ci.iloc[0] > 0:
    print("\n✓ Confidence interval does NOT include zero (positive effect)")
else:
    print("\n⚠ Confidence interval INCLUDES zero (effect not statistically significant)")

# Check significance at different levels
print(f"\nSignificant at 0.05 level? {did_pvalue < 0.05}")
print(f"Significant at 0.01 level? {did_pvalue < 0.01}")

# Raw changes
print("\n=== RAW CHANGES ===")
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print(f"Treatment: {treat_before:.1f} → {treat_after:.1f} (change: {treat_after-treat_before:+.1f})")
print(f"Control: {control_before:.1f} → {control_after:.1f} (change: {control_after-control_before:+.1f})")
print(f"DiD: {(treat_after-treat_before) - (control_after-control_before):.1f}")

# Effect size relative to baseline
print(f"\n=== EFFECT SIZE CONTEXT ===")
print(f"Baseline average score: ~50")
print(f"Effect size: {did_coef:.1f} points = {did_coef/50*100:.1f}% improvement")
