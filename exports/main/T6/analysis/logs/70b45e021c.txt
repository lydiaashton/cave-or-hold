df['did'] = df['switched_branch'] * df['after_switch']

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check robustness - event study approach
# Look at each week's effect separately
df['did_week'] = 0
for week in range(15, 23):
    df[f'week_{week}'] = ((df['week'] == week) & (df['switched_branch'] == 1)).astype(int)

# Regression with week-specific treatment effects (relative to week 14 as baseline)
week_cols = [f'week_{w}' for w in range(15, 23)]
formula = 'service_score ~ switched_branch + week + ' + ' + '.join(week_cols)
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study - Treatment Effect by Week:")
for week in range(15, 23):
    coef = model_event.params[f'week_{week}']
    se = model_event.bse[f'week_{week}']
    pval = model_event.pvalues[f'week_{week}']
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"Week {week}: {coef:.2f} (SE: {se:.2f}, p={pval:.3f}) {sig}")

# Check if effect is consistent or driven by specific weeks
post_data = df[df['week'] >= 15]
print("\n\nPost-period average treatment effect by week:")
for week in range(15, 23):
    week_df = df[df['week'] == week]
    treatment_mean = week_df[week_df['switched_branch'] == 1]['service_score'].mean()
    control_mean = week_df[week_df['switched_branch'] == 0]['service_score'].mean()
    diff = treatment_mean - control_mean
    print(f"Week {week}: Treatment={treatment_mean:.2f}, Control={control_mean:.2f}, Diff={diff:.2f}")

# Bootstrap confidence interval for the DiD estimate
print("\n\nBootstrap check on DiD estimate:")
np.random.seed(42)
did_estimates = []
for i in range(1000):
    # Resample branches with replacement
    branches = df['branch_id'].unique()
    sampled_branches = np.random.choice(branches, size=len(branches), replace=True)
    boot_df = df[df['branch_id'].isin(sampled_branches)]
    
    control_before = boot_df[(boot_df['switched_branch'] == 0) & (boot_df['after_switch'] == 0)]['service_score'].mean()
    control_after = boot_df[(boot_df['switched_branch'] == 0) & (boot_df['after_switch'] == 1)]['service_score'].mean()
    treatment_before = boot_df[(boot_df['switched_branch'] == 1) & (boot_df['after_switch'] == 0)]['service_score'].mean()
    treatment_after = boot_df[(boot_df['switched_branch'] == 1) & (boot_df['after_switch'] == 1)]['service_score'].mean()
    
    did = (treatment_after - treatment_before) - (control_after - control_before)
    did_estimates.append(did)

print(f"Bootstrap mean DiD: {np.mean(did_estimates):.2f}")
print(f"Bootstrap 95% CI: [{np.percentile(did_estimates, 2.5):.2f}, {np.percentile(did_estimates, 97.5):.2f}]")
print(f"Bootstrap std: {np.std(did_estimates):.2f}")

import pandas as pd
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Look at the raw weekly patterns more carefully
weekly = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
weekly.columns = ['Control', 'Treatment']

# Create a cleaner visualization
fig, axes = plt.subplots(2, 1, figsize=(10, 10))

# Top panel: Raw weekly averages
ax1 = axes[0]
ax1.plot(weekly.index, weekly['Control'].values, marker='o', label='Control (No Switch)', linewidth=2, markersize=6)
ax1.plot(weekly.index, weekly['Treatment'].values, marker='s', label='Treatment (Switched)', linewidth=2, markersize=6)

ax1.axvline(x=14.5, color='red', linestyle='--', label='Switch (Week 15)', linewidth=2)
ax1.set_xlabel('Week', fontsize=12)
ax1.set_ylabel('Average Service Score', fontsize=12)
ax1.set_title('Figure 1: Raw Weekly Service Scores by Group', fontsize=14)
ax1.legend()
ax1.grid(True, alpha=0.3)
ax1.set_xticks(range(6, 23))

# Bottom panel: Difference between treatment and control each week
ax2 = axes[1]
diff = weekly['Treatment'] - weekly['Control']
ax2.bar(diff.index.astype(str), diff.values, color='steelblue', alpha=0.7)
ax2.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax2.axvline(x=8.5, color='red', linestyle='--', linewidth=2, label='Switch (Week 15)')
ax2.set_xlabel('Week', fontsize=12)
ax2.set_ylabel('Treatment - Control (points)', fontsize=12)
ax2.set_title('Figure 2: Weekly Gap Between Treatment and Control Groups', fontsize=14)
ax2.legend()
ax2.grid(True, alpha=0.3, axis='y')
ax2.set_xticks(range(0, 17))
ax2.set_xticklabels([str(i) for i in range(6, 23)], rotation=45)

plt.tight_layout()
plt.savefig('service_scores_detailed.png', dpi=150)
print("Detailed plot saved as service_scores_detailed.png")

# Pre vs post gap
pre_gap = weekly.loc[6:14, 'Treatment'] - weekly.loc[6:14, 'Control']
post_gap = weekly.loc[15:22, 'Treatment'] - weekly.loc[15:22, 'Control']

print(f"\nAverage gap (Treatment - Control):")

print(f"Change in gap: {post_gap.mean() - pre_gap.mean():.2f} points")
