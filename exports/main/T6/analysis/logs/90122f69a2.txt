import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check parallel trends: average service score by week for each group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_score']

# Pivot for easier plotting
weekly_pivot = weekly_avg.pivot(index='week', columns='switched_branch', values='avg_score')
weekly_pivot.columns = ['Control (no switch)', 'Treatment (switched)']

print("Weekly averages by group:")
print(weekly_pivot)

# Create visualization
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Plot 1: Service score trends by week
ax1 = axes[0]
ax1.plot(weekly_pivot.index, weekly_pivot['Control (no switch)'], 'o-', label='Control (no switch)', linewidth=2, markersize=8)
ax1.plot(weekly_pivot.index, weekly_pivot['Treatment (switched)'], 's-', label='Treatment (switched)', linewidth=2, markersize=8)
ax1.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch point (week 15)')
ax1.set_xlabel('Week', fontsize=12)
ax1.set_ylabel('Average Service Score', fontsize=12)
ax1.set_title('Service Score Trends by Group Over Time', fontsize=14, fontweight='bold')
ax1.legend(loc='best', fontsize=10)
ax1.grid(True, alpha=0.3)
ax1.set_xticks(range(6, 23))

# Plot 2: Distribution of service scores
ax2 = axes[1]
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score']
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score']
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score']
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score']

box_data = [before_control, after_control, before_treatment, after_treatment]
labels = ['Control\nBefore', 'Control\nAfter', 'Treatment\nBefore', 'Treatment\nAfter']
colors = ['#3498db', '#2ecc71', '#e74c3c', '#f39c12']

bp = ax2.boxplot(box_data, labels=labels, patch_artist=True, widths=0.6)
for patch, color in zip(bp['boxes'], colors):
    patch.set_facecolor(color)
    patch.set_alpha(0.7)

ax2.set_ylabel('Service Score', fontsize=12)
ax2.set_title('Service Score Distribution by Group and Period', fontsize=14, fontweight='bold')
ax2.grid(True, alpha=0.3, axis='y')

plt.tight_layout()
plt.savefig('service_score_analysis.png', dpi=150, bbox_inches='tight')
print("\nPlot saved as 'service_score_analysis.png'")

# Check parallel trends more formally - compare trends before switch
print("\n" + "=" * 60)
print("PARALLEL TRENDS CHECK (Pre-switch weeks 6-14)")
print("=" * 60)

pre_switch = df[df['week'] < 15].copy()
pre_switch['week_centered'] = pre_switch['week'] - pre_switch['week'].mean()

# Regress service score on week trend interacted with treatment group
model_trends = ols('service_score ~ switched_branch * week_centered', data=pre_switch).fit(cov_type='cluster', cov_kwds={'groups': pre_switch['branch_id']})
print(model_trends.summary())

print("\nInterpretation: The interaction term (switched_branch:week_centered) tests whether")
print("the two groups had different trends BEFORE the switch. If not significant, parallel")
print("trends assumption holds.")

print("\n" + "=" * 60)
print("TREATMENT EFFECT WITH CONFIDENCE INTERVAL")
print("=" * 60)

df['treatment'] = df['switched_branch'] * df['after_switch']
model_full = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

treatment_se = model_full.bse['treatment']
treatment_pvalue = model_full.pvalues['treatment']

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Recalculate everything from scratch to verify
df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple DiD calculation
means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean()
print("Mean service scores by group:")
print(means)
print()

# DiD = (Treatment_After - Treatment_Before) - (Control_After - Control_Before)
did = (means.loc[(1, 1)] - means.loc[(1, 0)]) - (means.loc[(0, 1)] - means.loc[(0, 0)])
print(f"Simple DiD estimate: {did:.3f}")
print()

# Regression with clustered SE
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Regression results:")
print(f"  Treatment coefficient: {model.params['treatment']:.3f}")
print(f"  Standard error: {model.bse['treatment']:.3f}")
print(f"  P-value: {model.pvalues['treatment']:.4f}")
print(f"  95% CI: [{model.params['treatment'] - 1.96*model.bse['treatment']:.3f}, {model.params['treatment'] + 1.96*model.bse['treatment']:.3f}]")
print()

# Check if effect is practically significant
baseline = df[df['after_switch'] == 0]['service_score'].mean()
print(f"Baseline mean service score: {baseline:.2f}")
print(f"Effect as % of baseline: {(model.params['treatment'] / baseline) * 100:.1f}%")
print()

# Check the distribution of effects
print("Number of branches:")
print(df.groupby('switched_branch')['branch_id'].nunique())
print()

# Check for any red flags in the data
print("Service score range:", df['service_score'].min(), "to", df['service_score'].max())
print("Any missing values?", df.isnull().sum().sum())
