import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Pre-switch period: weeks 6-14 (after_switch = 0)
# Post-switch period: weeks 15-22 (after_switch = 1)

print("=== Service Scores by Group and Period ===")

# Before switch (weeks 6-14)
before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

print("\nBEFORE switch (weeks 6-14):")
print(f"  Switched branches: mean={before[before['switched_branch']==1]['service_score'].mean():.2f}, n={len(before[before['switched_branch']==1])}")
print(f"  Non-switched branches: mean={before[before['switched_branch']==0]['service_score'].mean():.2f}, n={len(before[before['switched_branch']==0])}")

print("\nAFTER switch (weeks 15-22):")
print(f"  Switched branches: mean={after[after['switched_branch']==1]['service_score'].mean():.2f}, n={len(after[after['switched_branch']==1])}")
print(f"  Non-switched branches: mean={after[after['switched_branch']==0]['service_score'].mean():.2f}, n={len(after[after['switched_branch']==0])}")

# Calculate differences
switched_before = before[before['switched_branch']==1]['service_score'].mean()
switched_after = after[after['switched_branch']==1]['service_score'].mean()
control_before = before[before['switched_branch']==0]['service_score'].mean()
control_after = after[after['switched_branch']==0]['service_score'].mean()

diff_switched = switched_after - switched_before
diff_control = control_after - control_before
did_estimate = diff_switched - diff_control

print(f"\n=== Difference-in-Differences Estimate ===")
print(f"Change in switched branches: {diff_switched:.2f}")
print(f"Change in control branches: {diff_control:.2f}")
print(f"DiD Estimate (treatment effect): {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is our treatment effect

model = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== DiD Regression Results ===")
print(model.summary())

# Extract the interaction term (treatment effect)
treatment_effect = model.params['switched_branch:after_switch']
treatment_pvalue = model.pvalues['switched_branch:after_switch']
print(f"\nTreatment effect: {treatment_effect:.3f}")
print(f"P-value: {treatment_pvalue:.4f}")

# Run DiD with branch fixed effects
# This controls for time-invariant branch characteristics

print("=== DiD with Branch Fixed Effects ===")

            subset = df[(df['switched_branch']==1) & (df['after_switch'] == (0 if period=='before' else 1))]
        else:
            subset = df[(df['switched_branch']==0) & (df['after_switch'] == (0 if period=='before' else 1))]
        print(f"   {group.capitalize()} branches, {period}: {subset['service_score'].mean():.2f}")

# 5. Probability the effect is positive
# Using the estimate and SE, we can approximate this
from scipy import stats
z = te / te_se
prob_positive = 1 - stats.norm.cdf(0, loc=te, scale=te_se)
print(f"\n5. Probability the true effect is positive: ~{prob_positive*100:.0f}%")
print(f"   (Based on normal approximation of the sampling distribution)")

# 6. Power analysis - what effect could we detect?
n_treatment = 340
n_control = 238
overall_std = df['service_score'].std()
# Approximate MDE for 80% power at alpha=0.05
mde = 2.8 * overall_std * np.sqrt(1/n_treatment + 1/n_control)
print(f"\n6. Minimum detectable effect (80% power): ~{mde:.2f} points")
print(f"   - Our estimate ({te:.2f}) is below this threshold")
print(f"   - We may be underpowered to detect the true effect")
