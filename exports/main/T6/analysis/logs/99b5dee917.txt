import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check balance between treatment and control groups
print("Number of branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate average service scores by group and period
print("\n=== Average Service Scores ===")
print("\nBefore switch (weeks 6-14, after_switch=0):")
before = df[df['after_switch']==0].groupby('switched_branch')['service_score'].mean()
print(before)

print("\nAfter switch (weeks 15-22, after_switch=1):")
after = df[df['after_switch']==1].groupby('switched_branch')['service_score'].mean()
print(after)

# Calculate the differences
print("\n=== Difference-in-Differences Calculation ===")
control_before = before[0]
control_after = after[0]
treatment_before = before[1]
treatment_after = after[1]

control_diff = control_after - control_before
treatment_diff = treatment_after - treatment_before
did_estimate = treatment_diff - control_diff

print(f"Control group (no switch): {control_before:.2f} -> {control_after:.2f} (change: {control_diff:.2f})")
print(f"Treatment group (switch): {treatment_before:.2f} -> {treatment_after:.2f} (change: {treatment_diff:.2f})")
print(f"\nDifference-in-Differences estimate: {did_estimate:.3f}")

# Run a regression for DiD
print("\n=== Regression Analysis ===")
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model.summary())

# Check pre-trends (weeks 6-14 only)
print("\n=== Pre-Treatment Trends Check ===")
pre_data = df[df['after_switch']==0]
pre_model = ols('service_score ~ switched_branch * week', data=pre_data).fit()
print(pre_model.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Create a summary table for the board
print("=" * 70)
print("SUMMARY TABLE FOR BOARD")
print("=" * 70)

# Group sizes
n_treatment = df[df['switched_branch']==1]['branch_id'].nunique()
n_control = df[df['switched_branch']==0]['branch_id'].nunique()
print(f"\nNumber of branches:")
print(f"  Treatment (switched): {n_treatment}")
print(f"  Control (no switch): {n_control}")

# Pre-post means
pre_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
post_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
pre_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
post_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print(f"\nAverage Service Scores:")
print(f"  Treatment group: {pre_treatment:.2f} (before) -> {post_treatment:.2f} (after), change: {post_treatment - pre_treatment:.2f}")
print(f"  Control group:   {pre_control:.2f} (before) -> {post_control:.2f} (after), change: {post_control - pre_control:.2f}")

# Main DiD result
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
did_coef = model.params['treatment']
did_se = model.bse['treatment']
did_pvalue = model.pvalues['treatment']
ci_lower = model.conf_int().loc['treatment', 0]
ci_upper = model.conf_int().loc['treatment', 1]

print(f"\n=== MAIN RESULT ===")
print(f"Difference-in-Differences Estimate: {did_coef:.2f} points")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {did_pvalue:.3f}")
print(f"Statistically significant: {'Yes' if did_pvalue < 0.05 else 'No'}")

# Effect size relative to baseline
print(f"\nEffect size: {did_coef:.2f} points is about {100*did_coef/pre_treatment:.1f}% of baseline treatment group score")

# Check parallel trends more formally
print("\n=== PARALLEL TRENDS TEST ===")
pre_data = df[df['after_switch']==0]
pre_model = ols('service_score ~ switched_branch * week', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
interaction_pvalue = pre_model.pvalues['switched_branch:week']
print(f"Pre-treatment trend interaction p-value: {interaction_pvalue:.3f}")
print(f"Parallel trends assumption satisfied: {'Yes' if interaction_pvalue > 0.05 else 'No'}")

# Create the trend plot
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
plt.figure(figsize=(12, 6))
for switched in [0, 1]:

    label = 'Control (No Switch)' if switched == 0 else 'Treatment (Switched)'

plt.xlabel('Week', fontsize=12)

print("\nTrend plot saved as 'service_score_trends.png'")

# Additional robustness: check if effect varies over time
print("\n=== EFFECT PERSISTENCE ===")

df['post_late'] = ((df['after_switch']==1) & (df['week'] > 18)).astype(int)

print("=" * 70)
print("CRITICAL RE-EXAMINATION")

# 1. Check baseline differences
print("\n=== BASELINE DIFFERENCES ===")
pre_data = df[df['after_switch']==0]

print(f"  Treatment branches: {pre_data[pre_data['switched_branch']==1]['service_score'].mean():.2f}")
print(f"  Control branches:   {pre_data[pre_data['switched_branch']==0]['service_score'].mean():.2f}")

print("\n=== BRANCH SELECTION BIAS CHECK ===")
branch_avg_pre = pre_data.groupby('branch_id')['service_score'].mean().reset_index()

print("Average pre-switch score by switch status:")
print(branch_avg_pre.groupby('switched')['service_score'].describe())

# 3. Effect size context
print("\n=== EFFECT SIZE CONTEXT ===")
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

overall_std = df['service_score'].std()
print(f"DiD estimate: {did:.2f} points")
print(f"Overall service score SD: {overall_std:.2f} points")
print(f"Effect in SD units: {did/overall_std:.2f} SD")

# Typical Cohen's d interpretations
if did/overall_std < 0.2:
    print("Interpretation: SMALL effect (Cohen's d < 0.2)")
elif did/overall_std < 0.5:
    print("Interpretation: MEDIUM effect (Cohen's d 0.2-0.5)")

    print("Interpretation: LARGE effect (Cohen's d > 0.5)")
