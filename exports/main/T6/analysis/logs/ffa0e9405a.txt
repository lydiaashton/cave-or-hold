import pandas as pd
import numpy as np
import statsmodels.api as sm
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Simple DiD regression
# service_score = beta0 + beta1*treatment + beta2*post + beta3*did + error
# beta3 is the treatment effect

X = df[['treatment', 'post', 'did']]
X = sm.add_constant(X)
y = df['service_score']

model = sm.OLS(y, X).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# DiD with controls (staffing and visits)
X_ctrl = df[['treatment', 'post', 'did', 'staffing_index', 'visits_index']]
X_ctrl = sm.add_constant(X_ctrl)
model_ctrl = sm.OLS(y, X_ctrl).fit()
print("\n=== DiD Regression with Controls ===")
print(model_ctrl.summary())

# Clustered standard errors by branch
print("\n=== DiD with Clustered SE (by branch) ===")
model_cluster = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_cluster.summary())

import pandas as pd
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# Event study / dynamic DiD - check effects by week
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']

# Create week dummies interacted with treatment
# Use week 14 as reference (last pre-switch week)
weeks = list(range(6, 23))
for w in weeks:
    if w != 14:  # reference week
        df[f'week_{w}'] = (df['week'] == w).astype(int)
        df[f'week_{w}_treat'] = df[f'week_{w}'] * df['treatment']

# Run regression with all week-treatment interactions
treat_week_cols = [f'week_{w}_treat' for w in weeks if w != 14]
week_cols = [f'week_{w}' for w in weeks if w != 14]

X = df[treat_week_cols + week_cols + ['treatment']]
X = sm.add_constant(X)
y = df['service_score']

model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study: Treatment Effect by Week ===")
print("Week 14 is the reference (last pre-switch week)")
print("\nCoefficient (Treatment vs Control difference relative to week 14):")
for w in weeks:
    if w != 14:
        coef = model.params[f'week_{w}_treat']
        pval = model.pvalues[f'week_{w}_treat']
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"Week {w}: {coef:6.2f} (p={pval:.3f}) {sig}")

# Check if pre-trends are flat (weeks 6-14 should not be significantly different from 0)
print("\n=== Pre-Trend Test (Weeks 6-13 vs Week 14) ===")
pre_weeks = [w for w in weeks if w < 14 and w != 14]
pre_coefs = [model.params[f'week_{w}_treat'] for w in pre_weeks]
pre_pvals = [model.pvalues[f'week_{w}_treat'] for w in pre_weeks]
print(f"Pre-trend coefficients range: {min(pre_coefs):.2f} to {max(pre_coefs):.2f}")
print(f"All pre-trend p-values > 0.05: {all(p > 0.05 for p in pre_pvals)}")

print("   This VIOLATES a core DiD assumption.\n")

import pandas as pd

df = pd.read_csv('branches.csv')

# For weeks 19-22, all observations are after_switch=1
# So we need to compare to the pre-switch baseline differently

print("=== More Careful Robustness Checks ===\n")

# Simple comparison: treatment vs control in late post period
late_post = df[df['week'] >= 19]
late_treatment = late_post[late_post['switched_branch'] == 1]['service_score'].mean()
late_control = late_post[late_post['switched_branch'] == 0]['service_score'].mean()
print(f"Weeks 19-22: Treatment avg = {late_treatment:.2f}, Control avg = {late_control:.2f}")
print(f"             Difference = {late_treatment - late_control:.2f}\n")

# But we need to account for pre-existing differences
# Treatment pre-average
pre_treatment = df[(df['switched_branch'] == 1) & (df['week'] < 15)]['service_score'].mean()
pre_control = df[(df['switched_branch'] == 0) & (df['week'] < 15)]['service_score'].mean()
print(f"Pre-switch (weeks 6-14): Treatment avg = {pre_treatment:.2f}, Control avg = {pre_control:.2f}")
print(f"                         Difference = {pre_treatment - pre_control:.2f}\n")

# The DiD is (post_diff - pre_diff)
post_diff = late_treatment - late_control
pre_diff = pre_treatment - pre_control
did_late = post_diff - pre_diff
print(f"DiD for weeks 19-22: {post_diff:.2f} - ({pre_diff:.2f}) = {did_late:.2f}")

# Also check if the effect is consistent across all post weeks
print("\n=== Post-Switch Weekly Differences (Treatment - Control) ===")
post_weeks = df[df['week'] >= 15]
for w in range(15, 23):
    week_data = post_weeks[post_weeks['week'] == w]
    t_avg = week_data[week_data['switched_branch'] == 1]['service_score'].mean()
    c_avg = week_data[week_data['switched_branch'] == 0]['service_score'].mean()
    diff = t_avg - c_avg
    print(f"Week {w}: Treatment - Control = {diff:+.2f}")
