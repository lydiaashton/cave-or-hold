import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# Create groups for analysis
# Group 1: Switched branches, before switch
# Group 2: Switched branches, after switch
# Group 3: Non-switched branches, before switch
# Group 4: Non-switched branches, after switch

# Calculate mean service scores for each group
grouped = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print("Service scores by group:")
print(grouped)
print()

# Calculate the difference-in-differences
# For switched branches: after - before
switched_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
switched_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
non_switched_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
non_switched_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print("Mean service scores:")
print(f"Switched branches - Before: {switched_before:.3f}, After: {switched_after:.3f}, Change: {switched_after - switched_before:.3f}")
print(f"Non-switched branches - Before: {non_switched_before:.3f}, After: {non_switched_after:.3f}, Change: {non_switched_after - non_switched_before:.3f}")
print()

# Difference-in-differences estimate
did_estimate = (switched_after - switched_before) - (non_switched_after - non_switched_before)
print(f"Difference-in-Differences estimate: {did_estimate:.3f}")
print()

# Let's also look at the weekly trends
weekly_trends = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
print("Weekly trends (service score by week and switch status):")
print(weekly_trends)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched + beta2*after + beta3*(switched*after) + error
# beta3 is the DiD estimate

df['switched_after'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + switched_after', data=df).fit()
print("Basic DiD Regression:")
print(model.summary())
print()

# Add branch fixed effects (this controls for time-invariant branch characteristics)
# We'll use branch_id as categorical
model_fe = ols('service_score ~ C(branch_id) + after_switch + switched_after', data=df).fit()
print("DiD Regression with Branch Fixed Effects:")
print(model_fe.summary())
print()

# Let's also check if staffing and visits explain some of the effect
model_controls = ols('service_score ~ switched_branch + after_switch + switched_after + staffing_index + visits_index', data=df).fit()
print("DiD Regression with Controls:")
print(model_controls.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['switched_after'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + switched_after', data=df).fit()

# Get the key coefficient
print("=== KEY RESULTS ===")
print(f"DiD Estimate (switched_after coefficient): {model.params['switched_after']:.3f}")
print(f"Standard Error: {model.bse['switched_after']:.3f}")
print(f"95% CI: [{model.conf_int().loc['switched_after'][0]:.3f}, {model.conf_int().loc['switched_after'][1]:.3f}]")
print(f"P-value: {model.pvalues['switched_after']:.6f}")
print()

# Check parallel trends assumption - compare pre-trends between switched and non-switched
pre_data = df[df['after_switch'] == 0]
pre_trends = pre_data.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
print("Pre-trends (weeks 6-14):")
print(pre_trends)
print()

# Calculate pre-trend differences
pre_data_copy = pre_data.copy()
pre_data_copy['week_centered'] = pre_data_copy['week'] - 14  # center at switch week
pre_trend_model = ols('service_score ~ switched_branch * week_centered', data=pre_data_copy).fit()
print("Pre-trend interaction test:")
print(pre_trend_model.params)
print(f"P-value for interaction: {pre_trend_model.pvalues['switched_branch:week_centered']:.4f}")
print()

# Let's also look at effect size in context
overall_mean = df['service_score'].mean()
