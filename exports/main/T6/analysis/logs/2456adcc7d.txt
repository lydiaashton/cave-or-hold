# Run difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the treatment effect

model = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

df['branch_mean'] = df.groupby('branch_id')['service_score'].transform('mean')
df['service_score_demeaned'] = df['service_score'] - df['branch_mean']
df['treatment_demeaned'] = df['treatment'] - df.groupby('branch_id')['treatment'].transform('mean')
df['after_switch_demeaned'] = df['after_switch'] - df.groupby('branch_id')['after_switch'].transform('mean')

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Look at pre-trends more carefully
print("=== Pre-Switch Weekly Averages (Weeks 6-14) ===")
pre_data = df[df['week'] < 15]
pre_weekly = pre_data.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
pre_weekly.columns = ['Control', 'Treatment']
pre_weekly['Gap'] = pre_weekly['Treatment'] - pre_weekly['Control']
print(pre_weekly)

print("\n=== Gap Trend Before Switch ===")
# Is the gap widening before the switch?
pre_weekly_reset = pre_weekly.reset_index()
gap_trend = np.polyfit(pre_weekly_reset['week'], pre_weekly_reset['Gap'], 1)
print(f"Gap trend (slope): {gap_trend[0]:.3f} points per week")
print(f"Gap at week 6: {pre_weekly.loc[6, 'Gap']:.2f}")
print(f"Gap at week 14: {pre_weekly.loc[14, 'Gap']:.2f}")

# Post-switch gap
print("\n=== Post-Switch Weekly Averages (Weeks 15-22) ===")
post_data = df[df['week'] >= 15]
post_weekly = post_data.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
post_weekly.columns = ['Control', 'Treatment']
post_weekly['Gap'] = post_weekly['Treatment'] - post_weekly['Control']
print(post_weekly)
print(f"\nAverage post-switch gap: {post_weekly['Gap'].mean():.2f}")
print(f"Average pre-switch gap: {pre_weekly['Gap'].mean():.2f}")
print(f"Gap change: {post_weekly['Gap'].mean() - pre_weekly['Gap'].mean():.2f}")

# Visualize pre-trends
plt.figure(figsize=(12, 5))

plt.subplot(1, 2, 1)
plt.plot(pre_weekly.index, pre_weekly['Control'], 'o-', label='Control', linewidth=2)
plt.plot(pre_weekly.index, pre_weekly['Treatment'], 's-', label='Treatment', linewidth=2)
plt.title('Pre-Switch Trends (Weeks 6-14)', fontsize=12)
plt.xlabel('Week')
plt.ylabel('Service Score')
plt.legend()
plt.grid(True, alpha=0.3)

plt.subplot(1, 2, 2)
plt.plot(pre_weekly.index, pre_weekly['Gap'], 'o-', color='purple', linewidth=2)
plt.axhline(y=0, color='gray', linestyle='--')
plt.title('Treatment-Control Gap (Pre-Switch)', fontsize=12)
plt.xlabel('Week')
plt.ylabel('Gap (Treatment - Control)')
plt.grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig('pretrends_check.png', dpi=150)
print("\nPre-trends visualization saved as pretrends_check.png")
