import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Run a Difference-in-Differences regression
# service_score = beta0 + beta1*treated + beta2*post + beta3*(treated*post) + error
# beta3 is the DiD estimate

df['treated_post'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()
print("Basic DiD Regression:")
print(model.summary())

# Now let's add branch fixed effects and week fixed effects for a more robust analysis
# This controls for time-invariant branch characteristics and common time trends

# Create dummy variables for branch and week
df_with_dummies = df.copy()
df_with_dummies = pd.get_dummies(df_with_dummies, columns=['branch_id', 'week'], drop_first=True)

# This might be too many dummies, let's use a different approach with fixed effects
# Using linearmodels for panel data would be ideal, but let's try with statsmodels

# Alternative: regression with branch and week as categorical variables
model_fe = ols('service_score ~ C(branch_id) + C(week) + switched_branch * after_switch', data=df).fit()
