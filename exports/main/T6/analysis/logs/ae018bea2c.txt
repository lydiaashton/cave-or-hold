import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Count branches by treatment status
print("Branches by treatment status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate average service scores by group and time period
print("\n\nAverage service scores by group and period:")
df['period'] = df['after_switch'].map({0: 'Before', 1: 'After'})
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='period', 
                       aggfunc='mean')
pivot.columns = ['After', 'Before']
pivot = pivot[['Before', 'After']]
pivot['Change'] = pivot['After'] - pivot['Before']
print(pivot)

# Calculate the difference-in-differences estimate manually
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treated_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treated_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n\nDetailed averages:")
print(f"Control (old procedure) - Before: {control_before:.3f}, After: {control_after:.3f}, Change: {control_after - control_before:.3f}")
print(f"Treated (new procedure) - Before: {treated_before:.3f}, After: {treated_after:.3f}, Change: {treated_after - treated_before:.3f}")

did_estimate = (treated_after - treated_before) - (control_after - control_before)
print(f"\nDifference-in-Differences estimate: {did_estimate:.3f}")
