import pandas as pd

import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Double-check the key DiD analysis
df['treatment'] = df['switched_branch'] * df['after_switch']

# Clustered regression (most appropriate for this panel data)
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== KEY REGRESSION RESULTS (Clustered SE) ===")
print(f"\nTreatment effect coefficient: {model.params['treatment']:.3f}")
print(f"Standard error: {model.bse['treatment']:.3f}")
print(f"t-statistic: {model.tvalues['treatment']:.3f}")

coef = model.params['treatment']
se = model.bse['treatment']
ci_lower = coef - 1.96 * se
ci_upper = coef + 1.96 * se
print(f"\n95% Confidence Interval: [{ci_lower:.3f}, {ci_upper:.3f}]")

# Effect size
print(f"\n=== EFFECT SIZE CONTEXT ===")
print(f"Mean service score (all): {df['service_score'].mean():.2f}")
print(f"Std dev of service score: {df['service_score'].std():.2f}")
