import pandas as pd
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Event study: create dummies for each week relative to treatment
# Week 15 is the first post-treatment week (relative week 0)
df['rel_week'] = df['week'] - 15

# Create event study dummies (relative to week -1, which is week 14)
# We'll use week 14 as the reference period
event_study_vars = []
for w in range(-9, 8):  # weeks -9 to +7 (week 6 to 22, relative to week 15)
    if w != -1:  # omit week 14 (relative week -1) as reference
        var_name = f'rel_week_{w}'
        df[var_name] = ((df['rel_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(var_name)

# Run event study regression with branch and week fixed effects
formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(branch_id)'
event_model = smf.ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Regression Results:")
print("="*60)
print("Coefficients for treatment interaction terms (reference: week 14):")
print("-"*60)
for var in event_study_vars:
    coef = event_model.params[var]
    pval = event_model.pvalues[var]
    ci = event_model.conf_int().loc[var]
    week = int(var.split('_')[-1])
    actual_week = week + 15
    period = "Pre" if week < 0 else "Post"
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"Week {actual_week} (rel {week:2d}) [{period:4s}]: coef={coef:6.2f}, p={pval:.3f} {sig}, 95% CI=[{ci[0]:.2f}, {ci[1]:.2f}]")

# Extract post-treatment coefficients
post_coefs = []
for var in event_study_vars:
    week = int(var.split('_')[-1])
    if week >= 0:
        post_coefs.append(event_model.params[var])

print(f"\nAverage post-treatment effect: {sum(post_coefs)/len(post_coefs):.2f}")
