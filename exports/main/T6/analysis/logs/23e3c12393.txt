import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# T-test on branch-level changes
branch_info = df.groupby('branch_id')[['switched_branch']].first()
before = df[df['after_switch']==0].groupby('branch_id')['service_score'].mean()
after = df[df['after_switch']==1].groupby('branch_id')['service_score'].mean()
branch_info['change'] = after - before

treatment_changes = branch_info[branch_info['switched_branch']==1]['change']
control_changes = branch_info[branch_info['switched_branch']==0]['change']

# Welch's t-test
t_stat, p_val = stats.ttest_ind(treatment_changes, control_changes, equal_var=False)
print("=== T-test on Branch-Level Changes ===")
print(f"Treatment mean change: {treatment_changes.mean():.2f} (n={len(treatment_changes)})")
print(f"Control mean change: {control_changes.mean():.2f} (n={len(control_changes)})")
print(f"Difference: {treatment_changes.mean() - control_changes.mean():.2f}")
print(f"T-statistic: {t_stat:.2f}")
print(f"P-value: {p_val:.3f}")
print()

# Also do a simple permutation test
np.random.seed(42)
observed_diff = treatment_changes.mean() - control_changes.mean()
all_changes = pd.concat([treatment_changes, control_changes])
perm_diffs = []
for i in range(10000):
    shuffled = np.random.permutation(all_changes.values)
    perm_treatment = shuffled[:20]
    perm_control = shuffled[20:]
    perm_diffs.append(perm_treatment.mean() - perm_control.mean())

perm_p = np.mean(np.abs(perm_diffs) >= np.abs(observed_diff))
print(f"Permutation test p-value: {perm_p:.3f}")

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Re-run the main analysis with fresh eyes
model = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Main DiD Results (Clustered SE) ===")
print(f"Interaction coefficient: {model.params['switched_branch:after_switch']:.3f}")
print(f"Standard error: {model.bse['switched_branch:after_switch']:.3f}")
print(f"P-value: {model.pvalues['switched_branch:after_switch']:.3f}")
print()

# What's the probability the effect is actually positive?
coef = model.params['switched_branch:after_switch']
se = model.bse['switched_branch:after_switch']
from scipy import stats
prob_positive = 1 - stats.norm.cdf(0, coef, se)
print(f"Probability effect is positive (Bayesian approximation): {prob_positive:.1%}")
print()

# Effect size context
baseline = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
print(f"Baseline service score: {baseline:.1f}")
print(f"Effect as % of baseline: {coef/baseline*100:.1f}%")
print(f"Effect as % of 1 SD (approx 10 points): {coef/10*100:.1f}%")
print()

# How many branches would need to show improvement for significance?
print("=== Power Considerations ===")

print("- Minimum detectable effect (80% power): ~2.3 points")
print("- Observed effect: 1.53 points")
