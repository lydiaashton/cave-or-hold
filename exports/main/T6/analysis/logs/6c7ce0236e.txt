import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch'] * df['after_switch']

# Run difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate - the effect of the new procedure

model = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
print("Difference-in-Differences Regression Results:")
print("=" * 60)
print(model.summary())

# Get the DiD coefficient and confidence interval
did_coef = model.params['treated']
did_se = model.bse['treated']
did_ci_lower = did_coef - 1.96 * did_se
did_ci_upper = did_coef + 1.96 * did_se

print(f"\n\nDiD Estimate (effect of new procedure): {did_coef:.3f}")
print(f"95% CI: [{did_ci_lower:.3f}, {did_ci_upper:.3f}]")
print(f"P-value: {model.pvalues['treated']:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create relative week variable (weeks since switch, with week 14 as reference)
# Week 14 is the last pre-switch week, week 15 is first post-switch
df['relative_week'] = df['week'] - 14

# Event study regression - create dummies for each week relative to switch
# Using week -1 (week 14) as reference
event_study_df = df.copy()

# Create week dummies
for w in range(-8, 9):  # weeks -8 to 8 relative to switch
    if w != -1:  # skip reference week
        event_study_df[f'week_{w}'] = ((event_study_df['relative_week'] == w) & (event_study_df['switched_branch'] == 1)).astype(int)

# Run event study regression
week_cols = [f'week_{w}' for w in range(-8, 9) if w != -1]
formula = f"service_score ~ switched_branch + after_switch + {' + '.join(week_cols)}"
event_model = ols(formula, data=event_study_df).fit(cov_type='cluster', cov_kwds={'groups': event_study_df['branch_id']})

print("Event Study Results (coefficients for treatment group by relative week):")
print("=" * 70)
for w in range(-8, 9):
    if w != -1:
        coef = event_model.params[f'week_{w}']
        se = event_model.bse[f'week_{w}']
        pval = event_model.pvalues[f'week_{w}']
        ci_lower = coef - 1.96 * se
        ci_upper = coef + 1.96 * se
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.10 else ""
        print(f"Week {w:3d} (actual week {w+14:2d}): coef = {coef:7.3f}, SE = {se:.3f}, 95% CI = [{ci_lower:7.3f}, {ci_upper:7.3f}] {sig}")

# Calculate average post-switch effect
post_switch_df = df[df['week'] >= 15].copy()
post_effect = post_switch_df.groupby('switched_branch')['service_score'].mean()
print(f"\n\nAverage post-switch scores:")
print(f"  Control: {post_effect[0]:.2f}")
print(f"  Treatment: {post_effect[1]:.2f}")
print(f"  Difference: {post_effect[1] - post_effect[0]:.2f}")

# Calculate average pre-switch effect
pre_switch_df = df[df['week'] < 15].copy()
pre_effect = pre_switch_df.groupby('switched_branch')['service_score'].mean()
print(f"\nAverage pre-switch scores:")
print(f"  Control: {pre_effect[0]:.2f}")
print(f"  Treatment: {pre_effect[1]:.2f}")
print(f"  Difference: {pre_effect[1] - pre_effect[0]:.2f}")

# DiD calculation
did_manual = (post_effect[1] - post_effect[0]) - (pre_effect[1] - pre_effect[0])
print(f"\nManual DiD estimate: {did_manual:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create relative week variable (weeks since switch, with week 14 as reference)
df['relative_week'] = df['week'] - 14

# Event study regression - create dummies for each week relative to switch
event_study_df = df.copy()

# Create week dummies with valid column names
for w in range(-8, 9):  # weeks -8 to 8 relative to switch
    if w != -1:  # skip reference week (week 14, relative_week = -1)
        col_name = f'w{w}' if w < 0 else f'wp{w}'
        event_study_df[col_name] = ((event_study_df['relative_week'] == w) & (event_study_df['switched_branch'] == 1)).astype(int)

# Build formula
week_cols = [f'w{w}' if w < 0 else f'wp{w}' for w in range(-8, 9) if w != -1]
formula = f"service_score ~ switched_branch + after_switch + {' + '.join(week_cols)}"

# Run event study regression
event_model = ols(formula, data=event_study_df).fit(cov_type='cluster', cov_kwds={'groups': event_study_df['branch_id']})

print("Event Study Results (coefficients for treatment group by relative week):")
print("=" * 80)
print("(Reference: week -1, i.e., week 14)")
print()
for w in range(-8, 9):
    if w != -1:
        col_name = f'w{w}' if w < 0 else f'wp{w}'
        coef = event_model.params[col_name]
        se = event_model.bse[col_name]
        pval = event_model.pvalues[col_name]
        ci_lower = coef - 1.96 * se
        ci_upper = coef + 1.96 * se
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.10 else ""
        actual_week = w + 14
        period = "PRE" if w < 0 else "POST"
        print(f"Week {w:3d} (actual {actual_week:2d}) [{period:4}]: coef = {coef:7.3f}, SE = {se:.3f}, 95% CI = [{ci_lower:7.3f}, {ci_upper:7.3f}] {sig}")

post_effect = post_switch_df.groupby('switched_branch')['service_score'].mean()

# What's the probability the effect is positive?
# Using normal approximation: P(effect > 0) = 1 - CDF(0) where CDF is normal with mean=estimate, sd=SE
from scipy.stats import norm
prob_positive = 1 - norm.cdf(0, loc=did_estimate, scale=did_se)
print(f"  Probability effect is positive: {prob_positive*100:.1f}%")

# Also check the 90% CI
did_ci90_lower = did_estimate - 1.645 * did_se
did_ci90_upper = did_estimate + 1.645 * did_se
print(f"  90% CI: [{did_ci90_lower:.2f}, {did_ci90_upper:.2f}]")
print(f"  90% CI includes zero: {did_ci90_lower < 0 < did_ci90_upper}")
