import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run a regression with the DiD specification
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*treatment + error
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Difference-in-Differences Regression Results:")
print(model.summary())

# Also run with cluster-robust standard errors at branch level
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nWith Cluster-Robust Standard Errors (clustered by branch):")
print(model_clustered.summary())

# Check pre-trends: compare trends before week 15 for both groups
print("\n\nPre-trend check (weeks 6-14):")
pre_period = df[df['week'] < 15]
pre_model = ols('service_score ~ switched_branch + week + switched_branch:week', data=pre_period).fit(cov_type='cluster', cov_kwds={'groups': pre_period['branch_id']})
print(pre_model.summary())

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main DiD model
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# What's the probability the effect is actually positive?
# Using the coefficient and SE, we can estimate this
coef = model.params['treatment']
se = model.bse['treatment']

# Z-score for testing if effect > 0
z = coef / se
# Probability that true effect is positive (one-tailed)
from scipy.stats import norm
prob_positive = norm.cdf(z)

print(f"Treatment coefficient: {coef:.2f}")
print(f"Standard error: {se:.2f}")
print(f"Z-score: {z:.2f}")
print(f"Probability true effect is positive: {prob_positive:.1%}")
print(f"P-value (two-tailed): {model.pvalues['treatment']:.3f}")

# What would we need to detect this effect with 80% power?
# Rough power calculation
print(f"\nCurrent sample: {len(df)} observations")
print(f"Effect size (Cohen's d approx): {coef / df['service_score'].std():.2f}")

# Minimum detectable effect with current sample
# For 80% power, alpha=0.05, two-tailed
n = len(df)
# Rough approximation
mde = 1.96 * se * 2  # Very rough
print(f"Approximate minimum detectable effect: ~{mde:.1f} points")
