import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Count branches by switch status
switch_counts = df.groupby('branch_id')['switched_branch'].first().value_counts()
print("Branches by switch status:")
print(switch_counts)

# Create a pivot to see average service scores by group and period
df['period'] = np.where(df['after_switch'] == 0, 'Before', 'After')
df['group'] = np.where(df['switched_branch'] == 1, 'Switched', 'Control')

# Average service scores by group and period
pivot = df.pivot_table(values='service_score', index='group', columns='period', aggfunc='mean')
print("\nAverage service scores by group and period:")
print(pivot)

# Calculate the DiD estimate manually
before_switched = df[(df['group'] == 'Switched') & (df['period'] == 'Before')]['service_score'].mean()
after_switched = df[(df['group'] == 'Switched') & (df['period'] == 'After')]['service_score'].mean()
before_control = df[(df['group'] == 'Control') & (df['period'] == 'Before')]['service_score'].mean()
after_control = df[(df['group'] == 'Control') & (df['period'] == 'After')]['service_score'].mean()

print(f"\nBefore - Switched: {before_switched:.2f}")
print(f"After - Switched: {after_switched:.2f}")
print(f"Change in Switched: {after_switched - before_switched:.2f}")
print(f"\nBefore - Control: {before_control:.2f}")
print(f"After - Control: {after_control:.2f}")
print(f"Change in Control: {after_control - before_control:.2f}")

did_estimate = (after_switched - before_switched) - (after_control - before_control)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

from statsmodels.formula.api import ols

# Run DiD regression with robust standard errors
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate

model = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='HC1')
print("DiD Regression Results (Robust SE):")
print(model.summary())

# Compare the pre-period average to see if higher-performing branches were selected

branch_pre_avg.columns = ['branch_id', 'pre_avg_score']

# Merge back to get switch status
branch_pre_avg = branch_pre_avg.merge(
    df[['branch_id', 'switched_branch']].drop_duplicates(), 
    on='branch_id'
)

print("\nPre-period average scores by switch status:")
print(branch_pre_avg.groupby('switched_branch')['pre_avg_score'].agg(['mean', 'std', 'count']))

# T-test for difference in pre-period averages
from scipy import stats
control_pre = branch_pre_avg[branch_pre_avg['switched_branch'] == 0]['pre_avg_score']
treated_pre = branch_pre_avg[branch_pre_avg['switched_branch'] == 1]['pre_avg_score']
t_stat, p_val = stats.ttest_ind(treated_pre, control_pre)
print(f"\nT-test for pre-period difference: t={t_stat:.2f}, p={p_val:.3f}")
