import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Check treatment/control group sizes
print("Treatment vs Control branches:")
treatment_branches = df[df['switched_branch'] == 1]['branch_id'].nunique()
control_branches = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"Treatment branches (switched_branch=1): {treatment_branches}")
print(f"Control branches (switched_branch=0): {control_branches}")

# Check pre/post periods
print("\nPre vs Post periods:")
pre_weeks = df[df['after_switch'] == 0]['week'].nunique()
post_weeks = df[df['after_switch'] == 1]['week'].nunique()
print(f"Pre-switch weeks (after_switch=0): {pre_weeks} weeks ({df[df['after_switch']==0]['week'].min()}-{df[df['after_switch']==0]['week'].max()})")
print(f"Post-switch weeks (after_switch=1): {post_weeks} weeks ({df[df['after_switch']==1]['week'].min()}-{df[df['after_switch']==1]['week'].max()})")

# Mean service scores by group and period
print("\nMean service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the difference-in-differences manually
print("\n--- Difference-in-Differences Calculation ---")
# Treatment group
treatment_pre = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_post = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_diff = treatment_post - treatment_pre

# Control group
control_pre = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_post = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
control_diff = control_post - control_pre

# DiD estimate
did_estimate = treatment_diff - control_diff

print(f"Treatment group: Pre={treatment_pre:.2f}, Post={treatment_post:.2f}, Change={treatment_diff:.2f}")
print(f"Control group: Pre={control_pre:.2f}, Post={control_post:.2f}, Change={control_diff:.2f}")
print(f"Difference-in-Differences estimate: {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Run DiD regression
# Basic DiD model
model1 = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit()
print("=== Basic DiD Regression ===")
print(model1.summary())

# DiD with branch fixed effects
model2 = ols('service_score ~ switched_branch + after_switch + treatment_post + C(branch_id)', data=df).fit()
print("\n=== DiD with Branch Fixed Effects ===")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Main DiD model with cluster-robust SE
model = ols('service_score ~ switched_branch + after_switch + treatment_post + staffing_index + visits_index + C(branch_id)', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract the treatment effect
treatment_effect = model.params['treatment_post']
treatment_se = model.bse['treatment_post']
treatment_pvalue = model.pvalues['treatment_post']

print("=== MAIN RESULTS ===")
print(f"Treatment Effect (DiD estimate): {treatment_effect:.2f} points")
print(f"Standard Error: {treatment_se:.2f}")
print(f"95% CI: [{treatment_effect - 1.96*treatment_se:.2f}, {treatment_effect + 1.96*treatment_se:.2f}]")
print(f"P-value: {treatment_pvalue:.3f}")

# Calculate percentage improvement relative to baseline
treatment_baseline = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
pct_improvement = (treatment_effect / treatment_baseline) * 100
print(f"\nBaseline treatment group mean: {treatment_baseline:.2f}")
print(f"Percentage improvement: {pct_improvement:.1f}%")

# Check if effect is meaningful relative to typical variation
service_score_std = df['service_score'].std()
effect_in_std_units = treatment_effect / service_score_std
print(f"\nService score SD: {service_score_std:.2f}")
print(f"Effect size in SD units: {effect_in_std_units:.2f}")

# Additional analysis: effect by branch (heterogeneity)
print("\n=== HETEROGENEITY CHECK ===")
# Calculate branch-level pre-post changes
branch_changes = df.groupby(['branch_id', 'switched_branch']).apply(
    lambda x: x[x['after_switch']==1]['service_score'].mean() - x[x['after_switch']==0]['service_score'].mean()
).reset_index()
branch_changes.columns = ['branch_id', 'switched_branch', 'change']

print("Treatment branches (switched_branch=1) changes:")
treatment_changes = branch_changes[branch_changes['switched_branch'] == 1]['change']
print(f"  Mean change: {treatment_changes.mean():.2f}")
print(f"  Std dev: {treatment_changes.std():.2f}")
print(f"  Min: {treatment_changes.min():.2f}, Max: {treatment_changes.max():.2f}")
print(f"  Branches with positive change: {(treatment_changes > 0).sum()} / {len(treatment_changes)}")

print("\nControl branches (switched_branch=0) changes:")
control_changes = branch_changes[branch_changes['switched_branch'] == 0]['change']
print(f"  Mean change: {control_changes.mean():.2f}")
print(f"  Std dev: {control_changes.std():.2f}")
print(f"  Min: {control_changes.min():.2f}, Max: {control_changes.max():.2f}")
print(f"  Branches with positive change: {(control_changes > 0).sum()} / {len(control_changes)}")
