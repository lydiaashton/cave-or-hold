import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Count unique branches in each group
print("=== Branch Counts ===")
switched_branches = df[df['switched_branch'] == 1]['branch_id'].nunique()
control_branches = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"Switched branches: {switched_branches}")
print(f"Control branches: {control_branches}")
print(f"Total branches: {df['branch_id'].nunique()}")

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate

df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\n=== DiD Regression Results ===")
print(model.summary())

# Also run with cluster-robust standard errors by branch
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Regression with Cluster-Robust SE ===")
print(model_clustered.summary())

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate branch-level before/after changes
branch_stats = df.groupby(['branch_id', 'switched_branch']).agg({
    'service_score': ['mean', 'std', 'count']
}).reset_index()
branch_stats.columns = ['branch_id', 'switched_branch', 'mean_score', 'std_score', 'count']

# Get before and after for each branch
before = df[df['after_switch'] == 0].groupby('branch_id')['service_score'].mean()
after = df[df['after_switch'] == 1].groupby('branch_id')['service_score'].mean()

branch_changes = pd.DataFrame({
    'branch_id': before.index,
    'before': before.values,
    'after': after.values
})
branch_changes['change'] = branch_changes['after'] - branch_changes['before']

# Merge with switched status
switched_status = df.groupby('branch_id')['switched_branch'].first().reset_index()
branch_changes = branch_changes.merge(switched_status, on='branch_id')

print("=== Branch-Level Changes ===")
print("\nSwitched branches (n=90):")
switched_changes = branch_changes[branch_changes['switched_branch'] == 1]['change']
print(f"  Mean change: {switched_changes.mean():.2f}")
print(f"  Std change: {switched_changes.std():.2f}")
print(f"  % positive: {(switched_changes > 0).mean()*100:.1f}%")
print(f"  Min change: {switched_changes.min():.2f}")
print(f"  Max change: {switched_changes.max():.2f}")

print("\nControl branches (n=60):")
control_changes = branch_changes[branch_changes['switched_branch'] == 0]['change']
print(f"  Mean change: {control_changes.mean():.2f}")
print(f"  Std change: {control_changes.std():.2f}")
print(f"  % positive: {(control_changes > 0).mean()*100:.1f}%")
print(f"  Min change: {control_changes.min():.2f}")
print(f"  Max change: {control_changes.max():.2f}")

# Statistical test
from scipy import stats
t_stat, p_val = stats.ttest_ind(switched_changes, control_changes)
print(f"\n=== Two-sample t-test ===")
print(f"Difference in changes: {switched_changes.mean() - control_changes.mean():.2f}")
print(f"t-statistic: {t_stat:.2f}")
print(f"p-value: {p_val:.4f}")
