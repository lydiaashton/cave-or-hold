import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Count branches by switch status
branch_switch = df.groupby('branch_id')['switched_branch'].first()
print("=== Branch Switch Status ===")
print(f"Total branches: {len(branch_switch)}")
print(f"Switched branches: {branch_switch.sum()} ({branch_switch.mean()*100:.1f}%)")
print(f"Control branches: {(branch_switch == 0).sum()} ({(1-branch_switch.mean())*100:.1f}%)")

# Average service scores by group and period
print("\n=== Average Service Scores ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate DiD manually
control_pre = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_post = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treated_pre = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treated_post = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nControl group: Pre={control_pre:.2f}, Post={control_post:.2f}, Change={control_post-control_pre:.2f}")
print(f"Treated group: Pre={treated_pre:.2f}, Post={treated_post:.2f}, Change={treated_post-treated_pre:.2f}")
print(f"\nDifference-in-Differences: {(treated_post - treated_pre) - (control_post - control_pre):.2f}")

# Run DiD regression
print("\n=== DiD Regression ===")
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print(model.summary().tables[1])

# Cluster-robust standard errors by branch
print("\n=== DiD Regression with Cluster-Robust SE (by branch) ===")
model_cluster = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_cluster.summary().tables[1])

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the visualization
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Plot 1: Parallel trends check (pre-period only)
pre_data = df[df['week'] < 15]
weekly_pre = pre_data.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

ax1 = axes[0]
for group, label in [(0, 'Control (No Switch)'), (1, 'Treatment (Switched)')]:
    subset = weekly_pre[weekly_pre['switched_branch'] == group]
    ax1.plot(subset['week'], subset['service_score'], marker='o', linewidth=2, label=label)
ax1.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')
ax1.set_xlabel('Week', fontsize=12)
ax1.set_ylabel('Average Service Score', fontsize=12)
ax1.set_title('Pre-Switch Trends: Parallel Trends Assumption', fontsize=14, fontweight='bold')
ax1.legend()
ax1.grid(True, alpha=0.3)
ax1.set_xticks(range(6, 15))

# Plot 2: Full event study with gap
weekly_full = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
pivot_full = weekly_full.pivot(index='week', columns='switched_branch', values='service_score')
pivot_full['gap'] = pivot_full[1] - pivot_full[0]

ax2 = axes[1]
weeks = pivot_full.index
ax2.plot(weeks, pivot_full[0], marker='o', linewidth=2, label='Control (No Switch)', color='blue')
ax2.plot(weeks, pivot_full[1], marker='s', linewidth=2, label='Treatment (Switched)', color='green')
ax2.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')
ax2.fill_between(weeks, pivot_full[0], pivot_full[1], alpha=0.3, color='green', label='Treatment Effect')
ax2.set_xlabel('Week', fontsize=12)
ax2.set_ylabel('Average Service Score', fontsize=12)
ax2.set_title('Service Scores: Treatment vs Control Over Time', fontsize=14, fontweight='bold')
ax2.legend()
ax2.grid(True, alpha=0.3)
ax2.set_xticks(range(6, 23))

plt.tight_layout()
