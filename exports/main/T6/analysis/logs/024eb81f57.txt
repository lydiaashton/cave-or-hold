import pandas as pd
import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched + beta2*after + beta3*(switched*after) + error
# beta3 is the DiD estimator

model = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Basic DiD Regression:")
print(model.summary())
print()

# Now let's add branch fixed effects and week fixed effects for a more robust analysis
# This controls for time-invariant branch characteristics and common time trends

df_with_dummies = df.copy()

# Create branch dummies (fixed effects)
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)

# Create week dummies (fixed effects)  
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)

# Combine
X = pd.concat([df[['switched_branch', 'after_switch', 'treatment']], 
               branch_dummies, 
               week_dummies], axis=1)
y = df['service_score']

# Add constant
X = sm.add_constant(X)

# Fit model
fe_model = sm.OLS(y, X).fit()
print("DiD with Branch and Week Fixed Effects:")
print(fe_model.summary())

}).reset_index()
weekly_summary.columns = ['switched_branch', 'week', 'mean', 'std', 'n']
weekly_summary['se'] = weekly_summary['std'] / (weekly_summary['n'] ** 0.5)

# Create the plot with confidence intervals
fig, ax = plt.subplots(figsize=(12, 6))

for group in [0, 1]:
    subset = weekly_summary[weekly_summary['switched_branch'] == group]
    label = 'Control (Old Procedure)' if group == 0 else 'Treatment (New Procedure)'
    color = 'blue' if group == 0 else 'green'
    ax.plot(subset['week'], subset['mean'], marker='o', label=label, linewidth=2, color=color)
    ax.fill_between(subset['week'], 
                    subset['mean'] - 1.96 * subset['se'],
                    subset['mean'] + 1.96 * subset['se'],
                    alpha=0.2, color=color)

# Add vertical line at switch point
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch implemented (week 15)')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Mean Service Score', fontsize=12)
ax.set_title('Service Scores by Group Over Time (with 95% Confidence Intervals)', fontsize=14)
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_scores_with_ci.png', dpi=150)
print("Plot saved as service_scores_with_ci.png")

)

print(f"Pre-period gap std dev: {pre_gap.std():.2f} points")
