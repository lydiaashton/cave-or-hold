import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster
from scipy import stats

df = pd.read_csv('branches.csv')

# Create treatment variable
df['treatment'] = df['switched_branch'] * df['after_switch']

# Fixed effects model with branch dummies
print("=== Fixed Effects Model (with branch FEs) ===")
fe_model = ols('service_score ~ treatment + staffing_index + visits_index + C(branch_id)', data=df).fit()
print(fe_model.summary())

# Cluster-robust standard errors by branch
print("\n=== Cluster-Robust Standard Errors (by branch) ===")
cluster_se = cov_cluster(fe_model, df['branch_id'])
cluster_se_diag = np.sqrt(np.diag(cluster_se))

treatment_idx = fe_model.params.index.get_loc('treatment')
print(f"Treatment coefficient: {fe_model.params['treatment']:.4f}")
print(f"Cluster-robust SE: {cluster_se_diag[treatment_idx]:.4f}")
t_stat = fe_model.params['treatment'] / cluster_se_diag[treatment_idx]
print(f"t-statistic: {t_stat:.4f}")

# Approximate p-value (two-tailed)
p_value = 2 * (1 - stats.t.cdf(abs(t_stat), df=len(df['branch_id'].unique()) - 1))
print(f"p-value: {p_value:.4f}")

# Confidence interval
ci_lower = fe_model.params['treatment'] - 1.96 * cluster_se_diag[treatment_idx]
ci_upper = fe_model.params['treatment'] + 1.96 * cluster_se_diag[treatment_idx]
print(f"95% CI: [{ci_lower:.4f}, {ci_upper:.4f}]")

# Also check event study - effect by week after switch
print("\n=== Event Study Analysis ===")
df['weeks_since_switch'] = df['week'] - 14  # 0 = week 14 (last pre-switch week), 1 = week 15 (first post-switch)
df['post_period'] = (df['week'] >= 15).astype(int)

# Create interaction terms for each week with treatment
event_results = []
for week in range(15, 23):
    week_df = df[df['week'] == week].copy()
    control_avg = week_df[week_df['switched_branch'] == 0]['service_score'].mean()
    treatment_avg = week_df[week_df['switched_branch'] == 1]['service_score'].mean()
    diff = treatment_avg - control_avg
    event_results.append({'week': week, 'treatment_avg': treatment_avg, 'control_avg': control_avg, 'diff': diff})

event_df = pd.DataFrame(event_results)
print(event_df)

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster
from scipy import stats

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Final model with branch fixed effects and cluster-robust SE
fe_model = ols('service_score ~ treatment + staffing_index + visits_index + C(branch_id)', data=df).fit()
cluster_se = cov_cluster(fe_model, df['branch_id'])
cluster_se_diag = np.sqrt(np.diag(cluster_se))

treatment_idx = fe_model.params.index.get_loc('treatment')
coef = fe_model.params['treatment']
se = cluster_se_diag[treatment_idx]
t_stat = coef / se
p_value = 2 * (1 - stats.t.cdf(abs(t_stat), df=len(df['branch_id'].unique()) - 1))
ci_lower = coef - 1.96 * se
ci_upper = coef + 1.96 * se

# Summary statistics
avg_score = df['service_score'].mean()
treated_branches = df[df['switched_branch'] == 1]['branch_id'].nunique()
control_branches = df[df['switched_branch'] == 0]['branch_id'].nunique()

# Pre and post averages by group
pre_treated = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
post_treated = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
