import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
branches_switched = df[df['switched_branch'] == 1]['branch_id'].nunique()
branches_control = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"Branches that switched: {branches_switched}")
print(f"Branches that didn't switch (control): {branches_control}")
print(f"Total branches: {branches_switched + branches_control}")

# Check balance - do all branches have data for all weeks?
weeks_per_branch = df.groupby('branch_id')['week'].nunique()
print(f"\nWeeks per branch - min: {weeks_per_branch.min()}, max: {weeks_per_branch.max()}, mean: {weeks_per_branch.mean():.1f}")

# Calculate average service scores by group and period
print("\n=== Average Service Scores ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Difference-in-differences calculation
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nTreatment group (switched): Before={before_treatment:.2f}, After={after_treatment:.2f}, Change={after_treatment - before_treatment:.2f}")
print(f"Control group (not switched): Before={before_control:.2f}, After={after_control:.2f}, Change={after_control - before_control:.2f}")

diff_in_diff = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {diff_in_diff:.2f}")

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run difference-in-differences regression with standard errors clustered by branch
# DID = after_switch * switched_branch interaction

# First, simple OLS
model = ols('service_score ~ C(switched_branch) * C(after_switch)', data=df).fit()
print("=== Simple OLS Regression ===")
print(model.summary())

# Now with cluster-robust standard errors by branch
model_clustered = ols('service_score ~ C(switched_branch) * C(after_switch)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== OLS with Cluster-Robust SE (by branch) ===")

# Also control for staffing and visits
model_controls = ols('service_score ~ C(switched_branch) * C(after_switch) + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== With staffing and visits controls ===")
print(model_controls.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study / dynamic DID - estimate effect for each week relative to switch
# Create relative week variable (week 15 = 0, week 14 = -1, etc.)
df['relative_week'] = df['week'] - 15

# Create dummies for each relative week (excluding week -1 as reference)
for rw in sorted(df['relative_week'].unique()):
    if rw != -1:  # reference category
        df[f'rel_week_{rw}'] = (df['relative_week'] == rw).astype(int)

# Run regression with all relative week dummies interacted with treatment
# Only include switched_branch interaction terms
formula = 'service_score ~ C(switched_branch) + '
for rw in sorted(df['relative_week'].unique()):
    if rw != -1:
        formula += f'C(switched_branch):rel_week_{rw} + '
formula = formula[:-3]  # remove trailing ' + '

model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results ===")
print("Coefficients for treatment interaction with each relative week (reference: week -1)")
print()

# Extract the interaction coefficients
coef_dict = {}
for rw in sorted(df['relative_week'].unique()):
    if rw == -1:
        coef_dict[rw] = 0  # reference
        se_dict = {rw: 0}
    else:
        var_name = f'C(switched_branch)[T.1]:rel_week_{rw}'
        if var_name in model.params:
            coef_dict[rw] = model.params[var_name]
            se_dict[rw] = model.bse[var_name]

for rw in sorted(coef_dict.keys()):
    if rw == -1:
        print(f"Week {rw+15} (reference): 0.00")
    else:
        coef = coef_dict[rw]
        se = se_dict[rw]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        sig = "***" if abs(coef/se) > 2.58 else "**" if abs(coef/se) > 1.96 else "*" if abs(coef/se) > 1.64 else ""
        print(f"Week {rw+15}: {coef:.2f} (SE: {se:.2f}), 95% CI: [{ci_low:.2f}, {ci_high:.2f}] {sig}")

# Create event study plot
fig, ax = plt.subplots(figsize=(10, 6))

weeks = sorted(coef_dict.keys())
coefs = [coef_dict[w] for w in weeks]
ses = [se_dict[w] for w in weeks]

ax.errorbar(weeks, coefs, yerr=[1.96*s for s in ses], fmt='o', capsize=5, label='Treatment Effect')
ax.axvline(x=-0.5, color='red', linestyle='--', label='Switch occurred')
ax.axhline(y=0, color='gray', linestyle='-', alpha=0.5)
ax.set_xlabel('Relative Week (0 = week of switch)', fontsize=12)
ax.set_ylabel('Effect on Service Score', fontsize=12)
ax.set_title('Event Study: Treatment Effect by Week', fontsize=14)
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(weeks)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nSaved event study plot to event_study.png")

import pandas as pd

import matplotlib.pyplot as plt

# Simpler approach: create relative week and use it directly
df['relative_week'] = df['week'] - 15

# Create a cleaner event study using a different approach
# We'll estimate the effect for each post-period week separately

results = []
for post_week in range(15, 23):  # weeks 15-22
    # Create a dummy for this specific week
    df['week_dummy'] = (df['week'] == post_week).astype(int)
    
    # Run DID for this week only (comparing to all pre-period weeks)
    pre_data = df[df['week'] < 15]
    this_week_data = df[df['week'] == post_week]
    combined = pd.concat([pre_data, this_week_data])
    combined['after'] = (combined['week'] >= 15).astype(int)
    
    model = ols('service_score ~ C(switched_branch) * after', data=combined).fit(
        cov_type='cluster', cov_kwds={'groups': combined['branch_id']}
    )
    
    # Get the interaction coefficient
    coef = model.params['C(switched_branch)[T.1]:after[T.1]']
    se = model.bse['C(switched_branch)[T.1]:after[T.1]']
    results.append({'week': post_week, 'coef': coef, 'se': se})

for _, row in results_df.iterrows():
    ci_low = row['coef'] - 1.96 * row['se']
    ci_high = row['coef'] + 1.96 * row['se']
    t_stat = row['coef'] / row['se']
    sig = "***" if abs(t_stat) > 2.58 else "**" if abs(t_stat) > 1.96 else "*" if abs(t_stat) > 1.64 else ""
    print(f"Week {row['week']}: {row['coef']:.2f} (SE: {row['se']:.2f}), 95% CI: [{ci_low:.2f}, {ci_high:.2f}] {sig}")

import pandas as pd
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Recalculate the main DID estimate
model = ols('service_score ~ C(switched_branch) * C(after_switch)', data=df).fit(
    cov_type='cluster', cov_kwds={'groups': df['branch_id']}
)

did_coef = model.params['C(switched_branch)[T.1]:C(after_switch)[T.1]']
did_se = model.bse['C(switched_branch)[T.1]:C(after_switch)[T.1]']
did_pvalue = model.pvalues['C(switched_branch)[T.1]:C(after_switch)[T.1]']

print("=== VERIFICATION OF MAIN RESULT ===")
