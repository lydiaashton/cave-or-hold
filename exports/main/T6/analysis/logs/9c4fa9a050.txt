import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create treatment and period indicators
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treatment'] * df['post']

# Calculate means by group and period
print("=== Service Score Means by Group and Period ===\n")

# Control group (switched_branch=0)
control_pre = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_post = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

# Treatment group (switched_branch=1)
treat_pre = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treat_post = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"Control Group (didn't switch):")
print(f"  Pre-period (weeks 6-14): {control_pre:.2f}")
print(f"  Post-period (weeks 15-22): {control_post:.2f}")
print(f"  Change: {control_post - control_pre:.2f}")

print(f"\nTreatment Group (switched):")
print(f"  Pre-period (weeks 6-14): {treat_pre:.2f}")
print(f"  Post-period (weeks 15-22): {treat_post:.2f}")
print(f"  Change: {treat_post - treat_pre:.2f}")

# Difference-in-differences estimate
did_estimate = (treat_post - treat_pre) - (control_post - control_pre)
print(f"\n=== Difference-in-Differences Estimate ===")
print(f"DiD = (Treatment Change) - (Control Change)")
print(f"DiD = {treat_post - treat_pre:.2f} - {control_post - control_pre:.2f} = {did_estimate:.2f}")

# Run a regression for DiD with standard errors
import statsmodels.api as sm
import statsmodels.formula.api as smf

# Simple DiD regression
model = smf.ols('service_score ~ treatment + post + treated_post', data=df).fit()
print("\n=== DiD Regression Results ===")
print(model.summary())

# Cluster-robust standard errors by branch
model_clustered = smf.ols('service_score ~ treatment + post + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Regression with Cluster-Robust SE (by branch) ===")
print(model_clustered.summary())

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Add covariates to the model
print("=== DiD with Covariates ===")
model_with_covariates = smf.ols('service_score ~ treatment + post + treated_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_with_covariates.summary())

# Check if covariates differ by treatment group
print("\n=== Covariate Balance Check ===")
covariate_means = df.groupby('switched_branch')[['staffing_index', 'visits_index']].mean()
print(covariate_means)

# Check post-period covariate differences
print("\nPost-period covariate means:")
post_df = df[df['after_switch'] == 1]
print(post_df.groupby('switched_branch')[['staffing_index', 'visits_index']].mean())

# Event study - check effects by week
print("\n=== Event Study Analysis ===")
df['week_relative'] = df['week'] - 14  # week 14 is the last pre-switch week

# Create week dummies interacted with treatment
event_study_df = df.copy()
for w in range(-8, 9):  # weeks -8 to +8 relative to switch
    event_study_df[f'week_{w}'] = ((event_study_df['week_relative'] == w) & (event_study_df['switched_branch'] == 1)).astype(int)

# Run event study regression (omitting week -1 as reference)
week_cols = [f'week_{w}' for w in range(-8, 9) if w != -1]
formula = 'service_score ~ ' + ' + '.join(week_cols) + ' + C(week)'
event_model = smf.ols(formula, data=event_study_df).fit(cov_type='cluster', cov_kwds={'groups': event_study_df['branch_id']})

# Extract coefficients for the treatment weeks
print("\nEvent Study Coefficients (relative to week -1):")
results = []
for w in range(-8, 9):
    if w != -1:
        coef = event_model.params[f'week_{w}']
        se = event_model.bse[f'week_{w}']
        pval = event_model.pvalues[f'week_{w}']
        results.append({'week_relative': w, 'coef': coef, 'se': se, 'pval': pval})

event_results = pd.DataFrame(results)
print(event_results.round(3))

# Plot event study
import matplotlib.pyplot as plt

fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(event_results['week_relative'], event_results['coef'], 
            yerr=1.96 * event_results['se'], 
            fmt='o', capsize=5, label='Coefficient ± 95% CI')
ax.axhline(y=0, color='red', linestyle='--', linewidth=2)
ax.axvline(x=-0.5, color='gray', linestyle='-', linewidth=2, label='Switch point')
ax.set_xlabel('Weeks Relative to Switch (0 = first post-switch week)', fontsize=12)
ax.set_ylabel('Effect on Service Score', fontsize=12)
ax.set_title('Event Study: Treatment Effect Over Time', fontsize=14, fontweight='bold')
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150, bbox_inches='tight')
print("\nEvent study plot saved as 'event_study.png'")

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treat'] = df['switched_branch']
df['post_period'] = df['after_switch']
df['treat_post'] = df['switched_branch'] * df['after_switch']

# Event study - check effects by week
print("=== Event Study Analysis ===")
df['week_relative'] = df['week'] - 14  # week 14 is the last pre-switch week

# Create week dummies interacted with treatment - use different naming
event_study_df = df.copy()
for w in range(-8, 9):  # weeks -8 to +8 relative to switch
    col_name = f'w{abs(w)}_neg{1 if w < 0 else 0}' if w != -1 else None  # skip week -1 as reference
    if w != -1:
        event_study_df[f'w{w}'] = ((event_study_df['week_relative'] == w) & (event_study_df['treat'] == 1)).astype(int)

# Run event study regression (omitting week -1 as reference)
week_cols = [f'w{w}' for w in range(-8, 9) if w != -1]
formula = 'service_score ~ ' + ' + '.join(week_cols) + ' + C(week)'
event_model = smf.ols(formula, data=event_study_df).fit(cov_type='cluster', cov_kwds={'groups': event_study_df['branch_id']})

# Extract coefficients for the treatment weeks
print("\nEvent Study Coefficients (relative to week -1):")
results = []
for w in range(-8, 9):
    if w != -1:
        coef = event_model.params[f'w{w}']
        se = event_model.bse[f'w{w}']
        pval = event_model.pvalues[f'w{w}']
        results.append({'week_relative': w, 'coef': coef, 'se': se, 'pval': pval})

event_results = pd.DataFrame(results)
print(event_results.round(3))

# Plot event study
fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(event_results['week_relative'], event_results['coef'], 
            yerr=1.96 * event_results['se'], 
            fmt='o', capsize=5, label='Coefficient ± 95% CI')
ax.axhline(y=0, color='red', linestyle='--', linewidth=2)
ax.axvline(x=-0.5, color='gray', linestyle='-', linewidth=2, label='Switch point')
ax.set_xlabel('Weeks Relative to Switch (0 = first post-switch week)', fontsize=12)
ax.set_ylabel('Effect on Service Score', fontsize=12)
ax.set_title('Event Study: Treatment Effect Over Time', fontsize=14, fontweight='bold')
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150, bbox_inches='tight')
print("\nEvent study plot saved as 'event_study.png'")

# Also print a summary interpretation
print("\n=== Event Study Interpretation ===")
pre_period = event_results[event_results['week_relative'] < 0]
post_period = event_results[event_results['week_relative'] >= 0]

print(f"Pre-period coefficients (should be near 0 if parallel trends hold):")
print(f"  Mean: {pre_period['coef'].mean():.2f}")
print(f"  Range: [{pre_period['coef'].min():.2f}, {pre_period['coef'].max():.2f}]")
print(f"  Any significant at 5%? {(pre_period['pval'] < 0.05).sum()} of {len(pre_period)}")

print(f"\nPost-period coefficients (treatment effects):")
print(f"  Mean: {post_period['coef'].mean():.2f}")
print(f"  Range: [{post_period['coef'].min():.2f}, {post_period['coef'].max():.2f}]")
print(f"  Any significant at 5%? {(post_period['pval'] < 0.05).sum()} of {len(post_period)}")

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treat'] = df['switched_branch']
df['post_period'] = df['after_switch']
df['treat_post'] = df['switched_branch'] * df['after_switch']

# Event study - check effects by week
print("=== Event Study Analysis ===")
df['week_relative'] = df['week'] - 14  # week 14 is the last pre-switch week

# Create week dummies interacted with treatment - use safe naming
event_study_df = df.copy()
week_dummies = {}
for w in range(-8, 9):
    if w != -1:  # skip week -1 as reference
        if w < 0:
            col_name = f'pre{abs(w)}'  # pre1, pre2, ..., pre8
        elif w == 0:
            col_name = 'post0'
        else:
            col_name = f'post{w}'  # post1, post2, ..., post8
        week_dummies[w] = col_name
        event_study_df[col_name] = ((event_study_df['week_relative'] == w) & (event_study_df['treat'] == 1)).astype(int)

# Run event study regression
week_cols = list(week_dummies.values())
formula = 'service_score ~ ' + ' + '.join(week_cols) + ' + C(week)'
event_model = smf.ols(formula, data=event_study_df).fit(cov_type='cluster', cov_kwds={'groups': event_study_df['branch_id']})

# Extract coefficients for the treatment weeks
print("\nEvent Study Coefficients (relative to week -1):")
results = []
for w, col_name in week_dummies.items():
    coef = event_model.params[col_name]
    se = event_model.bse[col_name]
    pval = event_model.pvalues[col_name]
    results.append({'week_relative': w, 'coef': coef, 'se': se, 'pval': pval})

event_results = pd.DataFrame(results)
print(event_results.round(3))

# Plot event study
fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(event_results['week_relative'], event_results['coef'], 
            yerr=1.96 * event_results['se'],

ax.axhline(y=0, color='red', linestyle='--', linewidth=2)

ax.set_xlabel('Weeks Relative to Switch (0 = first post-switch week)', fontsize=12)

# Main DiD model with cluster-robust SE
model = smf.ols('service_score ~ switched_branch + after_switch + treat_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
