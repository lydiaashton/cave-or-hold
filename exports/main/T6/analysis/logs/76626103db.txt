import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# DiD with branch fixed effects (using dummy variables)
# This controls for time-invariant branch characteristics
print("DiD with Branch Fixed Effects:")

# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_fe = pd.concat([df, branch_dummies], axis=1)

# Also add week fixed effects
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df_fe = pd.concat([df_fe, week_dummies], axis=1)

# Variables to include
branch_vars = [c for c in df_fe.columns if c.startswith('branch')]
week_vars = [c for c in df_fe.columns if c.startswith('week')]

# Main DiD variables + controls
main_vars = ['switched_branch', 'after_switch', 'treated', 'staffing_index', 'visits_index']
all_vars = main_vars + branch_vars + week_vars

# This will be a large model, so let's use a simpler approach with statsmodels
formula = 'service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index + C(branch_id) + C(week)'
model_fe = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# Re-examine the pre-trends more carefully
print("=" * 70)
print("CRITICAL ASSESSMENT: PRE-TRENDS ANALYSIS")
print("=" * 70)

# Weekly gaps between treatment and control
weekly = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack()
gap = weekly.loc[1] - weekly.loc[0]

print("\nTreatment-Control Gap by Week:")
print("-" * 50)
for w in range(6, 23):
    marker = " <-- SWITCH" if w == 15 else ""
    print(f"Week {w:2d}: {gap[w]:7.2f}{marker}")

# Fit pre-trend line (weeks 6-14)
pre_df = df[df['week'] < 15].copy()
pre_df['week_num'] = pre_df['week']

# What was the average weekly divergence before the switch?
model_pre = ols('service_score ~ switched_branch * week_num', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})

print("\n" + "=" * 70)
print("PRE-TREND STATISTICS (Weeks 6-14)")
print("=" * 70)
print(f"\nInteraction coefficient (differential trend): {model_pre.params['switched_branch:week_num']:.3f}")
print(f"Standard error: {model_pre.bse['switched_branch:week_num']:.3f}")
print(f"P-value: {model_pre.pvalues['switched_branch:week_num']:.4f}")

# Interpret: treatment branches were gaining X points per week MORE than control
diff_per_week = model_pre.params['switched_branch:week_num']
print(f"\nInterpretation: Treatment branches were improving by")
print(f"{diff_per_week:.3f} points PER WEEK faster than control branches BEFORE the switch")

# Over 8 weeks of pre-period, that's:
pre_period_gain = diff_per_week * 8
print(f"\nOver the 8-week pre-period, this amounts to {pre_period_gain:.2f} points")

# Post-period is 8 weeks (weeks 15-22)
# If the pre-trend continued, what would we expect?
expected_post_gain = diff_per_week * 8
print(f"\nIf this pre-trend continued unchanged, we'd expect {expected_post_gain:.2f} points")
print(f"of additional gain in the 8-week post-period WITHOUT any treatment effect")

# Actual DiD estimate
means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean()
did = (means.loc[1, 1] - means.loc[1, 0]) - (means.loc[0, 1] - means.loc[0, 0])
print(f"\nObserved DiD estimate: {did:.2f} points")
print(f"\nAdjusted estimate (subtracting pre-trend): {did - pre_period_gain:.2f} points")

print("\n" + "=" * 70)
print("ASSESSMENT")
print("=" * 70)
if did > pre_period_gain:
    print(f"The observed effect ({did:.2f}) exceeds the pre-trend ({pre_period_gain:.2f})")
    print(f"Residual treatment effect: ~{did - pre_period_gain:.2f} points")
else:
    print(f"The pre-trend ({pre_period_gain:.2f}) explains most or all of the observed effect ({did:.2f})")
    print("Cannot confidently attribute improvement to the new procedure")
