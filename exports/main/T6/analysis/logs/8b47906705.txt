import pandas as pd
import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Difference-in-differences regression
# service_score = beta0 + beta1*switched + beta2*after + beta3*(switched*after) + error
# beta3 is the treatment effect

model = smf.ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("=== DiD Regression Results ===")
print(model.summary())

# Also control for staffing and visits
model2 = smf.ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit()
print("\n=== DiD Regression with Controls ===")
print(model2.summary())

# Cluster-robust standard errors by branch
model3 = smf.ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Regression with Controls (Cluster-Robust SE by Branch) ===")
print(model3.summary())

import pandas as pd
import statsmodels.formula.api as smf
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create event time variable
df['event_time'] = df['week'] - 15  # Week 15 is when switch happened

# Create dummies for each event time, interacted with treatment
# Reference period: event_time = -1 (week 14, just before switch)
# Use 'm' for negative, 'p' for positive to avoid issues with patsy
for t in range(-8, 8):
    if t == -1:
        continue  # reference period
    if t < 0:
        var_name = f'treated_m{abs(t)}'
    elif t == 0:
        var_name = 'treated_p0'
    else:
        var_name = f'treated_p{t}'
    df[var_name] = ((df['event_time'] == t) & (df['switched_branch'] == 1)).astype(int)
    
# Build formula
event_vars = []
for t in range(-8, 8):
    if t == -1:
        continue
    if t < 0:
        event_vars.append(f'treated_m{abs(t)}')
    elif t == 0:
        event_vars.append('treated_p0')
    else:
        event_vars.append(f'treated_p{t}')

formula = 'service_score ~ ' + ' + '.join(event_vars) + ' + C(branch_id) + C(week)'
model_event = smf.ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients and confidence intervals
event_results = []
for t in range(-8, 8):
    if t == -1:
        event_results.append({'event_time': t, 'coef': 0, 'se': 0, 'ci_low': 0, 'ci_high': 0})
    else:
        if t < 0:
            var = f'treated_m{abs(t)}'
        elif t == 0:
            var = 'treated_p0'
        else:
            var = f'treated_p{t}'
        coef = model_event.params[var]
        se = model_event.bse[var]
        event_results.append({
            'event_time': t,
            'coef': coef,
            'se': se,
            'ci_low': coef - 1.96 * se,
            'ci_high': coef + 1.96 * se
        })

event_df = pd.DataFrame(event_results)
print("=== Event Study Results ===")
print(event_df)

# Plot event study
plt.figure(figsize=(10, 6))
plt.errorbar(event_df['event_time'], event_df['coef'], 
             yerr=[event_df['coef'] - event_df['ci_low'], event_df['ci_high'] - event_df['coef']],
             fmt='o', capsize=5, label='Treatment Effect')
plt.axhline(y=0, color='black', linestyle='--', alpha=0.5)
plt.axvline(x=-0.5, color='red', linestyle='-', alpha=0.5, label='Switch occurred')
plt.xlabel('Weeks Relative to Switch (0 = Week 15)')
plt.ylabel('Effect on Service Score')
plt.title('Event Study: Treatment Effect Over Time')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(-8, 8))
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as event_study.png")

# Check pre-trends (weeks -8 to -2 should be close to 0)
pre_trend_coefs = event_df[event_df['event_time'] < -1]['coef']
print(f"\nPre-trend coefficients (should be near 0):")
print(pre_trend_coefs.values)
print(f"Average pre-trend coefficient: {pre_trend_coefs.mean():.2f}")
print(f"Max absolute pre-trend coefficient: {pre_trend_coefs.abs().max():.2f}")

# Post-trend effects
post_trend_coefs = event_df[event_df['event_time'] >= 0]['coef']
print(f"\nPost-switch coefficients:")
print(post_trend_coefs.values)
print(f"Average post-switch coefficient: {post_trend_coefs.mean():.2f}")

import pandas as pd

df = pd.read_csv('branches.csv')

print("\nSwitched Branches:")
before_switched = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_switched = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
print(f"  Before: {before_switched:.2f}")
print(f"  After:  {after_switched:.2f}")
print(f"  Change: {after_switched - before_switched:.2f}")

print("\nControl Branches (kept old procedure):")
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
print(f"  Before: {before_control:.2f}")
print(f"  After:  {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")

print("\n=== Interpretation ===")
print("The DiD estimate of +2.14 means switched branches scored 2.14 points")
print("HIGHER than they would have without the new procedure.")
print("\nHowever, in ABSOLUTE terms, switched branches still saw scores DECLINE")
print(f"from {before_switched:.2f} to {after_switched:.2f} (-{before_switched - after_switched:.2f} points)")
print("\nThe new procedure APPEARS TO HAVE PROTECTED branches from a larger decline,")
print("rather than causing an absolute improvement.")

print("\n=== System-wide trend by week ===")
