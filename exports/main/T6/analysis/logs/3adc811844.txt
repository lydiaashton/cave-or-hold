import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch fixed effects
# Model: service_score ~ treated_post + after_switch + C(branch_id)
# We include after_switch to capture the common time trend
# We include branch fixed effects to control for time-invariant branch characteristics

# Using OLS with branch fixed effects (dummy variables for each branch)
model = ols('service_score ~ treated_post + after_switch + C(branch_id)', data=df)
results = model.fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== DIFFERENCE-IN-DIFFERENCES REGRESSION ===")
print("\nRegression Results (with branch fixed effects, clustered SEs):")
print(results.summary().tables[1])

# Extract the key coefficient
did_coef = results.params['treated_post']
did_se = results.bse['treated_post']
did_pvalue = results.pvalues['treated_post']
did_ci_lower = results.conf_int().loc['treated_post'][0]
did_ci_upper = results.conf_int().loc['treated_post'][1]

print(f"\n=== KEY RESULT ===")
print(f"DiD Coefficient: {did_coef:.3f}")
print(f"Standard Error: {did_se:.3f}")
print(f"95% CI: [{did_ci_lower:.3f}, {did_ci_upper:.3f}]")
print(f"P-value: {did_pvalue:.4f}")
print(f"Statistically significant at 5% level: {did_pvalue < 0.05}")

import pandas as pd

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Main DiD specification with branch fixed effects
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Model 1: Basic DiD with branch FE
model1 = ols('service_score ~ treated_post + after_switch + C(branch_id)', data=df)
results1 = model1.fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== MODEL 1: BASIC DiD WITH BRANCH FE ===")
print(f"treated_post coefficient: {results1.params['treated_post']:.3f}")
print(f"Standard error: {results1.bse['treated_post']:.3f}")
print(f"P-value: {results1.pvalues['treated_post']:.4f}")
print(f"95% CI: [{results1.conf_int().loc['treated_post'][0]:.3f}, {results1.conf_int().loc['treated_post'][1]:.3f}]")

# Model 2: DiD with branch FE and covariates (absorbing branch FE differently)
# Use demeaning approach to avoid collinearity
df_group = df.groupby('branch_id')
df['service_score_demeaned'] = df['service_score'] - df_group['service_score'].transform('mean')
df['treated_post_demeaned'] = df['treated_post'] - df_group['treated_post'].transform('mean')
df['after_switch_demeaned'] = df['after_switch'] - df_group['after_switch'].transform('mean')
df['staffing_index_demeaned'] = df['staffing_index'] - df_group['staffing_index'].transform('mean')
df['visits_index_demeaned'] = df['visits_index'] - df_group['visits_index'].transform('mean')

# Fixed effects regression using within transformation
model2 = ols('service_score_demeaned ~ treated_post_demeaned + after_switch_demeaned + staffing_index_demeaned + visits_index_demeaned', data=df)
