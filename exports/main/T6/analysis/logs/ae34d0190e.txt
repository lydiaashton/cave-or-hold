import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

# Simple DiD regression
# service_score = beta0 + beta1*treated + beta2*post + beta3*treated_post + error
# beta3 is the DiD estimator

model = ols('service_score ~ treated + post + treated_post', data=df).fit()
print("=== Simple DiD Regression ===")
print(model.summary())

# Now with branch fixed effects (more rigorous)
# This controls for time-invariant branch characteristics
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ treated_post + post + C(branch_id_str)', data=df).fit()
print("\n=== DiD with Branch Fixed Effects ===")
# Just show the key coefficients
print(model_fe.params[['treated_post', 'post']])
print("\nStandard errors:")
print(model_fe.bse[['treated_post', 'post']])
print("\nConfidence intervals:")
print(model_fe.conf_int().loc[['treated_post', 'post']])

# Create week dummies relative to switch
week_effects = []
for w in range(-8, 9):  # weeks -8 to +8 relative to switch
    if w != -1:  # omit week -1 as reference
        df[f'week_{w}'] = ((df['week'] - 15) == w).astype(int)
        df[f'week_{w}_treated'] = df[f'week_{w}'] * df['switched_branch']

# Run event study regression
formula = 'service_score ~ ' + ' + '.join([f'week_{w}_treated' for w in range(-8, 9) if w != -1]) + ' + C(branch_id_str) + C(week)'
model_event = ols(formula, data=df.assign(branch_id_str=df['branch_id'].astype(str))).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for treated interactions
event_study_results = []
for w in range(-8, 9):
    if w != -1:
        coef = model_event.params[f'week_{w}_treated']
        se = model_event.bse[f'week_{w}_treated']
