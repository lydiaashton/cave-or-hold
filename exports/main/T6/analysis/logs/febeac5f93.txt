import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Create treatment interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary: Average service scores by group and period
print("=== Service Score Summary ===")
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Calculate DiD manually
print("\n=== Difference-in-Differences Calculation ===")

# Treatment group (switched_branch=1)
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treatment_diff = treatment_after - treatment_before

# Control group (switched_branch=0)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_diff = control_after - control_before

# DiD estimate
did_estimate = treatment_diff - control_diff

print(f"Treatment group - Before: {treatment_before:.2f}, After: {treatment_after:.2f}, Change: {treatment_diff:.2f}")
print(f"Control group - Before: {control_before:.2f}, After: {control_after:.2f}, Change: {control_diff:.2f}")
print(f"\nDiD Estimate (Treatment Effect): {did_estimate:.2f}")

# Weekly trends visualization data
print("\n=== Weekly Averages by Group ===")
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
print(weekly_avg.to_string())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create treatment interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with cluster-robust standard errors
# Model: service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*treatment + error
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

# Get cluster-robust standard errors (clustered by branch)
cov_cluster = sm.cov_type.get_robustcov_results(model, 'cluster', groups=df['branch_id'])

print("=== DiD Regression Results ===")
print(cov_cluster.summary())

# Extract the treatment effect
treatment_effect = cov_cluster.params['treatment']
treatment_se = cov_cluster.bse['treatment']
treatment_pvalue = cov_cluster.pvalues['treatment']

print(f"\nTreatment Effect: {treatment_effect:.3f}")
print(f"Standard Error: {treatment_se:.3f}")
print(f"P-value: {treatment_pvalue:.3f}")
print(f"95% CI: [{treatment_effect - 1.96*treatment_se:.3f}, {treatment_effect + 1.96*treatment_se:.3f}]")

import pandas as pd

pre_weekly = pre_treatment.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Plot pre-treatment trends
fig, ax = plt.subplots(figsize=(10, 6))

for group in [0, 1]:
    group_data = pre_weekly[pre_weekly['switched_branch'] == group]
    label = 'Switched Branches' if group == 1 else 'Non-Switched Branches'
    ax.plot(group_data['week'], group_data['service_score'], marker='o', label=label, linewidth=2)

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Pre-Treatment Trends: Service Scores by Group', fontsize=14)
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()

print("Saved pre_treatment_trends.png")

# Test for parallel trends statistically
# Regress service_score on week, switched_branch, and their interaction (pre-treatment only)
pre_treatment = pre_treatment.copy()
pre_treatment['week_centered'] = pre_treatment['week'] - 10  # Center around middle of pre-period

parallel_trends_model = ols('service_score ~ week_centered * switched_branch', data=pre_treatment).fit(cov_type='cluster', cov_kwds={'groups': pre_treatment['branch_id']})

interaction_coef = parallel_trends_model.params['week_centered:switched_branch']
