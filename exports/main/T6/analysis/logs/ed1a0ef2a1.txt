import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Run DiD regression with robust standard errors
# Basic DiD: service_score ~ switched_branch * after_switch
model = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='HC1')
print("=== Basic DiD Regression ===")
print(model.summary())

# Now let's add branch fixed effects and week fixed effects for a more robust analysis
# This controls for time-invariant branch characteristics and common time trends
print("\n=== DiD with Branch and Week Fixed Effects ===")
# Create dummy variables for branch and week
df_model = df.copy()
df_model = pd.get_dummies(df_model, columns=['branch_id'], prefix='branch', drop_first=True)
df_model = pd.get_dummies(df_model, columns=['week'], prefix='week', drop_first=True)

# Run regression with fixed effects
fe_cols = [c for c in df_model.columns if c.startswith('branch_') or c.startswith('week_')]
formula = 'service_score ~ switched_branch * after_switch + ' + ' + '.join(fe_cols)
model_fe = ols(formula, data=df_model).fit(cov_type='cluster', cov_kwds={'groups': df_model['branch_id']})
print(model_fe.summary())

# Extract the interaction term coefficient (the DiD estimate)
print(f"\nDiD coefficient (interaction term): {model_fe.params['switched_branch:after_switch']:.3f}")
print(f"Standard error: {model_fe.bse['switched_branch:after_switch']:.3f}")
print(f"95% CI: [{model_fe.conf_int().loc['switched_branch:after_switch'][0]:.3f}, {model_fe.conf_int().loc['switched_branch:after_switch'][1]:.3f}]")
print(f"P-value: {model_fe.pvalues['switched_branch:after_switch']:.4f}")

import pandas as pd
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create event study with renamed columns
df_event = df.copy()
df_event['weeks_to_switch'] = df_event['week'] - 15

# Create indicators for each week relative to switch (using week 14 as reference, which is -1)
week_indicators = {}
for w in range(-9, 8):
    if w != -1:
        col_name = f'w{abs(w)}_neg' if w < 0 else f'w{w}_pos'
        df_event[col_name] = ((df_event['weeks_to_switch'] == w) & (df_event['switched_branch'] == 1)).astype(int)
        week_indicators[w] = col_name

print("Event study column mapping:")
for w, col in week_indicators.items():
    print(f"  Week {15+w} (relative {w:+d}): {col}")

# Run regression
from statsmodels.formula.api import ols
event_cols = list(week_indicators.values())
formula = 'service_score ~ ' + ' + '.join(event_cols) + ' + C(week)'
model_event = ols(formula, data=df_event).fit(cov_type='cluster', cov_kwds={'groups': df_event['branch_id']})

print("\nEvent study coefficients (relative to week 14, control group baseline):")
results = []
for w in range(-9, 8):
    if w != -1:
        col = week_indicators[w]
        coef = model_event.params[col]
        se = model_event.bse[col]
        pval = model_event.pvalues[col]
        ci_lower = model_event.conf_int().loc[col][0]
        ci_upper = model_event.conf_int().loc[col][1]
        results.append((w, coef, se, pval, ci_lower, ci_upper))
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"  Week {15+w:2d} (relative {w:+3d}): {coef:6.3f} (SE: {se:.3f}) [{ci_lower:.2f}, {ci_upper:.2f}] {sig}")

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
ws = [r[0] for r in results]
coefs = [r[1] for r in results]
ses = [r[2] for r in results]
ci_lower = [r[4] for r in results]
ci_upper = [r[5] for r in results]

ax.errorbar(ws, coefs, yerr=[1.96*s for s in ses], fmt='o', capsize=5, markersize=8, linewidth=2)
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')
ax.set_xlabel('Weeks Relative to Switch', fontsize=12)
ax.set_ylabel('Effect on Service Score (vs Week 14 baseline)', fontsize=12)
ax.set_title('Event Study: Effect of New Procedure on Service Scores', fontsize=14)
ax.set_xticks(ws)
ax.set_xticklabels([f'{15+w}' for w in ws], rotation=45)
ax.grid(True, alpha=0.3)
ax.legend()
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")
