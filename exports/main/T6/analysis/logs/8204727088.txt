import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data

# Create the interaction term for DiD
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit()
print("=== Simple Difference-in-Differences Regression ===")

import pandas as pd
import numpy as np

from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression with robust standard errors clustered by branch
model = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit()
print("=== Simple Difference-in-Differences Regression ===")
print(f"Treatment effect (treatment_post coefficient): {model.params['treatment_post']:.3f}")
print(f"Standard error: {model.bse['treatment_post']:.3f}")
print(f"P-value: {model.pvalues['treatment_post']:.4f}")
print(f"95% CI: [{model.conf_int().loc['treatment_post'][0]:.3f}, {model.conf_int().loc['treatment_post'][1]:.3f}]")

# DiD with branch fixed effects (absorbing switched_branch since it's time-invariant)
print("\n=== DiD with Branch Fixed Effects ===")
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ after_switch + treatment_post + C(branch_id_str)', data=df).fit()
print(f"Treatment effect (treatment_post coefficient): {model_fe.params['treatment_post']:.3f}")
print(f"Standard error: {model_fe.bse['treatment_post']:.3f}")
print(f"P-value: {model_fe.pvalues['treatment_post']:.4f}")
print(f"95% CI: [{model_fe.conf_int().loc['treatment_post'][0]:.3f}, {model_fe.conf_int().loc['treatment_post'][1]:.3f}]")

# DiD with branch and week fixed effects
print("\n=== DiD with Branch and Week Fixed Effects ===")
