df = pd.read_csv('branches.csv')

# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimator

model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("Difference-in-Differences Regression Results:")

from statsmodels.stats.diagnostic import het_breuschpagan

# Let's use a two-way fixed effects model properly

# Create the interaction term

# Run regression with branch fixed effects (switched_branch drops out due to collinearity)
# This is the standard DiD with unit fixed effects

se_cl = np.sqrt(np.diag(cov_cl))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Use linearmodels for proper panel data estimation with clustered SEs
try:
    from linearmodels.panel import PanelOLS
    
    # Set up panel data
    df_panel = df.set_index(['branch_id', 'week'])
    
    # Run PanelOLS with entity (branch) fixed effects and clustered SEs
    mod = PanelOLS(df_panel['service_score'], 
                   df_panel[['after_switch', 'treatment']], 
                   entity_effects=True)
    res = mod.fit(cov_type='clustered', cluster_entity=True)
    print("PanelOLS with Branch Fixed Effects and Clustered SEs:")
    print("="*60)
    print(res)
except ImportError:
    print("linearmodels not available, using manual approach")
    
    # Manual clustered SE calculation
    model = ols('service_score ~ after_switch + treatment + C(branch_id)', data=df).fit()
    
    # Get residuals and cluster by branch
    df['resid'] = model.resid
    
    # Calculate cluster-robust SEs
    G = df['branch_id'].nunique()  # number of clusters
    K = model.df_model + 1  # number of parameters
    
    # Get the design matrix
    X = model.model.exog
    
    # Calculate sandwich estimator
    scores = []
    for g in df['branch_id'].unique():
        mask = df['branch_id'] == g
        X_g = X[mask]
        resid_g = df.loc[mask, 'resid'].values
        score_g = (X_g.T * resid_g).sum(axis=1)
        scores.append(score_g)
    
    scores = np.array(scores)
    bread = np.linalg.inv(X.T @ X)
    meat = scores.T @ scores
    cov_cluster = (G / (G - 1)) * ((G - 1) / (G - K)) * bread @ meat @ bread
    
    se_cluster = np.sqrt(np.diag(cov_cluster))
    
    print("\nManual Clustered SEs:")
    for i, name in enumerate(model.params.index):
        print(f"{name}: coef={model.params[name]:.3f}, cluster_se={se_cluster[i]:.3f}")
