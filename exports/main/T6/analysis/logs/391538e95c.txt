import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch*after_switch) + error
# beta3 is the DiD estimate

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Also run with branch fixed effects to control for time-invariant branch characteristics
print("\n=== DiD with Branch Fixed Effects ===")
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df, branch_dummies], axis=1)

# Build formula with fixed effects
fe_cols = [col for col in df_with_fe.columns if col.startswith('branch')]
fe_formula = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(fe_cols)
model_fe = ols(fe_formula, data=df_with_fe).fit()
print(model_fe.summary())

# Check parallel trends - compare pre-trends between groups
print("\n=== Parallel Trends Check (Pre-Switch Weeks 6-14) ===")
pre_data = df[df['week'] < 15]

# Test if there's a significant difference in trends before the switch
pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()
pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_data).fit()
print(pre_model.summary())
print("\nInteraction term (switched_branch:week_centered) tests parallel trends.")
print("If not significant, parallel trends assumption holds.")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.diagnostic import het_breuschpagan

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with clustered standard errors at branch level
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

# Get cluster-robust standard errors
cluster_robust_se = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("=== DiD Regression with Cluster-Robust Standard Errors ===")
print(cluster_robust_se.summary())

# Also check the effect size relative to the baseline
baseline_mean = df[df['after_switch']==0]['service_score'].mean()
treatment_effect = cluster_robust_se.params['treatment']
print(f"\n=== Effect Size ===")
print(f"Baseline average service score: {baseline_mean:.2f}")
print(f"Estimated treatment effect: {treatment_effect:.2f} points")
print(f"Percentage change: {treatment_effect/baseline_mean*100:.1f}%")

# Confidence interval
ci_lower = cluster_robust_se.conf_int().loc['treatment'][0]
ci_upper = cluster_robust_se.conf_int().loc['treatment'][1]
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")

# Let's also look at the distribution of changes
print("\n=== Distribution of Score Changes ===")
branch_scores = df.pivot(index='branch_id', columns='after_switch', values='service_score')
branch_scores['before_mean'] = df[df['after_switch']==0].groupby('branch_id')['service_score'].mean()
branch_scores['after_mean'] = df[df['after_switch']==1].groupby('branch_id')['service_score'].mean()
branch_scores['change'] = branch_scores['after_mean'] - branch_scores['before_mean']

# Merge with switch status
switch_status = df.groupby('branch_id')['switched_branch'].first()
branch_scores['switched'] = switch_status

switched_changes = branch_scores[branch_scores['switched']==1]['change']
control_changes = branch_scores[branch_scores['switched']==0]['change']

print(f"  Min: {switched_changes.min():.2f}, Max: {switched_changes.max():.2f}")
