print("\nAfter switch (weeks 15-22):")
after = df[df['after_switch']==1].groupby('switched_branch')['service_score'].mean()
print(after)

print("\n=== Difference-in-Differences Calculation ===")
# Treatment group (switched=1)
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_diff = treat_after - treat_before

# Control group (switched=0)

ctrl_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
ctrl_diff = ctrl_after - ctrl_before

did_effect = treat_diff - ctrl_diff

print(f"\nTreatment group (switched): Before={treat_before:.2f}, After={treat_after:.2f}, Change={treat_diff:.2f}")
print(f"Control group (not switched): Before={ctrl_before:.2f}, After={ctrl_after:.2f}, Change={ctrl_diff:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks before/after switch)
# Week 15 is when switch happened, so relative_week = week - 14 (week 14 is -0, week 15 is 1)
df['relative_week'] = df['week'] - 14

# Event study regression - interact switched_branch with each week dummy
# Use week 14 as reference (last pre-treatment week)
weeks = list(range(6, 23))
pre_weeks = [w for w in weeks if w < 14]  # 6-13
post_weeks = [w for w in weeks if w >= 15]  # 15-22

# Create week dummies and interactions
for w in weeks:
    if w != 14:  # reference week
        df[f'week_{w}'] = (df['week'] == w).astype(int)
        df[f'switched_x_week_{w}'] = ((df['week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Build formula
interactions = ' + '.join([f'switched_x_week_{w}' for w in weeks if w != 14])
week_dummies = ' + '.join([f'week_{w}' for w in weeks if w != 14])
formula = f'service_score ~ {interactions} + {week_dummies} + staffing_index + visits_index'

model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results (coefficients for switched_branch x week interactions) ===")
print("Reference week: 14 (last pre-treatment week)")
print("\nPre-treatment coefficients (should be ~0 if parallel trends hold):")
for w in pre_weeks:
    coef = model.params[f'switched_x_week_{w}']
    se = model.bse[f'switched_x_week_{w}']
    pval = model.pvalues[f'switched_x_week_{w}']
    ci_low = coef - 1.96*se
    ci_high = coef + 1.96*se

    ci_low = coef - 1.96*se
