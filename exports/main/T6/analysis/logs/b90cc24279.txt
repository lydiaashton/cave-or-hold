# Run DiD regression with cluster-robust standard errors
# service_score = beta0 + beta1*treatment + beta2*post + beta3*(treatment*post) + error
# beta3 is the DiD estimate

model = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Test for pre-trends: interact treatment with week dummies for pre-period only
# Create week dummies
pre_data = df[df['week'] <= 14].copy()
pre_data['week_centered'] = pre_data['week'] - 10  # Center around middle of pre-period

# Test if treatment branches had different trends in pre-period
model_trend = ols('service_score ~ switched_branch * week_centered', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print("=== Pre-Trend Test (Treatment x Week interaction in pre-period) ===")
print(model_trend.summary())

# Also do a more formal event study with week fixed effects
df['week_cat'] = df['week'].astype('category')
# Use week 14 as reference (last pre-switch week)
df['week_14_ref'] = (df['week'] == 14).astype(int)

# Create dummies for each week relative to switch
for w in range(6, 23):
    if w != 14:
        df[f'week_{w}'] = (df['week'] == w).astype(int)

# Event study regression
event_vars = ' + '.join([f'week_{w}' for w in range(6, 23) if w != 14])
event_formula = f'service_score ~ switched_branch + {event_vars} + switched_branch:week_6 + switched_branch:week_7 + switched_branch:week_8 + switched_branch:week_9 + switched_branch:week_10 + switched_branch:week_11 + switched_branch:week_12 + switched_branch:week_13 + switched_branch:week_15 + switched_branch:week_16 + switched_branch:week_17 + switched_branch:week_18 + switched_branch:week_19 + switched_branch:week_20 + switched_branch:week_21 + switched_branch:week_22'

model_event = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Event Study Coefficients (Treatment x Week interactions) ===")
coef_df = pd.DataFrame({
    'coef': model_event.params,
    'pvalue': model_event.pvalues,
    'ci_lower': model_event.conf_int()[0],
    'ci_upper': model_event.conf_int()[1]
})
# Filter to just the interaction terms
interaction_terms = [c for c in coef_df.index if 'switched_branch:week' in c]
print(coef_df.loc[interaction_terms].sort_index())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Two-way fixed effects model (branch + week FE)
# This is the gold standard for DiD with panel data
# Note: switched_branch is absorbed by branch FE, after_switch is absorbed by week FE
# Only the interaction term remains

# Create week dummies (excluding one reference week)
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df = pd.concat([df, week_dummies], axis=1)

# Get the week dummy column names
week_cols = [c for c in df.columns if c.startswith('week_')]

# Build formula with branch FE and week FE
branch_fe = ' + '.join([f'C(branch_id)' for _ in range(1)])  # Will be expanded by statsmodels
week_fe = ' + '.join(week_cols)

# The interaction term is what we care about
formula = f'service_score ~ switched_branch:after_switch + C(branch_id) + {week_fe}'

model_twfe = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Two-Way Fixed Effects DiD Model ===")
# Just show the key coefficient
print(f"\nTreatment Effect (switched_branch:after_switch):")
print(f"  Coefficient: {model_twfe.params['switched_branch:after_switch']:.3f}")

print(f"  p-value: {model_twfe.pvalues['switched_branch:after_switch']:.4f}")
