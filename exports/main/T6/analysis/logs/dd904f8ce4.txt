import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary table: mean service scores by group and period
summary_table = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).unstack()
print("Mean Service Scores by Group and Period:")
print(summary_table)
print()

# Calculate the DiD estimate manually
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
pre_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
post_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("Pre-period (weeks 6-14):")
print(f"  Control group mean: {pre_control:.2f}")
print(f"  Treatment group mean: {pre_treatment:.2f}")
print(f"  Difference: {pre_treatment - pre_control:.2f}")
print()
print("Post-period (weeks 15-22):")
print(f"  Control group mean: {post_control:.2f}")
print(f"  Treatment group mean: {post_treatment:.2f}")
print(f"  Difference: {post_treatment - post_control:.2f}")
print()

did_estimate = (post_treatment - pre_treatment) - (post_control - pre_control)
print(f"Difference-in-Differences Estimate: {did_estimate:.2f}")
print()

# Now run regression with branch fixed effects
# DiD regression: service_score = beta0 + beta1*after_switch + beta2*switched_branch + beta3*treatment + error
# With branch fixed effects to control for time-invariant branch characteristics

# Model 1: Simple DiD without controls
model1 = ols('service_score ~ after_switch + switched_branch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Model 1: Simple DiD (clustered SEs by branch)")
print(model1.summary())
print()

# Model 2: DiD with branch fixed effects
df['branch_id'] = df['branch_id'].astype(str)
model2 = ols('service_score ~ after_switch + treatment + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Model 2: DiD with Branch Fixed Effects (clustered SEs by branch)")
print(model2.summary())
print()

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Pre-trend test
df_pre = df[df['week'] < 15].copy()
df_pre['week_centered'] = df_pre['week'] - df_pre['week'].mean()

model_pre = ols('service_score ~ switched_branch * week_centered + C(branch_id)', data=df_pre).fit(cov_type='cluster', cov_kwds={'groups': df_pre['branch_id']})
print("Pre-trend test (interaction of switched_branch with week trend):")
print(model_pre.summary().tables[1].iloc[-2:])
print()

# Let's also run a more flexible event study specification
# Create relative week variable (weeks since switch, with week 15 = 0)
df['relative_week'] = df['week'] - 15

# Create dummies for each week relative to switch
for w in range(-9, 8):  # weeks -9 to +7 relative to week 15
    df[f'week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (omitting week -1 as reference)
event_study_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = f'service_score ~ {" + ".join(event_study_vars)} + C(branch_id) + C(week)'
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Coefficients (treatment effect by week relative to switch):")
coef_table = model_event.summary().tables[1]
for var in event_study_vars:
    row = coef_table[coef_table.index.str.contains(var)]
    if len(row) > 0:
        print(row[['coef', 'P>|z|']].to_string())
print()

# Create visualization
fig, ax = plt.subplots(figsize=(12, 6))

# Plot weekly means
