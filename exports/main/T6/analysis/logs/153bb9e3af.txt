import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run DiD regression with cluster-robust standard errors
# Model: service_score = β0 + β1*treated + β2*post + β3*(treated*post) + ε
# β3 is our DiD estimator

model = ols('service_score ~ switched_branch * after_switch', data=df).fit()

# Get cluster-robust standard errors at branch level
cov_cluster = model.get_robustcov_results(cov_type='cluster', clusters=df['branch_id'])

print("=== DiD Regression Results ===")
print(cov_cluster.summary())

# Extract the interaction term coefficient

bse = cov_cluster.bse

print(f"switched_branch:after_switch (DiD effect): {params['switched_branch:after_switch']:.3f} (SE: {bse['switched_branch:after_switch']:.3f}, p={pvalues['switched_branch:after_switch']:.4f})")

print("\n=== Weekly Averages by Group ===")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Main DiD model
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
cov_cluster = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Extract the interaction term
params = pd.Series(cov_cluster.params, index=model.params.index)
bse = pd.Series(cov_cluster.bse, index=model.params.index)
pvalues = pd.Series(cov_cluster.pvalues, index=model.params.index)

did_coef = params['switched_branch:after_switch']
did_se = bse['switched_branch:after_switch']
did_pvalue = pvalues['switched_branch:after_switch']

print("=== MAIN DiD RESULTS ===")
print(f"Coefficient: {did_coef:.3f}")
print(f"Standard Error: {did_se:.3f}")
print(f"P-value: {did_pvalue:.4f}")
print(f"95% CI: [{did_coef - 1.96*did_se:.3f}, {did_coef + 1.96*did_se:.3f}]")

# Context
pre_mean = df[df['after_switch']==0]['service_score'].mean()
print(f"\nBaseline (pre-switch mean): {pre_mean:.1f}")
print(f"Effect as % of baseline: {did_coef/pre_mean*100:.1f}%")

# Check if effect is practically meaningful
print(f"\n=== PRACTICAL SIGNIFICANCE ===")
print(f"Is 2.48 points meaningful on a ~50 point scale?")
print(f"Standard deviation of service_score: {df['service_score'].std():.1f}")
print(f"Effect as % of 1 SD: {did_coef/df['service_score'].std()*100:.1f}%")

# Check post-switch weeks individually
post = df[df['after_switch']==1]
print("\n=== POST-SWITCH AVERAGES BY GROUP ===")
print(post.groupby('switched_branch')['service_score'].agg(['mean', 'std']))
