import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['did_interaction'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with robust standard errors
# Model 1: Basic DiD
model1 = ols('service_score ~ switched_branch + after_switch + did_interaction', data=df).fit(cov_type='HC1')
print("Model 1: Basic Difference-in-Differences")
print(model1.summary())

# Model 2: DiD with control variables
model2 = ols('service_score ~ switched_branch + after_switch + did_interaction + staffing_index + visits_index', data=df).fit(cov_type='HC1')
print("\n" + "="*80)
print("Model 2: DiD with Control Variables")
print(model2.summary())

# Model 3: With branch fixed effects
df['branch_id_str'] = df['branch_id'].astype(str)
model3 = ols('service_score ~ after_switch + did_interaction + staffing_index + visits_index + C(branch_id_str)', data=df).fit(cov_type='HC1')
print("\n" + "="*80)
print("Model 3: DiD with Branch Fixed Effects")
print(model3.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Check parallel trends - average service scores by week for each group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
print("Weekly averages by group:")
print(weekly_avg.pivot(index='week', columns='switched_branch', values='service_score'))

# Plot the trends
plt.figure(figsize=(10, 6))
for group in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == group]
    label = 'New Procedure (Treatment)' if group == 1 else 'Old Procedure (Control)'
    plt.plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2)

plt.axvline(x=14.5, color='red', linestyle='--', label='Switch Point (Week 15)')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Score Trends: Treatment vs Control Groups')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('trends_plot.png', dpi=150, bbox_inches='tight')
print("\nPlot saved as trends_plot.png")

# Check pre-trends statistically (weeks 6-14 only)
pre_data = df[df['week'] <= 14].copy()
pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()
pre_trend_model = ols('service_score ~ switched_branch * week_centered', data=pre_data).fit(cov_type='HC1')
print("\n" + "="*80)
print("Pre-trend test (interaction tests if trends differ before switch):")
print(pre_trend_model.summary())

# Get the coefficient and confidence interval from the main model
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ after_switch + did_interaction + staffing_index + visits_index + C(branch_id_str)', data=df).fit(cov_type='HC1')
did_coef = model_fe.params['did_interaction']
did_se = model_fe.bse['did_interaction']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se
print(f"\nDiD Effect Estimate: {did_coef:.2f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {model_fe.pvalues['did_interaction']:.3f}")
