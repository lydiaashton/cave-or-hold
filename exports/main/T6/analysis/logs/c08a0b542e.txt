import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster

# Load the data
df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD with branch and week fixed effects
# This controls for both time-invariant branch characteristics and common time shocks
model_twfe = ols('service_score ~ C(branch_id) + C(week) + treatment', data=df).fit()
cov_twfe = cov_cluster(model_twfe, df['branch_id'])
se_twfe = np.sqrt(np.diag(cov_twfe))

print("DiD with Branch and Week Fixed Effects (Two-Way FE):")
print(f"treatment coefficient: {model_twfe.params['treatment']:.3f}")
print(f"Clustered SE for treatment: {se_twfe[model_twfe.params.index.get_loc('treatment')]:.3f}")
print(f"t-statistic: {model_twfe.params['treatment'] / se_twfe[model_twfe.params.index.get_loc('treatment')]:.2f}")
print(f"p-value: {model_twfe.pvalues['treatment']:.6f}")

# Also run with controls
model_twfe_controls = ols('service_score ~ C(branch_id) + C(week) + treatment + staffing_index + visits_index', data=df).fit()
cov_twfe_controls = cov_cluster(model_twfe_controls, df['branch_id'])
se_twfe_controls = np.sqrt(np.diag(cov_twfe_controls))

print("\n\nDiD with Branch + Week FE and Controls:")
print(f"treatment coefficient: {model_twfe_controls.params['treatment']:.3f}")
print(f"Clustered SE for treatment: {se_twfe_controls[model_twfe_controls.params.index.get_loc('treatment')]:.3f}")
print(f"t-statistic: {model_twfe_controls.params['treatment'] / se_twfe_controls[model_twfe_controls.params.index.get_loc('treatment')]:.2f}")
print(f"p-value: {model_twfe_controls.pvalues['treatment']:.6f}")

# Calculate confidence interval
coef = model_twfe_controls.params['treatment']
se = se_twfe_controls[model_twfe_controls.params.index.get_loc('treatment')]
ci_lower = coef - 1.96 * se
ci_upper = coef + 1.96 * se
print(f"\n95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")

# Effect size relative to mean
mean_score = df['service_score'].mean()
print(f"\nEffect as % of mean score: {coef/mean_score*100:.1f}%")
