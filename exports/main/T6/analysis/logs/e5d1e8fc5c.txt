print("\nWeeks in data:")
print(df['week'].value_counts().sort_index())

# Service score summary by group and period
print("\nService Score Summary by Group and Period:")
df['period'] = df['after_switch'].map({0: 'Before', 1: 'After'})
df['group'] = df['switched_branch'].map({0: 'Control', 1: 'Treatment'})

summary = df.groupby(['group', 'period'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Calculate the DiD manually
print("\n--- Difference-in-Differences Calculation ---")

# Treatment group
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_diff = treatment_after - treatment_before

# Control group
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
control_diff = control_after - control_before

# DiD estimate
did_estimate = treatment_diff - control_diff

print(f"Treatment (switched) - Before: {treatment_before:.3f}, After: {treatment_after:.3f}, Change: {treatment_diff:.3f}")
print(f"Control (not switched) - Before: {control_before:.3f}, After: {control_after:.3f}, Change: {control_diff:.3f}")
print(f"\nDiD Estimate (Treatment Change - Control Change): {did_estimate:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create a cleaner parallel trends visualization data
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
weekly_avg.columns = ['Control', 'Treatment']
weekly_avg = weekly_avg.reset_index()

# Plot parallel trends
fig, ax = plt.subplots(figsize=(10, 6))
ax.plot(weekly_avg['week'], weekly_avg['Control'], marker='o', label='Control (No Switch)', linewidth=2)

plt.savefig('parallel_trends.png', dpi=150, bbox_inches='tight')
print("Saved parallel_trends.png")

# Pre-treatment trend test - regress service score on treatment, week, and their interaction (pre-period only)

pre_model = ols('service_score ~ switched_branch + week + switched_branch:week', data=pre_data).fit(cov_type='HC3')

# The key coefficient is switched_branch:week - if significant, parallel trends assumption may be violated
interaction_coef = pre_model.params['switched_branch:week']
interaction_pval = pre_model.pvalues['switched_branch:week']
print(f"\nPre-treatment interaction coefficient: {interaction_coef:.4f}, p-value: {interaction_pval:.4f}")

# Main DiD with robust SE
print("\n=== Main DiD Results ===")
did_model = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='HC3')
print(did_model.summary().tables[1][['coef', 'std err', 'P>|z|']])

# DiD with covariates
print("\n=== DiD with Covariates ===")
did_cov_model = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='HC3')
print(did_cov_model.summary().tables[1][['coef', 'std err', 'P>|z|']])

# Effect size interpretation
did_effect = did_model.params['switched_branch:after_switch']
did_se = did_model.bse['switched_branch:after_switch']
print(f"\nDiD Effect: {did_effect:.2f} points (SE: {did_se:.2f})")
print(f"95% CI: [{did_effect - 1.96*did_se:.2f}, {did_effect + 1.96*did_se:.2f}]")

# Compare to baseline average service score
baseline_avg = df['service_score'].mean()
print(f"\nBaseline average service score: {baseline_avg:.2f}")
print(f"Effect as % of baseline: {did_effect/baseline_avg*100:.1f}%")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study approach - estimate effect for each week relative to switch
# Create relative week indicator (0 = week 14, the last pre-switch week)
df['relative_week'] = df['week'] - 14

# Create indicators for each week relative to switch, omitting week -1 (week 13) as reference
# Actually, let's use week 14 (relative_week=0) as the reference since it's the last pre-period
weeks = sorted(df['relative_week'].unique())
print("Relative weeks:", weeks)

# Create event study variables
for w in weeks:
    if w != 0:  # reference period
        df[f'week_{w}'] = (df['relative_week'] == w).astype(int)

# Interact each week indicator with treatment
# This gives us the treatment effect for each week relative to the reference
event_study_vars = '+'.join([f'switched_branch:week_{w}' for w in weeks if w != 0])
event_study_formula = f'service_score ~ switched_branch + {" + ".join([f"week_{w}" for w in weeks if w != 0])} + {event_study_vars}'

print("\n=== Event Study Model ===")
event_model = ols(event_study_formula, data=df).fit(cov_type='HC3')

# Extract the interaction coefficients (treatment effects by week)
treatment_effects = {}
for w in weeks:
    if w != 0:
        coef_name = f'switched_branch:week_{w}'
        if coef_name in event_model.params:
            treatment_effects[w] = {
                'coef': event_model.params[coef_name],
                'se': event_model.bse[coef_name],
                'pval': event_model.pvalues[coef_name]
            }

# Print results
print("\nEvent Study Results (Treatment Effect by Relative Week):")
print(f"{'Week':>6} {'Relative':>8} {'Effect':>8} {'SE':>8} {'95% CI Low':>10} {'95% CI High':>11} {'Signif':>7}")
print("-" * 70)
for w in sorted(treatment_effects.keys()):
    te = treatment_effects[w]
    ci_low = te['coef'] - 1.96 * te['se']
    ci_high = te['coef'] + 1.96 * te['se']
    signif = '***' if te['pval'] < 0.01 else ('**' if te['pval'] < 0.05 else ('*' if te['pval'] < 0.1 else ''))
    print(f"{w+14:>6} {w:>8} {te['coef']:>8.2f} {te['se']:>8.2f} {ci_low:>10.2f} {ci_high:>11.2f} {signif:>7}")

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
weeks_sorted = sorted(treatment_effects.keys())
effects = [treatment_effects[w]['coef'] for w in weeks_sorted]
ci_low = [treatment_effects[w]['coef'] - 1.96 * treatment_effects[w]['se'] for w in weeks_sorted]
ci_high = [treatment_effects[w]['coef'] + 1.96 * treatment_effects[w]['se'] for w in weeks_sorted]

ax.plot(weeks_sorted, effects, marker='o', linewidth=2, markersize=8, label='Treatment Effect')
ax.fill_between(weeks_sorted, ci_low, ci_high, alpha=0.3, label='95% CI')
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=0.5, color='red', linestyle='--', linewidth=2, label='Switch Point')
ax.set_xlabel('Relative Week (0 = Week 14, last pre-switch week)')
ax.set_ylabel('Treatment Effect on Service Score')
ax.set_title('Event Study: Treatment Effect by Week')
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(weeks_sorted)
plt.savefig('event_study.png', dpi=150, bbox_inches='tight')
print("\nSaved event_study.png")
