import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Proper fixed effects model using branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df, branch_dummies], axis=1)

# Build formula with branch fixed effects
fe_cols = [c for c in df_with_fe.columns if c.startswith('branch_')]
formula = 'service_score ~ after_switch + treatment + staffing_index + visits_index + ' + ' + '.join(fe_cols)

model_fe_full = ols(formula, data=df_with_fe).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("DiD with Branch Fixed Effects (Full Model):")
print("=" * 60)
print(model_fe_full.summary())

# Extract just the treatment effect
print("\n\nTreatment effect with branch FE:", model_fe_full.params['treatment'])
print("Std error:", model_fe_full.bse['treatment'])
print("P-value:", model_fe_full.pvalues['treatment'])
print("95% CI:", model_fe_full.conf_int().loc['treatment'].values)
