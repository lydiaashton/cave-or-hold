import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Number of unique branches
print("Number of unique branches:", df['branch_id'].nunique())

# How many branches switched vs didn't switch
switched_counts = df.groupby('branch_id')['switched_branch'].first().value_counts()
print("\nBranches by switch status:")
print(switched_counts)

# Weeks covered
print("\nWeeks covered:", sorted(df['week'].unique()))

# Create a pivot to see before/after averages by group
df['period'] = df['after_switch'].map({0: 'Before', 1: 'After'})
df['group'] = df['switched_branch'].map({0: 'Control', 1: 'Treatment'})

# Summary table: average service score by group and period
summary = df.groupby(['group', 'period'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("\nService Score by Group and Period:")
print(summary)

# Calculate the difference-in-differences manually
before_control = df[(df['group'] == 'Control') & (df['period'] == 'Before')]['service_score'].mean()
after_control = df[(df['group'] == 'Control') & (df['period'] == 'After')]['service_score'].mean()
before_treatment = df[(df['group'] == 'Treatment') & (df['period'] == 'Before')]['service_score'].mean()
after_treatment = df[(df['group'] == 'Treatment') & (df['period'] == 'After')]['service_score'].mean()

print("\n--- Difference-in-Differences Calculation ---")
print(f"Control group: Before={before_control:.2f}, After={after_control:.2f}, Change={after_control - before_control:.2f}")
print(f"Treatment group: Before={before_treatment:.2f}, After={after_treatment:.2f}, Change={after_treatment - before_treatment:.2f}")
did_effect = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDiD Estimate (Treatment effect): {did_effect:.2f}")

import pandas as pd

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create DiD variables
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treated'] * df['post']

# Run DiD regression with branch fixed effects
# Using OLS with branch as categorical variable

# Basic DiD regression
model = ols('service_score ~ treated + post + did + C(branch_id)', data=df).fit()
print("=== DiD Regression with Branch Fixed Effects ===")
print(model.summary())

# Also run a simpler version without fixed effects for comparison
model_simple = ols('service_score ~ treated + post + did', data=df).fit()

from statsmodels.formula.api import ols

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create DiD variables
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treated'] * df['post']

# Model 1: Basic DiD with branch fixed effects
model1 = ols('service_score ~ treated + post + did + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Model 2: DiD with covariates (staffing and visits)
model2 = ols('service_score ~ treated + post + did + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Model 3: DiD with week fixed effects instead of just post
df['week_cat'] = df['week'].astype(str)
model3 = ols('service_score ~ treated + did + C(week_cat) + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Model Comparison ===")
print("\nModel 1: Basic DiD (branch FE)")
print(f"DID coefficient: {model1.params['did']:.3f} (SE: {model1.bse['did']:.3f}, p={model1.pvalues['did']:.4f})")

print("\nModel 2: DiD + Covariates (staffing, visits)")
print(f"DID coefficient: {model2.params['did']:.3f} (SE: {model2.bse['did']:.3f}, p={model2.pvalues['did']:.4f})")
print(f"Staffing coefficient: {model2.params['staffing_index']:.3f} (SE: {model2.bse['staffing_index']:.3f}, p={model2.pvalues['staffing_index']:.4f})")
print(f"Visits coefficient: {model2.params['visits_index']:.3f} (SE: {model2.bse['visits_index']:.3f}, p={model2.pvalues['visits_index']:.4f})")

print("\nModel 3: DiD with Week FE (branch FE)")
print(f"DID coefficient: {model3.params['did']:.3f} (SE: {model3.bse['did']:.3f}, p={model3.pvalues['did']:.4f})")

# Check if covariates are balanced between groups
print("\n=== Covariate Balance Check ===")
cov_balance = df.groupby('switched_branch')[['staffing_index', 'visits_index']].mean()
print(cov_balance.round(3))
