import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create a treatment interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# First, let's see the 2x2 table for DiD
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES SUMMARY")
print("=" * 60)

# Calculate means for each group
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nControl Group (no switch):")
print(f"  Before (weeks 6-14): {before_control:.2f}")
print(f"  After (weeks 15-22): {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")

print(f"\nTreatment Group (switched):")
print(f"  Before (weeks 6-14): {before_treatment:.2f}")
print(f"  After (weeks 15-22): {after_treatment:.2f}")
print(f"  Change: {after_treatment - before_treatment:.2f}")

# DiD estimate
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

# Count branches in each group
print(f"\nNumber of branches:")
print(f"  Control (no switch): {(df['switched_branch'] == 0).sum()}")
print(f"  Treatment (switched): {(df['switched_branch'] == 1).sum()}")

# Now run regression with branch fixed effects
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Model 1: Simple DiD
model1 = ols('service_score ~ treatment + C(switched_branch) + C(after_switch)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 1: Simple DiD (with group and time fixed effects)")
print(model1.summary().tables[1])

# Model 2: DiD with branch fixed effects
model2 = ols('service_score ~ treatment + C(after_switch) + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 2: DiD with Branch Fixed Effects")
print(model2.summary().tables[1])

# Model 3: DiD with controls
model3 = ols('service_score ~ treatment + C(after_switch) + C(branch_id) + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 3: DiD with Branch FE + Controls")

# 1. Event study with proper variable names
print("\n1. EVENT STUDY - EFFECT BY WEEK POST-SWITCH")
print("-" * 40)

df['week_relative'] = df['week'] - 15  # week 15 = 0 (first post-switch week)

# Create week dummies for treatment group
event_dummies = []
for w in range(-9, 8):  # weeks -9 to 7 relative to switch
    col_name = f'treat_week{w:+d}'
    df[col_name] = ((df['week_relative'] == w) & (df['switched_branch'] == 1)).astype(int)
    event_dummies.append(col_name)

# Run event study - exclude week -1 as reference
ref_week = 'treat_week-1'
event_cols = [c for c in event_dummies if c != ref_week]
formula = 'service_score ~ ' + ' + '.join(event_cols) + ' + C(branch_id) + C(week)'
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nEvent study coefficients (relative to week -1 baseline):")
print("(Negative weeks = pre-switch, Positive weeks = post-switch)")
print()
for w in range(-9, 8):
    col = f'treat_week{w:+d}'
    if col == ref_week:
        print(f"Week {w}: (reference)")
    elif col in event_model.params.index:
        coef = event_model.params[col]
        pval = event_model.pvalues[col]
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        marker = "POST" if w >= 0 else "PRE"
        print(f"Week {w:3d} [{marker}]: {coef:+.2f} (p={pval:.3f}) {sig}")

    elif col in event_model.params.index:

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

print("=" * 60)
print("REFINED ANALYSIS - EVENT STUDY")
print("=" * 60)

# Create week relative to switch (week 15 = 0)
df['week_rel'] = df['week'] - 15

# Create cleaner variable names for event study dummies
for w in range(-9, 8):
    # Use underscore prefix for negative numbers
    if w < 0:
        col_name = f'treat_w{abs(w)}b'  # 'b' for before
    elif w == 0:
        col_name = 'treat_w0'
    else:
        col_name = f'treat_w{w}a'  # 'a' for after
    
    df[col_name] = ((df['week_rel'] == w) & (df['switched_branch'] == 1)).astype(int)

# Map back to week numbers for reporting
col_to_week = {}
for w in range(-9, 8):
    if w < 0:
        col_to_week[f'treat_w{abs(w)}b'] = w
    elif w == 0:
        col_to_week['treat_w0'] = w
    else:
        col_to_week[f'treat_w{w}a'] = w

# Run event study - exclude week -1 as reference
ref_col = 'treat_w1b'  # week -1
event_cols = [c for c in df.columns if c.startswith('treat_w') and c != ref_col]

# Build formula carefully
formula_parts = event_cols + ['C(branch_id)']
formula = 'service_score ~ ' + ' + '.join(formula_parts)

event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nEvent study coefficients (relative to week -1 baseline):")
print("(Negative = pre-switch, Zero/Positive = post-switch)")
print()

# Collect results
event_results = []
for col, week in col_to_week.items():
    if col == ref_col:
        event_results.append({'week': week, 'coef': 0, 'se': 0, 'pval': 1, 'pre': week < 0})
        print(f"Week {week:3d}: (reference)")
    elif col in event_model.params.index:
        coef = event_model.params[col]
        se = event_model.bse[col]
        pval = event_model.pvalues[col]
        event_results.append({'week': week, 'coef': coef, 'se': se, 'pval': pval, 'pre': week < 0})
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        period = "PRE" if week < 0 else "POST"
        print(f"Week {week:3d} [{period}]: {coef:+.2f} (SE={se:.2f}, p={pval:.3f}) {sig}")

# Plot event study
event_df = pd.DataFrame(event_results)
event_df = event_df.sort_values('week')

fig, ax = plt.subplots(figsize=(10, 5))
ax.errorbar(event_df['week'], event_df['coef'], yerr=1.96*event_df['se'], 
            fmt='o', capsize=4, color='blue')
ax.axhline(y=0, color='gray', linestyle='--')
ax.axvline(x=-0.5, color='red', linestyle='-', linewidth=2, label='Switch')
ax.set_xlabel('Weeks Relative to Switch (0 = first week after switch)')
ax.set_ylabel('Service Score Difference vs Control')
ax.set_title('Event Study: Treatment Effect by Week')
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-9, 8))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)

# Summarize pre vs post
print("\n" + "=" * 60)
print("PRE VS POST SUMMARY")
print("=" * 60)

pre_event = event_df[event_df['pre'] == True]
post_event = event_df[event_df['pre'] == False]

print(f"\nPre-switch effects (should cluster around 0):")
print(f"  Mean: {pre_event['coef'].mean():.2f}")
print(f"  Range: [{pre_event['coef'].min():.2f}, {pre_event['coef'].max():.2f}]")
print(f"  Any significant? {(pre_event['pval'] < 0.05).sum()} of {len(pre_event)}")

print(f"\nPost-switch effects:")
print(f"  Mean: {post_event['coef'].mean():.2f}")
print(f"  Range: [{post_event['coef'].min():.2f}, {post_event['coef'].max():.2f}]")
print(f"  Any significant? {(post_event['pval'] < 0.05).sum()} of {len(post_event)}")

# Final DiD estimate
print("\n" + "=" * 60)
print("FINAL ESTIMATE")
print("=" * 60)

# Main DiD model
model = ols('service_score ~ treatment + C(after_switch) + C(branch_id)', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print(f"\nTreatment effect: {model.params['treatment']:.2f}")
print(f"95% CI: [{model.conf_int().loc['treatment'][0]:.2f}, {model.conf_int().loc['treatment'][1]:.2f}]")
print(f"P-value: {model.pvalues['treatment']:.4f}")

# Effect size
baseline = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
print(f"\nBaseline score: {baseline:.2f}")
print(f"Effect as % of baseline: {(model.params['treatment'] / baseline) * 100:.1f}%")
