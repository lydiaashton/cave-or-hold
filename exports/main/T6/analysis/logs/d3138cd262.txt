import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD table: average service scores by group and period
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES SUMMARY TABLE")
print("=" * 60)

# Before period (after_switch = 0)
before_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()

# After period (after_switch = 1)
after_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\n{'Group':<20} {'Before (wk 6-14)':<20} {'After (wk 15-22)':<20} {'Change':<15}")
print("-" * 75)
print(f"{'Switched (Treatment)':<20} {before_switched:>18.2f} {after_switched:>20.2f} {after_switched - before_switched:>14.2f}")
print(f"{'Non-Switched (Control)':<20} {before_control:>18.2f} {after_control:>20.2f} {after_control - before_control:>14.2f}")
print("-" * 75)
did_estimate = (after_switched - before_switched) - (after_control - before_control)
print(f"{'Difference-in-Differences:':<60} {did_estimate:>14.2f}")

# Now run regression with standard errors
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Clustered standard errors by branch
print("\n" + "=" * 60)
print("CLUSTERED STANDARD ERRORS (by branch)")
print("=" * 60)

# Use cluster-robust standard errors
model1_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators (week 14 is the last pre-switch week)
# Let's use week 14 as the reference (last week before switch)
df['relative_week'] = df['week'] - 14

# Create event study dummies for each week relative to switch
# We'll use week -8 (week 6) as baseline
event_study_weeks = list(range(-8, 9))  # -8 to 8 (week 6 to 22, relative to week 14)

# Create dummy variables for each relative week (excluding baseline week -8)
for w in event_study_weeks:
    if w != -8:  # baseline
        df[f'week_{w}'] = (df['relative_week'] == w).astype(int)

# Also create treatment interacted with each week
for w in event_study_weeks:
    if w != -8:  # baseline
        df[f'treatment_week_{w}'] = df['switched_branch'] * df[f'week_{w}']

# Build event study regression
# Baseline is week -8 (relative week), so we exclude it
treatment_cols = [f'treatment_week_{w}' for w in event_study_weeks if w != -8]
control_cols = [f'week_{w}' for w in event_study_weeks if w != -8]

# Create formula
formula = 'service_score ~ switched_branch + ' + ' + '.join(treatment_cols + control_cols)
print("Event Study Model Formula (partial):")
print(formula[:200] + "...")

# Run event study regression with clustered SE
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract treatment coefficients
treatment_coefs = []
treatment_se = []
treatment_weeks = []
for w in event_study_weeks:
    if w == -8:
        treatment_coefs.append(0)  # baseline
        treatment_se.append(0)
        treatment_weeks.append(w)
    else:
        coef_name = f'treatment_week_{w}'
        if coef_name in event_model.params:
            treatment_coefs.append(event_model.params[coef_name])
            treatment_se.append(event_model.bse[coef_name])
            treatment_weeks.append(w)

print("\n" + "=" * 70)
print("EVENT STUDY RESULTS")
print("=" * 70)
print(f"{'Relative Week':>15} {'Actual Week':>15} {'Coefficient':>15} {'Std Error':>15}")
print("-" * 70)
for i, w in enumerate(treatment_weeks):
    actual_week = w + 14
    coef = treatment_coefs[i]
    se = treatment_se[i]
    print(f"{w:>15} {actual_week:>15} {coef:>15.3f} {se:>15.3f}")

# Plot event study
plt.figure(figsize=(12, 6))
plt.errorbar(treatment_weeks, treatment_coefs, yerr=[1.96*s for s in treatment_se], 
             fmt='o', capsize=5, markersize=8, linewidth=2, color='blue')
plt.axhline(y=0, color='black', linestyle='-', linewidth=1)
plt.axvline(x=0.5, color='red', linestyle='--', linewidth=2, label='Switch Point')
plt.xlabel('Relative Week (0 = Week 14, last pre-switch week)', fontsize=12)
plt.ylabel('Treatment Effect (vs. Week 6 baseline)', fontsize=12)
plt.title('Event Study: Treatment Effect by Week', fontsize=14, fontweight='bold')
plt.grid(True, alpha=0.3)
plt.xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study chart saved as 'event_study.png'")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators (week 14 is the last pre-switch week)
df['relative_week'] = df['week'] - 14

# Create event study dummies with safe names (prefix 'w' to avoid issues with negative numbers)
event_study_weeks = list(range(-8, 9))  # -8 to 8 (week 6 to 22, relative to week 14)

# Create dummy variables for each relative week (excluding baseline week -8)
for w in event_study_weeks:
    if w != -8:  # baseline
        df[f'w{w}'] = (df['relative_week'] == w).astype(int)

# Also create treatment interacted with each week
for w in event_study_weeks:
    if w != -8:  # baseline
        df[f'tw{w}'] = df['switched_branch'] * df[f'w{w}']

# Build event study regression
treatment_cols = [f'tw{w}' for w in event_study_weeks if w != -8]
control_cols = [f'w{w}' for w in event_study_weeks if w != -8]

# Create formula
formula = 'service_score ~ switched_branch + ' + ' + '.join(treatment_cols + control_cols)

# Run event study regression with clustered SE
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract treatment coefficients
treatment_coefs = []
treatment_se = []
treatment_weeks = []
for w in event_study_weeks:
    if w == -8:
        treatment_coefs.append(0)  # baseline
        treatment_se.append(0)
        treatment_weeks.append(w)
    else:
        coef_name = f'tw{w}'
        if coef_name in event_model.params:
            treatment_coefs.append(event_model.params[coef_name])
            treatment_se.append(event_model.bse[coef_name])
            treatment_weeks.append(w)

print("=" * 70)
print("EVENT STUDY RESULTS")
print("=" * 70)
print(f"{'Relative Week':>15} {'Actual Week':>15} {'Coefficient':>15} {'Std Error':>15}")
print("-" * 70)
for i, w in enumerate(treatment_weeks):
    actual_week = w + 14
    coef = treatment_coefs[i]
    se = treatment_se[i]
    print(f"{w:>15} {actual_week:>15} {coef:>15.3f} {se:>15.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators (week 14 is the last pre-switch week)
df['relative_week'] = df['week'] - 14

# Create event study dummies with safe names (use 'rw' prefix for relative week)
event_study_weeks = list(range(-8, 9))  # -8 to 8 (week 6 to 22, relative to week 14)

# Create dummy variables for each relative week (excluding baseline week -8)
for w in event_study_weeks:
    if w != -8:  # baseline
        # Use 'n' prefix for negative, 'p' for positive, 'z' for zero
        if w < 0:
            col_name = f'rw_n{abs(w)}'
        elif w == 0:
            col_name = 'rw_z0'
        else:
            col_name = f'rw_p{w}'
        df[col_name] = (df['relative_week'] == w).astype(int)

# Also create treatment interacted with each week
for w in event_study_weeks:
    if w != -8:  # baseline
        if w < 0:
            col_name = f'rw_n{abs(w)}'
            tw_col = f'tw_n{abs(w)}'
        elif w == 0:
            col_name = 'rw_z0'
            tw_col = 'tw_z0'
        else:
            col_name = f'rw_p{w}'
            tw_col = f'tw_p{w}'
        df[tw_col] = df['switched_branch'] * df[col_name]

# Build event study regression
treatment_cols = []
control_cols = []
for w in event_study_weeks:
    if w != -8:  # baseline
        if w < 0:
            treatment_cols.append(f'tw_n{abs(w)}')
            control_cols.append(f'rw_n{abs(w)}')
        elif w == 0:
            treatment_cols.append('tw_z0')
            control_cols.append('rw_z0')
        else:
            treatment_cols.append(f'tw_p{w}')
            control_cols.append(f'rw_p{w}')

# Create formula
formula = 'service_score ~ switched_branch + ' + ' + '.join(treatment_cols + control_cols)
print("Formula length:", len(formula))
print("Formula (first 300 chars):", formula[:300])

# Run event study regression with clustered SE
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract treatment coefficients
treatment_coefs = []
treatment_se = []
treatment_weeks = []
for w in event_study_weeks:
    if w == -8:
        treatment_coefs.append(0)  # baseline
        treatment_se.append(0)
        treatment_weeks.append(w)
    else:
        if w < 0:
            coef_name = f'tw_n{abs(w)}'
        elif w == 0:
            coef_name = 'tw_z0'
        else:
            coef_name = f'tw_p{w}'
        if coef_name in event_model.params:
            treatment_coefs.append(event_model.params[coef_name])
            treatment_se.append(event_model.bse[coef_name])
            treatment_weeks.append(w)

print("\n" + "=" * 70)
print("EVENT STUDY RESULTS")
print("=" * 70)
print(f"{'Relative Week':>15} {'Actual Week':>15} {'Coefficient':>15} {'Std Error':>15}")
print("-" * 70)
for i, w in enumerate(treatment_weeks):
    actual_week = w + 14
    coef = treatment_coefs[i]
    se = treatment_se[i]
    print(f"{w:>15} {actual_week:>15} {coef:>15.3f} {se:>15.3f}")

# Plot event study
plt.figure(figsize=(12, 6))
plt.errorbar(treatment_weeks, treatment_coefs, yerr=[1.96*s for s in treatment_se],

import pandas as pd

# Final summary statistics
print("=" * 70)
print("FINAL ANALYSIS SUMMARY")
print("=" * 70)

# Basic DiD estimate with clustered SE
from statsmodels.formula.api import ols
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
did_coef = model.params['treatment']
did_se = model.bse['treatment']
did_pval = model.pvalues['treatment']
did_ci_low = did_coef - 1.96 * did_se
did_ci_high = did_coef + 1.96 * did_se

print(f"\nMain DiD Estimate: +{did_coef:.2f} points")
print(f"95% Confidence Interval: [{did_ci_low:.2f}, {did_ci_high:.2f}]")
print(f"P-value: {did_pval:.4f}")

# Effect size
baseline = df[df['week'] <= 14]['service_score'].mean()
pct_improvement = (did_coef / baseline) * 100
print(f"\nBaseline service score (pre-switch average): {baseline:.2f}")
print(f"Percentage improvement: {pct_improvement:.1f}%")

# Branch counts
n_switched = df['branch_id'][df['switched_branch'] == 1].nunique()
n_control = df['branch_id'][df['switched_branch'] == 0].nunique()
print(f"\nBranches that switched: {n_switched} ({n_switched/(n_switched+n_control)*100:.1f}%)")
print(f"Branches that did not switch: {n_control} ({n_control/(n_switched+n_control)*100:.1f}%)")

# Post-switch average for switched branches
post_switch_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
print(f"\nAverage service score for switched branches after switch: {post_switch_switched:.2f}")

# Control variables effect
model_with_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print(f"  Visits coefficient: {model_with_controls.params['visits_index']:.3f} (p={model_with_controls.pvalues['visits_index']:.4f})")

# Note about pre-trends
print("\n" + "=" * 70)
print("IMPORTANT CAVEAT: PRE-EXISTING TRENDS")

print("improving relative to non-switched branches BEFORE the procedure change.")

import pandas as pd

df = pd.read_csv('branches.csv')

# Create the DiD interaction term

# Final summary statistics
print("=" * 70)
print("FINAL ANALYSIS SUMMARY")
print("=" * 70)

# Basic DiD estimate with clustered SE
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
did_coef = model.params['treatment']
did_se = model.bse['treatment']
did_pval = model.pvalues['treatment']
did_ci_low = did_coef - 1.96 * did_se
did_ci_high = did_coef + 1.96 * did_se

print(f"\nMain DiD Estimate: +{did_coef:.2f} points")
print(f"95% Confidence Interval: [{did_ci_low:.2f}, {did_ci_high:.2f}]")
print(f"P-value: {did_pval:.4f}")

# Effect size
baseline = df[df['week'] <= 14]['service_score'].mean()
pct_improvement = (did_coef / baseline) * 100
print(f"\nBaseline service score (pre-switch average): {baseline:.2f}")
print(f"Percentage improvement: {pct_improvement:.1f}%")

# Branch counts
n_switched = df['branch_id'][df['switched_branch'] == 1].nunique()
n_control = df['branch_id'][df['switched_branch'] == 0].nunique()
print(f"\nBranches that switched: {n_switched} ({n_switched/(n_switched+n_control)*100:.1f}%)")
print(f"Branches that did not switch: {n_control} ({n_control/(n_switched+n_control)*100:.1f}%)")

post_switch_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
print(f"\nAverage service score for switched branches after switch: {post_switch_switched:.2f}")

# Test if switched branches had different slopes before the switch

print("CRITICAL ANALYSIS: PRE-TREND EXAMINATION")

print(f"  P-value: {pre_trend.pvalues['switched_branch:week_centered']:.4f}")

# Calculate expected vs actual post-switch difference
