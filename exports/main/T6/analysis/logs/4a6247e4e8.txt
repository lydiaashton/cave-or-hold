import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Run DiD regression with branch fixed effects
# First, let's do a simple DiD regression
model1 = ols('service_score ~ treatment + post + did', data=df).fit()
print("=== Simple DiD Regression ===")
print(model1.summary().tables[1])

# Now with branch fixed effects (using dummy variables)
print("\n=== DiD with Branch Fixed Effects ===")
df_with_dummies = pd.get_dummies(df, columns=['branch_id'], drop_first=True)
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_id_')]

X = df_with_dummies[['treatment', 'post', 'did'] + branch_cols]
X = sm.add_constant(X)
y = df_with_dummies['service_score']

model2 = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df_with_dummies['branch_id']})
print(model2.summary().tables[1].iloc[:5])  # Just show first few rows including our key vars

print(f"\nDiD coefficient: {model2.params['did']:.3f}")
print(f"DiD p-value: {model2.pvalues['did']:.4f}")
print(f"DiD 95% CI: [{model2.conf_int().loc['did', 0]:.3f}, {model2.conf_int().loc['did', 1]:.3f}]")

model_controls = ols('service_score ~ post + did + staffing_index + visits_index + C(branch_id)',

df['post'] = df['after_switch']

# Robustness: Check effect in immediate post period vs later
