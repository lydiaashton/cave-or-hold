from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create relative week indicators (event study)
# Week 14 is the last pre-period week, so we'll use it as reference
df['relative_week'] = df['week'] - 14  # So week 14 = 0, week 15 = 1, etc.

# Create interaction terms for each week with treatment
# We'll omit week 0 (week 14) as the reference period
weeks = sorted(df['week'].unique())
pre_weeks = [w for w in weeks if w < 14]  # weeks 6-13
post_weeks = [w for w in weeks if w >= 15]  # weeks 15-22

# Create dummy variables for each week (excluding week 14 as reference)
for w in weeks:
    if w != 14:
        df[f'week_{w}'] = (df['week'] == w).astype(int)
        df[f'week_{w}_treated'] = df[f'week_{w}'] * df['switched_branch']

# Build formula for event study
pre_terms = ' + '.join([f'week_{w}_treated' for w in pre_weeks])
post_terms = ' + '.join([f'week_{w}_treated' for w in post_weeks])
week_controls = ' + '.join([f'week_{w}' for w in weeks if w != 14])

formula = f'service_score ~ switched_branch + {week_controls} + {pre_terms} + {post_terms}'

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Recreate event study model
weeks = sorted(df['week'].unique())
pre_weeks = [w for w in weeks if w < 14]
post_weeks = [w for w in weeks if w >= 15]

for w in weeks:
    if w != 14:
        df[f'week_{w}'] = (df['week'] == w).astype(int)
        df[f'week_{w}_treated'] = df[f'week_{w}'] * df['switched_branch']

week_controls = ' + '.join([f'week_{w}' for w in weeks if w != 14])
pre_terms = ' + '.join([f'week_{w}_treated' for w in pre_weeks])
post_terms = ' + '.join([f'week_{w}_treated' for w in post_weeks])
formula = f'service_score ~ switched_branch + {week_controls} + {pre_terms} + {post_terms}'

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients and confidence intervals
results = []
for w in sorted(pre_weeks + [14] + post_weeks):
    if w == 14:
        results.append({'week': w, 'coef': 0, 'se': 0, 'lower': 0, 'upper': 0, 'period': 'reference'})
    else:
        coef = model_event.params[f'week_{w}_treated']
        se = model_event.bse[f'week_{w}_treated']
        results.append({
            'week': w, 
            'coef': coef, 
            'se': se,
            'lower': coef - 1.96 * se,
            'upper': coef + 1.96 * se,
            'period': 'pre' if w < 14 else 'post'
        })

results_df = pd.DataFrame(results)

fig, ax = plt.subplots(figsize=(12, 6))
