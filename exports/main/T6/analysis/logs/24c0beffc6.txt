import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks relative to switch at week 15)
df['rel_week'] = df['week'] - 15

# Event study: create dummies for each week relative to switch
# Use week -1 (week 14) as reference
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    if w != -1:  # skip reference week
        df[f'week_{w}'] = ((df['rel_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
event_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks relative to switch at week 15)
df['rel_week'] = df['week'] - 15

# Event study: create dummies for each week relative to switch with proper names
# Use week -1 (week 14) as reference
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    if w != -1:  # skip reference week
        var_name = f'w{abs(w)}_{"neg" if w < 0 else "pos"}' if w != 0 else 'w0'
        df[var_name] = ((df['rel_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Build formula
event_vars = []
var_map = {}
for w in range(-9, 8):
    if w == -1:
        continue  # reference
    elif w == 0:
        var_name = 'w0'
    else:
        var_name = f'w{abs(w)}_{"neg" if w < 0 else "pos"}'
    event_vars.append(var_name)
    var_map[var_name] = w

formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Results (coefficients for treatment group by week relative to switch):")
print("(Reference: week -1, i.e., week 14)")
print("\n" + "="*80)

event_results = []
for var_name, w in sorted(var_map.items(), key=lambda x: x[1]):
    coef = event_model.params[var_name]
    se = event_model.bse[var_name]
    pval = event_model.pvalues[var_name]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"Week {w:3d}: {coef:7.3f} (SE: {se:.3f}, p={pval:.3f}) {sig}")
    event_results.append({
        'rel_week': w,
        'coef': coef,
        'se': se,
        'ci_low': coef - 1.96*se,
        'ci_high': coef + 1.96*se
    })

# Add reference week
event_results.append({'rel_week': -1, 'coef': 0, 'se': 0, 'ci_low': 0, 'ci_high': 0})
event_results.sort(key=lambda x: x['rel_week'])
event_df = pd.DataFrame(event_results)

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
colors = ['gray' if w < 0 else 'green' for w in event_df['rel_week']]
ax.bar(event_df['rel_week'], event_df['coef'],

ax.set_xlabel('Weeks Relative to Switch', fontsize=12)
ax.set_ylabel('Effect on Service Score', fontsize=12)
ax.set_title('Event Study: Treatment Effect by Week\n(Reference: Week -1)', fontsize=14, fontweight='bold')
ax.legend()
ax.grid(True, alpha=0.3, axis='y')
ax.set_xticks(range(-9, 8))
plt.tight_layout()

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Look at event study more carefully - compare pre vs post pattern
print("Event Study Coefficients (treatment effect by week relative to switch)")
print("Reference week: -1 (week 14)")
print("="*70)

# Recreate event study with proper variable names
df['rel_week'] = df['week'] - 15

event_vars = []
var_map = {}
for w in range(-9, 8):
    if w == -1:
        continue
    elif w == 0:
        var_name = 'w0'
    else:
        var_name = f'w{abs(w)}_{"neg" if w < 0 else "pos"}'
    event_vars.append(var_name)
    var_map[var_name] = w
    df[var_name] = ((df['rel_week'] == w) & (df['switched_branch'] == 1)).astype(int)

formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

pre_coefs = []
post_coefs = []

for var_name, w in sorted(var_map.items(), key=lambda x: x[1]):
    coef = event_model.params[var_name]
    se = event_model.bse[var_name]
    pval = event_model.pvalues[var_name]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    
    if w < 0:
        pre_coefs.append(coef)
    else:
        post_coefs.append(coef)
    
    bar = "█" * int(abs(coef) * 3)
    direction = "▸" if coef > 0 else "◂"
    print(f"Week {w:3d}: {direction} {coef:6.2f} (SE: {se:.2f}, p={pval:.3f}) {sig} {bar}")

print("\n" + "="*70)
print(f"Pre-period coefficients (weeks -9 to -2): mean={np.mean(pre_coefs):.2f}, range=[{min(pre_coefs):.2f}, {max(pre_coefs):.2f}]")
print(f"Post-period coefficients (weeks 0 to 7): mean={np.mean(post_coefs):.2f}, range=[{min(post_coefs):.2f}, {max(post_coefs):.2f}]")

# Test if post-period effects are jointly different from pre-period
print("\n\nComparing average post vs pre effect:")
print(f"Average post-period effect: {np.mean(post_coefs):.2f}")
print(f"Average pre-period effect: {np.mean(pre_coefs):.2f}")
print(f"Difference: {np.mean(post_coefs) - np.mean(pre_coefs):.2f}")
