import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary table: mean service scores by group and period
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES SUMMARY")
print("=" * 60)

# Before period (weeks 6-14)
before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

print("\nMean Service Scores by Group and Period:")
print("-" * 50)

before_treatment = before[before['switched_branch'] == 1]['service_score'].mean()
before_control = before[before['switched_branch'] == 0]['service_score'].mean()
after_treatment = after[after['switched_branch'] == 1]['service_score'].mean()
after_control = after[after['switched_branch'] == 0]['service_score'].mean()

print(f"Before (weeks 6-14):")
print(f"  Treatment (switched) branches: {before_treatment:.2f}")
print(f"  Control (non-switched) branches: {before_control:.2f}")
print(f"  Difference: {before_treatment - before_control:.2f}")

print(f"\nAfter (weeks 15-22):")
print(f"  Treatment (switched) branches: {after_treatment:.2f}")
print(f"  Control (non-switched) branches: {after_control:.2f}")
print(f"  Difference: {after_treatment - after_control:.2f}")

# DiD estimate
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

# Now run regression
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary())

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary())

# Clustered standard errors by branch
print("\n" + "=" * 60)
print("CLUSTERED STANDARD ERRORS (by branch)")
print("=" * 60)

from statsmodels.stats.sandwich_covariance import cov_cluster

# Refit with clustered SE
model1_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
cov_clustered = cov_cluster(model1_clustered, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))
print("\nModel 1 with Clustered SE:")
print(f"Intercept: {model1_clustered.params['Intercept']:.3f} (SE: {se_clustered[0]:.3f})")
print(f"switched_branch: {model1_clustered.params['switched_branch']:.3f} (SE: {se_clustered[1]:.3f})")
print(f"after_switch: {model1_clustered.params['after_switch']:.3f} (SE: {se_clustered[2]:.3f})")
print(f"treatment (DiD effect): {model1_clustered.params['treatment']:.3f} (SE: {se_clustered[3]:.3f})")

# Calculate t-stat and p-value for treatment effect
treatment_coef = model1_clustered.params['treatment']
treatment_se = se_clustered[3]
t_stat = treatment_coef / treatment_se
from scipy import stats
p_value = 2 * (1 - stats.t.cdf(abs(t_stat), df=len(df) - 4))
print(f"t-statistic: {t_stat:.3f}, p-value: {p_value:.4f}")
