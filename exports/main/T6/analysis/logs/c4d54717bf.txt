import matplotlib.pyplot as plt

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimator - the effect of the treatment

model = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

# Load the data
df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run the main model with clustered SE
model = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get the treatment effect and confidence interval
treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pvalue = model.pvalues['treatment']

# 95% CI
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se

print("=== Treatment Effect Summary ===")
print(f"Estimated effect: {treatment_coef:.2f} points")
print(f"Standard error: {treatment_se:.2f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {treatment_pvalue:.4f}")
print(f"Statistically significant: {'Yes' if treatment_pvalue < 0.05 else 'No'}")

# Check pre-trends: interact switched_branch with week dummies for pre-period only
pre_df = df[df['week'] < 15].copy()
pre_df['treatment'] = pre_df['switched_branch'] * pre_df['after_switch']  # Should be all 0 in pre-period

# Test if there's a trend difference before the switch
pre_model = smf.ols('service_score ~ switched_branch + week + switched_branch:week', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print("\n=== Pre-Trend Test (Weeks 6-14) ===")
print(pre_model.summary())

# Effect size relative to baseline
baseline_score = df['service_score'].mean()
print(f"\n=== Effect Size ===")
print(f"Baseline average service score: {baseline_score:.2f}")
print(f"Treatment effect as % of baseline: {(treatment_coef / baseline_score) * 100:.1f}%")
