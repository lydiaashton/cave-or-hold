import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Run the basic DiD regression
print("=== Difference-in-Differences Regression ===")
model = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model.summary())

# Run DiD with controls (staffing and visits)
print("\n=== DiD Regression with Controls ===")
model_controls = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_controls.summary())

# Check parallel trends - event study approach
print("\n=== Event Study / Parallel Trends Check ===")
# Create week dummies interacted with treatment
df['week_cat'] = df['week'].astype('category')
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=False)

# For event study, we'll use week 14 as the reference (last pre-treatment week)
pre_weeks = [6, 7, 8, 9, 10, 11, 12, 13]
post_weeks = [15, 16, 17, 18, 19, 20, 21, 22]

# Create interaction terms for each week with treatment
for w in range(6, 23):
    if w != 14:  # Reference week
        df[f'week{w}_treated'] = ((df['week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (excluding week 14 as reference)
event_study_cols = ['switched_branch', 'after_switch'] + [f'week{w}_treated' for w in range(6, 23) if w != 14]
event_study_formula = 'service_score ~ ' + ' + '.join(event_study_cols)
model_event = ols(event_study_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the week-treatment interactions
print("\nWeek-by-week treatment effects (relative to week 14):")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Proper event study with week fixed effects
# Create week dummies (using week 14 as reference)
for w in range(6, 23):
    df[f'week_{w}'] = (df['week'] == w).astype(int)

# Create treatment-week interactions (for all weeks except reference week 14)
for w in range(6, 23):
    if w != 14:
        df[f'treated_week{w}'] = ((df['week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Build the formula - include week dummies (excluding week 14) and treatment-week interactions
week_cols = [f'week_{w}' for w in range(6, 23) if w != 14]
interaction_cols = [f'treated_week{w}' for w in range(6, 23) if w != 14]

# Formula: service_score ~ switched_branch + week_dummies + treatment_week_interactions
# Note: after_switch is collinear with week dummies, so we exclude it
formula = 'service_score ~ switched_branch + ' + ' + '.join(week_cols + interaction_cols)

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study with Week Fixed Effects ===")
print("\nPre-treatment coefficients (should be ~0 if parallel trends hold):")
pre_weeks = [6, 7, 8, 9, 10, 11, 12, 13]
for w in pre_weeks:
    coef = model_event.params.get(f'treated_week{w}', None)
    se = model_event.bse.get(f'treated_week{w}', None)
    pval = model_event.pvalues.get(f'treated_week{w}', None)
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    ci_low = coef - 1.96*se
    ci_high = coef + 1.96*se
    print(f"Week {w}: coef={coef:.2f}, 95% CI=[{ci_low:.2f}, {ci_high:.2f}], p={pval:.3f} {sig}")

print("\nPost-treatment coefficients (treatment effect):")
post_weeks = [15, 16, 17, 18, 19, 20, 21, 22]
for w in post_weeks:
    coef = model_event.params.get(f'treated_week{w}', None)
    se = model_event.bse.get(f'treated_week{w}', None)
    pval = model_event.pvalues.get(f'treated_week{w}', None)
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    ci_low = coef - 1.96*se
    ci_high = coef + 1.96*se
    print(f"Week {w}: coef={coef:.2f}, 95% CI=[{ci_low:.2f}, {ci_high:.2f}], p={pval:.3f} {sig}")

# Calculate average post-treatment effect
print("\n=== Average Post-Treatment Effect ===")
post_coefs = [model_event.params[f'treated_week{w}'] for w in post_weeks]
avg_effect = np.mean(post_coefs)
# Simple average of standard errors (not quite right but gives a sense)
avg_se = np.mean([model_event.bse[f'treated_week{w}'] for w in post_weeks])
print(f"Average post-treatment effect: {avg_effect:.2f} (approx. SE: {avg_se:.2f})")
