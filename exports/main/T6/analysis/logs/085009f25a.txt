import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from scipy import stats
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate mean service scores by group and period
print("=== Mean Service Scores by Group and Period ===\n")

# Pre-period (after_switch = 0)
pre_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()

# Post-period (after_switch = 1)
post_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"Treatment Group (switched branches):")
print(f"  Pre-period mean: {pre_treatment:.2f}")
print(f"  Post-period mean: {post_treatment:.2f}")
print(f"  Change: {post_treatment - pre_treatment:.2f}")
print()
print(f"Control Group (non-switched branches):")
print(f"  Pre-period mean: {pre_control:.2f}")
print(f"  Post-period mean: {post_control:.2f}")
print(f"  Change: {post_control - pre_control:.2f}")
print()

# Difference-in-Differences estimate
did_estimate = (post_treatment - pre_treatment) - (post_control - pre_control)
print(f"DiD Estimate (treatment effect): {did_estimate:.2f}")
print()

# Run DiD regression
print("=== DiD Regression Results ===\n")
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create the treatment indicator
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate weekly means by group
weekly_means = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_means['group'] = weekly_means['switched_branch'].map({0: 'Control (No Switch)', 1: 'Treatment (Switched)'})

# Create the plot
fig, ax = plt.subplots(figsize=(12, 6))

# Plot lines for each group
for group_name, group_data in weekly_means.groupby('group'):
    marker = 'o' if 'Treatment' in group_name else 's'
    ax.plot(group_data['week'], group_data['service_score'], marker=marker, 
            linewidth=2.5, markersize=8, label=group_name)

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch implemented')

# Add shaded region for post-period
ax.axvspan(14.5, 22.5, alpha=0.15, color='green', label='Post-switch period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Mean Service Score', fontsize=12)
ax.set_title('Service Scores Over Time: Treatment vs Control Groups', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=10)
ax.set_xticks(range(6, 23))
ax.grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("Plot saved as 'service_scores_trend.png'")

# Also calculate confidence intervals for the DiD estimate
print("\n=== Effect Size and Confidence Interval ===\n")

# Bootstrap for confidence interval
np.random.seed(42)
n_bootstrap = 1000
did_estimates = []

for i in range(n_bootstrap):
    # Sample branches with replacement
    sampled_branches = df['branch_id'].unique()
    boot_branches = np.random.choice(sampled_branches, size=len(sampled_branches), replace=True)
    boot_df = df[df['branch_id'].isin(boot_branches)]
    
    # Calculate DiD
    pre_treatment = boot_df[(boot_df['switched_branch'] == 1) & (boot_df['after_switch'] == 0)]['service_score'].mean()
    pre_control = boot_df[(boot_df['switched_branch'] == 0) & (boot_df['after_switch'] == 0)]['service_score'].mean()
    post_treatment = boot_df[(boot_df['switched_branch'] == 1) & (boot_df['after_switch'] == 1)]['service_score'].mean()
    post_control = boot_df[(boot_df['switched_branch'] == 0) & (boot_df['after_switch'] == 1)]['service_score'].mean()
    
    did = (post_treatment - pre_treatment) - (post_control - pre_control)
    did_estimates.append(did)

ci_lower = np.percentile(did_estimates, 2.5)
ci_upper = np.percentile(did_estimates, 97.5)
print(f"Bootstrap 95% CI for DiD estimate: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"Point estimate: {np.mean(did_estimates):.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# DiD with covariates (staffing and visits)
print("=== DiD Regression with Covariates ===\n")
model_covariates = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
                       data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_covariates.summary())

# Check if treatment effect varies by branch characteristics
print("\n=== Summary Statistics by Group ===\n")
summary_stats = df.groupby('switched_branch')[['service_score', 'staffing_index', 'visits_index']].agg(['mean', 'std', 'count'])
print(summary_stats)

# Pre-post comparison within each group
print("\n=== Pre-Post Changes Within Groups ===\n")
for switched in [0, 1]:
    group = df[df['switched_branch'] == switched]
    pre = group[group['after_switch'] == 0]['service_score']
    post = group[group['after_switch'] == 1]['service_score']
    t_stat, p_val = stats.ttest_ind(pre, post)
    print(f"Switched={switched}: Pre mean={pre.mean():.2f}, Post mean={post.mean():.2f}, Change={post.mean()-pre.mean():.2f}, p-value={p_val:.3f}")

# Check for any time-varying effects (event study style)
print("\n=== Weekly Treatment Effects (Event Study) ===\n")
df['post_week'] = df['week'] - 14  # weeks since switch (negative = before)
df['post_week'] = df['post_week'].apply(lambda x: max(-8, min(8, x)))  # cap at -8 and 8

# Create week dummies interacted with treatment
for w in range(-8, 9):
    if w != -1:  # omit week -1 as reference
        df[f'week_{w}'] = ((df['post_week'] == w) & (df['switched_branch'] == 1)).astype(int)

week_vars = [f'week_{w}' for w in range(-8, 9) if w != -1]
event_study_formula = f'service_score ~ switched_branch + after_switch + {" + ".join(week_vars)}'
event_model = ols(event_study_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the week effects
print("Week-specific treatment effects (relative to week 14):")
for w in range(-8, 9):
    if w != -1:
        coef = event_model.params[f'week_{w}']
        se = event_model.bse[f'week_{w}']
        pval = event_model.pvalues[f'week_{w}']
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"Week {14+w}: {coef:6.2f} (SE: {se:.2f}) {sig}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from scipy import stats

# Load the data
df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Check for any time-varying effects (event study style)
print("=== Weekly Treatment Effects (Event Study) ===\n")
df['post_week'] = df['week'] - 14  # weeks since switch (negative = before)
df['post_week'] = df['post_week'].apply(lambda x: max(-8, min(8, x)))  # cap at -8 and 8

# Create week dummies interacted with treatment (using valid column names)
for w in range(-8, 9):
    if w != -1:  # omit week -1 as reference
        col_name = f'w{abs(w)}_{"m" if w < 0 else "p"}{w}' if w != 0 else 'w0'
        if w < 0:
            col_name = f'w{abs(w)}m'
        elif w > 0:
            col_name = f'w{w}p'
        else:
            col_name = 'w0'
        df[col_name] = ((df['post_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Get all week columns
week_cols = []
for w in range(-8, 9):
    if w != -1:
        if w < 0:
            week_cols.append(f'w{abs(w)}m')
        elif w > 0:
            week_cols.append(f'w{w}p')
        else:
            week_cols.append('w0')

event_study_formula = f'service_score ~ switched_branch + after_switch + {" + ".join(week_cols)}'
event_model = ols(event_study_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the week effects
print("Week-specific treatment effects (relative to week 14):")
week_labels = []
for w in range(-8, 9):
    if w != -1:
        if w < 0:
            col_name = f'w{abs(w)}m'
        elif w > 0:
            col_name = f'w{w}p'
        else:
            col_name = 'w0'
        coef = event_model.params[col_name]
        se = event_model.bse[col_name]
        pval = event_model.pvalues[col_name]
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        week_labels.append(f"Week {14+w}: {coef:6.2f} (SE: {se:.2f}) {sig}")
        
for label in week_labels:
    print(label)
