model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run regression with branch fixed effects
print("=== DiD with Branch Fixed Effects ===")
# Convert branch_id to categorical for fixed effects
df['branch_id_cat'] = df['branch_id'].astype('category')

# Use C() notation for categorical variables
model_fe = ols('service_score ~ C(branch_id_cat) + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_fe.summary())

# Extract the treatment effect
print(f"\n=== Key Result with Branch FE ===")
print(f"Treatment effect: {model_fe.params['treatment']:.3f}")
print(f"95% CI: [{model_fe.conf_int().loc['treatment'][0]:.3f}, {model_fe.conf_int().loc['treatment'][1]:.3f}]")
print(f"P-value: {model_fe.pvalues['treatment']:.4f}")

# Check robustness - effect by week after switch
print("\n=== Dynamic Effects (Week-by-Week After Switch) ===")
df['weeks_since_switch'] = df['week'] - 14
df['weeks_since_switch'] = df['weeks_since_switch'].apply(lambda x: max(0, x))

# Create interaction for each post-switch week
for week in range(15, 23):
    df[f'post_week_{week}'] = ((df['week'] == week) & (df['switched_branch'] == 1)).astype(int)

# Run regression with week-specific effects
week_vars = [f'post_week_{w}' for w in range(15, 23)]
formula = f'service_score ~ C(branch_id_cat) + {" + ".join(week_vars)} + staffing_index + visits_index'
model_dynamic = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nWeek-specific treatment effects:")
for week in range(15, 23):
    coef = model_dynamic.params[f'post_week_{week}']
    pval = model_dynamic.pvalues[f'post_week_{week}']
    ci = model_dynamic.conf_int().loc[f'post_week_{week}']
    print(f"  Week {week}: {coef:.2f} (p={pval:.3f}, 95% CI: [{ci[0]:.2f}, {ci[1]:.2f}])")

import pandas as pd
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main DiD model with branch fixed effects
model = ols('service_score ~ C(branch_id) + after_switch + treatment + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== KEY STATISTICS ===")
print(f"\nTreatment Effect (DiD coefficient): {model.params['treatment']:.3f}")
print(f"Standard Error: {model.bse['treatment']:.3f}")
print(f"95% CI: [{model.conf_int().loc['treatment'][0]:.3f}, {model.conf_int().loc['treatment'][1]:.3f}]")
print(f"P-value: {model.pvalues['treatment']:.4f}")

# Simple group means for verification
print("\n=== GROUP MEANS (Verification) ===")
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
before_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
after_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"Control - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control-before_control:.2f}")
print(f"Treatment - Before: {before_treatment:.2f}, After: {after_treatment:.2f}, Change: {after_treatment-before_treatment:.2f}")
did_manual = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nManual DiD calculation: {did_manual:.2f}")

# Effect size context
print("\n=== EFFECT SIZE CONTEXT ===")
overall_mean = df['service_score'].mean()
overall_std = df['service_score'].std()
print(f"Overall service score mean: {overall_mean:.2f}")
print(f"Overall service score std: {overall_std:.2f}")
print(f"Effect as % of mean: {model.params['treatment']/overall_mean*100:.1f}%")
print(f"Effect in standard deviations: {model.params['treatment']/overall_std:.2f} SD")

# Staffing/visits coefficients for comparison
print("\n=== COMPARISON TO OTHER FACTORS ===")
print(f"Staffing index coefficient: {model.params['staffing_index']:.3f} (p={model.pvalues['staffing_index']:.4f})")
print(f"Visits index coefficient: {model.params['visits_index']:.3f} (p={model.pvalues['visits_index']:.4f})")
