import pandas as pd
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Calculate mean service scores by week for each group
weekly_scores = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Create the plot
fig, ax = plt.subplots(figsize=(10, 6))

for group in [0, 1]:
    group_data = weekly_scores[weekly_scores['switched_branch'] == group]
    label = 'Switched branches' if group == 1 else 'Non-switched branches'
    ax.plot(group_data['week'], group_data['service_score'], marker='o', label=label, linewidth=2)

# Add vertical line at week 15 (switch point)
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch occurred')
ax.axvspan(15, 22, alpha=0.2, color='red', label='Post-switch period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Mean Service Score', fontsize=12)
ax.set_title('Service Scores Over Time by Group', fontsize=14)
ax.legend(loc='best')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("Plot saved as service_scores_trend.png")

# Check parallel trends statistically - compare trends in pre-period
pre_period = df[df['week'] < 15]

# Run a regression on pre-period only to test if trends were parallel
pre_period = pre_period.copy()
pre_period['week_centered'] = pre_period['week'] - pre_period['week'].mean()
pre_period['trend_interaction'] = pre_period['switched_branch'] * pre_period['week_centered']

model_trends = ols('service_score ~ switched_branch + week_centered + trend_interaction', data=pre_period).fit(cov_type='cluster', cov_kwds={'groups': pre_period['branch_id']})
print("\n=== Parallel Trends Test (Pre-period only) ===")
print(model_trends.summary())
print(f"\nTrend interaction coefficient: {model_trends.params['trend_interaction']:.4f}")
print(f"P-value: {model_trends.pvalues['trend_interaction']:.4f}")
print("(If not significant, parallel trends assumption holds)")
