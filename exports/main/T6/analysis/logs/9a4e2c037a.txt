print(f"\nBaseline average service score: {baseline_score:.2f}")

for bid in treatment_branches:
    branch_data = df[df['branch_id']==bid]
    before = branch_data[branch_data['after_switch']==0]['service_score'].mean()

    after = branch_data[branch_data['after_switch']==1]['service_score'].mean()
    control_changes.append({'branch_id': bid, 'before': before, 'after': after, 'change': after - before})

print(f"  Std dev: {branch_effects_df['change'].std():.2f}")
print(f"  Min change: {branch_effects_df['change'].min():.2f}")
print(f"  Max change: {branch_effects_df['change'].max():.2f}")
