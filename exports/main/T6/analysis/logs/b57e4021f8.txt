import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Count branches by switch status
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate average service scores by group and time period
print("\n\nAverage service scores:")
print(df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']))

# Calculate the DiD manually
print("\n\nDifference-in-Differences calculation:")
# Before switch (after_switch=0)
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
before_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
# After switch (after_switch=1)
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
after_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"Control group - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")
print(f"Treatment group - Before: {before_treatment:.2f}, After: {after_treatment:.2f}, Change: {after_treatment - before_treatment:.2f}")
print(f"\nDiD estimate: {(after_treatment - before_treatment) - (after_control - before_control):.2f}")

# Weekly trends by group
print("\n\nWeekly average service scores by group:")
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(weekly)

# Basic DiD model

print("Basic DiD Regression (without clustering):")
print(model.summary())

# With cluster-robust standard errors at branch level
model_cluster = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

model_controls = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with controls (clustered by branch):")
print(model_controls.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check parallel trends - compare trends before the switch
# Create week dummies interacted with treatment
df['week_cat'] = df['week'].astype(str)

# Event study - create relative week indicators
# Week 14 is the last pre-switch week, use as reference
df['rel_week'] = df['week'] - 14

# Create indicators for each week relative to switch, excluding week 0 (reference)
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != 0:  # skip reference week
        df[f'rel_{w}'] = ((df['rel_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
event_vars = [f'rel_{w}' for w in range(-8, 9) if w != 0]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Coefficients (relative to week 14):")
for w in range(-8, 9):
    if w != 0:
        coef = model_event.params[f'rel_{w}']
        se = model_event.bse[f'rel_{w}']
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        sig = "***" if model_event.pvalues[f'rel_{w}'] < 0.01 else "**" if model_event.pvalues[f'rel_{w}'] < 0.05 else "*" if model_event.pvalues[f'rel_{w}'] < 0.1 else ""
        print(f"Week {14+w}: {coef:.2f} (SE: {se:.2f}) 95% CI: [{ci_low:.2f}, {ci_high:.2f}] {sig}")

# Pre-trend test: test if all pre-period coefficients are jointly zero
pre_weeks = [f'rel_{w}' for w in range(-8, 0)]
hypothesis = ', '.join([f'{w} = 0' for w in pre_weeks])
pre_trend_test = model_event.f_test(hypothesis)
print(f"\n\nPre-trend test (all pre-period coeffs = 0): F={pre_trend_test.fvalue:.2f}, p={pre_trend_test.pvalue:.4f}")

# Post-period test
post_weeks = [f'rel_{w}' for w in range(1, 9)]
hypothesis_post = ', '.join([f'{w} = 0' for w in post_weeks])
post_test = model_event.f_test(hypothesis_post)
print(f"Post-period test (all post-period coeffs = 0): F={post_test.fvalue:.2f}, p={post_test.pvalue:.4f}")

# Event study - create relative week indicators
# Week 14 is the last pre-switch week, use as reference
df['rel_week'] = df['week'] - 14

# Create indicators for each week relative to switch, with safe names
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != 0:  # skip reference week
        col_name = f'rw_{abs(w)}_{"neg" if w < 0 else "pos"}'
        df[col_name] = ((df['rel_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Build formula
event_vars = []
for w in range(-8, 9):
    if w != 0:
        col_name = f'rw_{abs(w)}_{"neg" if w < 0 else "pos"}'
        event_vars.append(col_name)

formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Coefficients (relative to week 14):")
results = []
for w in range(-8, 9):
    if w != 0:
        col_name = f'rw_{abs(w)}_{"neg" if w < 0 else "pos"}'
        coef = model_event.params[col_name]
        se = model_event.bse[col_name]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        sig = "***" if model_event.pvalues[col_name] < 0.01 else "**" if model_event.pvalues[col_name] < 0.05 else "*" if model_event.pvalues[col_name] < 0.1 else ""
        results.append((w, coef, se, ci_low, ci_high, sig))
        print(f"Week {14+w} (rel {w:+d}): {coef:.2f} (SE: {se:.2f}) 95% CI: [{ci_low:.2f}, {ci_high:.2f}] {sig}")

# Pre-trend test: test if all pre-period coefficients are jointly zero
pre_vars = [f'rw_{abs(w)}_neg' for w in range(-8, 0)]
hypothesis = ', '.join([f'{v} = 0' for v in pre_vars])
pre_trend_test = model_event.f_test(hypothesis)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Re-examine the event study more carefully
df['rel_week'] = df['week'] - 14

# Create indicators for each week relative to switch, with safe names
for w in range(-8, 9):
    if w != 0:
        col_name = f'rw_{abs(w)}_{"neg" if w < 0 else "pos"}'
        df[col_name] = ((df['rel_week'] == w) & (df['switched_branch'] == 1)).astype(int)

event_vars = []
for w in range(-8, 9):
    if w != 0:
        col_name = f'rw_{abs(w)}_{"neg" if w < 0 else "pos"}'
        event_vars.append(col_name)

formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== EVENT STUDY COEFFICIENTS ===\n")
print("Pre-period (should be ~0 if parallel trends hold):")
pre_coefs = []
for w in range(-8, 0):
    col_name = f'rw_{abs(w)}_neg'
    coef = model_event.params[col_name]
    se = model_event.bse[col_name]
    pval = model_event.pvalues[col_name]
    pre_coefs.append((w, coef, pval))
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Week {14+w} (rel {w:+d}): {coef:+.2f} (p={pval:.3f}) {sig}")

print("\nPost-period (treatment effect):")
post_coefs = []
for w in range(1, 9):
    col_name = f'rw_{w}_pos'
    coef = model_event.params[col_name]
    se = model_event.bse[col_name]
    pval = model_event.pvalues[col_name]
    post_coefs.append((w, coef, pval))
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Week {14+w} (rel +{w}): {coef:+.2f} (p={pval:.3f}) {sig}")

# Joint tests
print("\n=== JOINT TESTS ===")
pre_vars = [f'rw_{abs(w)}_neg' for w in range(-8, 0)]
hypothesis = ', '.join([f'{v} = 0' for v in pre_vars])
pre_trend_test = model_event.f_test(hypothesis)
print(f"Pre-period coefficients jointly = 0: F={pre_trend_test.fvalue:.2f}, p={pre_trend_test.pvalue:.4f}")

post_vars = [f'rw_{w}_pos' for w in range(1, 9)]
hypothesis_post = ', '.join([f'{v} = 0' for v in post_vars])
post_test = model_event.f_test(hypothesis_post)

# Average post-period effect
print("\n=== AVERAGE EFFECTS ===")

print(f"Average post-period coefficient: {avg_post:.2f}")

print(f"\nPost weeks significantly positive: {sig_positive} of 8")
