import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# Create treatment groups
# Treatment = switched branches after the switch
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary: branches by switch status
print("=== Branches by Switch Status ===")
switch_summary = df.groupby('switched_branch')['branch_id'].nunique()
print(switch_summary)
print(f"\nTotal branches: {df['branch_id'].nunique()}")
print(f"Switched branches: {switch_summary.get(1, 0)}")
print(f"Non-switched branches: {switch_summary.get(0, 0)}")

# Average service scores by group and period
print("\n=== Average Service Scores ===")
before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

print("\nBEFORE switch (weeks 6-14):")
print(before.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

print("\nAFTER switch (weeks 15-22):")
print(after.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

# Difference-in-Differences calculation
print("\n=== Difference-in-Differences ===")
# Before period
before_switched = before[before['switched_branch'] == 1]['service_score'].mean()
before_control = before[before['switched_branch'] == 0]['service_score'].mean()
before_diff = before_switched - before_control

# After period
after_switched = after[after['switched_branch'] == 1]['service_score'].mean()
after_control = after[after['switched_branch'] == 0]['service_score'].mean()
after_diff = after_switched - after_control

# DiD estimate
did_estimate = after_diff - before_diff

print(f"Before - Switched avg: {before_switched:.3f}")
print(f"Before - Control avg: {before_control:.3f}")
print(f"Before difference (switched - control): {before_diff:.3f}")
print()
print(f"After - Switched avg: {after_switched:.3f}")
print(f"After - Control avg: {after_control:.3f}")
print(f"After difference (switched - control): {after_diff:.3f}")
print()
print(f"DiD Estimate (change in difference): {did_estimate:.3f}")

# Weekly trends
print("\n=== Weekly Average Service Scores ===")
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(weekly)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create treatment variable
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch fixed effects
print("=== Difference-in-Differences Regression ===")
print("\nModel 1: Basic DiD (no fixed effects)")
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model1.summary().tables[1])

print("\n" + "="*60)
print("Model 2: With Branch Fixed Effects")
# Add branch fixed effects
df_with_fe = df.copy()
branch_dummies = pd.get_dummies(df_with_fe['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df_with_fe, branch_dummies], axis=1)

# Build formula with branch FEs
branch_cols = [c for c in df_with_fe.columns if c.startswith('branch_')]
formula = f"service_score ~ switched_branch + after_switch + treatment + {' + '.join(branch_cols)}"
model2 = ols(formula, data=df_with_fe).fit()
print(f"\nTreatment coefficient: {model2.params['treatment']:.4f}")
print(f"Standard error: {model2.bse['treatment']:.4f}")
print(f"t-statistic: {model2.tvalues['treatment']:.4f}")
print(f"p-value: {model2.pvalues['treatment']:.6f}")
print(f"95% CI: [{model2.conf_int().loc['treatment', 0]:.4f}, {model2.conf_int().loc['treatment', 1]:.4f}]")

print("\n" + "="*60)
print("Model 3: With Branch FEs and Control Variables")
formula3 = f"service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index + {' + '.join(branch_cols)}"
model3 = ols(formula3, data=df_with_fe).fit()
print(f"\nTreatment coefficient: {model3.params['treatment']:.4f}")
print(f"Standard error: {model3.bse['treatment']:.4f}")
print(f"t-statistic: {model3.tvalues['treatment']:.4f}")
print(f"p-value: {model3.pvalues['treatment']:.6f}")
print(f"95% CI: [{model3.conf_int().loc['treatment', 0]:.4f}, {model3.conf_int().loc['treatment', 1]:.4f}]")
print(f"\nR-squared: {model3.rsquared:.4f}")

print(f"   Treatment effect: {model_basic.params['treatment']:.2f} points")

print(f"   p-value: {model_basic.pvalues['treatment']:.4f}")

# Pre-trend adjusted: include linear pre-trend
print("\n2. ADJUSTING FOR PRE-EXISTING TREND:")
# Create a variable for branch-specific pre-trend
# Calculate each switched branch's trend in the pre-period
pre_data = df[df['week'] < 15].copy()
branch_trends = pre_data.groupby('branch_id').apply(
    lambda x: pd.Series({
        'pre_trend': np.polyfit(x['week'], x['service_score'], 1)[0] if len(x) > 1 else 0
    })
).reset_index()

df['trend_adjustment'] = df['switched_branch'] * df['after_switch'] * df['pre_trend']

X2 = df[['switched_branch', 'after_switch', 'treatment', 'pre_trend', 'trend_adjustment']]
X2 = sm.add_constant(X2)
model_adjusted = sm.OLS(y, X2).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
