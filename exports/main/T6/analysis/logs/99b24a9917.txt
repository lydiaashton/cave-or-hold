import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Number of unique branches
print("Number of unique branches:", df['branch_id'].nunique())
print("Switched branches:", df[df['switched_branch']==1]['branch_id'].nunique())
print("Non-switched branches:", df[df['switched_branch']==0]['branch_id'].nunique())

# Average service scores by group and period
print("\n=== Average Service Scores ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the difference-in-differences manually
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
before_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print("\n=== DiD Components ===")
print(f"Control (non-switched) before: {before_control:.2f}")
print(f"Control (non-switched) after: {after_control:.2f}")
print(f"Control change: {after_control - before_control:.2f}")
print()
print(f"Treatment (switched) before: {before_treatment:.2f}")
print(f"Treatment (switched) after: {after_treatment:.2f}")
print(f"Treatment change: {after_treatment - before_treatment:.2f}")
print()
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"Difference-in-Differences estimate: {did_estimate:.2f}")

# Check for pre-trends (weeks 6-14)
print("\n=== Pre-trend Check (weeks 6-14) ===")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate

model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("=== Simple DiD Regression ===")
print(model.summary())

# Now let's add branch fixed effects and week fixed effects for a more robust analysis
# This controls for time-invariant branch characteristics and common time trends

df['switched_after'] = df['switched_branch'] * df['after_switch']

# Branch fixed effects regression
model_fe = ols('service_score ~ switched_after + C(branch_id) + C(week)', data=df).fit()
print("\n=== DiD with Branch and Week Fixed Effects ===")
print(model_fe.summary())

# Cluster-robust standard errors at the branch level
model_fe_cluster = ols('service_score ~ switched_after + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD with Branch and Week Fixed Effects (Clustered SEs) ===")
print(model_fe_cluster.summary())
