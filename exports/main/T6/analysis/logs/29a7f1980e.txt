treated_change = treated_after - treated_before

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create event time relative to switch (week 15)
# Weeks 6-14 are before (negative), weeks 15-22 are after (positive)
df['event_time'] = df['week'] - 15

# Event study regression - interact switched_branch with each week dummy
# Use week 14 as reference (just before switch)
df['week_cat'] = df['week'].astype(str)

# Create dummies for each week except week 14 (reference)
weeks = sorted(df['week'].unique())
reference_week = 14

# Build formula for event study
# We'll interact switched_branch with week indicators
event_study_vars = []
for w in weeks:
    if w != reference_week:
        event_study_vars.append(f'switched_branch:week_cat("{w}")')

formula = f'service_score ~ switched_branch + week_cat + {" + ".join(event_study_vars)}'

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results (relative to week 14) ===")
# Extract just the interaction terms
coef_df = pd.DataFrame({

    'std_err': model_event.bse,
    'p_value': model_event.pvalues
})

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create event time relative to switch (week 15)
df['event_time'] = df['week'] - 15

# Create week dummies and interactions manually
weeks = sorted(df['week'].unique())
reference_week = 14

for w in weeks:
    df[f'week_{w}'] = (df['week'] == w).astype(int)
    if w != reference_week:
        df[f'treated_week_{w}'] = df['switched_branch'] * (df['week'] == w).astype(int)

# Build regression formula
pre_weeks = [w for w in weeks if w < reference_week]
post_weeks = [w for w in weeks if w > reference_week]

# Include week dummies (excluding reference) and interactions
week_dummies = ' + '.join([f'week_{w}' for w in weeks if w != reference_week])
interactions = ' + '.join([f'treated_week_{w}' for w in weeks if w != reference_week])

formula = f'service_score ~ switched_branch + {week_dummies} + {interactions}'

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results ===")
print("(Coefficients on treated_week_X show difference relative to week 14)")
print("\nPre-switch period (testing parallel trends):")
for w in pre_weeks:
    coef = model_event.params[f'treated_week_{w}']
    se = model_event.bse[f'treated_week_{w}']
    pval = model_event.pvalues[f'treated_week_{w}']
    ci_low = coef - 1.96 * se
    ci_high = coef + 1.96 * se
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"Week {w}: coef={coef:.2f}, SE={se:.2f}, 95%CI=[{ci_low:.2f}, {ci_high:.2f}] {sig}")

print("\nPost-switch period (treatment effects):")
for w in post_weeks:
    coef = model_event.params[f'treated_week_{w}']

    ci_low = coef - 1.96 * se
    ci_high = coef + 1.96 * se
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""

df = pd.read_csv('branches.csv')

# Event study coefficients from previous analysis
weeks = list(range(6, 23))
reference_week = 14

# Recreate the event study to get all coefficients
for w in weeks:
    df[f'week_{w}'] = (df['week'] == w).astype(int)
    if w != reference_week:
        df[f'treated_week_{w}'] = df['switched_branch'] * (df['week'] == w).astype(int)

week_dummies = ' + '.join([f'week_{w}' for w in weeks if w != reference_week])
interactions = ' + '.join([f'treated_week_{w}' for w in weeks if w != reference_week])
formula = f'service_score ~ switched_branch + {week_dummies} + {interactions}'

from statsmodels.formula.api import ols
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients and confidence intervals
event_coefs = []
event_se = []
event_weeks = []

for w in weeks:
    if w == reference_week:
        event_coefs.append(0)  # Reference week
        event_se.append(0)
    else:
        event_coefs.append(model_event.params[f'treated_week_{w}'])
        event_se.append(model_event.bse[f'treated_week_{w}'])
    event_weeks.append(w)
