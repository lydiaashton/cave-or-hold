# Run DiD regression with robust standard errors
# Model: service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch*after_switch) + error

# Basic DiD model
model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='HC1')
print("=== Basic DiD Regression ===")
print(model1.summary().tables[1])

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Calculate the effect for each switched branch
# Compare their average post-switch score to their average pre-switch score
# Then compare to the control group's change

# Pre and post averages by branch
branch_stats = df.groupby(['branch_id', 'switched_branch']).agg(
    pre_avg=('service_score', lambda x: x[df.loc[x.index, 'week'] < 15].mean() if len(x[df.loc[x.index, 'week'] < 15]) > 0 else np.nan),
    post_avg=('service_score', lambda x: x[df.loc[x.index, 'week'] >= 15].mean() if len(x[df.loc[x.index, 'week'] >= 15]) > 0 else np.nan),
    pre_count=('week', lambda x: len(x[x < 15])),
    post_count=('week', lambda x: len(x[x >= 15]))
).reset_index()

# Calculate change for each branch
branch_stats['change'] = branch_stats['post_avg'] - branch_stats['pre_avg']

# Compare treatment vs control changes
treatment_changes = branch_stats[branch_stats['switched_branch']==1]['change']
control_changes = branch_stats[branch_stats['switched_branch']==0]['change']

print(f"  Mean change: {treatment_changes.mean():.2f}")

print(f"  Min: {treatment_changes.min():.2f}")
print(f"  Max: {treatment_changes.max():.2f}")
print(f"  % with positive change: {(treatment_changes > 0).mean()*100:.1f}%")
