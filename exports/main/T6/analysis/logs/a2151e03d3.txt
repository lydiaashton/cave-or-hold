import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Run difference-in-differences regression
# Basic DiD model
model1 = ols('service_score ~ treated + post + DiD', data=df).fit()
print("Basic DiD Model:")
print(model1.summary())

# DiD with branch fixed effects
model2 = ols('service_score ~ treated + post + DiD + C(branch_id)', data=df).fit()
print("\n\nDiD with Branch Fixed Effects:")
print(model2.summary())

# DiD with branch fixed effects and control variables
model3 = ols('service_score ~ treated + post + DiD + staffing_index + visits_index + C(branch_id)', data=df).fit()
print("\n\nDiD with Branch FE and Controls:")
print(model3.summary())
