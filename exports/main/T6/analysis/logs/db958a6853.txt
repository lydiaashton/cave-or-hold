import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Model 1: Basic DiD
print("=== Model 1: Basic Difference-in-Differences ===")
model1 = ols('service_score ~ switched_branch + after_switch + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model1.summary().tables[1])

# Model 2: DiD with controls
print("\n=== Model 2: DiD with Controls (staffing_index, visits_index) ===")
model2 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model2.summary().tables[1])

# Model 3: With branch fixed effects
print("\n=== Model 3: DiD with Branch Fixed Effects ===")
df_with_dummies = pd.get_dummies(df, columns=['branch_id'], drop_first=True)
X = df_with_dummies[['switched_branch', 'after_switch', 'did', 'staffing_index', 'visits_index']]
X = sm.add_constant(X)
# Add branch dummies
for col in df_with_dummies.columns:
    if col.startswith('branch_id_'):
        X[col] = df_with_dummies[col]
y = df_with_dummies['service_score']
model3 = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("DID coefficient:", model3.params['did'], "Std Error:", model3.bse['did'], "p-value:", model3.pvalues['did'])

# Alternative: use formula with C() for fixed effects
print("\n=== Model 3b: Using Formula with Branch FE ===")
model3b = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model3b.summary().tables[1].loc[['did', 'Intercept']])

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the event study data
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].agg(['mean', 'std', 'count']).reset_index()
weekly_pivot = weekly_avg.pivot(index='week', columns='switched_branch', values=['mean', 'std', 'count'])

# Calculate the difference and confidence intervals
diff = weekly_pivot['mean'][1] - weekly_pivot['mean'][0]

# Standard error of difference
se_diff = np.sqrt(
    (weekly_pivot['std'][1]**2 / weekly_pivot['count'][1]) + 
    (weekly_pivot['std'][0]**2 / weekly_pivot['count'][0])
)

ci_lower = diff - 1.96 * se_diff
ci_upper = diff + 1.96 * se_diff

print("=== Weekly Differences with 95% CI ===")
event_study = pd.DataFrame({
    'week': diff.index,
    'diff': diff.values,
    'ci_lower': ci_lower.values,
    'ci_upper': ci_upper.values
})
event_study['post_switch'] = event_study['week'] >= 15
print(event_study.round(2))

# Create a plot
fig, ax = plt.subplots(figsize=(10, 6))
ax.plot(event_study['week'], event_study['diff'], 'o-', label='Difference (Switched - Control)', linewidth=2)
ax.fill_between(event_study['week'], event_study['ci_lower'], event_study['ci_upper'], alpha=0.3, label='95% CI')
