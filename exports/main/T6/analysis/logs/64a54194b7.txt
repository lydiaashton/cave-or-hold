import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load data

# Run difference-in-differences regression
# Basic DiD model
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("=== Basic Difference-in-Differences Regression ===")
print(model.summary())

# Model with branch fixed effects
print("\n\n=== Model with Branch Fixed Effects ===")
# Create branch dummies
df_temp = df.copy()
df_temp['branch_id'] = df_temp['branch_id'].astype('category')
model_fe = ols('service_score ~ switched_branch * after_switch + C(branch_id)', data=df_temp).fit()
# Only show key coefficients
print(model_fe.summary().tables[1].iloc[:10])

# Model with branch fixed effects and robust standard errors (clustered by branch)
print("\n\n=== Model with Branch FE and Clustered SEs ===")
model_fe_robust = ols('service_score ~ switched_branch * after_switch + C(branch_id)', data=df_temp).fit(cov_type='cluster', cov_kwds={'groups': df_temp['branch_id']})

did_coef = model_fe_robust.params['switched_branch:after_switch']

print(f"P-value: {did_pvalue:.4f}")
