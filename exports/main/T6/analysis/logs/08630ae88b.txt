import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks relative to switch at week 15)
# Week 14 is -1, week 15 is 0, etc.
df['relative_week'] = df['week'] - 15

# Event study regression with week fixed effects
# We'll use week 14 as the reference (last pre-switch week)
df['week_factor'] = df['week'].astype(str)

# Create interaction terms for each week with treatment
# Reference week is 14 (last pre-period week)
weeks = sorted(df['week'].unique())
ref_week = 14

# Create dummies for each week (except reference)
for w in weeks:
    if w != ref_week:
        df[f'week_{w}'] = (df['week'] == w).astype(int)
        df[f'treat_week_{w}'] = ((df['week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Build formula
pre_weeks = [w for w in weeks if w < ref_week]
post_weeks = [w for w in weeks if w > ref_week]

# Simple event study - just post weeks interactions
post_terms = ' + '.join([f'treat_week_{w}' for w in post_weeks])
pre_terms = ' + '.join([f'treat_week_{w}' for w in pre_weeks])

formula = f'service_score ~ switched_branch + C(week) + {post_terms}'

model_event = smf.ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results (Post-Period Effects) ===")
print(f"Reference week: {ref_week}")
print("\nCoefficients for treatment interactions:")
for w in post_weeks:
    coef = model_event.params[f'treat_week_{w}']
    se = model_event.bse[f'treat_week_{w}']
    pval = model_event.pvalues[f'treat_week_{w}']
    print(f"  Week {w}: coef={coef:.2f}, SE={se:.2f}, p={pval:.3f}")

# Average post-period effect
post_coefs = [model_event.params[f'treat_week_{w}'] for w in post_weeks]
print(f"\nAverage post-period effect: {np.mean(post_coefs):.2f}")

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# DiD model with controls
did_model = smf.ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Key DiD Results ===")
print(f"Treatment effect (switched_branch:after_switch): {did_model.params['switched_branch:after_switch']:.3f}")
print(f"Standard error: {did_model.bse['switched_branch:after_switch']:.3f}")
print(f"p-value (two-tailed): {did_model.pvalues['switched_branch:after_switch']:.3f}")
print(f"t-statistic: {did_model.tvalues['switched_branch:after_switch']:.3f}")

# 95% CI
coef = did_model.params['switched_branch:after_switch']
se = did_model.bse['switched_branch:after_switch']
ci_lower = coef - 1.96 * se
ci_upper = coef + 1.96 * se
print(f"95% CI: [{ci_lower:.3f}, {ci_upper:.3f}]")

# One-tailed p-value (since they had a directional hypothesis)
one_tailed_p = did_model.pvalues['switched_branch:after_switch'] / 2
print(f"One-tailed p-value: {one_tailed_p:.3f}")

# Also check the simple DiD without controls
