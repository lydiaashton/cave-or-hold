import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary table: average service scores by group and period
summary_table = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).unstack()
print("Service Score Summary by Group and Period:")
print(summary_table)
print()

# Calculate the DiD estimate manually
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("Mean Service Scores:")
print(f"Control (not switched) - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")
print(f"Treatment (switched) - Before: {before_treatment:.2f}, After: {after_treatment:.2f}, Change: {after_treatment - before_treatment:.2f}")
print(f"\nDifference-in-Differences Estimate: {(after_treatment - before_treatment) - (after_control - before_control):.2f}")
print()

# Run DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("DiD Regression Results (without controls):")
print(model.summary())
print()

# Run DiD regression with controls
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("DiD Regression Results (with controls):")
print(model_controls.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Simpler FE model - treatment and after_switch only (switched_branch absorbed by FE)
# Actually, let's use a different approach - just verify our main results are robust

# Main DiD without controls
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
cr1 = model1.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Main DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
cr2 = model2.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Pre-trend adjusted DiD (include pre-period trend interaction)
df['pre_trend'] = df['week'] * df['switched_branch']
model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index + pre_trend', data=df).fit()
cr3 = model3.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("=== Robustness Checks ===")
print("\n1. DiD without controls:")
print(f"   Treatment effect: {cr1.params[3]:.2f} (SE: {cr1.bse[3]:.2f}, p: {cr1.pvalues[3]:.3f})")

print("\n2. DiD with controls (main specification):")
print(f"   Treatment effect: {cr2.params[3]:.2f} (SE: {cr2.bse[3]:.2f}, p: {cr2.pvalues[3]:.3f})")

print("\n3. DiD with pre-trend adjustment:")
print(f"   Treatment effect: {cr3.params[3]:.2f} (SE: {cr3.bse[3]:.2f}, p: {cr3.pvalues[3]:.3f})")

post_df = df[df['after_switch'] == 1]
