print(f"  Change: {treatment_change:.2f}")

# Main DiD results
model = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
