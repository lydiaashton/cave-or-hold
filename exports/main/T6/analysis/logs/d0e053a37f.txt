import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run a difference-in-differences regression
# service_score = β0 + β1*switched_branch + β2*after_switch + β3*(switched_branch * after_switch) + ε
# β3 is the DiD estimator - the effect of the treatment

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Difference-in-Differences Regression Results:")
print("=" * 60)
print(model.summary())

# Also run with cluster-robust standard errors by branch
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n" + "=" * 60)
print("With Cluster-Robust Standard Errors (by branch):")
print("=" * 60)
print(model_clustered.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create week-specific treatment effects for post-switch weeks
df['treatment'] = df['switched_branch'] * df['after_switch']

# Event study: create indicators for each week relative to switch
# Week 14 is the last pre-switch week (baseline)
df['week_relative'] = df['week'] - 14  # So week 14 = 0, week 15 = 1, etc.

# Create dummies for each week relative to switch (excluding week 0 as baseline)
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != 0:  # Skip baseline
        df[f'week_{w}'] = ((df['week_relative'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
week_cols = [f'week_{w}' for w in range(-8, 9) if w != 0]
formula = 'service_score ~ switched_branch + ' + ' + '.join(week_cols)
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Regression (week-specific treatment effects):")
print("=" * 70)
print("Baseline: Week 14 (last pre-switch week)")
print("Coefficients show treatment effect for each week relative to baseline")
print("=" * 70)

# Extract and display the week-specific coefficients
results = []
for w in range(-8, 9):
    if w != 0:
        coef = model_event.params[f'week_{w}']
        se = model_event.bse[f'week_{w}']
        pval = model_event.pvalues[f'week_{w}']
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        results.append({
            'week': 14 + w,
            'relative_week': w,
            'coef': coef,
            'se': se,
            'pval': pval,
            'ci_low': ci_low,
            'ci_high': ci_high
        })

results_df = pd.DataFrame(results)
print(results_df.to_string(index=False, float_format='%.3f'))

# Calculate average post-switch effect
post_weeks = results_df[results_df['relative_week'] > 0]
avg_post_effect = post_weeks['coef'].mean()
print(f"\nAverage post-switch effect (weeks 15-22): {avg_post_effect:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create week-specific treatment effects for post-switch weeks
df['treatment'] = df['switched_branch'] * df['after_switch']

# Event study: create indicators for each week relative to switch
# Week 14 is the last pre-switch week (baseline)
df['week_relative'] = df['week'] - 14  # So week 14 = 0, week 15 = 1, etc.

# Create dummies for each week relative to switch (excluding week 0 as baseline)
# Use w prefix to avoid issues with negative numbers in column names
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != 0:  # Skip baseline
        df[f'w{abs(w)}_neg{1 if w < 0 else 0}'] = ((df['week_relative'] == w) & (df['switched_branch'] == 1)).astype(int)

# Build formula
week_cols = []
for w in range(-8, 9):
    if w != 0:
        neg = 1 if w < 0 else 0
        week_cols.append(f'w{abs(w)}_neg{neg}')

formula = 'service_score ~ switched_branch + ' + ' + '.join(week_cols)
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Regression (week-specific treatment effects):")
print("=" * 70)
print("Baseline: Week 14 (last pre-switch week)")
print("Coefficients show treatment effect for each week relative to baseline")
print("=" * 70)

# Extract and display the week-specific coefficients
results = []
for w in range(-8, 9):
    if w != 0:
        neg = 1 if w < 0 else 0
        col = f'w{abs(w)}_neg{neg}'
        coef = model_event.params[col]
        se = model_event.bse[col]
        pval = model_event.pvalues[col]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        results.append({
            'week': 14 + w,
            'relative_week': w,
            'period': 'Pre' if w < 0 else 'Post',
            'coef': coef,
            'se': se,
            'pval': pval,
            'ci_low': ci_low,
            'ci_high': ci_high
        })

results_df = pd.DataFrame(results)
print(results_df.to_string(index=False, float_format='%.3f'))

# Calculate average post-switch effect
post_weeks = results_df[results_df['relative_week'] > 0]
avg_post_effect = post_weeks['coef'].mean()
print(f"\nAverage post-switch effect (weeks 15-22): {avg_post_effect:.2f}")

# Check pre-trends - are pre-switch coefficients significantly different from zero?
pre_weeks = results_df[results_df['relative_week'] < 0]
print(f"\nPre-switch coefficients (should be close to 0 if parallel trends hold):")
print(pre_weeks[['week', 'coef', 'pval']].to_string(index=False, float_format='%.3f'))
