import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate mean service scores by group and period
print("=" * 60)
print("MEAN SERVICE SCORES BY GROUP AND PERIOD")
print("=" * 60)

# Pre-period (after_switch = 0)
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
pre_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()

# Post-period (after_switch = 1)
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
post_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nControl Group (no switch):")
print(f"  Pre-period mean: {pre_control:.2f}")
print(f"  Post-period mean: {post_control:.2f}")
print(f"  Change: {post_control - pre_control:.2f}")

print(f"\nTreatment Group (switched):")
print(f"  Pre-period mean: {pre_treatment:.2f}")
print(f"  Post-period mean: {post_treatment:.2f}")
print(f"  Change: {post_treatment - pre_treatment:.2f}")

# Difference-in-Differences estimate
did_estimate = (post_treatment - pre_treatment) - (post_control - pre_control)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

# Run DiD regression
print("\n" + "=" * 60)
print("DIFFERENCE-IN-DIFFERENCES REGRESSION")
print("=" * 60)

# Basic DiD model
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")

model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Test pre-trends: restrict to pre-period only
pre_period = df[df['after_switch'] == 0].copy()
pre_model = ols('service_score ~ switched_branch + C(week)', data=pre_period).fit(cov_type='cluster', cov_kwds={'groups': pre_period['branch_id']})
print("=" * 60)
print("PRE-TRENDS TEST (pre-period only, weeks 6-14)")
print("=" * 60)
print(pre_model.summary().tables[1])

# More flexible DiD: allow treatment effect to vary by post-period week
print("\n" + "=" * 60)
print("FLEXIBLE DiD: Treatment effect by post-period week")
print("=" * 60)

# Create interaction of treatment with each post-period week
df['post_week'] = df['week'].apply(lambda x: x if x >= 15 else None)
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Model with week fixed effects and treatment interaction
model_flex = ols('service_score ~ switched_branch + C(week) + treatment_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_flex.summary().tables[1])

# Confidence interval for the main effect
print("\n" + "=" * 60)
print("EFFECT SIZE SUMMARY")
print("=" * 60)
main_model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
treatment_coef = main_model.params['treatment']
treatment_se = main_model.bse['treatment']
treatment_ci_low = treatment_coef - 1.96 * treatment_se
treatment_ci_high = treatment_coef + 1.96 * treatment_se

print(f"Point estimate: {treatment_coef:.2f}")
print(f"95% CI: [{treatment_ci_low:.2f}, {treatment_ci_high:.2f}]")
print(f"P-value: {main_model.pvalues['treatment']:.4f}")

# How many branches improved?
branch_effects = df.groupby(['branch_id', 'switched_branch'])['service_score'].mean().unstack()
branch_effects['diff'] = branch_effects[1] - branch_effects[0]
# But this isn't quite right - we need pre-post differences per branch

# Better: calculate pre-post change for each branch
branch_pre = df[df['after_switch'] == 0].groupby('branch_id')['service_score'].mean()
branch_post = df[df['after_switch'] == 1].groupby('branch_id')['service_score'].mean()
branch_changes = branch_post - branch_pre
branch_changes_df = pd.DataFrame({
    'pre_mean': branch_pre,
    'post_mean': branch_post,
    'change': branch_changes
})
branch_changes_df['switched'] = df.groupby('branch_id')['switched_branch'].first()

print("\n" + "=" * 60)
print("BRANCH-LEVEL CHANGES")
