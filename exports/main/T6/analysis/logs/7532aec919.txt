import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Count branches by switch status
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Create treatment group indicator (switched_branch) and post period indicator (after_switch)
# DD estimator is the interaction: switched_branch * after_switch

# First, let's look at average service scores by group and period
print("\n\nAverage service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the difference-in-differences manually
# Treatment group (switched=1): before vs after
# Control group (switched=0): before vs after

treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nTreatment group (switched branches):")
print(f"  Before switch: {treatment_before:.2f}")
print(f"  After switch: {treatment_after:.2f}")
print(f"  Change: {treatment_after - treatment_before:.2f}")

print(f"\nControl group (non-switched branches):")
print(f"  Before switch: {control_before:.2f}")
print(f"  After switch: {control_after:.2f}")
print(f"  Change: {control_after - control_before:.2f}")

dd_estimate = (treatment_after - treatment_before) - (control_after - control_before)
print(f"\nDifference-in-Differences estimate: {dd_estimate:.2f}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate average service score by week for treatment and control groups
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()

weekly_avg_pivot.columns = ['Control (no switch)', 'Treatment (switched)']

print("Weekly average service scores by group:")
print(weekly_avg_pivot)

# Calculate the difference between treatment and control each week
weekly_avg_pivot['Difference (T-C)'] = weekly_avg_pivot['Treatment (switched)'] - weekly_avg_pivot['Control (no switch)']

# Check pre-trends: average difference before week 15 vs after
pre_switch_diff = weekly_avg_pivot.loc[6:14, 'Difference (T-C)'].mean()
post_switch_diff = weekly_avg_pivot.loc[15:22, 'Difference (T-C)'].mean()
print(f"\nAverage difference before switch (weeks 6-14): {pre_switch_diff:.2f}")
print(f"Average difference after switch (weeks 15-22): {post_switch_diff:.2f}")

ax.legend(loc='best', fontsize=10)
ax.grid(True, alpha=0.3)
ax.set_xticks(weeks)

plt.tight_layout()
plt.savefig('service_scores_by_week.png', dpi=150)
print("\n\nPlot saved as 'service_scores_by_week.png'")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Test for pre-trends: interact switched_branch with week dummies for pre-period only
# If parallel trends hold, these should not be significantly different from zero

df_pre = df[df['week'] < 15].copy()
df_pre['switched_x_week'] = df_pre['switched_branch'] * df_pre['week']

# Model with week fixed effects and treatment x week interaction for pre-period
# Using week 6 as reference
model_pretrend = ols('service_score ~ switched_branch + C(week) + switched_branch:C(week)', data=df_pre).fit(cov_type='cluster', cov_kwds={'groups': df_pre['branch_id']})

print("Pre-trends test (weeks 6-14 only):")
print("Testing whether treatment and control had different trends before the switch")
print("\nCoefficients for switched_branch:week interactions:")
for coef_name in model_pretrend.params.index:
    if 'switched_branch:C(week)' in coef_name:
        coef = model_pretrend.params[coef_name]
        se = model_pretrend.bse[coef_name]
        pval = model_pretrend.pvalues[coef_name]
        print(f"  {coef_name}: {coef:.3f} (SE: {se:.3f}, p={pval:.3f})")

# Also run an event study style model for all weeks
df['week_centered'] = df['week'] - 15  # Center at switch week
df['switched_x_week_c'] = df['switched_branch'] * df['week_centered']

# Simple model to check if there's a differential trend
model_trend = ols('service_score ~ switched_branch * week_centered + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDifferential trend test (full period):")
print(model_trend.summary().tables[1])

# Calculate confidence interval for the main DD estimate
model_dd = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
dd_coef = model_dd.params['switched_branch:after_switch']
dd_se = model_dd.bse['switched_branch:after_switch']
ci_lower = dd_coef - 1.96 * dd_se
ci_upper = dd_coef + 1.96 * dd_se
print(f"\n\nDD Estimate: {dd_coef:.2f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"Standard Error: {dd_se:.2f}")
print(f"P-value: {model_dd.pvalues['switched_branch:after_switch']:.3f}")
