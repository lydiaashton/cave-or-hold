import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main model with controls
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get the treatment effect and confidence interval
treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pvalue = model.pvalues['treatment']

# 95% confidence interval
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se

print("=== MAIN RESULTS ===")
print(f"Treatment effect (DiD estimate): {treatment_coef:.3f}")
print(f"Standard error (clustered): {treatment_se:.3f}")
print(f"95% Confidence interval: [{ci_lower:.3f}, {ci_upper:.3f}]")
print(f"P-value: {treatment_pvalue:.4f}")
print(f"Statistically significant: {'Yes' if treatment_pvalue < 0.05 else 'No'}")

# Calculate the effect size relative to the baseline
baseline_score = df['service_score'].mean()
print(f"\nBaseline average service score: {baseline_score:.2f}")
print(f"Effect as % of baseline: {(treatment_coef / baseline_score) * 100:.2f}%")

# Also get the pre-switch average for treatment group
treatment_pre_avg = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
print(f"Pre-switch treatment group average: {treatment_pre_avg:.2f}")
print(f"Effect as % of treatment pre-average: {(treatment_coef / treatment_pre_avg) * 100:.2f}%")
