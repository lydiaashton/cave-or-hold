import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term
df['treated'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
print("=== Simple Difference-in-Differences Regression ===")
print(model.summary())

# Now with branch fixed effects and week fixed effects
df['switched_branch'] = df['switched_branch'].astype(int)

# Summary statistics by group and period
print("=== Mean Service Scores by Group and Period ===")

                       index='switched_branch', 
                       columns='after_switch', 
                       aggfunc='mean')
print(pivot)

print("\n=== DiD Calculation ===")
# Control group change
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_change = control_after - control_before

# Treatment group change
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_change = treat_after - treat_before

did_effect = treat_change - control_change

print(f"Control (non-switched): Before={control_before:.2f}, After={control_after:.2f}, Change={control_change:.2f}")
print(f"Treatment (switched): Before={treat_before:.2f}, After={treat_after:.2f}, Change={treat_change:.2f}")
print(f"DiD Effect (Treatment - Control): {did_effect:.2f}")
