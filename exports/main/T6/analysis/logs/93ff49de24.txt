import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = β0 + β1*switched_branch + β2*after_switch + β3*(switched_branch * after_switch) + ε
# β3 is the DiD estimator

df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# Now with branch fixed effects (more robust)
# This controls for time-invariant branch characteristics
print("\n=== DiD with Branch Fixed Effects ===")
df_with_dummies = pd.get_dummies(df, columns=['branch_id'], drop_first=True)
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_id_')]
formula = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(branch_cols)
model_fe = ols(formula, data=df_with_dummies).fit()
print(f"R-squared: {model_fe.rsquared:.3f}")
print(f"\nCoefficients of interest:")
print(f"  switched_branch: {model_fe.params['switched_branch']:.3f} (SE: {model_fe.bse['switched_branch']:.3f})")
print(f"  after_switch: {model_fe.params['after_switch']:.3f} (SE: {model_fe.bse['after_switch']:.3f})")
print(f"  treatment (DiD): {model_fe.params['treatment']:.3f} (SE: {model_fe.bse['treatment']:.3f})")

# Get p-value for treatment effect
from scipy import stats
t_stat = model_fe.params['treatment'] / model_fe.bse['treatment']
p_value = 2 * (1 - stats.t.cdf(abs(t_stat), df=model_fe.df_resid))
print(f"  t-statistic: {t_stat:.3f}, p-value: {p_value:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run regression and get cluster-robust standard errors
print("=== DiD with Cluster-Robust SE (clustered by branch) ===")
model = ols('service_score ~ after_switch + treatment + staffing_index + visits_index + C(branch_id)', data=df).fit()

# Get cluster-robust covariance matrix
cov_clustered = cov_cluster(model, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))

print(f"\nCoefficients with Cluster-Robust SE:")

p_value = 2 * (1 - stats.norm.cdf(abs(t_stat)))

print("=== Key Statistics ===")
print(f"Treatment effect (point estimate): {treat_coef:.2f} points")
print(f"Cluster-robust SE: {treat_se:.2f}")
print(f"t-statistic: {t_stat:.2f}")
print(f"p-value: {p_value:.3f}")
print(f"95% CI: [{treat_coef - 1.96*treat_se:.2f}, {treat_coef + 1.96*treat_se:.2f}]")

# What does this mean?
print(f"\n=== Interpretation ===")
print(f"The 95% CI includes zero, so we CANNOT rule out no effect.")
print(f"The 95% CI also includes effects up to +{treat_coef + 1.96*treat_se:.1f} points.")
print(f"Probability of observing this result if true effect = 0: {p_value*100:.1f}%")
