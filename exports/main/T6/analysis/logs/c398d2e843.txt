df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

print(model.summary())

# Now let's add branch fixed effects to control for time-invariant branch characteristics
# This is a more robust specification
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df_with_dummies['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Alternative: use statsmodels with cluster-robust standard errors

# Weekly averages by group

# Linear trend for each group in pre-period

for group in [0, 1]:
    group_data = pre_data[pre_data['switched_branch'] == group].copy()
    model = smf.ols('service_score ~ week', data=group_data).fit(cov_type='cluster', cov_kwds={'groups': group_data['branch_id']})
    print(f"\nGroup {group} (switched={group}):")
    print(f"  Week coefficient: {model.params['week']:.3f} (p={model.pvalues['week']:.4f})")

# Now let's run the full model with covariates
print("\n=== Full DiD Model with Covariates ===")
df['treatment'] = df['switched_branch'] * df['after_switch']
full_model = smf.ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
                     data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(full_model.summary())

# Event study - check effect by week after switch
print("\n=== Event Study: Effect by Week ===")
df['weeks_since_switch'] = df['week'] - 14  # 0 = week 14 (last pre-week), 1 = week 15 (first post-week)
df['post_week'] = df['weeks_since_switch'].apply(lambda x: max(0, x))  # 0 for all pre-weeks, then 1, 2, 3...

# Create dummies for each post-week
for w in range(1, 9):
    df[f'post_week_{w}'] = ((df['weeks_since_switch'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
post_week_vars = ' + '.join([f'post_week_{w}' for w in range(1, 9)])
event_model = smf.ols(f'service_score ~ switched_branch + after_switch + staffing_index + visits_index + {post_week_vars}', 
                      data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(event_model.summary())

# Extract the post-week coefficients

for w in range(1, 9):
    coef = event_model.params[f'post_week_{w}']
    se = event_model.bse[f'post_week_{w}']
    pval = event_model.pvalues[f'post_week_{w}']
    print(f"  Week {14+w}: {coef:.2f} (SE={se:.2f}, p={pval:.4f})")
