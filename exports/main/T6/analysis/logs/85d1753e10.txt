import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

# Load the data
df = pd.read_csv('branches.csv')

# Pre-trends test: interact switched_branch with week indicators for pre-period only
pre_period = df[df['week'] < 15].copy()
pre_period['week_cat'] = pre_period['week'].astype('category')

# Create week dummies for pre-period (using week 6 as reference)
for w in range(7, 15):
    pre_period[f'week_{w}'] = (pre_period['week'] == w).astype(int)
    pre_period[f'week_{w}_treated'] = pre_period[f'week_{w}'] * pre_period['switched_branch']

# Regress service_score on week dummies and their interactions with treatment
pre_trend_vars = '+'.join([f'week_{w}' for w in range(7, 15)])
pre_trend_interactions = '+'.join([f'week_{w}_treated' for w in range(7, 15)])

pre_trend_model = smf.ols(f'service_score ~ switched_branch + {pre_trend_vars} + {pre_trend_interactions}', data=pre_period).fit(cov_type='cluster', cov_kwds={'groups': pre_period['branch_id']})

print("=== Pre-Trends Test (Weeks 6-14) ===")
print("Testing if treatment and control had different trends before the switch")
print("\nCoefficients on interaction terms (should be ~0 if parallel trends hold):")
for w in range(7, 15):
    coef = pre_trend_model.params[f'week_{w}_treated']
    pval = pre_trend_model.pvalues[f'week_{w}_treated']
    print(f"  Week {w} x Treated: {coef:.3f} (p={pval:.3f})")

# Joint test of pre-trends
# Test if all interaction terms are jointly zero
interaction_cols = [f'week_{w}_treated' for w in range(7, 15)]
joint_test = pre_trend_model.f_test(interaction_cols)
print(f"\nJoint test of all pre-trend interactions: F={joint_test.fvalue:.2f}, p={joint_test.pvalue:.3f}")

# Calculate effect size in practical terms
print("\n=== Effect Size Interpretation ===")
# Get baseline average service score
baseline = df[df['week'] < 15]['service_score'].mean()
did_effect = 2.123  # from earlier regression
pct_change = (did_effect / baseline) * 100
print(f"Baseline average service score: {baseline:.2f}")
print(f"DiD estimated effect: +{did_effect:.2f} points")
print(f"Percentage improvement: {pct_change:.1f}%")

# Also compare to standard deviation
std_score = df['service_score'].std()
effect_in_std = did_effect / std_score
print(f"Service score standard deviation: {std_score:.2f}")
print(f"Effect size in standard deviations: {effect_in_std:.2f} SD")

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

# Load the data
df = pd.read_csv('branches.csv')

# Robustness check: Add branch fixed effects
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Model with branch fixed effects
df['branch_id_cat'] = df['branch_id'].astype('category')
model_fe = smf.ols('service_score ~ branch_id_cat + after_switch + treated_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Robustness Check: Branch Fixed Effects Model ===")
print(model_fe.summary())

# Extract key coefficients
print("\n=== Summary of Key Estimates ===")
print(f"Simple DiD:                    {smf.ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']}).params['treated_post']:.3f}")
print(f"DiD with controls:             {smf.ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']}).params['treated_post']:.3f}")
