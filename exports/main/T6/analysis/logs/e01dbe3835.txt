import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*treated + beta2*post + beta3*(treated*post) + error
# beta3 is our DiD estimate

df['treated_post'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()
print("Simple DiD Regression:")
print(model.summary())

# Now with branch fixed effects and week fixed effects (more robust)
df['branch_id'] = df['branch_id'].astype(str)
df['week'] = df['week'].astype(str)

model_fe = ols('service_score ~ switched_branch + after_switch + treated_post + C(branch_id) + C(week)', data=df).fit()
print("\n\nDiD Regression with Branch and Week Fixed Effects:")
print(model_fe.summary())

# Extract the key coefficient
did_coef = model_fe.params['treated_post']
did_se = model_fe.bse['treated_post']
did_pvalue = model_fe.pvalues['treated_post']
print(f"\n\nKey Result: DiD coefficient = {did_coef:.3f}, SE = {did_se:.3f}, p-value = {did_pvalue:.3f}")
