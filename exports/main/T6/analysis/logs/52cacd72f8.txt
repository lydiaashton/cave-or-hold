import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch']  # 1 if branch switched
df['post'] = df['after_switch']  # 1 if after week 15
df['DiD'] = df['treated'] * df['post']  # Interaction term

# Basic DiD calculation - average scores by group and period
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES: BASIC CALCULATION")
print("=" * 60)

# Before period (weeks 6-14)
before = df[df['post'] == 0]
after = df[df['post'] == 1]

# Control group (not switched)
control_before = before[before['treated'] == 0]['service_score'].mean()
control_after = after[after['treated'] == 0]['service_score'].mean()

# Treatment group (switched)
treat_before = before[before['treated'] == 1]['service_score'].mean()
treat_after = after[after['treated'] == 1]['service_score'].mean()

print(f"\nControl Group (did not switch):")
print(f"  Before (weeks 6-14): {control_before:.2f}")
print(f"  After (weeks 15-22): {control_after:.2f}")
print(f"  Change: {control_after - control_before:.2f}")

print(f"\nTreatment Group (switched):")
print(f"  Before (weeks 6-14): {treat_before:.2f}")
print(f"  After (weeks 15-22): {treat_after:.2f}")
print(f"  Change: {treat_after - treat_before:.2f}")

# DiD estimate
did_estimate = (treat_after - treat_before) - (control_after - control_before)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")
print(f"  (Treatment change) - (Control change) = {treat_after - treat_before:.2f} - {control_after - control_before:.2f}")

# Now run regression with standard errors
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD regression
model = ols('service_score ~ treated + post + DiD', data=df).fit()
print("\nBasic DiD Regression (robust SE):")
print(model.summary())

# Add branch fixed effects
print("\n" + "=" * 60)
print("WITH BRANCH FIXED EFFECTS")
print("=" * 60)

# Create branch dummies
df_with_fe = df.copy()
branch_dummies = pd.get_dummies(df_with_fe['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df_with_fe, branch_dummies], axis=1)

# Build formula with branch FEs
fe_cols = [col for col in df_with_fe.columns if col.startswith('branch_')]
formula = 'service_score ~ treated + post + DiD + ' + ' + '.join(fe_cols)

model_fe = ols(formula, data=df_with_fe).fit(cov_type='HC3')
print(f"\nDiD with Branch Fixed Effects (clustered SE by branch):")
print(f"DiD coefficient: {model_fe.params['DiD']:.4f}")
print(f"Std Error: {model_fe.bse['DiD']:.4f}")
print(f"t-statistic: {model_fe.tvalues['DiD']:.4f}")
print(f"p-value: {model_fe.pvalues['DiD']:.4f}")
print(f"95% CI: [{model_fe.conf_int().loc['DiD'][0]:.4f}, {model_fe.conf_int().loc['DiD'][1]:.4f}]")

# Check parallel trends - plot averages over time
print("\n" + "=" * 60)
print("PARALLEL TRENDS CHECK")
print("=" * 60)

# Average service score by week and group
weekly_avg = df.groupby(['week', 'treated'])['service_score'].mean().unstack()
weekly_avg.columns = ['Control', 'Treatment']
weekly_avg['Diff'] = weekly_avg['Treatment'] - weekly_avg['Control']

print("\nWeekly averages by group:")
print(weekly_avg.round(2))

# Pre-treatment trend (weeks 6-14)
pre_trend = weekly_avg.loc[6:14, 'Diff']
print(f"\nPre-treatment trend (Treatment - Control):")
print(pre_trend.round(2))
print(f"Average pre-treatment difference: {pre_trend.mean():.2f}")
print(f"Pre-treatment trend slope: {pre_trend.diff().mean():.2f} per week")
