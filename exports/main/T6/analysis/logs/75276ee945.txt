import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = β0 + β1*switched_branch + β2*after_switch + β3*(switched_branch * after_switch) + ε
# β3 is the DiD estimator

df['treatment_interaction'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment_interaction', data=df).fit()
print("Simple DiD Regression:")
print(model.summary())

# Now with branch fixed effects (more robust)
# This controls for time-invariant branch characteristics
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df_with_dummies['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Build formula with branch fixed effects
branch_cols = [col for col in df_with_dummies.columns if col.startswith('branch_')]
formula = 'service_score ~ switched_branch + after_switch + treatment_interaction + ' + ' + '.join(branch_cols)

model_fe = ols(formula, data=df_with_dummies).fit()
print("\n\nDiD Regression with Branch Fixed Effects:")
print(f"Coefficient on treatment_interaction: {model_fe.params['treatment_interaction']:.3f}")
print(f"Standard Error: {model_fe.bse['treatment_interaction']:.3f}")
print(f"P-value: {model_fe.pvalues['treatment_interaction']:.4f}")
print(f"95% CI: [{model_fe.conf_int().loc['treatment_interaction', 0]:.3f}, {model_fe.conf_int().loc['treatment_interaction', 1]:.3f}]")
