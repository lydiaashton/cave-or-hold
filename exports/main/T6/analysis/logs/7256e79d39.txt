import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check treatment/control group sizes
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Create treatment interaction term (switched_branch * after_switch)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate average service scores by group and period
print("\n\nAverage service scores by group and period:")
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='after_switch', 
                       aggfunc='mean')
print(pivot)

# Calculate the DiD manually
control_pre = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_post = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treatment_pre = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_post = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print("\n\nDetailed breakdown:")
print(f"Control (no switch) - Pre: {control_pre:.2f}, Post: {control_post:.2f}, Change: {control_post-control_pre:.2f}")
print(f"Treatment (switched) - Pre: {treatment_pre:.2f}, Post: {treatment_post:.2f}, Change: {treatment_post-treatment_pre:.2f}")
print(f"\nDifference-in-Differences: {(treatment_post-treatment_pre) - (control_post-control_pre):.2f}")

# Run DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\n\nDiD Regression Results:")
print(model.summary())

# Also run with cluster-robust standard errors by branch
model_cluster = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with Cluster-Robust SE (by branch):")
print(model_cluster.summary())

print("\n\nDiD Regression with controls (clustered SE):")

# Check parallel trends more formally - interaction of week with switched_branch for pre-period only
df_pre = df[df['week'] <= 14].copy()
df_pre['week_centered'] = df_pre['week'] - df_pre['week'].mean()

import pandas as pd
import matplotlib.pyplot as plt
import numpy as np

df = pd.read_csv('branches.csv')

# Calculate weekly averages by group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg['group'] = weekly_avg['switched_branch'].map({0: 'Control (No Switch)', 1: 'Treatment (Switched)'})

# Create the plot
fig, ax = plt.subplots(figsize=(10, 6))

for group_name, group_data in weekly_avg.groupby('group'):
    ax.plot(group_data['week'], group_data['service_score'], 
            marker='o', linewidth=2, markersize=6, label=group_name)

# Add vertical line at switch point
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=1.5, label='Switch implemented')
ax.axvspan(15, 22, alpha=0.1, color='green', label='Post-switch period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Scores by Branch Group Over Time', fontsize=14, fontweight='bold')
ax.legend(loc='best')
ax.set_xticks(range(6, 23))
ax.grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("Plot saved as service_scores_trend.png")

# Also calculate confidence intervals for the treatment effect
from statsmodels.stats.api import get_var_cov
print("\n\nTreatment effect estimate with 95% CI:")
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
effect = model.params['treatment']
se = model.bse['treatment']
ci_lower = effect - 1.96 * se
ci_upper = effect + 1.96 * se
print(f"Effect: {effect:.2f} points")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {model.pvalues['treatment']:.3f}")
