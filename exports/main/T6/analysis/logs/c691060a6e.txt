import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated_after'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch and week fixed effects
# This controls for time-invariant branch characteristics and common time trends
print("=== DiD Regression with Branch and Week Fixed Effects ===")
model = ols('service_score ~ switched_branch + after_switch + treated_after + C(branch_id) + C(week)', data=df).fit()
print(model.summary().tables[1])

# Also run a simpler model with just the key variables
print("\n=== Simple DiD Regression ===")
simple_model = ols('service_score ~ switched_branch + after_switch + treated_after', data=df).fit()
print(simple_model.summary().tables[1])

# Model with controls (staffing and visits)
print("\n=== DiD with Controls (no FE) ===")
control_model = ols('service_score ~ switched_branch + after_switch + treated_after + staffing_index + visits_index', data=df).fit()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treated_after'] = df['switched_branch'] * df['after_switch']

print("="*70)
print("RECONSIDERING THE ANALYSIS")
print("="*70)

# 1. Simple DiD (no trend controls)
simple = ols('service_score ~ switched_branch + after_switch + treated_after', data=df).fit()
print("\n1. SIMPLE DiD (no trend controls):")
print(f"   Treatment effect: +{simple.params['treated_after']:.2f} points")
print(f"   95% CI: [{simple.conf_int().loc['treated_after'][0]:.2f}, {simple.conf_int().loc['treated_after'][1]:.2f}]")
print(f"   P-value: {simple.pvalues['treated_after']:.4f}")

# 2. DiD with linear time trends
trend_model = ols('service_score ~ switched_branch + after_switch + treated_after + week + switched_branch:week + after_switch:week', data=df).fit()
print("\n2. DiD WITH LINEAR TREND CONTROLS:")
print(f"   Treatment effect: {trend_model.params['treated_after']:.2f} points")
print(f"   95% CI: [{trend_model.conf_int().loc['treated_after'][0]:.2f}, {trend_model.conf_int().loc['treated_after'][1]:.2f}]")
print(f"   P-value: {trend_model.pvalues['treated_after']:.4f}")

# 3. Event study - discontinuity at switch
print("\n3. DISCONTINUITY AT SWITCH POINT:")
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
gap = weekly[1] - weekly[0]
print(f"   Gap week 13: {gap[13]:.2f}")
print(f"   Gap week 14: {gap[14]:.2f}")
print(f"   Gap week 15: {gap[15]:.2f}")
print(f"   Gap week 16: {gap[16]:.2f}")
print(f"   Change from week 14 to 15: {gap[15] - gap[14]:.2f}")

# 4. Pre vs Post average gap
print("\n4. AVERAGE GAP COMPARISON:")
print(f"   Average pre-switch gap (weeks 6-14): {gap[6:15].mean():.2f}")
print(f"   Average post-switch gap (weeks 15-22): {gap[15:23].mean():.2f}")
print(f"   Difference: {gap[15:23].mean() - gap[6:15].mean():.2f}")

# 5. Within-branch changes
print("\n5. WITHIN-BRANCH CHANGES:")
branch_sum = df.groupby(['branch_id', 'switched_branch']).apply(
    lambda x: pd.Series({
        'pre': x[x['after_switch']==0]['service_score'].mean(),
        'post': x[x['after_switch']==1]['service_score'].mean(),
        'change': x[x['after_switch']==1]['service_score'].mean() - x[x['after_switch']==0]['service_score'].mean()
    })
).reset_index()

ctrl_change = branch_sum[branch_sum['switched_branch']==0]['change'].mean()
treat_change = branch_sum[branch_sum['switched_branch']==1]['change'].mean()
print(f"   Control avg change: +{ctrl_change:.2f}")
print(f"   Treatment avg change: +{treat_change:.2f}")
print(f"   Difference: +{treat_change - ctrl_change:.2f}")

# 6. Key question: Is the pre-trend difference large enough to explain the post difference?
print("\n" + "="*70)
print("CRITICAL ASSESSMENT")
print("="*70)
pre_trend_diff = 0.346  # points per week faster for treated
weeks_post = 8  # weeks 15-22
expected_gap_increase = pre_trend_diff * weeks_post / 2  # average over post period
print(f"\nIf pre-trend continued unchanged:")
print(f"   Pre-trend difference: {pre_trend_diff:.2f} points/week")
print(f"   Expected additional gap over 8 weeks: ~{expected_gap_increase:.1f} to {pre_trend_diff * weeks_post:.1f} points")
print(f"   Observed gap increase: {gap[15:23].mean() - gap[6:15].mean():.2f} points")

print("\n" + "="*70)
print("HONEST CONCLUSION")
print("="*70)
print("""
The key question: Can pre-existing trends explain the observed difference?

Arguments for NO EFFECT:
  - Pre-trends were significantly different (treated improving faster)
  - No discontinuity at the switch point (week 14→15)
  - Trend-adjusted model shows no significant effect

Arguments for POSSIBLE EFFECT:
  - Simple DiD shows +1.9 points (highly significant)
  - Within-branch changes show +1.9 points difference (p<0.0001)
  - Post-switch gap is larger than pre-switch gap

VERDICT: The pre-existing trend differences create substantial uncertainty.
         We CANNOT confidently say the procedure improved scores.
         The evidence is INCONCLUSIVE, not definitively negative.
""")
