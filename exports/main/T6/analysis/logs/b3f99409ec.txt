# Load the data
df = pd.read_csv('branches.csv')

# Event study: estimate effect for each week relative to switch
# Create relative week variable (week 15 = 0, week 14 = -1, etc.)
df['relative_week'] = df['week'] - 15

# Create dummies for each relative week (excluding week -1 as reference)
# We'll estimate the effect for each week relative to the baseline

# First, let's create a cleaner event study visualization
# Calculate average service score by group and relative week
event_data = df.copy()
event_data['relative_week'] = event_data['week'] - 15

# Group by switched status and relative week
event_avg = event_data.groupby(['switched_branch', 'relative_week'])['service_score'].agg(['mean', 'std', 'count']).reset_index()
event_avg['se'] = event_avg['std'] / np.sqrt(event_avg['count'])

# Calculate the difference between treatment and control for each relative week
event_wide = event_avg.pivot(index='relative_week', columns='switched_branch', values=['mean', 'se'])
event_wide.columns = ['_'.join(col).strip() for col in event_wide.columns.values]
event_wide = event_wide.reset_index()

event_wide['diff'] = event_wide['mean_1'] - event_wide['mean_0']
event_wide['diff_se'] = np.sqrt(event_wide['se_1']**2 + event_wide['se_0']**2)
event_wide['diff_ci_lower'] = event_wide['diff'] - 1.96 * event_wide['diff_se']

# Build formula with all branch dummies

formula = f"service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index + {' + '.join(branch_cols)}"

# This might be computationally intensive, so let's use a simpler approach with statsmodels
# Using PanelOLS or just absorbing fixed effects

# Alternative: use linearmodels for fixed effects
try:
    from linearmodels.panel import PanelOLS
    
    # Set up panel data
    df_panel = df.copy()
    df_panel = df_panel.set_index(['branch_id', 'week'])
    
    # DiD with branch and week fixed effects
