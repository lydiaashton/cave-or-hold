import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create visualization

# More detailed pre-trends analysis

pre_df = df[df['week'] < 15].copy()

# Test if there's any systematic difference in trends before the switch
# Week effects interacted with treatment group

pre_model = ols('service_score ~ C(branch_id) + week_centered + week_centered:switched_branch', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})

        'change': x[x['after_switch']==1]['service_score'].mean() - x[x['after_switch']==0]['service_score'].mean()

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
from scipy import stats

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main model
model_fe = ols('service_score ~ C(branch_id) + after_switch + treatment + staffing_index + visits_index', 
               data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

effect = model_fe.params['treatment']
se = model_fe.bse['treatment']
ci_low = effect - 1.96 * se
ci_high = effect + 1.96 * se

# Calculate probability that effect is positive (one-sided p-value)
t_stat = effect / se
prob_positive = stats.norm.cdf(t_stat)

print("=== Reconsidering the Evidence ===")
print(f"Point estimate: {effect:.2f} points")
print(f"Standard error: {se:.2f}")
print(f"95% CI: [{ci_low:.2f}, {ci_high:.2f}]")
print(f"Two-sided p-value: {model_fe.pvalues['treatment']:.3f}")
print(f"One-sided p-value (effect > 0): {1 - stats.norm.cdf(t_stat):.3f}")
print(f"Probability effect is positive: {prob_positive*100:.1f}%")

# What portion of the CI is positive?
if ci_low < 0:
    portion_positive = ci_high / (ci_high - ci_low)
else:
    portion_positive = 1.0
print(f"\nPortion of 95% CI that is positive: {portion_positive*100:.0f}%")

# Effect size context
baseline = df['service_score'].mean()
print(f"\nEffect as % of average score: {(effect/baseline)*100:.1f}%")
print(f"Effect relative to score SD: {effect/df['service_score'].std():.2f} standard deviations")

# How many branches would need to show improvement for this to be meaningful?
branch_summary = df.groupby(['branch_id', 'switched_branch']).apply(
    lambda x: x[x['after_switch']==1]['service_score'].mean() - x[x['after_switch']==0]['service_score'].mean()
).reset_index()
branch_summary.columns = ['branch_id', 'switched_branch', 'change']

treatment_improved = branch_summary[(branch_summary['switched_branch']==1) & (branch_summary['change']>0)]
print(f"\nTreatment branches that improved: {len(treatment_improved)} / {branch_summary[branch_summary['switched_branch']==1].shape[0]}")
