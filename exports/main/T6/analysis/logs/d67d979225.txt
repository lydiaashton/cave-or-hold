print("\n")

# Regression coefficient interpretation
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
did_coef = model.params['switched_branch:after_switch']
print(f"DiD Coefficient (interaction term): {did_coef:+.3f}")
print(f"P-value: {model.pvalues['switched_branch:after_switch']:.4f}")
print()

# Interpretation
print("INTERPRETATION:")
print("A POSITIVE DiD coefficient means the treatment group (new procedure)")
print("improved MORE (or declined LESS) than the control group (old procedure).")
print()
print(f"Since the coefficient is {did_coef:+.3f} (positive and significant),")
print("the NEW PROCEDURE had a POSITIVE effect on service scores.")
print()
