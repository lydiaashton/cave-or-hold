import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Create visualization
fig, ax = plt.subplots(figsize=(10, 6))

# Plot weekly averages
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
for switched in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == switched]
    label = 'Switched Branches' if switched == 1 else 'Non-Switched Branches'
    ax.plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2, markersize=6)

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')
ax.axvspan(15, 22, alpha=0.1, color='green', label='Post-Switch Period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Scores: Switched vs Non-Switched Branches', fontsize=14, fontweight='bold')
ax.legend(loc='upper left')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("Plot saved as service_scores_trend.png")

# Run regression with branch fixed effects
df['branch_id'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id'].astype(int)})
print("\n=== DiD with Branch Fixed Effects (Clustered SE) ===")
print(model_fe.summary())

# Extract treatment effect
treatment_coef = model_fe.params['treatment']
treatment_pval = model_fe.pvalues['treatment']
treatment_se = model_fe.bse['treatment']
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se

print(f"\nTreatment Effect: {treatment_coef:.2f} points")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {treatment_pval:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main DiD model with clustered SE
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pval = model.pvalues['treatment']
treatment_ci_lower = treatment_coef - 1.96 * treatment_se
treatment_ci_upper = treatment_coef + 1.96 * treatment_se

print("=== Treatment Effect Summary ===")
print(f"Point estimate: {treatment_coef:.2f} points")
print(f"Standard error: {treatment_se:.2f}")
print(f"95% CI: [{treatment_ci_lower:.2f}, {treatment_ci_upper:.2f}]")
print(f"P-value: {treatment_pval:.3f}")
print(f"T-statistic: {treatment_coef/treatment_se:.2f}")

print("\n=== Interpretation Thresholds ===")
print(f"Significant at 0.10 level? {treatment_pval < 0.10}")
print(f"Significant at 0.05 level? {treatment_pval < 0.05}")
print(f"CI excludes zero? {treatment_ci_lower > 0 or treatment_ci_upper < 0}")

# Power consideration
print("\n=== Sample Size Context ===")
print(f"Total observations: {len(df)}")
print(f"Branches: {df['branch_id'].nunique()}")
print(f"Post-switch weeks per branch: 8")
print(f"Switched branches: {df['switched_branch'].sum()}")
print(f"Control branches: {len(df) - df['switched_branch'].sum()}")

# Effect size in context
baseline = df['service_score'].mean()
print(f"\nBaseline mean: {baseline:.2f}")
print(f"Effect as % of baseline: {treatment_coef/baseline*100:.1f}%")
print(f"Effect as % of SD: {treatment_coef/df['service_score'].std()*100:.1f}% of one standard deviation")
