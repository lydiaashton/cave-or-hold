import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Check balance between treatment and control
print("=== Treatment vs Control Group Sizes ===")
treatment_branches = df[df['switched_branch'] == 1]['branch_id'].nunique()
control_branches = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"Treatment branches (switched): {treatment_branches}")
print(f"Control branches (did not switch): {control_branches}")
print(f"Total branches: {treatment_branches + control_branches}")

# Check pre-trends: average service score by week for treatment vs control
print("\n=== Pre-Trends Check (Weeks 6-14, before switch) ===")
pre_switch = df[df['week'] < 15].copy()
pre_trends = pre_switch.groupby(['week', 'switched_branch'])['service_score'].agg(['mean', 'std', 'count']).unstack('switched_branch')
print(pre_trends)

# Overall means before and after for each group
print("\n=== Mean Service Scores by Group and Period ===")
df['period'] = df['after_switch'].map({0: 'Before', 1: 'After'})
summary = df.groupby(['switched_branch', 'period'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Simple difference-in-differences calculation
print("\n=== Simple DiD Calculation ===")
# Treatment before
treat_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
# Treatment after
treat_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
# Control before
ctrl_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
# Control after
ctrl_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"Treatment (switched) - Before: {treat_before:.3f}, After: {treat_after:.3f}, Change: {treat_after - treat_before:.3f}")
print(f"Control (not switched) - Before: {ctrl_before:.3f}, After: {ctrl_after:.3f}, Change: {ctrl_after - ctrl_before:.3f}")
did_estimate = (treat_after - treat_before) - (ctrl_after - ctrl_before)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run DiD regression with branch fixed effects
# The key coefficient is the interaction: switched_branch * after_switch
print("=== Difference-in-Differences Regression ===")
print("\nModel 1: Basic DiD (no fixed effects)")
model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print(model1.summary().tables[1])

print("\n" + "="*60)
print("Model 2: DiD with Branch Fixed Effects")
# Add branch fixed effects
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df_with_dummies['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_')]
formula = 'service_score ~ switched_branch * after_switch + ' + ' + '.join(branch_cols)
model2 = ols(formula, data=df_with_dummies).fit()

# Print just the key coefficients
print("\nKey coefficients:")
print(model2.params[['switched_branch', 'after_switch', 'switched_branch:after_switch']])
print("\nKey coefficient statistics:")
print(model2.bse[['switched_branch', 'after_switch', 'switched_branch:after_switch']])
print(model2.pvalues[['switched_branch', 'after_switch', 'switched_branch:after_switch']])

# Get the DiD estimate with confidence interval
did_coef = model2.params['switched_branch:after_switch']
did_se = model2.bse['switched_branch:after_switch']
did_pvalue = model2.pvalues['switched_branch:after_switch']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se

print(f"\n=== DiD Estimate Summary ===")
print(f"Coefficient: {did_coef:.3f}")
print(f"Standard Error: {did_se:.3f}")
print(f"95% CI: [{ci_lower:.3f}, {ci_upper:.3f}]")
print(f"P-value: {did_pvalue:.6f}")
print(f"Statistically significant at 5% level: {did_pvalue < 0.05}")

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Model with week fixed effects (more manageable than branch FEs)
print("=== Model with Week Fixed Effects ===")

df_week_fe = pd.concat([df, week_dummies], axis=1)

week_cols = [c for c in df_week_fe.columns if c.startswith('week_')]
formula = 'service_score ~ switched_branch * after_switch + ' + ' + '.join(week_cols)

print(model_week_fe.pvalues[['switched_branch', 'after_switch', 'switched_branch:after_switch']])

# Cluster-robust standard errors by branch

cov_cluster = model_week_fe.get_robustcov_results(cov_type='cluster', groups=df_week_fe['branch_id'])

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators with safe names
df['relative_week'] = df['week'] - 14  # Week 14 becomes 0, week 15 becomes 1, etc.

# Create indicators for each relative week (excluding week 0 as reference)
event_study_vars = []
for rw in range(-8, 9):
    if rw != 0:  # Skip reference week
        # Use safe variable names
        sign = 'm' if rw < 0 else 'p'
        abs_rw = abs(rw)
        var_name = f'rel_week_{sign}{abs_rw}'
        df[var_name] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(var_name)

# Run event study regression
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_study_vars)
model_event = ols(formula, data=df).fit()

print("=== Event Study Results ===")
print("Coefficients for treatment group in each relative week (reference: week 14)")
print("\nPre-switch weeks (should be near zero if parallel trends hold):")
for rw in range(-8, 0):
    sign = 'm' if rw < 0 else 'p'
    abs_rw = abs(rw)
    var_name = f'rel_week_{sign}{abs_rw}'
    coef = model_event.params[var_name]
    se = model_event.bse[var_name]
    pval = model_event.pvalues[var_name]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Week {14+rw} (rel {rw}): {coef:.2f} (SE: {se:.2f}){sig}")

print("\nPost-switch weeks (should show positive effect if intervention worked):")
for rw in range(1, 9):

    abs_rw = abs(rw)
    var_name = f'rel_week_{sign}{abs_rw}'
    coef = model_event.params[var_name]
    se = model_event.bse[var_name]
    pval = model_event.pvalues[var_name]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Week {14+rw} (rel {rw}): {coef:.2f} (SE: {se:.2f}){sig}")

# Summary statistics for the event study
print("\n=== Event Study Summary ===")
pre_coefs = []
for rw in range(-8, 0):
    sign = 'm' if rw < 0 else 'p'
    abs_rw = abs(rw)
    var_name = f'rel_week_{sign}{abs_rw}'
    pre_coefs.append(model_event.params[var_name])

post_coefs = []
for rw in range(1, 9):
    sign = 'm' if rw < 0 else 'p'
    abs_rw = abs(rw)
    var_name = f'rel_week_{sign}{abs_rw}'
    post_coefs.append(model_event.params[var_name])

print(f"Average pre-switch coefficient: {np.mean(pre_coefs):.2f}")
