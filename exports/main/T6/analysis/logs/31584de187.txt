import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Calculate group means
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES ANALYSIS")
print("=" * 60)

# Before period (after_switch=0)
before_treatment = df[(df['after_switch'] == 0) & (df['switched_branch'] == 1)]['service_score'].mean()
before_control = df[(df['after_switch'] == 0) & (df['switched_branch'] == 0)]['service_score'].mean()

# After period (after_switch=1)
after_treatment = df[(df['after_switch'] == 1) & (df['switched_branch'] == 1)]['service_score'].mean()
after_control = df[(df['after_switch'] == 1) & (df['switched_branch'] == 0)]['service_score'].mean()

print("\nAverage Service Scores:")
print(f"  Treatment group (switched branches):")
print(f"    Before (weeks 6-14): {before_treatment:.2f}")
print(f"    After (weeks 15-22): {after_treatment:.2f}")
print(f"    Change: {after_treatment - before_treatment:.2f}")
print(f"  Control group (non-switched branches):")
print(f"    Before (weeks 6-14): {before_control:.2f}")
print(f"    After (weeks 15-22): {after_control:.2f}")
print(f"    Change: {after_control - before_control:.2f}")

# DiD estimate
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\n  DiD Estimate (treatment effect): {did_estimate:.2f}")

# Run regression with standard errors
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD regression
model = ols('service_score ~ treated + post + DiD', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nBasic DiD Regression (clustered by branch):")
print(model.summary().tables[1])

# Add controls
model_controls = ols('service_score ~ treated + post + DiD + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nDiD Regression with Controls (clustered by branch):")
print(model_controls.summary().tables[1])

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Run regression with robust standard errors
model = ols('service_score ~ treated + post + DiD', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Key regression results:")
print(f"DiD Coefficient: {model.params['DiD']:.3f}")
print(f"Standard Error: {model.bse['DiD']:.3f}")
print(f"t-statistic: {model.tvalues['DiD']:.3f}")
print(f"P-value: {model.pvalues['DiD']:.3f}")
print(f"95% CI: [{model.conf_int().loc['DiD', 0]:.3f}, {model.conf_int().loc['DiD', 1]:.3f}]")

# What would we need for significance?
print("\nFor context:")
print(f"Sample size: {len(df)} observations, {df['branch_id'].nunique()} branches")
print(f"Treatment branches: {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"Control branches: {df[df['switched_branch']==0]['branch_id'].nunique()}")

# Power analysis - what effect could we detect?
n_treatment = df[df['switched_branch']==1]['branch_id'].nunique()
n_control = df[df['switched_branch']==0]['branch_id'].nunique()
sd = df['service_score'].std()

print(f"\nService score SD: {sd:.2f}")
print(f"With this sample size, minimum detectable effect (80% power, alpha=0.05): ~{0.8 * sd / np.sqrt(n_treatment):.2f} points")
