import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Calculate means by group and period
print("=" * 60)
print("SERVICE SCORE AVERAGES BY GROUP AND PERIOD")
print("=" * 60)

# Before switch (after_switch=0)
before_switched = df[(df['after_switch'] == 0) & (df['switched_branch'] == 1)]['service_score'].mean()
before_control = df[(df['after_switch'] == 0) & (df['switched_branch'] == 0)]['service_score'].mean()

# After switch (after_switch=1)
after_switched = df[(df['after_switch'] == 1) & (df['switched_branch'] == 1)]['service_score'].mean()
after_control = df[(df['after_switch'] == 1) & (df['switched_branch'] == 0)]['service_score'].mean()

print(f"\nControl Group (didn't switch):")
print(f"  Before (weeks 6-14): {before_control:.2f}")
print(f"  After (weeks 15-22): {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")

print(f"\nTreatment Group (switched):")
print(f"  Before (weeks 6-14): {before_switched:.2f}")
print(f"  After (weeks 15-22): {after_switched:.2f}")
print(f"  Change: {after_switched - before_switched:.2f}")

# DiD estimate
did_estimate = (after_switched - before_switched) - (after_control - before_control)
print(f"\n{'=' * 60}")
print(f"DIFFERENCE-IN-DIFFERENCES ESTIMATE: {did_estimate:.2f}")
print(f"{'=' * 60}")

# Run regression with standard errors clustered by branch
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()
print("\nBasic OLS (no clustering):")
print(model.summary().tables[1])

# Clustered standard errors by branch
model_clustered = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nWith cluster-robust SE (by branch):")
print(model_clustered.summary().tables[1])

pre_data['week_centered'] = pre_data['week'] - 10

print("\nPre-period trend (weeks 6-14):")
print(f"  Intercept (at week 10): {pre_trend.params['Intercept']:.2f}")
print(f"  Weekly slope: {pre_trend.params['week_centered']:.2f} points/week")

    actual = weekly.loc[week, 'diff']
    weeks_from_10 = week - 10
    predicted = pre_trend.params['Intercept'] + pre_trend.params['week_centered'] * weeks_from_10
    print(f"  Week {week}: Predicted={predicted:.2f}, Actual={actual:.2f}, Difference={actual-predicted:.2f}")
