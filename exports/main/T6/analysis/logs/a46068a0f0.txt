import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Count branches by switch status
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate mean service scores by group and period
print("\nMean service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the difference-in-differences manually
before_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print(f"\nTreatment group (switched): Before={before_treatment:.2f}, After={after_treatment:.2f}, Change={after_treatment-before_treatment:.2f}")
print(f"Control group (not switched): Before={before_control:.2f}, After={after_control:.2f}, Change={after_control-before_control:.2f}")
print(f"\nDifference-in-Differences estimate: {(after_treatment - before_treatment) - (after_control - before_control):.2f}")

# Run DiD regression
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

model = ols('service_score ~ treatment + post + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression Results:")
print(model.summary())

import pandas as pd

pre_by_week = pre_data.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
print(pre_by_week)

# Calculate average pre-period scores by group
pre_treatment = pre_data[pre_data['switched_branch']==1]['service_score'].mean()
pre_control = pre_data[pre_data['switched_branch']==0]['service_score'].mean()
print(f"\nPre-period average - Treatment: {pre_treatment:.2f}, Control: {pre_control:.2f}, Difference: {pre_treatment - pre_control:.2f}")

# Run DiD with controls
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

model_with_controls = ols('service_score ~ treatment + post + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with controls:")
print(model_with_controls.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Branch-level analysis: calculate before/after change for each branch
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']

# Calculate branch-level before and after averages
before = df[df['after_switch']==0].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
before.columns = ['branch_id', 'switched_branch', 'score_before']

after = df[df['after_switch']==1].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
after.columns = ['branch_id', 'switched_branch', 'score_after']

branch_level = before.merge(after, on=['branch_id', 'switched_branch'])
branch_level['change'] = branch_level['score_after'] - branch_level['score_before']

print("Branch-level changes:")
print(branch_level.groupby('switched_branch')['change'].describe().round(2))

# Test if the average change differs between groups
treatment_change = branch_level[branch_level['switched_branch']==1]['change']
control_change = branch_level[branch_level['switched_branch']==0]['change']

from scipy import stats
t_stat, p_val = stats.ttest_ind(treatment_change, control_change)
print(f"\nTwo-sample t-test on branch-level changes:")
print(f"Treatment mean change: {treatment_change.mean():.2f}")
print(f"Control mean change: {control_change.mean():.2f}")
print(f"Difference: {treatment_change.mean() - control_change.mean():.2f}")
print(f"T-statistic: {t_stat:.2f}, p-value: {p_val:.4f}")

# Effect size (Cohen's d)
pooled_std = np.sqrt((treatment_change.var() + control_change.var()) / 2)
cohens_d = (treatment_change.mean() - control_change.mean()) / pooled_std
print(f"Cohen's d (effect size): {cohens_d:.2f}")

# Check distribution of changes
print("\nDistribution of changes by group:")
print("\nTreatment group (switched) changes:")
print(branch_level[branch_level['switched_branch']==1]['change'].describe().round(2))
print("\nControl group (not switched) changes:")
print(branch_level[branch_level['switched_branch']==0]['change'].describe().round(2))

# How many branches improved vs declined?
print("\nBranches with positive change:")
print(branch_level.groupby('switched_branch').apply(lambda x: (x['change'] > 0).sum()))
print("\nBranches with negative change:")
print(branch_level.groupby('switched_branch').apply(lambda x: (x['change'] < 0).sum()))
