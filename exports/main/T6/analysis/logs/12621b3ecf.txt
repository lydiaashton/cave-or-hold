import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Check how many branches switched vs didn't
branches = df[['branch_id', 'switched_branch']].drop_duplicates()
print("Branches that switched:", (branches['switched_branch'] == 1).sum())
print("Branches that didn't switch:", (branches['switched_branch'] == 0).sum())

# Calculate average service scores by group and period
print("\n=== Average Service Scores ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Difference-in-Differences calculation
# Before period (after_switch=0), After period (after_switch=1)
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n=== DiD Components ===")
print(f"Control group (didn't switch): Before={before_control:.2f}, After={after_control:.2f}, Change={after_control - before_control:.2f}")
print(f"Treatment group (switched): Before={before_treatment:.2f}, After={after_treatment:.2f}, Change={after_treatment - before_treatment:.2f}")
print(f"\nDifference-in-Differences estimate: {(after_treatment - before_treatment) - (after_control - before_control):.2f}")

# Run a regression for statistical significance
# DiD regression: service_score ~ switched_branch * after_switch
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("\n=== DiD Regression Results ===")
print(model.summary())

# Also check with robust standard errors (clustered by branch)
model_clustered = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Regression with Clustered SE (by branch) ===")
print(model_clustered.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check pre-trends: compare service scores by week for both groups before week 15
print("=== Pre-Trend Check (Weeks 6-14) ===")
pre_trend = df[df['week'] < 15].groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(pre_trend.round(2))

# Check post-trends by week
print("\n=== Post-Trend Check (Weeks 15-22) ===")
post_trend = df[df['week'] >= 15].groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(post_trend.round(2))

# Create a week-relative variable for event study
df['weeks_since_switch'] = df['week'] - 15

# Event study regression (relative to week 14 as baseline)
# Create week dummies
df['week_cat'] = df['week'].astype(str)
df['baseline_week'] = (df['week'] == 14).astype(int)

# Event study: interact each week with treatment
event_study_vars = []
for w in range(6, 23):
    if w != 14:  # baseline
        df[f'week_{w}'] = (df['week'] == w).astype(int)
        df[f'treated_week_{w}'] = ((df['week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(f'treated_week_{w}')

# Run event study regression
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'week_{w}' for w in range(6, 23) if w != 14]) + ' + ' + ' + '.join(event_study_vars)
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n=== Event Study Coefficients (Treatment Effect by Week) ===")
for w in range(6, 23):
    if w < 15:
        coef = event_model.params.get(f'treated_week_{w}', 0)
        se = event_model.bse.get(f'treated_week_{w}', 0)
        print(f"Week {w} (pre): coef={coef:.2f}, SE={se:.2f}")
    else:
        coef = event_model.params.get(f'treated_week_{w}', 0)
        se = event_model.bse.get(f'treated_week_{w}', 0)
        print(f"Week {w} (post): coef={coef:.2f}, SE={se:.2f}")

# Check if staffing or visits explain the effect
print("\n=== Regression with Controls ===")
model_controls = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_controls.summary().tables[1])

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create event study plot data
event_data = []
for w in range(6, 23):
    if w != 14:  # baseline
        # Calculate the coefficient from our event study
        before_control = df[(df['switched_branch'] == 0) & (df['week'] == 14)]['service_score'].mean()
        week_control = df[(df['switched_branch'] == 0) & (df['week'] == w)]['service_score'].mean()
        week_treatment = df[(df['switched_branch'] == 1) & (df['week'] == w)]['service_score'].mean()
        
        # DiD for this week relative to week 14
        did = (week_treatment - week_control) - (df[(df['switched_branch'] == 1) & (df['week'] == 14)]['service_score'].mean() - before_control)
        event_data.append({'week': w, 'effect': did, 'period': 'pre' if w < 15 else 'post'})

event_df = pd.DataFrame(event_data)
event_df.loc[event_df['week'] == 14, 'effect'] = 0  # baseline

# Create the plot
fig, ax = plt.subplots(figsize=(10, 6))

# Plot pre and post periods
pre_df = event_df[event_df['period'] == 'pre']
post_df = event_df[event_df['period'] == 'post']

ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (Week 15)')
ax.axhline(y=0, color='gray', linestyle='-', linewidth=0.5)

ax.scatter(pre_df['week'], pre_df['effect'], color='blue', s=100, label='Pre-switch', zorder=5)
ax.scatter(post_df['week'], post_df['effect'], color='green', s=100, label='Post-switch', zorder=5)

# Connect points with lines
ax.plot(pre_df['week'], pre_df['effect'], color='blue', alpha=0.5)
ax.plot(post_df['week'], post_df['effect'], color='green', alpha=0.5)

# Add baseline point
ax.scatter([14], [0], color='gray', s=100, zorder=5)

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Treatment Effect (Service Score Points)', fontsize=12)
ax.set_title('Event Study: Effect of New Procedure on Service Scores\n(Week 14 = Baseline)', fontsize=14)
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("Event study plot saved as event_study.png")

# Calculate confidence intervals for the main DiD estimate
from statsmodels.stats.weightstats import DescrStatsW

# Get the DiD estimate with CI
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score']
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score']
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score']
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score']

# Calculate changes
treatment_change = treatment_after.mean() - treatment_before.mean()
control_change = control_after.mean() - control_before.mean()
did_estimate = treatment_change - control_change

# Standard error (simplified, using the regression SE)
did_se = 0.28  # from clustered regression
did_ci_lower = did_estimate - 1.96 * did_se
did_ci_upper = did_estimate + 1.96 * did_se

print(f"\n=== Final DiD Estimate ===")
print(f"Point estimate: {did_estimate:.2f} points")
print(f"95% CI: [{did_ci_lower:.2f}, {did_ci_upper:.2f}]")
print(f"Standard error: {did_se:.2f}")

# Calculate practical significance
avg_score = df['service_score'].mean()
pct_improvement = (did_estimate / avg_score) * 100
print(f"\nAverage service score: {avg_score:.1f}")
