import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Basic DiD regression (no fixed effects)
print("=" * 60)
print("MODEL 1: Basic DiD Regression")
print("=" * 60)
model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print(model1.summary())

# DiD with controls
print("\n" + "=" * 60)
print("MODEL 2: DiD with staffing and visits controls")
print("=" * 60)
model2 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit()
print(model2.summary())

# DiD with branch fixed effects
print("\n" + "=" * 60)
print("MODEL 3: DiD with Branch Fixed Effects")
print("=" * 60)
df['branch_id_str'] = df['branch_id'].astype(str)
model3 = ols('service_score ~ switched_branch * after_switch + C(branch_id_str)', data=df).fit()
print(model3.summary())

# DiD with branch and week fixed effects
print("\n" + "=" * 60)
print("MODEL 4: DiD with Branch and Week Fixed Effects")
print("=" * 60)
df['week_str'] = df['week'].astype(str)
model4 = ols('service_score ~ switched_branch * after_switch + C(branch_id_str) + C(week_str)', data=df).fit()
print(model4.summary())
