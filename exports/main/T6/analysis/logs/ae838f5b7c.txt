import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create week dummies relative to switch (week 15 is the switch)
# Week 14 will be the reference period
df['relative_week'] = df['week'] - 14  # So week 14 = 0, week 15 = 1, etc.

# Create dummies for each week (excluding week 14 as reference)
week_dummies = pd.get_dummies(df['relative_week'], prefix='week')
df = pd.concat([df, week_dummies], axis=1)

# Build the event study regression
# We'll interact each week dummy with switched_branch
# Reference week is week 14 (relative_week = 0)
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'switched_branch:week_{w}' for w in range(-8, 9) if w != 0]) + ' + staffing_index + visits_index'

# Actually, let me do this more carefully
# Create interaction terms manually
for w in range(-8, 9):
    if w != 0:  # Skip reference week
        df[f'switched_x_week{w}'] = df['switched_branch'] * (df['relative_week'] == w).astype(int)

# Build formula with all interaction terms
interaction_terms = ' + '.join([f'switched_x_week{w}' for w in range(-8, 9) if w != 0])
formula = f'service_score ~ switched_branch + after_switch + {interaction_terms} + staffing_index + visits_index'

model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Regression Results:")
print("="*80)
# Print only the interaction terms
for name in model.params.index:
    if 'switched_x_week' in name:
        coef = model.params[name]
        se = model.bse[name]
        pval = model.pvalues[name]
        print(f"{name:20s}: {coef:7.3f} (SE: {se:.3f}, p={pval:.3f})")

import pandas as pd

df = pd.read_csv('branches.csv')

# Let's do a cleaner event study with branch and week fixed effects
# Create dummies for branch and week
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)

# Create interaction of switched_branch with each week (relative to week 14)
df['relative_week'] = df['week'] - 14
interaction_cols = []
for w in range(-8, 9):
    if w != 0:  # Skip reference week
        col_name = f'tx_w{w:+03d}'
        df[col_name] = df['switched_branch'] * (df['relative_week'] == w).astype(int)
        interaction_cols.append(col_name)

X_cols = interaction_cols + ['staffing_index', 'visits_index']
X = pd.concat([branch_dummies, week_dummies, df[X_cols]], axis=1)
X = sm.add_constant(X, has_constant='skip')  # Already have dummies
y = df['service_score']

model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
