model_fe = ols('service_score ~ after_switch + treatment + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Branch Fixed Effects Model ===")
print(model_fe.summary().tables[1])

# Check pre-trends more formally - interact treatment with pre-period weeks
df['pre_period'] = (df['week'] < 15).astype(int)
df['pre_trend'] = df['switched_branch'] * df['week'] * df['pre_period']

# Test if pre-trends differ
model_pretrend = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
                     data=df[df['week'] < 15]).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
