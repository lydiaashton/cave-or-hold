import pandas as pd

df = pd.read_csv('branches.csv')

print()

# Calculate the difference-in-differences manually

# Model 1: Basic DiD (no controls)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Model with controls - this is our main specification
model = ols('service_score ~ switched_branch + after_switch + treatment_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== MAIN RESULTS ===")
print(f"Treatment effect (treatment_post coefficient): {model.params['treatment_post']:.2f}")
print(f"Standard error: {model.bse['treatment_post']:.2f}")
print(f"95% CI: [{model.conf_int().loc['treatment_post'][0]:.2f}, {model.conf_int().loc['treatment_post'][1]:.2f}]")
print(f"P-value: {model.pvalues['treatment_post']:.4f}")
print()

# Check parallel trends - compare pre-period trends
print("=== PARALLEL TRENDS CHECK ===")
pre_data = df[df['after_switch'] == 0]
pre_weekly = pre_data.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
pre_weekly['diff'] = pre_weekly[1] - pre_weekly[0]
print("Pre-period weekly differences (treatment - control):")
print(pre_weekly[['diff']].round(2))
print()

# Test if pre-period trend is flat
pre_data = pre_data.copy()
pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()
trend_model = ols('service_score ~ switched_branch * week_centered', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print("Pre-period interaction test (should be non-significant for parallel trends):")
interaction_coef = trend_model.params['switched_branch:week_centered']
print(f"Interaction coefficient: {interaction_coef:.3f}, p-value: {trend_model.pvalues['switched_branch:week_centered']:.3f}")
print()

# Effect size relative to baseline
baseline = df[df['after_switch'] == 0]['service_score'].mean()
effect_size_pct = (model.params['treatment_post'] / baseline) * 100
print("=== EFFECT SIZE ===")
print(f"Baseline service score (pre-period avg): {baseline:.2f}")
print(f"Treatment effect: {model.params['treatment_post']:.2f} points")
print(f"Relative improvement: {effect_size_pct:.1f}%")
