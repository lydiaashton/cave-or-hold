import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate means for the 4 groups
print("=== Difference-in-Differences Summary ===\n")

# Before switch (after_switch = 0)
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()

# After switch (after_switch = 1)
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("Control Group (no switch):")
print(f"  Before: {before_control:.2f}")
print(f"  After:  {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")

print("\nTreatment Group (switched):")
print(f"  Before: {before_treatment:.2f}")
print(f"  After:  {after_treatment:.2f}")
print(f"  Change: {after_treatment - before_treatment:.2f}")

# DiD estimate
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

# Run regression for statistical significance
print("\n=== Regression Analysis ===\n")

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model.summary())

# Check branch counts
print("\n=== Branch Counts ===")
print(f"Total branches: {df['branch_id'].nunique()}")
print(f"Switched branches: {df[df['switched_branch'] == 1]['branch_id'].nunique()}")

# Compare switching vs non-switching branches before the switch
print("Before switch (weeks 6-14):")

# Test if they're significantly different before the switch
pre_switch_test = ols('service_score ~ switched_branch', data=before_data).fit(cov_type='cluster', cov_kwds={'groups': before_data['branch_id']})

before_data_copy = before_data.copy()

print(f"  Coefficient: {pre_trend_model.params['trend_interaction']:.3f} (p={pre_trend_model.pvalues['trend_interaction']:.3f})")

# Check if switching branches differ on other dimensions
