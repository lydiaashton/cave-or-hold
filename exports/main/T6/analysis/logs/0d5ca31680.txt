import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment_after'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with robust standard errors
# Model: service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch*after_switch) + error
model = ols('service_score ~ switched_branch + after_switch + treatment_after', data=df).fit(cov_type='HC3')

print("Difference-in-Differences Regression Results")
print("=" * 60)
print(model.summary())

# Extract key coefficient
did_coef = model.params['treatment_after']
did_se = model.bse['treatment_after']
did_pvalue = model.pvalues['treatment_after']
did_ci_low = did_coef - 1.96 * did_se
did_ci_high = did_coef + 1.96 * did_se

print(f"\n\nKey DiD Coefficient (treatment_after):")
print(f"  Estimate: {did_coef:.3f}")
print(f"  Std Error: {did_se:.3f}")
print(f"  95% CI: [{did_ci_low:.3f}, {did_ci_high:.3f}]")
print(f"  p-value: {did_pvalue:.4f}")
