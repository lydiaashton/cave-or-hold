print("Treatment branches (switched_branch=1):", branches.sum())

# Average service scores by group and period

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression
# Basic DiD model
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# Now with branch fixed effects (more robust)
# This controls for time-invariant branch characteristics
model_fe = ols('service_score ~ after_switch + treatment + C(branch_id)', data=df).fit()
print("\n=== DiD with Branch Fixed Effects ===")
print(model_fe.summary())

# With cluster-robust standard errors at branch level
from statsmodels.stats.contrast import ContrastResults

# Let's use a different approach for clustered SEs
import statsmodels.api as sm
X = sm.add_constant(df[['after_switch', 'treatment']])
# Add branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
X = pd.concat([X, branch_dummies], axis=1)
y = df['service_score']

model_cluster = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD with Branch FE and Clustered SEs ===")
print(model_cluster.summary())
print(f"\nTreatment effect (treatment coefficient): {model_cluster.params['treatment']:.3f}")
print(f"Standard error: {model_cluster.bse['treatment']:.3f}")
print(f"95% CI: [{model_cluster.conf_int().loc['treatment'][0]:.3f}, {model_cluster.conf_int().loc['treatment'][1]:.3f}]")

# Calculate effect for different groups
