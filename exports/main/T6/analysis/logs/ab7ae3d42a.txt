import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
switched_counts = df.groupby('branch_id')['switched_branch'].first().value_counts()
print("Branches by switch status:")
print(switched_counts)
print(f"\nTotal branches: {len(df.groupby('branch_id'))}")
print(f"Switched: {switched_counts.get(1, 0)}")
print(f"Did not switch: {switched_counts.get(0, 0)}")

# Average service scores by group and period
print("\n\nAverage service scores by group and period:")
grouped = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(grouped)

# Calculate the difference-in-differences manually
print("\n\nDifference-in-Differences Calculation:")
# Control group (didn't switch): before and after
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
control_diff = control_after - control_before

# Treatment group (switched): before and after
treat_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treat_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
treat_diff = treat_after - treat_before

did = treat_diff - control_diff

print(f"Control (no switch): Before={control_before:.2f}, After={control_after:.2f}, Change={control_diff:.2f}")
print(f"Treatment (switch): Before={treat_before:.2f}, After={treat_after:.2f}, Change={treat_diff:.2f}")
print(f"\nDifference-in-Differences estimate: {did:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*treatment + beta2*post + beta3*(treatment*post) + error
# beta3 is our DiD estimate

df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit()
print("Simple DiD Regression:")
print(model.summary())

# Now with branch fixed effects (more robust)
# This controls for time-invariant branch characteristics
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df_with_dummies['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Build formula with branch fixed effects
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch')]
formula = 'service_score ~ switched_branch + after_switch + treatment_post + ' + ' + '.join(branch_cols)

model_fe = ols(formula, data=df_with_dummies).fit()
print("\n\nDiD Regression with Branch Fixed Effects:")
print(model_fe.summary())

# Let's also check the parallel trends assumption - compare trends before the switch
print("\n\nParallel Trends Check - Pre-period trends by week:")
pre_period = df[df['week'] < 15].copy()
pre_trends = pre_period.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
