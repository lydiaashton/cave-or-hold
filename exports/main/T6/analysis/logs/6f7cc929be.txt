import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model 1: Basic DiD without controls
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Model 1: Basic DiD (no controls) ===")
print(model1.summary())

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\n=== Model 2: DiD with controls ===")
print(model2.summary())

# Model 3: With branch fixed effects (more robust)
# Create dummy variables for each branch
df_with_fe = df.copy()
branch_dummies = pd.get_dummies(df_with_fe['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df_with_fe, branch_dummies], axis=1)

# Build formula with branch fixed effects
fe_cols = [col for col in df_with_fe.columns if col.startswith('branch')]
fe_formula = 'service_score ~ after_switch + treatment + staffing_index + visits_index + ' + ' + '.join(fe_cols)
model3 = ols(fe_formula, data=df_with_fe).fit()
print("\n=== Model 3: DiD with Branch Fixed Effects ===")
# Just print the key coefficients
print(f"Treatment effect (treatment coefficient): {model3.params['treatment']:.3f}")
print(f"Standard error: {model3.bse['treatment']:.3f}")
print(f"95% CI: [{model3.params['treatment'] - 1.96*model3.bse['treatment']:.3f}, {model3.params['treatment'] + 1.96*model3.bse['treatment']:.3f}]")
print(f"P-value: {model3.pvalues['treatment']:.4f}")

# Cluster-robust standard errors by branch
from statsmodels.stats.sandwich_covariance import cov_cluster
model3_clustered = model3.get_robustcov_results(cov_type='cluster', groups=df_with_fe['branch_id'])
print(f"\nWith clustered SEs by branch:")
print(f"Treatment effect: {model3_clustered.params['treatment']:.3f}")
print(f"Clustered SE: {model3_clustered.bse['treatment']:.3f}")
print(f"95% CI: [{model3_clustered.params['treatment'] - 1.96*model3_clustered.bse['treatment']:.3f}, {model3_clustered.params['treatment'] + 1.96*model3_clustered.bse['treatment']:.3f}]")
print(f"P-value: {model3_clustered.pvalues['treatment']:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model with branch fixed effects using a different approach
# Use C() for categorical variables
model_fe = ols('service_score ~ C(branch_id) + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("=== Model with Branch Fixed Effects ===")
print(f"Treatment effect: {model_fe.params['treatment']:.3f}")
print(f"Standard error: {model_fe.bse['treatment']:.3f}")
print(f"95% CI: [{model_fe.params['treatment'] - 1.96*model_fe.bse['treatment']:.3f}, {model_fe.params['treatment'] + 1.96*model_fe.bse['treatment']:.3f}]")
print(f"P-value: {model_fe.pvalues['treatment']:.4f}")

# Try cluster-robust SEs
try:
    model_fe_clustered = model_fe.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
    print(f"\nWith clustered SEs by branch:")
    print(f"Treatment effect: {model_fe_clustered.params['treatment']:.3f}")
    print(f"Clustered SE: {model_fe_clustered.bse['treatment']:.3f}")
    print(f"95% CI: [{model_fe_clustered.params['treatment'] - 1.96*model_fe_clustered.bse['treatment']:.3f}, {model_fe_clustered.params['treatment'] + 1.96*model_fe_clustered.bse['treatment']:.3f}]")
    print(f"P-value: {model_fe_clustered.pvalues['treatment']:.4f}")
except Exception as e:
    print(f"Clustering error: {e}")

# Event study - check for pre-trends
# Create week dummies interacted with treatment
df['relative_week'] = df['week'] - 15  # Week 15 is the switch (0 = first week after)
df['pre_post'] = np.where(df['relative_week'] < 0, 'pre', 'post')

# Event study model
event_df = df.copy()
# Create indicators for each week relative to switch, interacted with treatment
for w in range(-9, 8):  # weeks -9 to -1 (pre), 0 to 7 (post)
    event_df[f'week_{w}'] = ((event_df['relative_week'] == w) & (event_df['switched_branch'] == 1)).astype(int)

# Reference week is -1 (week 14, last pre-switch week)
pre_weeks = [f'week_{w}' for w in range(-9, -1)]  # -9 to -2
post_weeks = [f'week_{w}' for w in range(0, 8)]   # 0 to 7

# Run event study
event_formula = 'service_score ~ C(branch_id) + ' + ' + '.join(pre_weeks + post_weeks) + ' + staffing_index + visits_index'
# Drop week_-1 as reference
event_formula = 'service_score ~ C(branch_id) + ' + ' + '.join([f'week_{w}' for w in range(-9, -1)] + [f'week_{w}' for w in range(0, 8)]) + ' + staffing_index + visits_index'

model_event = ols(event_formula, data=event_df).fit()

print("\n=== Event Study Results (coefficients relative to week 14) ===")
print("\nPre-switch weeks (should be near 0 if parallel trends hold):")
for w in range(-9, -1):
    coef = model_event.params[f'week_{w}']
    se = model_event.bse[f'week_{w}']
    print(f"  Week {15+w}: {coef:.2f} (SE: {se:.2f})")

print("\nPost-switch weeks (treatment effect over time):")
for w in range(0, 8):
    coef = model_event.params[f'week_{w}']
    se = model_event.bse[f'week_{w}']
    print(f"  Week {15+w}: {coef:.2f} (SE: {se:.2f})")
