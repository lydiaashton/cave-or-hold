import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated_after'] = df['switched_branch'] * df['after_switch']

# Calculate mean service scores by group and period
print("Mean service scores by group and period:")
means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean()
print(means)
print()

# Calculate the DiD estimate manually
control_before = means[(0, 0)]
control_after = means[(0, 1)]
treated_before = means[(1, 0)]
treated_after = means[(1, 1)]

control_change = control_after - control_before
treated_change = treated_after - treated_before
did_estimate = treated_change - control_change

print("Difference-in-Differences Calculation:")
print(f"Control group (no switch): {control_before:.2f} → {control_after:.2f}, change = {control_change:.2f}")
print(f"Treated group (switched): {treated_before:.2f} → {treated_after:.2f}, change = {treated_change:.2f}")
print(f"DiD estimate (treated change - control change): {did_estimate:.2f}")
print()

# Run DiD regression with branch fixed effects
# This controls for time-invariant branch characteristics
print("DiD Regression with Branch Fixed Effects:")
model = ols('service_score ~ switched_branch * after_switch + C(branch_id)', data=df).fit()
print(model.summary())
