import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create event study variables (relative week to switch)
df['relative_week'] = df['week'] - 15  # week 15 is the switch, so 0
df['relative_week'] = df['relative_week'].clip(lower=-9, upper=7)  # cap at -9 (week 6) and 7 (week 22)

# Create dummies for each relative week (excluding week -1 as reference)
for rw in range(-9, 8):
    if rw != -1:  # skip reference category
        df[f'rel_week_{rw}'] = (df['relative_week'] == rw).astype(int)

# Run event study regression
event_study_vars = [f'rel_week_{rw}' for rw in range(-9, 8) if rw != -1]
formula = 'service_score ~ switched_branch + ' + ' + '.join(event_study_vars)
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Regression ===")
print(event_model.summary())

# Extract coefficients for the interaction terms
coeffs = []
weeks = []
for rw in range(-9, 8):
    if rw != -1:
        coef = event_model.params[f'rel_week_{rw}']
        se = event_model.bse[f'rel_week_{rw}']
        coeffs.append(coef)
        weeks.append(rw)

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
ax.errorbar(weeks, coeffs, yerr=[1.96*event_model.bse[f'rel_week_{w}'] for w in weeks], 
            fmt='o', capsize=5, markersize=8, linewidth=2)
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch Point')
ax.set_xlabel('Relative Week (0 = Switch Week)', fontsize=12)
ax.set_ylabel('Coefficient (vs Week -1)', fontsize=12)
ax.set_title('Event Study: Effect of New Procedure Over Time', fontsize=14)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-9, 8))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as event_study.png")

# Print the coefficients
print("\n=== Event Study Coefficients ===")
for rw, coef in zip(weeks, coeffs):
    se = event_model.bse[f'rel_week_{rw}']
    pval = event_model.pvalues[f'rel_week_{rw}']
    print(f"Week {rw+15} (relative {rw:+d}): coef={coef:.2f}, SE={se:.2f}, p={pval:.3f}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create event study variables (relative week to switch)
df['relative_week'] = df['week'] - 15  # week 15 is the switch, so 0
df['relative_week'] = df['relative_week'].clip(lower=-9, upper=7)

# Create dummies for each relative week with cleaner names
for rw in range(-9, 8):
    if rw != -1:  # skip reference category
        col_name = f'rel_wk_{abs(rw)}_{"neg" if rw < 0 else "pos"}' if rw != 0 else 'rel_wk_0'
        df[col_name] = (df['relative_week'] == rw).astype(int)

# Build formula
event_study_vars = []
for rw in range(-9, 8):
    if rw != -1:
        if rw == 0:
            event_study_vars.append('rel_wk_0')
        elif rw < 0:
            event_study_vars.append(f'rel_wk_{abs(rw)}_neg')
        else:
            event_study_vars.append(f'rel_wk_{rw}_pos')

formula = 'service_score ~ switched_branch + ' + ' + '.join(event_study_vars)
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
