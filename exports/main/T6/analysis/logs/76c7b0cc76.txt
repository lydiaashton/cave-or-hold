import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Create a group variable for DiD
df['group'] = df['switched_branch'].astype(str) + '_' + df['after_switch'].astype(str)

# Mean service scores by group
print("Mean service scores by group:")
group_means = df.groupby('group')['service_score'].agg(['mean', 'std', 'count'])
print(group_means)

# Calculate DiD manually
# Treatment group (switched=1)
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

# Control group (switched=0)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print("\n=== Difference-in-Differences Calculation ===")
print(f"Treatment (switched) before: {treatment_before:.3f}")
print(f"Treatment (switched) after:  {treatment_after:.3f}")
print(f"Control (non-switched) before: {control_before:.3f}")
print(f"Control (non-switched) after:  {control_after:.3f}")

treatment_change = treatment_after - treatment_before
control_change = control_after - control_before
did_effect = treatment_change - control_change

print(f"\nTreatment change: {treatment_change:.3f}")
print(f"Control change: {control_change:.3f}")
print(f"\nDiD Effect (Treatment change - Control change): {did_effect:.3f}")

weekly_avg.columns = ['week', 'switched_branch', 'avg_score']
