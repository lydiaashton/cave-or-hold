import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Calculate the difference-in-differences manually
group_period_means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
print("Mean service scores by group and period:")
print(group_period_means)

# DiD calculation
switched_diff = group_period_means.loc[1, 1] - group_period_means.loc[1, 0]
control_diff = group_period_means.loc[0, 1] - group_period_means.loc[0, 0]
did_estimate = switched_diff - control_diff

print(f"\nSwitched branches change (After - Before): {switched_diff:.3f}")
print(f"Control branches change (After - Before): {control_diff:.3f}")
print(f"Difference-in-Differences estimate: {did_estimate:.3f}")

# Now run a regression with DiD specification
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimator

df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\n\nDiD Regression Results:")
print(model.summary())

# Let's also cluster standard errors by branch
# Using statsmodels with cluster robust SE
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression Results (Clustered by branch):")
print(model_clustered.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create relative week (weeks since switch, with week 14 as reference)
df['relative_week'] = df['week'] - 14

# Create dummies for each relative week (excluding week -1 as reference)
# We'll interact these with switched_branch
for rw in range(-8, 9):
    if rw != -1:  # Skip reference period
        df[f'rel_week_{rw}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Event study regression
# Reference period is week -1 (the week before switch)
event_vars = [f'rel_week_{rw}' for rw in range(-8, 9) if rw != -1]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the event study
coeffs = []
cis_lower = []
cis_upper = []
weeks = []

for rw in range(-8, 9):
    if rw == -1:
        # Reference period
        coeffs.append(0)
        cis_lower.append(0)
        cis_upper.append(0)
        weeks.append(rw)
    else:
        var_name = f'rel_week_{rw}'
        if var_name in model_event.params:
            coeffs.append(model_event.params[var_name])
            cis_lower.append(model_event.conf_int().loc[var_name, 0])
            cis_upper.append(model_event.conf_int().loc[var_name, 1])
            weeks.append(rw)

# Create event study plot
fig, ax = plt.subplots(figsize=(12, 6))

ax.errorbar(weeks, coeffs, yerr=[np.array(coeffs) - np.array(cis_lower), np.array(cis_upper) - np.array(coeffs)], 
            fmt='o', capsize=5, markersize=8, linewidth=2, color='blue', ecolor='gray')

ax.axhline(y=0, color='red', linestyle='--', linewidth=2)
ax.axvline(x=-0.5, color='green', linestyle='-', linewidth=2, label='Switch Point')

ax.set_xlabel('Weeks Relative to Switch (Week -1 = Reference)', fontsize=12)
ax.set_ylabel('Effect on Service Score (vs. Control)', fontsize=12)
ax.set_title('Event Study: Service Score Effect by Week Relative to Switch', fontsize=14, fontweight='bold')
ax.set_xticks(range(-8, 9))
ax.grid(True, alpha=0.3)
ax.legend()

plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("Event study plot saved as 'event_study.png'")

# Print the event study coefficients
print("\nEvent Study Coefficients (relative to week -1):")
for i, rw in enumerate(weeks):
    print(f"Week {rw:2d}: {coeffs[i]:6.2f} (95% CI: {cis_lower[i]:6.2f} to {cis_upper[i]:6.2f})")

for rw in range(-8, 9):

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create relative week
df['relative_week'] = df['week'] - 14

# For a proper event study in DiD, we want to see how the gap between switched and control
# changes over time relative to a pre-period baseline
# Let's use week -8 (the earliest) as the reference

# Create interactions of switched_branch with each relative week
for rw in range(-8, 9):
    df[f'int_rw_{rw+10}'] = (df['switched_branch'] == 1) & (df['relative_week'] == rw)

# Event study with week -8 as reference (drop that interaction)
# Include switched_branch main effect and after_switch main effect
event_vars = [f'int_rw_{rw+10}' for rw in range(-7, 9)]  # Exclude week -8
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients
# The coefficient on switched_branch is the baseline difference at week -8
# The interaction terms show how that difference changes in each subsequent week
coeffs = []
cis_lower = []
cis_upper = []
weeks = []

# Week -8 is the reference (just the switched_branch coefficient)
base_coef = model_event.params['switched_branch']
base_ci = model_event.conf_int().loc['switched_branch']
coeffs.append(base_coef)
cis_lower.append(base_ci[0])
cis_upper.append(base_ci[1])
weeks.append(-8)

for rw in range(-7, 9):
    var_name = f'int_rw_{rw+10}'
    # The effect at week rw is the baseline + the interaction
    effect = base_coef + model_event.params[var_name]
    # For CI, we need the variance of the sum
    # Var(a+b) = Var(a) + Var(b) + 2*Cov(a,b)
    var_base = model_event.cov_params().loc['switched_branch', 'switched_branch']
    var_int = model_event.cov_params().loc[var_name, var_name]
    cov = model_event.cov_params().loc['switched_branch', var_name]
    se = np.sqrt(var_base + var_int + 2*cov)
    
    coeffs.append(effect)
    cis_lower.append(effect - 1.96 * se)
    cis_upper.append(effect + 1.96 * se)
    weeks.append(rw)

# Create event study plot
fig, ax = plt.subplots(figsize=(12, 6))

ax.errorbar(weeks, coeffs, yerr=[np.array(coeffs) - np.array(cis_lower), np.array(cis_upper) - np.array(coeffs)], 
            fmt='o', capsize=5, markersize=8, linewidth=2, color='blue', ecolor='gray')
