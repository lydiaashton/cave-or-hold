import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run a simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Simple DiD Regression:")
print(model.summary())

# Now run with branch fixed effects
print("\n\nDiD with Branch Fixed Effects:")
model_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit()
print(model_fe.summary())

# Cluster-robust standard errors at branch level

model_fe_cluster = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD with branch fixed effects (switched_branch is absorbed by FE)
print("DiD with Branch Fixed Effects (treatment effect only):")
model_fe = ols('service_score ~ after_switch + treatment + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_fe.summary())

# Also run with week fixed effects to control for time trends
print("\n\nDiD with Branch and Week Fixed Effects:")
model_fe_week = ols('service_score ~ treatment + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_fe_week.summary())

# Check parallel trends - compare pre-treatment trends between groups
print("\n\nParallel Trends Check - Average service score by week and group:")
pre_trend = df[df['week'] < 15].groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(pre_trend)

# Calculate week-by-week differences
print("\nWeek-by-week differences (treatment - control) before switch:")
for week in range(6, 15):
    treated = df[(df['week'] == week) & (df['switched_branch'] == 1)]['service_score'].mean()
    control = df[(df['week'] == week) & (df['switched_branch'] == 0)]['service_score'].mean()
    print(f"Week {week}: {treated - control:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple event study with renamed variables
df['relative_week'] = df['week'] - 15  # week 15 is when switch happened

# Create indicators for each week relative to switch with proper names
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    df[f'rw{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (omitting rw-1 as reference)
event_vars = [f'rw{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ ' + ' + '.join(event_vars) + ' + C(branch_id) + C(week)'
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event study coefficients (relative to week -1):")
event_results = []
for w in range(-9, 8):
    if w != -1:
        var = f'rw{w}'
        coef = event_model.params[var]
        se = event_model.bse[var]
        pval = event_model.pvalues[var]
        ci_low = event_model.conf_int().loc[var][0]
        ci_high = event_model.conf_int().loc[var][1]
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        event_results.append({'week': w, 'coef': coef, 'se': se, 'pval': pval, 'ci_low': ci_low, 'ci_high': ci_high})
        print(f"Week {w:3d}: {coef:6.2f} (SE={se:.2f}, p={pval:.3f}) {sig}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple event study with renamed variables - use 'pre' and 'post' naming
df['relative_week'] = df['week'] - 15  # week 15 is when switch happened

# Create indicators with safe names
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    if w < 0:
        df[f'pre{abs(w)}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
    else:
        df[f'post{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (omitting pre1 as reference - week 14)
pre_vars = [f'pre{w}' for w in range(9, 1, -1)]  # pre9, pre8, ..., pre2
post_vars = [f'post{w}' for w in range(0, 8)]    # post0, post1, ..., post7
event_vars = pre_vars + post_vars

formula = 'service_score ~ ' + ' + '.join(event_vars) + ' + C(branch_id) + C(week)'
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event study coefficients (relative to week -1 / pre1):")
event_results = []
for w in range(-9, 8):
    if w < 0:
        var = f'pre{abs(w)}'
    else:
        var = f'post{w}'
    
    if w == -1:  # reference period
        event_results.append({'week': w, 'coef': 0, 'se': 0, 'pval': np.nan, 'ci_low': 0, 'ci_high': 0})
        continue
        
    coef = event_model.params[var]
    se = event_model.bse[var]
    pval = event_model.pvalues[var]
    ci_low = event_model.conf_int().loc[var][0]
    ci_high = event_model.conf_int().loc[var][1]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    event_results.append({'week': w, 'coef': coef, 'se': se, 'pval': pval, 'ci_low': ci_low, 'ci_high': ci_high})
    print(f"Week {w:3d}: {coef:6.2f} (SE={se:.2f}, p={pval:.3f}) {sig}")

# Create a plot of the event study
fig, ax = plt.subplots(figsize=(10, 6))
weeks = [r['week'] for r in event_results]
coefs = [r['coef'] for r in event_results]
ci_low = [r['ci_low'] for r in event_results]
ci_high = [r['ci_high'] for r in event_results]

ax.errorbar(weeks, coefs, yerr=[np.array(coefs) - np.array(ci_low), np.array(ci_high) - np.array(coefs)], 
            fmt='o', capsize=5, label='Treatment Effect')
ax.axhline(y=0, color='red', linestyle='--', label='No Effect')
ax.axvline(x=-0.5, color='gray', linestyle=':', label='Switch Point')
ax.set_xlabel('Weeks Relative to Switch (0 = first week after switch)')
ax.set_ylabel('Effect on Service Score')
ax.set_title('Event Study: Effect of New Procedure on Service Score')
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved to event_study.png")
