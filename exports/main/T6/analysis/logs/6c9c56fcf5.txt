import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Basic Difference-in-Differences Regression ===")
print(model.summary())

# DiD with cluster-robust standard errors (clustered by branch)
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD with Cluster-Robust Standard Errors ===")
print(model_clustered.summary())

# DiD with controls
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD with Controls (Clustered SE) ===")
print(model_controls.summary())

import pandas as pd

matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate average service scores by group and period
print("=== Average Service Scores by Group and Period ===")
pre_data = df[df['after_switch'] == 0]
post_data = df[df['after_switch'] == 1]

print("\nPre-switch (weeks 6-14):")
print(pre_data.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

print("\nPost-switch (weeks 15-22):")
print(post_data.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

# Calculate the DiD manually
control_pre = pre_data[pre_data['switched_branch'] == 0]['service_score'].mean()
control_post = post_data[post_data['switched_branch'] == 0]['service_score'].mean()
treatment_pre = pre_data[pre_data['switched_branch'] == 1]['service_score'].mean()
treatment_post = post_data[post_data['switched_branch'] == 1]['service_score'].mean()

print("\n=== Manual DiD Calculation ===")
print(f"Control (non-switched) pre: {control_pre:.2f}")
print(f"Control (non-switched) post: {control_post:.2f}")
print(f"Control change: {control_post - control_pre:.2f}")
print(f"\nTreatment (switched) pre: {treatment_pre:.2f}")
print(f"Treatment (switched) post: {treatment_post:.2f}")
print(f"Treatment change: {treatment_post - treatment_pre:.2f}")
print(f"\nDifference-in-Differences: {(treatment_post - treatment_pre) - (control_post - control_pre):.2f}")

# Check parallel trends - plot weekly averages
print("\n=== Weekly Averages by Group ===")
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(weekly_avg)

# Create a plot
fig, ax = plt.subplots(figsize=(10, 6))
weeks = weekly_avg.index
ax.plot(weeks, weekly_avg[0], 'o-', label='Control (Old Procedure)', linewidth=2, markersize=6)

ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (Week 15)')
ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Scores by Week: Treatment vs Control Groups', fontsize=14)
ax.legend(fontsize=11)
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("\nPlot saved as service_scores_trend.png")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Test parallel trends: regress treatment status on pre-period trends
# Only use pre-switch data (weeks 6-14)
pre_data = df[df['after_switch'] == 0].copy()

# Check if there's a differential trend before the switch
# Create a continuous week variable centered at week 14 (last pre period)
pre_data['week_centered'] = pre_data['week'] - 14
pre_data['trend_interaction'] = pre_data['switched_branch'] * pre_data['week_centered']

# Regress service score on week trend and its interaction with treatment
pre_trend_model = ols('service_score ~ switched_branch + week_centered + trend_interaction', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print("=== Pre-Trend Test (Parallel Trends Assumption) ===")
print(pre_trend_model.summary())

# More robust: DiD with branch fixed effects
# This controls for time-invariant branch characteristics
df['branch_id_str'] = df['branch_id'].astype(str)

# DiD with branch fixed effects (using dummy variables)
# This is computationally intensive but more robust
print("\n=== DiD with Branch Fixed Effects ===")
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df.reset_index(drop=True), branch_dummies.reset_index(drop=True)], axis=1)

# Get the column names for branch dummies
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_')]

# Build formula
formula = 'service_score ~ after_switch + treatment + staffing_index + visits_index + ' + ' + '.join(branch_cols)
fe_model = ols(formula, data=df_with_dummies).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"Coefficient on treatment: {fe_model.params['treatment']:.4f}")
print(f"Standard error: {fe_model.bse['treatment']:.4f}")
print(f"P-value: {fe_model.pvalues['treatment']:.6f}")
print(f"95% CI: [{fe_model.conf_int().loc['treatment', 0]:.2f}, {fe_model.conf_int().loc['treatment', 1]:.2f}]")

# Also run with week fixed effects for additional robustness
print("\n=== DiD with Branch and Week Fixed Effects ===")
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df_full = pd.concat([df.reset_index(drop=True), week_dummies.reset_index(drop=True)], axis=1)
branch_dummies2 = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_full = pd.concat([df_full.reset_index(drop=True), branch_dummies2.reset_index(drop=True)], axis=1)

week_cols = [c for c in df_full.columns if c.startswith('week_')]
branch_cols2 = [c for c in df_full.columns if c.startswith('branch_')]

formula_full = 'service_score ~ treatment + staffing_index + visits_index + ' + ' + '.join(week_cols) + ' + ' + ' + '.join(branch_cols2)
fe_full_model = ols(formula_full, data=df_full).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"Coefficient on treatment: {fe_full_model.params['treatment']:.4f}")
print(f"Standard error: {fe_full_model.bse['treatment']:.4f}")
print(f"P-value: {fe_full_model.pvalues['treatment']:.6f}")
print(f"95% CI: [{fe_full_model.conf_int().loc['treatment', 0]:.2f}, {fe_full_model.conf_int().loc['treatment', 1]:.2f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Use C() notation for fixed effects in statsmodels - much cleaner
print("=== DiD with Branch Fixed Effects (using C() notation) ===")
fe_model = ols('service_score ~ treatment + staffing_index + visits_index + C(branch_id)', 
               data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"Coefficient on treatment: {fe_model.params['treatment']:.4f}")
print(f"Standard error: {fe_model.bse['treatment']:.4f}")
print(f"P-value: {fe_model.pvalues['treatment']:.6f}")
print(f"95% CI: [{fe_model.conf_int().loc['treatment', 0]:.2f}, {fe_model.conf_int().loc['treatment', 1]:.2f}]")

print("\n=== DiD with Branch and Week Fixed Effects ===")
fe_full_model = ols('service_score ~ treatment + staffing_index + visits_index + C(branch_id) + C(week)', 
                    data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"Coefficient on treatment: {fe_full_model.params['treatment']:.4f}")
print(f"Standard error: {fe_full_model.bse['treatment']:.4f}")
print(f"P-value: {fe_full_model.pvalues['treatment']:.6f}")
print(f"95% CI: [{fe_full_model.conf_int().loc['treatment', 0]:.2f}, {fe_full_model.conf_int().loc['treatment', 1]:.2f}]")

# Event study - check effects by week relative to switch
print("\n=== Event Study Analysis ===")
df['relative_week'] = df['week'] - 15  # Week 15 is when switch happened (0 = first post period)
# Create indicators for each relative week
for rw in range(-9, 8):  # weeks -9 to +7 relative to switch
    df[f'rel_week_{rw}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Use week -1 as reference (last pre-period)
event_study_vars = [f'rel_week_{rw}' for rw in range(-9, 8) if rw != -1]
formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(branch_id) + C(week)'
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event study coefficients (relative to week -1):")
for rw in range(-9, 8):
    if rw != -1:
        var_name = f'rel_week_{rw}'
        coef = event_model.params[var_name]
        se = event_model.bse[var_name]
        pval = event_model.pvalues[var_name]
        period = "Pre" if rw < 0 else "Post"
        print(f"  Week {rw:+d} ({period}): {coef:.2f} (SE: {se:.2f}, p={pval:.3f})")

import pandas as pd

# Look at the pre-trend more carefully
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly['diff'] = weekly[1] - weekly[0]

print("=== Weekly Treatment-Control Differences ===")
print(weekly)

# Pre-period trend (weeks 6-14)
pre_diffs = weekly[weekly.index < 15]['diff']
print(f"\nPre-switch differences by week:")
for week, diff in pre_diffs.items():
    print(f"  Week {week}: {diff:.2f}")

print(f"\nPre-switch trend: Difference grew from {pre_diffs.iloc[0]:.2f} (week 6) to {pre_diffs.iloc[-1]:.2f} (week 14)")
print(f"That's a {pre_diffs.iloc[-1] - pre_diffs.iloc[0]:.2f} point widening BEFORE the switch")

# Post-period
post_diffs = weekly[weekly.index >= 15]['diff']
print(f"\nPost-switch differences by week:")
for week, diff in post_diffs.items():
    print(f"  Week {week}: {diff:.2f}")

print(f"\nPost-switch trend: Difference grew from {post_diffs.iloc[0]:.2f} (week 15) to {post_diffs.iloc[-1]:.2f} (week 22)")
print(f"That's a {post_diffs.iloc[-1] - post_diffs.iloc[0]:.2f} point widening AFTER the switch")

# Compare the rate of change
pre_slope = (pre_diffs.iloc[-1] - pre_diffs.iloc[0]) / (len(pre_diffs) - 1)
post_slope = (post_diffs.iloc[-1] - post_diffs.iloc[0]) / (len(post_diffs) - 1)

print(f"\n=== CRITICAL COMPARISON ===")
print(f"Pre-switch: gap widened by {pre_diffs.iloc[-1] - pre_diffs.iloc[0]:.2f} points over 8 weeks ({pre_slope:.2f}/week)")
print(f"Post-switch: gap widened by {post_diffs.iloc[-1] - post_diffs.iloc[0]:.2f} points over 7 weeks ({post_slope:.2f}/week)")
print(f"\nThe rate of widening actually SLOWED after the switch")

# Simple comparison: average post minus average pre
simple_did = post_diffs.mean() - pre_diffs.mean()
print(f"\nSimple DiD estimate: {simple_did:.2f} points")

# But if we extrapolate the pre-trend, what would we expect?
# If the gap continued widening at the pre-switch rate, what would post average be?
expected_post_if_trend_continued = pre_diffs.iloc[-1] + (pre_slope * 4)  # roughly 4 weeks into post period on average
print(f"If pre-trend continued, expected average post diff: ~{expected_post_if_trend_continued:.2f}")
print(f"Actual average post diff: {post_diffs.mean():.2f}")
print(f"Adjustment for pre-trend: {post_diffs.mean() - expected_post_if_trend_continued:.2f}")
