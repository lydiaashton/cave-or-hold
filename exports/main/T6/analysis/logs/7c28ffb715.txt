# Load the data
df = pd.read_csv('branches.csv')

# Create the treatment interaction term (DiD term)
df['treated'] = df['switched_branch'] * df['after_switch']

# Basic DiD table: means by group and period
print("=== Difference-in-Differences Summary ===\n")

# Before period (after_switch = 0)
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()

# After period (after_switch = 1)
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print("Service Score Means:")

print(f"  Control group (no switch):  Before = {before_control:.2f}, After = {after_control:.2f}, Change = {after_control - before_control:.2f}")
print(f"\n  DiD Estimate (Treatment change - Control change): {(after_treatment - before_treatment) - (after_control - before_control):.2f}")

# Run DiD regression
print("\n=== DiD Regression ===")
model = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# Main DiD with cluster-robust SEs
model = ols('service_score ~ C(branch_id) + after_switch + treated', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract the treatment effect
treatment_effect = model.params['treated']
treatment_se = model.bse['treated']
treatment_pvalue = model.pvalues['treated']

# 95% confidence interval
ci_lower = treatment_effect - 1.96 * treatment_se
ci_upper = treatment_effect + 1.96 * treatment_se

print("=== Treatment Effect Estimate ===")
print(f"Point estimate: {treatment_effect:.2f} points")
print(f"Standard error: {treatment_se:.2f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {treatment_pvalue:.3f}")

# What does this mean practically?
print("\n=== Interpretation ===")
print(f"The data is consistent with effects ranging from {ci_lower:.1f} to {ci_upper:.1f} points.")
print(f"This includes zero (no effect) and potentially meaningful positive effects.")
print(f"\nStatistical significance threshold: p < 0.05")
print(f"Our p-value: {treatment_pvalue:.3f}")
print(f"Conclusion: Not statistically significant at conventional levels")

# Power consideration
n_treatment = 20
n_control = 14
print(f"\n=== Sample Size ===")
print(f"Treatment branches: {n_treatment}")
print(f"Control branches: {n_control}")
print(f"This is a relatively small sample for detecting modest effects.")
