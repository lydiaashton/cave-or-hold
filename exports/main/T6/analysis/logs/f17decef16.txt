import pandas as pd

import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = β0 + β1*treatment + β2*post + β3*(treatment*post) + ε
# β3 is the DiD estimate

df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['treatment_post'] = df['treatment'] * df['post']

# Basic DiD regression
model = ols('service_score ~ treatment + post + treatment_post', data=df).fit()
print("Basic Difference-in-Differences Regression:")
print(model.summary())

# Now with branch fixed effects (more robust)
# This controls for time-invariant branch characteristics
df = pd.read_csv('branches.csv')
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Add branch fixed effects
model_fe = ols('service_score ~ switched_branch + after_switch + treatment_post + C(branch_id)', data=df).fit()
