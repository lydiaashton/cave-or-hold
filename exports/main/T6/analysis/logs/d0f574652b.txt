import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['branch_id_str'] = df['branch_id'].astype(str)

# Proper event study - compare each week to a baseline (week 14, the week before switch)
print("=== Proper Event Study (vs Week 14 baseline) ===")

# Get week 14 as baseline
baseline = df[df['week'] == 14].groupby('switched_branch')['service_score'].mean()
print(f"Baseline (Week 14) - Control: {baseline[0]:.2f}, Treatment: {baseline[1]:.2f}")
print(f"Baseline gap (Treatment - Control): {baseline[1] - baseline[0]:.2f}")

# Calculate DiD for each post-switch week relative to week 14
event_results = []
for w in range(15, 23):
    week_data = df[df['week'] == w]
    baseline_data = df[df['week'] == 14]
    
    # Treatment change from baseline
    treatment_week = week_data[week_data['switched_branch'] == 1]['service_score'].mean()
    treatment_base = baseline_data[baseline_data['switched_branch'] == 1]['service_score'].mean()
    treatment_change = treatment_week - treatment_base
    
    # Control change from baseline
    control_week = week_data[week_data['switched_branch'] == 0]['service_score'].mean()
    control_base = baseline_data[baseline_data['switched_branch'] == 0]['service_score'].mean()
    control_change = control_week - control_base
    
    # DiD
    did = treatment_change - control_change
    
    event_results.append({
        'week': w,
        'week_relative': w - 14,
        'treatment_score': treatment_week,
        'control_score': control_week,
        'treatment_change': treatment_change,
        'control_change': control_change,
        'did': did
    })
