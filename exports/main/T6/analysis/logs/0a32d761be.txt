import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treated'] * df['post']

# Calculate summary statistics by group and period
print("=== Service Score Summary by Group and Period ===\n")

# Treatment group (switched branches)
treated_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score']
treated_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score']

# Control group (non-switched branches)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score']
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score']

print("Treatment Group (Switched Branches):")
print(f"  Before (weeks 6-14):  n={len(treated_before)}, mean={treated_before.mean():.2f}, std={treated_before.std():.2f}")
print(f"  After (weeks 15-22):  n={len(treated_after)}, mean={treated_after.mean():.2f}, std={treated_after.std():.2f}")
print(f"  Change: {treated_after.mean() - treated_before.mean():.2f}")

print("\nControl Group (Non-Switched Branches):")
print(f"  Before (weeks 6-14):  n={len(control_before)}, mean={control_before.mean():.2f}, std={control_before.std():.2f}")
print(f"  After (weeks 15-22):  n={len(control_after)}, mean={control_after.mean():.2f}, std={control_after.std():.2f}")
print(f"  Change: {control_after.mean() - control_before.mean():.2f}")

# Difference-in-Differences calculation
did_estimate = (treated_after.mean() - treated_before.mean()) - (control_after.mean() - control_before.mean())
print(f"\n=== Difference-in-Differences Estimate: {did_estimate:.2f} ===")

# Run DiD regression
print("\n=== DiD Regression Results ===")
model = ols('service_score ~ treated + post + did', data=df).fit()
print(model.summary())

# Also run with branch fixed effects
print("\n=== DiD Regression with Branch Fixed Effects ===")
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ treated + post + did + C(branch_id_str)', data=df).fit()
print(model_fe.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check parallel trends - average service score by week for each group
print("=== Parallel Trends Check: Average Service Score by Week ===\n")
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(weekly_avg)

# Calculate pre-trend difference (weeks 6-14)
pre_period = df[df['week'] < 15]
post_period = df[df['week'] >= 15]

print("\n=== Pre-Period (weeks 6-14) Average by Week ===")
pre_weekly = pre_period.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(pre_weekly)
print("\nDifference (Treatment - Control) by week:")
print(pre_weekly[1] - pre_weekly[0])

# Run regression with week fixed effects and cluster-robust SE
print("\n=== DiD with Branch and Week Fixed Effects ===")
df['week_str'] = df['week'].astype(str)
model_full = ols('service_score ~ treated + post + did + C(branch_id_str) + C(week_str)', data=df).fit()
print(model_full.summary())

# Cluster-robust standard errors at branch level
print("\n=== Cluster-Robust Standard Errors (clustered by branch) ===")
cov_clusters = model_full.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(cov_clusters.summary())

# Create visualization
plt.figure(figsize=(12, 6))

# Plot average service score by week for each group
weeks = range(6, 23)
treated_means = []
control_means = []

for w in weeks:
    treated_means.append(df[(df['week']==w) & (df['switched_branch']==1)]['service_score'].mean())
    control_means.append(df[(df['week']==w) & (df['switched_branch']==0)]['service_score'].mean())

plt.plot(weeks, treated_means, 'o-', label='Switched Branches (Treatment)', linewidth=2, markersize=8)
plt.plot(weeks, control_means, 's-', label='Non-Switched Branches (Control)', linewidth=2, markersize=8)
plt.axvline(x=14.5, color='red', linestyle='--', label='Switch Point (Week 15)')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Score Trends: Treatment vs Control Groups')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('service_score_trends.png', dpi=150)
print("\nPlot saved as service_score_trends.png")

# Also check if there are any covariates that might explain the effect
print("\n=== Controlling for Staffing and Visits ===")
model_covariates = ols('service_score ~ treated + post + did + staffing_index + visits_index + C(branch_id_str) + C(week_str)', data=df).fit()
cov_clusters_cov = model_covariates.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(cov_clusters_cov.summary())
