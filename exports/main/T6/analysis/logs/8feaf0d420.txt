import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Cluster-robust standard errors at branch level
df['branch_id'] = df['branch_id'].astype(str)

model_fe = ols('service_score ~ switched_branch:after_switch + C(branch_id) + C(week)', data=df).fit()

# Get cluster-robust standard errors
from statsmodels.stats.sandwich_covariance import cov_cluster
cov_clustered = cov_cluster(model_fe, df['branch_id'].astype('category').cat.codes)
se_robust = np.sqrt(np.diag(cov_clustered))

print("=== Cluster-Robust Standard Errors (by branch) ===")
treatment_idx = list(model_fe.params.index).index('switched_branch:after_switch')
print(f"Treatment effect: {model_fe.params['switched_branch:after_switch']:.4f}")
print(f"Cluster-robust SE: {se_robust[treatment_idx]:.4f}")
t_stat = model_fe.params['switched_branch:after_switch'] / se_robust[treatment_idx]
print(f"t-statistic: {t_stat:.2f}")
# Approximate p-value
from scipy import stats
p_val = 2 * (1 - stats.t.cdf(abs(t_stat), df=len(df['branch_id'].unique()) - 1))
print(f"P-value: {p_val:.6f}")

# Effect by post-period week (dynamic effects)
df['branch_id'] = df['branch_id'].astype(int)
post_weeks = df[df['after_switch'] == 1].groupby('week').apply(
    lambda x: x[x['switched_branch'] == 1]['service_score'].mean() - x[x['switched_branch'] == 0]['service_score'].mean()
)
print("\n=== Treatment Effect by Week (Post-Period) ===")
print(post_weeks)
print(f"\nAverage post-period effect: {post_weeks.mean():.2f}")

# 1. Simple difference-in-differences calculation
before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

before_treated = before[before['switched_branch'] == 1]['service_score'].mean()
before_control = before[before['switched_branch'] == 0]['service_score'].mean()
after_treated = after[after['switched_branch'] == 1]['service_score'].mean()
after_control = after[after['switched_branch'] == 0]['service_score'].mean()

did_simple = (after_treated - before_treated) - (after_control - before_control)

print("\n1. SIMPLE DiD CALCULATION:")

# 2. Regression with fixed effects
df['branch_id'] = df['branch_id'].astype(str)
model = ols('service_score ~ switched_branch:after_switch + C(branch_id) + C(week)', data=df).fit()

print("\n2. FIXED EFFECTS REGRESSION:")

print(f"   Standard error: {model.bse['switched_branch:after_switch']:.3f}")
print(f"   P-value: {model.pvalues['switched_branch:after_switch']:.6f}")

# 3. Cluster-robust standard errors
from statsmodels.stats.sandwich_covariance import cov_cluster
cov_clustered = cov_cluster(model, df['branch_id'].astype('category').cat.codes)
se_robust = np.sqrt(np.diag(cov_clustered))
treatment_idx = list(model.params.index).index('switched_branch:after_switch')
se_cluster = se_robust[treatment_idx]
effect = model.params['switched_branch:after_switch']
t_stat = effect / se_cluster
df_clusters = df['branch_id'].nunique() - 1
p_cluster = 2 * (1 - stats.t.cdf(abs(t_stat), df=df_clusters))
