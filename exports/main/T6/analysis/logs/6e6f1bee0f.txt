import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate mean service scores by group and period
print("=== Mean Service Scores by Group and Period ===")
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='after_switch', 
                       aggfunc='mean')
print(pivot)
print()

# Calculate differences
print("=== Differences ===")
before_diff = pivot[0][1] - pivot[0][0]  # Control group change
after_diff = pivot[1][1] - pivot[1][0]   # Treatment group change
did_estimate = after_diff - before_diff  # DiD estimate
print(f"Control group (no switch) before->after change: {before_diff:.3f}")
print(f"Treatment group (switched) before->after change: {after_diff:.3f}")
print(f"Difference-in-Differences estimate: {did_estimate:.3f}")
print()

# Run regression for DiD with standard errors
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Regression Results (DiD) ===")
print(model.summary())
print()

# Cluster-robust standard errors by branch
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Regression Results with Cluster-Robust SE (by branch) ===")
print(model_clustered.summary())
