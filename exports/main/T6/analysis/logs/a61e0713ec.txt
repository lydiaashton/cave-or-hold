# Calculate the DiD components
print("\n=== Difference-in-Differences Calculation ===")

# Before period (after_switch = 0)
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()

# After period (after_switch = 1)
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['did_interaction'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + did_interaction', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# DiD with cluster-robust standard errors (clustered by branch)
model_clustered = ols('service_score ~ switched_branch + after_switch + did_interaction', data=df).fit()

print("\n=== DiD Regression with Cluster-Robust SE (by branch) ===")

import pandas as pd

print("1. MANUAL DIFFERENCE-IN-DIFFERENCES CALCULATION")
print("-" * 50)

before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()

control_change = after_control - before_control
treatment_change = after_treatment - before_treatment
did_effect = treatment_change - control_change

print(f"p-value:         {did_pval:.4f}")
print(f"95% CI:          [{did_ci_low:.3f}, {did_ci_high:.3f}]")
