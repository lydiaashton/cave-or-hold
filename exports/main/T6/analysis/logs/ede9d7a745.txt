import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Count branches by switch status
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate how many branches switched
n_switched = df[df['switched_branch'] == 1]['branch_id'].nunique()
n_control = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"\nSwitched branches: {n_switched}")
print(f"Control branches: {n_control}")

# Average service scores by group and period
print("\n\nAverage service scores by group and period:")
df['period'] = np.where(df['after_switch'] == 0, 'Before', 'After')
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='period', 
                       aggfunc='mean')
print(pivot)

# Calculate the difference-in-differences manually
before_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nSwitched group: Before={before_switched:.2f}, After={after_switched:.2f}, Change={after_switched - before_switched:.2f}")
print(f"Control group: Before={before_control:.2f}, After={after_control:.2f}, Change={after_control - before_control:.2f}")

did_effect = (after_switched - before_switched) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {did_effect:.2f}")

# First, let's do a simple DiD without fixed effects
