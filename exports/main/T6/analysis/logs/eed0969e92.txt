import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Add branch fixed effects
# Create dummy variables for each branch
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df, branch_dummies], axis=1)

# Regression with branch fixed effects
fe_cols = [c for c in df_with_fe.columns if c.startswith('branch_')]
formula = 'service_score ~ switched_branch + after_switch + treatment_post + staffing_index + visits_index + ' + ' + '.join(fe_cols)

# This will be a large model, let's use a more efficient approach
# Actually, let's use the within estimator approach or just control for staffing/visits

# Model 1: DiD with controls (no FE)
model_controls = ols('service_score ~ switched_branch + after_switch + treatment_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("DiD Regression with Controls (staffing_index, visits_index)")
print("="*60)
print(model_controls.summary().tables[1])

# Model 2: Branch fixed effects (using a more efficient method)
# We'll use the linearmodels package if available, otherwise do a simpler approach
try:
    from linearmodels.panel import PanelOLS
    
    # Prepare panel data
    panel_df = df.set_index(['branch_id', 'week'])
    panel_df['treatment_post'] = panel_df['switched_branch'] * panel_df['after_switch']
    
    # Fixed effects model
    fe_model = PanelOLS(panel_df['service_score'], 
                        panel_df[['after_switch', 'treatment_post', 'staffing_index', 'visits_index']],
                        entity_effects=True)
    fe_result = fe_model.fit(cov_type='clustered', cluster_entity=True)
    print("\n\nPanel Model with Branch Fixed Effects")
    print("="*60)
    print(fe_result)
except ImportError:
    print("\nlinearmodels not available, using alternative FE approach")
    # Alternative: demean the data within each branch
    branch_means = df.groupby('branch_id')[['service_score', 'after_switch', 'treatment_post', 'staffing_index', 'visits_index']].transform('mean')
    df_demeaned = df[['service_score', 'after_switch', 'treatment_post', 'staffing_index', 'visits_index']] - branch_means
    
    fe_model_alt = ols('service_score ~ after_switch + treatment_post + staffing_index + visits_index', 
                       data=df_demeaned).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
    print("\n\nWithin-Estimator (Branch Fixed Effects)")
    print("="*60)
