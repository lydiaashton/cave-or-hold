import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Verify branch counts
control_branches = df[df['switched_branch'] == 0]['branch_id'].nunique()
treatment_branches = df[df['switched_branch'] == 1]['branch_id'].nunique()
print(f"Control branches (didn't switch): {control_branches}")
print(f"Treatment branches (switched): {treatment_branches}")
print(f"Total branches: {control_branches + treatment_branches}")

# Calculate average service scores by group and period
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("\nService scores by group and period:")
print(summary)

# Calculate the difference-in-differences
# Control group change
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
control_change = control_after - control_before

# Treatment group change
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_change = treatment_after - treatment_before

# Difference-in-differences
did = treatment_change - control_change

print(f"\nControl group: before={control_before:.2f}, after={control_after:.2f}, change={control_change:.2f}")
print(f"Treatment group: before={treatment_before:.2f}, after={treatment_after:.2f}, change={treatment_change:.2f}")
print(f"\nDifference-in-Differences estimate: {did:.2f}")

# Check pre-trends (weeks 6-14)
print("\n--- Pre-trend check (weeks 6-14) ---")
pre_trend = df[df['after_switch'] == 0].groupby(['switched_branch', 'week'])['service_score'].mean().unstack()
print(pre_trend.round(2))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Difference-in-differences regression
# service_score = β0 + β1*treatment + β2*post + β3*(treatment*post) + ε
# β3 is the DiD estimate

df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['treatment_post'] = df['treatment'] * df['post']

# Basic DiD regression
model = ols('service_score ~ treatment + post + treatment_post', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# DiD with branch fixed effects
print("\n=== DiD with Branch Fixed Effects ===")
model_fe = ols('service_score ~ treatment_post + C(branch_id)', data=df).fit()
print(model_fe.summary())

# DiD with branch and week fixed effects
print("\n=== DiD with Branch and Week Fixed Effects ===")
model_fe_week = ols('service_score ~ treatment_post + C(branch_id) + C(week)', data=df).fit()
print(model_fe_week.summary())

# DiD with controls (staffing and visits)
print("\n=== DiD with Branch/Week FE and Controls ===")
model_full = ols('service_score ~ treatment_post + C(branch_id) + C(week) + staffing_index + visits_index', data=df).fit()
