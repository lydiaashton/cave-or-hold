# Run DiD regression
# Basic model: service_score ~ switched_branch * after_switch
model_basic = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate average service score by week and treatment group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_score']

# Create a plot
plt.figure(figsize=(10, 6))

# Plot treatment and control groups over time
for group in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == group]
    label = 'Treatment (Switched)' if group == 1 else 'Control (No Switch)'
    plt.plot(subset['week'], subset['avg_score'], marker='o', linewidth=2, label=label, markersize=6)

# Add vertical line at switch point
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (Week 15)')
plt.axvspan(15, 22, alpha=0.1, color='red', label='Post-Switch Period')

plt.xlabel('Week', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Scores Over Time: Treatment vs Control Groups', fontsize=14, fontweight='bold')
plt.legend(loc='best')
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_scores_over_time.png', dpi=150)
print("Plot saved as 'service_scores_over_time.png'")

# Check parallel trends in pre-period
pre_period = df[df['week'] < 15]
pre_trends = pre_period.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()

# Run regression on pre-period only to test parallel trends
pre_period_copy = pre_period.copy()
pre_period_copy['week_centered'] = pre_period_copy['week'] - pre_period_copy['week'].mean()
model_trends = ols('service_score ~ switched_branch * week_centered', data=pre_period_copy).fit(cov_type='cluster', cov_kwds={'groups': pre_period_copy['branch_id']})

print("\n=== Parallel Trends Test (Pre-Period Only) ===")
print("Testing if trends differed before the switch...")
print(f"Interaction term (trend difference): {model_trends.params['switched_branch:week_centered']:.3f}")
print(f"Standard error: {model_trends.bse['switched_branch:week_centered']:.3f}")
print(f"P-value: {model_trends.pvalues['switched_branch:week_centered']:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators
# Week 14 is the last pre-switch week, so we'll use that as reference
df['relative_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create event study dummies (excluding reference period)
# We'll use week 13 as the reference (one period before switch)
event_study_vars = []
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != -1:  # exclude week 13 as reference
        df[f'rel_week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(f'rel_week_{w}')

# Run event study regression
event_study_formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(branch_id)'
model_event = ols(event_study_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results (Relative to Week 13) ===")
print("Coefficients for treatment group in each week relative to switch:")
print("(Week -1 is the reference period, omitted)")
print()

results = []
for w in range(-8, 9):
    if w != -1:
        coef = model_event.params[f'rel_week_{w}']
        se = model_event.bse[f'rel_week_{w}']
        pval = model_event.pvalues[f'rel_week_{w}']
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        results.append({'relative_week': w, 'coef': coef, 'se': se, 'pval': pval, 'ci_low': ci_low, 'ci_high': ci_high})
        marker = '*' if pval < 0.05 else ''
        print(f"Week {w:3d}: coef = {coef:6.2f}, SE = {se:.2f}, 95% CI = [{ci_low:6.2f}, {ci_high:6.2f}] {marker}")

event_results = pd.DataFrame(results)

# Summary of pre and post effects
pre_effects = event_results[event_results['relative_week'] < 0]
post_effects = event_results[event_results['relative_week'] > 0]

print(f"\n=== Summary ===")
print(f"Pre-switch weeks (should be ~0 if parallel trends hold):")
print(f"  Mean coefficient: {pre_effects['coef'].mean():.2f}")
print(f"  Range: [{pre_effects['coef'].min():.2f}, {pre_effects['coef'].max():.2f}]")
print(f"\nPost-switch weeks (treatment effect):")
print(f"  Mean coefficient: {post_effects['coef'].mean():.2f}")
print(f"  Range: [{post_effects['coef'].min():.2f}, {post_effects['coef'].max():.2f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators
# Week 14 is the last pre-switch week, week 13 will be reference
df['relative_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create event study dummies with valid names (prefix with 'w')
event_study_vars = []
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != -1:  # exclude week 13 as reference
        var_name = f'w{abs(w)}_neg' if w < 0 else f'w{w}_pos'
        if w == 0:
            var_name = 'w0'
        df[var_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(var_name)

# Run event study regression with branch fixed effects
event_study_formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(branch_id)'
model_event = ols(event_study_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results (Relative to Week 13) ===")
print("Coefficients for treatment group in each week relative to switch:")
print("(Week -1 is the reference period, omitted)")
print()

results = []
for w in range(-8, 9):
    if w != -1:
        if w < 0:
            var_name = f'w{abs(w)}_neg'
        elif w == 0:
            var_name = 'w0'
        else:
            var_name = f'w{w}_pos'
        
        coef = model_event.params[var_name]
        se = model_event.bse[var_name]
        pval = model_event.pvalues[var_name]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        results.append({'relative_week': w, 'coef': coef, 'se': se, 'pval': pval, 'ci_low': ci_low, 'ci_high': ci_high})
        marker = '*' if pval < 0.05 else ''
        print(f"Week {w:3d}: coef = {coef:6.2f}, SE = {se:.2f}, 95% CI = [{ci_low:6.2f}, {ci_high:6.2f}] {marker}")

event_results = pd.DataFrame(results)

# Summary of pre and post effects
pre_effects = event_results[event_results['relative_week'] < 0]
post_effects = event_results[event_results['relative_week'] >= 0]

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main DiD model
model = ols('service_score ~ treatment + C(branch_id) + C(week) + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

treatment_effect = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pval = model.pvalues['treatment']

print("=== Key Statistics ===")
print(f"Treatment effect: {treatment_effect:.2f} points")
print(f"Standard error: {treatment_se:.2f}")
print(f"P-value: {treatment_pval:.3f}")
print(f"95% CI: [{treatment_effect - 1.96*treatment_se:.2f}, {treatment_effect + 1.96*treatment_se:.2f}]")

# Calculate power-relevant metrics
print(f"\n=== Context ===")
print(f"Average service score (all data): {df['service_score'].mean():.1f}")
print(f"Effect as % of average: {treatment_effect/df['service_score'].mean()*100:.1f}%")
print(f"Effect relative to SD: {treatment_effect/df['service_score'].std():.2f} standard deviations")

# Post-switch means
post = df[df['after_switch'] == 1]
print(f"\n=== Post-Switch Averages ===")
print(f"Control branches: {post[post['switched_branch']==0]['service_score'].mean():.2f}")
print(f"Treatment branches: {post[post['switched_branch']==1]['service_score'].mean():.2f}")
print(f"Difference: {post[post['switched_branch']==1]['service_score'].mean() - post[post['switched_branch']==0]['service_score'].mean():.2f}")

# Pre-switch means
pre = df[df['after_switch'] == 0]
print(f"\n=== Pre-Switch Averages ===")
print(f"Control branches: {pre[pre['switched_branch']==0]['service_score'].mean():.2f}")
print(f"Treatment branches: {pre[pre['switched_branch']==1]['service_score'].mean():.2f}")
print(f"Difference: {pre[pre['switched_branch']==1]['service_score'].mean() - pre[pre['switched_branch']==0]['service_score'].mean():.2f}")

# DiD calculation
pre_diff = pre[pre['switched_branch']==1]['service_score'].mean() - pre[pre['switched_branch']==0]['service_score'].mean()
post_diff = post[post['switched_branch']==1]['service_score'].mean() - post[post['switched_branch']==0]['service_score'].mean()
did = post_diff - pre_diff
print(f"\n=== Simple DiD Calculation ===")
print(f"Pre-switch gap (T-C): {pre_diff:.2f}")
print(f"Post-switch gap (T-C): {post_diff:.2f}")
print(f"Change in gap (DiD): {did:.2f}")
