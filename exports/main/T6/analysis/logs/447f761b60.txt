import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run DiD regression with clustered standard errors
# Basic DiD model: service_score ~ switched_branch * after_switch
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("Basic DiD Regression (without clustering):")
print(model.summary())

# Now with branch fixed effects and clustered SEs
# Create dummies for branches
df = df.copy()
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Regression with branch fixed effects
model_fe = ols('service_score ~ did + C(branch_id)', data=df).fit()
print("\n\nDiD with Branch Fixed Effects:")
print(model_fe.summary())

# Clustered standard errors by branch
from statsmodels.stats.sandwich_covariance import cov_cluster
cov_clustered = cov_cluster(model_fe, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))
print("\n\nClustered Standard Errors (by branch):")
for i, (param, se) in enumerate(zip(model_fe.params.index, se_clustered)):
    print(f"{param}: coef={model_fe.params[param]:.4f}, SE={se:.4f}")

# Get the DID coefficient specifically
did_coef = model_fe.params['did']
did_se = se_clustered[model_fe.params.index.get_loc('did')]
did_t = did_coef / did_se
did_p = 2 * (1 - sm.stats.norm.cdf(abs(did_t)))
print(f"\n\nDID Effect: {did_coef:.3f}")
print(f"Clustered SE: {did_se:.3f}")
print(f"t-statistic: {did_t:.3f}")
print(f"p-value: {did_p:.4f}")
print(f"95% CI: [{did_coef - 1.96*did_se:.3f}, {did_coef + 1.96*did_se:.3f}]")
