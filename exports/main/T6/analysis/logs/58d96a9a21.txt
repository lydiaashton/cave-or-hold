import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create summary table: average service scores by group and period
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).unstack()
print("Service Score Summary by Group and Period:")
print(summary)

# Calculate DiD manually
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n=== Difference-in-Differences Calculation ===")
print(f"Treatment group (switched branches):")
print(f"  Before: {before_treatment:.3f}")
print(f"  After:  {after_treatment:.3f}")
print(f"  Change: {after_treatment - before_treatment:.3f}")
print(f"\nControl group (non-switched branches):")
print(f"  Before: {before_control:.3f}")
print(f"  After:  {after_control:.3f}")
print(f"  Change: {after_control - before_control:.3f}")

did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDiD Estimate: {did_estimate:.3f}")

# Run regression for statistical significance
# DiD regression: service_score ~ switched_branch * after_switch
df['treatment'] = df['switched_branch'] * df['after_switch']

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\n=== Regression Results ===")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from linearmodels.panel import PanelOLS
import warnings
warnings.filterwarnings('ignore')

df = pd.read_csv('branches.csv')

# Set up panel data
df = df.set_index(['branch_id', 'week'])

# Run panel regression with branch fixed effects and clustered SEs
# This is the gold standard for DiD analysis
model = PanelOLS.from_formula('service_score ~ switched_branch + after_switch + treatment + EntityEffects', data=df)
result = model.fit(cov_type='clustered', cluster_entity=True)

print("=== Panel Regression with Branch Fixed Effects ===")
print(result)

# Also check parallel trends assumption - plot trends over time
df_reset = df.reset_index()

# Calculate weekly averages by group
weekly_avg = df_reset.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print("\n=== Weekly Averages by Group ===")
print(weekly_avg)

# Create visualization
import matplotlib.pyplot as plt

fig, ax = plt.subplots(figsize=(10, 6))
weeks = weekly_avg.index
ax.plot(weeks, weekly_avg[0], 'o-', label='Control (No Switch)', linewidth=2, markersize=6)
ax.plot(weeks, weekly_avg[1], 's-', label='Treatment (Switched)', linewidth=2, markersize=6)
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch Week (15)')
ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Score Trends: Treatment vs Control Groups', fontsize=14)
ax.legend(loc='best')
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("\nPlot saved as service_score_trends.png")

# Check pre-trends (weeks 6-14)
pre_treatment = df_reset[df_reset['week'] < 15]
pre_trend_model = ols('service_score ~ week * switched_branch', data=pre_treatment).fit()
print("\n=== Pre-Treatment Trend Test (Weeks 6-14) ===")
print(f"Interaction coefficient (week:switched_branch): {pre_trend_model.params['week:switched_branch']:.4f}")
print(f"P-value: {pre_trend_model.pvalues['week:switched_branch']:.4f}")
