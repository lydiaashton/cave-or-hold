import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Calculate the difference-in-differences manually
# Treatment group (switched_branch=1)
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

# Control group (switched_branch=0)
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

# DiD estimate
did_estimate = (treatment_after - treatment_before) - (control_after - control_before)

print("=== Difference-in-Differences Calculation ===")
print(f"\nTreatment group (switched branches):")
print(f"  Before switch (weeks 6-14): {treatment_before:.2f}")
print(f"  After switch (weeks 15-22): {treatment_after:.2f}")
print(f"  Change: {treatment_after - treatment_before:.2f}")

print(f"\nControl group (non-switched branches):")
print(f"  Before switch (weeks 6-14): {control_before:.2f}")
print(f"  After switch (weeks 15-22): {control_after:.2f}")
print(f"  Change: {control_after - control_before:.2f}")

print(f"\n=== DiD Estimate: {did_estimate:.2f} ===")
print("This is the estimated effect of the new procedure on service scores.")

# Now run a regression with standard errors
# DiD regression: service_score ~ switched_branch * after_switch
df['interaction'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + interaction', data=df).fit()
print("\n=== DiD Regression Results ===")
print(model.summary())

# Cluster-robust standard errors by branch
model_clustered = ols('service_score ~ switched_branch + after_switch + interaction', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print(model_clustered.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from scipy import stats

df = pd.read_csv('branches.csv')

df['interaction'] = df['switched_branch'] * df['after_switch']

model_with_covariates = ols('service_score ~ switched_branch + after_switch + interaction + staffing_index + visits_index', 
                            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== DiD Regression with Covariates (Cluster-Robust SE) ===")
print(model_with_covariates.summary())

# Extract the key coefficient
did_coef = model_with_covariates.params['interaction']
did_se = model_with_covariates.bse['interaction']
did_pval = model_with_covariates.pvalues['interaction']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se

print(f"\n=== Key Result Summary ===")
print(f"Estimated effect of new procedure: {did_coef:.2f} points")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {did_pval:.4f}")

# Event study: effect by week relative to switch
print("\n=== Event Study: Treatment Effect by Week ===")
pre_avg = df[df['week'] < 15].groupby('branch_id')['service_score'].mean()

results = []
for week in range(15, 23):
    week_data = df[df['week'] == week].copy()
    week_data['pre_avg'] = week_data['branch_id'].map(pre_avg)
    week_data['diff_from_pre'] = week_data['service_score'] - week_data['pre_avg']
    
    # Compare treatment vs control in the change from pre-period
    treatment_change = week_data[week_data['switched_branch'] == 1]['diff_from_pre'].mean()
    control_change = week_data[week_data['switched_branch'] == 0]['diff_from_pre'].mean()
    did_week = treatment_change - control_change
    
    # Simple t-test
    treatment_vals = week_data[week_data['switched_branch'] == 1]['diff_from_pre']

    t_stat, p_val = stats.ttest_ind(treatment_vals, control_vals)

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# The issue: parallel trends assumption is violated
# Treatment branches were already improving faster than control before the switch

# Let's estimate what the "true" effect might be after adjusting for pre-trend differences

# Pre-switch period: weeks 6-14
# The treatment group was gaining 0.213 points/week more than control

# If this pre-trend continued, what would we expect post-switch without the intervention?
pre_trend_diff = 0.213  # treatment was gaining this much more per week already

# Post-switch period: weeks 15-22 (8 weeks)
# Expected gap increase from pre-trend alone: 0.213 * 8 = 1.70 points

# Actual observed gap increase post-switch:
post_data = df[df['week'] >= 15]
pre_data = df[df['week'] < 15]

# Average gap pre-switch (weeks 6-14)
pre_gap = pre_data.groupby('week').apply(lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()).mean()

# Average gap post-switch (weeks 15-22)  
post_gap = post_data.groupby('week').apply(lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()).mean()

actual_gap_increase = post_gap - pre_gap
expected_gap_from_trend = pre_trend_diff * 8  # 8 weeks post-switch
adjusted_effect = actual_gap_increase - expected_gap_from_trend

print("=== Adjusting for Pre-Existing Trend Differences ===")
print(f"Pre-switch gap (avg weeks 6-14): {pre_gap:.2f} points")
print(f"Post-switch gap (avg weeks 15-22): {post_gap:.2f} points")
print(f"Actual gap increase: {actual_gap_increase:.2f} points")
print(f"Pre-trend difference: {pre_trend_diff:.3f} points/week")
print(f"Expected gap increase from pre-trend alone (8 weeks): {expected_gap_from_trend:.2f} points")
print(f"\n**Trend-adjusted effect estimate: {adjusted_effect:.2f} points**")

# Alternative: Use only the first 2 weeks post-switch (less time for trends to diverge)
early_post = df[(df['week'] >= 15) & (df['week'] <= 16)].copy()
early_post['interaction'] = early_post['switched_branch'] * early_post['after_switch']
model_early = ols('service_score ~ switched_branch + after_switch + interaction', 
                  data=early_post).fit(cov_type='cluster', cov_kwds={'groups': early_post['branch_id']})

print(f"\n=== DiD Using Only Weeks 15-16 ===")
print(f"Effect: {model_early.params['interaction']:.2f} (95% CI: {model_early.conf_int().loc['interaction'][0]:.2f} to {model_early.conf_int().loc['interaction'][1]:.2f})")

# Another approach: synthetic control style - match on pre-period levels
# Compare branches with similar pre-switch service scores
pre_avg_by_branch = df[df['week'] < 15].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()

# Match treatment branches to control branches with similar pre-averages
print("\n=== Matched Analysis (by pre-switch service score) ===")

# Create bins for pre-switch averages
pre_avg_by_branch['pre_bin'] = pd.cut(pre_avg_by_branch['service_score'], bins=10)

# For each bin, calculate the post-pre difference for treatment and control
post_avg_by_branch = df[df['week'] >= 15].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
post_avg_by_branch = post_avg_by_branch.rename(columns={'service_score': 'post_score'})

matched = pre_avg_by_branch.merge(post_avg_by_branch, on=['branch_id', 'switched_branch'])
matched['change'] = matched['post_score'] - matched['service_score']

# Within each bin, calculate DiD
bin_results = []
for bin_name in matched['pre_bin'].unique():
    bin_data = matched[matched['pre_bin'] == bin_name]
    if len(bin_data[bin_data['switched_branch']==1]) > 0 and len(bin_data[bin_data['switched_branch']==0]) > 0:
        treat_change = bin_data[bin_data['switched_branch']==1]['change'].mean()
        control_change = bin_data[bin_data['switched_branch']==0]['change'].mean()
        did = treat_change - control_change
        n_treat = len(bin_data[bin_data['switched_branch']==1])
        n_control = len(bin_data[bin_data['switched_branch']==0])
        bin_results.append({'bin': str(bin_name), 'DiD': did, 'n_treat': n_treat, 'n_control': n_control})

bin_df = pd.DataFrame(bin_results)
print(bin_df.to_string())
print(f"\nWeighted average DiD across bins: {(bin_df['DiD'] * bin_df['n_treat']).sum() / bin_df['n_treat'].sum():.2f}")
