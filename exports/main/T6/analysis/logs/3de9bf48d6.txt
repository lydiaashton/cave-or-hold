import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Calculate means for each group-period combination
print("=== Group-Period Means ===")
group_means = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(group_means)

# Calculate the difference-in-differences manually
print("\n=== Manual DiD Calculation ===")
# Control group (switched_branch=0)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_change = control_after - control_before

# Treatment group (switched_branch=1)
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_change = treat_after - treat_before

# Difference in differences
did = treat_change - control_change

print(f"Control group: {control_before:.2f} -> {control_after:.2f} (change: {control_change:.2f})")
print(f"Treatment group: {treat_before:.2f} -> {treat_after:.2f} (change: {treat_change:.2f})")
print(f"Difference-in-Differences estimate: {did:.2f}")

# Now run regression with DiD
print("\n=== Regression Analysis ===")

# Basic DiD model
model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("Model 1: Basic DiD (no controls)")
print(model1.summary())

# DiD with controls
model2 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with controls (staffing_index, visits_index)")
print(model2.summary())

# Clustered standard errors by branch
print("\n=== Clustered Standard Errors (by branch) ===")
model3 = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model3.summary())

model4 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 4: DiD with controls, clustered SEs")
print(model4.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: estimate effect for each week relative to switch
# Create relative week variable (week 15 = 0, week 14 = -1, etc.)
df['relative_week'] = df['week'] - 15

# Create dummies for each relative week
for rw in range(-9, 8):  # weeks -9 to +7 relative to switch
    df[f'rel_week_{rw}'] = (df['relative_week'] == rw).astype(int)

# Event study regression (omitting week -1 as reference)
# Using only post-switch interaction for treatment effect
event_study_vars = [f'rel_week_{rw}' for rw in range(-9, 8) if rw != -1]
formula = 'service_score ~ switched_branch * (' + ' + '.join(event_study_vars) + ')'

# This is complex, let me do a simpler approach - estimate treatment effect by relative week
print("=== Event Study Analysis ===")
results = []
for rw in range(-9, 8):
    if rw == -1:  # reference period
        continue
    subset = df[(df['relative_week'] >= -9) & (df['relative_week'] <= 7)]
    subset['treatment_period'] = (subset['relative_week'] == rw).astype(int)
    model = ols('service_score ~ switched_branch * treatment_period', data=subset).fit(cov_type='cluster', cov_kwds={'groups': subset['branch_id']})
    coef = model.params['switched_branch:treatment_period']
    se = model.bse['switched_branch:treatment_period']
    pval = model.pvalues['switched_branch:treatment_period']
    results.append({'relative_week': rw, 'coef': coef, 'se': se, 'pval': pval})

event_df = pd.DataFrame(results)
print(event_df)

# Plot event study
plt.figure(figsize=(12, 6))
plt.errorbar(event_df['relative_week'], event_df['coef'], 
             yerr=1.96*event_df['se'], 
             fmt='o', capsize=5, markersize=8, linewidth=2)
plt.axhline(y=0, color='gray', linestyle='--', alpha=0.5)
plt.axvline(x=-0.5, color='red', linestyle='-', linewidth=2, label='Switch Point')
plt.xlabel('Relative Week (0 = Week 15, first week after switch)', fontsize=12)
plt.ylabel('Treatment Effect (Difference-in-Differences)', fontsize=12)
plt.title('Event Study: Treatment Effect by Week', fontsize=14, fontweight='bold')
plt.xticks(range(-9, 8))
plt.grid(True, alpha=0.3)
plt.legend()
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")

# DiD excluding week 14 (to check if pre-trend issue affects results)
print("\n=== DiD Excluding Week 14 ===")
df_no14 = df[df['week'] != 14].copy()
model_no14 = ols('service_score ~ switched_branch * after_switch', data=df_no14).fit(cov_type='cluster', cov_kwds={'groups': df_no14['branch_id']})
print(model_no14.summary())
print(f"\nTreatment effect (excluding week 14): {model_no14.params['switched_branch:after_switch']:.2f} (p={model_no14.pvalues['switched_branch:after_switch']:.4f})")

# DiD using only weeks 6-13 as pre-period (excluding week 14)
print("\n=== DiD with Clean Pre-Period (Weeks 6-13 only) ===")
df_clean = df[df['week'] <= 13].copy()

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# More careful event study with proper relative time dummies
df['relative_week'] = df['week'] - 15  # week 15 = 0

# Create the event study regression properly
# We'll use week -1 as the reference category
event_study_results = []

for rw in range(-9, 8):
    if rw == -1:
        continue
    
    # Create indicator for this specific relative week
    df[f'is_rel_week_{rw}'] = (df['relative_week'] == rw).astype(int)
    
# Build formula for event study
# service_score ~ switched_branch + sum of relative week dummies + switched_branch * relative week dummies
# Actually, let's do it differently - estimate the interaction for each period

# Simpler approach: for each relative week, estimate the gap between treatment and control
# relative to the gap in week -1

gaps_by_week = df.groupby(['relative_week', 'switched_branch'])['service_score'].mean().unstack()
gaps_by_week['gap'] = gaps_by_week[1] - gaps_by_week[0]

# Normalize to week -1 (reference)
reference_gap = gaps_by_week.loc[-1, 'gap']
gaps_by_week['normalized_gap'] = gaps_by_week['gap'] - reference_gap

print("=== Gaps Between Treatment and Control by Relative Week ===")
print(gaps_by_week[['gap', 'normalized_gap']])

# Now let's do a proper event study regression with all relative week dummies
# Using the approach from Callaway & Sant'Anna or similar

# Create dummies for each relative week
for rw in range(-9, 8):
    df[f'd_{rw}'] = (df['relative_week'] == rw).astype(int)

# Regression with all relative week interactions (omitting -1)
rel_weeks = [f'd_{rw}' for rw in range(-9, 8) if rw != -1]
interaction_terms = [f'switched_branch:{d}' for d in rel_weeks]

# Build formula
formula_parts = ['switched_branch'] + rel_weeks + interaction_terms
formula = 'service_score ~ ' + ' + '.join(formula_parts)

model_es = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n=== Event Study Regression Results ===")
es_results = []
for rw in range(-9, 8):
    if rw == -1:
        es_results.append({'relative_week': rw, 'coef': 0, 'se': 0, 'pval': np.nan, 'ci_low': 0, 'ci_high': 0})
        continue
    coef_key = f'switched_branch:d_{rw}'
    coef = model_es.params[coef_key]
    se = model_es.bse[coef_key]
    pval = model_es.pvalues[coef_key]
    es_results.append({
        'relative_week': rw, 
        'coef': coef, 
        'se': se, 
        'pval': pval,
        'ci_low': coef - 1.96*se,
        'ci_high': coef + 1.96*se
    })

es_df = pd.DataFrame(es_results)

import pandas as pd

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Final analysis - clean and simple
print("=" * 70)
print("FINAL ANALYSIS FOR BOARD SUMMARY")
print("=" * 70)

# Main DiD estimate
model_main = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', 
                 data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

treatment_effect = model_main.params['switched_branch:after_switch']
treatment_se = model_main.bse['switched_branch:after_switch']
treatment_pval = model_main.pvalues['switched_branch:after_switch']
ci_low = treatment_effect - 1.96 * treatment_se
ci_high = treatment_effect + 1.96 * treatment_se

print(f"\nMAIN FINDING:")
print(f"  Treatment effect: +{treatment_effect:.2f} points on service score")
print(f"  95% Confidence Interval: [{ci_low:.2f}, {ci_high:.2f}]")
print(f"  P-value: {treatment_pval:.4f}")
print(f"  Statistically significant: {'YES' if treatment_pval < 0.05 else 'NO'}")

# Effect size
baseline_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
baseline_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
print(f"\nBASELINE SCORES (Pre-switch):")
print(f"  Treatment branches: {baseline_treatment:.1f}")
print(f"  Control branches: {baseline_control:.1f}")
print(f"  Effect as % of baseline: {treatment_effect/baseline_treatment*100:.1f}%")

# Post-switch scores
post_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
post_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
print(f"\nPOST-SWITCH SCORES:")
print(f"  Treatment branches: {post_treatment:.1f}")
print(f"  Control branches: {post_control:.1f}")

# Robustness check - excluding week 14
df_no14 = df[df['week'] != 14].copy()
model_robust = ols('service_score ~ switched_branch * after_switch', 
                   data=df_no14).fit(cov_type='cluster', cov_kwds={'groups': df_no14['branch_id']})
treatment_effect_robust = model_robust.params['switched_branch:after_switch']
print(f"\nROBUSTNESS CHECK (excluding week 14):")
print(f"  Treatment effect: +{treatment_effect_robust:.2f} points")
