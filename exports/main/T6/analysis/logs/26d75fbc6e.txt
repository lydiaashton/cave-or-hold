import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary table: mean service scores by group and period
summary_table = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("=== Service Score by Group and Period ===")
print(summary_table)
print()

# Calculate the DiD estimate manually
# Control group (switched_branch=0): before and after
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_change = control_after - control_before

# Treatment group (switched_branch=1): before and after
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_change = treat_after - treat_before

# DiD estimate
did_estimate = treat_change - control_change

print("=== Difference-in-Differences Calculation ===")
print(f"Control group (non-switched): Before={control_before:.2f}, After={control_after:.2f}, Change={control_change:.2f}")
print(f"Treatment group (switched): Before={treat_before:.2f}, After={treat_after:.2f}, Change={treat_change:.2f}")
print(f"DiD Estimate (Treatment Effect): {did_estimate:.2f}")
print()

# Run regression with robust standard errors
# Basic DiD model
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='HC3')
print("=== Basic DiD Regression ===")
print(model1.summary().tables[1])
print()

# DiD model with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='HC3')
print("=== DiD Regression with Controls ===")
print(model2.summary().tables[1])
print()

# Check parallel trends: plot average service scores by week for both groups
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
print("=== Weekly Averages by Group ===")
print(weekly_avg.pivot(index='week', columns='switched_branch', values='service_score').round(2))

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check parallel trends more carefully
# Calculate the difference between treatment and control groups by week
weekly_diff = df.groupby('week').apply(
    lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()
).reset_index()
weekly_diff.columns = ['week', 'diff']

print("=== Pre-treatment trend (difference between treatment and control by week) ===")
print(weekly_diff)
print()

# Check if pre-trends are significantly different
pre_period = weekly_diff[weekly_diff['week'] < 15]
post_period = weekly_diff[weekly_diff['week'] >= 15]

print("Pre-period average difference (weeks 6-14):", pre_period['diff'].mean())
print("Post-period average difference (weeks 15-22):", post_period['diff'].mean())
print()

# Event study: create relative week indicators
# Week 14 is the last pre-switch week, so let's use that as reference
df['relative_week'] = df['week'] - 14

# Create event study dummies for each week relative to switch (excluding reference week 0)
# We'll use week -1 (week 13) as the reference
event_study_vars = []
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != -1:  # exclude reference week
        var_name = f'week_{w}'
        df[var_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(var_name)

# Run event study regression
event_study_formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_study_vars)
model_es = ols(event_study_formula, data=df).fit(cov_type='HC3')

print("=== Event Study Coefficients (relative to week 13) ===")
es_results = []
for w in range(-8, 9):
    if w != -1:
        var_name = f'week_{w}'
        coef = model_es.params[var_name]
        se = model_es.bse[var_name]
        pval = model_es.pvalues[var_name]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        es_results.append({'relative_week': w, 'coef': coef, 'se': se, 'pval': pval, 'ci_low': ci_low, 'ci_high': ci_high})

es_df = pd.DataFrame(es_results)
print(es_df.round(3))
print()

# Check if pre-treatment coefficients are jointly significant
pre_treatment_vars = [f'week_{w}' for w in range(-8, 0) if w != -1]
print("Pre-treatment coefficients (should be close to 0 if parallel trends hold):")
for var in pre_treatment_vars:
    print(f"  {var}: {model_es.params[var]:.3f} (p={model_es.pvalues[var]:.3f})")

# Model with branch fixed effects
# We need to create branch dummies

    df_fe[col] = branch_dummies[col]

# Build formula with branch fixed effects
branch_vars = branch_dummies.columns.tolist()
fe_formula = 'service_score ~ after_switch + treatment + staffing_index + visits_index + ' + ' + '.join(branch_vars)

# This will be a large model, let's try it

# Just show the key coefficients

for var in key_vars:

    se = model_fe.bse[var]
    pval = model_fe.pvalues[var]
    print(f"{var}: coef={coef:.3f}, se={se:.3f}, p={pval:.3f}")
