import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch fixed effects
# First, let's do a simple OLS with cluster-robust standard errors
model = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()
print("=== Simple DiD Regression (no fixed effects) ===")
print(model.summary())

# Now with branch fixed effects
# We need to create dummy variables for each branch
df_with_fe = df.copy()
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df_with_fe, branch_dummies], axis=1)

# Build the formula with all branch dummies
branch_cols = [c for c in branch_dummies.columns]
formula = f'service_score ~ switched_branch + after_switch + treated_post + {" + ".join(branch_cols)}'
model_fe = ols(formula, data=df_with_fe).fit(cov_type='cluster', cov_kwds={'groups': df_with_fe['branch_id']})
print("\n=== DiD Regression with Branch Fixed Effects (clustered SEs) ===")
print(model_fe.summary())

# Extract the key coefficient
did_coef = model_fe.params['treated_post']
did_se = model_fe.bse['treated_post']
did_pvalue = model_fe.pvalues['treated_post']
print(f"\n=== Key Result ===")
print(f"DiD Coefficient: {did_coef:.3f}")
print(f"Standard Error: {did_se:.3f}")
print(f"P-value: {did_pvalue:.4f}")
print(f"95% CI: [{did_coef - 1.96*did_se:.3f}, {did_coef + 1.96*did_se:.3f}]")

treat_change = treat_after - treat_before
