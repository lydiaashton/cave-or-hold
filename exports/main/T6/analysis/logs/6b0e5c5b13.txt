import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Summary table: mean service scores by group and period
print("=" * 60)
print("MEAN SERVICE SCORES BY GROUP AND PERIOD")
print("=" * 60)

# Before period (after_switch = 0)
before = df[df['after_switch'] == 0]
before_control = before[before['switched_branch'] == 0]['service_score'].mean()
before_treated = before[before['switched_branch'] == 1]['service_score'].mean()

# After period (after_switch = 1)
after = df[df['after_switch'] == 1]
after_control = after[after['switched_branch'] == 0]['service_score'].mean()
after_treated = after[after['switched_branch'] == 1]['service_score'].mean()

print(f"\nControl Group (no switch):")
print(f"  Before (weeks 6-14): {before_control:.2f}")
print(f"  After (weeks 15-22): {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")

print(f"\nTreatment Group (switched):")
print(f"  Before (weeks 6-14): {before_treated:.2f}")
print(f"  After (weeks 15-22): {after_treated:.2f}")
print(f"  Change: {after_treated - before_treated:.2f}")

# Difference-in-differences estimate
did_estimate = (after_treated - before_treated) - (after_control - before_control)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

# Now run the regression
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD model
model1 = ols('service_score ~ treated + post + DiD', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# DiD model with controls
model2 = ols('service_score ~ treated + post + DiD + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with controls (staffing_index, visits_index)")
print(model2.summary().tables[1])

# Cluster-robust standard errors by branch
print("\n" + "=" * 60)
print("CLUSTER-ROBUST STANDARD ERRORS (by branch)")
print("=" * 60)
model2_cluster = model2.get_robustcov_results(cov_type='cluster', clusters=df['branch_id'])
print(model2_cluster.summary().tables[1])

import pandas as pd

import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Basic DiD model
model = ols('service_score ~ treated + post + DiD', data=df).fit()
model_cluster = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# One-tailed p-value (since we had a directional hypothesis)
did_coef = model_cluster.params[3]
did_se = model_cluster.bse[3]
two_tailed_p = model_cluster.pvalues[3]

# One-tailed p-value for positive effect
from scipy import stats
t_stat = did_coef / did_se
one_tailed_p = 1 - stats.t.cdf(t_stat, df=33)  # 34 branches - 1

print(f"One-tailed p-value (testing for positive effect): {one_tailed_p:.3f}")
