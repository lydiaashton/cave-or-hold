# Basic DiD regression
print("=== Difference-in-Differences Regression ===")
print("\nModel 1: Basic DiD (no controls)")
model1 = ols('service_score ~ switched_branch + after_switch + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model1.summary())

print("\n\nModel 2: DiD with controls (staffing_index, visits_index)")
model2 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks since switch)
# Week 15 is the first week after switch, so relative_week = week - 14
# This makes week 14 = 0 (last pre week), week 15 = 1, etc.
# But for event study, we want to use week 14 as reference
df['relative_week'] = df['week'] - 14  # week 14 becomes 0

# Event study regression
# Create dummies for each week relative to switch, excluding week 14 as reference
event_study_df = df.copy()
for w in range(-8, 9):  # weeks -8 to 8 relative to week 14
    if w != 0:  # exclude reference week
        event_study_df[f'week_{w}'] = ((event_study_df['relative_week'] == w) & (event_study_df['switched_branch'] == 1)).astype(int)

# Build formula
week_vars = [f'week_{w}' for w in range(-8, 9) if w != 0]
formula = 'service_score ~ C(branch_id) + ' + ' + '.join(week_vars) + ' + staffing_index + visits_index'

model_es = ols(formula, data=event_study_df).fit(cov_type='cluster', cov_kwds={'groups': event_study_df['branch_id']})

# Extract coefficients and confidence intervals
results = []
for w in range(-8, 9):
    if w == 0:
        results.append({'relative_week': w, 'coef': 0, 'ci_lower': 0, 'ci_upper': 0})
    else:
        var_name = f'week_{w}'
        coef = model_es.params[var_name]
        ci = model_es.conf_int().loc[var_name]
        results.append({'relative_week': w, 'coef': coef, 'ci_lower': ci[0], 'ci_upper': ci[1]})

es_results = pd.DataFrame(results)
print("=== Event Study Results ===")
print(es_results)

# Plot
fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(es_results['relative_week'], es_results['coef'], 
            yerr=[es_results['coef'] - es_results['ci_lower'], es_results['ci_upper'] - es_results['coef']],
            fmt='o', capsize=5, label='Treatment effect')
ax.axhline(y=0, color='red', linestyle='--', label='No effect')
ax.axvline(x=0, color='gray', linestyle='-', alpha=0.5, label='Switch week')
ax.set_xlabel('Weeks relative to switch (0 = week 14, last pre-switch week)')
ax.set_ylabel('Service Score Effect')
ax.set_title('Event Study: Effect of New Procedure on Service Score')
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved to event_study.png")

event_study_df = df.copy()
for w in range(-8, 9):
    if w != 0:  # exclude reference week (week 14)
        # Use prefix to avoid negative sign issues
        var_name = f'rw_{abs(w)}_{"neg" if w < 0 else "pos"}'
        event_study_df[var_name] = ((event_study_df['relative_week'] == w) & (event_study_df['switched_branch'] == 1)).astype(int)

# Build formula
week_vars = [f'rw_{abs(w)}_{"neg" if w < 0 else "pos"}' for w in range(-8, 9) if w != 0]
formula = 'service_score ~ C(branch_id) + ' + ' + '.join(week_vars) + ' + staffing_index + visits_index'

model_es = ols(formula, data=event_study_df).fit(cov_type='cluster', cov_kwds={'groups': event_study_df['branch_id']})
