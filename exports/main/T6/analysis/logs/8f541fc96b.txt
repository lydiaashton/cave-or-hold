import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study / dynamic DiD - check effect by week relative to treatment
df['weeks_relative'] = df['week'] - 15  # Week 15 is when switch happened (0 = treatment week)

# Create indicators for each week relative to treatment, interacted with treatment
# Using week 14 as reference (last pre-treatment week)
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=False)
df = pd.concat([df, week_dummies], axis=1)

# Run regression with week fixed effects and treatment interaction
# This is a simplified event study
print("=== Event Study: Treatment Effect by Week ===")

# For each post-treatment week, estimate the differential effect
results = []
for wk in range(15, 23):
    df[f'post_wk_{wk}'] = (df['week'] == wk).astype(int)
    df[f'interact_post_wk_{wk}'] = df['switched_branch'] * df[f'post_wk_{wk}']

# Run regression with all post-treatment week interactions
formula = 'service_score ~ switched_branch + '
formula += ' + '.join([f'week_{w}' for w in range(6, 15)]) + ' + '  # Pre-treatment week FEs
formula += ' + '.join([f'interact_post_wk_{w}' for w in range(15, 23)])  # Post-treatment interactions
formula += ' + C(week)'  # Full week fixed effects

# Simpler approach: just DiD with week FE
model_week_fe = ols('service_score ~ switched_branch * after_switch + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD with Week Fixed Effects (Cluster-Robust SE) ===")
print(model_week_fe.summary())

# Branch fixed effects model (more rigorous)
print("\n=== DiD with Branch and Week Fixed Effects ===")
model_fe = ols('service_score ~ switched_branch * after_switch + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_fe.summary())

# Check balance - are there systematic differences between switched and non-switched branches?
print("\n=== Branch Characteristics Comparison ===")
branch_summary = df.groupby('switched_branch').agg({
    'service_score': ['mean', 'std'],
    'staffing_index': ['mean', 'std'],
    'visits_index': ['mean', 'std']
}).round(3)
print(branch_summary)

# Per-branch average before and after
branch_avg = df.groupby(['branch_id', 'switched_branch']).agg({
    'service_score': 'mean',
    'after_switch': 'mean'  # This gives proportion of post-period observations
}).reset_index()

# Actually, let's compute before/after for each branch
branch_before = df[df['after_switch'] == 0].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_before.columns = ['branch_id', 'switched_branch', 'score_before']
branch_after = df[df['after_switch'] == 1].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_after.columns = ['branch_id', 'switched_branch', 'score_after']

branch_change = branch_before.merge(branch_after, on=['branch_id', 'switched_branch'])
branch_change['change'] = branch_change['score_after'] - branch_change['score_before']

print("\n=== Distribution of Branch-Level Changes ===")
print(branch_change.groupby('switched_branch')['change'].describe())

# T-test on branch-level changes
from scipy import stats
control_changes = branch_change[branch_change['switched_branch'] == 0]['change']
treatment_changes = branch_change[branch_change['switched_branch'] == 1]['change']

t_stat, p_val = stats.ttest_ind(treatment_changes, control_changes)
print(f"\n=== T-test on Branch-Level Changes ===")
print(f"Control mean change: {control_changes.mean():.3f}")
print(f"Treatment mean change: {treatment_changes.mean():.3f}")
print(f"Difference: {treatment_changes.mean() - control_changes.mean():.3f}")
print(f"T-statistic: {t_stat:.3f}, p-value: {p_val:.3f}")

import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# Recalculate the key DiD estimate with precise p-value
from statsmodels.formula.api import ols

# Basic DiD with cluster-robust SE
model = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

did_coef = model.params['switched_branch:after_switch']
did_se = model.bse['switched_branch:after_switch']
did_p = model.pvalues['switched_branch:after_switch']
did_ci_lower = model.conf_int().loc['switched_branch:after_switch', 0]
did_ci_upper = model.conf_int().loc['switched_branch:after_switch', 1]

print("=== Precise DiD Results ===")
print(f"Point estimate: {did_coef:.2f} points")
print(f"Standard error: {did_se:.2f}")
print(f"95% Confidence interval: [{did_ci_lower:.2f}, {did_ci_upper:.2f}]")
print(f"Two-sided p-value: {did_p:.3f}")
print(f"One-sided p-value (effect > 0): {did_p/2:.3f}")

# What does this mean practically?
print(f"\n=== Interpretation ===")
print(f"The 95% CI includes zero, so we cannot reject the null hypothesis at α=0.05")
print(f"However, the entire CI is: [{did_ci_lower:.1f} to {did_ci_upper:.1f}]")
print(f"This means plausible values range from a {abs(did_ci_lower):.1f} point DECREASE to a {did_ci_upper:.1f} point INCREASE")

# Effect size
overall_std = df['service_score'].std()
print(f"\nEffect size (Cohen's d): {did_coef / overall_std:.3f}")
print(f"Convention: 0.2=small, 0.5=medium, 0.8=large")

# Power analysis approximation
n_treatment = 20  # branches
n_control = 14  # branches

print(f"\n=== Study Design ===")

print(f"Control branches: {n_control}")

print(f"Total post-treatment observations: {n_treatment * post_weeks} vs {n_control * post_weeks}")
