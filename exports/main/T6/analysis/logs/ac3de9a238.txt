import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# Basic DiD model
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("Basic DiD Regression:")
print(model.summary())

# DiD with cluster-robust standard errors (clustered by branch)
model_clustered = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD with Cluster-Robust SE (by branch):")
print(model_clustered.summary())

# Add control variables
model_controls = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD with Controls and Cluster-Robust SE:")
print(model_controls.summary())

# Check parallel trends - look at pre-period trends
pre_data = df[df['after_switch'] == 0].copy()
pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()

# Regression of service_score on week for each group in pre-period
pre_treatment = pre_data[pre_data['switched_branch'] == 1]
pre_control = pre_data[pre_data['switched_branch'] == 0]

trend_treatment = ols('service_score ~ week', data=pre_treatment).fit()
trend_control = ols('service_score ~ week', data=pre_control).fit()

print("\n\nPre-period trends:")

import pandas as pd

df = pd.read_csv('branches.csv')

# Create a visualization of trends by week for both groups
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()

# Plot
fig, ax = plt.subplots(figsize=(12, 6))

for group in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == group]
    label = 'Treatment (New Procedure)' if group == 1 else 'Control (Old Procedure)'
    color = '#2E86AB' if group == 1 else '#A23B72'
    ax.plot(subset['week'], subset['service_score'], marker='o', linewidth=2, markersize=6, label=label, color=color)

# Add vertical line at switch point
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch Point (Week 15)')
ax.axvspan(14.5, 22, alpha=0.1, color='gray', label='Post-Switch Period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Service Score', fontsize=12)
ax.set_title('Service Score Trends: Treatment vs Control Groups', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=10)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('trends_plot.png', dpi=150)
print("Plot saved as trends_plot.png")

# Let's also check pre-trends more formally with an event study approach
# Create relative week variable (weeks relative to switch at week 15)
df['relative_week'] = df['week'] - 15

# Only use pre-period for parallel trends check
pre_df = df[df['week'] < 15].copy()

# Run regression with week fixed effects interacted with treatment
# This tests if pre-trends differ
pre_df['switched_branch'] = pre_df['switched_branch'].astype(int)
pre_df['week_cat'] = pre_df['week'].astype(str)

# Create week dummies interacted with treatment
for w in range(6, 15):
    pre_df[f'week_{w}'] = (pre_df['week'] == w).astype(int)
    pre_df[f'week_{w}_treat'] = pre_df[f'week_{w}'] * pre_df['switched_branch']

# Use week 6 as baseline
treat_vars = [f'week_{w}_treat' for w in range(7, 15)]
week_vars = [f'week_{w}' for w in range(7, 15)]

formula = 'service_score ~ ' + ' + '.join(treat_vars + week_vars)
model = ols(formula, data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})

print("\n\nPre-Trends Test (Treatment x Week interactions, week 6 as baseline):")
print("Testing if treatment and control had different trends before the switch")
print("\nCoefficients for treatment x week interactions:")
for w in range(7, 15):
    coef = model.params[f'week_{w}_treat']
    pval = model.pvalues[f'week_{w}_treat']
    print(f"Week {w}: coef={coef:.3f}, p={pval:.3f}")

# Joint test
from statsmodels.stats.anova import anova_lm
# Test if all interaction terms are jointly zero
hypotheses = [f"week_{w}_treat = 0" for w in range(7, 15)]
from statsmodels.sandbox.regression.predstd import wald_test
wald = model.wald_test(hypotheses)
print(f"\nJoint test of all pre-trend interactions: F={wald.statistic:.2f}, p={wald.pvalue:.4f}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: estimate effect for each week relative to switch
# Create relative week (week 15 is the switch, so relative_week = week - 15)
df['relative_week'] = df['week'] - 15

# Create dummies for each relative week
for rw in range(-9, 8):  # weeks -9 to +7 (week 6 to 22)
    df[f'rel_week_{rw}'] = (df['relative_week'] == rw).astype(int)
    df[f'rel_week_{rw}_treat'] = df[f'rel_week_{rw}'] * df['switched_branch']

# Use relative week -9 (week 6) as baseline
treat_vars = [f'rel_week_{rw}_treat' for rw in range(-8, 8)]
week_vars = [f'rel_week_{rw}' for rw in range(-8, 8)]

formula = 'service_score ~ ' + ' + '.join(treat_vars + week_vars)
model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients and confidence intervals for treatment interactions
results = []
for rw in range(-8, 8):
    var = f'rel_week_{rw}_treat'
    coef = model.params[var]
    ci_low = model.conf_int().loc[var, 0]
    ci_high = model.conf_int().loc[var, 1]
    pval = model.pvalues[var]
    results.append({'relative_week': rw, 'coef': coef, 'ci_low': ci_low, 'ci_high': ci_high, 'pval': pval})

event_study = pd.DataFrame(results)
print("Event Study Results (coefficient = treatment effect for each week relative to switch):")
print(event_study.to_string(index=False))

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))

# Pre-period (negative relative weeks)
pre = event_study[event_study['relative_week'] < 0]
ax.errorbar(pre['relative_week'], pre['coef'], 
            yerr=[pre['coef'] - pre['ci_low'], pre['ci_high'] - pre['coef']],
            fmt='o', color='#A23B72', label='Pre-Switch', capsize=5, markersize=8)

# Post-period (non-negative relative weeks)
post = event_study[event_study['relative_week'] >= 0]
ax.errorbar(post['relative_week'], post['coef'], 
            yerr=[post['coef'] - post['ci_low'], post['ci_high'] - post['coef']],
            fmt='o', color='#2E86AB', label='Post-Switch', capsize=5, markersize=8)

ax.axhline(y=0, color='gray', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='gray', linestyle='--', linewidth=2, label='Switch Point')
ax.set_xlabel('Weeks Relative to Switch (0 = Week 15)', fontsize=12)
ax.set_ylabel('Treatment Effect (Service Score Points)', fontsize=12)
ax.set_title('Event Study: Weekly Treatment Effects on Service Score', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=10)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-9, 8))

plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\n\nEvent study plot saved as event_study.png")

print(f"\nAverage post-switch effect: {post_effects.mean():.2f} points")
print(f"Range of post-switch effects: [{post_effects.min():.2f}, {post_effects.max():.2f}]")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: estimate effect for each week relative to switch
# Create relative week (week 15 is the switch, so relative_week = week - 15)
df['relative_week'] = df['week'] - 15

# Create dummies for each relative week with safe names
for rw in range(-9, 8):  # weeks -9 to +7 (week 6 to 22)
    safe_name = f'w{rw}'  # e.g., w-9, w-8, ..., w7
    df[f'{safe_name}'] = (df['relative_week'] == rw).astype(int)
    df[f'{safe_name}_treat'] = df[f'{safe_name}'] * df['switched_branch']

# Use relative week -9 (week 6) as baseline
treat_vars = [f'w{rw}_treat' for rw in range(-8, 8)]
week_vars = [f'w{rw}' for rw in range(-8, 8)]

formula = 'service_score ~ ' + ' + '.join(treat_vars + week_vars)
model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients and confidence intervals for treatment interactions
results = []
for rw in range(-8, 8):
    var = f'w{rw}_treat'
    coef = model.params[var]
    ci_low = model.conf_int().loc[var, 0]
    ci_high = model.conf_int().loc[var, 1]
    pval = model.pvalues[var]
    results.append({'relative_week': rw, 'coef': coef, 'ci_low': ci_low, 'ci_high': ci_high, 'pval': pval})

event_study = pd.DataFrame(results)
print("Event Study Results (coefficient = treatment effect for each week relative to switch):")
print(event_study.to_string(index=False))

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))

# Pre-period (negative relative weeks)
pre = event_study[event_study['relative_week'] < 0]
ax.errorbar(pre['relative_week'], pre['coef'], 
            yerr=[pre['coef'] - pre['ci_low'], pre['ci_high'] - pre['coef']],
            fmt='o', color='#A23B72', label='Pre-Switch', capsize=5, markersize=8)

# Post-period (non-negative relative weeks)
post = event_study[event_study['relative_week'] >= 0]
ax.errorbar(post['relative_week'], post['coef'], 
            yerr=[post['coef'] - post['ci_low'], post['ci_high'] - post['coef']],
            fmt='o', color='#2E86AB', label='Post-Switch', capsize=5, markersize=8)

ax.axhline(y=0, color='gray', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='gray', linestyle='--', linewidth=2, label='Switch Point')
ax.set_xlabel('Weeks Relative to Switch (0 = Week 15)', fontsize=12)
ax.set_ylabel('Treatment Effect (Service Score Points)', fontsize=12)
ax.set_title('Event Study: Weekly Treatment Effects on Service Score', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=10)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-9, 8))

plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\n\nEvent study plot saved as event_study.png")

print(f"\nAverage post-switch effect: {post_effects.mean():.2f} points")
print(f"Range of post-switch effects: [{post_effects.min():.2f}, {post_effects.max():.2f}]")
print(f"All post effects positive? {(post_effects > 0).all()}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: estimate effect for each week relative to switch
# Create relative week (week 15 is the switch, so relative_week = week - 15)
df['relative_week'] = df['week'] - 15

# Create dummies for each relative week with safe names (prefix with 'd' for dummy)
for rw in range(-9, 8):  # weeks -9 to +7 (week 6 to 22)
    safe_name = f'd{rw + 10}'  # shift by 10 to make all positive: d1, d2, ..., d17
    df[f'{safe_name}'] = (df['relative_week'] == rw).astype(int)
    df[f'{safe_name}_treat'] = df[f'{safe_name}'] * df['switched_branch']

# Map back for results
rw_map = {i: i - 10 for i in range(1, 18)}  # d1 -> -9, d2 -> -8, etc.

# Use d1 (relative week -9, i.e., week 6) as baseline
treat_vars = [f'd{i}_treat' for i in range(2, 18)]
week_vars = [f'd{i}' for i in range(2, 18)]

formula = 'service_score ~ ' + ' + '.join(treat_vars + week_vars)
model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients and confidence intervals for treatment interactions
results = []
for i in range(2, 18):
    var = f'd{i}_treat'
    rw = rw_map[i]
    coef = model.params[var]
    ci_low = model.conf_int().loc[var, 0]
    ci_high = model.conf_int().loc[var, 1]
    pval = model.pvalues[var]
    results.append({'relative_week': rw, 'coef': coef, 'ci_low': ci_low, 'ci_high': ci_high, 'pval': pval})

event_study = pd.DataFrame(results)
print("Event Study Results (coefficient = treatment effect for each week relative to switch):")
print(event_study.to_string(index=False))

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))

# Pre-period (negative relative weeks)
pre = event_study[event_study['relative_week'] < 0]
ax.errorbar(pre['relative_week'], pre['coef'], 
            yerr=[pre['coef'] - pre['ci_low'], pre['ci_high'] - pre['coef']],
            fmt='o', color='#A23B72', label='Pre-Switch', capsize=5, markersize=8)

# Post-period (non-negative relative weeks)
post = event_study[event_study['relative_week'] >= 0]
ax.errorbar(post['relative_week'], post['coef'], 
            yerr=[post['coef'] - post['ci_low'], post['ci_high'] - post['coef']],
            fmt='o', color='#2E86AB', label='Post-Switch', capsize=5, markersize=8)

ax.axhline(y=0, color='gray', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='gray', linestyle='--', linewidth=2, label='Switch Point')
ax.set_xlabel('Weeks Relative to Switch (0 = Week 15)', fontsize=12)
ax.set_ylabel('Treatment Effect (Service Score Points)', fontsize=12)
ax.set_title('Event Study: Weekly Treatment Effects on Service Score', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=10)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-9, 8))

plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\n\nEvent study plot saved as event_study.png")

# Summary statistics for post-period effects
post_effects = event_study[event_study['relative_week'] >= 0]['coef']
print(f"\nAverage post-switch effect: {post_effects.mean():.2f} points")
print(f"Range of post-switch effects: [{post_effects.min():.2f}, {post_effects.max():.2f}]")
print(f"All post effects positive? {(post_effects > 0).all()}")
print(f"Post effects significantly > 0? {(post_effects - event_study[event_study['relative_week'] >= 0]['ci_low'] > 0).all()}")
