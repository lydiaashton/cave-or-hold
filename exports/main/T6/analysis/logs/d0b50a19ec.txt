import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# Create groups
# Treatment group: switched_branch = 1
# Control group: switched_branch = 0
# Before: after_switch = 0 (weeks 6-14)
# After: after_switch = 1 (weeks 15-22)

# Summary by group and period
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("Service Score by Group and Period:")
print(summary)
print()

# Calculate means for each combination
means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean()
print("Means:")
print(means)
print()

# Difference-in-Differences calculation
# Treatment group change: after - before for switched_branch=1
# Control group change: after - before for switched_branch=0
# DID = Treatment change - Control change

treatment_before = means.loc[(1, 0)]
treatment_after = means.loc[(1, 1)]
control_before = means.loc[(0, 0)]
control_after = means.loc[(0, 1)]

treatment_change = treatment_after - treatment_before
control_change = control_after - control_before
did_estimate = treatment_change - control_change

print("Difference-in-Differences Analysis:")
print(f"Treatment group (switched) before: {treatment_before:.2f}")
print(f"Treatment group (switched) after: {treatment_after:.2f}")
print(f"Treatment change: {treatment_change:.2f}")
print()
print(f"Control group (not switched) before: {control_before:.2f}")
print(f"Control group (not switched) after: {control_after:.2f}")
print(f"Control change: {control_change:.2f}")
print()
print(f"DID Estimate (Treatment change - Control change): {did_estimate:.2f}")
print()

# Check how many branches are in each group
branch_groups = df.groupby('branch_id')['switched_branch'].first()
print(f"Total branches: {len(branch_groups)}")
print(f"Switched branches: {branch_groups.sum()}")
print(f"Non-switched branches: {(branch_groups == 0).sum()}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DID
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Run DID regression
model = ols('service_score ~ treatment + post + did', data=df).fit()
print("DID Regression (without controls):")
print(model.summary())
print()

# Run DID regression with controls (staffing_index, visits_index)
model_controls = ols('service_score ~ treatment + post + did + staffing_index + visits_index', data=df).fit()
print("DID Regression (with controls):")
print(model_controls.summary())
print()

# Cluster-robust standard errors by branch
model_clustered = ols('service_score ~ treatment + post + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("DID Regression (clustered SE by branch):")
print(model_clustered.summary())

weekly_trends.columns = ['switched_branch', 'week', 'service_score']

# Plot trends
plt.figure(figsize=(12, 6))

for group in [0, 1]:
    subset = weekly_trends[weekly_trends['switched_branch'] == group]
    label = 'Switched branches' if group == 1 else 'Non-switched branches'
    plt.plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2)

plt.axvline(x=14.5, color='red', linestyle='--', label='Switch week (15)')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Score Trends by Branch Group')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('trends.png', dpi=150)
print("Plot saved as trends.png")

# Check parallel trends before the switch (weeks 6-14)
pre_switch = df[df['week'] <= 14]
pre_trends = pre_switch.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

import pandas as pd
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Pre-switch trends analysis - check parallel trends assumption
pre_switch = df[df['week'] <= 14].copy()
pre_trends = pre_switch.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Pivot to see the gap over time
pivot = pre_trends.pivot(index='week', columns='switched_branch', values='service_score')
pivot['gap'] = pivot[1] - pivot[0]

print("Pre-Switch Trends (Weeks 6-14):")
print(pivot)
print()

# Calculate trend (slope) for each group before switch
from scipy import stats

weeks_pre = pre_trends['week'].unique()
treatment_pre = pre_trends[pre_trends['switched_branch'] == 1].sort_values('week')['service_score']
control_pre = pre_trends[pre_trends['switched_branch'] == 0].sort_values('week')['service_score']

# Linear regression for pre-trends
slope_treatment, intercept_treatment, r_t, p_t, se_t = stats.linregress(weeks_pre, treatment_pre.values)
slope_control, intercept_control, r_c, p_c, se_c = stats.linregress(weeks_pre, control_pre.values)

print(f"Treatment group pre-trend slope: {slope_treatment:.3f} per week (p={p_t:.3f})")
print(f"Control group pre-trend slope: {slope_control:.3f} per week (p={p_c:.3f})")
print(f"Difference in slopes: {slope_treatment - slope_control:.3f} per week")
print()

# The gap is widening - let's quantify this
slope_gap, _, _, p_gap, _ = stats.linregress(weeks_pre, pivot['gap'].values)
print(f"Gap widening slope: {slope_gap:.3f} per week (p={p_gap:.3f})")
print()

# Plot pre-trends
plt.figure(figsize=(10, 5))
plt.plot(pivot.index, pivot[0], marker='o', label='Control (Non-Switched)', linewidth=2)
plt.plot(pivot.index, pivot[1], marker='s', label='Treatment (Switched)', linewidth=2)
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Pre-Switch Trends: Parallel Trends Assumption Check')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 15))
plt.tight_layout()
plt.savefig('pre_trends.png', dpi=150)
print("Pre-trends plot saved as pre_trends.png")
