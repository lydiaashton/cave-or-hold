import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary table: Average service scores by group and period
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES SUMMARY TABLE")
print("=" * 60)

# Before period (after_switch = 0)
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()

# After period (after_switch = 1)
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"  After (weeks 15-22): {after_treatment:.2f}")
print(f"  Change: {after_treatment - before_treatment:.2f}")

# DiD estimate
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

# Now run regression
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Better event study with pre-period weeks included
# Create week dummies for all weeks, using week 6 as reference
for w in range(7, 23):
    df[f'week_{w}'] = (df['week'] == w).astype(int)

# Event study with all weeks relative to week 6
# This shows if there's a break in trend at week 15
event_formula = 'service_score ~ switched_branch + ' + ' + '.join([f'week_{w}' for w in range(7, 23)]) + ' + ' + ' + '.join([f'switched_branch:week_{w}' for w in range(7, 23)])

event_model2 = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("="*70)
print("FULL EVENT STUDY (All weeks, reference = week 6)")
print("="*70)
print("\nPre-switch interaction terms (should be ~0 if parallel trends hold):")
for w in range(7, 15):
    term = f'switched_branch:week_{w}'
    coef = event_model2.params[term]
    se = event_model2.bse[term]
    pval = event_model2.pvalues[term]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Week {w}: {coef:6.2f} (SE: {se:.2f}, p: {pval:.3f}) {sig}")

print("\nPost-switch interaction terms (treatment effects):")
for w in range(15, 23):
    term = f'switched_branch:week_{w}'
    coef = event_model2.params[term]
    se = event_model2.bse[term]
    pval = event_model2.pvalues[term]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Week {w}: {coef:6.2f} (SE: {se:.2f}, p: {pval:.3f}) {sig}")

# Alternative: Control for linear pre-trend
df['pre_trend'] = np.where(df['week'] < 15, df['week'] - 6, 8)  # Linear trend that stops at week 14
df['post_trend'] = np.where(df['week'] >= 15, df['week'] - 14, 0)  # Post-switch trend
df['treatment_post_trend'] = df['switched_branch'] * df['post_trend']

# Model with pre-trend control
model_with_trend = ols('service_score ~ switched_branch + after_switch + treatment + pre_trend + post_trend + treatment_post_trend + staffing_index + visits_index', 
                       data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n" + "="*70)
print("MODEL CONTROLLING FOR PRE-EXISTING TREND")
print("="*70)
print(model_with_trend.summary().tables[1])

# Calculate the implied treatment effect at different post-switch weeks
print("\n" + "="*70)
print("IMPLIED TREATMENT EFFECT BY WEEK (controlling for pre-trend)")
print("="*70)
# The treatment effect is: treatment coef + treatment_post_trend coef * weeks_since_switch
treatment_base = model_with_trend.params['treatment']
trend_interaction = model_with_trend.params['treatment_post_trend']
