import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many branches switched?
print("Branches by treatment status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Check balance - average service scores before and after by group
print("\n\nAverage service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the DiD manually
# Treatment group: switched_branch=1
# Control group: switched_branch=0
# Pre: after_switch=0, Post: after_switch=1

treatment_pre = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_post = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
control_pre = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_post = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print(f"\nTreatment (switched) pre: {treatment_pre:.2f}")
print(f"Treatment (switched) post: {treatment_post:.2f}")
print(f"Change in treatment: {treatment_post - treatment_pre:.2f}")
print(f"\nControl (not switched) pre: {control_pre:.2f}")
print(f"Control (not switched) post: {control_post:.2f}")
print(f"Change in control: {control_post - control_pre:.2f}")

did_estimate = (treatment_post - treatment_pre) - (control_post - control_pre)
print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with robust standard errors
# Model 1: Simple DiD
model1 = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='HC3')
print("Model 1: Simple Difference-in-Differences")
print(model1.summary())

# Model 2: DiD with controls
model2 = smf.ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='HC3')
print("\n\nModel 2: DiD with Controls (staffing_index, visits_index)")
print(model2.summary())

# Model 3: With branch fixed effects
model3 = smf.ols('service_score ~ after_switch + treatment + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='HC3')
print("\n\nModel 3: DiD with Branch Fixed Effects")
print(model3.summary().summary_col())

# Extract key coefficient
did_coef = model3.params['treatment']
did_se = model3.bse['treatment']
print(f"\n\nKey DiD estimate (with branch FE and controls): {did_coef:.3f} (SE: {did_se:.3f})")
print(f"95% CI: [{did_coef - 1.96*did_se:.3f}, {did_coef + 1.96*did_se:.3f}]")
