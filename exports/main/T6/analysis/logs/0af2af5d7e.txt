import pandas as pd
import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Difference-in-Differences regression
# service_score = beta0 + beta1*treated + beta2*post + beta3*(treated*post) + error
# beta3 is the DiD estimator - the effect of the new procedure

# Create interaction term
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch fixed effects
# Using OLS with branch as categorical variable
df['branch_id_str'] = df['branch_id'].astype(str)

# Model 1: Simple DiD without fixed effects
model1 = smf.ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()
print("=== Model 1: Simple DiD (no fixed effects) ===")
print(model1.summary())

# Model 2: DiD with branch fixed effects
model2 = smf.ols('service_score ~ switched_branch + after_switch + treated_post + C(branch_id)', data=df).fit()
print("\n=== Model 2: DiD with Branch Fixed Effects ===")
print(model2.summary())

# Model 3: DiD with branch fixed effects and controls
model3 = smf.ols('service_score ~ switched_branch + after_switch + treated_post + C(branch_id) + staffing_index + visits_index', data=df).fit()
print("\n=== Model 3: DiD with Branch FE + Controls ===")
print(model3.summary())
