import pandas as pd
import statsmodels.api as sm
from statsmodels.regression.linear_model import OLS

df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']

# Add branch fixed effects using dummy variables
# This is computationally intensive but gives us the cleanest estimate
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [col for col in df_with_fe.columns if col.startswith('branch_')]
formula = f"service_score ~ switched_branch + after_switch + did + staffing_index + visits_index + {' + '.join(branch_cols)}"

# This will be slow, let's use a simpler approach with linearmodels or just report the within estimator
# Actually, let's use the absorbing approach - regress out branch means first

# Within transformation (demeaning by branch)
df['service_score_demeaned'] = df.groupby('branch_id')['service_score'].transform(lambda x: x - x.mean())
df['switched_branch_demeaned'] = df.groupby('branch_id')['switched_branch'].transform(lambda x: x - x.mean())
df['after_switch_demeaned'] = df.groupby('branch_id')['after_switch'].transform(lambda x: x - x.mean())
df['did_demeaned'] = df.groupby('branch_id')['did'].transform(lambda x: x - x.mean())
df['staffing_index_demeaned'] = df.groupby('branch_id')['staffing_index'].transform(lambda x: x - x.mean())
df['visits_index_demeaned'] = df.groupby('branch_id')['visits_index'].transform(lambda x: x - x.mean())

# Run FE regression on demeaned data (note: switched_branch is time-invariant so it drops out)
X = df[['after_switch_demeaned', 'did_demeaned', 'staffing_index_demeaned', 'visits_index_demeaned']]
X = sm.add_constant(X)
y = df['service_score_demeaned']

model_fe = OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Fixed Effects Model (within estimator, clustered SE):")
print(model_fe.summary())

# Also run a simpler model without controls for comparison
X_simple = df[['after_switch_demeaned', 'did_demeaned']]
X_simple = sm.add_constant(X_simple)
model_fe_simple = OLS(y, X_simple).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nFixed Effects Model (simple, no controls):")
print(model_fe_simple.summary())
