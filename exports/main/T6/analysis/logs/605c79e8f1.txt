import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks since switch, with week 14 as reference)
# Week 15 is the first post-switch week, so relative_week = week - 14
df['relative_week'] = df['week'] - 14

# Create dummies for each week (excluding week 14 as reference)
# We'll use weeks 6-13 as pre, 15-22 as post
pre_weeks = list(range(6, 14))  # weeks 6-13
post_weeks = list(range(15, 23))  # weeks 15-22

# Create interaction terms for event study
for w in pre_weeks + post_weeks:
    if w != 14:  # skip reference week
        df[f'week_{w}'] = (df['week'] == w).astype(int)
        df[f'week_{w}_treated'] = ((df['week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Event study regression (using week 14 as reference)
# We interact each week dummy with treatment status
event_study_vars = '+'.join([f'week_{w}_treated' for w in pre_weeks + post_weeks if w != 14])
formula = f'service_score ~ switched_branch + {" + ".join([f"week_{w}" for w in pre_weeks + post_weeks if w != 14])} + {event_study_vars} + staffing_index + visits_index'

model_es = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the interaction terms (these are the event study estimates)
coefficients = []
stderrs = []
weeks = []
for w in sorted(pre_weeks + post_weeks):
    if w != 14:
        coef_name = f'week_{w}_treated'
        coefficients.append(model_es.params[coef_name])
        stderrs.append(model_es.bse[coef_name])
        weeks.append(w)

# Create event study plot
fig, ax = plt.subplots(figsize=(12, 6))

# Actually, let's do a cleaner event study with proper specification

# Create dummies for each week relative to week 13
weeks = list(range(6, 23))  # weeks 6-22
ref_week = 13

# Build the formula properly
pre_weeks = [w for w in weeks if w < ref_week]  # weeks 6-12

# Create interaction terms
for w in weeks:
    if w != ref_week:
        df[f'w{w}'] = (df['week'] == w).astype(int)
        df[f'w{w}_treated'] = ((df['week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Event study formula
week_dummies = '+'.join([f'w{w}' for w in weeks if w != ref_week])
interaction_terms = '+'.join([f'w{w}_treated' for w in weeks if w != ref_week])
