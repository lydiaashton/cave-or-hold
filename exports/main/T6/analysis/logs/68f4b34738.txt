import pandas as pd
import numpy as np
import statsmodels.api as sm
import statsmodels.formula.api as smf
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# Basic DiD: service_score ~ switched_branch * after_switch
model_basic = smf.ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Basic DiD Regression (with cluster-robust SEs by branch):")
print(model_basic.summary())

# Add controls (staffing_index, visits_index)
model_controls = smf.ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with controls:")
print(model_controls.summary())

# Check pre-trends: look at the difference between groups before the switch
print("\n\nPre-trend check (weeks 6-14):")
pre_period = df[df['week'] < 15]
pre_trend = pre_period.groupby('week').apply(
    lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()
)
print(pre_trend)
print(f"\nAverage pre-period gap: {pre_trend.mean():.2f}")
print(f"Std dev of pre-period gap: {pre_trend.std():.2f}")

# Plot the trends
plt.figure(figsize=(12, 6))
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
plt.plot(weekly.index, weekly[0], 'o-', label='Control (No Switch)', linewidth=2, markersize=8)
plt.plot(weekly.index, weekly[1], 's-', label='Treatment (Switched)', linewidth=2, markersize=8)
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')
plt.xlabel('Week')
plt.ylabel('Service Score')
plt.title('Service Scores by Week: Treatment vs Control Groups')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.savefig('trends.png', dpi=150, bbox_inches='tight')
print("\nPlot saved as trends.png")

import pandas as pd

import statsmodels.api as sm
import statsmodels.formula.api as smf
from scipy import stats

df = pd.read_csv('branches.csv')

# More detailed analysis
# Get the coefficient and confidence interval from the model with controls
model = smf.ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

did_coef = model.params['switched_branch:after_switch']
did_se = model.bse['switched_branch:after_switch']
did_pvalue = model.pvalues['switched_branch:after_switch']

print(f"DiD Coefficient: {did_coef:.2f}")
print(f"Standard Error: {did_se:.2f}")
print(f"P-value: {did_pvalue:.3f}")
print(f"95% CI: [{did_coef - 1.96*did_se:.2f}, {did_coef + 1.96*did_se:.2f}]")
print(f"90% CI: [{did_coef - 1.645*did_se:.2f}, {did_coef + 1.645*did_se:.2f}]")

# Effect size relative to baseline
baseline = df['service_score'].mean()
print(f"\nBaseline average service score: {baseline:.2f}")
print(f"Effect as % of baseline: {did_coef/baseline*100:.1f}%")

# Check distribution of branch-level changes
print("\n\nBranch-level analysis:")
branch_pivot = df.pivot_table(index='branch_id', columns=['switched_branch', 'after_switch'], values='service_score', aggfunc='mean')

# For treatment branches: calculate before/after change
treatment_branches = df[df['switched_branch']==1]['branch_id'].unique()
control_branches = df[df['switched_branch']==0]['branch_id'].unique()

treatment_changes = []
for b in treatment_branches:
    before = df[(df['branch_id']==b) & (df['after_switch']==0)]['service_score'].mean()
    after = df[(df['branch_id']==b) & (df['after_switch']==1)]['service_score'].mean()
    treatment_changes.append(after - before)

control_changes = []
for b in control_branches:
    before = df[(df['branch_id']==b) & (df['after_switch']==0)]['service_score'].mean()
    after = df[(df['branch_id']==b) & (df['after_switch']==1)]['service_score'].mean()
    control_changes.append(after - before)

print(f"\nTreatment branches (n={len(treatment_branches)}):")
print(f"  Mean change: {np.mean(treatment_changes):.2f}")
print(f"  Std dev: {np.std(treatment_changes, ddof=1):.2f}")
print(f"  Min: {np.min(treatment_changes):.2f}, Max: {np.max(treatment_changes):.2f}")
print(f"  % with positive change: {np.mean(np.array(treatment_changes) > 0)*100:.0f}%")

print(f"\nControl branches (n={len(control_branches)}):")
print(f"  Mean change: {np.mean(control_changes):.2f}")
print(f"  Std dev: {np.std(control_changes, ddof=1):.2f}")
print(f"  Min: {np.min(control_changes):.2f}, Max: {np.max(control_changes):.2f}")
print(f"  % with positive change: {np.mean(np.array(control_changes) > 0)*100:.0f}%")

# T-test on the changes
from scipy.stats import ttest_ind
t_stat, t_pval = ttest_ind(treatment_changes, control_changes)
print(f"\nT-test comparing changes: t={t_stat:.2f}, p={t_pval:.3f}")

# Power analysis - what effect could we detect?
n_treatment = len(treatment_branches)
n_control = len(control_branches)
pooled_std = np.sqrt(((n_treatment-1)*np.std(treatment_changes, ddof=1)**2 + (n_control-1)*np.std(control_changes, ddof=1)**2) / (n_treatment + n_control - 2))
print(f"\nPooled std of changes: {pooled_std:.2f}")

# Minimum detectable effect at 80% power, alpha=0.05
from statsmodels.stats.power import TTestIndPower
power_analysis = TTestIndPower()
mde = power_analysis.solve_power(effect_size=None, nobs1=n_treatment, alpha=0.05, power=0.8, ratio=n_control/n_treatment)
print(f"Minimum detectable effect size (Cohen's d) at 80% power: {mde:.2f}")
print(f"In score points: {mde * pooled_std:.2f}")
