import pandas as pd

matplotlib.use('Agg')
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols
from scipy import stats

df = pd.read_csv('branches.csv')

# Event study: estimate effect for each week relative to switch
# Week 14 is the last pre-switch week, so we'll use it as reference
df['relative_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create dummies for each relative week
# We'll estimate the difference between treatment and control for each week
# relative to week 14 (the reference)

event_study_results = []
for rw in sorted(df['relative_week'].unique()):
    if rw == 0:  # Skip reference week
        continue
    week_dummy = (df['relative_week'] == rw).astype(int)
    interaction = week_dummy * df['switched_branch']
    
    # Regress on switched_branch, week dummy, and interaction
    temp_df = df.copy()
    temp_df['week_dummy'] = week_dummy
    temp_df['interaction'] = interaction
    
    model = ols('service_score ~ switched_branch + week_dummy + interaction', data=temp_df).fit()
    cov = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
    
    coef_names = model.params.index.tolist()
    interaction_idx = coef_names.index('interaction')
    
    event_study_results.append({
        'relative_week': rw,
        'coef': cov.params[interaction_idx],
        'se': cov.bse[interaction_idx],
        'ci_lower': cov.params[interaction_idx] - 1.96 * cov.bse[interaction_idx],
        'ci_upper': cov.params[interaction_idx] + 1.96 * cov.bse[interaction_idx]
    })

event_df = pd.DataFrame(event_study_results)
print("Event Study Results:")
print(event_df.round(3))

# Plot event study
fig, ax = plt.subplots(figsize=(10, 6))
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=0.5, color='red', linestyle='--', linewidth=2, label='Switch')

# Pre-period (negative relative weeks)
pre = event_df[event_df['relative_week'] < 0]
post = event_df[event_df['relative_week'] > 0]

ax.errorbar(pre['relative_week'], pre['coef'], yerr=1.96*pre['se'], 
            fmt='o', label='Pre-period', markersize=8, capsize=4, color='blue')
ax.errorbar(post['relative_week'], post['coef'], yerr=1.96*post['se'], 
            fmt='s', label='Post-period', markersize=8, capsize=4, color='green')
