import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create treatment group indicator (interaction of switched_branch and after_switch)
df['treatment'] = df['switched_branch'] * df['after_switch']

# === 1. Summary Statistics ===
print("=" * 60)
print("SUMMARY STATISTICS")
print("=" * 60)

# Before switch (weeks 6-14)
before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

print("\n--- Before Switch (Weeks 6-14) ---")
print("Switched branches (treatment group):")
print(f"  Mean service score: {before[before['switched_branch']==1]['service_score'].mean():.2f}")
print(f"  N observations: {len(before[before['switched_branch']==1])}")
print("Non-switched branches (control group):")
print(f"  Mean service score: {before[before['switched_branch']==0]['service_score'].mean():.2f}")
print(f"  N observations: {len(before[before['switched_branch']==0])}")

print("\n--- After Switch (Weeks 15-22) ---")
print("Switched branches (treatment group):")
print(f"  Mean service score: {after[after['switched_branch']==1]['service_score'].mean():.2f}")
print(f"  N observations: {len(after[after['switched_branch']==1])}")
print("Non-switched branches (control group):")
print(f"  Mean service score: {after[after['switched_branch']==0]['service_score'].mean():.2f}")
print(f"  N observations: {len(after[after['switched_branch']==0])}")

# === 2. Difference-in-Differences Calculation ===
print("\n" + "=" * 60)
print("DIFFERENCE-IN-DIFFERENCES CALCULATION")
print("=" * 60)

# Mean scores by group and period
switched_before = before[before['switched_branch']==1]['service_score'].mean()
switched_after = after[after['switched_branch']==1]['service_score'].mean()
control_before = before[before['switched_branch']==0]['service_score'].mean()
control_after = after[after['switched_branch']==0]['service_score'].mean()

print(f"\nSwitched branches: {switched_before:.2f} -> {switched_after:.2f} (change: {switched_after - switched_before:.2f})")
print(f"Control branches:  {control_before:.2f} -> {control_after:.2f} (change: {control_after - control_before:.2f})")
print(f"\nDifference-in-Differences estimate: {(switched_after - switched_before) - (control_after - control_before):.2f}")

# === 3. Regression Analysis ===
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nBasic DiD Model (service_score ~ switched_branch + after_switch + treatment):")
print(model.summary())

# DiD with branch fixed effects
print("\n" + "=" * 60)
print("DIFFERENCE-IN-DIFFERENCES WITH BRANCH FIXED EFFECTS")
print("=" * 60)

# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df, branch_dummies], axis=1)

# Build formula with fixed effects
fe_cols = [c for c in df_with_fe.columns if c.startswith('branch_')]
fe_formula = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(fe_cols)
model_fe = ols(fe_formula, data=df_with_fe).fit()
print(f"\nDiD with Branch Fixed Effects:")
print(f"Treatment effect coefficient: {model_fe.params['treatment']:.4f}")
print(f"Standard error: {model_fe.bse['treatment']:.4f}")
print(f"P-value: {model_fe.pvalues['treatment']:.4f}")
print(f"95% CI: [{model_fe.conf_int().loc['treatment', 0]:.2f}, {model_fe.conf_int().loc['treatment', 1]:.2f}]")

# DiD with branch and week fixed effects
print("\n" + "=" * 60)
print("DIFFERENCE-IN-DIFFERENCES WITH BRANCH AND WEEK FIXED EFFECTS")
print("=" * 60)

week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df_full_fe = pd.concat([df.reset_index(drop=True), branch_dummies.reset_index(drop=True), week_dummies.reset_index(drop=True)], axis=1)

week_cols = [c for c in df_full_fe.columns if c.startswith('week_')]
full_formula = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(fe_cols) + ' + ' + ' + '.join(week_cols)
model_full = ols(full_formula, data=df_full_fe).fit()
print(f"\nDiD with Branch and Week Fixed Effects:")
print(f"Treatment effect coefficient: {model_full.params['treatment']:.4f}")
print(f"Standard error: {model_full.bse['treatment']:.4f}")
print(f"P-value: {model_full.pvalues['treatment']:.4f}")
print(f"95% CI: [{model_full.conf_int().loc['treatment', 0]:.2f}, {model_full.conf_int().loc['treatment', 1]:.2f}]")

# === 4. Check Parallel Trends (Pre-treatment trends) ===

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# === Clustered Standard Errors ===
print("=" * 60)
print("ROBUSTNESS CHECK: CLUSTERED STANDARD ERRORS (by branch)")
print("=" * 60)

# Fit model and get clustered SEs
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
cov_cl = cov_cluster(model, df['branch_id'])
se_cl = np.sqrt(np.diag(cov_cl))

print("\nDiD with Clustered Standard Errors (by branch):")
print(f"Treatment effect coefficient: {model.params['treatment']:.4f}")
print(f"Clustered SE: {se_cl['treatment']:.4f}")
t_stat = model.params['treatment'] / se_cl['treatment']
p_val = 2 * (1 - sm.stats.norm.cdf(abs(t_stat)))
print(f"t-statistic: {t_stat:.3f}")
print(f"P-value: {p_val:.4f}")
print(f"95% CI: [{model.params['treatment'] - 1.96*se_cl['treatment']:.2f}, {model.params['treatment'] + 1.96*se_cl['treatment']:.2f}]")

# === Effect Size ===
print("\n" + "=" * 60)
print("EFFECT SIZE INTERPRETATION")
print("=" * 60)
