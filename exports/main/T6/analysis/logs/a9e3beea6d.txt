import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD table: average service scores by group and period
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES SUMMARY TABLE")
print("=" * 60)

# Before period (weeks 6-14)
before = df[df['after_switch'] == 0]
# After period (weeks 15-22)
after = df[df['after_switch'] == 1]

# Control group (switched_branch = 0)
control_before = before[before['switched_branch'] == 0]['service_score'].mean()
control_after = after[after['switched_branch'] == 0]['service_score'].mean()
control_diff = control_after - control_before

# Treatment group (switched_branch = 1)
treat_before = before[before['switched_branch'] == 1]['service_score'].mean()
treat_after = after[after['switched_branch'] == 1]['service_score'].mean()
treat_diff = treat_after - treat_before

# DiD estimate
did_estimate = treat_diff - control_diff

print(f"\nControl Group (No Switch):")
print(f"  Before (weeks 6-14):  {control_before:.2f}")
print(f"  After (weeks 15-22):  {control_after:.2f}")
print(f"  Change:               {control_diff:.2f}")

print(f"\nTreatment Group (Switched):")
print(f"  Before (weeks 6-14):  {treat_before:.2f}")
print(f"  After (weeks 15-22):  {treat_after:.2f}")
print(f"  Change:               {treat_diff:.2f}")

print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")
print(f"  (Treatment change - Control change)")

# Now run regression with standard errors
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Model 1: Basic DiD
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 1: Basic DiD (clustered SE by branch)")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 2: DiD with staffing and visits controls (clustered SE by branch)")
print(model2.summary().tables[1])

# Check parallel trends - plot average scores by week for each group
print("\n" + "=" * 60)
print("PARALLEL TRENDS CHECK - Weekly Averages")
print("=" * 60)

weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_avg.columns = ['Control (No Switch)', 'Treatment (Switched)']
weekly_avg['Diff'] = weekly_avg['Treatment (Switched)'] - weekly_avg['Control (No Switch)']

print("\nWeekly averages by group:")
print(weekly_avg.round(2))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check pre-trends more carefully
# Create week dummies interacted with treatment
print("=" * 60)
print("PRE-TREND ANALYSIS")
print("=" * 60)

# Create indicators for each week
for week in range(6, 23):
    df[f'week_{week}'] = (df['week'] == week).astype(int)

# Use week 6 as reference, interact each week with switched_branch
# This shows the difference between treatment and control for each week
pre_switch_weeks = [6, 7, 8, 9, 10, 11, 12, 13, 14]
post_switch_weeks = [15, 16, 17, 18, 19, 20, 21, 22]

# Calculate treatment-control difference for each week
weekly_diffs = []
for week in range(6, 23):
    week_data = df[df['week'] == week]
    control_mean = week_data[week_data['switched_branch'] == 0]['service_score'].mean()
    treat_mean = week_data[week_data['switched_branch'] == 1]['service_score'].mean()
    diff = treat_mean - control_mean
    weekly_diffs.append({'week': week, 'diff': diff, 'period': 'pre' if week < 15 else 'post'})

weekly_diffs_df = pd.DataFrame(weekly_diffs)
print("\nTreatment - Control difference by week:")
print(weekly_diffs_df.round(2))

# Average pre-trend difference (weeks 6-14)
pre_avg_diff = weekly_diffs_df[weekly_diffs_df['period'] == 'pre']['diff'].mean()
# Average post-trend difference (weeks 15-22)
post_avg_diff = weekly_diffs_df[weekly_diffs_df['period'] == 'post']['diff'].mean()

print(f"\nAverage pre-switch difference (weeks 6-14): {pre_avg_diff:.2f}")
print(f"Average post-switch difference (weeks 15-22): {post_avg_diff:.2f}")
print(f"Change in difference: {post_avg_diff - pre_avg_diff:.2f}")

# Check if pre-trend was already increasing
pre_trends = weekly_diffs_df[weekly_diffs_df['period'] == 'pre']
print(f"\nPre-trend slope (change in diff per week, weeks 6-14):")
pre_trend_slope = (pre_trends.iloc[-1]['diff'] - pre_trends.iloc[0]['diff']) / (pre_trends.iloc[-1]['week'] - pre_trends.iloc[0]['week'])
print(f"  {pre_trend_slope:.2f} points per week")

# Event study regression
print("\n" + "=" * 60)
print("EVENT STUDY REGRESSION")
print("=" * 60)

# Create relative week variable (week 15 = 0)
df['relative_week'] = df['week'] - 15

# Create week indicators relative to switch (omitting week -9 as reference)
for rw in range(-9, 8):
    if rw != -9:  # omit reference
        df[f'rel_week_{rw}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Event study model
event_study_formula = 'service_score ~ switched_branch + ' + ' + '.join([f'rel_week_{rw}' for rw in range(-8, 8)])
model_es = ols(event_study_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nEvent Study Coefficients (relative to week 6, treatment x control difference):")
coef_table = model_es.summary().tables[1]
print(coef_table)

# Extract and plot event study coefficients
event_coefs = []
for rw in range(-9, 8):
    if rw == -9:
        event_coefs.append({'relative_week': rw, 'coef': 0, 'se': 0, 'pval': 1})  # reference
    else:
        coef_name = f'rel_week_{rw}'
        if coef_name in model_es.params.index:
            coef = model_es.params[coef_name]
            se = model_es.bse[coef_name]
            pval = model_es.pvalues[coef_name]
            event_coefs.append({'relative_week': rw, 'coef': coef, 'se': se, 'pval': pval})

event_coefs_df = pd.DataFrame(event_coefs)
print("\nEvent study coefficients:")
print(event_coefs_df.round(3))

# Check if pre-trends are statistically different from zero
pre_event_coefs = event_coefs_df[event_coefs_df['relative_week'] < 0]
print(f"\nPre-trend coefficients (should be ~0 if parallel trends hold):")
print(pre_event_coefs)

# Test if pre-trends are jointly significant
pre_week_vars = [f'rel_week_{rw}' for rw in range(-8, 0)]
from statsmodels.stats.anova import anova_lm

# F-test for pre-trend coefficients
hypotheses = [f"{var} = 0" for var in pre_week_vars]
f_test = model_es.f_test(hypotheses)
print(f"\nJoint F-test for pre-trend coefficients (weeks -8 to -1):")
print(f"  F-statistic: {f_test.fvalue:.2f}")
print(f"  P-value: {f_test.pvalue:.4f}")

# Plot event study
plt.figure(figsize=(10, 6))
plt.errorbar(event_coefs_df['relative_week'], event_coefs_df['coef'], 
             yerr=1.96 * event_coefs_df['se'], fmt='o', capsize=5, linewidth=2, markersize=8)
plt.axhline(y=0, color='gray', linestyle='--')
plt.axvline(x=-0.5, color='red', linestyle='-', linewidth=2, label='Switch Point')
plt.xlabel('Weeks Relative to Switch (0 = Week 15)', fontsize=12)
plt.ylabel('Treatment Effect (vs. Week 6 Reference)', fontsize=12)
plt.title('Event Study: Treatment Effect by Week', fontsize=14, fontweight='bold')
plt.grid(True, alpha=0.3)
plt.xticks(range(-9, 8))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150, bbox_inches='tight')
print("\nEvent study plot saved as 'event_study.png'")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create relative week variable (week 15 = 0)
df['relative_week'] = df['week'] - 15

# Create week indicators relative to switch with cleaner names
for rw in range(-9, 8):
    df[f'r{rw:+d}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Event study model - use r_9 as reference (week 6)
# Build formula manually to avoid patsy issues
pre_vars = [f'r{rw:+d}' for rw in range(-8, 0)]  # weeks -8 to -1
post_vars = [f'r{rw:+d}' for rw in range(0, 8)]  # weeks 0 to 7
all_vars = pre_vars + post_vars

formula = 'service_score ~ switched_branch + ' + ' + '.join(all_vars)
model_es = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=" * 60)
print("EVENT STUDY REGRESSION")
print("=" * 60)
print("\nCoefficients for treatment x week interactions:")
print("(Reference: Week 6, omitted)")
print()

# Extract coefficients
event_results = []
for rw in range(-9, 8):
    var_name = f'r{rw:+d}'
    if rw == -9:
        event_results.append({'relative_week': rw, 'coef': 0.0, 'se': 0.0, 'pval': 1.0, 'ci_low': 0.0, 'ci_high': 0.0})
    elif var_name in model_es.params.index:
        coef = model_es.params[var_name]
        se = model_es.bse[var_name]
        pval = model_es.pvalues[var_name]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        event_results.append({'relative_week': rw, 'coef': coef, 'se': se, 'pval': pval, 'ci_low': ci_low, 'ci_high': ci_high})

event_df = pd.DataFrame(event_results)
event_df['period'] = event_df['relative_week'].apply(lambda x: 'pre' if x < 0 else ('switch' if x == 0 else 'post'))

print(event_df.round(3).to_string(index=False))

# Test pre-trends
print("\n" + "=" * 60)
print("PRE-TREND TESTS")
print("=" * 60)

# Individual pre-trend coefficients
pre_event = event_df[event_df['relative_week'] < 0]
print("\nPre-switch coefficients (should be ~0 if parallel trends hold):")
print(pre_event[['relative_week', 'coef', 'se', 'pval']].round(3).to_string(index=False))

# Are any pre-trends significantly different from 0?
sig_pre = pre_event[pre_event['pval'] < 0.05]
print(f"\nPre-trends significantly different from 0 (p<0.05): {len(sig_pre)} out of {len(pre_event)}")

# Joint F-test for all pre-trends
from statsmodels.stats.contrast import ContrastMatrix

pre_var_names = [f'r{rw:+d}' for rw in range(-8, 0)]
r_matrix = np.zeros((len(pre_var_names), len(model_es.params)))
for i, var in enumerate(pre_var_names):
    if var in model_es.params.index:
        r_matrix[i, model_es.params.index.get_loc(var)] = 1

f_test = model_es.f_test(r_matrix)
print(f"\nJoint F-test for ALL pre-trend coefficients:")
print(f"  F-statistic: {f_test.fvalue:.2f}")
print(f"  P-value: {f_test.pvalue:.4f}")
if f_test.pvalue < 0.05:
    print("  *** PRE-TRENDS ARE SIGNIFICANTLY DIFFERENT FROM ZERO ***")
    print("  *** Parallel trends assumption is VIOLATED ***")

for i, rw in enumerate(range(-9, 8), start=1):
    df[f'w{i}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Event study model - use w1 (week 6) as reference
# Build formula: weeks 2-17 (w2 to w17)
formula_vars = [f'w{i}' for i in range(2, 18)]
formula = 'service_score ~ switched_branch + ' + ' + '.join(formula_vars)

model_es = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

event_results = []
for i, rw in enumerate(range(-9, 8), start=1):

    if i == 1:
        event_results.append({'relative_week': rw, 'week': rw+15, 'coef': 0.0, 'se': 0.0, 'pval': 1.0})
    elif var_name in model_es.params.index:
