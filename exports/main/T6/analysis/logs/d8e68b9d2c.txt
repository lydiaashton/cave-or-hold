import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['treatment_post'] = df['treatment'] * df['post']

# Basic DiD regression
model = ols('service_score ~ treatment + post + treatment_post', data=df).fit()
print("Basic Difference-in-Differences Regression:")
print(model.summary())

# Now with branch fixed effects
print("\n\nDiD with Branch Fixed Effects:")
df_with_dummies = pd.get_dummies(df, columns=['branch_id'], drop_first=True)
fixed_effects_cols = [c for c in df_with_dummies.columns if c.startswith('branch_id')]
formula = 'service_score ~ treatment + post + treatment_post + ' + ' + '.join(fixed_effects_cols)
model_fe = ols(formula, data=df_with_dummies).fit()
print(model_fe.summary())

# Let's also check if we should control for staffing and visits
print("\n\nDiD with Branch FE and Controls:")
formula_controls = 'service_score ~ treatment + post + treatment_post + staffing_index + visits_index + ' + ' + '.join(fixed_effects_cols)

import pandas as pd
import numpy as np
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['treatment_post'] = df['treatment'] * df['post']

# Use a cleaner approach: absorb branch fixed effects by demeaning
# First, calculate branch means
branch_means = df.groupby('branch_id')['service_score'].transform('mean')
df['service_score_demeaned'] = df['service_score'] - branch_means

# Also demean the other variables
for col in ['treatment', 'post', 'treatment_post', 'staffing_index', 'visits_index']:
    col_means = df.groupby('branch_id')[col].transform('mean')
    df[f'{col}_demeaned'] = df[col] - col_means

# Run regression on demeaned data (this is equivalent to fixed effects)
X = df[['treatment_demeaned', 'post_demeaned', 'treatment_post_demeaned', 'staffing_index_demeaned', 'visits_index_demeaned']]
X = sm.add_constant(X)
y = df['service_score_demeaned']

model_fe_clean = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Fixed Effects Model (clustered SEs by branch):")
print(model_fe_clean.summary())

# Also run without controls
X_simple = df[['treatment_demeaned', 'post_demeaned', 'treatment_post_demeaned']]
X_simple = sm.add_constant(X_simple)
model_fe_simple = sm.OLS(y, X_simple).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nFixed Effects Model Simple (clustered SEs by branch):")
print(model_fe_simple.summary())

# Check parallel trends: compare pre-period trends between groups
print("\n\nParallel Trends Check - Pre-period service scores by week:")
pre_data = df[df['after_switch'] == 0]
pre_summary = pre_data.groupby(['switched_branch', 'week'])['service_score'].mean().unstack()
print(pre_summary)

# Calculate week-by-week differences in pre-period
print("\n\nPre-period weekly averages:")
for week in range(6, 15):
    week_data = df[df['week'] == week]
    control_mean = week_data[week_data['switched_branch'] == 0]['service_score'].mean()
    treatment_mean = week_data[week_data['switched_branch'] == 1]['service_score'].mean()
    diff = treatment_mean - control_mean
    print(f"Week {week}: Control={control_mean:.2f}, Treatment={treatment_mean:.2f}, Diff={diff:.2f}")
