import pandas as pd
import numpy as np
import statsmodels.api as sm
import statsmodels.formula.api as smf
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary table: mean service scores by group and period
summary_table = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("Service Score by Group and Period:")
print(summary_table)
print()

# Calculate the DiD estimate manually
# Control group (switched_branch=0)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_change = control_after - control_before

# Treatment group (switched_branch=1)
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_change = treat_after - treat_before

did_estimate = treat_change - control_change

print("Difference-in-Differences Calculation:")
print(f"Control group - Before: {control_before:.2f}, After: {control_after:.2f}, Change: {control_change:.2f}")
print(f"Treatment group - Before: {treat_before:.2f}, After: {treat_after:.2f}, Change: {treat_change:.2f}")
print(f"DiD Estimate (Treatment effect): {did_estimate:.2f}")
print()

# Run regression with DiD
model = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("DiD Regression Results:")
print(model.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
import statsmodels.formula.api as smf
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create event study variables - relative week to treatment (week 15)
df['relative_week'] = df['week'] - 15

# For event study, create dummies for each week relative to treatment
# Use week 14 (relative_week = -1) as reference
weeks = sorted(df['week'].unique())
week_dummies = {}
for w in weeks:
    if w != 14:  # reference week
        df[f'week_{w}'] = (df['week'] == w).astype(int)

# Event study regression - interact each week dummy with treatment
# Only include pre-treatment weeks for parallel trends check
event_study_vars = [f'week_{w}' for w in weeks if w != 14]
interaction_vars = [f'week_{w}_treat' for w in weeks if w != 14]

for w in weeks:
    if w != 14:
        df[f'week_{w}_treat'] = df[f'week_{w}'] * df['switched_branch']

# Run event study with pre-treatment weeks only (weeks 6-14)
pre_treatment_df = df[df['week'] <= 14].copy()
pre_treatment_vars = [f'week_{w}_treat' for w in range(6, 14)]
if len(pre_treatment_vars) > 0:
    formula = 'service_score ~ ' + ' + '.join(pre_treatment_vars)
    model_pre = smf.ols(formula, data=pre_treatment_df).fit(cov_type='cluster', cov_kwds={'groups': pre_treatment_df['branch_id']})
    print("Pre-treatment trends test (weeks 6-13 vs week 14):")
    print(model_pre.summary())
    print()

# Full event study
full_vars = [f'week_{w}_treat' for w in weeks if w != 14]
formula_full = 'service_score ~ ' + ' + '.join(full_vars)
model_event = smf.ols(formula_full, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the interaction terms
coef_dict = {}
for w in weeks:
    if w == 14:
        coef_dict[w] = 0  # reference
    else:
        coef_dict[w] = model_event.params.get(f'week_{w}_treat', 0)
