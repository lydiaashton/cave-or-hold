import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with cluster-robust standard errors
# Model 1: Simple DiD
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Model 1: Simple DiD ===")
print(model1.summary())

# Model 2: DiD with covariates
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\n=== Model 2: DiD with Covariates ===")
print(model2.summary())

# Model 3: With branch fixed effects and cluster-robust SEs
# This controls for time-invariant branch characteristics
df_demeaned = df.copy()
# Demean within each branch
branch_means = df.groupby('branch_id')['service_score'].transform('mean')
df_demeaned['service_score_demeaned'] = df['service_score'] - branch_means

# Also demean the other variables
for col in ['switched_branch', 'after_switch', 'treatment', 'staffing_index', 'visits_index']:
    col_means = df.groupby('branch_id')[col].transform('mean')
    df_demeaned[f'{col}_demeaned'] = df[col] - col_means

model3 = ols('service_score_demeaned ~ after_switch_demeaned + treatment_demeaned + staffing_index_demeaned + visits_index_demeaned', data=df_demeaned).fit()
print("\n=== Model 3: Branch Fixed Effects ===")
print(model3.summary())

# Cluster-robust standard errors at branch level
from statsmodels.stats.sandwich_covariance import cov_cluster
cov_club = cov_cluster(model2, df['branch_id'])
se_club = np.sqrt(np.diag(cov_club))
print("\n=== Model 2 with Cluster-Robust SEs (clustered by branch) ===")
print(f"treatment coefficient: {model2.params['treatment']:.4f}")
print(f"Cluster-robust SE: {se_club[model2.params.index.get_loc('treatment')]:.4f}")
t_stat = model2.params['treatment'] / se_club[model2.params.index.get_loc('treatment')]
print(f"t-statistic: {t_stat:.2f}")
p_value = 2 * (1 - sm.stats.norm.cdf(abs(t_stat)))
print(f"p-value: {p_value:.4f}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create week dummies for event study
# Use week 14 as reference (last pre-period week)
df['week_14'] = (df['week'] == 14).astype(int)

# Create interaction terms for each week with treatment
for w in range(6, 23):
    if w != 14:  # Skip reference week
        df[f'week_{w}'] = (df['week'] == w).astype(int)
        df[f'treat_week_{w}'] = df['switched_branch'] * df[f'week_{w}']

# Event study regression
event_study_vars = '+'.join([f'treat_week_{w}' for w in range(6, 23) if w != 14])
week_vars = '+'.join([f'week_{w}' for w in range(6, 23) if w != 14])

formula = f'service_score ~ switched_branch + {week_vars} + {event_study_vars} + staffing_index + visits_index'
model_es = ols(formula, data=df).fit()

# Extract treatment-week coefficients
print("=== Event Study Coefficients (relative to week 14) ===")
coeffs = []
for w in range(6, 23):
    if w == 14:
        coeffs.append((w, 0, 0, 0))  # Reference week
    else:
        coef = model_es.params[f'treat_week_{w}']
        se = model_es.bse[f'treat_week_{w}']
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        coeffs.append((w, coef, ci_low, ci_high))
        print(f"Week {w}: coef={coef:.2f}, 95% CI=[{ci_low:.2f}, {ci_high:.2f}]")

# Create a plot
plt.figure(figsize=(12, 6))
weeks = [c[0] for c in coeffs]
coefs = [c[1] for c in coeffs]
ci_low = [c[2] for c in coeffs]
ci_high = [c[3] for c in coeffs]

plt.plot(weeks, coefs, 'o-', label='Treatment Effect', linewidth=2, markersize=8)
plt.fill_between(weeks, ci_low, ci_high, alpha=0.3, label='95% CI')
plt.axvline(x=14.5, color='red', linestyle='--', label='Switch Point (Week 15)')
plt.axhline(y=0, color='gray', linestyle='-', alpha=0.5)
plt.xlabel('Week')
plt.ylabel('Service Score Difference (Treatment vs Control)')
plt.title('Event Study: Weekly Treatment Effects\n(Week 14 as Reference)')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved to event_study.png")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Better event study: interact treatment with week indicators, using week 6 as baseline
# This shows the evolution of the treatment-control gap over time

# Create week dummies
for w in range(6, 23):
    df[f'week_{w}'] = (df['week'] == w).astype(int)

# Create treatment x week interactions
for w in range(6, 23):
    df[f'treat_x_week_{w}'] = df['switched_branch'] * df[f'week_{w}']

# Regression with week 6 as reference
week_vars = '+'.join([f'week_{w}' for w in range(7, 23)])
treat_week_vars = '+'.join([f'treat_x_week_{w}' for w in range(7, 23)])

formula = f'service_score ~ switched_branch + week_6 + {week_vars} + {treat_week_vars} + staffing_index + visits_index'
model_es2 = ols(formula, data=df).fit()

# Extract the treatment x week coefficients (these show the gap relative to week 6)
print("=== Treatment-Control Gap by Week (relative to week 6) ===")
print("(Positive values = treatment group ahead of control)")
results = []
for w in range(6, 23):
    if w == 6:
        # Reference week - the switched_branch coefficient gives the week 6 gap
        coef = model_es2.params['switched_branch']
        se = model_es2.bse['switched_branch']
    else:
        coef = model_es2.params[f'treat_x_week_{w}']
        se = model_es2.bse[f'treat_x_week_{w}']
    ci_low = coef - 1.96 * se
    ci_high = coef + 1.96 * se
    results.append((w, coef, ci_low, ci_high))
    sig = "***" if ci_low > 0 or ci_high < 0 else ""
    print(f"Week {w}: gap={coef:.2f}, 95% CI=[{ci_low:.2f}, {ci_high:.2f}] {sig}")

# Plot

weeks = [r[0] for r in results]
coefs = [r[1] for r in results]
ci_low = [r[2] for r in results]
ci_high = [r[3] for r in results]

plt.fill_between(weeks, ci_low, ci_high, alpha=0.3, color='blue', label='95% CI')
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')

pre_gaps = [r[1] for r in results if r[0] < 15]

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Look at the weekly gaps more carefully
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
gaps = weekly_avg[1] - weekly_avg[0]

print("=== Weekly Treatment-Control Gaps ===")
print("(Positive = Treatment ahead of Control)")
print()
for week in range(6, 23):
    gap = gaps[week]
    marker = " <-- SWITCH" if week == 15 else ""
    print(f"Week {week:2d}: {gap:6.2f}{marker}")

print()
print("=== Pre-period trend (weeks 6-14) ===")
pre_weeks = list(range(6, 15))
pre_gaps = [gaps[w] for w in pre_weeks]
# Linear trend
pre_slope = (pre_gaps[-1] - pre_gaps[0]) / (len(pre_gaps) - 1)
print(f"Gap at week 6: {pre_gaps[0]:.2f}")
print(f"Gap at week 14: {pre_gaps[-1]:.2f}")
print(f"Pre-period slope: {pre_slope:.2f} points per week")

print()
print("=== Post-period trend (weeks 15-22) ===")
post_weeks = list(range(15, 23))
post_gaps = [gaps[w] for w in post_weeks]
post_slope = (post_gaps[-1] - post_gaps[0]) / (len(post_gaps) - 1)
print(f"Gap at week 15: {post_gaps[0]:.2f}")
print(f"Gap at week 22: {post_gaps[-1]:.2f}")
print(f"Post-period slope: {post_slope:.2f} points per week")

print()
print("=== Key Question: Was there a discontinuity at week 15? ===")
# Compare week 14 to week 15 specifically
week14_gap = gaps[14]
week15_gap = gaps[15]
jump = week15_gap - week14_gap
print(f"Week 14 gap: {week14_gap:.2f}")
print(f"Week 15 gap: {week15_gap:.2f}")
print(f"Jump at switch: {jump:.2f}")

# Compare to average pre-period week-to-week change
pre_changes = [pre_gaps[i+1] - pre_gaps[i] for i in range(len(pre_gaps)-1)]
avg_pre_change = np.mean(pre_changes)
print(f"Average pre-period week-to-week change: {avg_pre_change:.2f}")

print()
print("=== Visual: Gap over time ===")
plt.figure(figsize=(10, 5))
plt.plot(range(6, 23), [gaps[w] for w in range(6, 23)], 'o-', linewidth=2, markersize=8)
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point')
plt.axhline(y=0, color='gray', linestyle='-', alpha=0.5)
# Add trend lines
plt.plot(pre_weeks, [pre_gaps[0] + pre_slope * (w - 6) for w in pre_weeks], 'g--', alpha=0.7, label='Pre-trend')
plt.plot(post_weeks, [post_gaps[0] + post_slope * (w - 15) for w in post_weeks], 'orange', '--', alpha=0.7, label='Post-trend')
plt.xlabel('Week')
plt.ylabel('Treatment-Control Gap')
plt.title('Was There a Break in Trend at Week 15?')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('trend_check.png', dpi=150)
print("Plot saved to trend_check.png")
