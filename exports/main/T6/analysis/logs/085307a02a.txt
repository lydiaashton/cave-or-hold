import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# DiD with branch fixed effects (drop switched_branch as it's collinear with branch FEs)
model_fe = ols('service_score ~ after_switch + did + C(branch_id)', data=df).fit()
print("DiD with Branch Fixed Effects (dropping switched_branch):")
print(model_fe.summary())

# Get the DiD coefficient and confidence interval
did_coef = model_fe.params['did']
did_se = model_fe.bse['did']
did_ci_lower = did_coef - 1.96 * did_se
did_ci_upper = did_coef + 1.96 * did_se
did_pvalue = model_fe.pvalues['did']

print(f"\n\nDiD Estimate: {did_coef:.3f}")
print(f"95% CI: [{did_ci_lower:.3f}, {did_ci_upper:.3f}]")
print(f"P-value: {did_pvalue:.3f}")

# Also run with robust standard errors clustered by branch
# Using linearmodels for cluster-robust SEs
try:
    from linearmodels.panel import PanelOLS
    
    # Prepare panel data
    df_panel = df.copy()
    df_panel = df_panel.set_index(['branch_id', 'week'])
    
    # Model with cluster-robust SEs
    mod = PanelOLS(df_panel['service_score'], 
                   df_panel[['after_switch', 'did']], 
                   entity_effects=True)
    res = mod.fit(cov_type='clustered', cluster_entity=True)
    print("\n\nPanelOLS with Cluster-Robust Standard Errors:")
    print(res)
except ImportError:
    print("\n\nlinearmodels not available, using basic OLS with HAC errors")
    # Use HAC robust standard errors as alternative
    model_hac = ols('service_score ~ after_switch + did + C(branch_id)', data=df).fit(cov_type='HAC', cov_kwds={'maxlags': 4})
    print(model_hac.summary())

import pandas as pd
import numpy as np
from linearmodels.panel import PanelOLS
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']

print("="*60)
print("RECONSIDERING THE EVIDENCE")
print("="*60)

# 1. Simple before/after for treatment branches
treatment = df[df['switched_branch'] == 1]
treat_before = treatment[treatment['after_switch'] == 0]['service_score'].mean()
treat_after = treatment[treatment['after_switch'] == 1]['service_score'].mean()
print(f"\n1. Treatment branches alone:")
print(f"   Before: {treat_before:.2f} → After: {treat_after:.2f}")
print(f"   Change: {treat_after - treat_before:.2f}")

# 2. What happened to control branches?
control = df[df['switched_branch'] == 0]
ctrl_before = control[control['after_switch'] == 0]['service_score'].mean()
ctrl_after = control[control['after_switch'] == 1]['service_score'].mean()
print(f"\n2. Control branches alone:")
print(f"   Before: {ctrl_before:.2f} → After: {ctrl_after:.2f}")
print(f"   Change: {ctrl_after - ctrl_before:.2f}")

# 3. The DiD calculation
did_simple = (treat_after - treat_before) - (ctrl_after - ctrl_before)
print(f"\n3. Difference-in-Differences:")
print(f"   (+1.26) - (-0.68) = {did_simple:.2f}")

# 4. How many treatment branches actually improved?
print(f"\n4. Branch-level outcomes:")
branch_data = df.groupby(['branch_id', 'after_switch'])['service_score'].mean().unstack()
branch_data['change'] = branch_data[1] - branch_data[0]
branch_data['switched'] = df.groupby('branch_id')['switched_branch'].first()

treat_branches = branch_data[branch_data['switched'] == 1]['change']
ctrl_branches = branch_data[branch_data['switched'] == 0]['change']

print(f"   Treatment: {len(treat_branches)} branches, {sum(treat_branches > 0)} improved ({100*sum(treat_branches > 0)/len(treat_branches):.0f}%)")
print(f"   Control: {len(ctrl_branches)} branches, {sum(ctrl_branches > 0)} improved ({100*sum(ctrl_branches > 0)/len(ctrl_branches):.0f}%)")

# 5. Statistical significance - both approaches
print(f"\n5. Statistical tests:")

# OLS with branch FE (non-clustered SEs)
df_panel = df.set_index(['branch_id', 'week'])
mod = PanelOLS(df_panel['service_score'], df_panel[['after_switch', 'did']], entity_effects=True)
res_clustered = mod.fit(cov_type='clustered', cluster_entity=True)
res_nonclustered = mod.fit(cov_type='unadjusted')

print(f"   With cluster-robust SEs: p = {res_clustered.pvalues['did']:.3f}")
