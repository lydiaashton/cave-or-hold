import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Summary table: average service scores by group and period
summary_table = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).unstack()
print("Average Service Scores by Group and Period:")
print(summary_table)
print()

# Calculate the DiD estimate manually
# Treatment group: before and after
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
# Control group: before and after
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print("Treatment group (switched branches):")
print(f"  Before: {treatment_before:.2f}")
print(f"  After: {treatment_after:.2f}")
print(f"  Change: {treatment_after - treatment_before:.2f}")
print()
print("Control group (non-switched branches):")
print(f"  Before: {control_before:.2f}")
print(f"  After: {control_after:.2f}")
print(f"  Change: {control_after - control_before:.2f}")
print()
did_estimate = (treatment_after - treatment_before) - (control_after - control_before)
print(f"Difference-in-Differences estimate: {did_estimate:.2f}")
print()

# Run DiD regression
# Model 1: Simple DiD
model1 = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Model 1: Simple DiD Regression")
print(model1.summary())
print()

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Model 2: DiD Regression with Controls")
print(model2.summary())
print()

# Check parallel trends - look at pre-treatment trends
pre_data = df[df['week'] < 15]

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Robustness check: event study approach - look at each week relative to switch
df['weeks_relative'] = df['week'] - 15  # 0 is the switch week, negative is before, positive is after

# Create week dummies for event study
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    df[f'week_{w}'] = (df['weeks_relative'] == w).astype(int)

# Event study regression (omitting week -1 as reference)
event_study_vars = ' + '.join([f'week_{w}' for w in range(-9, 8) if w != -1])
event_study_formula = f'service_score ~ switched_branch + after_switch + {event_study_vars}'

# This might be too many variables, let's simplify
# Let's just look at pre vs post more carefully

# Check if effect is consistent across post-treatment weeks
post_data = df[df['week'] >= 15].copy()
post_data['weeks_after'] = post_data['week'] - 14  # 1, 2, 3... weeks after switch

# Interaction of treated with weeks after
post_data['treated_weeks_after'] = post_data['switched_branch'] * post_data['weeks_after']

post_model = ols('service_score ~ switched_branch + weeks_after + treated_weeks_after', 
                 data=post_data).fit(cov_type='cluster', cov_kwds={'groups': post_data['branch_id']})
print("Post-treatment trend analysis:")
print(post_model.summary())

# Also check if there's any immediate vs delayed effect
# Split post period into early (weeks 15-18) and late (weeks 19-22)
df['post_early'] = ((df['week'] >= 15) & (df['week'] <= 18)).astype(int)
df['post_late'] = ((df['week'] >= 19) & (df['week'] <= 22)).astype(int)
df['treated_post_early'] = df['switched_branch'] * df['post_early']
df['treated_post_late'] = df['switched_branch'] * df['post_late']

phase_model = ols('service_score ~ switched_branch + post_early + post_late + treated_post_early + treated_post_late', 
                  data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nPhase analysis (early vs late post-treatment):")
print(phase_model.summary())

print("\nEarly post period effect (weeks 15-18):", phase_model.params['treated_post_early'])
print("Late post period effect (weeks 19-22):", phase_model.params['treated_post_late'])
