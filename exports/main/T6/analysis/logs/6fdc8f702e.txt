import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster
from scipy import stats

# Load the data
df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main model with cluster-robust SE
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
cov_robust = cov_cluster(model, group=df['branch_id'])
se_robust = np.sqrt(np.diag(cov_robust))

# Get treatment effect
treatment_coef = model.params['treatment']
treatment_se = se_robust[model.params.index.get_loc('treatment')]
t_stat = treatment_coef / treatment_se
p_val = 2 * (1 - stats.norm.cdf(abs(t_stat)))
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se

print("=== STATISTICAL EVIDENCE ===")
print(f"Point estimate: {treatment_coef:.2f} points")
print(f"Clustered SE: {treatment_se:.2f}")
print(f"t-statistic: {t_stat:.2f}")
print(f"p-value: {p_val:.3f}")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"90% CI: [{treatment_coef - 1.645 * treatment_se:.2f}, {treatment_coef + 1.645 * treatment_se:.2f}]")

# Probability that effect is positive (one-sided)
prob_positive = stats.norm.cdf(t_stat)
print(f"\nProbability effect is positive: {prob_positive:.1%}")

# What sample size would we need for significance?
# Current power calculation
print(f"\n=== INTERPRETATION ===")
print(f"Conventionally significant at 5% level? {'Yes' if p_val < 0.05 else 'No'}")
print(f"Conventionally significant at 10% level? {'Yes' if p_val < 0.10 else 'No'}")

# Check if CI includes zero
print(f"95% CI includes zero? {'Yes' if ci_lower < 0 < ci_upper else 'No'}")

# Effect size
baseline = df[df['week'] < 15]['service_score'].mean()
print(f"\nEffect as % of baseline: {100 * treatment_coef / baseline:.1f}%")
