import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Basic DiD calculation - average scores by group and period
print("=== Difference-in-Differences Summary ===\n")

# Before period (after_switch = 0)
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()

# After period (after_switch = 1)
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print("BEFORE the switch (weeks 6-14):")
print(f"  Treatment group (switched branches): {before_treatment:.2f}")
print(f"  Control group (non-switched branches): {before_control:.2f}")
print(f"  Difference: {before_treatment - before_control:.2f}")

print("\nAFTER the switch (weeks 15-22):")
print(f"  Treatment group (switched branches): {after_treatment:.2f}")
print(f"  Control group (non-switched branches): {after_control:.2f}")
print(f"  Difference: {after_treatment - after_control:.2f}")

# DiD estimate
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\n=== DiD Estimate: {did_estimate:.2f} ===")
print("(Change in treatment group) - (Change in control group)")

# Now run a regression with standard errors
print("\n=== Regression Analysis ===")
model = ols('service_score ~ treated + post + DiD', data=df).fit()
print(model.summary())

# Clustered standard errors by branch
print("\n=== Regression with Clustered Standard Errors (by branch) ===")
model_clustered = ols('service_score ~ treated + post + DiD', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_clustered.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Let's look at this from a different angle - event study style
# Calculate the difference between treatment and control for each week
weekly_diff = df.groupby('week').apply(
    lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()
).reset_index()
weekly_diff.columns = ['week', 'treatment_control_diff']

print("=== Weekly Treatment-Control Differences ===")
print(weekly_diff)

# Plot the differences over time
fig, ax = plt.subplots(figsize=(12, 5))
ax.bar(weekly_diff['week'], weekly_diff['treatment_control_diff'], 
       color=['#2E86AB' if w < 15 else '#A23B72' for w in weekly_diff['week']], alpha=0.7)
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch occurred')
ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Treatment - Control Difference', fontsize=12)
ax.set_title('Weekly Difference in Service Scores: Treatment vs Control\n(Positive = Treatment performing better)', fontsize=14)
ax.legend()
ax.grid(True, alpha=0.3, axis='y')
ax.set_xticks(range(6, 23))
plt.tight_layout()
plt.savefig('weekly_diff.png', dpi=150)
print("\nPlot saved as 'weekly_diff.png'")

# Now let's run a regression with week fixed effects to control for time trends
print("\n=== Regression with Week Fixed Effects ===")
df_weekly_fe = df.copy()
# Create week dummies (using week 6 as reference)
for w in range(7, 23):
    df_weekly_fe[f'week_{w}'] = (df_weekly_fe['week'] == w).astype(int)

# Build formula
fe_vars = ' + '.join([f'week_{w}' for w in range(7, 23)])
formula = f'service_score ~ switched_branch + after_switch + switched_branch * after_switch + {fe_vars}'
model_fe = ols(formula, data=df_weekly_fe).fit(cov_type='cluster', cov_kwds={'groups': df_weekly_fe['branch_id']})
print(model_fe.summary())
print(f"\nDiD coefficient: {model_fe.params['switched_branch:after_switch']:.3f}")
print(f"DiD p-value: {model_fe.pvalues['switched_branch:after_switch']:.4f}")

# Also control for staffing and visits
print("\n=== Regression with Controls (staffing_index, visits_index) ===")
model_controls = ols('service_score ~ switched_branch + after_switch + switched_branch * after_switch + staffing_index + visits_index', 
                     data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_controls.summary())
print(f"\nDiD coefficient: {model_controls.params['switched_branch:after_switch']:.3f}")
print(f"DiD p-value: {model_controls.pvalues['switched_branch:after_switch']:.4f}")
