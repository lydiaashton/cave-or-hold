import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch'] * df['after_switch']

# Run a difference-in-differences regression with branch fixed effects
# First, let's do a simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
print("Simple DiD Regression (no fixed effects):")
print(model.summary().tables[1])

# Now with branch fixed effects (more robust)
df = df.copy()
df['branch_id'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ after_switch + treated + C(branch_id)', data=df).fit()

print(model_fe.summary().tables[1])

import pandas as pd
import numpy as np
import statsmodels.api as sm

from statsmodels.regression.linear_model import OLS

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# Create dummy variables for branch fixed effects

X = pd.concat([df[['after_switch', 'treated', 'staffing_index', 'visits_index']], branch_dummies], axis=1)
X = sm.add_constant(X)
y = df['service_score']

# Fit the model
model = OLS(y, X).fit()

model_robust = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

    ci_lower = coef - 1.96 * se

treated_coef = model_robust.params['treated']
