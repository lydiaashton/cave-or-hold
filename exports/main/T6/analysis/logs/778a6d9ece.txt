import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Calculate mean service scores by group and period
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("Mean Service Scores by Group and Period:")
print(summary)
print("\n")

# Calculate the DiD estimate manually
# Treatment group (switched=1): before and after
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print("Treatment (switched) branches:")
print(f"  Before switch: {treatment_before:.2f}")
print(f"  After switch: {treatment_after:.2f}")
print(f"  Change: {treatment_after - treatment_before:.2f}")
print("\nControl (non-switched) branches:")
print(f"  Before switch: {control_before:.2f}")
print(f"  After switch: {control_after:.2f}")
print(f"  Change: {control_after - control_before:.2f}")
print("\nDifference-in-Differences estimate:")
did_estimate = (treatment_after - treatment_before) - (control_after - control_before)
print(f"  DiD = {did_estimate:.2f}")

# Run regression with DiD
model = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
print("\n\nRegression Results (DiD Model):")
print(model.summary())

# Run regression with cluster-robust standard errors (clustered by branch)
model_clustered = ols('service_score ~ switched_branch + after_switch + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nRegression Results (Clustered by Branch):")
print(model_clustered.summary())

import pandas as pd
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']

# Event study design - estimate effects for each week relative to switch
df['relative_week'] = df['week'] - 15  # Week 15 is the switch (relative week 0)

# Create week dummies interacted with treatment
df['week_dummies'] = df['week'].astype(str)

# Event study: service_score ~ C(week) + switched_branch + C(week)*switched_branch (for pre-trends)
# Let's do a simpler approach: estimate week-specific treatment effects

event_study = []
for w in range(6, 23):
    week_data = df[df['week'] == w].copy()
    week_data['treated'] = week_data['switched_branch']
    if len(week_data['treated'].unique()) > 1:
        model = ols('service_score ~ treated', data=week_data).fit()
        event_study.append({
            'week': w,
            'relative_week': w - 15,
            'effect': model.params['treated'],
            'std_err': model.bse['treated'],
            'is_post': w >= 15
        })

event_df = pd.DataFrame(event_study)
event_df['ci_lower'] = event_df['effect'] - 1.96 * event_df['std_err']
event_df['ci_upper'] = event_df['effect'] + 1.96 * event_df['std_err']

print("Event Study Results (weekly treatment effects):")
print(event_df.round(3))

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
ax.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch implemented')
ax.axhline(y=0, color='black', linestyle='-', linewidth=0.5)

# Pre-treatment
pre = event_df[event_df['relative_week'] < 0]
ax.errorbar(pre['relative_week'], pre['effect'], 
            yerr=[pre['effect'] - pre['ci_lower'], pre['ci_upper'] - pre['effect']],
            fmt='o', color='blue', capsize=5, label='Pre-switch', markersize=8)

# Post-treatment
post = event_df[event_df['relative_week'] >= 0]
ax.errorbar(post['relative_week'], post['effect'], 
            yerr=[post['effect'] - post['ci_lower'], post['ci_upper'] - post['effect']],
            fmt='o', color='green', capsize=5, label='Post-switch', markersize=8)

ax.set_xlabel('Weeks Relative to Switch (0 = Week 15)')
ax.set_ylabel('Treatment Effect (Treatment - Control)')

ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-9, 8))
