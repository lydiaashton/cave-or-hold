import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*treatment + beta2*post + beta3*(treatment*post) + error
# beta3 is the DiD estimate

df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['treatment_post'] = df['treatment'] * df['post']

# Basic DiD regression
model = ols('service_score ~ treatment + post + treatment_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Basic DiD Regression (clustered by branch) ===")
print(model.summary())

# Also run with branch fixed effects and week fixed effects for more robustness
df['branch_id_str'] = df['branch_id'].astype(str)
df['week_str'] = df['week'].astype(str)

# DiD with branch and week fixed effects
model_fe = ols('service_score ~ treatment_post + C(branch_id_str) + C(week_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD with Branch and Week Fixed Effects ===")
print(model_fe.summary())

# Extract the key coefficient
did_coef = model_fe.params['treatment_post']
did_se = model_fe.bse['treatment_post']
did_pvalue = model_fe.pvalues['treatment_post']
print(f"\nDiD Estimate: {did_coef:.3f}")
print(f"Standard Error: {did_se:.3f}")
print(f"P-value: {did_pvalue:.4f}")
print(f"95% CI: [{did_coef - 1.96*did_se:.3f}, {did_coef + 1.96*did_se:.3f}]")
