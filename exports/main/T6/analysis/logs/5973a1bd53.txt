import pandas as pd
import numpy as np
import statsmodels.api as sm
import statsmodels.formula.api as smf

# Load the data
df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched + beta2*after + beta3*(switched*after) + error
# beta3 is the DiD estimate

df['interaction'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = smf.ols('service_score ~ switched_branch + after_switch + interaction', data=df)
results = model.fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Basic DiD Regression (clustered SEs by branch) ===")
print(results.summary())

# Add control variables (staffing_index, visits_index)
model2 = smf.ols('service_score ~ switched_branch + after_switch + interaction + staffing_index + visits_index', data=df)
results2 = model2.fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n=== DiD Regression with Controls (clustered SEs by branch) ===")
print(results2.summary())

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

# Load the data
df = pd.read_csv('branches.csv')

# Run DiD with branch fixed effects for more robust estimation
# This controls for time-invariant branch characteristics
df['week_factor'] = df['week'].astype('category')

# DiD with branch fixed effects
model_fe = smf.ols('service_score ~ switched_branch + after_switch + interaction + C(branch_id)', data=df)
results_fe = model_fe.fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== DiD with Branch Fixed Effects ===")
print(results_fe.summary())

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

# Load the data
df = pd.read_csv('branches.csv')
df['interaction'] = df['switched_branch'] * df['after_switch']

# Test parallel trends - check if pre-treatment trends were different
# Use only pre-treatment data (weeks 6-14)
pre_data = df[df['week'] <= 14].copy()
pre_data['week_centered'] = pre_data['week'] - 10  # Center around middle of pre-period

# Regress service score on week trend, switched status, and their interaction
# If parallel trends hold, the interaction should not be significant
model_trend = smf.ols('service_score ~ switched_branch * week_centered', data=pre_data)
results_trend = model_trend.fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})

print("=== Parallel Trends Test (Pre-treatment period only) ===")
print(results_trend.summary())

# The key coefficient is switched_branch:week_centered
# If not significant, parallel trends assumption is supported
trend_interaction_pval = results_trend.pvalues['switched_branch:week_centered']
print(f"\nParallel trends test p-value: {trend_interaction_pval:.4f}")
if trend_interaction_pval > 0.05:
    print("Result: Parallel trends assumption is SUPPORTED (no significant difference in pre-trends)")
else:
    print("Result: Parallel trends assumption may be VIOLATED")

# Also run event study - effect by week relative to switch
df['weeks_relative'] = df['week'] - 15  # Week 15 is the switch (0 = first week after)

# Create week dummies relative to switch
for w in range(-9, 8):  # weeks -9 to -1 (before), 0 to 7 (after)
    df[f'week_{w}'] = (df['weeks_relative'] == w).astype(int)

# Use week -1 as reference (last week before switch)
# Regress on switched, week dummies, and their interactions
week_cols = [f'week_{w}' for w in range(-9, 8) if w != -1]  # exclude reference week
interaction_cols = [f'switched_branch:{col}' for col in week_cols]

formula = 'service_score ~ switched_branch + ' + ' + '.join(week_cols) + ' + ' + ' + '.join([f'switched_branch * {col}' for col in week_cols])

    coef_name = f'switched_branch:week_{w}'
    if coef_name in results_event.params:

import pandas as pd

# Load the data
df = pd.read_csv('branches.csv')

df['weeks_relative'] = df['week'] - 15  # Week 15 is the switch (0 = first week after)

ax.set_title('Event Study: Service Scores Before and After Procedure Change', fontsize=14)

ax.grid(True, alpha=0.3)

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

# Load the data
df = pd.read_csv('branches.csv')
df['interaction'] = df['switched_branch'] * df['after_switch']

# Re-run the main DiD regression
model = smf.ols('service_score ~ switched_branch + after_switch + interaction', data=df)
results = model.fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== MAIN DiD REGRESSION (VERIFICATION) ===")
print(f"Interaction coefficient (treatment effect): {results.params['interaction']:.3f}")
print(f"Standard error: {results.bse['interaction']:.3f}")
print(f"95% CI: [{results.params['interaction'] - 1.96*results.bse['interaction']:.3f}, {results.params['interaction'] + 1.96*results.bse['interaction']:.3f}]")
print(f"P-value: {results.pvalues['interaction']:.4f}")
print(f"Significant at 5% level: {results.pvalues['interaction'] < 0.05}")

# Check the weekly pattern more carefully around the switch
print("\n=== WEEKLY AVERAGES AROUND SWITCH (Weeks 13-17) ===")
weekly = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack()
print(weekly.loc[:, 13:17].round(2))

# Check if there's any pre-trend difference
pre_data = df[df['week'] < 15].copy()
pre_data['week_num'] = pre_data['week']
model Pretrend = smf.ols('service_score ~ switched_branch * week_num', data=pre_data)
results_pretrend = model_pretrend.fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print(f"\n=== PRE-TREND TEST ===")
print(f"Interaction p-value: {results_pretrend.pvalues['switched_branch:week_num']:.4f}")
print(f"Parallel trends supported: {results_pretrend.pvalues['switched_branch:week_num'] > 0.05}")

# Calculate the effect size in context
overall_mean = df['service_score'].mean()
overall_std = df['service_score'].std()
effect_size_cohens_d = results.params['interaction'] / overall_std
print(f"\n=== EFFECT SIZE IN CONTEXT ===")
print(f"Overall mean service score: {overall_mean:.2f}")
print(f"Overall std dev: {overall_std:.2f}")
print(f"Effect as % of mean: {(results.params['interaction'] / overall_mean) * 100:.1f}%")
print(f"Cohen's d: {effect_size_cohens_d:.2f}")
