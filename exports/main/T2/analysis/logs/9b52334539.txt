import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Main DiD model with controls
model = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get the treatment effect
treatment_effect = model.params['treated_post']
treatment_se = model.bse['treated_post']
treatment_pval = model.pvalues['treated_post']
ci_lower = treatment_effect - 1.96 * treatment_se
ci_upper = treatment_effect + 1.96 * treatment_se

print("MAIN RESULTS:")
print(f"Treatment effect (new procedure): {treatment_effect:.2f} points")
print(f"Standard error: {treatment_se:.2f}")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {treatment_pval:.4f}")

# Calculate effect size relative to baseline
baseline_score = df[df['after_switch']==0]['service_score'].mean()
effect_size_pct = (treatment_effect / baseline_score) * 100
print(f"\nBaseline average score: {baseline_score:.2f}")
print(f"Effect as % of baseline: {effect_size_pct:.1f}%")

# Create a visualization of the parallel trends
plt.figure(figsize=(10, 6))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create week dummies
for w in range(6, 23):
    df[f'week_{w}'] = (df['week'] == w).astype(int)

# Event study regression with week 14 as reference
week_vars = [f'week_{w}' for w in range(6, 23) if w != 14]
formula = 'service_score ~ switched_branch + ' + ' + '.join(week_vars) + ' + ' + ' + '.join([f'switched_branch * week_{w}' for w in range(6, 23) if w != 14])

model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients and standard errors for interaction terms
weeks = list(range(6, 23))
coefs = []
ses = []
for w in weeks:
    if w != 14:
        coef = model.params.get(f'switched_branch:week_{w}', 0)
        se = model.bse.get(f'switched_branch:week_{w}', 0)
    else:
        coef = 0  # Reference week
        se = 0
    coefs.append(coef)
    ses.append(se)

# Create event study plot
plt.figure(figsize=(12, 6))
plt.errorbar(weeks, coefs, yerr=[1.96*s for s in ses], fmt='o', capsize=5, 
             markersize=8, linewidth=2, color='steelblue', label='Treatment Effect by Week')
plt.axhline(y=0, color='gray', linestyle='-', linewidth=1)
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')

# Shade pre and post periods
plt.axvspan(6, 14.5, alpha=0.2, color='orange', label='Pre-Switch Period')
plt.axvspan(14.5, 22, alpha=0.2, color='green', label='Post-Switch Period')

plt.xlabel('Week', fontsize=12)
plt.ylabel('Treatment Effect (vs Week 14)', fontsize=12)
plt.title('Event Study: Treatment Effect by Week\n(Week 14 is Reference)', fontsize=14, fontweight='bold')
plt.legend(loc='best')
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("Event study plot saved as 'event_study.png'")

# Print the coefficients for the report
print("\nEvent Study Coefficients (treatment effect relative to week 14):")
print("-" * 50)
for w, coef, se in zip(weeks, coefs, ses):
    marker = "***" if w >= 15 and coef > 1.96*se else ""
    print(f"Week {w:2d}: {coef:6.2f} (SE: {se:5.2f}) {marker}")
