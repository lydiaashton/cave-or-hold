import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Run the main DiD model with cluster-robust standard errors
model = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get the DiD coefficient
did_coef = model.params['switched_branch:after_switch']
did_se = model.bse['switched_branch:after_switch']
did_pval = model.pvalues['switched_branch:after_switch']
did_ci_lower = did_coef - 1.96 * did_se
did_ci_upper = did_coef + 1.96 * did_se

print("Main Difference-in-Differences Results (with controls):")
print(f"  Coefficient: {did_coef:.2f}")
print(f"  Standard Error: {did_se:.2f}")
print(f"  95% Confidence Interval: [{did_ci_lower:.2f}, {did_ci_upper:.2f}]")
print(f"  P-value: {did_pval:.3f}")

# Calculate effect size relative to baseline
baseline_score = df[df['after_switch'] == 0]['service_score'].mean()
effect_size_pct = (did_coef / baseline_score) * 100
print(f"\n  Baseline average service score: {baseline_score:.1f}")
print(f"  Effect size: {effect_size_pct:.1f}% of baseline")

# Power analysis - how many observations do we have?
n_obs = len(df)
n_branches = df['branch_id'].nunique()
n_treated = df['switched_branch'].sum() // 17  # approximately
n_control = (df['switched_branch'] == 0).sum() // 17

print(f"\n  Total observations: {n_obs}")
print(f"  Number of branches: {n_branches}")
print(f"  Weeks per branch: 17")

# Check if staffing changes might explain results
print("\n\nStaffing Index by Group and Period:")
staffing_summary = df.groupby(['switched_branch', 'after_switch'])['staffing_index'].mean().unstack()
print(staffing_summary)

print("\n\nVisits Index by Group and Period:")
visits_summary = df.groupby(['switched_branch', 'after_switch'])['visits_index'].mean().unstack()
print(visits_summary)
