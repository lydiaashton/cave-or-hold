import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the treatment effect

df['treatment'] = df['switched_branch'] * df['after_switch']

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Also run with cluster-robust standard errors at branch level
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== With Cluster-Robust Standard Errors (by branch) ===")
print(model_clustered.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Final model with controls and cluster-robust SE
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get the treatment effect and confidence interval
treatment_effect = model.params['treatment']
treatment_se = model.bse['treatment']
ci_lower = treatment_effect - 1.96 * treatment_se
ci_upper = treatment_effect + 1.96 * treatment_se
p_value = model.pvalues['treatment']

print("=== FINAL RESULTS ===")
print(f"Treatment Effect: {treatment_effect:.2f} points")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {p_value:.4f}")
print(f"Statistically significant: {p_value < 0.05}")

# Calculate percentage improvement relative to baseline
baseline_score = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
pct_improvement = (treatment_effect / baseline_score) * 100
print(f"\nBaseline score (switched branches before): {baseline_score:.2f}")
print(f"Percentage improvement: {pct_improvement:.1f}%")

# Also show the raw before/after for switched branches
print(f"\nSwitched branches - Before: {baseline_score:.2f}, After: {df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean():.2f}")
print(f"Control branches - Before: {df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean():.2f}, After: {df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean():.2f}")
