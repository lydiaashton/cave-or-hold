import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Create treatment group indicator (switched_branch * after_switch interaction for DiD)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Before/After summary by group
print("=== SERVICE SCORE SUMMARY ===\n")

# Before switch (weeks 6-14)
before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

print("BEFORE SWITCH (Weeks 6-14):")
print(before.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

print("\nAFTER SWITCH (Weeks 15-22):")
print(after.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

# Calculate differences
switched_before = before[before['switched_branch'] == 1]['service_score'].mean()
switched_after = after[after['switched_branch'] == 1]['service_score'].mean()
control_before = before[before['switched_branch'] == 0]['service_score'].mean()
control_after = after[after['switched_branch'] == 0]['service_score'].mean()

print("\n=== DIFFERENCE-IN-DIFFERENCES CALCULATION ===")
print(f"Switched branches: {switched_before:.2f} -> {switched_after:.2f} (change: {switched_after - switched_before:.2f})")
print(f"Control branches:  {control_before:.2f} -> {control_after:.2f} (change: {control_after - control_before:.2f})")
print(f"\nDifference-in-Differences estimate: {(switched_after - switched_before) - (control_after - control_before):.2f}")

# Weekly trends
print("\n=== WEEKLY AVERAGE SERVICE SCORES ===")
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly.columns = ['Control', 'Switched']
weekly['Diff'] = weekly['Switched'] - weekly['Control']
print(weekly)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Difference-in-Differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimator - the causal effect of the new procedure

model = ols('service_score ~ switched_branch + after_switch + switched_branch:after_switch', data=df).fit()
print("=== DIFFERENCE-IN-DIFFERENCES REGRESSION ===")
print(model.summary())

# Also run with branch fixed effects for more robust estimation
print("\n=== REGRESSION WITH BRANCH FIXED EFFECTS ===")
df['branch_id'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ switched_branch:after_switch + C(branch_id)', data=df).fit()
print(model_fe.summary())

# Check parallel trends - look at pre-treatment trends
print("\n=== PRE-TREATMENT TRENDS CHECK ===")
pre_data = df[df['week'] < 15]
pre_model = ols('service_score ~ C(week) * switched_branch', data=pre_data).fit()
print(pre_model.summary())
