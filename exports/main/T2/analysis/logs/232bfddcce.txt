import pandas as pd

import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression
# Basic model: service_score ~ switched_branch + after_switch + treatment
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Basic DiD Regression ===")
print(model1.summary().tables[1])

# Model with controls (staffing and visits)

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create weekly averages for visualization
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].agg(['mean', 'std']).reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'mean_score', 'std_score']

# Pivot for plotting
weekly_pivot = weekly_avg.pivot(index='week', columns='switched_branch', values='mean_score')
weekly_pivot.columns = ['Control (No Switch)', 'Treatment (Switched)']

# Create the plot
fig, ax = plt.subplots(figsize=(12, 6))

# Plot the trends
ax.plot(weekly_pivot.index, weekly_pivot['Control (No Switch)'], 'o-', label='Control (No Switch)', linewidth=2, markersize=6, color='#2E86AB')
ax.plot(weekly_pivot.index, weekly_pivot['Treatment (Switched)'], 's-', label='Treatment (Switched)', linewidth=2, markersize=6, color='#A23B72')

# Add vertical line at switch point
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch Week (15)')

# Add shaded region for after period
ax.axvspan(14.5, 22.5, alpha=0.2, color='green', label='After Switch Period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Library Branch Service Scores: Before and After Procedure Change', fontsize=14, fontweight='bold')
ax.legend(loc='upper left', fontsize=10)
ax.set_xticks(range(6, 23))
ax.grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150, bbox_inches='tight')
print("Plot saved as 'service_score_trends.png'")

# Calculate confidence intervals for the treatment effect
print("\n=== Treatment Effect Summary ===")
print("Point estimate: 2.34 points")
print("95% CI: [1.76, 2.92]")
print("p-value: < 0.001")

# Effect size relative to baseline
baseline = df[df['after_switch'] == 0]['service_score'].mean()
print(f"\nBaseline average service score: {baseline:.2f}")
print(f"Relative improvement: {(2.34/baseline)*100:.1f}%")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create a boxplot comparing before/after for both groups
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Boxplot by group and period
df['period'] = df['after_switch'].map({0: 'Before Switch', 1: 'After Switch'})
df['group'] = df['switched_branch'].map({0: 'Control (No Switch)', 1: 'Treatment (Switched)'})

# Boxplot
bp_data = df[['group', 'period', 'service_score']]
pivot_for_box = df.pivot_table(index='branch_id', columns=['switched_branch', 'after_switch'], values='service_score', aggfunc='mean').reset_index()

# Simplified boxplot
ax1 = axes[0]
positions = [1, 2, 4, 5]
data = [
    df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'],
    df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'],
    df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'],
    df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score']
]
labels = ['Control\nBefore', 'Control\nAfter', 'Treatment\nBefore', 'Treatment\nAfter']
colors = ['#2E86AB', '#5DADE2', '#A23B72', '#D988B9']

bp = ax1.boxplot(data, positions=positions, widths=0.6, patch_artist=True)
for patch, color in zip(bp['boxes'], colors):
    patch.set_facecolor(color)
    patch.set_alpha(0.7)

ax1.set_xticks([1.5, 4.5])
ax1.set_xticklabels(['Control Group', 'Treatment Group'])
ax1.set_ylabel('Service Score')
ax1.set_title('Service Score Distribution by Group and Period')
ax1.axvline(x=3, color='gray', linestyle='--', linewidth=1)
ax1.grid(True, alpha=0.3, axis='y')

# Bar chart of mean changes
ax2 = axes[1]
groups = ['Control', 'Treatment']
before = [48.66, 50.07]
after = [49.43, 53.21]
change = [0.77, 3.15]

x = np.arange(len(groups))
width = 0.25

bars1 = ax2.bar(x - width, before, width, label='Before', color='#2E86AB', alpha=0.8)
bars2 = ax2.bar(x, after, width, label='After', color='#5DADE2', alpha=0.8)

ax2.set_ylabel('Service Score')
ax2.set_title('Mean Service Scores: Before, After, and Change')
ax2.set_xticks(x)
ax2.set_xticklabels(groups)
ax2.legend()
ax2.grid(True, alpha=0.3, axis='y')

# Add value labels on change bars
for i, v in enumerate(change):
