print("\nAfter switch (weeks 15-22):")
after = df[df['week'] >= 15].groupby('switched_branch')['service_score'].mean()
print(after)

# Calculate the differences
print("\n=== Difference-in-Differences Calculation ===")

control_before = before[0]
control_after = after[0]
control_diff = control_after - control_before

# Treatment group (switched_branch=1)
treat_before = before[1]
treat_after = after[1]
treat_diff = treat_after - treat_before

# DiD estimate
did_estimate = treat_diff - control_diff

print(f"\nControl group change: {control_diff:.3f}")
print(f"Treatment group change: {treat_diff:.3f}")
print(f"\nDiD Estimate (Treatment effect): {did_estimate:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create proper event study variables
# Week 14 is the last pre-switch week (reference)
df['relative_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create treatment interaction for each week
for w in df['week'].unique():
    if w != 14:  # Skip reference week
        df[f'treat_week{w}'] = ((df['week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
week_cols = [f'treat_week{w}' for w in sorted(df['week'].unique()) if w != 14]
formula = 'service_score ~ switched_branch + C(week) + ' + ' + '.join(week_cols)
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for treatment weeks
print("=== Event Study: Treatment Effects by Week (relative to week 14) ===")
results = []
for w in sorted(df['week'].unique()):
    if w == 14:
        results.append((w, 0, 0, 1.0))  # Reference week
    else:
        coef_name = f'treat_week{w}'
        if coef_name in event_model.params:
            coef = event_model.params[coef_name]
            se = event_model.bse[coef_name]
            pval = event_model.pvalues[coef_name]
            results.append((w, coef, se, pval))

for w, coef, se, pval in results:
    marker = '*' if pval < 0.1 else ' '
    print(f"Week {w:2d}: {coef:6.2f} (SE: {se:.2f}, p={pval:.3f}) {marker}")

# Create event study plot
fig, ax = plt.subplots(figsize=(12, 6))
weeks = [r[0] for r in results]
coefs = [r[1] for r in results]
ses = [r[2] for r in results]

ax.errorbar(weeks, coefs, yerr=1.96*np.array(ses), fmt='o', capsize=5, markersize=8, linewidth=2)
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch implemented')
ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Treatment Effect (vs Week 14)', fontsize=12)
ax.set_title('Event Study: Weekly Treatment Effects on Service Score', fontsize=14)
ax.set_xticks(weeks)
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study_plot.png', dpi=150)
print("\nEvent study plot saved as event_study_plot.png")

# Let's also check the average pre vs post more carefully
print("\n=== Summary Statistics ===")
print("\nPre-period (weeks 6-14):")
pre = df[df['week'] < 15]
print(pre.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

print("\nPost-period (weeks 15-22):")
post = df[df['week'] >= 15]
print(post.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

# Calculate confidence interval for DiD estimate
df['treatment'] = df['switched_branch'] * df['after_switch']
did_model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
did_coef = did_model.params['treatment']
did_se = did_model.bse['treatment']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se
print(f"\n=== DiD Estimate Summary ===")
print(f"Point estimate: {did_coef:.2f}")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {did_model.pvalues['treatment']:.3f}")
