import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check treatment/control balance
print("Treatment/Control Balance:")
print(df.groupby('switched_branch')['branch_id'].nunique())
print()

# Calculate pre/post means by treatment status
print("Service Score Means:")
pre_treatment = df[(df['after_switch'] == 0) & (df['switched_branch'] == 1)]['service_score'].mean()
pre_control = df[(df['after_switch'] == 0) & (df['switched_branch'] == 0)]['service_score'].mean()
post_treatment = df[(df['after_switch'] == 1) & (df['switched_branch'] == 1)]['service_score'].mean()
post_control = df[(df['after_switch'] == 1) & (df['switched_branch'] == 0)]['service_score'].mean()

print(f"Pre-switch, Treatment group: {pre_treatment:.2f}")
print(f"Pre-switch, Control group: {pre_control:.2f}")
print(f"Post-switch, Treatment group: {post_treatment:.2f}")
print(f"Post-switch, Control group: {post_control:.2f}")

# Calculate DiD estimate manually
diff_treatment = post_treatment - pre_treatment
diff_control = post_control - pre_control
did_estimate = diff_treatment - diff_control

print(f"\nDifference in Treatment (Post - Pre): {diff_treatment:.2f}")
print(f"Difference in Control (Post - Pre): {diff_control:.2f}")
print(f"Difference-in-Differences estimate: {did_estimate:.2f}")

# Run DiD regression
print("\n" + "="*60)
print("DiD Regression Results:")
print("="*60)

# Basic DiD model
model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit()
print("\n" + "="*60)
print("DiD Regression with Controls:")

print(model2.summary().tables[1])

# Cluster-robust standard errors by branch
print("\n" + "="*60)
print("DiD with Cluster-Robust SE (by branch):")
print("="*60)
model3 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model3.summary().tables[1])

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check parallel trends - average service score by week and treatment status
weekly_means = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print("Weekly Service Score Means by Treatment Status:")
print(weekly_means)

# Check pre-trends statistically (weeks 6-14)
pre_df = df[df['week'] < 15]
pre_trends = pre_df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

# Run regression to test parallel trends in pre-period
pre_df = df[df['week'] < 15].copy()
pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()
model_trends = ols('service_score ~ switched_branch * week_centered', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print("\n" + "="*60)
print("Pre-Trend Test (interaction should be non-significant):")
print("="*60)
print(model_trends.summary().tables[1])

# Create visualization
plt.figure(figsize=(12, 6))

# Plot treatment and control groups over time
weeks = sorted(df['week'].unique())
treatment_means = [df[(df['week'] == w) & (df['switched_branch'] == 1)]['service_score'].mean() for w in weeks]
control_means = [df[(df['week'] == w) & (df['switched_branch'] == 0)]['service_score'].mean() for w in weeks]

plt.plot(weeks, treatment_means, 'o-', label='Treatment (Switched)', linewidth=2, markersize=8)
plt.plot(weeks, control_means, 's-', label='Control (No Switch)', linewidth=2, markersize=8)

# Add vertical line at switch point
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')

plt.xlabel('Week', fontsize=12)
plt.ylabel('Service Score', fontsize=12)
plt.title('Service Score Trends: Treatment vs Control Groups', fontsize=14, fontweight='bold')
plt.legend(fontsize=11)
plt.grid(True, alpha=0.3)
plt.xticks(weeks)
plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("\nPlot saved as 'service_score_trends.png'")

# Calculate confidence intervals for the DiD estimate
from statsmodels.stats.weightstats import DescrStatsW

# Get the DiD estimate with CI from the model
model_final = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
did_coef = model_final.params['switched_branch:after_switch']
did_se = model_final.bse['switched_branch:after_switch']
did_ci_lower = did_coef - 1.96 * did_se
did_ci_upper = did_coef + 1.96 * did_se

print(f"\nDiD Estimate: {did_coef:.2f}")
print(f"95% CI: [{did_ci_lower:.2f}, {did_ci_upper:.2f}]")
print(f"P-value: {model_final.pvalues['switched_branch:after_switch']:.4f}")
