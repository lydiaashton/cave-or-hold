import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# Create treatment group indicator (switched AND after)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary by group and period
print("=== Service Score Summary ===")
print("\nBefore switch (weeks 6-14):")
before = df[df['after_switch'] == 0]
print("  Switched branches:", before[before['switched_branch'] == 1]['service_score'].mean())
print("  Non-switched branches:", before[before['switched_branch'] == 0]['service_score'].mean())

print("\nAfter switch (weeks 15-22):")
after = df[df['after_switch'] == 1]
print("  Switched branches:", after[after['switched_branch'] == 1]['service_score'].mean())
print("  Non-switched branches:", after[after['switched_branch'] == 0]['service_score'].mean())

# Difference-in-differences calculation
switched_before = before[before['switched_branch'] == 1]['service_score'].mean()
switched_after = after[after['switched_branch'] == 1]['service_score'].mean()
control_before = before[before['switched_branch'] == 0]['service_score'].mean()
control_after = after[after['switched_branch'] == 0]['service_score'].mean()

switched_change = switched_after - switched_before
control_change = control_after - control_before
did_estimate = switched_change - control_change

print(f"\n=== Difference-in-Differences ===")
print(f"Switched branches change: {switched_change:.2f}")
print(f"Control branches change: {control_change:.2f}")
print(f"DiD estimate (treatment effect): {did_estimate:.2f}")

# Check balance before the switch
print(f"\n=== Pre-switch Balance Check ===")
print(f"Switched branches (before): n={len(before[before['switched_branch'] == 1])}, mean={switched_before:.2f}, std={before[before['switched_branch'] == 1]['service_score'].std():.2f}")
print(f"Control branches (before): n={len(before[before['switched_branch'] == 0])}, mean={control_before:.2f}, std={before[before['switched_branch'] == 0]['service_score'].std():.2f}")

# Simple t-test on the treatment effect
# Create a dataset with one observation per branch-period
branch_period = df.groupby(['branch_id', 'after_switch'])['service_score'].mean().reset_index()
branch_period.columns = ['branch_id', 'period', 'avg_score']

# Merge back switch status
branch_info = df[['branch_id', 'switched_branch']].drop_duplicates()
branch_period = branch_period.merge(branch_info, on='branch_id')

# Calculate change for each branch
before_scores = branch_period[branch_period['period'] == 0][['branch_id', 'avg_score']].copy()
before_scores.columns = ['branch_id', 'before_score']
after_scores = branch_period[branch_period['period'] == 1][['branch_id', 'avg_score']].copy()
after_scores.columns = ['branch_id', 'after_score']

branch_changes = before_scores.merge(after_scores, on='branch_id')
branch_changes['change'] = branch_changes['after_score'] - branch_changes['before_score']
branch_changes = branch_changes.merge(branch_info, on='branch_id')

print(f"\n=== Branch-level Change Analysis ===")
switched_changes = branch_changes[branch_changes['switched_branch'] == 1]['change']
control_changes = branch_changes[branch_changes['switched_branch'] == 0]['change']

print(f"Switched branches: mean change = {switched_changes.mean():.2f}, std = {switched_changes.std():.2f}, n = {len(switched_changes)}")
print(f"Control branches: mean change = {control_changes.mean():.2f}, std = {control_changes.std():.2f}, n = {len(control_changes)}")

# Two-sample t-test
t_stat, p_value = stats.ttest_ind(switched_changes, control_changes)
print(f"\nTwo-sample t-test: t = {t_stat:.2f}, p-value = {p_value:.4f}")

# Confidence interval for the difference
diff_mean = switched_changes.mean() - control_changes.mean()
diff_std = np.sqrt(switched_changes.var()/len(switched_changes) + control_changes.var()/len(control_changes))
ci_lower = diff_mean - 1.96 * diff_std
ci_upper = diff_mean + 1.96 * diff_std
print(f"95% CI for treatment effect: [{ci_lower:.2f}, {ci_upper:.2f}]")

import pandas as pd

import matplotlib.pyplot as plt
from scipy import stats

df = pd.read_csv('branches.csv')

# Check parallel trends - average service score by week for each group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched', 'avg_score']

print("=== Weekly Trends ===")
print(weekly_avg.pivot(index='week', columns='switched', values='avg_score'))

# Plot the trends (save to file)
plt.figure(figsize=(10, 6))
for switched in [0, 1]:
    subset = weekly_avg[weekly_avg['switched'] == switched]
    label = 'Switched branches' if switched == 1 else 'Control branches'
    plt.plot(subset['week'], subset['avg_score'], marker='o', label=label, linewidth=2)

plt.axvline(x=14.5, color='red', linestyle='--', label='Switch occurred')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Score Trends by Branch Group')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('trends.png', dpi=150, bbox_inches='tight')
print("\nTrend plot saved to trends.png")

# Regression with controls
print("\n=== Regression Analysis ===")

# Simple DiD regression
df['const'] = 1
df['post'] = df['after_switch']
df['treated'] = df['switched_branch']
df['did'] = df['treated'] * df['post']

# OLS regression manually
X = df[['const', 'treated', 'post', 'did']].values
y = df['service_score'].values

# Add controls
X_with_controls = df[['const', 'treated', 'post', 'did', 'staffing_index', 'visits_index']].values

def ols(X, y):
    beta = np.linalg.inv(X.T @ X) @ X.T @ y
    y_pred = X @ beta
    residuals = y - y_pred
    n = len(y)
    k = X.shape[1]
    var_beta = np.linalg.inv(X.T @ X) * (residuals.T @ residuals) / (n - k)
    se_beta = np.sqrt(np.diag(var_beta))
    t_stats = beta / se_beta
    return beta, se_beta, t_stats

beta_simple, se_simple, t_simple = ols(X, y)
beta_ctrl, se_ctrl, t_ctrl = ols(X_with_controls, y)

print("Simple DiD Regression:")
print(f"  DID coefficient: {beta_simple[3]:.3f} (SE: {se_simple[3]:.3f}, t: {t_simple[3]:.2f})")

print("\nDiD Regression with Controls (staffing, visits):")
print(f"  DID coefficient: {beta_ctrl[3]:.3f} (SE: {se_ctrl[3]:.3f}, t: {t_ctrl[3]:.2f})")
print(f"  Staffing coefficient: {beta_ctrl[4]:.3f} (SE: {se_ctrl[4]:.3f})")
print(f"  Visits coefficient: {beta_ctrl[5]:.3f} (SE: {se_ctrl[5]:.3f})")

# Check if there's heterogeneity - effect size by branch
print("\n=== Heterogeneity Check ===")
branch_effects = df.groupby(['branch_id', 'switched_branch', 'after_switch'])['service_score'].mean().unstack()
branch_effects['change'] = branch_effects[1] - branch_effects[0]
switched_branches = df[['branch_id', 'switched_branch']].drop_duplicates()
switched_branches = switched_branches[switched_branches['switched_branch'] == 1]['branch_id']

switched_effects = branch_effects.loc[switched_branches, 'change']
print(f"Switched branches individual effects:")

print(f"  Max: {switched_effects.max():.2f}")
print(f"  Median: {switched_effects.median():.2f}")
print(f"  Number with positive effect: {(switched_effects > 0).sum()} of {len(switched_effects)}")

import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# Pre-trends analysis: compare trends before the switch (weeks 6-14)
pre_data = df[df['week'] <= 14].copy()

# Calculate week-to-week changes for each group
print("=== Pre-Trends Analysis (Weeks 6-14) ===")

# Average score by week for each group
pre_weekly = pre_data.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
pre_weekly.columns = ['week', 'switched', 'avg_score']

# Calculate linear trend for each group before switch
switched_pre = pre_weekly[pre_weekly['switched'] == 1]
control_pre = pre_weekly[pre_weekly['switched'] == 0]

# Linear regression for trends
def linear_trend(weeks, scores):
    X = np.column_stack([np.ones(len(weeks)), weeks])
    beta = np.linalg.inv(X.T @ X) @ X.T @ scores
    return beta[1]  # slope

switched_slope = linear_trend(switched_pre['week'].values, switched_pre['avg_score'].values)
control_slope = linear_trend(control_pre['week'].values, control_pre['avg_score'].values)

print(f"Pre-switch trend (switched branches): {switched_slope:.3f} points/week")
print(f"Pre-switch trend (control branches): {control_slope:.3f} points/week")
print(f"Difference in pre-trends: {switched_slope - control_slope:.3f} points/week")

# Statistical test for parallel pre-trends
# Compare the change from first week to last week before switch
switched_first = pre_data[(pre_data['switched_branch'] == 1) & (pre_data['week'] == 6)]['service_score'].mean()
switched_last = pre_data[(pre_data['switched_branch'] == 1) & (pre_data['week'] == 14)]['service_score'].mean()
control_first = pre_data[(pre_data['switched_branch'] == 0) & (pre_data['week'] == 6)]['service_score'].mean()
control_last = pre_data[(pre_data['switched_branch'] == 0) & (pre_data['week'] == 14)]['service_score'].mean()

print(f"\nSwitched: week 6 = {switched_first:.2f}, week 14 = {switched_last:.2f}, change = {switched_last - switched_first:.2f}")
print(f"Control: week 6 = {control_first:.2f}, week 14 = {control_last:.2f}, change = {control_last - control_first:.2f}")

# Power analysis - what effect size could we detect?
print("\n=== Power Considerations ===")
n_treated = 20
n_control = 14
pooled_std = np.sqrt((19 * 2.52**2 + 13 * 4.13**2) / (20 + 14 - 2))
print(f"Pooled standard deviation: {pooled_std:.2f}")

# Minimum detectable effect at 80% power, alpha=0.05
# Using approximation: MDE ≈ (t_alpha + t_beta) * SE
from scipy.stats import t
df_deg = n_treated + n_control - 2
t_alpha = t.ppf(0.975, df_deg)  # two-tailed
t_beta = t.ppf(0.80, df_deg)  # 80% power
se = pooled_std * np.sqrt(1/n_treated + 1/n_control)
mde = (t_alpha + t_beta) * se
print(f"Minimum detectable effect (80% power): ~{mde:.2f} points")
print(f"Observed effect: 1.91 points")
print(f"This study had limited power to detect small effects.")

# Effect size (Cohen's d)
cohen_d = 1.91 / pooled_std
print(f"\nEffect size (Cohen's d): {cohen_d:.2f}")
print("Interpretation: small-to-medium effect")

# Summary stats for the report
print("\n=== Final Summary Statistics ===")
print(f"Total observations: {len(df)}")
print(f"Branches: {df['branch_id'].nunique()} (switched: {df['switched_branch'].sum()}, control: {df['branch_id'].nunique() - df['switched_branch'].sum()})")
print(f"Weeks: {df['week'].min()} to {df['week'].max()}")
print(f"Switch occurred: week 15")
print(f"\nAverage service score (all): {df['service_score'].mean():.1f}")
print(f"Switched branches - before: {df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean():.1f}")
print(f"Switched branches - after: {df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean():.1f}")
print(f"Control branches - before: {df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean():.1f}")
print(f"Control branches - after: {df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean():.1f}")
