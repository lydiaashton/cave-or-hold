# Calculate weekly means
weekly_pre = pre_period.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Fit trend lines for pre-period
from scipy import stats

treated_pre = weekly_pre[weekly_pre['switched_branch'] == 1]

# Linear regression for pre-trends
control_slope, control_intercept, control_r, control_p, control_se = stats.linregress(control_pre['week'], control_pre['service_score'])

print(f"Treated group: slope = {treated_slope:.3f}, p-value = {treated_p:.3f}")
print(f"Difference in slopes: {treated_slope - control_slope:.3f}")

pre_period = pre_period.copy()
pre_period['week_centered'] = pre_period['week'] - pre_period['week'].mean()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create interaction term
df['treated'] = df['switched_branch'] * df['after_switch']

# Two-way fixed effects model (branch FE + week FE)
# This controls for time-invariant branch characteristics and common time shocks
df['branch_id_str'] = df['branch_id'].astype(str)
df['week_str'] = df['week'].astype(str)

# Run with branch and week fixed effects
model_fe = ols('service_score ~ treated + staffing_index + visits_index + C(branch_id_str) + C(week_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Two-Way Fixed Effects Model (Branch + Week FE):")
print("Treatment effect (treated coefficient):")
print(f"  Coefficient: {model_fe.params['treated']:.3f}")
print(f"  Std Error: {model_fe.bse['treated']:.3f}")
print(f"  95% CI: [{model_fe.conf_int().loc['treated'][0]:.3f}, {model_fe.conf_int().loc['treated'][1]:.3f}]")
print(f"  P-value: {model_fe.pvalues['treated']:.4f}")

# Also check without controls
model_fe_simple = ols('service_score ~ treated + C(branch_id_str) + C(week_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nTwo-Way Fixed Effects Model (without controls):")
print(f"  Coefficient: {model_fe_simple.params['treated']:.3f}")
print(f"  Std Error: {model_fe_simple.bse['treated']:.3f}")
print(f"  95% CI: [{model_fe_simple.conf_int().loc['treated'][0]:.3f}, {model_fe_simple.conf_int().loc['treated'][1]:.3f}]")
print(f"  P-value: {model_fe_simple.pvalues['treated']:.4f}")

# Event study - effect by week after treatment
df['weeks_since_switch'] = df['week'] - 15
df['weeks_since_switch'] = df['weeks_since_switch'].clip(-9, 8)  # Cap at -9 to 8

# Create dummies for each week relative to switch
for w in range(-9, 9):
    if w != -1:  # Use week -1 as reference
        df[f'week_{w}'] = (df['weeks_since_switch'] == w).astype(int)

# Run event study regression (week -1 is omitted reference)
event_cols = [f'week_{w}' for w in range(-9, 9) if w != -1]
event_formula = 'service_score ~ ' + ' + '.join(event_cols) + ' + C(branch_id_str) + C(week_str)'
model_event = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nEvent Study Coefficients (relative to week -1):")
event_results = []
for w in range(-9, 9):
    if w == -1:
        event_results.append((w, 0, 0, 0, 1))  # Reference
    else:
        col = f'week_{w}'
        coef = model_event.params[col]
        se = model_event.bse[col]
        pval = model_event.pvalues[col]
        event_results.append((w, coef, se, coef - 1.96*se, coef + 1.96*se))
        print(f"  Week {w:+2d}: coef={coef:.2f}, SE={se:.2f}, 95%CI=[{coef-1.96*se:.2f}, {coef+1.96*se:.2f}], p={pval:.3f}")

import pandas as pd

import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create interaction term
df['treated'] = df['switched_branch'] * df['after_switch']

# Event study - effect by week relative to switch
df['weeks_since_switch'] = df['week'] - 15

# Create dummies for each week relative to switch (use w prefix to avoid issues with negative numbers)
for w in range(-9, 9):
    if w != -1:  # Use week -1 as reference
        df[f'w{abs(w)}_neg' if w < 0 else f'w{w}_pos'] = (df['weeks_since_switch'] == w).astype(int)

# Build formula
event_cols = []
for w in range(-9, 9):
    if w != -1:
        if w < 0:
            event_cols.append(f'w{abs(w)}_neg')
        else:
            event_cols.append(f'w{w}_pos')

event_formula = 'service_score ~ ' + ' + '.join(event_cols) + ' + C(branch_id) + C(week)'
model_event = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Coefficients (relative to week -1):")
event_results = []
for w in range(-9, 9):
    if w == -1:
        event_results.append((w, 0, 0, 0, 0, 1))  # Reference
    else:
        if w < 0:
            col = f'w{abs(w)}_neg'
        else:
            col = f'w{w}_pos'
        coef = model_event.params[col]
        se = model_event.bse[col]
        pval = model_event.pvalues[col]
        ci_low = coef - 1.96*se
        ci_high = coef + 1.96*se
        event_results.append((w, coef, se, ci_low, ci_high, pval))
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"  Week {w:+2d}: coef={coef:6.2f}, SE={se:.2f}, 95%CI=[{ci_low:6.2f}, {ci_high:6.2f}], p={pval:.3f} {sig}")

# Create event study plot
plt.figure(figsize=(12, 6))
weeks = [r[0] for r in event_results]
coefs = [r[1] for r in event_results]
ci_low = [r[3] for r in event_results]
ci_high = [r[4] for r in event_results]
