import pandas as pd

df = pd.read_csv('branches.csv')

branches = df.groupby('branch_id')['switched_branch'].first()

print("Total branches:", len(branches))

# Calculate the difference-in-differences
# Treatment group (switched=1): before and after

control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()

control_change = control_after - control_before

# Difference-in-differences
did_estimate = treatment_change - control_change

print("\n=== Difference-in-Differences Analysis ===")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Run DiD regression with branch fixed effects
# Model: service_score = beta0 + beta1*treatment + beta2*post + beta3*did + branch_FE + error
# But since treatment is time-invariant, it gets absorbed by branch FE
# So we use: service_score = beta0 + beta2*post + beta3*did + branch_FE + error

# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)

# Prepare regression
X = df[['post', 'did']].copy()
X = pd.concat([X, branch_dummies], axis=1)
X = sm.add_constant(X)
y = df['service_score']

model = ols('service_score ~ post + did + C(branch_id)', data=df).fit()
print("=== DiD Regression with Branch Fixed Effects ===")
print(model.summary())
