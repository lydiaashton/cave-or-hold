import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Basic DiD calculation - mean service scores by group and period
print("=== Difference-in-Differences Analysis ===\n")

# Calculate means for each of the 4 groups
group_means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
print("Mean Service Scores by Group and Period:")
print(group_means)
print()

# Calculate the DiD estimator manually
# Treatment group change
treatment_pre = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_post = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_change = treatment_post - treatment_pre

# Control group change
control_pre = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_post = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
control_change = control_post - control_pre

# DiD estimator
did_estimator = treatment_change - control_change

print(f"Treatment group (switched branches):")
print(f"  Pre-switch mean: {treatment_pre:.2f}")
print(f"  Post-switch mean: {treatment_post:.2f}")
print(f"  Change: {treatment_change:.2f}")
print()
print(f"Control group (non-switched branches):")
print(f"  Pre-switch mean: {control_pre:.2f}")
print(f"  Post-switch mean: {control_post:.2f}")
print(f"  Change: {control_change:.2f}")
print()
print(f"Difference-in-Differences Estimate: {did_estimator:.2f}")
print()

# Run regression with controls
print("=== Regression Analysis ===\n")

# Model 1: Basic DiD
model1 = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
print("Model 1: Basic DiD (no controls)")
print(model1.summary().tables[1])
print()

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit()
print("Model 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])
print()

# Model 3: With branch fixed effects
model3 = ols('service_score ~ after_switch + did + staffing_index + visits_index + C(branch_id)', data=df).fit()
print("Model 3: DiD with branch fixed effects and controls")
# Just show the key coefficients
print("\nKey coefficients:")
print(f"  after_switch: {model3.params['after_switch']:.4f} (SE: {model3.bse['after_switch']:.4f})")
print(f"  did (treatment effect): {model3.params['did']:.4f} (SE: {model3.bse['did']:.4f})")
print(f"  p-value for did: {model3.pvalues['did']:.4f}")

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Event study analysis - estimate effect for each week relative to switch
# Using week 14 as the reference (last pre-period week)

# Create relative week indicators
df['rel_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create indicators for each relative week
for rw in range(-8, 9):  # weeks -8 to +8 relative to week 14
    df[f'rel_{rw}'] = (df['rel_week'] == rw).astype(int)

# Create interaction terms (treatment x relative week)
for rw in range(-8, 9):
    df[f'did_rel_{rw}'] = df['switched_branch'] * df[f'rel_{rw}']

# Event study regression (omit rel_0 as reference)
rel_vars = [f'did_rel_{rw}' for rw in range(-8, 9) if rw != 0]
rel_vars_str = ' + '.join(rel_vars)

# Full model with branch FE
event_study_formula = f'service_score ~ switched_branch + {" + ".join([f"rel_{rw}" for rw in range(-8, 9) if rw != 0])} + {rel_vars_str} + staffing_index + visits_index + C(branch_id)'

# This might be too many variables, let's simplify
# Just do the key event study without all the individual week dummies

# Alternative: estimate effect for each post-week separately with pre-week average as baseline
print("=== Event Study Analysis ===\n")

# Calculate treatment effect for each week relative to pre-period baseline
# Using a regression approach with week-specific treatment effects

results = []
for week in range(6, 23):
    week_data = df[df['week'] == week].copy()
    if len(week_data) == 0:
        continue
    
    # Simple comparison for this week
    treat_mean = week_data[week_data['switched_branch'] == 1]['service_score'].mean()
    control_mean = week_data[week_data['switched_branch'] == 0]['service_score'].mean()
    diff = treat_mean - control_mean
    
    results.append({
        'week': week,
        'period': 'pre' if week < 15 else 'post',
        'treat_mean': treat_mean,
        'control_mean': control_mean,
        'diff': diff

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Final analysis for the summary
df['did'] = df['switched_branch'] * df['after_switch']

# Main model with controls and branch FE
model = ols('service_score ~ after_switch + did + staffing_index + visits_index + C(branch_id)', data=df).fit()

# Key statistics
treatment_effect = model.params['did']
se = model.bse['did']
ci_low = treatment_effect - 1.96 * se
ci_high = treatment_effect + 1.96 * se
p_value = model.pvalues['did']

# Calculate percentage improvement
avg_score = df['service_score'].mean()
pct_improvement = (treatment_effect / avg_score) * 100

# Branch counts
