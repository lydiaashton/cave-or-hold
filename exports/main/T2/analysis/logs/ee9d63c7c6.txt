print("\n\nAverage service scores:")

pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='period', 
                       aggfunc='mean')
pivot.columns = ['Before (weeks 6-14)', 'After (weeks 15-22)']
pivot.index = pivot.index.map({0: 'Control (old procedure)', 1: 'Treatment (new procedure)'})
print(pivot)

# Calculate the difference-in-differences manually
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nControl group: Before={control_before:.2f}, After={control_after:.2f}, Change={control_after-control_before:.2f}")
print(f"Treatment group: Before={treatment_before:.2f}, After={treatment_after:.2f}, Change={treatment_after-treatment_before:.2f}")
did_effect = (treatment_after - treatment_before) - (control_after - control_before)
print(f"\nDifference-in-Differences estimate: {did_effect:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Pre-trends test: only use pre-switch data (weeks 6-14)
pre_df = df[df['week'] <= 14].copy()

# Create week dummies (using week 6 as reference)
for w in range(7, 15):
    pre_df[f'week_{w}'] = (pre_df['week'] == w).astype(int)

# Test for differential pre-trends
# Interact week dummies with switched_branch
formula = 'service_score ~ ' + ' + '.join([f'week_{w}:switched_branch' for w in range(7, 15)]) + ' + switched_branch'
model = ols(formula, data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print("Pre-trends test (weeks 7-14 vs week 6 baseline):")
print(model.summary())

# Joint test for all pre-trend interactions
pre_trend_coefs = [model.params[f'week_{w}:switched_branch'] for w in range(7, 15)]
print(f"\nPre-trend coefficients: {[round(c, 2) for c in pre_trend_coefs]}")

# F-test for joint significance
hypotheses = [f'week_{w}:switched_branch = 0' for w in range(7, 15)]
f_test = model.f_test(hypotheses)
print(f"\nJoint F-test for pre-trends: F={f_test.fvalue:.2f}, p={f_test.pvalue:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Better pre-trends test: test for differential linear time trend in pre-period
pre_df = df[df['week'] <= 14].copy()
pre_df['week_centered'] = pre_df['week'] - 10  # Center around middle of pre-period

# Test: does the treatment group have a different time trend before the switch?
model = ols('service_score ~ week_centered * switched_branch', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print("Pre-trends test (linear time trend interaction):")
print(model.summary())
print(f"\nInteraction coefficient (differential pre-trend): {model.params['week_centered:switched_branch']:.3f}, p={model.pvalues['week_centered:switched_branch']:.4f}")

# Also run an event study style regression to see effects by week
df['did'] = df['after_switch'] * df['switched_branch']

# Create week dummies relative to switch (week 15 is the first "after" week)
df['relative_week'] = df['week'] - 15

# Event study regression (week 14 as reference, which is the last pre-switch week)
event_df = df.copy()
for w in range(-9, 8):  # weeks -9 to +7 relative to week 15
    if w != -1:  # skip week 14 (reference)
        event_df[f'rel_week_{w}'] = ((event_df['relative_week'] == w) & (event_df['switched_branch'] == 1)).astype(int)

# Run event study
event_formula = 'service_score ~ ' + ' + '.join([f'rel_week_{w}' for w in range(-9, 8) if w != -1]) + ' + switched_branch + after_switch'
event_model = ols(event_formula, data=event_df).fit(cov_type='cluster', cov_kwds={'groups': event_df['branch_id']})

print("\n\nEvent Study Coefficients (treatment effects by week relative to switch):")
for w in range(-9, 8):
    if w != -1:
        coef = event_model.params.get(f'rel_week_{w}', 0)
        se = event_model.bse.get(f'rel_week_{w}', 0)
        print(f"Week {15+w} (relative {w:+d}): {coef:.2f} (SE: {se:.2f})")
