import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Count branches by switch status
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate average service scores by group and period
print("\n\nAverage service scores by group and period:")
df['period'] = df['after_switch'].map({0: 'Before', 1: 'After'})
df['group'] = df['switched_branch'].map({0: 'Control', 1: 'Treatment'})

pivot = df.pivot_table(values='service_score', 
                       index='group', 
                       columns='period', 
                       aggfunc='mean')
print(pivot)

# Calculate the difference-in-differences manually
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n\nDetailed averages:")
print(f"Control Before: {control_before:.3f}")
print(f"Control After: {control_after:.3f}")
print(f"Control Change: {control_after - control_before:.3f}")
print(f"\nTreatment Before: {treatment_before:.3f}")
print(f"Treatment After: {treatment_after:.3f}")
print(f"Treatment Change: {treatment_after - treatment_before:.3f}")

did_effect = (treatment_after - treatment_before) - (control_after - control_before)
print(f"\nDifference-in-Differences Estimate: {did_effect:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with robust standard errors
# Model 1: Basic DiD
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='HC1')
print("Model 1: Basic DiD Regression")
print(model1.summary())

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='HC1')
print("\n\nModel 2: DiD with Controls")
print(model2.summary())

# Model 3: DiD with branch fixed effects (using dummy variables for branches)
# This is more computationally intensive but controls for time-invariant branch characteristics
print("\n\nModel 3: DiD with Branch Fixed Effects")
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [col for col in df_with_dummies.columns if col.startswith('branch_')]
formula = 'service_score ~ after_switch + treatment + staffing_index + visits_index + ' + ' + '.join(branch_cols)
model3 = ols(formula, data=df_with_dummies).fit(cov_type='HC1')
print(f"R-squared: {model3.rsquared:.4f}")
print(f"Treatment effect coefficient: {model3.params['treatment']:.4f}")
print(f"Treatment effect std err: {model3.bse['treatment']:.4f}")
print(f"Treatment effect p-value: {model3.pvalues['treatment']:.6f}")

import pandas as pd

# Check parallel trends - average service score by week for each group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg['group'] = weekly_avg['switched_branch'].map({0: 'Control', 1: 'Treatment'})

print("Weekly averages by group:")
print(weekly_avg.pivot(index='week', columns='group', values='service_score'))

# Calculate the difference between treatment and control each week
pivot = weekly_avg.pivot(index='week', columns='group', values='service_score')
pivot['diff'] = pivot['Treatment'] - pivot['Control']
print("\n\nDifference (Treatment - Control) by week:")
print(pivot['diff'])

pre_trend = pivot.loc[6:14, 'diff']
print(f"\nPre-switch trend (weeks 6-14):")
print(f"Mean difference: {pre_trend.mean():.3f}")
print(f"Std dev: {pre_trend.std():.3f}")

post_trend = pivot.loc[15:22, 'diff']
print(f"\nPost-switch trend (weeks 15-22):")

print(f"Std dev: {post_trend.std():.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Add week as a categorical variable to control for time trends
df['week_cat'] = df['week'].astype('category')

# Run DiD with week fixed effects (controls for common time trends)
# This is more robust than just the after_switch indicator
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model with week fixed effects
model_week_fe = ols('service_score ~ switched_branch + treatment + staffing_index + visits_index + C(week)', data=df).fit(cov_type='HC1')
print("Model with Week Fixed Effects:")
print(f"Treatment effect: {model_week_fe.params['treatment']:.4f}")
print(f"Std Error: {model_week_fe.bse['treatment']:.4f}")
print(f"P-value: {model_week_fe.pvalues['treatment']:.6f}")
print(f"95% CI: [{model_week_fe.conf_int().loc['treatment', 0]:.4f}, {model_week_fe.conf_int().loc['treatment', 1]:.4f}]")

# Also check if effect varies by how long after switch
df['weeks_after'] = np.maximum(0, df['week'] - 14)  # 0 before switch, 1+ after
df['treatment_intensity'] = df['switched_branch'] * df['weeks_after']

model_intensity = ols('service_score ~ switched_branch + after_switch + treatment_intensity + staffing_index + visits_index', data=df).fit(cov_type='HC1')
print("\n\nModel with Treatment Intensity (weeks after switch):")
print(f"Treatment intensity effect (per week): {model_intensity.params['treatment_intensity']:.4f}")
print(f"Std Error: {model_intensity.bse['treatment_intensity']:.4f}")
print(f"P-value: {model_intensity.pvalues['treatment_intensity']:.6f}")

# Summary statistics for the board
print("\n\n=== SUMMARY STATISTICS ===")
print(f"Total observations: {len(df):,}")

print(f"\nAverage service score (all): {df['service_score'].mean():.2f}")
print(f"Average service score (treatment, before): {df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean():.2f}")

print(f"Average service score (control, after): {df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean():.2f}")
