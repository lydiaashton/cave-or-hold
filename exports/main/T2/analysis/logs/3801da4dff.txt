import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Count branches by treatment status
print("Branches by treatment status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Create treatment interaction term (DiD estimator)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary statistics by group and period
print("\n\nService Score Summary by Group and Period:")
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Calculate the raw DiD estimate
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
pre_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
post_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n\nRaw DiD Calculation:")
print(f"Control group - Pre: {pre_control:.3f}, Post: {post_control:.3f}, Change: {post_control - pre_control:.3f}")
print(f"Treatment group - Pre: {pre_treatment:.3f}, Post: {post_treatment:.3f}, Change: {post_treatment - pre_treatment:.3f}")
did_estimate = (post_treatment - pre_treatment) - (post_control - pre_control)
print(f"DiD Estimate (Treatment Effect): {did_estimate:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch and week fixed effects
# This is the gold standard for DiD analysis
print("=== DiD Regression with Branch and Week Fixed Effects ===")
model = ols('service_score ~ treatment + C(branch_id) + C(week)', data=df).fit()
print(model.summary().tables[1].as_string())

# Also run a simpler model with just the key variables
print("\n\n=== Simpler DiD Regression (no fixed effects) ===")
model_simple = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model_simple.summary().tables[1].as_string())

# Check if staffing and visits indices affect results
print("\n\n=== DiD Regression with Controls ===")
model_controls = ols('service_score ~ treatment + staffing_index + visits_index + C(branch_id) + C(week)', data=df).fit()
print(model_controls.summary().tables[1].as_string())

from statsmodels.formula.api import ols
import matplotlib

# Run DiD regression with branch and week fixed effects

model = ols('service_score ~ treatment + C(branch_id) + C(week)', data=df).fit()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators
# Week 14 is the last pre-switch week, so we'll use it as reference
df['relative_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create interaction terms for each week with treatment (use rel_ prefix)
for w in range(-8, 9):  # weeks -8 to 8 relative to week 14
    if w != 0:  # Skip reference week
        sign = 'm' if w < 0 else 'p'
        abs_w = abs(w)
        df[f'rel_{sign}{abs_w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
event_study_vars = []
for w in range(-8, 9):
    if w != 0:
        sign = 'm' if w < 0 else 'p'
        abs_w = abs(w)
        event_study_vars.append(f'rel_{sign}{abs_w}')

formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(branch_id) + C(week)'

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients
event_results = []
for w in range(-8, 9):
    if w == 0:
        event_results.append({'relative_week': w, 'coef': 0, 'se': 0, 'pval': np.nan, 'ci_low': 0, 'ci_high': 0})
    else:
        sign = 'm' if w < 0 else 'p'
        abs_w = abs(w)
        var_name = f'rel_{sign}{abs_w}'
        event_results.append({
            'relative_week': w,
            'coef': model_event.params[var_name],
            'se': model_event.bse[var_name],
            'pval': model_event.pvalues[var_name],
            'ci_low': model_event.conf_int().loc[var_name][0],
            'ci_high': model_event.conf_int().loc[var_name][1]
        })

event_df = pd.DataFrame(event_results)

print("Event Study Results (relative to week 14):")
print(event_df.round(3))

fig, ax = plt.subplots(figsize=(12, 6))
