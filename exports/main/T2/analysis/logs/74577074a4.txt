import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study - check effects by week relative to switch
df['weeks_relative'] = df['week'] - 15  # week 15 is period 0

# Create dummies for each week relative to switch (excluding week -1 as reference)
week_dummies = pd.get_dummies(df['weeks_relative'], prefix='week_rel')
df_event = df.join(week_dummies)

# Also create interaction with treated
for col in week_dummies.columns:
    df_event[f'{col}_treated'] = df_event[col] * df_event['treated']

# Run event study regression (excluding week -1 as reference)
control_vars = ['staffing_index', 'visits_index']
# Get all week_rel_treated columns except the reference
treated_interaction_cols = [c for c in df_event.columns if 'week_rel' in c and '_treated' in c and 'week_rel_-1' not in c]

# Build formula
formula = 'service_score ~ treated + ' + ' + '.join(treated_interaction_cols) + ' + ' + ' + '.join(control_vars)
event_model = ols(formula, data=df_event).fit()

# Extract coefficients for the interaction terms
coef_results = []
for col in treated_interaction_cols:
    week_num = int(col.replace('week_rel_', '').replace('_treated', ''))
    coef_results.append({
        'week_relative': week_num,
        'coef': event_model.params[col],
        'se': event_model.bse[col],
        'p_value': event_model.pvalues[col]

import pandas as pd

# By group
print("\nBy Group:")
for group_name, group_val in [('Control (no switch)', 0), ('Switched to new procedure', 1)]:
    group_data = df[df['switched_branch'] == group_val]
    before = group_data[group_data['after_switch']==0]['service_score'].mean()
    after = group_data[group_data['after_switch']==1]['service_score'].mean()
    print(f"  {group_name}:")

# DiD estimate from regression with controls and clustered SE
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster
from scipy import stats

df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treated'] * df['post']

model = ols('service_score ~ treated + post + did + staffing_index + visits_index', data=df).fit()
clustered_se = np.sqrt(np.diag(cov_cluster(model, df['branch_id'])))
did_coef = model.params['did']
did_se = clustered_se[model.params.index.get_loc('did')]
did_t = did_coef / did_se
did_p = 2 * (1 - stats.norm.cdf(np.abs(did_t)))
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se

print(f"\nDifference-in-Differences Estimate:")
print(f"  Effect of new procedure: {did_coef:.2f} points")
print(f"  95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
