import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Calculate average service scores by group and period
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("Service Score Summary by Group and Period:")
print(summary)

# Calculate the difference-in-differences manually
# Treatment group (switched_branch=1)
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

# Control group (switched_branch=0)
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n" + "="*60)
print("DIFFERENCE-IN-DIFFERENCES CALCULATION")
print("="*60)
print(f"\nTreatment Group (Switched Branches):")
print(f"  Before: {treatment_before:.2f}")
print(f"  After:  {treatment_after:.2f}")
print(f"  Change: {treatment_after - treatment_before:.2f}")

print(f"\nControl Group (Non-Switched Branches):")
print(f"  Before: {control_before:.2f}")
print(f"  After:  {control_after:.2f}")
print(f"  Change: {control_after - control_before:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD with cluster-robust standard errors (clustered by branch)
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("MODEL: DiD with Cluster-Robust Standard Errors (by branch)")
print("="*60)
print(model_clustered.summary())

# Also run with branch fixed effects and clustered SE
model_fe_clustered = ols('service_score ~ after_switch + treatment + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nMODEL: DiD with Branch FE + Cluster-Robust SE")
print("="*60)
# Extract just the key coefficients
print("\nKey Coefficients:")
print(f"  after_switch: {model_fe_clustered.params['after_switch']:.3f} (SE: {model_fe_clustered.bse['after_switch']:.3f}, p: {model_fe_clustered.pvalues['after_switch']:.3f})")
print(f"  treatment:    {model_fe_clustered.params['treatment']:.3f} (SE: {model_fe_clustered.bse['treatment']:.3f}, p: {model_fe_clustered.pvalues['treatment']:.3f})")

# Calculate 95% confidence interval for treatment effect
treatment_coef = model_fe_clustered.params['treatment']
treatment_se = model_fe_clustered.bse['treatment']
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se
print(f"\n95% Confidence Interval for Treatment Effect: [{ci_lower:.2f}, {ci_upper:.2f}]")

# Effect size relative to baseline
baseline = df[df['after_switch'] == 0]['service_score'].mean()
print(f"\nBaseline average service score: {baseline:.2f}")
print(f"Treatment effect as % of baseline: {(treatment_coef/baseline)*100:.1f}%")

import pandas as pd
import numpy as np

# Calculate before/after averages for each branch
before = df[df['after_switch'] == 0].groupby('branch_id')['service_score'].mean()
after = df[df['after_switch'] == 1].groupby('branch_id')['service_score'].mean()

branch_changes = pd.DataFrame({
    'before': before,
    'after': after,
    'change': after - before,
    'switched': df.groupby('branch_id')['switched_branch'].first()
})

print("Branch-Level Changes:")
print(branch_changes.head(10))

print("\n" + "="*60)
print("SUMMARY BY GROUP")
print("="*60)

# Treatment branches
treatment_changes = branch_changes[branch_changes['switched'] == 1]['change']
print(f"\nTreatment Branches (n={len(treatment_changes)}):")
print(f"  Mean change: {treatment_changes.mean():.2f}")
print(f"  Std dev: {treatment_changes.std():.2f}")
print(f"  Min: {treatment_changes.min():.2f}")
print(f"  Max: {treatment_changes.max():.2f}")
print(f"  % with positive change: {(treatment_changes > 0).mean()*100:.1f}%")

# Control branches
control_changes = branch_changes[branch_changes['switched'] == 0]['change']
print(f"\nControl Branches (n={len(control_changes)}):")
print(f"  Mean change: {control_changes.mean():.2f}")
print(f"  Std dev: {control_changes.std():.2f}")
print(f"  Min: {control_changes.min():.2f}")
print(f"  Max: {control_changes.max():.2f}")
print(f"  % with positive change: {(control_changes > 0).mean()*100:.1f}%")

# Statistical test on branch-level changes
