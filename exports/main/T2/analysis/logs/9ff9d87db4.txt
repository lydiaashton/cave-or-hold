import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD analysis
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run a difference-in-differences regression
# This controls for baseline differences between switched/non-switched branches
# and overall time trends (before/after period)
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Difference-in-Differences Regression (Basic):")
print(model.summary())

# Now let's add branch fixed effects to control for time-invariant branch characteristics
# This is a more robust approach
print("\n\n" + "="*60)
print("Fixed Effects Regression (controlling for branch and week):")
df['week_str'] = df['week'].astype(str)
model_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id) + C(week_str)', data=df).fit()
# Just show the key coefficients
print(model_fe.summary().tables[1].iloc[:4])

# Also run with staffing and visits as controls
print("\n\n" + "="*60)
print("DiD with staffing and visits controls:")
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print(model_controls.summary().tables[1].iloc[:5])

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Full model with all controls
print("Full Model with All Controls:")
model_full = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_full.summary().tables[1])

# Check effect heterogeneity - does the effect vary by branch characteristics?
# Let's see if branches with different baseline scores respond differently
print("\n\n" + "="*60)
print("Effect by Baseline Performance Quartile:")

# Calculate baseline (pre-switch) average for each branch
baseline = df[df['week'] < 15].groupby('branch_id')['service_score'].mean().reset_index()
baseline.columns = ['branch_id', 'baseline_score']
baseline['quartile'] = pd.qcut(baseline['baseline_score'], 4, labels=['Q1 (Low)', 'Q2', 'Q3', 'Q4 (High)'])

# Merge back to main data
df = df.merge(baseline[['branch_id', 'quartile']], on='branch_id')

# Calculate treatment effect by quartile
for q in ['Q1 (Low)', 'Q2', 'Q3', 'Q4 (High)']:
    subset = df[df['quartile'] == q]
    switched_post = subset[(subset['switched_branch'] == 1) & (subset['after_switch'] == 1)]['service_score'].mean()
    switched_pre = subset[(subset['switched_branch'] == 1) & (subset['after_switch'] == 0)]['service_score'].mean()
    control_post = subset[(subset['switched_branch'] == 0) & (subset['after_switch'] == 1)]['service_score'].mean()
    control_pre = subset[(subset['switched_branch'] == 0) & (subset['after_switch'] == 0)]['service_score'].mean()
    
    did_effect = (switched_post - switched_pre) - (control_post - control_pre)
    print(f"{q}: DiD Effect = {did_effect:.2f}")

# Summary stats for the board
print("\n\n" + "="*60)
print("SUMMARY STATISTICS FOR BOARD:")
print(f"Total branches: {df['branch_id'].nunique()}")
print(f"Branches that switched: {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"Branches that didn't switch: {df[df['switched_branch']==0]['branch_id'].nunique()}")
print(f"Weeks analyzed: {df['week'].min()} to {df['week'].max()}")
print(f"Switch occurred at week: 15")
print(f"\nTreatment effect (DiD estimate): +{model_full.params['treatment']:.2f} points")
print(f"95% CI: [{model_full.conf_int().loc['treatment'][0]:.2f}, {model_full.conf_int().loc['treatment'][1]:.2f}]")
print(f"P-value: {model_full.pvalues['treatment']:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Final model
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get key numbers
treatment_effect = model.params['treatment']
treatment_ci_low = model.conf_int().loc['treatment'][0]
treatment_ci_high = model.conf_int().loc['treatment'][1]
treatment_pvalue = model.pvalues['treatment']

# Pre/post means
switched_pre = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
switched_post = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
control_pre = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_post = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print("=== FINAL NUMBERS FOR REPORT ===")
print(f"Treatment effect: +{treatment_effect:.2f} points")
print(f"95% Confidence Interval: [{treatment_ci_low:.2f}, {treatment_ci_high:.2f}]")
print(f"P-value: {treatment_pvalue:.6f}")
print(f"\nSwitched branches: {switched_pre:.2f} (before) → {switched_post:.2f} (after) = +{switched_post - switched_pre:.2f}")
print(f"Control branches: {control_pre:.2f} (before) → {control_post:.2f} (after) = {control_post - control_pre:+.2f}")
print(f"\nRelative improvement: {(treatment_effect / control_pre) * 100:.1f}%")
