import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# Count branches by treatment status
print("Branches by treatment status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate average service scores by group and period
print("\n\nAverage service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
pivot.columns = ['Before (weeks 6-14)', 'After (weeks 15-22)']
pivot.index = ['Control (old procedure)', 'Treatment (new procedure)']
print(pivot)

# Calculate the differences
print("\n\nDifference-in-Differences Calculation:")
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

control_change = control_after - control_before
treatment_change = treatment_after - treatment_before
did_estimate = treatment_change - control_change

print(f"Control group: {control_before:.2f} -> {control_after:.2f} (change: {control_change:.2f})")
print(f"Treatment group: {treatment_before:.2f} -> {treatment_after:.2f} (change: {treatment_change:.2f})")
print(f"\nDiD Estimate (treatment effect): {did_estimate:.2f}")

# Run a regression-based DiD with standard errors
print("\n\nRegression-based DiD analysis:")
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Simple OLS regression
import statsmodels.api as sm
X = df[['treatment', 'post', 'did']]
X = sm.add_constant(X)
y = df['service_score']

model = sm.OLS(y, X).fit()
print(model.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.stats.multicomp import pairwise_tukeyhsd
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Cluster-robust standard errors at branch level
print("DiD with cluster-robust standard errors (clustered by branch):")
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

X = df[['treatment', 'post', 'did']]
X = sm.add_constant(X)
y = df['service_score']

model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model.summary())

# Check parallel trends - plot average service scores by week for each group
print("\n\nWeekly averages by group:")
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
print(weekly_avg.pivot(index='week', columns='switched_branch', values='service_score'))

# Create visualization
fig, ax = plt.subplots(figsize=(10, 6))

for group in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == group]
    label = 'Treatment (new procedure)' if group == 1 else 'Control (old procedure)'
    ax.plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2)

ax.axvline(x=14.5, color='red', linestyle='--', label='Switch point (week 15)')
ax.set_xlabel('Week')
ax.set_ylabel('Average Service Score')
ax.set_title('Service Scores Over Time by Group')
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("\nPlot saved as service_scores_trend.png")

# Pre-trend test: compare trends before the switch (weeks 6-14)
print("\n\nPre-trend analysis (weeks 6-14):")
pre_data = df[df['week'] < 15].copy()
pre_data['treatment'] = pre_data['switched_branch']
pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()
pre_data['interaction'] = pre_data['treatment'] * pre_data['week_centered']

X_pre = pre_data[['treatment', 'week_centered', 'interaction']]
X_pre = sm.add_constant(X_pre)
y_pre = pre_data['service_score']

model_pre = sm.OLS(y_pre, X_pre).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print(model_pre.summary())
print(f"\nPre-trend interaction coefficient: {model_pre.params['interaction']:.3f}, p-value: {model_pre.pvalues['interaction']:.3f}")

import pandas as pd
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# Add covariates to the model
print("DiD with covariates (staffing_index, visits_index):")
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

X = df[['treatment', 'post', 'did', 'staffing_index', 'visits_index']]
X = sm.add_constant(X)
y = df['service_score']

model_cov = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_cov.summary())

# Effect size relative to baseline
print("\n\nEffect size interpretation:")
baseline = df[df['after_switch'] == 0]['service_score'].mean()
effect = model_cov.params['did']
print(f"Baseline average service score: {baseline:.2f}")
print(f"Estimated treatment effect: {effect:.2f} points")
print(f"Effect as % of baseline: {effect/baseline*100:.1f}%")

# Check if effect is consistent across post-period weeks
print("\n\nPost-period weekly effects (treatment vs control difference):")
post_data = df[df['after_switch'] == 1]
post_diff = post_data.groupby('week').apply(
    lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()
)

import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import pandas as pd

df = pd.read_csv('branches.csv')

# Create a comprehensive figure
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Left panel: Trends over time
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

for group in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == group]
    label = 'New Procedure (20 branches)' if group == 1 else 'Old Procedure (14 branches)'
    axes[0].plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2, markersize=6)

axes[0].axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch implemented')
axes[0].axvspan(15, 22, alpha=0.2, color='green', label='Post-switch period')
axes[0].set_xlabel('Week', fontsize=12)
axes[0].set_ylabel('Average Service Score', fontsize=12)
axes[0].set_title('Service Score Trends by Procedure Type', fontsize=14, fontweight='bold')
axes[0].legend(loc='upper left')
axes[0].grid(True, alpha=0.3)
axes[0].set_xticks(range(6, 23))

# Right panel: Before/After comparison
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

x = ['Old Procedure', 'New Procedure']
before_vals = [before_control, before_treatment]
after_vals = [after_control, after_treatment]

x_pos = range(len(x))
width = 0.35

bars1 = axes[1].bar([p - width/2 for p in x_pos], before_vals, width, label='Before (Weeks 6-14)', color='steelblue')
bars2 = axes[1].bar([p + width/2 for p in x_pos], after_vals, width, label='After (Weeks 15-22)', color='coral')

axes[1].set_ylabel('Average Service Score', fontsize=12)
axes[1].set_title('Before/After Comparison by Procedure', fontsize=14, fontweight='bold')
axes[1].set_xticks(x_pos)
axes[1].set_xticklabels(x)
axes[1].legend()
axes[1].grid(True, alpha=0.3, axis='y')

# Add value labels on bars
for bar in bars1:
    height = bar.get_height()
    axes[1].annotate(f'{height:.1f}',
                    xy=(bar.get_x() + bar.get_width() / 2, height),
                    xytext=(0, 3), textcoords="offset points",
