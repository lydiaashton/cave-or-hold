import pandas as pd
import numpy as np

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Basic DiD table: average service scores by group and period
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES SUMMARY TABLE")
print("=" * 60)

# Calculate means for each group
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nControl Group (did not switch):")
print(f"  Before (weeks 6-14): {before_control:.2f}")

print(f"  Change: {after_control - before_control:.2f}")

print(f"  Before (weeks 6-14): {before_treatment:.2f}")
print(f"  After (weeks 15-22): {after_treatment:.2f}")
print(f"  Change: {after_treatment - before_treatment:.2f}")

# DiD estimate
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")

model = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()

# Create week dummies interacted with treatment
# Use week 14 as the reference period (last week before switch)
df['week_14'] = (df['week'] == 14).astype(int)

# Create interaction terms for each week with treatment
for week in range(6, 23):
    if week != 14:  # reference week
        df[f'week_{week}'] = (df['week'] == week).astype(int)
        df[f'week_{week}_treated'] = df[f'week_{week}'] * df['switched_branch']

# Build the formula for event study
control_vars = '+ staffing_index + visits_index'
week_terms = ' + '.join([f'week_{w}_treated' for w in range(6, 23) if w != 14])
week_fixed = ' + '.join([f'week_{w}' for w in range(6, 23) if w != 14])

formula = f'service_score ~ switched_branch + {week_fixed} + {week_terms} + staffing_index + visits_index'

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract the event study coefficients
print("EVENT STUDY RESULTS")
print("=" * 70)
print("Coefficients for treatment effects by week (relative to week 14)")
print("(Week 14 is the reference period, right before the switch)")

event_coefs = []
for week in range(6, 23):
    if week == 14:
        event_coefs.append((week, 0.0, 0.0, 0.0, 1.0))  # reference
    else:
        coef_name = f'week_{week}_treated'
        if coef_name in model_event.params.index:
            coef = model_event.params[coef_name]
            se = model_event.bse[coef_name]
            pval = model_event.pvalues[coef_name]
            event_coefs.append((week, coef, se, pval, 1 if pval < 0.05 else 0))
            marker = "*" if pval < 0.05 else " "
