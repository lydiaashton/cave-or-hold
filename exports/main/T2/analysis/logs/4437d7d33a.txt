import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Average service score by group and period
print("Average service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the differences
print("\n\nDifference-in-Differences calculation:")

# Treatment group (switched_branch=1)
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_diff = treatment_after - treatment_before

# Control group (switched_branch=0)
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
control_diff = control_after - control_before

# DiD estimate
did_estimate = treatment_diff - control_diff

print(f"Treatment group - Before: {treatment_before:.2f}, After: {treatment_after:.2f}, Change: {treatment_diff:.2f}")
print(f"Control group - Before: {control_before:.2f}, After: {control_after:.2f}, Change: {control_diff:.2f}")
print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators with simple names
df = df.copy()
df['relative_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create indicators for each relative week (excluding reference week 0)
event_vars = []
for rw in range(-8, 9):
    if rw != 0:
        # Use 'p' for positive and 'm' for minus/negative
        if rw < 0:
            var_name = f'rel_m{abs(rw)}'  # e.g., rel_m8 for week -8
        else:
            var_name = f'rel_p{rw}'  # e.g., rel_p1 for week +1
        df[var_name] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)
        event_vars.append(var_name)

# Run event study regression
event_formula = f'service_score ~ {" + ".join(event_vars)} + staffing_index + visits_index'
event_model = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Results (coefficients for treatment group by relative week):")
print("=" * 70)
print(f"{'Relative Week':<15} {'Coefficient':<15} {'Std Error':<15} {'P-value':<15}")
print("-" * 70)

results = []
for rw in range(-8, 9):
    if rw == 0:
        results.append((rw, np.nan, np.nan, np.nan, '(reference)'))
    else:
        if rw < 0:
            var_name = f'rel_m{abs(rw)}'
        else:
            var_name = f'rel_p{rw}'
        coef = event_model.params[var_name]
        se = event_model.bse[var_name]
        pval = event_model.pvalues[var_name]
        results.append((rw, coef, se, pval, ''))

for rw, coef, se, pval, note in results:
    if rw == 0:
        print(f"{rw:<15} {note:<15} {'-':<15} {'-':<15}")
    else:
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"{rw:<15} {coef:<15.3f} {se:<15.3f} {pval:<15.4f} {sig}")

print("\nNote: Coefficients show the difference in service scores for treatment vs control")
print("      at each relative week, compared to week 14 (reference).")
print("      Positive values indicate treatment group scoring higher than control.")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create visualization data
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
weekly_avg['group'] = weekly_avg['switched_branch'].map({0: 'Control (Old Procedure)', 1: 'Treatment (New Procedure)'})

# Create the plot
fig, ax = plt.subplots(figsize=(12, 7))

# Plot both groups
for group_name, group_data in weekly_avg.groupby('group'):
    marker = 'o' if 'New' in group_name else 's'
    color = '#2E86AB' if 'New' in group_name else '#A23B72'
    ax.plot(group_data['week'], group_data['service_score'], 
            marker=marker, markersize=8, linewidth=2.5, 
            label=group_name, color=color)

# Add vertical line at switch point
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch implemented')

# Labels and title
ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Library Branch Service Scores: Before and After Procedure Change', fontsize=14, fontweight='bold')
ax.legend(loc='upper left', fontsize=11)
ax.set_xticks(range(6, 23))
ax.grid(True, alpha=0.3)
ax.set_ylim(45, 60)

plt.savefig('service_scores_trend.png', dpi=150, bbox_inches='tight')

print("KEY STATISTICS FOR BOARD SUMMARY")
