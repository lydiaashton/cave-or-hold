import pandas as pd
import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch fixed effects
# This controls for time-invariant branch characteristics
print("=== DiD Regression with Branch Fixed Effects ===")
model = smf.ols('service_score ~ treatment + after_switch + C(branch_id)', data=df)
results = model.fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(results.summary().tables[1])

# Also run with week fixed effects to control for time trends
print("\n=== DiD Regression with Branch and Week Fixed Effects ===")
model2 = smf.ols('service_score ~ treatment + C(branch_id) + C(week)', data=df)
results2 = model2.fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
