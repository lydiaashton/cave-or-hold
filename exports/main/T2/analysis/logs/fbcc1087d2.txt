import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Count branches by switch status
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Check balance - how many observations per group/time period
print("\nObservations by group and period:")
df['period'] = df['after_switch'].map({0: 'Before', 1: 'After'})
df['group'] = df['switched_branch'].map({0: 'Control (No Switch)', 1: 'Treatment (Switched)'})
print(df.groupby(['group', 'period']).size())

# Average service scores by group and period
print("\nAverage service scores by group and period:")
print(df.groupby(['group', 'period'])['service_score'].agg(['mean', 'std', 'count']))

# Calculate the difference-in-differences manually
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n--- Difference-in-Differences Calculation ---")
print(f"Control group (no switch): Before={before_control:.2f}, After={after_control:.2f}, Change={after_control - before_control:.2f}")
print(f"Treatment group (switched): Before={before_treatment:.2f}, After={after_treatment:.2f}, Change={after_treatment - before_treatment:.2f}")
did_effect = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {did_effect:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run a difference-in-differences regression
# Basic DiD model
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# Now let's add branch fixed effects and week fixed effects for a more robust analysis
# Create dummy variables for branch and week
df_with_fe = df.copy()

# More robust: branch and week fixed effects
model_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id) + C(week)', data=df_with_fe).fit()
print("\n=== DiD with Branch and Week Fixed Effects ===")
print(model_fe.summary())

# Cluster-robust standard errors at branch level
model_fe_clustered = model_fe.get_robustcov_results(cov_type='cluster', groups=df_with_fe['branch_id'])
print("\n=== DiD with Branch and Week FE (Clustered SE at Branch Level) ===")
print(model_fe_clustered.summary())

week_cols = [f'week_{week}_demeaned' for week in df['week'].unique()]
fe_model = ols('service_score_demeaned ~ switched_branch_demeaned + after_switch_demeaned + treatment_demeaned + ' + ' + '.join(week_cols), data=df).fit()
