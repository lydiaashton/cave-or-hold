import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create week dummies relative to switch
# Week 14 is the last pre-switch week, use as reference
df['relative_week'] = df['week'] - 14

# Create indicators for each week relative to switch (excluding week 0 as reference)
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != 0:
        df[f'week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Event study regression
event_vars = [f'week_{w}' for w in range(-8, 9) if w != 0]
formula = 'service_score ~ switched_branch + ' + ' + '.join(event_vars) + ' + staffing_index + visits_index'

event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results (coefficients for treatment group by week relative to switch) ===")
print("Reference: Week 14 (last pre-switch week)")
print("\nPre-switch weeks (should be near 0 if parallel trends hold):")
for w in range(-8, 0):
    var = f'week_{w}'
    coef = event_model.params[var]
    pval = event_model.pvalues[var]
    ci_low, ci_high = event_model.conf_int().loc[var]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create week dummies relative to switch
# Week 14 is the last pre-switch week, use as reference
df['relative_week'] = df['week'] - 14

# Create indicators for each week relative to switch (excluding week 0 as reference)
# Use prefixed names to avoid patsy issues
for w in range(-8, 9):
    if w != 0:
        prefix = 'pre' if w < 0 else 'post'
        df[f'{prefix}_w{abs(w)}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Event study regression
pre_vars = [f'pre_w{i}' for i in range(8, 0, -1)]  # pre_w8 to pre_w1
post_vars = [f'post_w{i}' for i in range(1, 9)]   # post_w1 to post_w8
event_vars = pre_vars + post_vars

formula = 'service_score ~ switched_branch + ' + ' + '.join(event_vars) + ' + staffing_index + visits_index'

event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results ===")
print("Reference: Week 14 (last pre-switch week)")
print("\nPre-switch weeks (coefficients for treatment group vs control):")
for i, var in enumerate(pre_vars):
    w = 14 - (8 - i)  # week number
    coef = event_model.params[var]
    pval = event_model.pvalues[var]
    ci_low, ci_high = event_model.conf_int().loc[var]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Week {w} ({var}): {coef:6.2f} (p={pval:.3f}) [{ci_low:6.2f}, {ci_high:6.2f}] {sig}")

print("\nPost-switch weeks (treatment effect):")
for i, var in enumerate(post_vars):
    w = 14 + (i + 1)  # week number
    coef = event_model.params[var]
    pval = event_model.pvalues[var]
    ci_low, ci_high = event_model.conf_int().loc[var]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Week {w} ({var}): {coef:6.2f} (p={pval:.3f}) [{ci_low:6.2f}, {ci_high:6.2f}] {sig}")

weekly = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
