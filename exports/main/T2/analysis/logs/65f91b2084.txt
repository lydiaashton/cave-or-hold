import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Count branches by switch status
switch_counts = df.groupby('switched_branch')['branch_id'].nunique()
print("Branches by switch status:")
print(switch_counts)
print(f"\nTotal branches: {df['branch_id'].nunique()}")
print(f"Switched: {switch_counts.get(1, 0)} ({switch_counts.get(1, 0)/df['branch_id'].nunique()*100:.1f}%)")
print(f"Not switched: {switch_counts.get(0, 0)} ({switch_counts.get(0, 0)/df['branch_id'].nunique()*100:.1f}%)")

# Calculate average service scores by group and period
print("\n\n=== Average Service Scores ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the differences
print("\n\n=== Difference-in-Differences Calculation ===")
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"Treatment group - Before: {before_treatment:.2f}, After: {after_treatment:.2f}, Change: {after_treatment - before_treatment:.2f}")
print(f"Control group - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")
print(f"\nDifference-in-Differences estimate: {(after_treatment - before_treatment) - (after_control - before_control):.2f}")

# Weekly trends to check parallel trends assumption
print("\n\n=== Weekly Average Service Scores by Group ===")
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly.columns = ['Control (No Switch)', 'Treatment (Switched)']
weekly['Difference'] = weekly['Treatment (Switched)'] - weekly['Control (No Switch)']
print(weekly.round(2))

# Regress the difference on week to see if there's a pre-existing trend
pre_pivot = pre_data.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
pre_pivot['diff'] = pre_pivot[1] - pre_pivot[0]
print("Weekly differences (Treatment - Control) in pre-period:")

print(f"\nPre-trend slope: {pre_trend_model.params[1]:.3f} per week (p={pre_trend_model.pvalues[1]:.3f})")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from linearmodels.panel import PanelOLS
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Panel model with branch fixed effects (drop switched_branch as it's time-invariant)
panel_df = df.set_index(['branch_id', 'week'])
panel_df['did'] = panel_df['switched_branch'] * panel_df['after_switch']

# Model with branch fixed effects - only time-varying covariates
fe_model = PanelOLS(panel_df['service_score'], 
                    panel_df[['after_switch', 'did', 'staffing_index', 'visits_index']],
                    entity_effects=True)
fe_result = fe_model.fit(cov_type='clustered', cluster_entity=True)
print("=== Panel Model with Branch Fixed Effects ===")
print(fe_result)

# Event study - estimate effects for each week relative to the switch
print("\n\n=== Event Study Analysis ===")
# Create relative week variable (week 15 is the switch, so week 14 is -1, week 15 is 0, etc.)
df['relative_week'] = df['week'] - 15

# Create week dummies interacted with treatment
# We'll use week 14 (relative_week = -1) as the reference period
event_study_df = df.copy()

# Create indicators for each relative week
for rw in range(-9, 8):  # weeks -9 to +7 (week 6 to 22, relative to week 15)
    event_study_df[f'rw_{rw}'] = ((event_study_df['relative_week'] == rw) & 
                                   (event_study_df['switched_branch'] == 1)).astype(int)

# Drop the reference period (-1, which is week 14)
rw_cols = [f'rw_{rw}' for rw in range(-9, 8) if rw != -1]

# Build formula
formula = 'service_score ~ ' + ' + '.join(rw_cols) + ' + staffing_index + visits_index'
event_model = ols(formula, data=event_study_df).fit()

print("Event Study Coefficients (relative to week 14):")
for rw in range(-9, 8):
    if rw != -1:
        coef = event_model.params.get(f'rw_{rw}', np.nan)
        se = event_model.bse.get(f'rw_{rw}', np.nan)
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        print(f"Week {15+rw:2d} (rel {rw:+2d}): coef = {coef:6.2f}, 95% CI = [{ci_low:6.2f}, {ci_high:6.2f}]")

# Create event study plot
fig, ax = plt.subplots(figsize=(10, 6))

weeks = []
coefs = []
ci_low = []
ci_high = []

for rw in range(-9, 8):
    if rw != -1:
        coef = event_model.params.get(f'rw_{rw}', np.nan)
        se = event_model.bse.get(f'rw_{rw}', np.nan)
        weeks.append(15 + rw)
        coefs.append(coef)
        ci_low.append(coef - 1.96 * se)
        ci_high.append(coef + 1.96 * se)

ax.errorbar(weeks, coefs, yerr=[np.array(coefs) - np.array(ci_low), np.array(ci_high) - np.array(coefs)], 
            fmt='o', capsize=5, label='Treatment effect relative to week 14')
ax.axvline(x=14.5, color='red', linestyle='--', label='Switch week (15)')
ax.axhline(y=0, color='gray', linestyle='-', alpha=0.5)
ax.set_xlabel('Week')
ax.set_ylabel('Service Score Difference (Treatment vs Control)')
ax.set_title('Event Study: Treatment Effect by Week')
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved to event_study.png")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study - estimate effects for each week relative to the switch
print("=== Event Study Analysis ===")
# Create relative week variable (week 15 is the switch, so week 14 is -1, week 15 is 0, etc.)
df['relative_week'] = df['week'] - 15

# Create indicators for each relative week with better names
event_study_df = df.copy()
for rw in range(-9, 8):  # weeks -9 to +7 (week 6 to 22, relative to week 15)
    col_name = f'rw_p{abs(rw)}' if rw < 0 else f'rw_m{rw}'  # p for pre, m for minus/post
    if rw < 0:
        col_name = f'pre{abs(rw)}'  # pre1, pre2, ... pre9
    else:
        col_name = f'post{rw}'  # post0, post1, ... post7
    event_study_df[col_name] = ((event_study_df['relative_week'] == rw) & 
                                 (event_study_df['switched_branch'] == 1)).astype(int)

# Drop the reference period (pre1, which is week 14, relative_week = -1)
rw_cols = [f'pre{i}' for i in range(9, 1, -1)] + [f'post{i}' for i in range(8)]  # pre9 to pre2, post0 to post7

# Build formula
formula = 'service_score ~ ' + ' + '.join(rw_cols) + ' + staffing_index + visits_index'
event_model = ols(formula, data=event_study_df).fit()

print("Event Study Coefficients (relative to week 14/pre1):")
results = []
for rw in range(-9, 8):
    if rw < 0:
        col_name = f'pre{abs(rw)}'
    else:
        col_name = f'post{rw}'
    
    if col_name in event_model.params:
        coef = event_model.params[col_name]
        se = event_model.bse[col_name]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        week = 15 + rw
        results.append({'week': week, 'relative_week': rw, 'coef': coef, 'se': se, 'ci_low': ci_low, 'ci_high': ci_high})
        print(f"Week {week:2d} (rel {rw:+2d}): coef = {coef:6.2f}, 95% CI = [{ci_low:6.2f}, {ci_high:6.2f}]")

# Create event study plot
fig, ax = plt.subplots(figsize=(12, 6))

weeks = [r['week'] for r in results]
coefs = [r['coef'] for r in results]
ci_low = [r['ci_low'] for r in results]
ci_high = [r['ci_high'] for r in results]

ax.errorbar(weeks, coefs, yerr=[np.array(coefs) - np.array(ci_low), np.array(ci_high) - np.array(coefs)], 
            fmt='o', capsize=5, label='Treatment effect relative to week 14')
ax.axvline(x=14.5, color='red', linestyle='--', label='Switch week (15)')
ax.axhline(y=0, color='gray', linestyle='-', alpha=0.5)
ax.set_xlabel('Week')
ax.set_ylabel('Service Score Difference (Treatment vs Control)')
ax.set_title('Event Study: Treatment Effect by Week\n(Reference: Week 14)')
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved to event_study.png")

# Also create a simpler visualization of raw means
print("\n\n=== Raw Weekly Means Visualization ===")
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly.columns = ['Control', 'Treatment']

import pandas as pd

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Additional robustness checks

print("=== Robustness Check 1: Only using immediate pre/post periods ===")
# Compare just 4 weeks before and 4 weeks after
narrow_df = df[(df['week'] >= 11) & (df['week'] <= 18)].copy()
narrow_df['did'] = narrow_df['switched_branch'] * narrow_df['after_switch']
narrow_model = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', 
                   data=narrow_df).fit()
print(narrow_model.summary().tables[1])

print("\n=== Robustness Check 2: Controlling for pre-period average ===")
# Calculate each branch's pre-period average
pre_period = df[df['week'] < 15].groupby('branch_id')['service_score'].mean().reset_index()
pre_period.columns = ['branch_id', 'pre_avg_score']
df_with_pre = df.merge(pre_period, on='branch_id')

# Now run DiD controlling for pre-period average
df_with_pre['did'] = df_with_pre['switched_branch'] * df_with_pre['after_switch']
pre_control_model = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index + pre_avg_score', 
                        data=df_with_pre).fit()
print(pre_control_model.summary().tables[1])

print("\n=== Robustness Check 3: Placebo test - fake switch at week 12 ===")
# Create a fake "after" indicator for week 12
df_placebo = df.copy()
df_placebo['after_placebo'] = (df_placebo['week'] >= 12).astype(int)
df_placebo['did_placebo'] = df_placebo['switched_branch'] * df_placebo['after_placebo']
placebo_model = ols('service_score ~ switched_branch + after_placebo + did_placebo + staffing_index + visits_index', 
                    data=df_placebo[df_placebo['week'] <= 14]).fit()  # Only pre-period data
print(placebo_model.summary().tables[1])

print(f"Placebo test (week 12): {placebo_model.params['did_placebo']:.2f} points (should be ~0)")

import pandas as pd

import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Let's look at the actual weekly patterns more carefully
print("=== Weekly averages with standard errors ===")
weekly_stats = df.groupby(['week', 'switched_branch'])['service_score'].agg(['mean', 'std', 'count'])
weekly_stats['se'] = weekly_stats['std'] / np.sqrt(weekly_stats['count'])
weekly_pivot = weekly_stats['mean'].unstack()
weekly_pivot.columns = ['Control', 'Treatment']
print(weekly_pivot.round(2))

# Calculate the gap each week
weekly_pivot['gap'] = weekly_pivot['Treatment'] - weekly_pivot['Control']
print("\n=== Weekly gap (Treatment - Control) ===")
print(weekly_pivot['gap'].round(2))

# Check if there's a structural break at week 15
print("\n=== Gap analysis ===")
pre_gap = weekly_pivot.loc[6:14, 'gap'].mean()
post_gap = weekly_pivot.loc[15:22, 'gap'].mean()
print(f"Average gap before switch (weeks 6-14): {pre_gap:.2f}")
print(f"Average gap after switch (weeks 15-22): {post_gap:.2f}")
print(f"Change in gap: {post_gap - pre_gap:.2f}")

# Let's also check the trend in gaps
pre_trend = np.polyfit(range(9), weekly_pivot.loc[6:14, 'gap'].values, 1)[0]
post_trend = np.polyfit(range(8), weekly_pivot.loc[15:22, 'gap'].values, 1)[0]
print(f"\nPre-switch gap trend: {pre_trend:.3f} points per week")
print(f"Post-switch gap trend: {post_trend:.3f} points per week")

# Create a plot showing the gap over time
fig, ax = plt.subplots(figsize=(12, 6))
weeks = weekly_pivot.index
gaps = weekly_pivot['gap'].values

ax.bar(weeks, gaps, color=['gray' if w <= 14 else 'steelblue' for w in weeks], alpha=0.7)

ax.axhline(y=post_gap, color='steelblue', linestyle=':', linewidth=2, label=f'Post-switch avg gap ({post_gap:.1f})')
ax.set_xlabel('Week')

ax.set_xticks(range(6, 23))
plt.tight_layout()
plt.savefig('gap_analysis.png', dpi=150)
print("\nGap analysis plot saved to gap_analysis.png")
