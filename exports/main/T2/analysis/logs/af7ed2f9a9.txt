import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Count branches by switch status
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate pre and post averages by group
print("\n\nAverage service scores by group and period:")
pre_data = df[df['after_switch'] == 0]
post_data = df[df['after_switch'] == 1]

print("\nBefore switch (weeks 6-14):")
print(pre_data.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

print("\nAfter switch (weeks 15-22):")
print(post_data.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

# Calculate the difference-in-differences manually
control_pre = pre_data[pre_data['switched_branch'] == 0]['service_score'].mean()
control_post = post_data[post_data['switched_branch'] == 0]['service_score'].mean()
treated_pre = pre_data[pre_data['switched_branch'] == 1]['service_score'].mean()
treated_post = post_data[post_data['switched_branch'] == 1]['service_score'].mean()

print("\n\nDifference-in-Differences Calculation:")
print(f"Control group (no switch): {control_pre:.2f} -> {control_post:.2f}, change = {control_post - control_pre:.2f}")
print(f"Treated group (switch): {treated_pre:.2f} -> {treated_post:.2f}, change = {treated_post - treated_pre:.2f}")
did_estimate = (treated_post - treated_pre) - (control_post - control_pre)
print(f"\nDiD Estimate (treatment effect): {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# The key coefficient is the interaction between switched_branch and after_switch
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("DiD Regression Results:")
print(model.summary())

# Also run with branch fixed effects for more robust estimation
# This controls for time-invariant branch characteristics
print("\n\n" + "="*60)
print("DiD Regression with Branch Fixed Effects:")
model_fe = ols('service_score ~ switched_branch * after_switch + C(branch_id)', data=df).fit()
print(model_fe.summary())

# Let's also check the parallel trends assumption by looking at weekly trends
print("\n\n" + "="*60)
print("Weekly average service scores by group:")
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(weekly_avg)
