pre_level_model = ols('service_score ~ switched_branch', data=pre_switch).fit(cov_type='cluster', cov_kwds={'groups': pre_switch['branch_id']})
print(f"\nPre-switch level difference: {pre_level_model.params['switched_branch']:.3f} (p={pre_level_model.pvalues['switched_branch']:.3f})")

# Event study - check week-by-week effects
df['week_14'] = (df['week'] == 14).astype(int)  # Reference week
for w in range(6, 23):
    if w != 14:
        df[f'week_{w}'] = (df['week'] == w).astype(int)
        df[f'week_{w}_treated'] = df[f'week_{w}'] * df['switched_branch']

# Run event study regression (using week 14 as baseline)
event_study_vars = '+'.join([f'week_{w}_treated' for w in range(6, 23) if w != 14])
event_study_formula = f'service_score ~ switched_branch + after_switch + {event_study_vars}'

# This might be too many variables, let's do a simpler version
# Just check the interaction for each week
print("\n\nWeek-by-week treatment effects (relative to week 14 baseline):")
results = []
for w in range(6, 23):
    if w < 15:  # Pre-period
        subset = df[df['week'].isin([14, w])]
        subset['is_week_w'] = (subset['week'] == w).astype(int)
        subset['interaction'] = subset['is_week_w'] * subset['switched_branch']
        model = ols('service_score ~ switched_branch + is_week_w + interaction', data=subset).fit(cov_type='cluster', cov_kwds={'groups': subset['branch_id']})
        results.append({'week': w, 'effect': model.params['interaction'], 'pvalue': model.pvalues['interaction'], 'period': 'pre'})
    else:  # Post-period
        subset = df[df['week'].isin([14, w])]

        subset['interaction'] = subset['is_week_w'] * subset['switched_branch']
        model = ols('service_score ~ switched_branch + is_week_w + interaction', data=subset).fit(cov_type='cluster', cov_kwds={'groups': subset['branch_id']})
        results.append({'week': w, 'effect': model.params['interaction'], 'pvalue': model.pvalues['interaction'], 'period': 'post'})

results_df = pd.DataFrame(results)
