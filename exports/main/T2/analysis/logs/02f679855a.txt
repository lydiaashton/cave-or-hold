import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# Regression with DiD specification
# Basic model
model1 = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
print("=== Model 1: Basic DiD ===")
print(model1.summary())

# Model with branch fixed effects
df['branch_id_str'] = df['branch_id'].astype(str)
model2 = ols('service_score ~ after_switch + treated + C(branch_id_str)', data=df).fit()
print("\n=== Model 2: With Branch Fixed Effects ===")
print(model2.summary())

# Model with branch FE and controls
model3 = ols('service_score ~ after_switch + treated + C(branch_id_str) + staffing_index + visits_index', data=df).fit()
print("\n=== Model 3: With Branch FE and Controls ===")
print(model3.summary())

# Clustered standard errors at branch level
from statsmodels.stats.sandwich_covariance import cov_cluster
model4 = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
cov_clustered = cov_cluster(model4, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))
print("\n=== Model 4: Clustered SE at Branch Level ===")
print("Coefficients:")
print(model4.params)
print("\nClustered Standard Errors:")
print(se_clustered)
print("\nClustered p-values:")
from scipy import stats
t_stats = model4.params / se_clustered
p_values = 2 * (1 - stats.t.cdf(np.abs(t_stats), df=len(df) - len(model4.params)))
print(p_values)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster
from scipy import stats

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# Get clustered standard errors
model = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
cov_clustered = cov_cluster(model, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))

# Get the index for 'treated'
treated_idx = list(model.params.index).index('treated')
print("=== DiD Results with Clustered SE ===")
print(f"Treated coefficient: {model.params['treated']:.4f}")
print(f"Clustered SE: {se_clustered[treated_idx]:.4f}")
t_stat = model.params['treated'] / se_clustered[treated_idx]
print(f"t-statistic: {t_stat:.4f}")
p_val = 2 * (1 - stats.t.cdf(np.abs(t_stat), df=len(df) - len(model.params)))
print(f"p-value: {p_val:.4f}")
print(f"95% CI: [{model.params['treated'] - 1.96*se_clustered[treated_idx]:.4f}, {model.params['treated'] + 1.96*se_clustered[treated_idx]:.4f}]")

# Check parallel trends - plot average service scores by week for each group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print("\n=== Weekly Averages by Group ===")
print(weekly_avg)

# Calculate pre-trends
