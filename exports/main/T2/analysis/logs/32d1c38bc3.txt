import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Add branch fixed effects
print("=== DiD with Branch Fixed Effects ===")
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index + C(branch_id_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
# The switched_branch coefficient will be absorbed by branch FE, so we look at treatment
print(f"Treatment effect: {model_fe.params['treatment']:.3f}")
print(f"Std Error: {model_fe.bse['treatment']:.3f}")
print(f"P-value: {model_fe.pvalues['treatment']:.4f}")
print(f"95% CI: [{model_fe.conf_int().loc['treatment'][0]:.3f}, {model_fe.conf_int().loc['treatment'][1]:.3f}]")

# Event study / dynamic effects
print("\n=== Dynamic Effects (Week-by-Week Relative to Switch) ===")
df['relative_week'] = df['week'] - 14  # weeks relative to switch (week 15 is +1)
# Create dummies for each week interacted with treatment
for w in range(6, 23):
    df[f'week_{w}'] = (df['week'] == w).astype(int)
    df[f'treat_week_{w}'] = (df['switched_branch'] == 1).astype(int) * (df['week'] == w).astype(int)

# Reference week 14 (last pre-period week)
pre_weeks = [6, 7, 8, 9, 10, 11, 12, 13]
post_weeks = [15, 16, 17, 18, 19, 20, 21, 22]

# Dynamic DiD
formula = 'service_score ~ C(branch_id_str) + ' + ' + '.join([f'treat_week_{w}' for w in post_weeks]) + ' + staffing_index + visits_index'
model_dynamic = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Post-switch week effects (relative to week 14 baseline):")
for w in post_weeks:
    coef = model_dynamic.params[f'treat_week_{w}']
    se = model_dynamic.bse[f'treat_week_{w}']
    pval = model_dynamic.pvalues[f'treat_week_{w}']
    print(f"  Week {w}: {coef:.2f} (SE: {se:.2f}, p={pval:.3f})")
