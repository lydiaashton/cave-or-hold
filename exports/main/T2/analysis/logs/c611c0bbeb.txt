import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
print("Branches that switched:", df['switched_branch'].sum() // 17)
print("Branches that didn't switch:", (1 - df['switched_branch'].mean()) * 150)
print("\nSwitched branch distribution:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Weeks before and after
print("\nWeeks before switch (after_switch=0):", (df['after_switch'] == 0).sum())
print("Weeks after switch (after_switch=1):", (df['after_switch'] == 1).sum())

# Average service scores by group and period
print("\n=== Average Service Scores ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(pivot)

# Calculate the difference-in-differences manually
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nControl (no switch): Before={before_control:.2f}, After={after_control:.2f}, Change={after_control - before_control:.2f}")
print(f"Treatment (switched): Before={before_treatment:.2f}, After={after_treatment:.2f}, Change={after_treatment - before_treatment:.2f}")

diff_in_diff = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {diff_in_diff:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Simple DiD Regression ===")
print(model1.summary().tables[1])

# DiD with controls (staffing and visits)
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\n=== DiD Regression with Controls ===")
print(model2.summary().tables[1])

# Clustered standard errors by branch
model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD with Clustered SE (by branch) ===")
print(model3.summary().tables[1])

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check parallel trends - average service score by week for each group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
weekly_avg.columns = ['switched_branch', 'week', 'avg_score']

# Pivot for easier viewing
pivot_weekly = weekly_avg.pivot(index='week', columns='switched_branch', values='avg_score')
pivot_weekly.columns = ['Control (No Switch)', 'Treatment (Switched)']
print("=== Weekly Average Service Scores ===")
print(pivot_weekly)

# Calculate the difference between groups each week
pivot_weekly['Difference'] = pivot_weekly['Treatment (Switched)'] - pivot_weekly['Control (No Switch)']
print("\n=== Weekly Difference (Treatment - Control) ===")
print(pivot_weekly['Difference'])

# Check pre-trend (weeks 6-14) vs post-trend (weeks 15-22)
pre_diff = pivot_weekly.loc[6:14, 'Difference'].mean()
post_diff = pivot_weekly.loc[15:22, 'Difference'].mean()
print(f"\nAverage difference before switch (weeks 6-14): {pre_diff:.2f}")
print(f"Average difference after switch (weeks 15-22): {post_diff:.2f}")
print(f"Change in difference: {post_diff - pre_diff:.2f}")

# Create visualization
plt.figure(figsize=(12, 6))
plt.plot(pivot_weekly.index, pivot_weekly['Control (No Switch)'], 'o-', label='Control (No Switch)', linewidth=2, markersize=8)
plt.plot(pivot_weekly.index, pivot_weekly['Treatment (Switched)'], 's-', label='Treatment (Switched)', linewidth=2, markersize=8)
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')
plt.xlabel('Week', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Scores by Week: Treatment vs Control Groups', fontsize=14, fontweight='bold')
plt.legend(fontsize=11)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("\nPlot saved as service_scores_trend.png")

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Effect size in standard deviation units
overall_std = df['service_score'].std()
treatment_effect = 2.12  # from regression
print(f"Overall service score std dev: {overall_std:.2f}")
print(f"Treatment effect: {treatment_effect:.2f}")
print(f"Effect size (Cohen's d): {treatment_effect / overall_std:.2f} SD")

# Percentage improvement relative to baseline
baseline_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
print(f"\nBaseline treatment group mean: {baseline_treatment:.2f}")
print(f"Percentage improvement: {(treatment_effect / baseline_treatment) * 100:.1f}%")

# Check if effect varies by branch characteristics (if any patterns exist)
# Let's see the distribution of effects across branches
branch_effects = df.groupby('branch_id').apply(
    lambda x: x[x['after_switch'] == 1]['service_score'].mean() - x[x['after_switch'] == 0]['service_score'].mean()
).reset_index()
branch_effects.columns = ['branch_id', 'change']

# Merge with switched status
branch_switched = df.groupby('branch_id')['switched_branch'].first().reset_index()
branch_effects = branch_effects.merge(branch_switched, on='branch_id')

print("\n=== Distribution of Score Changes ===")
print("Switched branches:")
switched_changes = branch_effects[branch_effects['switched_branch'] == 1]['change']
print(f"  Mean: {switched_changes.mean():.2f}")
print(f"  Std: {switched_changes.std():.2f}")
print(f"  Min: {switched_changes.min():.2f}")
print(f"  Max: {switched_changes.max():.2f}")
print(f"  % positive: {(switched_changes > 0).mean() * 100:.1f}%")

print("\nControl branches:")
control_changes = branch_effects[branch_effects['switched_branch'] == 0]['change']
print(f"  Mean: {control_changes.mean():.2f}")
print(f"  Std: {control_changes.std():.2f}")
print(f"  Min: {control_changes.min():.2f}")
print(f"  Max: {control_changes.max():.2f}")
print(f"  % positive: {(control_changes > 0).mean() * 100:.1f}%")
