import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch fixed effects
# Model: service_score = β0 + β1*after_switch + β2*switched_branch + β3*(after_switch*switched_branch) + ε
# β3 is the DiD estimate

# First, simple OLS without fixed effects
model_simple = ols('service_score ~ after_switch + switched_branch + treatment', data=df).fit()
print("Simple OLS (no fixed effects):")
print(model_simple.summary().tables[1])

# Now with branch fixed effects (using dummy variables for each branch)
# This controls for time-invariant branch characteristics
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df, branch_dummies], axis=1)

# Build formula with branch dummies
branch_cols = [col for col in df_with_dummies.columns if col.startswith('branch_')]
formula = 'service_score ~ after_switch + switched_branch + treatment + ' + ' + '.join(branch_cols)

model_fe = ols(formula, data=df_with_dummies).fit()
print("\n" + "="*60)
print("With Branch Fixed Effects:")
print("="*60)
print(f"Treatment effect (coefficient on interaction): {model_fe.params['treatment']:.4f}")
print(f"Standard error: {model_fe.bse['treatment']:.4f}")
print(f"t-statistic: {model_fe.tvalues['treatment']:.4f}")
print(f"p-value: {model_fe.pvalues['treatment']:.6f}")
print(f"95% CI: [{model_fe.conf_int().loc['treatment', 0]:.4f}, {model_fe.conf_int().loc['treatment', 1]:.4f}]")

# Also cluster standard errors by branch
from statsmodels.stats.api import sandwich_covariance
cluster_se = model_fe.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(f"\nClustered SE (by branch): {cluster_se.bse['treatment']:.4f}")
print(f"Clustered p-value: {cluster_se.pvalues['treatment']:.6f}")
print(f"Clustered 95% CI: [{cluster_se.conf_int().loc['treatment', 0]:.4f}, {cluster_se.conf_int().loc['treatment', 1]:.4f}]")

import pandas as pd

import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Try a simpler approach - regression without branch FE but with robust SE
# This gives us a good estimate even if not as precise as FE
model = ols('service_score ~ after_switch + switched_branch + treatment', data=df).fit(cov_type='HC3')

print("DiD Regression with Robust Standard Errors")
print("="*70)
print(f"\nTreatment effect (after_switch * switched_branch):")
print(f"  Coefficient: {model.params['treatment']:.4f}")
print(f"  Standard error: {model.bse['treatment']:.4f}")
print(f"  p-value: {model.pvalues['treatment']:.6f}")
ci = model.conf_int().loc['treatment']
print(f"  95% CI: [{ci.iloc[0]:.4f}, {ci.iloc[1]:.4f}]")

# Also run with branch FE using a workaround
# Demean the data within each branch
df_demeaned = df.copy()
branch_means = df.groupby('branch_id')[['service_score', 'after_switch', 'switched_branch', 'treatment']].transform('mean')
for col in ['service_score', 'after_switch', 'switched_branch', 'treatment']:
    df_demeaned[col + '_dm'] = df[col] - branch_means[col]

# Note: after_switch and switched_branch will be collinear with branch FE, so we just use treatment
# Actually for DiD with FE, we can't include switched_branch (time-invariant)
# Let's use a different approach - just the interaction term with branch FE

# Run regression on demeaned data (this is equivalent to including branch FE)
# We need to include after_switch_dm because it varies within branch
model_fe = ols('service_score_dm ~ after_switch_dm + treatment_dm', data=df_demeaned).fit(cov_type='HC3')

print("\n" + "="*70)
print("DiD with Branch Fixed Effects (within-estimator)")
print("="*70)
print(f"\nTreatment effect:")
print(f"  Coefficient: {model_fe.params['treatment_dm']:.4f}")
