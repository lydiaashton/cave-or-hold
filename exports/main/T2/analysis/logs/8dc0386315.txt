import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create treatment interaction
df['treatment'] = df['switched_branch'] * df['after_switch']

# Event study / dynamic DiD - create relative week indicators
# Week 14 is the last pre-switch week, so we'll use it as reference
df['relative_week'] = df['week'] - 14  # So week 14 = 0, week 15 = 1, etc.

# Create indicators for each week relative to switch (excluding reference week 14)
# We'll interact these with switched_branch
for w in range(-8, 9):  # weeks -8 to 8 relative to week 14
    if w != 0:  # Skip reference week
        df[f'rel_week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (excluding reference period)
# Using weeks 6-22, relative weeks are -8 to 8
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'rel_week_{w}' for w in range(-8, 9) if w != 0])
model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Regression (relative to week 14):")
print(model.summary().tables[1])

# Extract the pre-trend coefficients (weeks -8 to -1)
print("\n\nPre-trend coefficients (should be close to 0 if parallel trends hold):")
pre_trend_coefs = []
for w in range(-8, 0):
    coef_name = f'rel_week_{w}'
    if coef_name in model.params.index:
        coef = model.params[coef_name]
        se = model.bse[coef_name]
        pre_trend_coefs.append((w, coef, se))
        print(f"Week {14+w}: coef = {coef:.3f}, SE = {se:.3f}")

# Test if pre-trends are jointly zero
# F-test for all pre-trend coefficients
pre_trend_vars = [f'rel_week_{w}' for w in range(-8, 0) if f'rel_week_{w}' in model.params.index]
print(f"\n\nJoint test of pre-trends (weeks 6-13):")
hypotheses = ', '.join([f'{v} = 0' for v in pre_trend_vars])
ftest = model.f_test(hypotheses)
print(f"F-statistic: {ftest.fvalue:.3f}, p-value: {ftest.pvalue:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create treatment interaction
df['treatment'] = df['switched_branch'] * df['after_switch']

# Event study / dynamic DiD - create relative week indicators
# Week 14 is the last pre-switch week, so we'll use it as reference
df['relative_week'] = df['week'] - 14  # So week 14 = 0, week 15 = 1, etc.

# Create indicators for each week relative to switch (excluding reference week 14)
# Rename to avoid negative sign issues
for w in range(-8, 9):  # weeks -8 to 8 relative to week 14
    if w != 0:  # Skip reference week
        col_name = f'rel_week_p{abs(w)}' if w < 0 else f'rel_week_m{w}'
        df[col_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Build formula
pre_weeks = [f'rel_week_p{w}' for w in range(8, 0, -1)]  # p8, p7, ..., p1
