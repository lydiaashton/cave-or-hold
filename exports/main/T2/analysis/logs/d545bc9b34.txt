import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# Model: service_score ~ switched_branch * after_switch + controls
model = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit()
print("=== DiD Regression Results ===")
print(model.summary())

# Also run a simpler model without controls
model_simple = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("\n=== Simple DiD Regression (no controls) ===")
print(model_simple.summary())

# Get the interaction term coefficient (this is the DiD estimate)
print(f"\n=== Key Coefficient (Interaction Term) ===")
print(f"With controls: {model.params['switched_branch:after_switch']:.3f} (p={model.pvalues['switched_branch:after_switch']:.4f})")
print(f"Without controls: {model_simple.params['switched_branch:after_switch']:.3f} (p={model_simple.pvalues['switched_branch:after_switch']:.4f})")

# Confidence interval
ci = model.conf_int().loc['switched_branch:after_switch']
print(f"\n95% CI for treatment effect: [{ci[0]:.2f}, {ci[1]:.2f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate average service score by week and group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_score']

# Calculate pre-switch trend differences
pre_switch_data = df[df['week'] < 15]
pre_trends = pre_switch_data.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
control_pre = pre_trends[pre_trends['switched_branch'] == 0].set_index('week')['service_score']
treatment_pre = pre_trends[pre_trends['switched_branch'] == 1].set_index('week')['service_score']

# Simple trend analysis (slope) for pre-period
control_slope = np.polyfit(control_pre.index, control_pre.values, 1)[0]
treatment_slope = np.polyfit(treatment_pre.index, treatment_pre.values, 1)[0]
print("=== Pre-Switch Trends (Weeks 6-14) ===")
print(f"Control group trend (slope): {control_slope:.3f} points/week")
print(f"Treatment group trend (slope): {treatment_slope:.3f} points/week")
print(f"Difference in slopes: {treatment_slope - control_slope:.3f} points/week")

# Run regression with clustered standard errors by branch
print("\n=== Regression with Clustered Standard Errors ===")
model_clustered = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_clustered.summary())

# Get the key coefficient
did_coef = model_clustered.params['switched_branch:after_switch']
did_se = model_clustered.bse['switched_branch:after_switch']
did_pvalue = model_clustered.pvalues['switched_branch:after_switch']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se

print(f"\n=== Treatment Effect (with clustered SEs) ===")
print(f"Estimate: {did_coef:.2f} points")
print(f"Standard Error: {did_se:.2f}")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {did_pvalue:.3f}")

# Effect size relative to baseline
baseline_avg = df[df['after_switch'] == 0]['service_score'].mean()
print(f"\n=== Effect Size ===")
print(f"Baseline average service score: {baseline_avg:.1f}")
print(f"Treatment effect as % of baseline: {100*did_coef/baseline_avg:.1f}%")
