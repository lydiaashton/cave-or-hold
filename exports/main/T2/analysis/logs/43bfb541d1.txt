import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# === DIFFERENCE-IN-DIFFERENCES ANALYSIS ===
# DiD = (Treatment_After - Treatment_Before) - (Control_After - Control_Before)

# Get the means
means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
print("=== DIFFERENCE-IN-DIFFERENCES ===")
print("\nMean service scores:")
print(means)

treatment_before = means.loc[1, 0]
treatment_after = means.loc[1, 1]
control_before = means.loc[0, 0]
control_after = means.loc[0, 1]

treatment_change = treatment_after - treatment_before
control_change = control_after - control_before
did_estimate = treatment_change - control_change

print(f"\nTreatment (switched) change: {treatment_change:.3f}")
print(f"Control (non-switched) change: {control_change:.3f}")
print(f"\nDiD Estimate (treatment effect): {did_estimate:.3f}")

# === STATISTICAL TESTING ===
# Create treatment variable (interaction of switched_branch and after_switch)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple regression: service_score ~ treatment + switched_branch + after_switch
import statsmodels.api as sm

X = df[['treatment', 'switched_branch', 'after_switch']]
X = sm.add_constant(X)
y = df['service_score']

model = sm.OLS(y, X).fit()
print("\n\n=== REGRESSION RESULTS (DiD) ===")
print(model.summary())

# === ROBUSTNESS: Add controls ===
X2 = df[['treatment', 'switched_branch', 'after_switch', 'staffing_index', 'visits_index']]
X2 = sm.add_constant(X2)
model2 = sm.OLS(y, X2).fit()
print("\n\n=== REGRESSION WITH CONTROLS ===")
print(model2.summary())

import pandas as pd
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# === FIXED EFFECTS MODEL (branch-level) ===
# This controls for time-invariant branch characteristics
print("=== FIXED EFFECTS MODEL ===")

# Create dummy variables for each branch
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)

# Variables for the model
X = pd.concat([df[['treatment', 'after_switch', 'staffing_index', 'visits_index']], branch_dummies], axis=1)
y = df['service_score']

# This might be computationally intensive, let's use a simpler approach with statsmodels
# Using PanelOLS would be ideal but let's try a simpler fixed effects approach

# Alternative: Use linearmodels package if available, otherwise do manual demeaning
try:
    from linearmodels.panel import PanelOLS
    
    # Prepare panel data
    df_panel = df.copy()
    df_panel = df_panel.set_index(['branch_id', 'week'])
    
    # Fixed effects model
    exog = df_panel[['treatment', 'after_switch', 'staffing_index', 'visits_index']]
    exog = sm.add_constant(exog)
    
    fe_model = PanelOLS(df_panel['service_score'], exog, entity_effects=True)
    fe_results = fe_model.fit(cov_type='clustered', cluster_entity=True)
    print(fe_results)
    
except ImportError:
    print("linearmodels not available, using manual fixed effects...")
    
    # Manual fixed effects: demean within each branch
    df['service_score_demeaned'] = df.groupby('branch_id')['service_score'].transform(lambda x: x - x.mean())
    df['treatment_demeaned'] = df.groupby('branch_id')['treatment'].transform(lambda x: x - x.mean())
    df['after_switch_demeaned'] = df.groupby('branch_id')['after_switch'].transform(lambda x: x - x.mean())
    df['staffing_demeaned'] = df.groupby('branch_id')['staffing_index'].transform(lambda x: x - x.mean())
    df['visits_demeaned'] = df.groupby('branch_id')['visits_index'].transform(lambda x: x - x.mean())
    
    X_fe = df[['treatment_demeaned', 'after_switch_demeaned', 'staffing_demeaned', 'visits_demeaned']]
    X_fe = sm.add_constant(X_fe)
    y_fe = df['service_score_demeaned']
    
    fe_model = sm.OLS(y_fe, X_fe).fit()
    print(fe_model.summary())
