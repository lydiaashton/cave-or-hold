import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Run difference-in-differences regression
# Basic DiD model
model_did = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit()
print("Basic DiD Model:")
print(model_did.summary())

# DiD with controls (staffing_index and visits_index)
model_did_controls = ols('service_score ~ switched_branch + after_switch + treatment_post + staffing_index + visits_index', data=df).fit()
print("\n\nDiD Model with Controls:")
print(model_did_controls.summary())

# Cluster-robust standard errors by branch
model_did_clustered = ols('service_score ~ switched_branch + after_switch + treatment_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Model with Controls (Clustered SE by branch):")
print(model_did_clustered.summary())

weekly_avg.columns = ['Control', 'Treatment']

fig, ax = plt.subplots(figsize=(12, 6))

ax.plot(pre_data['week'], pre_data['Treatment'], 's-', label='Treatment (New Procedure)', color='green', linewidth=2, markersize=8)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Simple test of parallel trends: compare trends in pre-period
pre_df = df[df['week'] < 15].copy()
pre_df['week_centered'] = pre_df['week'] - 10  # center around middle of pre-period
trend_model = ols('service_score ~ switched_branch * week_centered', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print("Pre-treatment trend test (interaction term shows if trends differ):")
print(trend_model.summary())

# Also run a more comprehensive DiD with week fixed effects to control for time trends
df['week_str'] = df['week'].astype(str)
model_fe = ols('service_score ~ switched_branch + after_switch + treatment_post + C(week_str) + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Model with Week Fixed Effects (Clustered SE by branch):")

print(f"P-value: {model_fe.pvalues['treatment_post']:.4f}")

df = pd.read_csv('branches.csv')

# Main DiD model
model_main = ols('service_score ~ switched_branch + after_switch + treatment_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

after = df[df['after_switch']==1].groupby('switched_branch')['service_score'].agg(['count', 'mean', 'std'])
