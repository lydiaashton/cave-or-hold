import numpy as np

df['treatment'] = df['switched_branch'] * df['after_switch']

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

    'before': before,
    'after': after,
    'switched': df.groupby('branch_id')['switched_branch'].first()
})
branch_df['change'] = branch_df['after'] - branch_df['before']

print("\nTreatment branches (switched=1):")
treatment_changes = branch_df[branch_df['switched'] == 1]['change']
print(f"  Mean change: {treatment_changes.mean():.2f}")
print(f"  Std dev: {treatment_changes.std():.2f}")
print(f"  Min: {treatment_changes.min():.2f}")
print(f"  Max: {treatment_changes.max():.2f}")
print(f"  Branches with positive change: {(treatment_changes > 0).sum()} / {len(treatment_changes)}")

print("\nControl branches (switched=0):")
control_changes = branch_df[branch_df['switched'] == 0]['change']
print(f"  Mean change: {control_changes.mean():.2f}")
print(f"  Std dev: {control_changes.std():.2f}")
print(f"  Min: {control_changes.min():.2f}")
print(f"  Max: {control_changes.max():.2f}")
print(f"  Branches with positive change: {(control_changes > 0).sum()} / {len(control_changes)}")

# T-test on branch-level changes
from scipy import stats
t_stat, t_pval = stats.ttest_ind(treatment_changes, control_changes)
print(f"\nTwo-sample t-test on branch-level changes:")
print(f"  t-statistic: {t_stat:.2f}")
print(f"  p-value: {t_pval:.3f}")
