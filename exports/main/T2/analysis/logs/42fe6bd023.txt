import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
print("=== Basic DiD Regression ===")
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model.summary())

# DiD with controls
print("\n=== DiD Regression with Controls ===")
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_controls.summary())

# Check pre-trends (weeks 6-14 only)
print("\n=== Pre-Trend Check (weeks 6-14) ===")
pre_df = df[df['after_switch'] == 0].copy()
# Add week as continuous to check trend
pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()
pre_df['treatment_trend'] = pre_df['switched_branch'] * pre_df['week_centered']

pre_model = ols('service_score ~ switched_branch + week_centered + treatment_trend', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print(pre_model.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate weekly averages by group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg['group'] = weekly_avg['switched_branch'].map({0: 'Control (Old Procedure)', 1: 'Treatment (New Procedure)'})

# Create the plot
fig, ax = plt.subplots(figsize=(10, 6))

for group in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == group]
    label = 'Control (Old Procedure)' if group == 0 else 'Treatment (New Procedure)'
    color = 'blue' if group == 0 else 'green'
    ax.plot(subset['week'], subset['service_score'], marker='o', linewidth=2, label=label, color=color)

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')
ax.axvspan(14.5, 22, alpha=0.1, color='green', label='Post-Switch Period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Service Score', fontsize=12)
ax.set_title('Service Scores by Group Over Time', fontsize=14, fontweight='bold')
ax.legend(loc='best')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("Plot saved as service_scores_trend.png")

# Also calculate confidence intervals for the DiD estimate
from statsmodels.stats.weightstats import DescrStatsW

# Get the treatment effect with bootstrapping for CI
np.random.seed(42)
n_bootstrap = 1000
did_estimates = []

for i in range(n_bootstrap):
    # Sample branches with replacement
    sampled_branches = np.random.choice(df['branch_id'].unique(), size=len(df['branch_id'].unique()), replace=True)
    boot_df = df[df['branch_id'].isin(sampled_branches)]
    
    if len(boot_df) > 100:
        pre_boot = boot_df[boot_df['after_switch'] == 0]
        post_boot = boot_df[boot_df['after_switch'] == 1]
        
        pre_switched = pre_boot[pre_boot['switched_branch'] == 1]['service_score'].mean()
        pre_control = pre_boot[pre_boot['switched_branch'] == 0]['service_score'].mean()
        post_switched = post_boot[post_boot['switched_branch'] == 1]['service_score'].mean()
        post_control = post_boot[post_boot['switched_branch'] == 0]['service_score'].mean()
        
        did = (post_switched - pre_switched) - (post_control - pre_control)
        did_estimates.append(did)

did_ci_lower = np.percentile(did_estimates, 2.5)
did_ci_upper = np.percentile(did_estimates, 97.5)
did_mean = np.mean(did_estimates)

print(f"\n=== Bootstrap Confidence Interval for DiD ===")
print(f"Mean DiD estimate: {did_mean:.2f}")
print(f"95% CI: [{did_ci_lower:.2f}, {did_ci_upper:.2f}]")
print(f"CI includes 0: {did_ci_lower <= 0 <= did_ci_upper}")

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Overall statistics
print("=== Overall Service Score Statistics ===")
print(f"Overall mean: {df['service_score'].mean():.2f}")
print(f"Overall std: {df['service_score'].std():.2f}")
print(f"Min: {df['service_score'].min():.2f}")
print(f"Max: {df['service_score'].max():.2f}")

# Effect size relative to baseline
pre_avg = df[df['after_switch'] == 0]['service_score'].mean()
did_effect = 1.71  # from earlier analysis
print(f"\n=== Effect Size Context ===")
print(f"Pre-switch average score: {pre_avg:.2f}")
print(f"DiD effect estimate: {did_effect:.2f}")
print(f"Effect as % of baseline: {(did_effect/pre_avg)*100:.1f}%")
print(f"Effect relative to std dev: {did_effect/df['service_score'].std():.2f} std devs")

# Check if effect varies by branch characteristics
print("\n=== Branch-Level Analysis ===")
branch_stats = df.groupby(['branch_id', 'switched_branch']).agg({
    'service_score': ['mean', 'std'],
    'week': 'count'
}).reset_index()
branch_stats.columns = ['branch_id', 'switched_branch', 'mean_score', 'std_score', 'n_weeks']

# For switched branches, compare pre vs post
switched_branches = df[df['switched_branch'] == 1]['branch_id'].unique()
print(f"\nSwitched branches (n={len(switched_branches)}):")
for bid in switched_branches[:5]:  # Show first 5
    branch_data = df[df['branch_id'] == bid]
    pre = branch_data[branch_data['after_switch'] == 0]['service_score'].mean()
    post = branch_data[branch_data['after_switch'] == 1]['service_score'].mean()
    print(f"  Branch {bid}: Pre={pre:.1f}, Post={post:.1f}, Change={post-pre:.1f}")

# Count how many switched branches improved vs declined
print("\n=== Direction of Change in Switched Branches ===")
improved = 0
declined = 0
for bid in switched_branches:
    branch_data = df[df['branch_id'] == bid]
    pre = branch_data[branch_data['after_switch'] == 0]['service_score'].mean()
    post = branch_data[branch_data['after_switch'] == 1]['service_score'].mean()
    if post > pre:
        improved += 1
    else:
        declined += 1

print(f"Branches with improved scores: {improved}/{len(switched_branches)}")
print(f"Branches with declined scores: {declined}/{len(switched_branches)}")
