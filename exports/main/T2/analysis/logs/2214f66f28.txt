import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run the basic DiD regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch*after_switch) + error
# beta3 is the DiD estimator

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Basic Difference-in-Differences Regression ===")
print(model.summary())

# Now let's add branch fixed effects and week fixed effects for a more robust analysis
# This controls for time-invariant branch characteristics and common time trends

print("\n\n=== DiD with Branch and Week Fixed Effects ===")
# Create dummy variables for branch and week
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)

# Combine
X = pd.concat([df[['switched_branch', 'after_switch', 'treatment']], 
               branch_dummies, 
               week_dummies], axis=1)
y = df['service_score']

# Add constant
X = sm.add_constant(X)

# Fit model
fe_model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(fe_model.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create the visualization
fig, ax = plt.subplots(figsize=(12, 6))

# Calculate mean service scores by week and group
weekly_means = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

# Plot
ax.plot(weekly_means.index, weekly_means[0], 'o-', label='Control (No Switch)', linewidth=2, markersize=8, color='blue')
ax.plot(weekly_means.index, weekly_means[1], 's-', label='Treatment (Switched)', linewidth=2, markersize=8, color='green')

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Implementation (Week 15)')

# Add shaded region for post-period
ax.axvspan(14.5, 22, alpha=0.2, color='red', label='Post-Switch Period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Mean Service Score', fontsize=12)
ax.set_title('Library Service Scores: Treatment vs Control Groups\nBefore and After Procedure Switch', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=11)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150, bbox_inches='tight')
print("Chart saved as 'service_scores_trend.png'")

# Also calculate effect size in percentage terms
treatment_pre = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_post = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
control_pre = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_post = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

did_effect = (treatment_post - treatment_pre) - (control_post - control_pre)
baseline = control_pre

print(f"\n=== Effect Size Summary ===")
print(f"DiD Effect: {did_effect:.2f} points")
print(f"Baseline (control pre): {baseline:.2f} points")
print(f"Relative effect: {did_effect/baseline*100:.1f}%")
