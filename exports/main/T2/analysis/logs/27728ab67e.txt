import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Calculate average service scores by group and period
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).reset_index()
print("Service scores by group and period:")
print(summary)

# Calculate the differences
print("\n--- Treatment Group (switched_branch=1) ---")
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
print(f"Before switch: {treatment_before:.2f}")
print(f"After switch: {treatment_after:.2f}")
print(f"Change: {treatment_after - treatment_before:.2f}")

print("\n--- Control Group (switched_branch=0) ---")
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
print(f"Before switch: {control_before:.2f}")
print(f"After switch: {control_after:.2f}")
print(f"Change: {control_after - control_before:.2f}")

# Difference-in-differences
diff_in_diff = (treatment_after - treatment_before) - (control_after - control_before)
print(f"\n--- Difference-in-Differences Estimate ---")
print(f"DiD = {diff_in_diff:.2f}")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster
from scipy import stats

df = pd.read_csv('branches.csv')
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Final comprehensive analysis
print("=" * 70)
print("COMPREHENSIVE ANALYSIS SUMMARY")
print("=" * 70)

# Basic statistics
print("\n1. SAMPLE CHARACTERISTICS")
print("-" * 40)
print(f"Total observations: {len(df)}")
print(f"Total branches: {df['branch_id'].nunique()}")
print(f"  - Treatment branches (switched): {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"  - Control branches (no switch): {df[df['switched_branch']==0]['branch_id'].nunique()}")
print(f"Weeks covered: {df['week'].min()} to {df['week'].max()}")
print(f"Switch implemented: Week 15")
print(f"Pre-switch weeks: 9 (weeks 6-14)")
print(f"Post-switch weeks: 8 (weeks 15-22)")

# Main DiD results
print("\n2. MAIN FINDINGS (Difference-in-Differences)")
print("-" * 40)

model = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()
cov_clustered = cov_cluster(model, df['branch_id'])
robust_se = np.sqrt(np.diag(cov_clustered))

treatment_coef = model.params['treated_post']
treatment_se = robust_se[list(model.params.index).index('treated_post')]
treatment_t = treatment_coef / treatment_se
treatment_p = 2 * (1 - stats.norm.cdf(abs(treatment_t)))
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se

print(f"Estimated treatment effect: {treatment_coef:.2f} points")
print(f"Cluster-robust SE: {treatment_se:.2f}")
print(f"t-statistic: {treatment_t:.2f}")
print(f"p-value: {treatment_p:.3f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")

# Simple before/after comparison
print("\n3. SIMPLE BEFORE/AFTER COMPARISON")
print("-" * 40)

treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print(f"Treatment group: {treatment_before:.2f} → {treatment_after:.2f} (change: +{treatment_after-treatment_before:.2f})")
print(f"Control group: {control_before:.2f} → {control_after:.2f} (change: {control_after-control_before:.2f})")
print(f"Net effect (DiD): +{treatment_coef:.2f} points")

# Practical significance
print("\n4. PRACTICAL SIGNIFICANCE")
print("-" * 40)
baseline_avg = df['service_score'].mean()
pct_change = (treatment_coef / baseline_avg) * 100
print(f"Average service score across all data: {baseline_avg:.1f}")
print(f"Treatment effect as % of baseline: {pct_change:.1f}%")
print(f"Interpretation: The new procedure is associated with approximately a {pct_change:.1f}% improvement in service scores")

# Heterogeneity
