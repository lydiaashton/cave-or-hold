import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate mean service scores by group and period
print("=== Mean Service Scores by Group and Period ===")
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='after_switch', 
                       aggfunc='mean')
print(pivot)

# Calculate the DiD estimate manually
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treated_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treated_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"\nControl (old procedure): Before={control_before:.2f}, After={control_after:.2f}, Change={control_after-control_before:.2f}")
print(f"Treated (new procedure): Before={treated_before:.2f}, After={treated_after:.2f}, Change={treated_after-treated_before:.2f}")
did_estimate = (treated_after - treated_before) - (control_after - control_before)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.3f}")

# Run DiD regression
print("\n=== DiD Regression (Basic) ===")
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model1.summary())

# Run DiD regression with controls
print("\n=== DiD Regression (With Controls) ===")
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print(model2.summary())

# Cluster-robust standard errors by branch
print("\n=== DiD Regression with Cluster-Robust SE (by branch) ===")
model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
cov_cluster = model3.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(cov_cluster.summary())

# Simple linear regression for pre-trends

treated_weeks = treated_pre.index.values
treated_scores = treated_pre.values

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Check how many branches are in each group
n_control = df['branch_id'][df['switched_branch']==0].nunique()
n_treated = df['branch_id'][df['switched_branch']==1].nunique()
print(f"Number of control branches: {n_control}")
print(f"Number of treated branches: {n_treated}")
print(f"Total branches: {n_control + n_treated}")

# Check balance of covariates between groups (pre-switch)
pre_df = df[df['after_switch']==0]
print("\n=== Pre-Switch Covariate Balance ===")
print("Staffing Index:")
print(f"  Control mean: {pre_df[pre_df['switched_branch']==0]['staffing_index'].mean():.3f}")
print(f"  Treated mean: {pre_df[pre_df['switched_branch']==1]['staffing_index'].mean():.3f}")

print("Visits Index:")
print(f"  Control mean: {pre_df[pre_df['switched_branch']==0]['visits_index'].mean():.3f}")
print(f"  Treated mean: {pre_df[pre_df['switched_branch']==1]['visits_index'].mean():.3f}")

# Confidence interval for the treatment effect
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
cov_cluster = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
treatment_coef = cov_cluster.params['treatment']
treatment_se = cov_cluster.bse['treatment']
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se
print(f"\n=== Treatment Effect (95% CI) ===")
print(f"Point estimate: {treatment_coef:.2f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {cov_cluster.pvalues['treatment']:.3f}")

# Check if the effect is consistent across post-switch weeks
post_df = df[df['after_switch']==1]
print("\n=== Post-Switch Weekly Effects ===")
for week in sorted(post_df['week'].unique()):
    week_data = post_df[post_df['week']==week]
    control_mean = week_data[week_data['switched_branch']==0]['service_score'].mean()
    treated_mean = week_data[week_data['switched_branch']==1]['service_score'].mean()
    diff = treated_mean - control_mean
    print(f"Week {week}: Control={control_mean:.1f}, Treated={treated_mean:.1f}, Diff={diff:.1f}")
