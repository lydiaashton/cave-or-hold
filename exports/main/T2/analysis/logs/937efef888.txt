import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment_interaction'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with cluster-robust standard errors
model = ols('service_score ~ switched_branch + after_switch + treatment_interaction + staffing_index + visits_index', data=df).fit()
cluster_robust_se = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("DiD Regression with Cluster-Robust Standard Errors (clustered by branch):")
print(cluster_robust_se.summary())

# Extract key coefficient
did_coef = cluster_robust_se.params['treatment_interaction']
did_se = cluster_robust_se.bse['treatment_interaction']
did_ci_lower = did_coef - 1.96 * did_se
did_ci_upper = did_coef + 1.96 * did_se
print(f"\n\nTreatment Effect (DiD): {did_coef:.2f}")
print(f"95% CI: [{did_ci_lower:.2f}, {did_ci_upper:.2f}]")
print(f"p-value: {cluster_robust_se.pvalues['treatment_interaction']:.4f}")

# Check parallel trends more formally - event study
print("\n\nEvent Study - Weekly effects:")
df['week_effect'] = df['week'].astype(str)
# Create week dummies interacted with treatment
for w in range(6, 15):  # Pre-treatment weeks
    df[f'week_{w}'] = (df['week'] == w).astype(int)
    df[f'treat_week_{w}'] = df['switched_branch'] * df[f'week_{w}']

# Run event study regression (using week 14 as reference)
event_vars = '+'.join([f'treat_week_{w}' for w in range(6, 15)])
event_model = ols(f'service_score ~ switched_branch + after_switch + staffing_index + visits_index + {event_vars}', data=df).fit()
print("\nPre-treatment coefficients (testing parallel trends):")
for w in range(6, 15):
    coef = event_model.params[f'treat_week_{w}']
    se = event_model.bse[f'treat_week_{w}']
    pval = event_model.pvalues[f'treat_week_{w}']
    print(f"Week {w}: coef={coef:.2f}, se={se:.2f}, p={pval:.3f}")

# Check if the effect is consistent over time

# Add weeks since treatment
df['weeks_since_switch'] = df['week'].apply(lambda w: max(0, w - 14))

# Check for any concerning patterns
print("Data quality checks:")
print(f"Missing values:\n{df.isnull().sum()}")

# Check if treatment and control branches had similar pre-treatment scores
pre_df = df[df['after_switch'] == 0]
pre_avg = pre_df.groupby('switched_branch')['service_score'].agg(['mean', 'std'])
print(f"\nPre-treatment service scores by group:\n{pre_avg}")

# T-test for pre-treatment equivalence
from scipy import stats
control_pre = pre_df[pre_df['switched_branch']==0]['service_score']
treatment_pre = pre_df[pre_df['switched_branch']==1]['service_score']
t_stat, p_val = stats.ttest_ind(control_pre, treatment_pre)
print(f"\nPre-treatment difference t-test: t={t_stat:.2f}, p={p_val:.3f}")

# Check variance in treatment effect by looking at individual branch changes
print("\n\nBranch-level analysis:")
branch_pre = df[df['after_switch']==0].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_post = df[df['after_switch']==1].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_pre.columns = ['branch_id', 'switched_branch', 'pre_score']
branch_post.columns = ['branch_id', 'switched_branch', 'post_score']
branch_change = branch_pre.merge(branch_post, on=['branch_id', 'switched_branch'])
branch_change['change'] = branch_change['post_score'] - branch_change['pre_score']

print(f"\nTreatment branches (n={len(branch_change[branch_change['switched_branch']==1])}):")
print(branch_change[branch_change['switched_branch']==1]['change'].describe())

print(f"\nControl branches (n={len(branch_change[branch_change['switched_branch']==0])}):")
print(branch_change[branch_change['switched_branch']==0]['change'].describe())
