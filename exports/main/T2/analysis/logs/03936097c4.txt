import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch'] * df['after_switch']

# Summary table: average service scores by group and period
print("=== Average Service Scores by Group and Period ===")
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(summary)

# Calculate the DiD estimate manually
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
before_treated = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_treated = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

did_estimate = (after_treated - before_treated) - (after_control - before_control)
print(f"\n=== Manual DiD Calculation ===")
print(f"Control group (no switch): Before={before_control:.2f}, After={after_control:.2f}, Change={after_control-before_control:.2f}")
print(f"Treated group (switched): Before={before_treated:.2f}, After={after_treated:.2f}, Change={after_treated-before_treated:.2f}")
print(f"DiD Estimate (treatment effect): {did_estimate:.2f}")

# Run regression with robust standard errors clustered by branch
print("\n=== DiD Regression (clustered by branch) ===")
model = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
print(model.get_robustcov_results(cov_type='cluster', cov_kwds={'groups': df['branch_id']}).summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import warnings
warnings.filterwarnings('ignore')

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# Main DiD with controls
print("=== DiD Regression with Controls ===")
model_controls = ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', data=df).fit()
robust_controls = model_controls.get_robustcov_results(cov_type='HC1')
print(robust_controls.summary())

# Event study: effect by week after switch
print("\n=== Event Study: Effect by Week ===")
df['post_week'] = df['week'] - 14  # weeks since switch (week 15 = 1, week 14 = 0)
df['post_week'] = df['post_week'].clip(lower=0)  # cap pre-period at 0

# Create week dummies for post-period
for w in range(1, 9):  # weeks 15-22
    df[f'post_{w}'] = ((df['week'] == 14 + w) & (df['switched_branch'] == 1)).astype(int)

# Regression with week-specific effects
week_vars = [f'post_{w}' for w in range(1, 9)]
formula = f'service_score ~ switched_branch + after_switch + {" + ".join(week_vars)}'
model_event = ols(formula, data=df).fit()
robust_event = model_event.get_robustcov_results(cov_type='HC1')

# Extract just the post-week coefficients
print("Week-specific treatment effects (relative to pre-period baseline):")
for w in range(1, 9):
    coef = robust_event.params[f'post_{w}']
    se = robust_event.bse[f'post_{w}']
    pval = robust_event.pvalues[f'post_{w}']
    print(f"  Week {14+w} (post week {w}): {coef:.2f} (SE={se:.2f}, p={pval:.3f})")

# Confidence intervals
print("\n=== 95% Confidence Intervals for Treatment Effect ===")
treated_coef = robust_controls.params['treated']
treated_se = robust_controls.bse['treated']
ci_lower = treated_coef - 1.96 * treated_se
ci_upper = treated_coef + 1.96 * treated_se
print(f"Point estimate: {treated_coef:.2f}")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
