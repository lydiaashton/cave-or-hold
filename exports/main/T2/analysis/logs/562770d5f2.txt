import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
switched_counts = df.groupby('branch_id')['switched_branch'].first().value_counts()
print("Branches by switch status:")
print(switched_counts)
print(f"\nTotal branches: {df['branch_id'].nunique()}")
print(f"Switched: {switched_counts.get(1, 0)}")
print(f"Not switched: {switched_counts.get(0, 0)}")

# Average service scores by group and period
print("\n\nAverage service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the difference-in-differences manually
# Treatment group (switched=1): after - before
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nTreatment group (switched):")
print(f"  Before: {treatment_before:.2f}")
print(f"  After: {treatment_after:.2f}")
print(f"  Change: {treatment_after - treatment_before:.2f}")

print(f"\nControl group (not switched):")
print(f"  Before: {control_before:.2f}")
print(f"  After: {control_after:.2f}")
print(f"  Change: {control_after - control_before:.2f}")

did_effect = (treatment_after - treatment_before) - (control_after - control_before)
print(f"\nDifference-in-Differences estimate: {did_effect:.2f}")

model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

import pandas as pd

from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check parallel trends: average service score by week for each group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
print("Weekly averages by group:")
print(weekly_avg.pivot(index='week', columns='switched_branch', values='service_score').round(2))

# Create a visualization
plt.figure(figsize=(12, 6))
for switched in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == switched]
    label = 'Switched branches' if switched == 1 else 'Non-switched branches'
    plt.plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2)

plt.axvline(x=14.5, color='red', linestyle='--', label='Switch occurred (week 15)')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Scores Over Time: Switched vs Non-Switched Branches')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.savefig('service_scores_trend.png', dpi=150, bbox_inches='tight')
print("\nPlot saved as service_scores_trend.png")

# Test for pre-trends: compare trends before week 15
pre_period = df[df['week'] < 15].copy()
pre_period['week_centered'] = pre_period['week'] - pre_period['week'].mean()

# Regress service_score on week trend interacted with switched_branch in pre-period
pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_period).fit()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model with clustered standard errors at branch level
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Model with Clustered Standard Errors (by branch):")
print(model_clustered.summary())

# Also run with branch fixed effects and clustered SE
df_cat = df.copy()
df_cat['branch_id'] = df_cat['branch_id'].astype('category')

# For fixed effects, we need to be careful about the switched_branch variable
# Since it's time-invariant, it gets absorbed by branch FEs
model_fe = ols('service_score ~ after_switch + treatment + staffing_index + visits_index + C(branch_id)', data=df_cat).fit(cov_type='cluster', cov_kwds={'groups': df_cat['branch_id']})

print("\n\nModel with Branch Fixed Effects and Clustered SE:")
print("\nKey coefficients:")
print(f"  after_switch: {model_fe.params['after_switch']:.3f} (SE: {model_fe.bse['after_switch']:.3f}, p: {model_fe.pvalues['after_switch']:.3f})")
print(f"  treatment: {model_fe.params['treatment']:.3f} (SE: {model_fe.bse['treatment']:.3f}, p: {model_fe.pvalues['treatment']:.3f})")
print(f"  staffing_index: {model_fe.params['staffing_index']:.3f} (SE: {model_fe.bse['staffing_index']:.3f}, p: {model_fe.pvalues['staffing_index']:.3f})")
print(f"  visits_index: {model_fe.params['visits_index']:.3f} (SE: {model_fe.bse['visits_index']:.3f}, p: {model_fe.pvalues['visits_index']:.3f})")

# Calculate confidence interval for treatment effect
treatment_effect = model_fe.params['treatment']
treatment_se = model_fe.bse['treatment']
ci_lower = treatment_effect - 1.96 * treatment_se
ci_upper = treatment_effect + 1.96 * treatment_se
print(f"\n95% CI for treatment effect: [{ci_lower:.2f}, {ci_upper:.2f}]")
