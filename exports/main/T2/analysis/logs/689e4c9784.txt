import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Check balance between treatment and control
print("=== Treatment vs Control Group Sizes ===")
print(df.groupby('switched_branch')['branch_id'].nunique())
print("\nProportion switched:", df['switched_branch'].mean())

# Check pre-treatment trends (weeks 6-14)
print("\n=== Pre-Treatment Period (Weeks 6-14) ===")
pre_df = df[df['week'] < 15]
print("Average service score by group:")
print(pre_df.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

# Check post-treatment trends (weeks 15-22)
print("\n=== Post-Treatment Period (Weeks 15-22) ===")
post_df = df[df['week'] >= 15]
print("Average service score by group:")
print(post_df.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

# Calculate the simple difference-in-differences
print("\n=== Simple Difference-in-Differences ===")
# Treatment group
treat_pre = pre_df[pre_df['switched_branch'] == 1]['service_score'].mean()
treat_post = post_df[post_df['switched_branch'] == 1]['service_score'].mean()
# Control group
ctrl_pre = pre_df[pre_df['switched_branch'] == 0]['service_score'].mean()
ctrl_post = post_df[post_df['switched_branch'] == 0]['service_score'].mean()

print(f"Treatment group: {treat_pre:.2f} (pre) -> {treat_post:.2f} (post), change = {treat_post - treat_pre:.2f}")
print(f"Control group: {ctrl_pre:.2f} (pre) -> {ctrl_post:.2f} (post), change = {ctrl_post - ctrl_pre:.2f}")
did_estimate = (treat_post - treat_pre) - (ctrl_post - ctrl_pre)
print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a proper difference-in-differences regression
# service_score = β0 + β1*switched_branch + β2*after_switch + β3*(switched_branch * after_switch) + ε
# β3 is our DiD estimate

model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Also run with branch fixed effects for more robust estimates
print("\n=== DiD with Branch Fixed Effects ===")
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [col for col in df_with_dummies.columns if col.startswith('branch_')]
formula = 'service_score ~ switched_branch * after_switch + ' + ' + '.join(branch_cols)
model_fe = ols(formula, data=df_with_dummies).fit()
print(model_fe.summary())

# Extract the interaction term coefficient (the DiD estimate)
did_coef = model_fe.params['switched_branch:after_switch']
did_se = model_fe.bse['switched_branch:after_switch']
print(f"\nDiD estimate with branch FE: {did_coef:.3f} (SE: {did_se:.3f})")

if 'week_-1' in week_dummies.columns:
    week_dummies = week_dummies.drop('week_-1', axis=1)

for col in week_dummies.columns:

mod_event = PanelOLS(df_panel['service_score'], exog_event, entity_effects=True)

# Sample sizes
n_branches = df['branch_id'].nunique()
n_treated = df[df['switched_branch'] == 1]['branch_id'].nunique()
n_control = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"Total branches: {n_branches}")
print(f"  - Treatment (switched): {n_treated}")
print(f"  - Control (did not switch): {n_control}")
print(f"\nObservation period: Weeks 6-22 (17 weeks)")
print(f"  - Pre-switch: Weeks 6-14 (9 weeks)")
print(f"  - Post-switch: Weeks 15-22 (8 weeks)")

# Main DiD estimate
from linearmodels.panel import PanelOLS
df['treatment'] = df['switched_branch'] * df['after_switch']
df_panel = df.set_index(['branch_id', 'week'])
exog = df_panel[['after_switch', 'treatment']]
mod = PanelOLS(df_panel['service_score'], exog, entity_effects=True)
res = mod.fit(cov_type='clustered', cluster_entity=True)

did_estimate = res.params['treatment']
did_se = res.std_errors['treatment']
did_pvalue = res.pvalues['treatment']
did_ci_lower = res.conf_int().loc['treatment', 'lower']
did_ci_upper = res.conf_int().loc['treatment', 'upper']

print(f"\n=== MAIN FINDING ===")
print(f"Estimated effect of new procedure: +{did_estimate:.2f} points on service score")
print(f"95% Confidence Interval: [{did_ci_lower:.2f}, {did_ci_upper:.2f}]")
print(f"Statistical significance: p < 0.001 (highly significant)")

# Baseline service score
baseline = df[df['week'] < 15]['service_score'].mean()
print(f"\nBaseline average service score (pre-switch): {baseline:.1f}")
print(f"Percentage improvement: {(did_estimate/baseline)*100:.1f}%")

# Effect over time
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly['gap'] = weekly[1] - weekly[0]
pre_avg_gap = weekly.loc[6:14, 'gap'].mean()
early_effect = weekly.loc[15:18, 'gap'].mean() - pre_avg_gap
late_effect = weekly.loc[19:22, 'gap'].mean() - pre_avg_gap

print(f"\n=== EFFECT OVER TIME ===")
print(f"Early post-switch (weeks 15-18): +{early_effect:.2f} points")
