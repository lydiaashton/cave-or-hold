import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run difference-in-differences regression
# service_score = beta0 + beta1*treatment + beta2*post + beta3*(treatment*post) + error
# beta3 is the DiD estimate

df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['treatment_post'] = df['treatment'] * df['post']

# Basic DiD regression
model = ols('service_score ~ treatment + post + treatment_post', data=df).fit()
print("Basic DiD Regression:")
print(model.summary())

# Add branch fixed effects
print("\n\nDiD with Branch Fixed Effects:")
model_fe = ols('service_score ~ treatment + post + treatment_post + C(branch_id)', data=df).fit()
print(model_fe.summary())

# Add week fixed effects too
print("\n\nDiD with Branch and Week Fixed Effects:")
model_fef = ols('service_score ~ treatment + post + treatment_post + C(branch_id) + C(week)', data=df).fit()
print(model_fef.summary())

# Manual clustered SE calculation
def cluster_se(model, clusters):
    """Calculate cluster-robust standard errors"""
    import numpy as np
    from scipy.sparse import csr_matrix
    
    # Get residuals and design matrix
    resid = model.resid.values
    X = model.model.exog
    
    # Get unique clusters
    unique_clusters = np.unique(clusters)
    n_clusters = len(unique_clusters)
    
    # Calculate cluster-robust variance
    bread = np.linalg.inv(X.T @ X)
    
    meat = np.zeros((X.shape[1], X.shape[1]))
    for c in unique_clusters:
        mask = clusters == c
        Xc = X[mask]
        rc = resid[mask]
        score = Xc.T * rc
        meat += np.outer(score, score)
    
    # Degrees of freedom adjustment
    n = len(resid)
    k = X.shape[1]
    df_adjustment = (n - 1) / (n - k) * n_clusters / (n_clusters - 1)
    
    cov = df_adjustment * bread @ meat @ bread
    se = np.sqrt(np.diag(cov))
    
    return se, cov

# Manual clustered SE calculation
def cluster_se(model, clusters):
    """Calculate cluster-robust standard errors"""
    import numpy as np
    
    # Get residuals and design matrix
    resid = model.resid.values
    X = model.model.exog
    
    # Get unique clusters
    unique_clusters = np.unique(clusters)
    n_clusters = len(unique_clusters)
    
    # Calculate cluster-robust variance
    bread = np.linalg.inv(X.T @ X)
    
    meat = np.zeros((X.shape[1], X.shape[1]))
    for c in unique_clusters:
        mask = clusters == c
        Xc = X[mask]
        rc = resid[mask]
        score = (Xc.T * rc).sum(axis=1)  # Sum scores within cluster
        meat += np.outer(score, score)
    
    # Degrees of freedom adjustment
    n = len(resid)
    k = X.shape[1]
    df_adjustment = (n - 1) / (n - k) * n_clusters / (n_clusters - 1)
    
    cov = df_adjustment * bread @ meat @ bread
    se = np.sqrt(np.diag(cov))
    
    return se, cov

clusters = df['branch_id'].values
