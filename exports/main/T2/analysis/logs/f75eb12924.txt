# Check balance - how many observations per group
print("\nObservations by group:")
print(df.groupby(['switched_branch', 'after_switch']).size())

# Calculate mean service scores by group and period
print("\nMean service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the DiD estimate manually
# Treatment group change
treatment_pre = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_post = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_change = treatment_post - treatment_pre

# Control group change
control_pre = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_post = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
control_change = control_post - control_pre

did_estimate = treatment_change - control_change

print(f"\n--- Difference-in-Differences Calculation ---")
print(f"Treatment group: Pre={treatment_pre:.2f}, Post={treatment_post:.2f}, Change={treatment_change:.2f}")
print(f"Control group: Pre={control_pre:.2f}, Post={control_post:.2f}, Change={control_change:.2f}")
print(f"DiD Estimate (Treatment - Control): {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Main DiD model with confidence interval
model = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

did_coef = model.params['switched_branch:after_switch']
did_se = model.bse['switched_branch:after_switch']
did_pvalue = model.pvalues['switched_branch:after_switch']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se

print("=== Main DiD Results ===")
print(f"Point estimate: {did_coef:.2f}")
print(f"Standard error: {did_se:.2f}")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {did_pvalue:.4f}")

# Calculate effect size relative to baseline
baseline_score = df[df['after_switch'] == 0]['service_score'].mean()
effect_size_pct = (did_coef / baseline_score) * 100
print(f"\nBaseline average score: {baseline_score:.2f}")
print(f"Effect as % of baseline: {effect_size_pct:.1f}%")

# Check robustness - exclude week 14 (potential pre-trend issue)
df_robust = df[df['week'] != 14].copy()
model_robust = ols('service_score ~ switched_branch * after_switch', data=df_robust).fit(cov_type='cluster', cov_kwds={'groups': df_robust['branch_id']})
print(f"\n=== Robustness check (excluding week 14) ===")
print(f"DiD estimate: {model_robust.params['switched_branch:after_switch']:.2f}")
print(f"P-value: {model_robust.pvalues['switched_branch:after_switch']:.4f}")

# Also check with branch fixed effects
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ switched_branch * after_switch + C(branch_id_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print(f"DiD estimate: {model_fe.params['switched_branch:after_switch']:.2f}")
print(f"P-value: {model_fe.pvalues['switched_branch:after_switch']:.4f}")
