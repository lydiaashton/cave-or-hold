import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Check balance - how many observations per group/time period?
print("\nObservations by group and period:")
print(df.groupby(['switched_branch', 'after_switch']).size().unstack(fill_value=0))

# Average service scores by group and period
print("\nAverage service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
print(pivot)

# Calculate the difference-in-differences manually
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treated_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treated_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nControl (no switch) - Before: {control_before:.2f}, After: {control_after:.2f}, Change: {control_after - control_before:.2f}")
print(f"Treated (switched) - Before: {treated_before:.2f}, After: {treated_after:.2f}, Change: {treated_after - treated_before:.2f}")
print(f"\nDifference-in-Differences estimate: {(treated_after - treated_before) - (control_after - control_before):.2f}")

# Check pre-trends - look at scores by week for each group
print("\n\nService scores by week and group:")
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack()
print(weekly_avg)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# Basic DiD: service_score ~ switched_branch * after_switch
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("Basic DiD Regression:")
print(model.summary())

# Add branch fixed effects
print("\n\nDiD with Branch Fixed Effects:")
model_fe = ols('service_score ~ switched_branch * after_switch + C(branch_id)', data=df).fit()
print(model_fe.summary())

# Add week fixed effects as well
print("\n\nDiD with Branch and Week Fixed Effects:")
model_twfe = ols('service_score ~ switched_branch * after_switch + C(branch_id) + C(week)', data=df).fit()
print(model_twfe.summary())

# Add control variables
print("\n\nDiD with Branch/Week FE and Controls:")
model_full = ols('service_score ~ switched_branch * after_switch + C(branch_id) + C(week) + staffing_index + visits_index', data=df).fit()
print(model_full.summary())

df = pd.read_csv('branches.csv')

# Get the key coefficient from the TWFE model
model_twfe = ols('service_score ~ switched_branch * after_switch + C(branch_id) + C(week)', data=df).fit()
print("Two-Way Fixed Effects DiD Model (key results):")
print(f"DiD coefficient (switched_branch:after_switch): {model_twfe.params['switched_branch:after_switch']:.3f}")

print(f"95% CI: [{model_twfe.conf_int().loc['switched_branch:after_switch'][0]:.3f}, {model_twfe.conf_int().loc['switched_branch:after_switch'][1]:.3f}]")

pre_data = df[df['week'] < 15]

pre_model = ols('service_score ~ switched_branch * week + C(branch_id)', data=pre_data).fit()
print(f"Interaction term (switched_branch:week) in pre-period: {pre_model.params['switched_branch:week']:.3f}")
print(f"p-value: {pre_model.pvalues['switched_branch:week']:.3f}")
