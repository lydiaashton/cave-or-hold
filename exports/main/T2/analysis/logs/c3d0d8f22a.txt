import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Calculate weekly averages by group for visualization
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg['group'] = weekly_avg['switched_branch'].map({0: 'Control (Old Procedure)', 1: 'Treatment (New Procedure)'})

# Create the plot
fig, ax = plt.subplots(figsize=(12, 6))

for group in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == group]
    label = 'Control (Old Procedure)' if group == 0 else 'Treatment (New Procedure)'
    ax.plot(subset['week'], subset['service_score'], marker='o', linewidth=2, markersize=8, label=label)

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch implemented (Week 15)')
ax.axvspan(15, 22, alpha=0.1, color='green', label='Post-switch period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Scores by Branch Group Over Time', fontsize=14)
ax.legend(loc='best')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("Plot saved as service_scores_trend.png")

# Now let's do DiD with cluster-robust standard errors
df['treated'] = df['switched_branch'] * df['after_switch']

# Simple DiD with cluster-robust SEs
model = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
cluster_robust_se = model.get_robustcov_results(cov_type='cluster', clusters=df['branch_id'])
print("\n=== DiD with Cluster-Robust Standard Errors ===")
print(cluster_robust_se.summary())

# Check pre-trends more formally - interaction of week and treatment group in pre-period
pre_data = df[df['week'] < 15].copy()
pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()
pre_trend_model = ols('service_score ~ switched_branch * week_centered', data=pre_data).fit()
print("\n=== Pre-Treatment Trend Test ===")
print(pre_trend_model.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster
from scipy import stats

# Load the data
df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# Full model with controls
model = ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', data=df).fit()
cov_cluster_robust = cov_cluster(model, df['branch_id'])
cluster_robust_se = np.sqrt(np.diag(cov_cluster_robust))

# Get the treatment effect
treated_coef = model.params['treated']
treated_se = cluster_robust_se[list(model.params.index).index('treated')]

# 95% confidence interval
ci_lower = treated_coef - 1.96 * treated_se
ci_upper = treated_coef + 1.96 * treated_se

print("=== Treatment Effect Estimate ===")
print(f"Point estimate: {treated_coef:.2f} points")
print(f"Cluster-robust SE: {treated_se:.2f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"p-value: {2 * (1 - stats.norm.cdf(abs(treated_coef / treated_se))):.3f}")

# Calculate the effect in percentage terms relative to baseline
baseline_score = df[df['week'] < 15]['service_score'].mean()
pct_effect = (treated_coef / baseline_score) * 100
print(f"\nBaseline average service score (pre-treatment): {baseline_score:.1f}")
print(f"Estimated effect as % of baseline: {pct_effect:.1f}%")

# Summary statistics for the board
print("\n=== Summary Statistics ===")
print(f"Total observations: {len(df)}")
print(f"Number of branches: {df['branch_id'].nunique()}")
print(f"Branches that switched: {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"Branches that didn't switch: {df[df['switched_branch']==0]['branch_id'].nunique()}")
print(f"Weeks of data: {df['week'].min()} to {df['week'].max()}")

# Average scores by group and period
print("\n=== Average Service Scores ===")
pre_scores = df[df['week'] < 15].groupby('switched_branch')['service_score'].mean()
post_scores = df[df['week'] >= 15].groupby('switched_branch')['service_score'].mean()

print(f"Control group (old procedure):")
print(f"  Pre-switch (weeks 6-14): {pre_scores[0]:.1f}")
print(f"  Post-switch (weeks 15-22): {post_scores[0]:.1f}")
print(f"  Change: {post_scores[0] - pre_scores[0]:.1f}")

print(f"\nTreatment group (new procedure):")
print(f"  Pre-switch (weeks 6-14): {pre_scores[1]:.1f}")
print(f"  Post-switch (weeks 15-22): {post_scores[1]:.1f}")
print(f"  Change: {post_scores[1] - pre_scores[1]:.1f}")
