import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

# === 1. Summary statistics by group and period ===
print("=" * 60)
print("SERVICE SCORE SUMMARY BY GROUP AND PERIOD")
print("=" * 60)

summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)
print()

# Calculate the DiD estimate manually
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
before_treated = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treated = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("Before Switch:")
print(f"  Control branches (no switch): {before_control:.2f}")
print(f"  Treated branches (switched):  {before_treated:.2f}")
print(f"  Difference: {before_treated - before_control:.2f}")
print()
print("After Switch:")
print(f"  Control branches (no switch): {after_control:.2f}")
print(f"  Treated branches (switched):  {after_treated:.2f}")
print(f"  Difference: {after_treated - after_control:.2f}")
print()

did_estimate = (after_treated - before_treated) - (after_control - before_control)
print(f"Difference-in-Differences Estimate: {did_estimate:.2f}")
print()

# === 2. Regression Analysis ===
print("=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD model
model1 = ols('service_score ~ treated + post + treated_post', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ treated + post + treated_post + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Clustered standard errors by branch
model3 = ols('service_score ~ treated + post + treated_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 3: DiD with controls (clustered SE by branch)")
print(model3.summary().tables[1])

# === 3. Check parallel trends - plot weekly averages ===
print("\n" + "=" * 60)
print("WEEKLY TRENDS BY GROUP")
print("=" * 60)

weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(weekly_avg)

# Plot
plt.plot(weekly_avg.index, weekly_avg['Control (No Switch)'], 'o-', label='Control (No Switch)', linewidth=2, markersize=8)
plt.plot(weekly_avg.index, weekly_avg['Treatment (Switched)'], 's-', label='Treatment (Switched)', linewidth=2, markersize=8)

# Add vertical line at week 15 (switch point)
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')
plt.axvspan(14.5, 22.5, alpha=0.1, color='red', label='Post-Switch Period')

plt.xlabel('Week', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Score Trends: Treatment vs Control Branches', fontsize=14, fontweight='bold')
plt.legend(loc='best', fontsize=10)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.ylim(40, 56)

plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("Plot saved as service_score_trends.png")

print("\n" + "=" * 60)
print("PRE-TRENDS ANALYSIS (Weeks 6-14 only)")
print("=" * 60)

pre_df = df[df['week'] < 15].copy()
pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()

# Test if trends were parallel before the switch
model_pre = ols('service_score ~ treated * week_centered', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print("\nPre-trends test (interaction between treated and week trend):")
print(model_pre.summary().tables[1])

# Also check average difference in pre-period
pre_summary = pre_df.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print("\nPre-period averages:")
print(pre_summary)

# === Effect size and confidence interval ===
print("\n" + "=" * 60)
print("EFFECT SIZE AND CONFIDENCE INTERVAL")
print("=" * 60)

model_full = ols('service_score ~ treated + post + treated_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

pct_change = (did_coef / baseline) * 100
