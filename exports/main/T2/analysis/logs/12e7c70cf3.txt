import pandas as pd

df = pd.read_csv('branches.csv')

# Service score summary by group and period
print("=== Service Score Summary ===")
print("\nOverall service score stats:")

summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Calculate the differences
print("\n\n=== Difference-in-Differences Calculation ===")

# Pre-period means (after_switch = 0)
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
pre_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()

# Post-period means (after_switch = 1)
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
post_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"  Pre-period mean: {pre_control:.3f}")
print(f"  Post-period mean: {post_control:.3f}")
print(f"  Change: {post_control - pre_control:.3f}")

print(f"  Pre-period mean: {pre_treatment:.3f}")
print(f"  Post-period mean: {post_treatment:.3f}")
print(f"  Change: {post_treatment - pre_treatment:.3f}")

# DiD estimate
did_estimate = (post_treatment - pre_treatment) - (post_control - pre_control)
print(f"\nDifference-in-Differences estimate: {did_estimate:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create event time relative to switch (week 15)
# Week 14 = -1, Week 15 = 0, etc.
df['event_time'] = df['week'] - 15

# Create indicators for each week relative to switch (omitting week -1 as reference)
# We'll create dummies for weeks -8 to -2 (pre) and 0 to 7 (post)
for t in list(range(-8, -1)) + list(range(0, 8)):
    df[f'week_{t}'] = ((df['event_time'] == t) & (df['switched_branch'] == 1)).astype(int)

# Event study regression (omitting week -1 as reference)
# Include switched_branch main effect and week fixed effects
event_study_vars = '+'.join([f'week_{t}' for t in list(range(-8, -1)) + list(range(0, 8))])
formula = f'service_score ~ switched_branch + C(week) + {event_study_vars}'

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for event time dummies
print("=== Event Study Results ===")
print("Coefficients for treatment group by week relative to switch (week -1 is reference):")
print()

event_coefs = []
for t in list(range(-8, -1)) + list(range(0, 8)):
    coef_name = f'week_{t}'
    if coef_name in model_event.params.index:
        coef = model_event.params[coef_name]
        se = model_event.bse[coef_name]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        pval = model_event.pvalues[coef_name]
        event_coefs.append({'week_rel': t, 'coef': coef, 'se': se, 'ci_low': ci_low, 'ci_high': ci_high, 'pval': pval})
        print(f"Week {t:+d}: coef={coef:.3f}, SE={se:.3f}, 95% CI=[{ci_low:.3f}, {ci_high:.3f}], p={pval:.4f}")

event_coefs_df = pd.DataFrame(event_coefs)
