import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster
from scipy import stats

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run regression with cluster-robust standard errors at branch level
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
cluster_robust_se = cov_cluster(model, df['branch_id'])

# Get the treatment effect with cluster-robust SE
treatment_coef = model.params['treatment']
# The cluster_robust_se is a numpy array, get the diagonal for SEs
cluster_se = np.sqrt(np.diag(cluster_robust_se))
treatment_se = cluster_se[model.params.index.get_loc('treatment')]
treatment_t = treatment_coef / treatment_se
treatment_p = 2 * (1 - stats.t.cdf(abs(treatment_t), df=len(df['branch_id'].unique()) - 1))

print("Regression coefficients with cluster-robust SEs:")
for i, param in enumerate(model.params.index):
    print(f"  {param}: coef={model.params[param]:.3f}, cluster_se={cluster_se[i]:.3f}")

print(f"\n=== Treatment Effect (Difference-in-Differences) ===")
print(f"Estimate: {treatment_coef:.2f} points")
print(f"Cluster-robust SE: {treatment_se:.2f}")
print(f"t-statistic: {treatment_t:.2f}")
print(f"p-value: {treatment_p:.4f}")
print(f"95% CI: [{treatment_coef - 1.96*treatment_se:.2f}, {treatment_coef + 1.96*treatment_se:.2f}]")

# Check pre-trends more formally - compare trends before week 15
pre_data = df[df['week'] < 15].copy()

pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_data).fit()

import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run regression with branch fixed effects (using C() for categorical)
# This is cleaner than creating dummies manually
model_fe = ols('service_score ~ C(branch_id) + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("=== Regression with Branch Fixed Effects ===")
# Just show the key coefficients
print(f"\nKey coefficients:")
print(f"  after_switch: {model_fe.params['after_switch']:.3f} (p={model_fe.pvalues['after_switch']:.3f})")
print(f"  treatment: {model_fe.params['treatment']:.3f} (p={model_fe.pvalues['treatment']:.3f})")
print(f"  staffing_index: {model_fe.params['staffing_index']:.3f} (p={model_fe.pvalues['staffing_index']:.3f})")
print(f"  visits_index: {model_fe.params['visits_index']:.3f} (p={model_fe.pvalues['visits_index']:.3f})")

# Calculate effect size relative to baseline
baseline_score = df['service_score'].mean()
treatment_effect = model_fe.params['treatment']
effect_size_pct = (treatment_effect / baseline_score) * 100
print(f"\n=== Effect Size ===")
print(f"Baseline average service score: {baseline_score:.1f}")
print(f"Treatment effect: {treatment_effect:.2f} points")
print(f"Relative effect: {effect_size_pct:.1f}%")

# Check if the effect varies by time since switch
df['weeks_since_switch'] = np.where(df['after_switch'] == 1, df['week'] - 14, 0)
df['treatment_interact'] = df['switched_branch'] * df['weeks_since_switch']

model_dynamic = ols('service_score ~ C(branch_id) + staffing_index + visits_index + switched_branch + after_switch + treatment + treatment_interact', data=df).fit()
print(f"\n=== Dynamic Effects (weeks since switch) ===")
print(f"  treatment (immediate effect): {model_dynamic.params['treatment']:.3f} (p={model_dynamic.pvalues['treatment']:.3f})")
print(f"  treatment_interact (effect per week): {model_dynamic.params['treatment_interact']:.3f} (p={model_dynamic.pvalues['treatment_interact']:.3f})")

# Summary statistics for the board
print("\n\n=== Summary Statistics for Board ===")
print(f"Total observations: {len(df)}")
print(f"Number of branches: {df['branch_id'].nunique()}")
print(f"Branches that switched: {df['switched_branch'].sum()}")
print(f"Branches that did not switch: {(df['switched_branch'] == 0).sum()}")
print(f"Weeks of data: {df['week'].min()} to {df['week'].max()}")
print(f"\nAverage service score (all): {df['service_score'].mean():.1f}")
print(f"Average service score (treatment, before): {df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean():.1f}")
print(f"Average service score (treatment, after): {df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean():.1f}")
print(f"Average service score (control, before): {df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean():.1f}")
print(f"Average service score (control, after): {df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean():.1f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Two-way fixed effects: branch + week
# Create week dummies
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df_week = pd.concat([df, week_dummies], axis=1)

# Build formula with branch and week fixed effects
branch_cols = [f'C(branch_id)']
week_cols = [col for col in df_week.columns if col.startswith('week_')]
formula = 'service_score ~ ' + ' + '.join(branch_cols) + ' + ' + ' + '.join(week_cols) + ' + treatment + staffing_index + visits_index'

model_twfe = ols(formula, data=df_week).fit()
print("=== Two-Way Fixed Effects (Branch + Week) ===")
print(f"treatment coefficient: {model_twfe.params['treatment']:.3f}")
print(f"treatment p-value: {model_twfe.pvalues['treatment']:.4f}")
print(f"treatment 95% CI: [{model_twfe.conf_int().loc['treatment'][0]:.2f}, {model_twfe.conf_int().loc['treatment'][1]:.2f}]")

# Try with cluster-robust SEs on the TWFE model
try:
    cluster_se_twfe = cov_cluster(model_twfe, df_week['branch_id'])
    treatment_se_twfe = np.sqrt(cluster_se_twfe.loc['treatment', 'treatment'])
    treatment_coef_twfe = model_twfe.params['treatment']
    treatment_t_twfe = treatment_coef_twfe / treatment_se_twfe
    from scipy import stats
    treatment_p_twfe = 2 * (1 - stats.t.cdf(abs(treatment_t_twfe), df=len(df['branch_id'].unique()) - 1))
    print(f"\nWith cluster-robust SEs:")
    print(f"treatment coefficient: {treatment_coef_twfe:.3f}")
    print(f"cluster-robust SE: {treatment_se_twfe:.3f}")
    print(f"p-value: {treatment_p_twfe:.4f}")
except Exception as e:
    print(f"Error with cluster SEs: {e}")

# Also check: what if we just compare the change in treatment vs control more simply
print("\n\n=== Simple Comparison ===")
# Calculate branch-level changes
branch_before = df[df['after_switch']==0].groupby(['branch_id', 'switched_branch'])['service_score'].mean().unstack()
branch_after = df[df['after_switch']==1].groupby(['branch_id', 'switched_branch'])['service_score'].mean().unstack()

treatment_branches_change = branch_after[1] - branch_before[1]
control_branches_change = branch_after[0] - branch_before[0]

print(f"Average change in treatment branches: {treatment_branches_change.mean():.2f}")
print(f"Average change in control branches: {control_branches_change.mean():.2f}")
print(f"DiD estimate (branch-level): {treatment_branches_change.mean() - control_branches_change.mean():.2f}")

# T-test on branch-level changes
from scipy import stats
t_stat, p_val = stats.ttest_ind(treatment_branches_change, control_branches_change)
print(f"\nT-test on branch-level changes:")
print(f"t-statistic: {t_stat:.2f}")
print(f"p-value: {p_val:.4f}")

import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# Calculate branch-level changes properly
treatment_branches = df[df['switched_branch']==1]['branch_id'].unique()
control_branches = df[df['switched_branch']==0]['branch_id'].unique()

treatment_changes = []
control_changes = []

for bid in treatment_branches:
    branch_data = df[df['branch_id']==bid]
    before = branch_data[branch_data['after_switch']==0]['service_score'].mean()
    after = branch_data[branch_data['after_switch']==1]['service_score'].mean()
    treatment_changes.append(after - before)

for bid in control_branches:
    branch_data = df[df['branch_id']==bid]
    before = branch_data[branch_data['after_switch']==0]['service_score'].mean()
    after = branch_data[branch_data['after_switch']==1]['service_score'].mean()
    control_changes.append(after - before)

treatment_changes = np.array(treatment_changes)
control_changes = np.array(control_changes)

print("=== Branch-Level Analysis ===")
print(f"Number of treatment branches: {len(treatment_changes)}")
print(f"Number of control branches: {len(control_changes)}")
print(f"\nTreatment branches - mean change: {treatment_changes.mean():.2f}, std: {treatment_changes.std():.2f}")
print(f"Control branches - mean change: {control_changes.mean():.2f}, std: {control_changes.std():.2f}")
print(f"\nDiD estimate: {treatment_changes.mean() - control_changes.mean():.2f}")

# T-test
t_stat, p_val = stats.ttest_ind(treatment_changes, control_changes, equal_var=False)
print(f"\nWelch's t-test:")
print(f"t-statistic: {t_stat:.2f}")
print(f"p-value: {p_val:.4f}")

# 95% CI for the DiD
se_diff = np.sqrt(treatment_changes.var()/len(treatment_changes) + control_changes.var()/len(control_changes))
ci_lower = (treatment_changes.mean() - control_changes.mean()) - 1.96 * se_diff
ci_upper = (treatment_changes.mean() - control_changes.mean()) + 1.96 * se_diff
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")

# Visualize the distribution of changes
import matplotlib.pyplot as plt
plt.figure(figsize=(10, 5))
plt.hist(treatment_changes, alpha=0.7, label='Treatment Branches', bins=10, edgecolor='black')
plt.hist(control_changes, alpha=0.7, label='Control Branches', bins=10, edgecolor='black')
plt.axvline(x=treatment_changes.mean(), color='blue', linestyle='--', linewidth=2, label=f'Treatment mean: {treatment_changes.mean():.1f}')
plt.axvline(x=control_changes.mean(), color='orange', linestyle='--', linewidth=2, label=f'Control mean: {control_changes.mean():.1f}')
plt.xlabel('Change in Service Score (After - Before)')
plt.ylabel('Number of Branches')
plt.title('Distribution of Service Score Changes by Group')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('branch_changes.png', dpi=150)
print("\nPlot saved as branch_changes.png")
