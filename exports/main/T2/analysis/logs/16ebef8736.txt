import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Check how many branches switched vs didn't
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Check balance - weeks per branch
print("\nWeeks per branch (should be 17 for all):")
weeks_per_branch = df.groupby('branch_id')['week'].nunique()
print(weeks_per_branch.describe())

# Create a pivot to see the data structure better
print("\nService score summary by group and period:")
df['period'] = df['after_switch'].map({0: 'Before', 1: 'After'})
df['group'] = df['switched_branch'].map({0: 'Control', 1: 'Treatment'})

summary = df.groupby(['group', 'period'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Calculate the difference-in-differences manually
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n--- Difference-in-Differences Calculation ---")
print(f"Control group - Before: {before_control:.3f}, After: {after_control:.3f}, Change: {after_control - before_control:.3f}")
print(f"Treatment group - Before: {before_treatment:.3f}, After: {after_treatment:.3f}, Change: {after_treatment - before_treatment:.3f}")
did_effect = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDiD Effect (Treatment change - Control change): {did_effect:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run DiD regression with standard errors
# Basic DiD: service_score ~ switched_branch * after_switch
did_model = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Basic DiD Regression (clustered by branch) ===")
print(did_model.summary())

# Add controls (staffing_index, visits_index)
did_model_controls = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n=== DiD Regression with Controls ===")
print(did_model_controls.summary())

# Check pre-trends: look at treatment vs control before the switch
print("\n=== Pre-Trend Check (weeks 6-14 only) ===")
pre_period = df[df['after_switch'] == 0]
pre_trend_model = ols('service_score ~ switched_branch + week', data=pre_period).fit(cov_type='cluster', cov_kwds={'groups': pre_period['branch_id']})
print(pre_trend_model.summary())

# Also check mean service scores by week for both groups
print("\n=== Service Score by Week and Group ===")
weekly_means = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_means.columns = ['Control', 'Treatment']
weekly_means['Difference'] = weekly_means['Treatment'] - weekly_means['Control']
print(weekly_means)

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create a visualization-ready summary
weekly_means = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_means.columns = ['Control (No Switch)', 'Treatment (Switched)']

# Calculate confidence intervals for the plot
weekly_stats = df.groupby(['week', 'switched_branch'])['service_score'].agg(['mean', 'std', 'count'])
weekly_stats['se'] = weekly_stats['std'] / np.sqrt(weekly_stats['count'])
weekly_stats['ci_lower'] = weekly_stats['mean'] - 1.96 * weekly_stats['se']
weekly_stats['ci_upper'] = weekly_stats['mean'] + 1.96 * weekly_stats['se']

print("Weekly statistics with confidence intervals:")
print(weekly_stats)

# Effect size as percentage
avg_score = df['service_score'].mean()
did_effect = 2.11
print(f"\nAverage service score across all data: {avg_score:.2f}")
print(f"DiD effect: {did_effect:.2f} points")
print(f"Effect as % of average: {(did_effect/avg_score)*100:.1f}%")

# Check if the effect is consistent across branches
branch_effects = df.pivot_table(index='branch_id', columns='after_switch', values='service_score', aggfunc='mean')
branch_effects['change'] = branch_effects[1] - branch_effects[0]
branch_effects['switched'] = df.groupby('branch_id')['switched_branch'].first()

treatment_changes = branch_effects[branch_effects['switched'] == 1]['change']
control_changes = branch_effects[branch_effects['switched'] == 0]['change']

print(f"\nTreatment branches (n={len(treatment_changes)}):")
print(f"  Mean change: {treatment_changes.mean():.2f}")
print(f"  Std dev: {treatment_changes.std():.2f}")
print(f"  Median change: {treatment_changes.median():.2f}")
print(f"  % with positive change: {(treatment_changes > 0).mean()*100:.1f}%")

print(f"\nControl branches (n={len(control_changes)}):")
print(f"  Mean change: {control_changes.mean():.2f}")
print(f"  Std dev: {control_changes.std():.2f}")
print(f"  Median change: {control_changes.median():.2f}")
print(f"  % with positive change: {(control_changes > 0).mean()*100:.1f}%")

import pandas as pd

# Create the trend plot

ax.plot(weekly_means.index, weekly_means[0], 'o-', label='Control (No Switch)', linewidth=2, markersize=6, color='#2E86AB')

ax.set_ylabel('Service Score', fontsize=12)
