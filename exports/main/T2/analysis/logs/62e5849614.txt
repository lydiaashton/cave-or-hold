import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

# Load the data
df = pd.read_csv('branches.csv')
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch fixed effects
# Using OLS with cluster-robust standard errors at branch level
model = smf.ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== DiD Regression Results ===")
print(model.summary())

# Also run with branch fixed effects
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = smf.ols('service_score ~ C(branch_id_str) + after_switch + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n=== DiD Regression with Branch Fixed Effects ===")
# Just show the key coefficients
print("\nKey coefficients:")
print(f"after_switch: {model_fe.params['after_switch']:.3f} (SE: {model_fe.bse['after_switch']:.3f})")
print(f"treated_post: {model_fe.params['treated_post']:.3f} (SE: {model_fe.bse['treated_post']:.3f})")

# Calculate p-value for treated_post
from scipy import stats
t_stat = model_fe.params['treated_post'] / model_fe.bse['treated_post']
df_resid = model_fe.df_resid
p_value = 2 * (1 - stats.t.cdf(abs(t_stat), df_resid))
print(f"t-statistic: {t_stat:.3f}, p-value: {p_value:.4f}")

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

# Load the data
df = pd.read_csv('branches.csv')
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Pre-trend test: interact treatment with week dummies for pre-period only
# Create week dummies for weeks 6-14 (pre-period)
pre_period_df = df[df['week'] < 15].copy()

# Test if there are differential pre-trends
# Regress service_score on treatment, week, and their interaction (pre-period only)
pre_period_df['week_centered'] = pre_period_df['week'] - pre_period_df['week'].mean()
pre_period_df['treat_week_interact'] = pre_period_df['switched_branch'] * pre_period_df['week_centered']

model_pre = smf.ols('service_score ~ switched_branch + week_centered + treat_week_interact', data=pre_period_df).fit(cov_type='cluster', cov_kwds={'groups': pre_period_df['branch_id']})
print("=== Pre-Trend Test (Weeks 6-14) ===")
print(f"Interaction coefficient (treat_week_interact): {model_pre.params['treat_week_interact']:.3f}")
print(f"Standard error: {model_pre.bse['treat_week_interact']:.3f}")
t_stat = model_pre.params['treat_week_interact'] / model_pre.bse['treat_week_interact']
from scipy import stats
p_value = 2 * (1 - stats.t.cdf(abs(t_stat), model_pre.df_resid))
print(f"t-statistic: {t_stat:.3f}, p-value: {p_value:.3f}")
print("Interpretation: Non-significant interaction suggests parallel pre-trends")

# Main DiD with controls
print("\n=== DiD with Controls (staffing_index, visits_index) ===")
model_controls = smf.ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"treated_post coefficient: {model_controls.params['treated_post']:.3f}")
print(f"Standard error: {model_controls.bse['treated_post']:.3f}")
t_stat = model_controls.params['treated_post'] / model_controls.bse['treated_post']
p_value = 2 * (1 - stats.t.cdf(abs(t_stat), model_controls.df_resid))
print(f"t-statistic: {t_stat:.3f}, p-value: {p_value:.4f}")

# Effect size relative to baseline
baseline = df[df['after_switch'] == 0]['service_score'].mean()
effect_size = model_controls.params['treated_post']
print(f"\n=== Effect Size ===")
print(f"Baseline mean service score (pre-period): {baseline:.2f}")
print(f"Estimated effect: {effect_size:.2f} points")
print(f"Relative effect: {effect_size/baseline*100:.1f}%")

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf
from scipy import stats

# Load the data
df = pd.read_csv('branches.csv')
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Final model with branch fixed effects
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = smf.ols('service_score ~ C(branch_id_str) + after_switch + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Key statistics
did_estimate = model_fe.params['treated_post']
did_se = model_fe.bse['treated_post']
t_stat = did_estimate / did_se
df_resid = model_fe.df_resid
p_value = 2 * (1 - stats.t.cdf(abs(t_stat), df_resid))

# Confidence interval
ci_lower = did_estimate - 1.96 * did_se
ci_upper = did_estimate + 1.96 * did_se

# Baseline stats
baseline = df[df['after_switch'] == 0]['service_score'].mean()
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treated_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treated_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"DiD Estimate: {did_estimate:.2f}")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {p_value:.4f}")
print(f"\nControl branches: {control_before:.2f} -> {control_after:.2f} (change: {control_after-control_before:.2f})")
print(f"Treatment branches: {treated_before:.2f} -> {treated_after:.2f} (change: {treated_after-treated_before:.2f})")
