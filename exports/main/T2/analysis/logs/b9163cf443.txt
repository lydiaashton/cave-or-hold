import pandas as pd
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate average service score by week and group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_avg.columns = ['Control (no switch)', 'Treated (switched)']

print("=== Weekly Average Service Scores ===")
print(weekly_avg)

# Create a plot
fig, ax = plt.subplots(figsize=(10, 6))
ax.plot(weekly_avg.index, weekly_avg['Control (no switch)'], marker='o', label='Control (no switch)', linewidth=2)
ax.plot(weekly_avg.index, weekly_avg['Treated (switched)'], marker='s', label='Treated (switched)', linewidth=2)

# Add vertical line at week 15 (switch point)
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch occurred')
ax.axvspan(14.5, 22.5, alpha=0.2, color='red', label='Post-switch period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Scores Over Time: Control vs Treated Branches', fontsize=14)
ax.legend(loc='best')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_scores_over_time.png', dpi=150)
print("\nPlot saved as 'service_scores_over_time.png'")

# Check parallel trends statistically (pre-treatment period only)
pre_df = df[df['week'] < 15].copy()
pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()
pre_df['treated'] = pre_df['switched_branch']  # In pre-period, just check if groups differ in trend

# Test for differential pre-trends
pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_df).fit()
pre_model_clustered = pre_model.get_robustcov_results(cov_type='cluster', groups=pre_df['branch_id'])
print("\n=== Pre-Treatment Parallel Trends Test ===")
print(pre_model_clustered.summary())

# The interaction term tests if trends were different before treatment
interaction_pvalue = pre_model_clustered.pvalues[3]  # switched_branch:week_centered
print(f"\nInteraction term p-value: {interaction_pvalue:.4f}")
if interaction_pvalue > 0.05:
    print("Conclusion: No evidence of differential pre-trends (parallel trends assumption holds)")
else:
    print("Conclusion: Evidence of differential pre-trends (parallel trends assumption may be violated)")

pre_df = df[df['week'] < 15].copy()
pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()

# Test for differential pre-trends
pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_df).fit()
pre_model_clustered = pre_model.get_robustcov_results(cov_type='cluster', groups=pre_df['branch_id'])
print("=== Pre-Treatment Parallel Trends Test ===")
print(pre_model_clustered.summary())

# The interaction term tests if trends were different before treatment
# Index 3 should be the interaction term
print(f"\nParameter names: {list(pre_model.params.index)}")
interaction_pvalue = pre_model_clustered.pvalues[3]  # switched_branch:week_centered
print(f"\nInteraction term p-value: {interaction_pvalue:.4f}")
if interaction_pvalue > 0.05:
    print("Conclusion: No evidence of differential pre-trends (parallel trends assumption holds)")
else:
    print("Conclusion: Evidence of differential pre-trends (parallel trends assumption may be violated)")
