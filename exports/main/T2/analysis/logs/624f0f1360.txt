import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Calculate average service scores by group and period
print("Average service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the difference-in-differences manually
# Before switch (after_switch=0)
before_switched = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()

# After switch (after_switch=1)
after_switched = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print(f"\nBefore switch:")
print(f"  Switched branches (treatment): {before_switched:.2f}")
print(f"  Non-switched branches (control): {before_control:.2f}")
print(f"  Difference: {before_switched - before_control:.2f}")

print(f"\nAfter switch:")
print(f"  Switched branches (treatment): {after_switched:.2f}")
print(f"  Non-switched branches (control): {after_control:.2f}")
print(f"  Difference: {after_switched - after_control:.2f}")

# Difference-in-differences estimate
did_estimate = (after_switched - before_switched) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")
print(f"  (Change in treatment) - (Change in control) = ({after_switched:.2f} - {before_switched:.2f}) - ({after_control:.2f} - {before_control:.2f})")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score ~ switched_branch * after_switch + branch fixed effects + week fixed effects
# This is the standard DiD specification

# Create the interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression without fixed effects first
model_simple = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Simple DiD Regression (no fixed effects):")
print(model_simple.summary())

print("\n" + "="*80 + "\n")

# DiD regression with branch fixed effects
# We need to convert branch_id to categorical
df['branch_id_cat'] = df['branch_id'].astype('category')
df['week_cat'] = df['week'].astype('category')

# Regression with branch fixed effects
model_branch_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id_cat)', data=df).fit()
print("DiD Regression with Branch Fixed Effects:")
print(model_branch_fe.summary())

print("\n" + "="*80 + "\n")

# Full model with branch and week fixed effects
model_full = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id_cat) + C(week_cat)', data=df).fit()
print("DiD Regression with Branch and Week Fixed Effects:")
print(model_full.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']
df['branch_id_cat'] = df['branch_id'].astype('category')
df['week_cat'] = df['week'].astype('category')

# Full model with branch and week fixed effects
model_full = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id_cat) + C(week_cat)', data=df).fit()

# Get just the key coefficients
print("Key coefficients from full DiD model (with branch and week fixed effects):")
print("="*80)
key_vars = ['switched_branch', 'after_switch', 'treatment']
for var in key_vars:
    coef = model_full.params[var]
    se = model_full.bse[var]
    t = model_full.tvalues[var]
    p = model_full.pvalues[var]
    ci_low = model_full.conf_int().loc[var, 0]
    ci_high = model_full.conf_int().loc[var, 1]
    print(f"{var}: coef={coef:.3f}, SE={se:.3f}, t={t:.2f}, p={p:.4f}, 95% CI=[{ci_low:.3f}, {ci_high:.3f}]")

print("\n" + "="*80)
print("\nNow with cluster-robust standard errors (clustered by branch):")
# Cluster-robust standard errors
cov_clustered = model_full.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
for var in key_vars:
    coef = cov_clustered.params[var]
    se = cov_clustered.bse[var]
    t = coef / se
    # Approximate p-value using normal distribution
    from scipy import stats
    p = 2 * (1 - stats.norm.cdf(abs(t)))
    ci_low = coef - 1.96 * se
    ci_high = coef + 1.96 * se
    print(f"{var}: coef={coef:.3f}, SE={se:.3f}, t={t:.2f}, p={p:.4f}, 95% CI=[{ci_low:.3f}, {ci_high:.3f}]")
