import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the treatment effect

df['interaction'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + interaction', data=df).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Add branch fixed effects (using branch_id as categorical)
# This controls for time-invariant branch characteristics
model_fe = ols('service_score ~ C(branch_id) + after_switch + interaction', data=df).fit()
print("\n=== DiD with Branch Fixed Effects ===")
# Just show the key coefficients
print(model_fe.params[['after_switch', 'interaction']])
print("\nStandard errors:")
print(model_fe.bse[['after_switch', 'interaction']])

# Also add week fixed effects to control for common time trends
model_full = ols('service_score ~ C(branch_id) + C(week) + interaction', data=df).fit()
print("\n=== DiD with Branch and Week Fixed Effects ===")
print(model_full.params[['interaction']])
print("\nStandard errors:")
print(model_full.bse[['interaction']])
print("\nConfidence interval for interaction term:")
conf_int = model_full.conf_int().loc['interaction']
print(f"95% CI: [{conf_int[0]:.3f}, {conf_int[1]:.3f}]")
