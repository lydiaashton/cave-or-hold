import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch*after_switch) + error
# beta3 is the DiD estimator - the effect of the new procedure

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Difference-in-Differences Regression Results:")
print("=" * 60)
print(model.summary())

# Also run with cluster-robust standard errors at branch level
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nWith Cluster-Robust Standard Errors (clustered by branch):")
print("=" * 60)
print(model_clustered.summary())

import pandas as pd
import numpy as np
from linearmodels.panel import PanelOLS

# Load the data
df = pd.read_csv('branches.csv')

# Create interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Set up panel data
df_panel = df.copy()
df_panel = df_panel.set_index(['branch_id', 'week'])

# Fixed effects model - switched_branch is absorbed, but treatment and after_switch remain
fe_model = PanelOLS(df_panel['service_score'], 
                    df_panel[['after_switch', 'treatment', 'staffing_index', 'visits_index']],
                    entity_effects=True)
fe_results = fe_model.fit(cov_type='clustered', cluster_entity=True)
print("Fixed Effects Model (with branch fixed effects):")
print("=" * 60)
print(fe_results)

# Also run time fixed effects to control for common time trends
fe_time_model = PanelOLS(df_panel['service_score'], 
                         df_panel[['after_switch', 'treatment', 'staffing_index', 'visits_index']],
                         entity_effects=True, time_effects=True)
fe_time_results = fe_time_model.fit(cov_type='clustered', cluster_entity=True)
print("\n\nFixed Effects Model (with branch AND time fixed effects):")
print("=" * 60)
print(fe_time_results)

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create a clearer visualization for the board
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Left panel: Average scores by group and period
group_means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
group_means.index = ['Control (Old Procedure)', 'Treatment (New Procedure)']
group_means.columns = ['Before (Weeks 6-14)', 'After (Weeks 15-22)']

ax1 = axes[0]
group_means.plot(kind='bar', ax=ax1, color=['#3498db', '#e74c3c'], width=0.7)
ax1.set_xlabel('')
ax1.set_ylabel('Average Service Score')
ax1.set_title('Service Scores: Before vs After Switch', fontsize=12, fontweight='bold')
ax1.legend(loc='upper right')
ax1.grid(axis='y', alpha=0.3)

# Add value labels
for i, (idx, row) in enumerate(group_means.iterrows()):
    for j, val in enumerate(row):
        ax1.text(i + j*0.35 - 0.1, val + 0.3, f'{val:.1f}', ha='center', va='bottom', fontsize=10)

# Right panel: Weekly trends
ax2 = axes[1]
for group, label, color in [(0, 'Control (Old Procedure)', '#3498db'), 
                             (1, 'Treatment (New Procedure)', '#e74c3c')]:
    weekly_avg = df[df['switched_branch'] == group].groupby('week')['service_score'].mean()
    ax2.plot(weekly_avg.index, weekly_avg.values, marker='o', label=label, linewidth=2.5, color=color)

ax2.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch Point')
ax2.axvspan(14.5, 22, alpha=0.1, color='green', label='Post-Switch Period')
ax2.set_xlabel('Week')
ax2.set_ylabel('Average Service Score')
ax2.set_title('Weekly Service Score Trends', fontsize=12, fontweight='bold')
ax2.legend(loc='upper right')
ax2.grid(alpha=0.3)
ax2.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('board_summary_plot.png', dpi=150, bbox_inches='tight')
print("Board summary plot saved as board_summary_plot.png")

# Calculate key statistics for the report
control_pre = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_post = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_pre = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_post = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

control_change = control_post - control_pre
treatment_change = treatment_post - treatment_pre
did_effect = treatment_change - control_change

print(f"\n=== KEY STATISTICS ===")
print(f"Control group (old procedure):")
print(f"  Before: {control_pre:.2f}")
print(f"  After: {control_post:.2f}")
print(f"  Change: {control_change:.2f}")
print(f"\nTreatment group (new procedure):")
print(f"  Before: {treatment_pre:.2f}")
print(f"  After: {treatment_post:.2f}")
print(f"  Change: {treatment_change:.2f}")
print(f"\nDifference-in-Differences Effect: {did_effect:.2f} points")
print(f"(This is the estimated impact of the new procedure)")
