import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create relative week variable (weeks relative to switch at week 15)
df['relative_week'] = df['week'] - 15

# Create week dummies interacted with treatment
# We'll use week 14 as the reference (last pre-switch week)
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=False)
df = pd.concat([df, week_dummies], axis=1)

# Create interaction terms for each week with switched_branch
for week in range(6, 23):
    df[f'week{week}_treated'] = df[f'week_{week}'] * df['switched_branch']

# Run event study regression (excluding week 14 as reference)
# We need to construct the formula dynamically
control_weeks = [f'week_{w}' for w in range(6, 23) if w != 14]
treatment_weeks = [f'week{w}_treated' for w in range(6, 23) if w != 14]

# Build formula
formula = 'service_score ~ switched_branch + ' + ' + '.join(control_weeks) + ' + ' + ' + '.join(treatment_weeks)

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for treatment weeks
treatment_coefs = {}
treatment_se = {}
for week in range(6, 23):
    if week != 14:
        coef_name = f'week{week}_treated'
        if coef_name in model_event.params:
            treatment_coefs[week] = model_event.params[coef_name]
            treatment_se[week] = model_event.bse[coef_name]

print("=== Event Study Results (Treatment Effect by Week) ===")
print("Week 14 is the reference period (coefficient = 0)")
for week in sorted(treatment_coefs.keys()):
    coef = treatment_coefs[week]
    se = treatment_se[week]
    ci_low = coef - 1.96 * se
    ci_high = coef + 1.96 * se
    sig = "***" if se > 0 and abs(coef/se) > 2.58 else "**" if se > 0 and abs(coef/se) > 1.96 else "*" if se > 0 and abs(coef/se) > 1.64 else ""
    period = "PRE" if week < 15 else "POST"
    print(f"Week {week} ({period}): coef={coef:.2f}, SE={se:.2f}, 95% CI=[{ci_low:.2f}, {ci_high:.2f}] {sig}")

# Plot event study
fig, ax = plt.subplots(figsize=(14, 7))

weeks = sorted(treatment_coefs.keys())
coefs = [treatment_coefs[w] for w in weeks]
ses = [treatment_se[w] for w in weeks]
ci_low = [c - 1.96 * s for c, s in zip(coefs, ses)]
ci_high = [c + 1.96 * s for c, s in zip(coefs, ses)]

# Create colors - red for pre, green for post
colors = ['red' if w < 15 else 'green' for w in weeks]

ax.scatter(weeks, coefs, c=colors, s=100, zorder=5)
ax.plot(weeks, coefs, c='gray', linestyle='--', alpha=0.5)

# Add confidence intervals
for i, (w, c, low, high) in enumerate(zip(weeks, coefs, ci_low, ci_high)):
