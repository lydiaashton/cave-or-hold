import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Let's look at the trends more carefully
# Calculate week-by-week differences between treatment and control
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_avg['diff'] = weekly_avg[1] - weekly_avg[0]
print("Weekly averages and differences:")
print(weekly_avg)

# The diff should be relatively stable before week 15 if parallel trends hold
print("\nPre-period (weeks 6-14) average difference:", weekly_avg.loc[6:14, 'diff'].mean())
print("Post-period (weeks 15-22) average difference:", weekly_avg.loc[15:22, 'diff'].mean())

# Let's also check if the pre-trend is driven by specific weeks
# Plot the difference over time
plt.figure(figsize=(10, 5))
plt.plot(weekly_avg.index, weekly_avg['diff'], marker='o', linewidth=2)
plt.axvline(x=14.5, color='red', linestyle='--', label='Switch implemented (week 15)')
plt.axhline(y=weekly_avg.loc[6:14, 'diff'].mean(), color='blue', linestyle=':', label='Pre-period avg diff')
plt.axhline(y=weekly_avg.loc[15:22, 'diff'].mean(), color='green', linestyle=':', label='Post-period avg diff')
plt.xlabel('Week')
plt.ylabel('Treatment - Control (Service Score)')
plt.title('Weekly Difference: Switched vs Non-Switched Branches')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('diff_over_time.png', dpi=150)
print("\nDifference plot saved as diff_over_time.png")

# Run a more flexible event study model
df['relative_week'] = df['week'] - 15  # week 15 is period 0
df['relative_week'] = df['relative_week'].clip(-9, 7)  # cap at -9 and 7

# Create dummies for each relative week (omitting week -1 as reference)
for rw in range(-9, 8):
    if rw != -1:
        df[f'rel_week_{rw}'] = (df['relative_week'] == rw).astype(int)

# Event study regression
event_cols = [f'rel_week_{rw}' for rw in range(-9, 8) if rw != -1]
event_formula = 'service_score ~ switched_branch + ' + ' + '.join(event_cols)
# Also add interactions
event_formula_full = 'service_score ~ switched_branch + after_switch + '
for rw in range(-9, 8):
    if rw != -1:
        event_formula_full += f' + switched_branch:rel_week_{rw}'

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Final DiD analysis with robust results
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main model with controls
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Final DiD Model Results:")
print(model.summary().tables[1])

# Get the treatment effect and confidence interval
treatment_effect = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pvalue = model.pvalues['treatment']
ci_lower = model.conf_int().loc['treatment', 0]
ci_upper = model.conf_int().loc['treatment', 1]

print(f"\n\n=== KEY RESULTS ===")
print(f"Treatment effect (new procedure): {treatment_effect:.2f} points")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {treatment_pvalue:.4f}")
print(f"Statistically significant: {treatment_pvalue < 0.05}")

# Calculate percentage improvement relative to baseline
baseline = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
pct_improvement = (treatment_effect / baseline) * 100
print(f"\nBaseline average (switched branches, before): {baseline:.2f}")
print(f"Percentage improvement: {pct_improvement:.1f}%")
