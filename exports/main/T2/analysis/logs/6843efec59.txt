import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create treatment indicator
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run difference-in-differences regression
# Basic DiD model
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Basic DiD Regression:")
print(model.summary())

# Add branch fixed effects (more rigorous)
# Create dummy variables for branches
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = df.copy()
for col in branch_dummies.columns:
    df_with_fe[col] = branch_dummies[col]

# Build formula with fixed effects
fe_cols = [col for col in df_with_fe.columns if col.startswith('branch_')]
formula = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(fe_cols)
model_fe = ols(formula, data=df_with_fe).fit()
print("\n\nDiD with Branch Fixed Effects:")
print(f"Treatment coefficient: {model_fe.params['treatment']:.4f}")
print(f"Treatment p-value: {model_fe.pvalues['treatment']:.4f}")
print(f"Treatment 95% CI: {model_fe.conf_int().loc['treatment'].values}")

# Also add week fixed effects for even more rigor
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df_with_all_fe = df.copy()
for col in branch_dummies.columns:
    df_with_all_fe[col] = branch_dummies[col]
for col in week_dummies.columns:
    df_with_all_fe[col] = week_dummies[col]

fe_cols_all = [col for col in df_with_all_fe.columns if col.startswith('branch_') or col.startswith('week_')]
formula_all = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(fe_cols_all)
model_all_fe = ols(formula_all, data=df_with_all_fe).fit()
print("\n\nDiD with Branch and Week Fixed Effects:")
print(f"Treatment coefficient: {model_all_fe.params['treatment']:.4f}")
print(f"Treatment p-value: {model_all_fe.pvalues['treatment']:.4f}")
print(f"Treatment 95% CI: {model_all_fe.conf_int().loc['treatment'].values}")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Check if staffing and visits changed differently for treatment vs control
print("Did staffing_index change differently for switched branches?")
staffing_model = ols('staffing_index ~ switched_branch + after_switch + treatment', data=df).fit()
print(f"Treatment effect on staffing: {staffing_model.params['treatment']:.4f} (p={staffing_model.pvalues['treatment']:.4f})")

print("\nDid visits_index change differently for switched branches?")
visits_model = ols('visits_index ~ switched_branch + after_switch + treatment', data=df).fit()
print(f"Treatment effect on visits: {visits_model.params['treatment']:.4f} (p={visits_model.pvalues['treatment']:.4f})")

# Add controls to main model
print("\n\nDiD with controls (staffing and visits):")
controls_model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print(controls_model.summary())

# Check for heterogeneous effects (does the effect vary by branch characteristics?)
# For now, just note the consistency
print("\n\nEffect consistency check:")
# Calculate effect for each switched branch
switched_branches = df[df['switched_branch']==1]['branch_id'].unique()
branch_effects = []
for bid in switched_branches:
    branch_data = df[df['branch_id']==bid]
    before = branch_data[branch_data['after_switch']==0]['service_score'].mean()
    after = branch_data[branch_data['after_switch']==1]['service_score'].mean()
    branch_effects.append(after - before)

print(f"Average within-branch change for switched branches: {np.mean(branch_effects):.2f}")
print(f"Std dev of within-branch changes: {np.std(branch_effects):.2f}")
print(f"Min change: {np.min(branch_effects):.2f}, Max change: {np.max(branch_effects):.2f}")
print(f"Percent of switched branches with positive change: {np.mean(np.array(branch_effects) > 0)*100:.1f}%")
