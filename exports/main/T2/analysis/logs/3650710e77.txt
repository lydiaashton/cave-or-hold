import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = β0 + β1*switched_branch + β2*after_switch + β3*(switched_branch * after_switch) + ε
# β3 is the DiD estimator - the effect of the new procedure

df['treatment_interaction'] = df['switched_branch'] * df['after_switch']

# Basic DiD model
model1 = ols('service_score ~ switched_branch + after_switch + treatment_interaction', data=df).fit()
print("Basic Difference-in-Differences Regression:")
print(model1.summary())

# Model with branch fixed effects (to control for time-invariant branch characteristics)
print("\n\n" + "="*80)
print("Model with Branch Fixed Effects:")
model2 = ols('service_score ~ switched_branch + after_switch + treatment_interaction + C(branch_id)', data=df).fit()
print(model2.summary())

# Model with additional controls (staffing and visits)
print("\n\n" + "="*80)
print("Model with Branch Fixed Effects and Controls:")
model3 = ols('service_score ~ switched_branch + after_switch + treatment_interaction + staffing_index + visits_index + C(branch_id)', data=df).fit()
print(model3.summary())
