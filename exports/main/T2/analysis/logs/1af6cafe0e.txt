import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch'] * df['after_switch']

# Summary table: mean service scores by group and period
summary_table = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).unstack()
print("Service Score by Group and Period:")
print(summary_table)
print()

# Calculate the DiD manually
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
before_treated = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_treated = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print("Mean Service Scores:")
print(f"Control (non-switched) - Before: {before_control:.3f}, After: {after_control:.3f}, Change: {after_control - before_control:.3f}")
print(f"Treated (switched) - Before: {before_treated:.3f}, After: {after_treated:.3f}, Change: {after_treated - before_treated:.3f}")
print(f"\nDifference-in-Differences Estimate: {(after_treated - before_treated) - (after_control - before_control):.3f}")
print()

# Run DiD regression
model = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
print("DiD Regression Results (without controls):")
print(model.summary())
print()

# Run DiD regression with controls
model_controls = ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', data=df).fit()
print("DiD Regression Results (with controls):")
print(model_controls.summary())
