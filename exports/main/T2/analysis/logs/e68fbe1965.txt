import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# === BASIC DiD CALCULATION ===
print("=== AVERAGE SERVICE SCORES ===")
# Before period (weeks 6-14), After period (weeks 15-22)
before = df[df['after_switch']==0]
after = df[df['after_switch']==1]

# Treatment group (switched_branch=1)
treat_before = before[before['switched_branch']==1]['service_score'].mean()
treat_after = after[after['switched_branch']==1]['service_score'].mean()

# Control group (switched_branch=0)
ctrl_before = before[before['switched_branch']==0]['service_score'].mean()
ctrl_after = after[after['switched_branch']==0]['service_score'].mean()

print(f"\nTreatment Group (switched branches):")
print(f"  Before (weeks 6-14): {treat_before:.2f}")
print(f"  After (weeks 15-22): {treat_after:.2f}")
print(f"  Change: {treat_after - treat_before:.2f}")

print(f"\nControl Group (non-switched branches):")
print(f"  Before (weeks 6-14): {ctrl_before:.2f}")
print(f"  After (weeks 15-22): {ctrl_after:.2f}")
print(f"  Change: {ctrl_after - ctrl_before:.2f}")

# DiD estimate
did_estimate = (treat_after - treat_before) - (ctrl_after - ctrl_before)
print(f"\n=== DIFFERENCE-IN-DIFFERENCES ESTIMATE ===")
print(f"DiD Effect: {did_estimate:.2f}")

# === REGRESSION DiD ===
print("\n=== REGRESSION DiD MODEL ===")
model = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
print(model.summary())

# Cluster-robust standard errors by branch
print("\n=== CLUSTER-ROBUST SE (by branch) ===")
model_cluster = ols('service_score ~ switched_branch + after_switch + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# === PARALLEL TRENDS CHECK ===
# Calculate average service score by week and group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_avg.columns = ['Control', 'Treatment']
weekly_avg['Diff'] = weekly_avg['Treatment'] - weekly_avg['Control']

print("=== WEEKLY AVERAGES BY GROUP ===")
print(weekly_avg)

# === PRE-TREND ANALYSIS ===
# Check if there's a significant trend difference before the switch
pre_data = df[df['week'] < 15].copy()
pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()

# Test for parallel pre-trends
pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print("\n=== PRE-TREND TEST (interaction should be non-significant) ===")
print(pre_model.summary())

# === VISUALIZATION ===
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Plot 1: Service scores over time by group
ax1 = axes[0]
for group in [0, 1]:
    group_data = df[df['switched_branch']==group].groupby('week')['service_score'].mean()
    label = 'Treatment (Switched)' if group == 1 else 'Control (Not Switched)'
    ax1.plot(group_data.index, group_data.values, marker='o', label=label, linewidth=2)

ax1.axvline(x=14.5, color='red', linestyle='--', label='Switch Week (15)')
ax1.set_xlabel('Week')
ax1.set_ylabel('Average Service Score')
ax1.set_title('Service Scores Over Time by Group')
ax1.legend()
ax1.grid(alpha=0.3)

# Plot 2: Difference between treatment and control over time
ax2 = axes[1]
ax2.plot(weekly_avg.index, weekly_avg['Diff'], marker='s', linewidth=2, color='purple')
ax2.axvline(x=14.5, color='red', linestyle='--', label='Switch Week (15)')
ax2.axhline(y=0, color='gray', linestyle='-', alpha=0.5)
ax2.set_xlabel('Week')
ax2.set_ylabel('Treatment - Control Difference')
ax2.set_title('Difference in Service Scores Over Time')
ax2.legend()
ax2.grid(alpha=0.3)

plt.tight_layout()
plt.savefig('service_scores_analysis.png', dpi=150)
print("\nPlot saved as 'service_scores_analysis.png'")

import pandas as pd
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']

# === MODEL WITH COVARIATES ===
print("=== DiD WITH COVARIATES (staffing_index, visits_index) ===")
model_cov = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', 
                data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_cov.summary())

# === EVENT STUDY / DYNAMIC EFFECTS ===
# Create week dummies relative to switch
df['relative_week'] = df['week'] - 15  # Week 15 is the switch (0), before is negative, after is positive

# Create indicators for each relative week
# Omit week -1 as reference
for w in range(-9, 8):  # weeks -9 to -1, 0 to 7 (relative to week 15)
    if w != -1:  # omit reference category
        df[f'week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
event_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
event_formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)
print("\n=== EVENT STUDY MODEL ===")
event_model = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the relative week indicators
coef_table = event_model.params
se_table = event_model.bse

print("\nEvent Study Coefficients (Treatment x Relative Week):")
print("-" * 50)
for w in range(-9, 8):
    if w != -1:
        var_name = f'week_{w}'
        coef = coef_table[var_name]
        se = se_table[var_name]
        pval = event_model.pvalues[var_name]
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        period = "Pre" if w < 0 else "Post"
        print(f"Week {w:3d} ({period:4s}): coef={coef:6.2f}, SE={se:6.2f}, p={pval:.3f} {sig}")

import pandas as pd
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']

# === EVENT STUDY / DYNAMIC EFFECTS ===
# Create week dummies relative to switch
df['relative_week'] = df['week'] - 15  # Week 15 is the switch (0), before is negative, after is positive

# Create indicators for each relative week with proper variable names
# Omit week -1 as reference
event_vars = []
for w in range(-9, 8):  # weeks -9 to -1, 0 to 7 (relative to week 15)
    if w != -1:  # omit reference category
        var_name = f'rw_{abs(w)}_neg' if w < 0 else f'rw_{w}_pos'
        df[var_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_vars.append(var_name)

# Run event study regression
event_formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)
print("=== EVENT STUDY MODEL ===")
event_model = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the relative week indicators
coef_table = event_model.params
se_table = event_model.bse

print("\nEvent Study Coefficients (Treatment x Relative Week, ref=week -1):")
print("-" * 60)
for w in range(-9, 8):
    if w != -1:
        var_name = f'rw_{abs(w)}_neg' if w < 0 else f'rw_{w}_pos'
        coef = coef_table[var_name]
        se = se_table[var_name]
        pval = event_model.pvalues[var_name]
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        period = "Pre" if w < 0 else "Post"
        print(f"Week {w:3d} ({period:4s}): coef={coef:6.2f}, SE={se:6.2f}, p={pval:.3f} {sig}")

# Create event study plot
import matplotlib.pyplot as plt

weeks = [w for w in range(-9, 8) if w != -1]
coefs = []
ses = []

    coefs.append(coef_table[var_name])
    ses.append(se_table[var_name])

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create relative week (0 = week 15, the first post-switch week)
df['relative_week'] = df['week'] - 15

# Create treatment x relative week interactions, omitting period -1 as reference
# We need week fixed effects too
event_vars = []
for w in range(-9, 8):
    if w != -1:  # omit reference category
        var_name = f'treat_rw{w}'
        df[var_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_vars.append(var_name)

# Also create week dummies for time fixed effects
week_vars = []
for w in range(6, 23):
    if w != 14:  # omit week 14 as reference (last pre-switch week)
        var_name = f'week_{w}'
        df[var_name] = (df['week'] == w).astype(int)
        week_vars.append(var_name)

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import statsmodels.api as sm
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# === PROPER EVENT STUDY USING C() NOTATION ===
df['relative_week'] = df['week'] - 15

# Use C() for categorical variables
# This is cleaner than creating many dummy variables
print("=== EVENT STUDY WITH C() NOTATION ===")

# Create treatment indicator
df['treated'] = df['switched_branch']

# Event study formula with relative week as categorical, interacted with treatment
# Reference period is relative_week = -1 (week 14)
event_formula = 'service_score ~ C(week) + treated:C(relative_week)'
event_model = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get the coefficients for the interaction terms
params = event_model.params
se = event_model.bse
pvalues = event_model.pvalues

print("\nEvent Study Coefficients (Treatment x Relative Week):")
print("-" * 60)
results = []
for idx in params.index:
    if 'C(relative_week)' in idx and 'treated' in idx:
        # Parse the week number from the index
        # Format is something like "treated:C(relative_week)[T.-9]"
        try:
            week_str = idx.split('[T.')[1].rstrip(']')
            week = int(week_str)
            coef = params[idx]
            se_val = se[idx]
            pval = pvalues[idx]
            sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
            period = "Pre" if week < 0 else "Post"
            results.append({'week': week, 'coef': coef, 'se': se_val, 'pval': pval, 'period': period, 'sig': sig})
            print(f"Week {week:3d} ({period:4s}): coef={coef:6.2f}, SE={se_val:6.2f}, p={pval:.3f} {sig}")
        except:
            pass

# Sort results by week
results.sort(key=lambda x: x['week'])

# Create event study plot
weeks = [r['week'] for r in results]
coefs = [r['coef'] for r in results]
ses = [r['se'] for r in results]

fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(weeks, coefs, yerr=1.96*np.array(ses), fmt='o', capsize=5, markersize=8, linewidth=2)
ax.axvline(x=-0.5, color='red', linestyle='--', label='Switch (week 15)')
ax.axhline(y=0, color='gray', linestyle='-', alpha=0.5)
ax.set_xlabel('Relative Week (0 = first week after switch)')
ax.set_ylabel('Effect on Service Score')
ax.set_title('Event Study: Dynamic Treatment Effects\n(Reference: Week 14, last pre-switch week)')
ax.legend()
ax.grid(alpha=0.3)
plt.tight_layout()
plt.savefig('event_study_final.png', dpi=150)
print("\nEvent study plot saved as 'event_study_final.png'")

# Check pre-trends specifically
print("\n=== PRE-TREND TEST (joint test of pre-period coefficients) ===")
pre_var_names = [idx for idx in params.index if 'C(relative_week)' in idx and 'treated' in idx and '[T.-' in idx]
if pre_var_names:
    f_test = event_model.f_test('+'.join(pre_var_names))
    print(f"Joint F-test for pre-trend coefficients:")
    print(f"  F-statistic: {f_test.fvalue[0][0]:.2f}")
    print(f"  P-value: {f_test.pvalue:.4f}")

import pandas as pd
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']

# Final model with cluster-robust SEs
model = ols('service_score ~ switched_branch + after_switch + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get key statistics
did_coef = model.params['did']
did_se = model.bse['did']
did_pval = model.pvalues['did']
did_ci_lower = did_coef - 1.96 * did_se
did_ci_upper = did_coef + 1.96 * did_se

# Calculate effect size relative to baseline
baseline_score = df[df['after_switch']==0]['service_score'].mean()
effect_pct = (did_coef / baseline_score) * 100

print("=== FINAL SUMMARY STATISTICS ===")
print(f"DiD Estimate: {did_coef:.2f} points")
print(f"95% CI: [{did_ci_lower:.2f}, {did_ci_upper:.2f}]")
print(f"P-value: {did_pval:.3f}")
print(f"Baseline average score: {baseline_score:.2f}")
print(f"Effect as % of baseline: {effect_pct:.1f}%")

# Number of observations
print(f"\nTotal observations: {len(df)}")
print(f"Branches: {df['branch_id'].nunique()}")
print(f"  - Treatment (switched): {df['switched_branch'].sum() // 17}")
print(f"  - Control (not switched): {(1 - df['switched_branch']).sum() // 17}")
print(f"Weeks: {df['week'].nunique()} (weeks {df['week'].min()}-{df['week'].max()})")
