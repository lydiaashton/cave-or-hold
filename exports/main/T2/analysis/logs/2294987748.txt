import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression with robust standard errors
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='HC3')
print("=== Basic DiD Regression (Robust SE) ===")
print(f"Treatment coefficient: {model.params['treatment']:.3f}")
print(f"Std Error: {model.bse['treatment']:.3f}")
print(f"95% CI: [{model.conf_int().loc['treatment'][0]:.3f}, {model.conf_int().loc['treatment'][1]:.3f}]")
print(f"P-value: {model.pvalues['treatment']:.4f}")

# DiD with branch fixed effects - extract just key coefficients
print("\n=== DiD with Branch Fixed Effects ===")
model_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print(f"Std Error: {model_fe.bse['treatment']:.3f}")

print(f"P-value: {model_fe.pvalues['treatment']:.4f}")

# DiD with branch fixed effects and controls
print("\n=== DiD with Branch FE and Controls ===")
