before_summary = before.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print(before_summary)

print("\nAFTER switch (weeks 15-22):")
after_summary = after.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print(after_summary)

# Difference-in-differences calculation
print("\n" + "="*60)
print("Difference-in-Differences Calculation")

# Mean scores
switched_before = before[before['switched_branch'] == 1]['service_score'].mean()

control_before = before[before['switched_branch'] == 0]['service_score'].mean()
control_after = after[after['switched_branch'] == 0]['service_score'].mean()

print(f"\nSwitched branches:  Before={switched_before:.2f}, After={switched_after:.2f}, Change={switched_after - switched_before:.2f}")
print(f"Control branches:   Before={control_before:.2f}, After={control_after:.2f}, Change={control_after - control_before:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate

model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("="*60)
print("Difference-in-Differences Regression Results")
print("="*60)
print(model.summary())

# Also run with cluster-robust standard errors by branch
print("\n" + "="*60)
print("Regression with Cluster-Robust Standard Errors (by branch)")
print("="*60)
model_clustered = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_clustered.summary())

# Check if staffing and visits indices affect results
print("\n" + "="*60)
print("Regression with Controls (staffing_index, visits_index)")
print("="*60)
model_controls = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_controls.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create trend data by week and group
trend_data = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

# Plot trends
fig, ax = plt.subplots(figsize=(14, 7))

ax.plot(trend_data.index, trend_data[0], marker='o', linewidth=2.5, label='Control (Old Procedure)', 
        markersize=8, color='#2E86AB')
ax.plot(trend_data.index, trend_data[1], marker='s', linewidth=2.5, label='Treatment (New Procedure)', 
        markersize=8, color='#A23B72')

# Add vertical line for switch point
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2.5, label='Procedure Switch (Week 15)')

# Shade the after period
ax.axvspan(14.5, 22.5, alpha=0.15, color='green', label='Post-Switch Period')

ax.set_xlabel('Week', fontsize=13, fontweight='bold')
ax.set_ylabel('Average Service Score', fontsize=13, fontweight='bold')
ax.set_title('Library Branch Service Scores: Impact of New Shelving Procedure', 
             fontsize=15, fontweight='bold', pad=15)
ax.legend(fontsize=11, loc='upper left')
ax.grid(True, alpha=0.3, linestyle='-')
ax.set_xticks(range(6, 23))
ax.set_ylim(45, 58)

# Add annotation for the effect
ax.annotate('+2.05 pts\n(estimated effect)', xy=(18, 53.5), fontsize=11, fontweight='bold',
            bbox=dict(boxstyle='round', facecolor='lightgreen', alpha=0.7))

plt.tight_layout()
plt.savefig('service_score_trends_final.png', dpi=150)
print("Saved: service_score_trends_final.png")

# Calculate effect size in terms of standard deviations
overall_std = df['service_score'].std()
effect_size = 2.05 / overall_std
print(f"\nEffect size (Cohen's d): {effect_size:.2f} standard deviations")
print(f"Overall service score std dev: {overall_std:.2f}")

# Branch-level analysis: how many branches improved?
branch_data = df.pivot_table(index='branch_id', columns='after_switch', values='service_score', aggfunc='mean')
branch_data['change'] = branch_data[1] - branch_data[0]
branch_data['switched'] = df.groupby('branch_id')['switched_branch'].first()

switched_branches = branch_data[branch_data['switched'] == 1]['change']
control_branches = branch_data[branch_data['switched'] == 0]['change']

print(f"\nBranch-level changes:")
print(f"Switched branches (n={len(switched_branches)}): mean change = {switched_branches.mean():.2f}, std = {switched_branches.std():.2f}")
print(f"Control branches (n={len(control_branches)}): mean change = {control_branches.mean():.2f}, std = {control_branches.std():.2f}")
print(f"Switched branches that improved: {(switched_branches > 0).sum()} / {len(switched_branches)} ({(switched_branches > 0).mean()*100:.1f}%)")
print(f"Control branches that improved: {(control_branches > 0).sum()} / {len(control_branches)} ({(control_branches > 0).mean()*100:.1f}%)")
