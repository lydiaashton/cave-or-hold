import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Fixed effects model - drop switched_branch since it's collinear with branch FE
df['branch_id'] = df['branch_id'].astype(str)
df['week'] = df['week'].astype(str)

model_fe = ols('service_score ~ after_switch + treatment + C(branch_id) + C(week)', data=df).fit()
print("DiD Regression with Branch and Week Fixed Effects (proper specification):")
print(model_fe.summary())

# Cluster robust standard errors at branch level
from statsmodels.stats.sandwich_covariance import cov_cluster
cov_clustered = cov_cluster(model_fe, df['branch_id'].astype(int))
se_clustered = np.sqrt(np.diag(cov_clustered))
print("\n\nClustered Standard Errors (by branch):")
print(f"Treatment effect: {model_fe.params['treatment']:.3f}")
print(f"Clustered SE: {se_clustered[model_fe.params.index.get_loc('treatment')]:.3f}")
t_stat = model_fe.params['treatment'] / se_clustered[model_fe.params.index.get_loc('treatment')]
print(f"t-statistic: {t_stat:.3f}")

# Check pre-trends more carefully - event study approach
# Create relative week indicators
df['week_num'] = df['week'].astype(int)
df['relative_week'] = df['week_num'] - 15  # week 15 is the switch week (0)

# Create indicators for each week relative to switch, excluding the reference period
pre_data = df[df['week_num'] < 15].copy()
post_data = df[df['week_num'] >= 15].copy()

print("\n\n=== Pre-Trend Analysis ===")
print("Average service scores by week for each group (before switch):")
pre_pivot = pre_data.groupby(['switched_branch', 'week_num'])['service_score'].mean().unstack()
print(pre_pivot)

# Calculate week-over-week changes for each group before switch
print("\nWeek-over-week changes (before switch):")
for group in [0, 1]:
    group_data = pre_data[pre_data['switched_branch'] == group].groupby('week_num')['service_score'].mean()
    changes = group_data.diff()
    print(f"\nGroup {group} (switched={group}):")
    print(changes)

print(f"t-statistic: {t_stat:.3f}")

# Approximate p-value (two-tailed)
from scipy import stats
p_value = 2 * (1 - stats.t.cdf(abs(t_stat), df=len(df) - model_controls.df_model - 1))

ci_lower = model_controls.params['treatment'] - 1.96 * se_clustered[param_names.get_loc('treatment')]
