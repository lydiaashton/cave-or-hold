import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = β0 + β1*switched_branch + β2*after_switch + β3*(switched_branch * after_switch) + ε
# β3 is the treatment effect

df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Check pre-trends: compare trends before the switch
print("\n\n=== Pre-Trends Check (weeks 6-14) ===")
pre_data = df[df['after_switch'] == 0].copy()

# Regression with week as continuous variable, interacted with switch status
pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()
pre_data['switched_x_week'] = pre_data['switched_branch'] * pre_data['week_centered']

pre_model = ols('service_score ~ switched_branch + week_centered + switched_x_week', data=pre_data).fit()
print(pre_model.summary())

# Also check average scores by week for both groups
print("\n\n=== Average Service Score by Week and Group ===")
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(weekly_avg)

# Run DiD regression with cluster-robust standard errors
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

# Cluster-robust standard errors by branch
cov_type = 'cluster'
cov_kwds = {'groups': df['branch_id']}
model_clustered = model.get_robustcov_results(cov_type=cov_type, cov_kwds=cov_kwds)

print("=== DiD Regression with Cluster-Robust SE (by branch) ===")
print(model_clustered.summary())

# Also try clustering by week to account for time shocks
model_week_clustered = model.get_robustcov_results(cov_type='cluster', cov_kwds={'groups': df['week']})
print("\n\n=== DiD Regression with Cluster-Robust SE (by week) ===")
print(model_week_clustered.summary())

# Two-way clustering (branch and week) - using a simple approximation
# For proper two-way clustering we'd need a different approach
print("\n\n=== Branch Fixed Effects Model ===")
# Add branch fixed effects
df_with_dummies = pd.get_dummies(df, columns=['branch_id'], drop_first=True)
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_id_')]

import pandas as pd
import numpy as np
import statsmodels.api as sm
from linearmodels.panel import PanelOLS

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Set up panel data
df_indexed = df.set_index(['branch_id', 'week'])

# Event study / parallel trends check
print("=== Event Study / Parallel Trends Check ===")
df_indexed['relative_week'] = df_indexed.index.get_level_values('week') - 15  # week 15 is when switch happened

# Create indicators for each week relative to switch
for w in range(-9, 8):  # weeks -9 to +7 relative to week 15
    df_indexed[f'week_{w}'] = (df_indexed['relative_week'] == w).astype(int)
    df_indexed[f'treated_week_{w}'] = df_indexed['switched_branch'] * df_indexed[f'week_{w}']

# Run regression with week dummies (omitting week -1 as reference)
treated_weeks = [f'treated_week_{w}' for w in range(-9, 7) if w != -1]  # omit week -1
week_dummies = [f'week_{w}' for w in range(-9, 8) if w != -1]

exog_event = sm.add_constant(df_indexed[week_dummies + treated_weeks])
endog = df_indexed['service_score']

event_model = PanelOLS(endog, exog_event, entity_effects=True)
event_results = event_model.fit(cov_type='clustered', cluster_entity=True)

# Print coefficients for treated weeks
print("\nPre-treatment period (weeks 6-14, relative -9 to -1):")
print("(Reference period is week 14, relative -1)")
pre_trend_coefs = []
for w in range(-9, -1):
    coef = event_results.params.get(f'treated_week_{w}', np.nan)
    pval = event_results.pvalues.get(f'treated_week_{w}', np.nan)
    pre_trend_coefs.append(coef)
    print(f"  Week {w+15} (relative {w}): coef={coef:.2f}, p={pval:.3f}")

print("\nPost-treatment period (weeks 15-22, relative 0 to 7):")
post_trend_coefs = []
