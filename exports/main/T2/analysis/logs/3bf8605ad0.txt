import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Difference-in-Differences regression
# DID = after_switch * switched_branch interaction
df['did'] = df['after_switch'] * df['switched_branch']

# Basic DID regression
model = ols('service_score ~ after_switch + switched_branch + did', data=df).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Add branch fixed effects
print("\n=== DID with Branch Fixed Effects ===")
model_fe = ols('service_score ~ after_switch + switched_branch + did + C(branch_id)', data=df).fit()
print(model_fe.summary())

# Cluster-robust standard errors by branch
print("\n=== DID with Branch Fixed Effects (Cluster-Robust SE) ===")
model_fe_cluster = ols('service_score ~ after_switch + switched_branch + did + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

from scipy import stats

# Using the within transformation approach

# Branch-level aggregation for cleaner analysis

# Aggregate to branch level: before and after means
branch_before = df[df['after_switch']==0].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_before.columns = ['branch_id', 'switched_branch', 'score_before']

branch_after = df[df['after_switch']==1].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_after.columns = ['branch_id', 'switched_branch', 'score_after']

branch_agg = branch_before.merge(branch_after, on=['branch_id', 'switched_branch'])
branch_agg['change'] = branch_agg['score_after'] - branch_agg['score_before']

print("=== Branch-Level Analysis ===")
print(f"\nTreatment branches (n={len(branch_agg[branch_agg['switched_branch']==1])}):")
print(f"  Mean change: {branch_agg[branch_agg['switched_branch']==1]['change'].mean():.2f}")
print(f"  Std change: {branch_agg[branch_agg['switched_branch']==1]['change'].std():.2f}")

print(f"\nControl branches (n={len(branch_agg[branch_agg['switched_branch']==0])}):")
print(f"  Mean change: {branch_agg[branch_agg['switched_branch']==0]['change'].mean():.2f}")
print(f"  Std change: {branch_agg[branch_agg['switched_branch']==0]['change'].std():.2f}")

# T-test on the changes
treatment_change = branch_agg[branch_agg['switched_branch']==1]['change']
control_change = branch_agg[branch_agg['switched_branch']==0]['change']
t_stat, p_val = stats.ttest_ind(treatment_change, control_change)
print(f"\nT-test on changes (treatment vs control): t={t_stat:.2f}, p={p_val:.3f}")
print(f"Difference in changes: {treatment_change.mean() - control_change.mean():.2f}")

# Confidence interval for the difference
diff = treatment_change.mean() - control_change.mean()
se_diff = np.sqrt(treatment_change.var()/len(treatment_change) + control_change.var()/len(control_change))
ci_lower = diff - 1.96 * se_diff
ci_upper = diff + 1.96 * se_diff
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")

# Also run the FE model and extract the key coefficient
print("\n=== Fixed Effects Regression (Key Results) ===")
model_fe = ols('service_score ~ after_switch + switched_branch + did + C(branch_id)', data=df).fit()
did_coef = model_fe.params['did']
did_se = model_fe.bse['did']
did_pval = model_fe.pvalues['did']
print(f"DID coefficient: {did_coef:.2f}")
print(f"Standard error: {did_se:.2f}")
print(f"P-value: {did_pval:.3f}")
print(f"95% CI: [{did_coef - 1.96*did_se:.2f}, {did_coef + 1.96*did_se:.2f}]")
