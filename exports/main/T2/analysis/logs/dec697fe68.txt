import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Run regression with cluster-robust standard errors at branch level
model = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
# Get cluster-robust standard errors
cov_type = 'cluster'
cov_kwds = {'groups': df['branch_id']}
model_robust = model.get_robustcov_results(cov_type=cov_type, cov_kwds=cov_kwds)

print("DiD Regression with Cluster-Robust Standard Errors (clustered by branch):")
print("=" * 70)
print(f"{'Variable':<20} {'Coefficient':>12} {'Std Error':>12} {'P-value':>12}")
print("-" * 70)
for var in ['switched_branch', 'after_switch', 'did']:
    coef = model_robust.params[var]
    se = model_robust.bse[var]
    pval = model_robust.pvalues[var]
    print(f"{var:<20} {coef:>12.3f} {se:>12.3f} {pval:>12.4f}")

print(f"\nR-squared: {model.rsquared:.4f}")
print(f"Number of clusters (branches): {df['branch_id'].nunique()}")

# Check pre-trends: compare trends before the switch (weeks 6-14)
print("\n\nPre-Trend Analysis (weeks 6-14 only):")
pre_df = df[df['week'] < 15].copy()
pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()

# Check if trends were parallel before the switch
pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_df).fit()
print("\nTesting for differential pre-trends:")
print(f"Interaction coefficient (switched_branch:week_centered): {pre_model.params['switched_branch:week_centered']:.3f}")
print(f"P-value: {pre_model.pvalues['switched_branch:week_centered']:.4f}")
if pre_model.pvalues['switched_branch:week_centered'] > 0.05:
    print("=> No significant difference in pre-trends (parallel trends assumption holds)")
else:
    print("=> WARNING: Significant difference in pre-trends!")

# Visualize the trends
plt.figure(figsize=(10, 6))
for switched in [0, 1]:
    group_data = df[df['switched_branch'] == switched].groupby('week')['service_score'].mean()
    label = 'Treatment (switched)' if switched == 1 else 'Control (not switched)'
    plt.plot(group_data.index, group_data.values, marker='o', label=label, linewidth=2)

plt.axvline(x=14.5, color='red', linestyle='--', label='Switch point (week 15)')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Score Trends: Treatment vs Control Groups')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('trends_plot.png', dpi=150)
print("\nTrends plot saved as 'trends_plot.png'")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check pre-trends: compare trends before the switch (weeks 6-14)
print("Pre-Trend Analysis (weeks 6-14 only):")
pre_df = df[df['week'] < 15].copy()
pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()

# Check if trends were parallel before the switch
pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_df).fit()
print("\nTesting for differential pre-trends:")
print(f"Interaction coefficient (switched_branch:week_centered): {pre_model.params['switched_branch:week_centered']:.3f}")
print(f"P-value: {pre_model.pvalues['switched_branch:week_centered']:.4f}")
if pre_model.pvalues['switched_branch:week_centered'] > 0.05:
    print("=> No significant difference in pre-trends (parallel trends assumption holds)")
else:
    print("=> WARNING: Significant difference in pre-trends!")

# Event study: estimate effect for each week relative to switch
print("\n\nEvent Study Analysis:")
df['relative_week'] = df['week'] - 15  # week 15 is the switch point (0)

# Create week dummies relative to switch, excluding week -1 as reference
df['post_switch'] = (df['relative_week'] >= 0).astype(int)

# Run event study regression
event_cols = []
for rw in sorted(df['relative_week'].unique()):
    if rw != -1:  # reference category
        df[f'rel_week_{rw}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)
        event_cols.append(f'rel_week_{rw}')

# Also need main effects
event_formula = 'service_score ~ switched_branch + ' + ' + '.join([f'rel_week_{rw}' for rw in sorted(df['relative_week'].unique()) if rw != -1])
event_model = ols(event_formula, data=df).fit()

# Get cluster-robust SE
event_robust = event_model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("\nEvent Study Coefficients (Treatment vs Control, relative to week 14):")
print(f"{'Week':<10} {'Relative':<10} {'Coefficient':>12} {'Std Error':>12} {'P-value':>12}")
print("-" * 60)
for rw in sorted(df['relative_week'].unique()):
    if rw == -1:
        print(f"{'Week':<10} {rw:<10} {'(reference)':>12} {'':>12} {'':>12}")
    else:
        coef = event_robust.params[f'rel_week_{rw}']
        se = event_robust.bse[f'rel_week_{rw}']
        pval = event_robust.pvalues[f'rel_week_{rw}']
        marker = '*' if pval < 0.05 else ''
        print(f"{'Week':<10} {rw:<10} {coef:>12.2f} {se:>12.2f} {pval:>12.3f}{marker}")

# Visualize event study
plt.figure(figsize=(12, 6))
weeks = []
coefs = []
ses = []
for rw in sorted(df['relative_week'].unique()):
    if rw != -1:
        weeks.append(rw)
        coefs.append(event_robust.params[f'rel_week_{rw}'])
        ses.append(event_robust.bse[f'rel_week_{rw}'])

plt.errorbar(weeks, coefs, yerr=1.96*np.array(ses), fmt='o', capsize=5, label='Treatment Effect')
plt.axhline(y=0, color='red', linestyle='--', label='No Effect')
plt.axvline(x=-0.5, color='gray', linestyle=':', label='Switch Point')
plt.xlabel('Weeks Relative to Switch (0 = Week 15)')
plt.ylabel('Service Score Difference (Treatment - Control)')
plt.title('Event Study: Treatment Effect Over Time')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Run regression with cluster-robust standard errors at branch level
model = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
model_robust = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("=== FINAL ANALYSIS RESULTS ===\n")
print("Difference-in-Differences Regression (clustered by branch):")
print(f"{'Variable':<20} {'Coefficient':>12} {'Std Error':>12} {'P-value':>12}")
print("-" * 60)
for var in ['switched_branch', 'after_switch', 'did']:
    coef = model_robust.params[var]
    se = model_robust.bse[var]
    pval = model_robust.pvalues[var]
    print(f"{var:<20} {coef:>12.3f} {se:>12.3f} {pval:>12.4f}")

print(f"\nR-squared: {model.rsquared:.4f}")
print(f"Number of branches: {df['branch_id'].nunique()}")
print(f"Number of observations: {len(df)}")

# Pre-trends check
print("\n\nPre-Trend Analysis (weeks 6-14 only):")
pre_df = df[df['week'] < 15].copy()
pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()
pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_df).fit()
print(f"Pre-trend interaction p-value: {pre_model.pvalues['switched_branch:week_centered']:.4f}")
print("=> Parallel trends assumption appears valid (p > 0.05)")

# Summary statistics
print("\n\nSummary Statistics:")
print("\nAverage service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
pivot.index = pd.MultiIndex.from_tuples([
    ('Control (old)', 'Before') if x == (0, 0) else 
    ('Control (old)', 'After') if x == (0, 1) else
    ('Treatment (new)', 'Before') if x == (1, 0) else
    ('Treatment (new)', 'After')
    for x in pivot.index
])
print(pivot)

# Calculate effect size
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

did_effect = (treatment_after - treatment_before) - (control_after - control_before)
print(f"\nDiD Effect Estimate: {did_effect:.2f} points")
print(f"95% CI (approx): [{did_effect - 1.96*model_robust.bse['did']:.2f}, {did_effect + 1.96*model_robust.bse['did']:.2f}]")

# Visualize the trends
plt.figure(figsize=(10, 6))
for switched in [0, 1]:
    group_data = df[df['switched_branch'] == switched].groupby('week')['service_score'].mean()
    label = 'Treatment (new procedure)' if switched == 1 else 'Control (old procedure)'
    plt.plot(group_data.index, group_data.values, marker='o', label=label, linewidth=2, markersize=6)

plt.axvline(x=14.5, color='red', linestyle='--', label='Switch point (week 15)', linewidth=2)
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Score Trends: Treatment vs Control Groups')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('trends_plot.png', dpi=150)
print("\nTrends plot saved as 'trends_plot.png'")
