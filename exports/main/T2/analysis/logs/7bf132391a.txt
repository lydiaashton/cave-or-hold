import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary table: Average service scores by group and period
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES SUMMARY TABLE")
print("=" * 60)

# Before period (weeks 6-14)
before = df[df['after_switch'] == 0]
before_switched = before[before['switched_branch'] == 1]['service_score'].mean()
before_control = before[before['switched_branch'] == 0]['service_score'].mean()

# After period (weeks 15-22)
after = df[df['after_switch'] == 1]
after_switched = after[after['switched_branch'] == 1]['service_score'].mean()
after_control = after[after['switched_branch'] == 0]['service_score'].mean()

print(f"\n{'Group':<20} {'Before (W6-14)':<18} {'After (W15-22)':<18} {'Change':<10}")
print("-" * 60)
print(f"{'Switched Branches':<20} {before_switched:<18.2f} {after_switched:<18.2f} {after_switched - before_switched:<10.2f}")
print(f"{'Control Branches':<20} {before_control:<18.2f} {after_control:<18.2f} {after_control - before_control:<10.2f}")
print("-" * 60)

# DiD estimate
did_estimate = (after_switched - before_switched) - (after_control - before_control)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")
print("(This is the estimated effect of the new procedure on service score)")

# Now run regression with controls
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Clustered standard errors by branch
print("\n" + "=" * 60)
print("CLUSTERED STANDARD ERRORS (by branch)")
print("=" * 60)

# Use cluster-robust standard errors
model2_clustered = model2.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(model2_clustered.summary().tables[1])

weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_score']

print("Weekly averages by group:")
print(weekly_avg.pivot(index='week', columns='switched_branch', values='avg_score'))

# Check for any pre-trends (weeks 6-14)
pre_period = df[df['week'] < 15]
pre_trend = pre_period.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()

# Calculate the difference between groups each week
pivot_pre = pre_trend.pivot(index='week', columns='switched_branch', values='service_score')
pivot_pre['diff'] = pivot_pre[1] - pivot_pre[0]
print("\nPre-period trend (difference between switched and control):")
print(pivot_pre)

# Statistical test for pre-trends
# Regress service score on week*switched interaction for pre-period only
pre_period = pre_period.copy()
pre_period['week_centered'] = pre_period['week'] - pre_period['week'].mean()
pre_period['interaction'] = pre_period['week_centered'] * pre_period['switched_branch']

from statsmodels.formula.api import ols
pre_trend_model = ols('service_score ~ switched_branch + week_centered + interaction', data=pre_period).fit()

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: estimate effect for each week relative to switch
# Create week dummies interacted with switched_branch
df = df.copy()
df['relative_week'] = df['week'] - 15  # week 15 is the switch week (0)

# Create indicators for each week relative to switch, interacted with treatment
weeks = sorted(df['relative_week'].unique())
print("Relative weeks:", weeks)

# Event study regression with week 14 (relative_week = -1) as reference
# This shows the effect for each week compared to the period just before switch
event_study_vars = []
for w in weeks:
    if w != -1:  # skip reference period
        df[f'week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(f'week_{w}')

# Run event study regression with controls
event_study_formula = f"service_score ~ switched_branch + after_switch + {' + '.join(event_study_vars)} + staffing_index + visits_index"
event_model = ols(event_study_formula, data=df).fit()
event_model_clustered = event_model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("\nEvent Study Results (relative to week 14, the week before switch):")
print("=" * 70)
coef_df = event_model_clustered.summary().tables[1]
print(coef_df)

# Extract coefficients for the event study
event_coefs = {}
event_se = {}
for w in weeks:
    if w != -1:
        var_name = f'week_{w}'
        if var_name in event_model_clustered.params.index:
            event_coefs[w] = event_model_clustered.params[var_name]
            event_se[w] = event_model_clustered.bse[var_name]

print("\nEvent Study Coefficients:")
for w in sorted(event_coefs.keys()):
    ci_low = event_coefs[w] - 1.96 * event_se[w]
    ci_high = event_coefs[w] + 1.96 * event_se[w]
    sig = "***" if event_model_clustered.pvalues[f'week_{w}'] < 0.01 else "**" if event_model_clustered.pvalues[f'week_{w}'] < 0.05 else "*" if event_model_clustered.pvalues[f'week_{w}'] < 0.1 else ""
    print(f"Week {w+15} (rel {w:+d}): {event_coefs[w]:.2f} [{ci_low:.2f}, {ci_high:.2f}] {sig}")
