print(switch_counts)
print(f"\nTotal branches: {df['branch_id'].nunique()}")
print(f"Switched branches: {switch_counts.get(1, 0)}")
print(f"Non-switched branches: {switch_counts.get(0, 0)}")

# Check data balance
print("\nObservations per branch (should be 17 weeks each):")
obs_per_branch = df.groupby('branch_id').size()
print(obs_per_branch.describe())

# Create a summary table of service scores by group and period
df['period'] = np.where(df['after_switch'] == 0, 'Before', 'After')
summary = df.groupby(['switched_branch', 'period'])['service_score'].agg(['mean', 'std', 'count'])
print("\nService Score Summary by Group and Period:")
print(summary)

# Calculate the difference-in-differences manually
before_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"After - Control branches: {after_control:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Run difference-in-differences regression
# Basic DiD: service_score ~ switched_branch * after_switch
model_basic = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Basic DiD Regression (with clustered standard errors):")
print(model_basic.summary())

# DiD with controls
model_controls = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Test parallel trends: interact switched_branch with week dummies for pre-period only
# This checks if the trends were different before the switch
pre_df = df[df['week'] < 15].copy()
pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()

# Regress service_score on switched_branch, week, and their interaction for pre-period
# If parallel trends hold, the interaction should not be significant
model_trends = ols('service_score ~ switched_branch * week_centered', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print("Parallel Trends Test (Pre-Period Only):")
print(model_trends.summary())

# Also check if there's any differential trend

weekly_diff = df.groupby('week').apply(lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean())
