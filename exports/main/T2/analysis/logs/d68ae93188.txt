import pandas as pd
import numpy as np
import statsmodels.api as sm
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch']  # 1 if branch switched
df['post'] = df['after_switch']  # 1 if after week 15
df['DiD'] = df['treated'] * df['post']  # interaction term

# Run DiD regression
# Y = β0 + β1*treated + β2*post + β3*DiD + ε
# β3 is the DiD estimator - the effect of the treatment

X = df[['treated', 'post', 'DiD']]
X = sm.add_constant(X)
y = df['service_score']

model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Difference-in-Differences Regression Results:")
print(model.summary())

print("\n\nKey coefficient (DiD interaction):", model.params['DiD'])
print("Standard error:", model.bse['DiD'])
print("95% CI:", model.conf_int().loc['DiD'].values)
print("P-value:", model.pvalues['DiD'])

import pandas as pd
import numpy as np
import statsmodels.api as sm

# Load the data
df = pd.read_csv('branches.csv')

# Create DiD variables
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Check parallel trends assumption - compare pre-trends
print("=== Parallel Trends Check (Pre-Switch Period: Weeks 6-14) ===")
pre = df[df['week'] < 15].copy()

# Regress service score on week and treated*week interaction in pre-period
pre['week_centered'] = pre['week'] - pre['week'].mean()
pre['treated_week'] = pre['treated'] * pre['week_centered']

X_pre = pre[['treated', 'week_centered', 'treated_week']]
X_pre = sm.add_constant(X_pre)
y_pre = pre['service_score']

model_pre = sm.OLS(y_pre, X_pre).fit(cov_type='cluster', cov_kwds={'groups': pre['branch_id']})
print("\nPre-trend interaction test:")
print(model_pre.summary())
print("\nPre-trend coefficient (treated_week):", model_pre.params['treated_week'])
print("P-value:", model_pre.pvalues['treated_week'])

# Now run full DiD with week fixed effects
print("\n\n=== DiD with Week Fixed Effects ===")
df_full = df.copy()
df_full['week_fe'] = df_full['week'].astype(str)
X_full = df_full[['treated', 'post', 'DiD']]
X_full = sm.add_constant(X_full)

# Add week dummies
week_dummies = pd.get_dummies(df_full['week'], prefix='week', drop_first=True)
X_full = pd.concat([X_full, week_dummies], axis=1)

y_full = df_full['service_score']
model_fe = sm.OLS(y_full, X_full).fit(cov_type='cluster', cov_kwds={'groups': df_full['branch_id']})
print("\nDiD coefficient with week FEs:", model_fe.params['DiD'])
print("P-value:", model_fe.pvalues['DiD'])

# Check if staffing/visits changed
print("\n\n=== Effect on Staffing and Visits Indices ===")
for outcome in ['staffing_index', 'visits_index']:
    X = df[['treated', 'post', 'DiD']]
    X = sm.add_constant(X)
    y = df[outcome]
    m = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
    print(f"\n{outcome}:")
    print(f"  DiD coefficient: {m.params['DiD']:.3f} (p={m.pvalues['DiD']:.3f})")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm

# Load the data
df = pd.read_csv('branches.csv')

# Calculate the difference between treatment and control each week
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
pivot = weekly_avg.pivot(index='week', columns='switched_branch', values='avg_score')
pivot['diff'] = pivot[1] - pivot[0]
pivot['week'] = pivot.index

print("Weekly difference (Treatment - Control):")
print(pivot[['diff']].round(2))

# Calculate average pre and post differences
pre_diff = pivot[pivot.index < 15]['diff'].mean()
post_diff = pivot[pivot.index >= 15]['diff'].mean()
print(f"\nAverage pre-switch difference: {pre_diff:.2f}")
print(f"Average post-switch difference: {post_diff:.2f}")
print(f"Change in difference (DiD estimate): {post_diff - pre_diff:.2f}")

# Effect size relative to baseline
baseline = df[df['week'] < 15]['service_score'].mean()
effect_size_pct = 1.77 / baseline * 100
print(f"\nBaseline average service score: {baseline:.2f}")
print(f"Effect size as % of baseline: {effect_size_pct:.1f}%")

# Calculate how many branches improved
branch_avg = df.groupby(['branch_id', 'switched_branch', 'after_switch'])['service_score'].mean().reset_index()
branch_avg.columns = ['branch_id', 'switched', 'after', 'avg_score']
