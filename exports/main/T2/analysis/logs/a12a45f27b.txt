import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Create treatment groups
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary by group and period
print("=== Service Score Summary ===")
print("\nBefore switch (weeks 6-14):")
before = df[df['after_switch'] == 0]
print(before.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

print("\nAfter switch (weeks 15-22):")
after = df[df['after_switch'] == 1]
print(after.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

# Difference-in-differences calculation
before_switched = before[before['switched_branch'] == 1]['service_score'].mean()
before_control = before[before['switched_branch'] == 0]['service_score'].mean()
after_switched = after[after['switched_branch'] == 1]['service_score'].mean()
after_control = after[after['switched_branch'] == 0]['service_score'].mean()

print("\n=== Difference-in-Differences ===")
print(f"Switched branches - Before: {before_switched:.2f}, After: {after_switched:.2f}, Change: {after_switched - before_switched:.2f}")
print(f"Control branches - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")
print(f"\nDiD Estimate (treatment effect): {(after_switched - before_switched) - (after_control - before_control):.2f}")

# Check how many branches switched
print("\n=== Branch Counts ===")
print(f"Total branches: {df['branch_id'].nunique()}")
print(f"Switched branches: {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"Control branches: {df[df['switched_branch']==0]['branch_id'].nunique()}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched + beta2*after + beta3*(switched*after) + error
# beta3 is the treatment effect

model = ols('service_score ~ switched_branch + after_switch + switched_branch:after_switch', data=df).fit()
print("=== DiD Regression (Basic) ===")
print(model.summary())

# Now with branch fixed effects (more rigorous)
# This controls for time-invariant branch characteristics
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df_with_dummies['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_')]
formula = 'service_score ~ switched_branch + after_switch + switched_branch:after_switch + ' + ' + '.join(branch_cols)

model_fe = ols(formula, data=df_with_dummies).fit()

import pandas as pd
import numpy as np
import statsmodels.api as sm

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Check if staffing and visits changed differently for treatment vs control
print("=== Covariate Balance Check ===")
print("\nStaffing Index - Before vs After:")
for group in [0, 1]:
    group_name = 'Switched' if group == 1 else 'Control'
    before = df[(df['switched_branch'] == group) & (df['after_switch'] == 0)]['staffing_index'].mean()
    after = df[(df['switched_branch'] == group) & (df['after_switch'] == 1)]['staffing_index'].mean()
    print(f"{group_name}: Before={before:.3f}, After={after:.3f}, Change={after-before:.3f}")

print("\nVisits Index - Before vs After:")
for group in [0, 1]:
    group_name = 'Switched' if group == 1 else 'Control'
    before = df[(df['switched_branch'] == group) & (df['after_switch'] == 0)]['visits_index'].mean()
    after = df[(df['switched_branch'] == group) & (df['after_switch'] == 1)]['visits_index'].mean()
    print(f"{group_name}: Before={before:.3f}, After={after:.3f}, Change={after-before:.3f}")

# Regression with covariates
print("\n=== DiD Regression with Covariates ===")
model_cov = sm.OLS.from_formula('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print(model_cov.summary())

# Calculate effect size relative to baseline
baseline = df[df['after_switch'] == 0]['service_score'].mean()
treatment_effect = model_cov.params['treatment']
print(f"\n=== Effect Size ===")
print(f"Baseline average service score: {baseline:.2f}")
print(f"Treatment effect: {treatment_effect:.2f}")
print(f"Relative improvement: {(treatment_effect/baseline)*100:.1f}%")

# Confidence interval
ci_lower = model_effect = model_cov.conf_int().loc['treatment'][0]
ci_upper = model_cov.conf_int().loc['treatment'][1]
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Check effect by post-switch week (is there a fade-out or ramp-up?)
post_period = df[df['after_switch'] == 1].copy()
post_period['weeks_since_switch'] = post_period['week'] - 14  # weeks since switch

print("=== Treatment Effect by Weeks Since Switch ===")
for week_offset in range(1, 9):
    week_data = post_period[post_period['weeks_since_switch'] == week_offset]
    switched_mean = week_data[week_data['switched_branch'] == 1]['service_score'].mean()
    control_mean = week_data[week_data['switched_branch'] == 0]['service_score'].mean()
    diff = switched_mean - control_mean
    print(f"Week {week_offset} post-switch: Switched={switched_mean:.2f}, Control={control_mean:.2f}, Diff={diff:.2f}")

# Also check distribution of effects
print("\n=== Distribution of Branch-Level Changes ===")
# Calculate change for each branch
branch_pivot = df.pivot_table(index='branch_id', columns='after_switch', values='service_score', aggfunc='mean')
branch_pivot['change'] = branch_pivot[1] - branch_pivot[0]

# Merge with switched status
switched_status = df.groupby('branch_id')['switched_branch'].first()
branch_pivot['switched'] = switched_status

switched_branches = branch_pivot[branch_pivot['switched'] == 1]['change']
control_branches = branch_pivot[branch_pivot['switched'] == 0]['change']

print(f"\nSwitched branches (n={len(switched_branches)}):")
print(f"  Mean change: {switched_branches.mean():.2f}")
print(f"  Median change: {switched_branches.median():.2f}")
print(f"  Std: {switched_branches.std():.2f}")
print(f"  % with positive change: {(switched_branches > 0).mean()*100:.1f}%")

print(f"\nControl branches (n={len(control_branches)}):")
print(f"  Mean change: {control_branches.mean():.2f}")
print(f"  Median change: {control_branches.median():.2f}")
print(f"  Std: {control_branches.std():.2f}")
print(f"  % with positive change: {(control_branches > 0).mean()*100:.1f}%")

# Statistical test
from scipy import stats
t_stat, p_val = stats.ttest_ind(switched_branches, control_branches)
print(f"\nT-test comparing changes: t={t_stat:.2f}, p={p_val:.4f}")
