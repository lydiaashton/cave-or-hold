import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many branches switched?
print("Branches by treatment status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Check balance - average service scores before and after by group
print("\n=== Service Score Summary ===")
print("\nPre-switch (weeks 6-14):")
pre = df[df['after_switch'] == 0]
print(pre.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

print("\nPost-switch (weeks 15-22):")
post = df[df['after_switch'] == 1]
print(post.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

# Calculate the 2x2 DiD manually
treated_pre = pre[pre['switched_branch'] == 1]['service_score'].mean()
treated_post = post[post['switched_branch'] == 1]['service_score'].mean()
control_pre = pre[pre['switched_branch'] == 0]['service_score'].mean()
control_post = post[post['switched_branch'] == 0]['service_score'].mean()

print("\n=== Difference-in-Differences Calculation ===")
print(f"Treated (switched) branches: Pre={treated_pre:.2f}, Post={treated_post:.2f}, Change={treated_post-treated_pre:.2f}")
print(f"Control (non-switched) branches: Pre={control_pre:.2f}, Post={control_post:.2f}, Change={control_post-control_pre:.2f}")
did_estimate = (treated_post - treated_pre) - (control_post - control_pre)
print(f"\nDiD Estimate (Treatment Effect): {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check parallel trends - average service score by week and treatment group
print("=== Parallel Trends Check (Pre-Period: Weeks 6-14) ===")
pre_period = df[df['week'] <= 14]
weekly_avg = pre_period.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(weekly_avg)
print("\nDifference (Treated - Control) by week:")
weekly_diff = weekly_avg[1] - weekly_avg[0]
print(weekly_diff)

# Run DiD regression
print("\n=== Regression Analysis ===")

# Basic DiD model
df['treatment'] = df['switched_branch'] * df['after_switch']

model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 1: Basic DiD (clustered SE by branch)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 2: DiD with controls (clustered SE by branch)")
print(model2.summary().tables[1])

# Branch fixed effects model
df['branch_id'] = df['branch_id'].astype(str)
model3 = ols('service_score ~ C(branch_id) + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 3: Branch Fixed Effects (clustered SE by branch)")
# Extract treatment coefficient
print(f"\nTreatment effect (with branch FE): {model3.params['treatment']:.3f}, SE: {model3.bse['treatment']:.3f}, p-value: {model3.pvalues['treatment']:.3f}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create a plot showing trends by treatment group
plt.figure(figsize=(12, 6))

# Calculate weekly averages by treatment group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

# Plot
plt.plot(weekly_avg.index, weekly_avg[0], 'o-', label='Control (Old Procedure)', linewidth=2, markersize=8)
plt.plot(weekly_avg.index, weekly_avg[1], 's-', label='Treatment (New Procedure)', linewidth=2, markersize=8)

# Add vertical line at switch point
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')

plt.xlabel('Week', fontsize=12)
plt.ylabel('Service Score', fontsize=12)
plt.title('Service Score Trends: Treatment vs Control Groups', fontsize=14, fontweight='bold')
plt.legend(fontsize=11)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("Plot saved as service_score_trends.png")

# Show the data
print("\nWeekly averages:")
print(weekly_avg)

# Calculate confidence interval for the treatment effect
from statsmodels.formula.api import ols
df['treatment'] = df['switched_branch'] * df['after_switch']
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se

print(f"\n=== Treatment Effect Summary ===")
print(f"Point estimate: {treatment_coef:.2f}")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {model.pvalues['treatment']:.3f}")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check if effect varies by time since switch
df['treatment'] = df['switched_branch'] * df['after_switch']

# Power analysis - what's the minimum detectable effect?
n_branches = df['branch_id'].nunique()
n_observations = len(df)
print(f"Total observations: {n_observations}")
print(f"Total branches: {n_branches}")
print(f"Treated branches: {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"Control branches: {df[df['switched_branch']==0]['branch_id'].nunique()}")

# Check variance
print(f"\nService score variance: {df['service_score'].var():.2f}")
print(f"Service score std dev: {df['service_score'].std():.2f}")

# Event study - check if there's any pattern in the effect
df['weeks_since_switch'] = df['week'] - 14
df['weeks_since_switch'] = df['weeks_since_switch'].apply(lambda x: max(0, x))

# Just show post-period treatment effects by week
post = df[df['after_switch'] == 1]
print("\n=== Post-Switch Weekly Differences (Treated - Control) ===")
weekly_diff = post.groupby('week').apply(
    lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()
)
print(weekly_diff)

# Effect size in practical terms
treatment_effect = 1.69
avg_score = df['service_score'].mean()
print(f"\n=== Practical Significance ===")
print(f"Average service score: {avg_score:.1f}")
print(f"Treatment effect: {treatment_effect:.1f} points ({treatment_effect/avg_score*100:.1f}% of average)")
