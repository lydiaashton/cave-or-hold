import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# With branch fixed effects, switched_branch is absorbed, so just include treatment and after_switch
print("=== Fixed Effects Model (treatment effect only) ===")
model_fe = ols('service_score ~ treatment + after_switch + C(branch_id)', data=df).fit()
print("\nKey coefficients:")
print(model_fe.params[['treatment', 'after_switch']])
print("\nKey p-values:")
print(model_fe.pvalues[['treatment', 'after_switch']])

# Cluster-robust standard errors
print("\n=== With Cluster-Robust SEs ===")
robust = model_fe.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print("Treatment effect:", robust.params['treatment'])
print("P-value:", robust.pvalues['treatment'])
print("95% CI:", robust.conf_int().loc['treatment'])

# Also check weekly trends visually
print("\n=== Weekly Average Service Scores ===")
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(weekly_avg)

# Create visualization
fig, ax = plt.subplots(figsize=(10, 6))
for switched in [0, 1]:
    subset = weekly_avg[switched]
    label = 'Switched branches' if switched == 1 else 'Control branches'
    ax.plot(subset.index, subset.values, marker='o', label=label, linewidth=2, markersize=6)

ax.axvline(x=14.5, color='red', linestyle='--', label='Switch implemented (week 15)')
ax.set_xlabel('Week')
ax.set_ylabel('Average Service Score')
ax.set_title('Service Scores by Branch Type Over Time')
ax.legend()

baseline = df[(df['after_switch']==0) & (df['switched_branch']==0)]['service_score'].mean()

# Treatment effect as percentage of baseline

# Event study regression
print("\n=== Event Study Regression ===")
# Create week dummies relative to switch
df['week_rel'] = df['week'] - 14  # week 14 is the last pre-switch week

# Create indicators for each week relative to switch
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    df[f'week_{w}'] = (df['week_rel'] == w).astype(int)

# Interact with treatment
for w in range(-8, 9):
    df[f'treat_week_{w}'] = df['switched_branch'] * df[f'week_{w}']

# Run regression omitting week -1 as reference
week_vars = '+'.join([f'treat_week_{w}' for w in range(-8, 8) if w != -1])
event_model = ols(f'service_score ~ {week_vars} + C(branch_id) + C(week_rel)', data=df).fit()

print("\nWeekly treatment effects (relative to week 13):")
for w in range(-8, 8):
    if w != -1:
        coef_name = f'treat_week_{w}'
        if coef_name in event_model.params.index:
            coef = event_model.params[coef_name]
            se = event_model.bse[coef_name]

# Treatment effect as percentage of baseline
treatment_effect = 2.32

print(f"Treatment effect: {treatment_effect:.2f} points ({pct_effect:.1f}% of baseline)")

# Simple event study - just show the weekly differences we already calculated
print("\n=== Weekly Differences (Switched - Control) ===")
weekly_diffs = []
for week in range(6, 23):
    week_data = df[df['week'] == week]
    if len(week_data) > 0:
        diff = week_data[week_data['switched_branch']==1]['service_score'].mean() - \
               week_data[week_data['switched_branch']==0]['service_score'].mean()
        weekly_diffs.append((week, diff))
        print(f"Week {week}: {diff:.2f}")
