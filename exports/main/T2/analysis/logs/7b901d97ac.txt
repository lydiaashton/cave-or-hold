treatment_change = treatment_after - treatment_before
control_change = control_after - control_before
did_estimate = treatment_change - control_change

print(f"\n1. SAMPLE SIZE:")
print(f"   - Total observations: {len(df)}")
print(f"   - Treatment branches (switched): {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"   - Control branches (no switch): {df[df['switched_branch']==0]['branch_id'].nunique()}")
print(f"   - Pre-switch weeks: 6-14 (9 weeks)")
print(f"   - Post-switch weeks: 15-22 (8 weeks)")

print(f"\n2. SERVICE SCORE CHANGES:")
print(f"   - Treatment group: {treatment_before:.1f} → {treatment_after:.1f} (change: {treatment_change:+.1f})")
print(f"   - Control group: {control_before:.1f} → {control_after:.1f} (change: {control_change:+.1f})")
print(f"   - Difference-in-Differences estimate: +{did_estimate:.1f} points")

# Regression with clustered SE
model = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
did_coef = model.params['switched_branch:after_switch']
did_se = model.bse['switched_branch:after_switch']
did_pvalue = model.pvalues['switched_branch:after_switch']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se

print(f"\n3. STATISTICAL SIGNIFICANCE:")
print(f"   - DiD coefficient: {did_coef:.2f}")
print(f"   - Clustered standard error: {did_se:.2f}")
print(f"   - p-value: {did_pvalue:.3f}")
print(f"   - 95% confidence interval: [{ci_lower:.2f}, {ci_upper:.2f}]")

# Effect size relative to baseline
baseline = df['service_score'].mean()
relative_effect = (did_coef / baseline) * 100
print(f"\n4. EFFECT SIZE:")
print(f"   - Baseline average score: {baseline:.1f}")
print(f"   - Relative improvement: {relative_effect:.1f}%")

# How many branches improved?
branch_changes = df.groupby(['branch_id', 'switched_branch', 'after_switch'])['service_score'].mean()
branch_changes = branch_changes.unstack(level=[1,2])

treatment_improved = 0
treatment_total = 0
for b in df[df['switched_branch']==1]['branch_id'].unique():
    if b in branch_changes.index:
        before = branch_changes.loc[b, (1, 0)]
        after = branch_changes.loc[b, (1, 1)]
        treatment_total += 1
        if after > before:
            treatment_improved += 1

control_improved = 0
control_total = 0
for b in df[df['switched_branch']==0]['branch_id'].unique():
    if b in branch_changes.index:
        before = branch_changes.loc[b, (0, 0)]
        after = branch_changes.loc[b, (0, 1)]
        control_total += 1
        if after > before:
            control_improved += 1

print(f"\n5. BRANCH-LEVEL OUTCOMES:")
print(f"   - Treatment branches that improved: {treatment_improved}/{treatment_total} ({100*treatment_improved/treatment_total:.0f}%)")
print(f"   - Control branches that improved: {control_improved}/{control_total} ({100*control_improved/control_total:.0f}%)")
