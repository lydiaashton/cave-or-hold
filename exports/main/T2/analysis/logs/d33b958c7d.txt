# Run DiD regression
print("\n=== DiD Regression ===")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create DiD variables
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

# Main DiD model with controls
model = ols('service_score ~ treated + post + treated_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get the DiD estimate and confidence interval
did_coef = model.params['treated_post']
did_se = model.bse['treated_post']
did_pvalue = model.pvalues['treated_post']

print("=== Main Results Summary ===")
print(f"DiD Estimate: {did_coef:.2f} points")
print(f"Standard Error: {did_se:.2f}")
print(f"95% CI: [{did_coef - 1.96*did_se:.2f}, {did_coef + 1.96*did_se:.2f}]")
print(f"P-value: {did_pvalue:.3f}")

# Calculate effect size relative to baseline
baseline_score = df[df['after_switch'] == 0]['service_score'].mean()
print(f"\nBaseline average score: {baseline_score:.1f}")
print(f"Effect as % of baseline: {did_coef/baseline_score*100:.1f}%")

# Also run a model with branch fixed effects to control for time-invariant branch characteristics
print("\n=== DiD with Branch Fixed Effects ===")
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ treated_post + post + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_fe.summary())

# Extract just the treated_post coefficient
fe_did = model_fe.params['treated_post']
fe_se = model_fe.bse['treated_post']
fe_p = model_fe.pvalues['treated_post']
print(f"\nWith Branch FE - DiD: {fe_did:.2f}, SE: {fe_se:.2f}, p-value: {fe_p:.3f}")
