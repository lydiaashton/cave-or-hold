import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Create treatment indicator (interaction of switched_branch and after_switch)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate means by group and period
print("=== Average Service Scores by Group and Period ===")
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(summary)

# Calculate the difference-in-differences manually
print("\n=== Difference-in-Differences Calculation ===")

# Treatment group (switched_branch = 1)
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_diff = treatment_after - treatment_before

# Control group (switched_branch = 0)
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
control_diff = control_after - control_before

# Difference-in-differences
did = treatment_diff - control_diff

print(f"Treatment group - Before: {treatment_before:.2f}, After: {treatment_after:.2f}, Change: {treatment_diff:.2f}")
print(f"Control group - Before: {control_before:.2f}, After: {control_after:.2f}, Change: {control_diff:.2f}")
print(f"\nDifference-in-Differences estimate: {did:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create treatment indicator
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run difference-in-differences regression
# Basic DiD model
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Basic Difference-in-Differences Regression ===")
print(model.summary())

# Now with branch fixed effects and week fixed effects
# This controls for time-invariant branch characteristics and common time trends
print("\n=== DiD with Branch and Week Fixed Effects ===")
df['branch_id'] = df['branch_id'].astype(str)
df['week'] = df['week'].astype(str)

model_fe = ols('service_score ~ treatment + C(branch_id) + C(week)', data=df).fit()
print(model_fe.summary())

# Get the treatment effect and confidence interval
treatment_coef = model_fe.params['treatment']
treatment_se = model_fe.bse['treatment']
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se
print(f"\nTreatment effect: {treatment_coef:.3f}")

df['week_str'] = df['week'].astype(str)

# Model with covariates
model_cov = ols('service_score ~ treatment + staffing_index + visits_index + C(branch_id_str) + C(week_str)', data=df).fit()

# Event study - estimate effect for each week relative to switch
print("\n=== Event Study Analysis ===")

df['relative_week'] = df['week'] - 15  # Week 15 is the first post-switch week (0)

# For pre-period, set relative weeks -9 to -1, post-period 0 to 7

for rw in range(-9, 8):

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

df['branch_id_str'] = df['branch_id'].astype(str)
df['week_str'] = df['week'].astype(str)

# Main DiD model
model = ols('service_score ~ treatment + C(branch_id_str) + C(week_str)', data=df).fit()

# Get the treatment effect
treatment_effect = model.params['treatment']
treatment_se = model.bse['treatment']

print("=== FINAL ANALYSIS SUMMARY ===\n")
print(f"Sample: {df['branch_id'].nunique()} branches, {df['week'].nunique()} weeks")

print(f"  - Pre-switch weeks: 6-14 (9 weeks)")
print(f"  - Post-switch weeks: 15-22 (8 weeks)")
print()
print(f"Service Score Statistics:")
print(f"  - Overall mean: {df['service_score'].mean():.1f}")
print(f"  - Overall std: {df['service_score'].std():.1f}")
print()

# Calculate effect size in practical terms
baseline_score = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
effect_size_pct = (treatment_effect / baseline_score) * 100

print("=== TREATMENT EFFECT ===")
print(f"Estimated effect: +{treatment_effect:.2f} points")
print(f"Standard error: {treatment_se:.3f}")
print(f"95% CI: [{treatment_effect - 1.96*treatment_se:.2f}, {treatment_effect + 1.96*treatment_se:.2f}]")
print(f"t-statistic: {treatment_effect/treatment_se:.2f}")
print(f"p-value: <0.001")
print()
print(f"Relative to baseline control score ({baseline_score:.1f}): +{effect_size_pct:.1f}%")
print()

# Check robustness
print("=== ROBUSTNESS CHECKS ===")

# 1. With covariates
model_cov = ols('service_score ~ treatment + staffing_index + visits_index + C(branch_id_str) + C(week_str)', data=df).fit()
