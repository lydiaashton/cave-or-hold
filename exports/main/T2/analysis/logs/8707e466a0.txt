import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
branches_switched = df[df['switched_branch'] == 1]['branch_id'].nunique()
branches_control = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"Branches that switched: {branches_switched}")
print(f"Branches that didn't switch (control): {branches_control}")
print(f"Total branches: {branches_switched + branches_control}")

# Average service scores by group and period
print("\n=== Average Service Scores ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the simple DiD estimate
# Treatment group: before vs after
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nTreatment group (switched branches):")
print(f"  Before: {treatment_before:.2f}")
print(f"  After: {treatment_after:.2f}")
print(f"  Change: {treatment_after - treatment_before:.2f}")

print(f"\nControl group (non-switched branches):")
print(f"  Before: {control_before:.2f}")
print(f"  After: {control_after:.2f}")
print(f"  Change: {control_after - control_before:.2f}")

# Difference-in-Differences estimate
did_estimate = (treatment_after - treatment_before) - (control_after - control_before)
print(f"\n=== Simple DiD Estimate: {did_estimate:.2f} ===")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# Parallel trends test with proper regression
pre_df = df[df['week'] < 15].copy()
week_dummies = pd.get_dummies(pre_df['week'], prefix='week', drop_first=True)
pre_df = pd.concat([pre_df.reset_index(drop=True), week_dummies], axis=1)

# Build formula for parallel trends test
week_cols = [c for c in pre_df.columns if c.startswith('week_')]
interaction_terms = [f'switched_branch:{c}' for c in week_cols]
formula = f"service_score ~ switched_branch + {' + '.join(week_cols)} + {' + '.join(interaction_terms)}"

model_trends = ols(formula, data=pre_df).fit()
print("=== Parallel Trends Test (Pre-period interactions) ===")
print("Testing if treatment and control had similar trends before the switch...")
print("\nInteraction coefficients (should be close to 0 if parallel trends hold):")
for coef in interaction_terms:
    if coef in model_trends.params:
        print(f"  {coef}: coef={model_trends.params[coef]:.3f}, p={model_trends.pvalues[coef]:.3f}")

# Let's also do a simpler test: regression of the difference on a time trend in pre-period
pre_df_simple = df[df['week'] < 15].copy()
# Calculate group means by week
weekly_means = pre_df_simple.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_means['diff'] = weekly_means[1] - weekly_means[0]
weekly_means['time'] = weekly_means.index

# Regress the difference on time
from scipy import stats
slope, intercept, r_value, p_value, std_err = stats.linregress(weekly_means['time'], weekly_means['diff'])
print(f"\n=== Simple Pre-Trend Test ===")
print(f"Regression of (Treatment - Control) difference on time (weeks 6-14):")
print(f"  Slope: {slope:.3f} (per week)")
print(f"  P-value: {p_value:.3f}")
print(f"  If parallel trends hold, slope should not be significantly different from 0")

# Also check the average pre-period difference
pre_period_diff = weekly_means['diff'].mean()
print(f"\n  Average pre-period difference: {pre_period_diff:.2f}")

# Let's also do an event study style analysis
print("\n=== Event Study Analysis ===")
# Create relative week variable (week 14 = 0, the last pre-treatment week)
df['relative_week'] = df['week'] - 14

# Create dummies for each relative week
for rw in range(-8, 9):  # weeks -8 to +8 relative to week 14

# Run event study regression (omitting week 0 as reference)

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Weekly trends data for visualization
weekly_trends = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

# Create a cleaner event study plot
fig, ax = plt.subplots(figsize=(10, 6))

# Plot both groups
ax.plot(weekly_trends.index, weekly_trends[0], 'o-', label='Control (No Switch)', linewidth=2, markersize=6, color='blue')
ax.plot(weekly_trends.index, weekly_trends[1], 's-', label='Treatment (Switched)', linewidth=2, markersize=6, color='green')

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')
ax.axvspan(14.5, 22, alpha=0.15, color='red')
ax.axvspan(6, 14.5, alpha=0.15, color='gray')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Score Trends: Treatment vs Control Groups\n(Pre-treatment divergence noted)', fontsize=14, fontweight='bold')
ax.legend(loc='best')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("Plot saved as 'service_score_trends.png'")

# Let's do a simpler event study by calculating differences week by week
print("\n=== Week-by-Week Treatment Effect (Difference in Differences from baseline) ===")
print("Using Week 6 as baseline for both groups")

baseline_control = df[(df['week'] == 6) & (df['switched_branch'] == 0)]['service_score'].mean()
baseline_treatment = df[(df['week'] == 6) & (df['switched_branch'] == 1)]['service_score'].mean()

print(f"\nBaseline (Week 6) - Control: {baseline_control:.2f}, Treatment: {baseline_treatment:.2f}")
print(f"Baseline difference: {baseline_treatment - baseline_control:.2f}")

print("\nRelative to baseline, the DiD effect by week:")
results = []
for w in range(6, 23):
    control_w = df[(df['week'] == w) & (df['switched_branch'] == 0)]['service_score'].mean()
    treatment_w = df[(df['week'] == w) & (df['switched_branch'] == 1)]['service_score'].mean()
    
    # DiD relative to week 6
    control_change = control_w - baseline_control
    treatment_change = treatment_w - baseline_treatment
    did_w = treatment_change - control_change
    
    results.append({'week': w, 'control': control_w, 'treatment': treatment_w, 'did': did_w})
    print(f"Week {w:2d}: Control={control_w:.2f}, Treatment={treatment_w:.2f}, DiD={did_w:.2f}")

results_df = pd.DataFrame(results)

# Average DiD in pre vs post period
pre_did = results_df[results_df['week'] < 15]['did'].mean()
post_did = results_df[results_df['week'] >= 15]['did'].mean()
print(f"\nAverage DiD Pre-period (weeks 6-14): {pre_did:.2f}")
print(f"Average DiD Post-period (weeks 15-22): {post_did:.2f}")
print(f"Change in DiD (Post - Pre): {post_did - pre_did:.2f}")

did_pvalue = model_clustered.pvalues['switched_branch:after_switch']
did_ci_lower = model_clustered.conf_int().loc['switched_branch:after_switch', 0]
did_ci_upper = model_clustered.conf_int().loc['switched_branch:after_switch', 1]
