import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary table: Average service scores by group and period
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES SUMMARY TABLE")
print("=" * 60)

# Before period (after_switch = 0)
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()

# After period (after_switch = 1)
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nControl Group (No Switch):")
print(f"  Before (weeks 6-14): {before_control:.2f}")
print(f"  After (weeks 15-22): {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")

print(f"\nTreatment Group (Switched):")
print(f"  Before (weeks 6-14): {before_treatment:.2f}")
print(f"  After (weeks 15-22): {after_treatment:.2f}")
print(f"  Change: {after_treatment - before_treatment:.2f}")

# DiD estimate
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

# Now run regression
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Clustered standard errors by branch
print("\n" + "=" * 60)
print("CLUSTERED STANDARD ERRORS (by branch)")
print("=" * 60)

# Get cluster-robust standard errors
cov_type = 'cluster'
cov_kwds = {'groups': df['branch_id']}
model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type=cov_type, cov_kwds=cov_kwds)
print("\nModel 3: DiD with controls and clustered SEs")
print(model3.summary().tables[1])

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check parallel trends: average service score by week for each group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_avg.columns = ['Control (No Switch)', 'Treatment (Switched)']
print("Weekly averages by group:")
print(weekly_avg)

# Calculate the difference between treatment and control each week
weekly_diff = weekly_avg['Treatment (Switched)'] - weekly_avg['Control (No Switch)']
print("\nWeekly difference (Treatment - Control):")
print(weekly_diff)

# Check pre-trends (weeks 6-14)
pre_trend_diff = weekly_diff[weekly_diff.index < 15]
print("\nPre-treatment trend (weeks 6-14):")
print(f"  Average gap: {pre_trend_diff.mean():.2f}")
print(f"  Std dev of gap: {pre_trend_diff.std():.2f}")

# Check if the gap changed significantly after week 14
post_trend_diff = weekly_diff[weekly_diff.index >= 15]
print(f"\nPost-treatment trend (weeks 15-22):")
print(f"  Average gap: {post_trend_diff.mean():.2f}")
print(f"  Std dev of gap: {post_trend_diff.std():.2f}")

# Create a visualization
plt.figure(figsize=(12, 6))
plt.plot(weekly_avg.index, weekly_avg['Control (No Switch)'], 'o-', label='Control (No Switch)', linewidth=2, markersize=8)
plt.plot(weekly_avg.index, weekly_avg['Treatment (Switched)'], 's-', label='Treatment (Switched)', linewidth=2, markersize=8)
