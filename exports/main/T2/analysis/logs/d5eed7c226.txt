import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression (no controls)
model_basic = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Basic DiD Regression (no controls):")
print(model_basic.summary())

# DiD regression with controls
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\n\nDiD Regression with controls:")
print(model_controls.summary())

# Cluster-robust standard errors by branch
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with clustered SEs (by branch):")
print(model_clustered.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Calculate average service score by week and group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg['switched_label'] = weekly_avg['switched_branch'].map({0: 'Control (Old Procedure)', 1: 'Treatment (New Procedure)'})

# Create the plot
fig, ax = plt.subplots(figsize=(12, 7))

# Plot lines for each group
for group in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == group]
    label = 'Control (Old Procedure)' if group == 0 else 'Treatment (New Procedure)'
    color = 'blue' if group == 0 else 'green'
    ax.plot(subset['week'], subset['service_score'], marker='o', linewidth=2.5, 
            markersize=8, label=label, color=color)

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')

# Add shading for post-period
ax.axvspan(15, 22, alpha=0.15, color='gray', label='Post-Switch Period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Scores Over Time: Treatment vs Control Branches', fontsize=14, fontweight='bold')
ax.legend(loc='upper left', fontsize=10)
ax.set_xticks(range(6, 23))
ax.grid(True, alpha=0.3)
ax.set_ylim(45, 55)

plt.tight_layout()
plt.savefig('service_scores_over_time.png', dpi=150)
print("Plot saved as 'service_scores_over_time.png'")

# Show the actual values for the summary
print("\nWeekly averages by group:")
print(weekly_avg.pivot(index='week', columns='switched_branch', values='service_score').round(2))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Pre-period analysis (weeks 6-14)
pre_df = df[df['after_switch'] == 0].copy()

# Check if there's a time trend difference pre-period
pre_df['week_centered'] = pre_df['week'] - 10  # Center around middle of pre-period
pre_df['interaction'] = pre_df['switched_branch'] * pre_df['week_centered']

model_pre = ols('service_score ~ switched_branch + week_centered + interaction', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print("Pre-period trend analysis (testing parallel trends):")
print(model_pre.summary())

# Calculate effect size in practical terms
post_df = df[df['after_switch'] == 1]
pre_avg_control = pre_df[pre_df['switched_branch'] == 0]['service_score'].mean()
post_avg_control = post_df[post_df['switched_branch'] == 0]['service_score'].mean()
pre_avg_treatment = pre_df[pre_df['switched_branch'] == 1]['service_score'].mean()
post_avg_treatment = post_df[post_df['switched_branch'] == 1]['service_score'].mean()

print(f"\nPre-period averages:")
print(f"  Control: {pre_avg_control:.2f}")
print(f"  Treatment: {pre_avg_treatment:.2f}")
print(f"  Difference: {pre_avg_treatment - pre_avg_control:.2f}")

print(f"\nPost-period averages:")
print(f"  Control: {post_avg_control:.2f}")
print(f"  Treatment: {post_avg_treatment:.2f}")
print(f"  Difference: {post_avg_treatment - post_avg_control:.2f}")

diff_in_diff = (post_avg_treatment - pre_avg_treatment) - (post_avg_control - pre_avg_control)
print(f"\nDifference-in-Differences estimate: {diff_in_diff:.2f}")

# Percentage change relative to baseline
pct_change = diff_in_diff / pre_avg_treatment * 100
print(f"Percentage improvement: {pct_change:.1f}%")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Calculate average service score by week and group with std error
weekly_stats = df.groupby(['week', 'switched_branch'])['service_score'].agg(['mean', 'std', 'count']).reset_index()
weekly_stats['se'] = weekly_stats['std'] / np.sqrt(weekly_stats['count'])
weekly_stats['switched_label'] = weekly_stats['switched_branch'].map({0: 'Control (Old Procedure)', 1: 'Treatment (New Procedure)'})

# Create the plot with confidence intervals
fig, ax = plt.subplots(figsize=(12, 7))

colors = ['blue', 'green']
for i, group in enumerate([0, 1]):
    subset = weekly_stats[weekly_stats['switched_branch'] == group]
    label = 'Control (Old Procedure)' if group == 0 else 'Treatment (New Procedure)'
    color = colors[i]
    
    ax.plot(subset['week'], subset['mean'], marker='o', linewidth=2.5, 
            markersize=8, label=label, color=color)
    ax.fill_between(subset['week'], 
                    subset['mean'] - 1.96 * subset['se'],
                    subset['mean'] + 1.96 * subset['se'],
                    alpha=0.2, color=color)

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')

# Add shading for post-period
ax.axvspan(15, 22, alpha=0.15, color='gray', label='Post-Switch Period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score (±95% CI)', fontsize=12)
ax.set_title('Service Scores Over Time: Treatment vs Control Branches', fontsize=14, fontweight='bold')
ax.legend(loc='upper left', fontsize=10)
ax.set_xticks(range(6, 23))
ax.grid(True, alpha=0.3)
ax.set_ylim(44, 56)

plt.tight_layout()
plt.savefig('service_scores_with_ci.png', dpi=150)
print("Plot saved as 'service_scores_with_ci.png'")
