import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Count branches by switch status
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Create a treatment indicator (interaction of switched_branch and after_switch)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate mean service scores by group and period
print("\n=== Mean Service Scores ===")
print("\nBy switched_branch and after_switch:")
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='after_switch', 
                       aggfunc='mean')
print(pivot)

# Calculate the differences
print("\n=== Difference-in-Differences Calculation ===")
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treated_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treated_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"Control group (didn't switch): Before={control_before:.2f}, After={control_after:.2f}, Change={control_after-control_before:.2f}")
print(f"Treated group (switched): Before={treated_before:.2f}, After={treated_after:.2f}, Change={treated_after-treated_before:.2f}")
print(f"\nDifference-in-Differences estimate: {(treated_after-treated_before) - (control_after-control_before):.2f}")

# Run DiD regression
print("\n=== DiD Regression (Simple) ===")
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model.summary())

# Run DiD regression with controls
print("\n=== DiD Regression (With Controls) ===")
model_ctrl = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print(model_ctrl.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Check for heterogeneous effects - does the effect vary by branch characteristics?
# Let's see the distribution of effects by branch
print("=== Service Score Changes by Branch ===")
branch_summary = df.groupby(['branch_id', 'switched_branch']).apply(
    lambda x: pd.Series({
        'before_mean': x[x['after_switch']==0]['service_score'].mean(),
        'after_mean': x[x['after_switch']==1]['service_score'].mean(),
        'change': x[x['after_switch']==1]['service_score'].mean() - x[x['after_switch']==0]['service_score'].mean()
    })
).reset_index()

print("\nTreated branches (switched_branch=1):")
treated = branch_summary[branch_summary['switched_branch']==1]
print(treated[['branch_id', 'before_mean', 'after_mean', 'change']].to_string(index=False))
print(f"\nAverage change for treated: {treated['change'].mean():.2f}")

print("\nControl branches (switched_branch=0):")
control = branch_summary[branch_summary['switched_branch']==0]
print(control[['branch_id', 'before_mean', 'after_mean', 'change']].to_string(index=False))
print(f"\nAverage change for control: {control['change'].mean():.2f}")

# Event study - check effects by week after switch
print("\n=== Event Study: Weekly Effects ===")
df['weeks_since_switch'] = df['week'] - 14  # weeks relative to switch (week 15 is week 1)
event_df = df[df['switched_branch']==1].copy()
event_control = df[df['switched_branch']==0].copy()

# Calculate difference in means by week
weekly_diff = []
for week in sorted(df['week'].unique()):
    treated_mean = event_df[event_df['week']==week]['service_score'].mean()
    control_mean = event_control[event_control['week']==week]['service_score'].mean()
    weekly_diff.append({'week': week, 'treated': treated_mean, 'control': control_mean, 'diff': treated_mean - control_mean})

weekly_diff_df = pd.DataFrame(weekly_diff)
print(weekly_diff_df.to_string(index=False))
