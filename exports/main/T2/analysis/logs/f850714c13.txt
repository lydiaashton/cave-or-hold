import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.regression.linear_model import OLS
from statsmodels.stats.multicomp import pairwise_tukeyhsd

df = pd.read_csv('branches.csv')

# Let's do a cleaner approach: within-branch analysis
# For each branch, calculate the before and after average, then compare changes

branch_summary = df.groupby(['branch_id', 'switched_branch']).agg({
    'service_score': lambda x: x[df.loc[x.index, 'after_switch']==0].mean() if (df.loc[x.index, 'after_switch']==0).any() else np.nan
}).reset_index()

# Actually, let me do this more carefully
before_scores = df[df['after_switch']==0].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
before_scores.columns = ['branch_id', 'switched_branch', 'before_score']

after_scores = df[df['after_switch']==1].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
after_scores.columns = ['branch_id', 'switched_branch', 'after_score']

branch_changes = before_scores.merge(after_scores, on=['branch_id', 'switched_branch'])
branch_changes['change'] = branch_changes['after_score'] - branch_changes['before_score']

print("Branch-level changes:")
print(branch_changes.head(10))

print("\nAverage change by group:")
print(branch_changes.groupby('switched_branch')['change'].agg(['mean', 'std', 'count']))

# T-test comparing changes between groups
from scipy import stats
control_changes = branch_changes[branch_changes['switched_branch']==0]['change']
treatment_changes = branch_changes[branch_changes['switched_branch']==1]['change']

t_stat, p_val = stats.ttest_ind(treatment_changes, control_changes, equal_var=False)
print(f"\nTwo-sample t-test (Welch's):")
print(f"  Treatment mean change: {treatment_changes.mean():.2f}")
print(f"  Control mean change: {control_changes.mean():.2f}")
print(f"  Difference: {treatment_changes.mean() - control_changes.mean():.2f}")
print(f"  t-statistic: {t_stat:.3f}")
print(f"  p-value: {p_val:.3f}")

# Confidence interval for the difference
diff = treatment_changes.mean() - control_changes.mean()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.regression.linear_model import OLS

df = pd.read_csv('branches.csv')

# Create the interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run regression with branch fixed effects (using dummies)
# Also cluster standard errors at branch level

# First, let's create branch dummies properly
X = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
X['after_switch'] = df['after_switch']
X['treatment'] = df['treatment']
X['staffing_index'] = df['staffing_index']
X['visits_index'] = df['visits_index']
X = sm.add_constant(X)

y = df['service_score']

model = OLS(y, X).fit()

# Now cluster standard errors at branch level
def cluster_se(model, cluster_var):
    """Calculate cluster-robust standard errors"""
    from statsmodels.stats.sandwich_covariance import cov_cluster
    clusters = cluster_var.values
    cov_clustered = cov_cluster(model, clusters)
    se_clustered = np.sqrt(np.diag(cov_clustered))
    return se_clustered, cov_clustered

cluster_se_vals, cov_clust = cluster_se(model, df['branch_id'])

# Get coefficient names
coef_names = model.params.index.tolist()

# Find the treatment coefficient
treatment_idx = coef_names.index('treatment')
treatment_coef = model.params['treatment']
treatment_se = cluster_se_vals[treatment_idx]
treatment_t = treatment_coef / treatment_se
# p-value from t-distribution with G-1 clusters (G = number of branches)
from scipy import stats
df_clusters = df['branch_id'].nunique() - 1
treatment_p = 2 * (1 - stats.t.cdf(abs(treatment_t), df_clusters))
treatment_ci_lower = treatment_coef - stats.t.ppf(0.975, df_clusters) * treatment_se
treatment_ci_upper = treatment_coef + stats.t.ppf(0.975, df_clusters) * treatment_se

print("=== Panel Regression with Branch FE and Clustered SE ===")
print(f"Treatment coefficient (DiD estimate): {treatment_coef:.3f}")
print(f"Clustered SE: {treatment_se:.3f}")
print(f"t-statistic: {treatment_t:.3f}")
print(f"p-value: {treatment_p:.3f}")
print(f"95% CI: [{treatment_ci_lower:.3f}, {treatment_ci_upper:.3f}]")
print(f"Degrees of freedom (clusters - 1): {df_clusters}")

# Also check without controls
X_simple = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
X_simple['after_switch'] = df['after_switch']
X_simple['treatment'] = df['treatment']

treatment_t_simple = treatment_coef_simple / treatment_se_simple
treatment_p_simple = 2 * (1 - stats.t.cdf(abs(treatment_t_simple), df_clusters))
