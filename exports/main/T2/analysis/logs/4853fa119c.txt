import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Run DiD regression with robust standard errors
# Model: service_score = beta0 + beta1*treatment + beta2*post + beta3*did + error
model = ols('service_score ~ treatment + post + did', data=df).fit(cov_type='HC3')
print("Difference-in-Differences Regression Results:")
print(model.summary())

# Also run with branch fixed effects (more robust)
# This controls for time-invariant branch characteristics
df_with_dummies = pd.get_dummies(df, columns=['branch_id'], drop_first=True)
fe_model = ols('service_score ~ treatment + post + did + ' + 
               ' + '.join([f'branch_id_{i}' for i in range(1, 34)]), 
               data=df_with_dummies).fit(cov_type='cluster', cov_kwds={'groups': df_with_dummies['branch_id']})
print("\n\nDiD with Branch Fixed Effects (clustered SEs):")
print(fe_model.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Basic DiD regression
model = ols('service_score ~ treatment + post + did', data=df).fit(cov_type='HC3')
print("Basic DiD Regression:")
print(f"  DiD coefficient: {model.params['did']:.2f}")
print(f"  Standard error: {model.bse['did']:.2f}")
print(f"  P-value: {model.pvalues['did']:.3f}")
print(f"  95% CI: [{model.conf_int().loc['did', 0]:.2f}, {model.conf_int().loc['did', 1]:.2f}]")

# DiD with controls
model_controls = ols('service_score ~ treatment + post + did + staffing_index + visits_index', data=df).fit(cov_type='HC3')
print("\nDiD with controls (staffing and visits):")
print(f"  DiD coefficient: {model_controls.params['did']:.2f}")
print(f"  Standard error: {model_controls.bse['did']:.2f}")
print(f"  P-value: {model_controls.pvalues['did']:.3f}")

# Calculate effect size relative to baseline
baseline = df[df['after_switch']==0]['service_score'].mean()
effect_size_pct = (model.params['did'] / baseline) * 100
print(f"\nEffect size: {model.params['did']:.2f} points ({effect_size_pct:.1f}% of baseline)")

# Create visualization
fig, ax = plt.subplots(figsize=(10, 6))

# Weekly averages by group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
switched = weekly_avg[weekly_avg['switched_branch']==1]
not_switched = weekly_avg[weekly_avg['switched_branch']==0]

ax.plot(not_switched['week'], not_switched['service_score'], 'o-', label='Control (No Switch)', linewidth=2, markersize=6)
ax.plot(switched['week'], switched['service_score'], 's-', label='Treatment (Switched)', linewidth=2, markersize=6)

# Add vertical line at switch point
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch occurred')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Service Score', fontsize=12)
ax.set_title('Service Scores by Group Over Time', fontsize=14)
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

df['did'] = df['treatment'] * df['post']

# Power analysis - what's the minimum detectable effect?
# Using the observed standard deviation and sample size
n_treatment = df[df['switched_branch']==1].shape[0]
n_control = df[df['switched_branch']==0].shape[0]
sd = df['service_score'].std()

# For a two-sample test with alpha=0.05, power=0.80
# MDE ≈ (t_alpha + t_beta) * sd * sqrt(1/n1 + 1/n2)
# Using approximate values
alpha = 0.05
power = 0.80
z_alpha = stats.norm.ppf(1 - alpha/2)  # 1.96
z_beta = stats.norm.ppf(power)  # 0.84

# But this is DiD, so we need to account for the variance of the difference-in-differences
# The standard error of DiD is about 1.64
se_did = 1.64
mde = (z_alpha + z_beta) * se_did
print(f"Minimum detectable effect (80% power, 5% significance): {mde:.2f} points")
