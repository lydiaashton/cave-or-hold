import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Check parallel trends - compare pre-treatment trends
pre_data = df[df['week'] < 15].copy()

# Average service score by week and treatment group
pre_trends = pre_data.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print("Pre-treatment trends (average service score by week):")
print(pre_trends)

# Calculate week-over-week changes for each group
print("\nWeek-over-week changes in pre-treatment period:")
for group in [0, 1]:
    group_data = pre_trends[group]
    changes = group_data.diff()
    print(f"\nSwitched={group}:")
    print(changes)

# Simple DiD regression
print("\n" + "="*60)
print("DIFFERENCE-IN-DIFFERENCES ANALYSIS")
print("="*60)

# Model 1: Basic DiD (no controls)
model1 = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with controls (staffing_index, visits_index)")
print(model2.summary().tables[1])

# Model 3: DiD with branch fixed effects
model3 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index + C(branch_id)', data=df).fit()
print("\nModel 3: DiD with branch fixed effects")
# Just show the DiD coefficient
did_coef = model3.params['did']
did_pval = model3.pvalues['did']
print(f"DID coefficient: {did_coef:.4f} (p-value: {did_pval:.4f})")

# Clustered standard errors by branch (more appropriate for panel data)
print("\n" + "="*60)
print("CLUSTERED STANDARD ERRORS (by branch)")
print("="*60)

# Use cluster-robust standard errors
model1_cluster = model1.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print("\nModel 1 with clustered SE:")
print(model1_cluster.summary().tables[1])

model2_cluster = model2.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print("\nModel 2 with clustered SE:")
print(model2_cluster.summary().tables[1])
