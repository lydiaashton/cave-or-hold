df = pd.read_csv('branches.csv')

df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary table: Average service scores by group and period
print("=" * 60)
print("AVERAGE SERVICE SCORES BY GROUP AND PERIOD")

summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(summary)
print()

# Calculate DiD manually
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
pre_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
post_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("Control group (didn't switch):")
print(f"  Pre-switch avg: {pre_control:.2f}")
print(f"  Post-switch avg: {post_control:.2f}")
print(f"  Change: {post_control - pre_control:.2f}")
print()
print("Treatment group (switched):")
print(f"  Pre-switch avg: {pre_treatment:.2f}")
print(f"  Post-switch avg: {post_treatment:.2f}")
print(f"  Change: {post_treatment - pre_treatment:.2f}")
print()
print("Difference-in-Differences estimate:")
did = (post_treatment - pre_treatment) - (post_control - pre_control)
print(f"  DiD = {did:.2f}")
print()

# Weekly trends by group

print("WEEKLY TRENDS BY GROUP")

import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load data
df = pd.read_csv('branches.csv')

df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES REGRESSION")
print("=" * 60)

# Basic DiD model
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 1: Basic DiD (with branch-clustered SEs)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 2: DiD with controls (staffing_index, visits_index)")
print(model2.summary().tables[1])

# Check pre-trends: interaction of switched_branch with week dummies (pre-period only)

pre_period = df[df['week'] < 15].copy()
pre_period['week_cat'] = pre_period['week'].astype('category')

# Regress service_score on switched_branch, week fixed effects, and their interaction
# Using week 6 as reference
model_pretrend = ols('service_score ~ switched_branch + C(week) + switched_branch:C(week)', data=pre_period).fit(cov_type='cluster', cov_kwds={'groups': pre_period['branch_id']})
print("\nPre-trend test (interaction terms show if trends differ by week):")
print(model_pretrend.summary().tables[1])

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from scipy import stats
from statsmodels.formula.api import ols

# Load data
df = pd.read_csv('branches.csv')

# Create relative week indicators (week 14 is reference, just before switch)
df['relative_week'] = df['week'] - 14
df['relative_week'] = df['relative_week'].clip(-8, 8)

# Create dummies with valid names (prefix with 'w' to avoid issues)
for rw in range(-8, 9):
    if rw != 0:
        sign = 'm' if rw < 0 else 'p'
        abs_rw = abs(rw)
        df[f'w_{sign}{abs_rw}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Event study regression
event_vars = [f'w_{"m" if rw < 0 else "p"}{abs(rw)}' for rw in range(-8, 9) if rw != 0]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=" * 60)

import pandas as pd

# Load data
df = pd.read_csv('branches.csv')

# Final DiD analysis with robust standard errors
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main DiD model with controls
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=" * 70)
print("FINAL DIFFERENCE-IN-DIFFERENCES ANALYSIS")
print("=" * 70)
print()
print("Model: service_score ~ switched_branch + after_switch + treatment + controls")
print("Standard errors clustered by branch")
print()
print(model.summary().tables[1])

# Key results
did_coef = model.params['treatment']
did_se = model.bse['treatment']
did_pval = model.pvalues['treatment']
did_ci_lower = did_coef - 1.96 * did_se
did_ci_upper = did_coef + 1.96 * did_se

print("\n" + "=" * 70)
print("KEY FINDINGS")
print("=" * 70)
print(f"\nTreatment Effect (DiD estimate): {did_coef:.2f} points")
print(f"95% Confidence Interval: [{did_ci_lower:.2f}, {did_ci_upper:.2f}]")
print(f"P-value: {did_pval:.4f}")
print(f"Statistically significant: {'Yes' if did_pval < 0.05 else 'No'}")

# Calculate effect size relative to baseline
baseline = df[df['after_switch'] == 0]['service_score'].mean()
effect_size_pct = (did_coef / baseline) * 100
print(f"\nBaseline average service score: {baseline:.2f}")
print(f"Effect size: {effect_size_pct:.1f}% improvement")
