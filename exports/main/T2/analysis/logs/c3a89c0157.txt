import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# === 1. Summary Statistics by Group and Period ===
print("=== Service Score by Group and Period ===")
pivot_table = df.pivot_table(
    values='service_score', 
    index='switched_branch', 
    columns='after_switch', 
    aggfunc=['mean', 'std', 'count']
)
print(pivot_table)

# Calculate the DiD estimate manually
print("\n=== Manual DiD Calculation ===")
# Control group (switched_branch=0)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_change = control_after - control_before

# Treatment group (switched_branch=1)
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_change = treat_after - treat_before

did_estimate = treat_change - control_change

print(f"Control group (non-switched): Before={control_before:.2f}, After={control_after:.2f}, Change={control_change:.2f}")
print(f"Treatment group (switched): Before={treat_before:.2f}, After={treat_after:.2f}, Change={treat_change:.2f}")
print(f"\nDifference-in-Differences Estimate: {did_estimate:.3f}")

# === 2. Regression Analysis ===
print("\n=== Regression Analysis ===")

# Model 1: Basic DiD
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Model 3: With branch fixed effects
print("\n=== Model with Branch Fixed Effects ===")
df['branch_id_str'] = df['branch_id'].astype(str)
model3 = ols('service_score ~ after_switch + treatment + staffing_index + visits_index + C(branch_id)', data=df).fit()
print(model3.summary().tables[1])
