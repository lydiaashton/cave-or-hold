import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Basic DiD: average scores by group and period
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES ANALYSIS")
print("=" * 60)

# Calculate means for the 4 groups
before_control = df[(df['treated']==0) & (df['post']==0)]['service_score'].mean()
before_treated = df[(df['treated']==1) & (df['post']==0)]['service_score'].mean()
after_control = df[(df['treated']==0) & (df['post']==1)]['service_score'].mean()
after_treated = df[(df['treated']==1) & (df['post']==1)]['service_score'].mean()

print("\nAverage Service Scores by Group and Period:")
print(f"  Control (not switched), Before: {before_control:.2f}")
print(f"  Treatment (switched), Before:   {before_treated:.2f}")
print(f"  Control (not switched), After:  {after_control:.2f}")
print(f"  Treatment (switched), After:    {after_treated:.2f}")

# Calculate the DiD estimator manually
diff_before = before_treated - before_control
diff_after = after_treated - after_control
did_estimator = diff_after - diff_before

print(f"\nDifference before (treated - control): {diff_before:.2f}")
print(f"Difference after (treated - control):  {diff_after:.2f}")
print(f"\nDiD Estimator (effect of new procedure): {did_estimator:.2f}")

# Now run regression
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD regression
model1 = ols('service_score ~ treated + post + DiD', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ treated + post + DiD + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Clustered standard errors by branch
print("\n" + "=" * 60)
print("CLUSTERED STANDARD ERRORS (by branch)")
print("=" * 60)

from statsmodels.stats.multicomp import pairwise_tukeyhsd

# Get clustered standard errors
cov_cluster = model2.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print("\nModel 2 with clustered SE:")
print(cov_cluster.summary().tables[1])

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check parallel trends: average service score by week for each group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()

print("Weekly averages by group:")
print(weekly_avg.pivot(index='week', columns='switched_branch', values='service_score'))

# Simple test: regression of service score on week, treated, and their interaction (pre-period only)
pre_data = df[df['week'] < 15].copy()
pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()
pre_model = ols('service_score ~ week_centered * switched_branch', data=pre_data).fit()
print("\n" + "=" * 60)
print("PARALLEL TRENDS TEST (Pre-period only, weeks 6-14)")
print("=" * 60)
print(pre_model.summary().tables[1])

# Also check post-period trends
post_data = df[df['week'] >= 15].copy()
post_model = ols('service_score ~ week * switched_branch', data=post_data).fit()
print("\n" + "=" * 60)
print("POST-PERIOD TRENDS (weeks 15-22)")
print("=" * 60)
print(post_model.summary().tables[1])

# Event study: estimate effect for each week relative to switch
print("\n" + "=" * 60)
print("EVENT STUDY ANALYSIS")
print("=" * 60)

# Create week dummies relative to week 14 (last pre-switch week as reference)
df_event = df.copy()
df_event['week_rel'] = df_event['week'] - 14  # week 14 = 0, week 15 = 1, etc.

# Create dummies for each week (excluding reference week 14)
for w in range(-8, 9):  # weeks -8 to 8 relative to week 14
    if w != 0:  # skip reference week
        df_event[f'week_{w}'] = ((df_event['week_rel'] == w) & (df_event['switched_branch'] == 1)).astype(int)

# Build formula
control_weeks = [f'week_{w}' for w in range(-8, 9) if w != 0]
formula = f"service_score ~ switched_branch + {' + '.join(control_weeks)}"

event_model = ols(formula, data=df_event).fit()
print("\nEvent study coefficients (interaction of treated x week dummies):")
print(event_model.summary().tables[1])

# Get just the week interaction coefficients
coef_df = event_model.summary().tables[1]
print("\nKey coefficients for interpretation:")
for idx, row in coef_df.iterrows():
    if 'week_' in str(row[0]):
        print(f"  {row[0]}: {row[1]:.2f} (p={row[3]:.3f})")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Simpler event study: create relative week dummies with valid names
df_event = df.copy()
df_event['week_rel'] = df_event['week'] - 14  # week 14 = 0 (reference)

# Create dummies with valid names
for w in range(-8, 9):
    if w != 0:
        col_name = f'w{abs(w)}_{"neg" if w < 0 else "pos"}'
        df_event[col_name] = ((df_event['week_rel'] == w) & (df_event['switched_branch'] == 1)).astype(int)

# Build formula - treatment main effect + all week interactions
control_vars = [c for c in df_event.columns if c.startswith('w')]
formula = f"service_score ~ switched_branch + {' + '.join(control_vars)}"

event_model = ols(formula, data=df_event).fit()

print("=" * 60)
print("EVENT STUDY: Effect by Week Relative to Switch")
print("(Reference: Week 14, last pre-switch week)")
print("=" * 60)

# Extract and display coefficients
results = []
for w in range(-8, 9):
    if w == 0:
        results.append((w, 0.0, 1.0, "Reference"))
    else:
        col_name = f'w{abs(w)}_{"neg" if w < 0 else "pos"}'
        coef = event_model.params[col_name]
        pval = event_model.pvalues[col_name]
        results.append((w, coef, pval, ""))

print(f"\n{'Week Rel':>10} {'Coefficient':>12} {'P-value':>10} {'Signif':>8}")
print("-" * 45)
for w, coef, pval, note in results:
    week_num = w + 14
    signif = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"{week_num:>10} {coef:>12.2f} {pval:>10.3f} {signif:>8} {note}")

print("\nNote: Coefficients show the difference-in-differences estimate")
print("      for each week (treated vs control, relative to week 14)")

# Pre-trends test: are pre-period coefficients jointly zero?
pre_cols = [f'w{abs(w)}_neg' for w in range(-8, 0)]
print("\n" + "=" * 60)
print("PRE-TRENDS F-TEST (weeks 6-14)")
print("=" * 60)
f_test = event_model.f_test(pre_cols)
print(f"F-statistic: {f_test.statistic[0][0]:.2f}")
print(f"P-value: {f_test.pvalue:.3f}")
if f_test.pvalue > 0.05:
    print("Conclusion: No significant pre-trends detected (parallel trends plausible)")
else:
    print("Conclusion: Significant pre-trends detected (parallel trends violated)")

# Post-period average effect
print("\n" + "=" * 60)
print("AVERAGE POST-PERIOD EFFECT (weeks 15-22)")
print("=" * 60)
post_cols = [f'w{abs(w)}_pos' for w in range(1, 9)]
# Average the post-period coefficients
post_coefs = [event_model.params[c] for c in post_cols]
avg_post_effect = np.mean(post_coefs)
print(f"Average post-period DiD effect: {avg_post_effect:.2f} points")

# Get standard error for average using covariance matrix
post_cov = event_model.cov_params().loc[post_cols, post_cols]
weights = np.ones(len(post_cols)) / len(post_cols)
avg_se = np.sqrt(weights @ post_cov @ weights)
print(f"Standard error: {avg_se:.2f}")
t_stat = avg_post_effect / avg_se
print(f"t-statistic: {t_stat:.2f}")
