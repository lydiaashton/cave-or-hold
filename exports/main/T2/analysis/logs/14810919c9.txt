import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# Get means more clearly
means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean()
print("Mean service scores:")
print(means)

# Treatment = switched_branch=1, Control = switched_branch=0
# Before = after_switch=0, After = after_switch=1

treatment_before = means.loc[(1, 0)]
treatment_after = means.loc[(1, 1)]
control_before = means.loc[(0, 0)]
control_after = means.loc[(0, 1)]

print(f"\nTreatment (switched) before: {treatment_before:.3f}")
print(f"Treatment (switched) after: {treatment_after:.3f}")
print(f"Control (non-switched) before: {control_before:.3f}")
print(f"Control (non-switched) after: {control_after:.3f}")

# Changes over time
treatment_change = treatment_after - treatment_before
control_change = control_after - control_before

print(f"\nTreatment change: {treatment_change:.3f}")
print(f"Control change: {control_change:.3f}")

# Difference-in-differences
did = treatment_change - control_change
print(f"\nDifference-in-Differences: {did:.3f}")
print(f"This means the new procedure is associated with a {did:.3f} point INCREASE in service score")

# Let's also run a regression to get statistical significance
import statsmodels.api as sm
import statsmodels.formula.api as smf

# Simple DiD regression
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['treatment_post'] = df['treatment'] * df['post']

model = smf.ols('service_score ~ treatment + post + treatment_post', data=df).fit()
print("\n\nRegression results (DiD model):")
print(model.summary())

# Cluster-robust standard errors by branch
model_clustered = smf.ols('service_score ~ treatment + post + treatment_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nRegression with cluster-robust SE (by branch):")
print(model_clustered.summary())

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create treatment and post variables
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['treatment_post'] = df['treatment'] * df['post']

# Model with branch and week fixed effects
# This is the most robust DiD specification
model_fe = smf.ols('service_score ~ treatment_post + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Model with Branch and Week Fixed Effects:")
print(model_fe.summary())

# Also control for staffing and visits
model_controls = smf.ols('service_score ~ treatment_post + staffing_index + visits_index + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nModel with Branch/Week FE + Controls:")
print(model_controls.summary())

# Let's also look at the effect by week after the switch to see if it's consistent
df['weeks_since_switch'] = df['week'] - 14  # weeks since switch (week 15 = 1, week 16 = 2, etc.)
df['weeks_since_switch'] = df['weeks_since_switch'].apply(lambda x: max(0, x))  # 0 before switch

# Effect by week after switch
post_df = df[df['week'] >= 15].copy()
weekly_effects = post_df.groupby('week').apply(
    lambda x: smf.ols('service_score ~ treatment', data=x).fit().params.get('treatment', np.nan)
)
print("\n\nTreatment effect by week (after switch):")
print(weekly_effects)

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create treatment and post variables
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['treatment_post'] = df['treatment'] * df['post']

# Model with branch and week fixed effects - extract just the treatment_post coefficient
model_fe = smf.ols('service_score ~ treatment_post + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Treatment effect with Branch and Week Fixed Effects:")
print(f"treatment_post coefficient: {model_fe.params['treatment_post']:.3f}")
print(f"Standard error: {model_fe.bse['treatment_post']:.3f}")
print(f"P-value: {model_fe.pvalues['treatment_post']:.6f}")
print(f"95% CI: [{model_fe.conf_int().loc['treatment_post'][0]:.3f}, {model_fe.conf_int().loc['treatment_post'][1]:.3f}]")

# Model with controls
model_controls = smf.ols('service_score ~ treatment_post + staffing_index + visits_index + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nTreatment effect with Branch/Week FE + Controls:")
print(f"treatment_post coefficient: {model_controls.params['treatment_post']:.3f}")
print(f"Standard error: {model_controls.bse['treatment_post']:.3f}")
print(f"P-value: {model_controls.pvalues['treatment_post']:.6f}")

# Let's also do an event study to see the effect over time
df['relative_week'] = df['week'] - 15  # week 15 = 0 (first week after switch)

# Create indicators for each week relative to switch
for w in range(-9, 8):  # weeks -9 to +7 relative to week 15
    df[f'rel_week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

event_model = smf.ols(f'service_score ~ {event_study_vars} + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
