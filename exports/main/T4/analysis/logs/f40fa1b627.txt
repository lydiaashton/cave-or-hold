import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Create a treatment group indicator (interaction of switched_branch and after_switch)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate means for the 4 groups in a DiD design
print("=== Difference-in-Differences Setup ===")
print("\nMean service scores by group:")

# Before switch (after_switch=0)
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
before_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()

# After switch (after_switch=1)
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
after_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"Control group (didn't switch):")
print(f"  Before: {before_control:.2f}")
print(f"  After:  {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")

print(f"\nTreatment group (switched):")
print(f"  Before: {before_treatment:.2f}")
print(f"  After:  {after_treatment:.2f}")
print(f"  Change: {after_treatment - before_treatment:.2f}")

# DiD estimate
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\n=== DiD Estimate: {did_estimate:.2f} ===")
print("This is the estimated effect of the new procedure on service score")

# Check parallel trends - compare trends before the switch
print("\n=== Pre-trends Check (weeks 6-14) ===")
pre_trends = df[df['after_switch']==0].groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
print(pre_trends)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a DiD regression with cluster-robust standard errors
# Model: service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is our DiD estimate

model = ols('service_score ~ switched_branch + after_switch + switched_branch:after_switch', data=df).fit()

# Get cluster-robust standard errors
cov_cluster = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("=== DiD Regression Results ===")
print(cov_cluster.summary())

fig, ax = plt.subplots(figsize=(12, 6))

for group in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == group]
    label = 'Treatment (New Procedure)' if group == 1 else 'Control (Old Procedure)'
    color = '#2E86AB' if group == 1 else '#A23B72'
    ax.plot(subset['week'], subset['service_score'], marker='o', linewidth=2, markersize=6, 
            label=label, color=color)

# Add vertical line at switch point
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch implemented (week 15)')

ax.set_title('Service Scores by Group Over Time', fontsize=14, fontweight='bold')
ax.legend(fontsize=11)

ax.set_xticks(range(6, 23))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a regression with branch fixed effects (more robust DiD)
# This controls for all time-invariant branch characteristics
df['week_demeaned'] = df.groupby('branch_id')['week'].transform(lambda x: x - x.mean())

# Two-way fixed effects model (branch + week FE)
# We'll use the interaction term as our treatment
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model with branch fixed effects
model_fe = ols('service_score ~ treatment + C(branch_id)', data=df).fit()
cov_fe = model_fe.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("=== Branch Fixed Effects Model ===")

print(f"  95% CI: [{cov_fe.conf_int().loc['treatment'][0]:.2f}, {cov_fe.conf_int().loc['treatment'][1]:.2f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Two-way fixed effects model (branch + week FE)
model_twfe = ols('service_score ~ treatment + C(branch_id) + C(week)', data=df).fit()
cov_twfe = model_twfe.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Extract treatment coefficient
treatment_coef = cov_twfe.params['treatment']
treatment_se = cov_twfe.bse['treatment']
treatment_p = cov_twfe.pvalues['treatment']

print("\n=== Simple DiD (for comparison) ===")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Two-way fixed effects model
model_twfe = ols('service_score ~ treatment + C(branch_id) + C(week)', data=df).fit()

# Get cluster-robust covariance matrix
cov_matrix = cov_cluster(model_twfe, df['branch_id'])

# Get standard errors from the covariance matrix
param_names = model_twfe.params.index
treatment_idx = list(param_names).index('treatment')
treatment_se = np.sqrt(cov_matrix[treatment_idx, treatment_idx])

treatment_coef = model_twfe.params['treatment']
treatment_t = treatment_coef / treatment_se
# Approximate p-value using normal distribution
from scipy import stats
treatment_p = 2 * (1 - stats.norm.cdf(abs(treatment_t)))
treatment_ci_low = treatment_coef - 1.96 * treatment_se
treatment_ci_high = treatment_coef + 1.96 * treatment_se

print("=== Two-Way Fixed Effects Results (Branch + Week) ===")
print(f"Treatment effect: {treatment_coef:.3f} points")
print(f"Standard error (clustered): {treatment_se:.3f}")
print(f"t-statistic: {treatment_t:.2f}")
print(f"95% CI: [{treatment_ci_low:.2f}, {treatment_ci_high:.2f}]")
print(f"p-value: {treatment_p:.6f}")

# Simple DiD for comparison
model_simple = ols('service_score ~ switched_branch + after_switch + switched_branch:after_switch', data=df).fit()
cov_matrix_simple = cov_cluster(model_simple, df['branch_id'])
simple_idx = list(model_simple.params.index).index('switched_branch:after_switch')
simple_se = np.sqrt(cov_matrix_simple[simple_idx, simple_idx])
simple_coef = model_simple.params['switched_branch:after_switch']
simple_t = simple_coef / simple_se
simple_p = 2 * (1 - stats.norm.cdf(abs(simple_t)))

print(f"\n=== Simple DiD (for comparison) ===")
print(f"Treatment effect: {simple_coef:.3f} points")
print(f"Standard error (clustered): {simple_se:.3f}")
print(f"t-statistic: {simple_t:.2f}")
print(f"p-value: {simple_p:.6f}")
