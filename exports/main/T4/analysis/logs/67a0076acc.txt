import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import statsmodels.stats.api as sms

# Load the data
df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD with branch fixed effects
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df, branch_dummies], axis=1)

# Build formula with fixed effects
fe_cols = [c for c in df_with_fe.columns if c.startswith('branch')]
formula = 'service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index + ' + ' + '.join(fe_cols)

model_fe = ols(formula, data=df_with_fe).fit(cov_type='cluster', cov_kwds={'groups': df_with_fe['branch_id']})
print("DiD Regression with Branch Fixed Effects (clustered SEs):")
print(model_fe.summary())

# Extract the treatment effect
treatment_coef = model_fe.params['treatment']
treatment_se = model_fe.bse['treatment']
treatment_pval = model_fe.pvalues['treatment']
treatment_ci = model_fe.conf_int().loc['treatment']

print(f"\n\nTreatment Effect: {treatment_coef:.2f} points")
print(f"Standard Error: {treatment_se:.2f}")
print(f"95% CI: [{treatment_ci[0]:.2f}, {treatment_ci[1]:.2f}]")
print(f"P-value: {treatment_pval:.3f}")

# Also calculate effect size relative to baseline
baseline_score = df[df['after_switch'] == 0]['service_score'].mean()
print(f"\nBaseline average service score: {baseline_score:.2f}")
print(f"Effect as % of baseline: {treatment_coef/baseline_score*100:.1f}%")
