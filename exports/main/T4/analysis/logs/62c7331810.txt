import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with cluster-robust standard errors at branch level
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("DiD Regression with Cluster-Robust SE (by branch):")
print(model_clustered.summary())

# Check parallel trends - plot pre-period trends
pre_period = df[df['week'] < 15]
pre_means = pre_period.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

post_period = df[df['week'] >= 15]
post_means = post_period.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

all_means = pd.concat([pre_means, post_means])

# Create plot
plt.figure(figsize=(10, 6))
for switched in [0, 1]:
    subset = all_means[all_means['switched_branch'] == switched]
    label = 'Treatment (switched)' if switched == 1 else 'Control (not switched)'
    plt.plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2)

plt.axvline(x=14.5, color='red', linestyle='--', label='Switch point (week 15)')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Scores Over Time: Treatment vs Control Groups')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('did_plot.png', dpi=150)
print("\nPlot saved as did_plot.png")

# Print the pre-period means for parallel trends check
print("\nPre-period means by week:")
print(pre_means.pivot(index='week', columns='switched_branch', values='service_score'))

# Post-period means
print("\nPost-period means by week:")
print(post_means.pivot(index='week', columns='switched_branch', values='service_score'))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run the main model
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get the treatment effect and confidence interval
treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_ci_lower = treatment_coef - 1.96 * treatment_se
treatment_ci_upper = treatment_coef + 1.96 * treatment_se

print(f"Treatment Effect: {treatment_coef:.2f}")
print(f"95% CI: [{treatment_ci_lower:.2f}, {treatment_ci_upper:.2f}]")
print(f"P-value: {model.pvalues['treatment']:.4f}")

# Calculate effect size relative to baseline
baseline_score = df[df['after_switch'] == 0]['service_score'].mean()
print(f"\nBaseline (pre-period) average service score: {baseline_score:.2f}")
print(f"Effect as % of baseline: {(treatment_coef / baseline_score) * 100:.1f}%")

# Check for heterogeneous effects - does the effect vary by branch characteristics?
# Let's see if there's variation across branches
branch_effects = df.groupby('branch_id').apply(
    lambda x: x.groupby('after_switch')['service_score'].mean().diff().iloc[-1] if len(x.groupby('after_switch')) > 1 else np.nan
)

# For treatment branches only
treatment_branches = df[df['switched_branch'] == 1]['branch_id'].unique()
control_branches = df[df['switched_branch'] == 0]['branch_id'].unique()

# Calculate branch-level pre-post changes
def calc_change(branch_id):
    branch_data = df[df['branch_id'] == branch_id]
    pre = branch_data[branch_data['after_switch'] == 0]['service_score'].mean()
    post = branch_data[branch_data['after_switch'] == 1]['service_score'].mean()
    return post - pre

treatment_changes = [calc_change(b) for b in treatment_branches]
control_changes = [calc_change(b) for b in control_branches]

print(f"\nTreatment branches (n={len(treatment_branches)}):")
print(f"  Mean change: {np.mean(treatment_changes):.2f}")
print(f"  Std dev: {np.std(treatment_changes):.2f}")
print(f"  Min: {np.min(treatment_changes):.2f}, Max: {np.max(treatment_changes):.2f}")

print(f"\nControl branches (n={len(control_branches)}):")
print(f"  Mean change: {np.mean(control_changes):.2f}")
print(f"  Std dev: {np.std(control_changes):.2f}")
print(f"  Min: {np.min(control_changes):.2f}, Max: {np.max(control_changes):.2f}")

# Event study - check effects by week relative to switch
df['weeks_since_switch'] = df['week'] - 15
df['weeks_since_switch'] = df['weeks_since_switch'].clip(-9, 7)  # cap at -9 to 7

# Create week dummies interacted with treatment
event_study_results = []
for w in range(-9, 8):
    if w == -1:  # reference period
        continue
    df[f'week_{w}'] = ((df['weeks_since_switch'] == w) & (df['switched_branch'] == 1)).astype(int)

event_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
event_formula = f'service_score ~ switched_branch + after_switch + {" + ".join(event_vars)} + staffing_index + visits_index'
event_model = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n\nEvent Study Results (relative to week 14 as baseline):")
for w in range(-9, 8):
    if w == -1:
        continue
    var_name = f'week_{w}'
    if var_name in event_model.params:
        coef = event_model.params[var_name]
        se = event_model.bse[var_name]
        pval = event_model.pvalues[var_name]
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        week_label = f"Week {15+w}" if w >= 0 else f"Week {15+w}"
        print(f"  {week_label}: {coef:.2f} (SE: {se:.2f}) {sig}")

import pandas as pd
import numpy as np

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run the main model
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get the treatment effect and confidence interval
treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_ci_lower = treatment_coef - 1.96 * treatment_se
treatment_ci_upper = treatment_coef + 1.96 * treatment_se

print(f"Treatment Effect: {treatment_coef:.2f}")
print(f"95% CI: [{treatment_ci_lower:.2f}, {treatment_ci_upper:.2f}]")
print(f"P-value: {model.pvalues['treatment']:.6f}")

# Calculate effect size relative to baseline
baseline_score = df[df['after_switch'] == 0]['service_score'].mean()
print(f"\nBaseline (pre-period) average service score: {baseline_score:.2f}")
print(f"Effect as % of baseline: {(treatment_coef / baseline_score) * 100:.1f}%")

# Event study - check effects by week relative to switch
df['weeks_since_switch'] = df['week'] - 15

# Create week dummies interacted with treatment - use proper variable names
event_vars = []
for w in range(-9, 8):
    if w == -1:  # reference period (week 14)
        continue
    var_name = f'wk{abs(w)}_{"neg" if w < 0 else "pos"}'
    df[var_name] = ((df['weeks_since_switch'] == w) & (df['switched_branch'] == 1)).astype(int)
    event_vars.append(var_name)

event_formula = f'service_score ~ switched_branch + after_switch + {" + ".join(event_vars)} + staffing_index + visits_index'
