import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Verify branch counts
switched_branches = df[df['switched_branch'] == 1]['branch_id'].nunique()
control_branches = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"Switched branches (treatment): {switched_branches}")
print(f"Control branches: {control_branches}")
print(f"Total branches: {switched_branches + control_branches}")

# Check weeks per branch
weeks_per_branch = df.groupby('branch_id')['week'].nunique()
print(f"\nWeeks per branch - min: {weeks_per_branch.min()}, max: {weeks_per_branch.max()}")

# Calculate average service scores by group and period
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("\nService scores by group and period:")
print(summary)

# Calculate the difference-in-differences manually
# Treatment group: before and after
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nTreatment (switched) - Before: {treatment_before:.2f}, After: {treatment_after:.2f}, Change: {treatment_after - treatment_before:.2f}")
print(f"Control (not switched) - Before: {control_before:.2f}, After: {control_after:.2f}, Change: {control_after - control_before:.2f}")

did_effect = (treatment_after - treatment_before) - (control_after - control_before)
print(f"\nDifference-in-Differences estimate: {did_effect:.2f}")

import pandas as pd
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Basic Difference-in-Differences Regression ===")
print(model1.summary())

# DiD with controls (staffing and visits)
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Regression with Controls ===")
print(model2.summary())

# Check pre-trends: compare trends before week 15
pre_data = df[df['week'] < 15].copy()
pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()

# Test if trends were parallel before the switch
pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create trend data by week and group
trend_data = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
trend_data['group'] = trend_data['switched_branch'].map({0: 'Control (Old Procedure)', 1: 'Treatment (New Procedure)'})

# Plot trends
plt.figure(figsize=(10, 6))
for group_name, group_data in trend_data.groupby('group'):
    plt.plot(group_data['week'], group_data['service_score'], marker='o', linewidth=2, label=group_name)

plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Score Trends: Treatment vs Control Groups')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("Saved trend plot to service_score_trends.png")

# Calculate confidence interval for the treatment effect
from statsmodels.formula.api import ols
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se

print(f"\nTreatment Effect: {treatment_coef:.2f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {model.pvalues['treatment']:.4f}")

# Effect size relative to baseline
control_before_mean = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
effect_size_pct = (treatment_coef / control_before_mean) * 100
print(f"\nEffect size as % of baseline control score: {effect_size_pct:.1f}%")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate confidence interval for the treatment effect
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se

print(f"Treatment Effect: {treatment_coef:.2f}")
print(f"Standard Error: {treatment_se:.2f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {model.pvalues['treatment']:.4f}")

# Effect size relative to baseline
control_before_mean = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
effect_size_pct = (treatment_coef / control_before_mean) * 100
print(f"\nEffect size as % of baseline control score: {effect_size_pct:.1f}%")

# Also check practical significance - what does a 2 point change mean?
print(f"\nBaseline control group average (before): {control_before_mean:.2f}")
