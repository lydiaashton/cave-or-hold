from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check parallel trends: compare pre-treatment trends between groups
# Create week dummies interacted with treatment for pre-period only
df_pre = df[df['week'] < 15].copy()

# Test if pre-trends are different
df_pre['branch_id'] = df_pre['branch_id'].astype(str)
pre_trend_model = ols('service_score ~ switched_branch * C(week) + C(branch_id)', data=df_pre).fit(cov_type='cluster', cov_kwds={'groups': df_pre['branch_id']})

# Check if the interaction terms (different trends) are jointly significant
# Get the week interaction coefficients
week_cols = [col for col in pre_trend_model.params.index if 'switched_branch:C(week)' in col]
print("Pre-treatment trend test:")
print(f"Number of interaction terms: {len(week_cols)}")

# F-test for joint significance
from statsmodels.stats.anova import anova_lm
# Actually, let's just look at the coefficients
print("\nPre-treatment interaction coefficients (should be near 0 if parallel trends hold):")
for col in week_cols:
    coef = pre_trend_model.params[col]
    se = pre_trend_model.bse[col]
    pval = pre_trend_model.pvalues[col]
    print(f"{col}: {coef:.2f} (SE: {se:.2f}, p: {pval:.3f})")

# Also calculate effect size relative to baseline
baseline_score = df[df['after_switch']==0]['service_score'].mean()

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators
# Week 14 is the last pre-treatment week, so relative week = week - 14
df['relative_week'] = df['week'] - 14

# Create indicators for each relative week (omit week 0 as reference)
# relative_week ranges from -8 (week 6) to 8 (week 22)
for rw in range(-8, 9):
    if rw != 0:
        df[f'rel_week_{rw}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression with branch fixed effects
df['branch_id_str'] = df['branch_id'].astype(str)

# Build formula dynamically
rel_week_vars = [f'rel_week_{rw}' for rw in range(-8, 9) if rw != 0]
formula = f"service_score ~ after_switch + {' + '.join(rel_week_vars)} + staffing_index + visits_index + C(branch_id_str)"

event_study = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the relative week indicators
event_coeffs = []
event_se = []
event_pval = []
for rw in range(-8, 9):
    if rw == 0:
        event_coeffs.append(0)  # reference category
        event_se.append(0)
        event_pval.append(1)
    else:
        col = f'rel_week_{rw}'
        event_coeffs.append(event_study.params[col])
        event_se.append(event_study.bse[col])
        event_pval.append(event_study.pvalues[col])

# Create event study plot
plt.figure(figsize=(12, 6))
relative_weeks = list(range(-8, 9))
plt.errorbar(relative_weeks, event_coeffs, yerr=[1.96*s for s in event_se],

plt.axvline(x=0.5, color='red', linestyle='--', linewidth=2, label='Treatment Start')
plt.xlabel('Relative Week (0 = Last Pre-Treatment Week)')
plt.ylabel('Effect on Service Score')
plt.title('Event Study: Service Score Effects by Relative Week')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(relative_weeks)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("Event study plot saved as event_study.png")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators with better names
# Week 14 is the last pre-treatment week, so relative week = week - 14
df['relative_week'] = df['week'] - 14

# Create indicators for each relative week (omit week 0 as reference)
# relative_week ranges from -8 (week 6) to 8 (week 22)
for rw in range(-8, 9):
    if rw != 0:
        # Use 'w' prefix to avoid patsy issues
        df[f'w{rw:+d}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression with branch fixed effects
df['branch_id_str'] = df['branch_id'].astype(str)

# Build formula dynamically
rel_week_vars = [f'w{rw:+d}' for rw in range(-8, 9) if rw != 0]
formula = f"service_score ~ after_switch + {' + '.join(rel_week_vars)} + staffing_index + visits_index + C(branch_id_str)"

event_study = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the relative week indicators
event_coeffs = []
event_se = []
for rw in range(-8, 9):
    if rw == 0:
        event_coeffs.append(0)  # reference category
        event_se.append(0)
    else:
        col = f'w{rw:+d}'
        event_coeffs.append(event_study.params[col])
        event_se.append(event_study.bse[col])

# Create event study plot

relative_weeks = list(range(-8, 9))

plt.legend()

plt.xticks(relative_weeks)

plt.savefig('event_study.png', dpi=150)
