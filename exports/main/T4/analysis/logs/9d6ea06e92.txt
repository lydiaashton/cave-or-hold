from statsmodels.formula.api import ols

# Run a difference-in-differences regression
# Basic DiD: service_score ~ switched_branch * after_switch
model = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Basic Difference-in-Differences Regression:")
print(model.summary())

# Also run with branch fixed effects and week fixed effects for more robust estimate
df['switched_after'] = df['switched_branch'] * df['after_switch']

# Add branch and week as categorical for fixed effects
model_fe = ols('service_score ~ switched_after + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators
# Week 14 is the last pre-switch week, so we'll use that as reference
df['relative_week'] = df['week'] - 14  # Week 14 becomes 0, week 15 becomes 1, etc.

# Create indicators for each week relative to switch (excluding week 14 as reference)
# We want to see if there are any pre-trends
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != 0:  # Skip reference week
        df[f'rel_week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression with branch and week fixed effects
# Only include the interaction terms for treated branches in each relative week
event_study_vars = [f'rel_week_{w}' for w in range(-8, 9) if w != 0]
formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(branch_id) + C(week)'

model_es = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the relative week indicators
print("Event Study Results (coefficients for treated branches in each relative week):")
print("(Reference: Week 14, pre-switch)")
print("-" * 60)

results = []
for w in range(-8, 9):
    if w != 0:
        coef_name = f'rel_week_{w}'
        if coef_name in model_es.params:
            coef = model_es.params[coef_name]
            se = model_es.bse[coef_name]
            pval = model_es.pvalues[coef_name]
            ci_low = coef - 1.96 * se
            ci_high = coef + 1.96 * se
            period = "Pre" if w < 0 else "Post"
            results.append((w, coef, se, pval, ci_low, ci_high, period))
            marker = "*" if pval < 0.05 else " "
            print(f"Week {14+w:2d} (rel {w:+2d}): {coef:7.2f} (SE: {se:.2f}), 95% CI: [{ci_low:.2f}, {ci_high:.2f}] {marker}")

pre_se = [r[2] for r in results if r[6] == "Pre"]

    results.append((w, coef, se, pval, ci_low, ci_high, period))
    marker = "*" if pval < 0.05 else " "
    print(f"Week {14+w:2d} (rel {w:+2d}): {coef:7.2f} (SE: {se:.2f}), 95% CI: [{ci_low:.2f}, {ci_high:.2f}] {marker}")

# Create event study plot
pre_weeks = [r[0] for r in results if r[6] == "Pre"]
pre_coefs = [r[1] for r in results if r[6] == "Pre"]
pre_se = [r[2] for r in results if r[6] == "Pre"]

post_weeks = [r[0] for r in results if r[6] == "Post"]
post_coefs = [r[1] for r in results if r[6] == "Post"]

# Plot pre-treatment coefficients

            color='blue', label='Pre-Switch', capsize=5, markersize=8, linewidth=2)

# Plot post-treatment coefficients
ax.errorbar(post_weeks, post_coefs, yerr=[1.96*s for s in post_se], fmt='s', 
            color='green', label='Post-Switch', capsize=5, markersize=8, linewidth=2)

# Reference line at zero
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
# Switch point
ax.axvline(x=0.5, color='red', linestyle='--', linewidth=2, label='Switch Point')

ax.set_xlabel('Relative Week (0 = Week 14, last pre-switch week)', fontsize=12)
ax.set_ylabel('Effect on Service Score (vs Control)', fontsize=12)
ax.set_title('Event Study: Treatment Effect by Week Relative to Switch', fontsize=14)
ax.legend(loc='best')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-8, 9))

plt.tight_layout()
plt.savefig('event_study.png', dpi=150)

# Check if treatment affected the other indices (they shouldn't if randomization was good)
# This is a "placebo" test - if the switch affected staffing or visits, that could be a concern

print("Testing if switch affected staffing_index and visits_index:")
print("=" * 60)

for outcome in ['staffing_index', 'visits_index']:
    model = ols(f'{outcome} ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
    interaction_coef = model.params['switched_branch:after_switch']
    interaction_se = model.bse['switched_branch:after_switch']
    interaction_pval = model.pvalues['switched_branch:after_switch']
    print(f"\n{outcome}:")
    print(f"  DiD coefficient: {interaction_coef:.3f} (SE: {interaction_se:.3f}), p-value: {interaction_pval:.3f}")

# Also run the main model controlling for these covariates
print("\n" + "=" * 60)
print("Main model with covariates (staffing_index, visits_index):")
model_with_covariates = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', 
                            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
