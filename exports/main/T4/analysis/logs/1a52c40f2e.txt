import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Count branches by switch status
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate average service scores by group and period
print("\n=== Average Service Scores ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the DiD estimate manually
# Treatment group (switched=1): After - Before
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treatment_change = treatment_after - treatment_before

# Control group (switched=0): After - Before
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_change = control_after - control_before

# DiD estimate
did_estimate = treatment_change - control_change

print(f"\n=== Difference-in-Differences Calculation ===")
print(f"Treatment group (switched): Before={treatment_before:.2f}, After={treatment_after:.2f}, Change={treatment_change:.2f}")
print(f"Control group (not switched): Before={control_before:.2f}, After={control_after:.2f}, Change={control_change:.2f}")
print(f"\nDiD Estimate (Treatment Change - Control Change): {did_estimate:.2f}")

# Check weekly trends by group
print("\n=== Weekly Service Scores by Group ===")
weekly = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack('switched_branch')
weekly.columns = ['Control (No Switch)', 'Treatment (Switched)']

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression
print("=== DiD Regression (Basic) ===")
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model1.summary())

# Run DiD regression with controls
print("\n=== DiD Regression (With Controls) ===")
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study design: estimate effect for each week relative to baseline (week 6)
# This helps visualize dynamic treatment effects
df['week_rel'] = df['week'] - 14  # Center around switch week (week 14 is last pre, 15 is first post)
df['week_cat'] = df['week'].astype(str)

# Create event study regression
# Use week 6 as baseline (omit it)
event_study = ols('service_score ~ C(week) + switched_branch:C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the interaction terms
print("=== Event Study: Treatment Effect by Week (relative to week 6 baseline) ===")
weeks = sorted(df['week'].unique())
results = []
for w in weeks:
    if w == 6:
        results.append({'week': w, 'coef': 0, 'se': 0, 'ci_low': 0, 'ci_high': 0})
        continue
    coef_name = f'switched_branch:C(week)[T.{w}]'
    if coef_name in event_study.params.index:
        coef = event_study.params[coef_name]
        se = event_study.bse[coef_name]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        results.append({'week': w, 'coef': coef, 'se': se, 'ci_low': ci_low, 'ci_high': ci_high})

results_df = pd.DataFrame(results)
print(results_df.to_string(index=False))

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
ax.errorbar(results_df['week'], results_df['coef'], 
            yerr=[results_df['coef'] - results_df['ci_low'], results_df['ci_high'] - results_df['coef']],
            fmt='o', capsize=5, linewidth=2, markersize=8)
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch implemented')
ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Treatment Effect (vs Control)', fontsize=12)
ax.set_title('Event Study: Weekly Treatment Effects on Service Score', fontsize=14, fontweight='bold')
ax.legend()
ax.set_xticks(range(6, 23))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")

# Check if there's a difference in variance between groups
print("\n=== Variance Analysis ===")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create relative week indicator (weeks before/after switch)
df['relative_week'] = df['week'] - 15  # Week 15 = 0 (first post-switch week)

# Create event study dummies with safe names (no special characters)
for w in range(-9, 8):  # -9 to -1 are pre, 0 to 7 are post
    col_name = f'w{w+10}'  # Shift by 10 to avoid negatives: w-9->w1, w-8->w2, ..., w0->w10, w1->w11, ...
    df[col_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (omit w1 as baseline, which corresponds to relative week -9)
# Build formula - include all except w1
formula_terms = [f'w{w+10}' for w in range(-8, 8)]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(formula_terms)
print("Formula:", formula)

event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients
print("\n=== Event Study Results ===")
results = []
for w in range(-9, 8):
    col_name = f'w{w+10}'
    if w == -9:
        results.append({'relative_week': w, 'period': 'pre', 'coef': 0, 'se': 0, 'pval': np.nan})
        continue
    if col_name in event_model.params.index:
        coef = event_model.params[col_name]
        se = event_model.bse[col_name]
        pval = event_model.pvalues[col_name]
        period = 'pre' if w < 0 else 'post'
        results.append({'relative_week': w, 'period': period, 'coef': coef, 'se': se, 'pval': pval})

results_df = pd.DataFrame(results)
print(results_df.to_string(index=False))

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
pre_data = results_df[results_df['period'] == 'pre']
post_data = results_df[results_df['period'] == 'post']

ax.errorbar(pre_data['relative_week'], pre_data['coef'], 
            yerr=1.96 * pre_data['se'],
            fmt='o', capsize=5, linewidth=2, markersize=8, color='blue', label='Pre-switch')
ax.errorbar(post_data['relative_week'], post_data['coef'], 
            yerr=1.96 * post_data['se'],
            fmt='s', capsize=5, linewidth=2, markersize=8, color='green', label='Post-switch')

ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch point')
ax.set_xlabel('Relative Week (0 = first week after switch)', fontsize=12)
ax.set_ylabel('Treatment Effect (vs Control)', fontsize=12)
ax.set_title('Event Study: Dynamic Treatment Effects on Service Score', fontsize=14, fontweight='bold')
ax.legend()
ax.set_xticks(range(-9, 8))
plt.tight_layout()
plt.savefig('event_study_dynamic.png', dpi=150)
print("\nEvent study plot saved as 'event_study_dynamic.png'")

# Test joint significance of pre-trends
print("\n=== Pre-Trend Joint Test ===")
pre_test_cols = [f'w{w+10}' for w in range(-8, 0)]
pre_test = event_model.f_test(pre_test_cols)
print(f"F-statistic: {pre_test.statistic:.2f}, p-value: {pre_test.pvalue:.3f}")

# Test joint significance of post-treatment effects
print("\n=== Post-Treatment Effect Joint Test ===")
post_test_cols = [f'w{w+10}' for w in range(0, 8)]
post_test = event_model.f_test(post_test_cols)
print(f"F-statistic: {post_test.statistic:.2f}, p-value: {post_test.pvalue:.3f}")

# Build formula with branch FEs
branch_cols = [c for c in df_with_fe.columns if c.startswith('branch')]
formula_fe = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(branch_cols)
model_fe = ols(formula_fe, data=df_with_fe).fit(cov_type='cluster', cov_kwds={'groups': df_with_fe['branch_id']})

# Alternative: use linearmodels for proper FE
print("\n=== Branch Fixed Effects (using within transformation) ===")
# Demean within each branch
df['service_score_demeaned'] = df.groupby('branch_id')['service_score'].transform(lambda x: x - x.mean())
df['after_switch_demeaned'] = df.groupby('branch_id')['after_switch'].transform(lambda x: x - x.mean())
df['treatment_demeaned'] = df.groupby('branch_id')['treatment'].transform(lambda x: x - x.mean())

# The switched_branch is time-invariant, so it gets differenced out

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create treatment interaction
df['treatment'] = df['switched_branch'] * df['after_switch']

# Alternative: use linearmodels for proper FE
print("=== Branch Fixed Effects (using within transformation) ===")
# Demean within each branch
df['service_score_demeaned'] = df.groupby('branch_id')['service_score'].transform(lambda x: x - x.mean())
df['after_switch_demeaned'] = df.groupby('branch_id')['after_switch'].transform(lambda x: x - x.mean())
df['treatment_demeaned'] = df.groupby('branch_id')['treatment'].transform(lambda x: x - x.mean())

# The switched_branch is time-invariant, so it gets differenced out
model_within = ols('service_score_demeaned ~ after_switch_demeaned + treatment_demeaned', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_within.summary().tables[1])

# Also try controlling for pre-period average score
print("\n=== DiD Controlling for Pre-Period Branch Average ===")
pre_avg = df[df['week'] < 15].groupby('branch_id')['service_score'].mean().reset_index()
pre_avg.columns = ['branch_id', 'pre_avg_score']
df_merged = df.merge(pre_avg, on='branch_id')

model_ctrl = ols('service_score ~ switched_branch + after_switch + treatment + pre_avg_score', data=df_merged).fit(cov_type='cluster', cov_kwds={'groups': df_merged['branch_id']})
print(model_ctrl.summary().tables[1])

# Synthetic control style: match on pre-period score
print("\n=== Matched Sample (by pre-period score tercile) ===")
df_merged['pre_tercile'] = pd.qcut(df_merged['pre_avg_score'], 3, labels=['low', 'mid', 'high'])

# Within each tercile, compare treatment vs control
matched = df_merged.copy()
model_matched = ols('service_score ~ switched_branch + after_switch + treatment + C(pre_tercile)', data=matched).fit(cov_type='cluster', cov_kwds={'groups': matched['branch_id']})
print(model_matched.summary().tables[1])
