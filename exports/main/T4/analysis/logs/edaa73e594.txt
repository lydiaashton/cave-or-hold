import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Count branches by switch status
branches = df[['branch_id', 'switched_branch']].drop_duplicates()
print("Branches by switch status:")
print(branches['switched_branch'].value_counts())
print("\nProportion switched:", branches['switched_branch'].mean())

# Calculate average service scores by group and period
print("\n\nAverage service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Create the 2x2 table for difference-in-differences
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n\n2x2 Difference-in-Differences Table:")
print(f"Control (no switch): Before={before_control:.2f}, After={after_control:.2f}, Change={after_control - before_control:.2f}")
print(f"Treatment (switch):  Before={before_treatment:.2f}, After={after_treatment:.2f}, Change={after_treatment - before_treatment:.2f}")
print(f"\nDifference-in-Differences: {(after_treatment - before_treatment) - (after_control - before_control):.2f}")

# Run a regression with DiD specification
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\n\nBasic DiD Regression:")
print(model.summary())

# Add branch fixed effects
df['branch_id'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit()
print("\n\nDiD Regression with Branch Fixed Effects:")
print(model_fe.summary())
