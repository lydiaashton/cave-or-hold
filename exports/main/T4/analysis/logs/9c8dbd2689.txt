import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Count switched vs non-switched branches
switched_counts = df.groupby('switched_branch')['branch_id'].nunique()
print("Number of branches by group:")
print(switched_counts)
print(f"\nTotal branches: {switched_counts.sum()}")
print(f"Switched: {switched_counts.get(1, 0)} ({switched_counts.get(1, 0)/switched_counts.sum()*100:.1f}%)")
print(f"Not switched: {switched_counts.get(0, 0)} ({switched_counts.get(0, 0)/switched_counts.sum()*100:.1f}%)")

# Calculate average service scores by group and period
print("\n\nAverage service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the difference-in-differences manually
# Before period: after_switch = 0
# After period: after_switch = 1
before_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n\nMean service scores:")
print(f"Switched branches - Before: {before_switched:.2f}, After: {after_switched:.2f}, Change: {after_switched - before_switched:.2f}")
print(f"Control branches - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")

did_effect = (after_switched - before_switched) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {did_effect:.2f}")
