import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression (no controls)
model_basic = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Basic DiD Model (no controls) ===")
print(model_basic.summary().tables[1])

# DiD regression with controls
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Model with Controls ===")
print(model_controls.summary().tables[1])

# Get the DiD coefficient and confidence interval
did_coef = model_controls.params['treatment']
did_se = model_controls.bse['treatment']
did_ci_lower = did_coef - 1.96 * did_se
did_ci_upper = did_coef + 1.96 * did_se
print(f"\nDiD Estimate: {did_coef:.3f}")
print(f"95% CI: [{did_ci_lower:.3f}, {did_ci_upper:.3f}]")
print(f"P-value: {model_controls.pvalues['treatment']:.4f}")

ax.legend(loc='best')

ax.set_xticks(range(6, 23))

# Calculate week-over-week changes in pre-period for each group

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create week dummies interacted with treatment
df['relative_week'] = df['week'] - 14  # Week 14 is the last pre-switch week
df['relative_week'] = df['relative_week'].clip(-8, 8)  # Cap at -8 to 8

# Create event study dummies (omit week -1 as reference)
event_study_df = df.copy()
for w in range(-8, 9):
    if w != -1:  # Skip reference week
        event_study_df[f'week_{w}'] = ((event_study_df['relative_week'] == w) & (event_study_df['switched_branch'] == 1)).astype(int)

# Build formula for event study
control_vars = ['switched_branch', 'staffing_index', 'visits_index']
week_vars = [f'week_{w}' for w in range(-8, 9) if w != -1]
formula = 'service_score ~ ' + ' + '.join(control_vars + week_vars)

model_event = ols(formula, data=event_study_df).fit(cov_type='cluster', cov_kwds={'groups': event_study_df['branch_id']})

# Extract coefficients for event study
event_coefs = []
event_ci_lower = []
event_ci_upper = []
event_weeks = []

for w in range(-8, 9):
    if w == -1:
        event_coefs.append(0)
        event_ci_lower.append(0)
        event_ci_upper.append(0)
        event_weeks.append(w)
    else:
        coef = model_event.params[f'week_{w}']
        se = model_event.bse[f'week_{w}']
        event_coefs.append(coef)
        event_ci_lower.append(coef - 1.96 * se)
        event_ci_upper.append(coef + 1.96 * se)
        event_weeks.append(w)

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
ax.bar(event_weeks, event_coefs, color=['gray' if w < 0 else 'steelblue' for w in event_weeks], alpha=0.7)
ax.errorbar(event_weeks, event_coefs, yerr=[np.array(event_coefs) - np.array(event_ci_lower), np.array(event_ci_upper) - np.array(event_coefs)], 
            fmt='none', ecolor='black', capsize=5, linewidth=1.5)
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch Point')
ax.set_xlabel('Relative Week (0 = First Week After Switch)', fontsize=12)
ax.set_ylabel('Effect on Service Score', fontsize=12)
ax.set_title('Event Study: Effect of New Procedure on Service Scores', fontsize=14, fontweight='bold')
ax.set_xticks(range(-8, 9))
ax.legend()
ax.grid(True, alpha=0.3, axis='y')
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("Event study plot saved as event_study.png")

# Print key event study coefficients
print("\n=== Event Study Results ===")
print("Pre-treatment weeks (should be near 0 if parallel trends hold):")
for i, w in enumerate(event_weeks):
    if w < 0:
        print(f"  Week {w}: {event_coefs[i]:.2f} [{event_ci_lower[i]:.2f}, {event_ci_upper[i]:.2f}]")

print("\nPost-treatment weeks:")
for i, w in enumerate(event_weeks):
    if w >= 0:
        print(f"  Week {w}: {event_coefs[i]:.2f} [{event_ci_lower[i]:.2f}, {event_ci_upper[i]:.2f}]")

# Calculate average post-treatment effect
post_coefs = [event_coefs[i] for i, w in enumerate(event_weeks) if w >= 0]
avg_post_effect = np.mean(post_coefs)
print(f"\nAverage post-treatment effect: {avg_post_effect:.2f}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create week dummies interacted with treatment
df['relative_week'] = df['week'] - 14  # Week 14 is the last pre-switch week

# Create event study dummies with safe names (w prefix for negative weeks)
event_study_df = df.copy()
for w in range(-8, 9):
    if w != -1:  # Skip reference week
        col_name = f'w{abs(w)}_neg' if w < 0 else f'w{w}_pos'
        event_study_df[col_name] = ((event_study_df['relative_week'] == w) & (event_study_df['switched_branch'] == 1)).astype(int)

# Build formula for event study
control_vars = ['switched_branch', 'staffing_index', 'visits_index']
week_vars = [f'w{abs(w)}_neg' if w < 0 else f'w{w}_pos' for w in range(-8, 9) if w != -1]
formula = 'service_score ~ ' + ' + '.join(control_vars + week_vars)

model_event = ols(formula, data=event_study_df).fit(cov_type='cluster', cov_kwds={'groups': event_study_df['branch_id']})

# Extract coefficients for event study
event_coefs = []
event_ci_lower = []
event_ci_upper = []
event_weeks = []
event_names = []

for w in range(-8, 9):
    if w == -1:
        event_coefs.append(0)
        event_ci_lower.append(0)
        event_ci_upper.append(0)
        event_weeks.append(w)
        event_names.append('ref')
    else:
        col_name = f'w{abs(w)}_neg' if w < 0 else f'w{w}_pos'
        coef = model_event.params[col_name]
        se = model_event.bse[col_name]
        event_coefs.append(coef)
        event_ci_lower.append(coef - 1.96 * se)
        event_ci_upper.append(coef + 1.96 * se)
        event_weeks.append(w)
        event_names.append(col_name)

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
colors = ['gray' if w < 0 else 'steelblue' for w in event_weeks]
ax.bar(event_weeks, event_coefs, color=colors, alpha=0.7)
ax.errorbar(event_weeks, event_coefs, 
            yerr=[np.array(event_coefs) - np.array(event_ci_lower), np.array(event_ci_upper) - np.array(event_coefs)], 
            fmt='none', ecolor='black', capsize=5, linewidth=1.5)
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch Point')
ax.set_xlabel('Relative Week (0 = First Week After Switch)', fontsize=12)
ax.set_ylabel('Effect on Service Score', fontsize=12)
ax.set_title('Event Study: Effect of New Procedure on Service Scores', fontsize=14, fontweight='bold')
ax.set_xticks(range(-8, 9))
ax.legend()
ax.grid(True, alpha=0.3, axis='y')
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("Event study plot saved as event_study.png")

# Print key event study coefficients
print("\n=== Event Study Results ===")
print("Pre-treatment weeks (should be near 0 if parallel trends hold):")
for i, w in enumerate(event_weeks):
    if w < 0:
        sig = "***" if event_ci_lower[i] > 0 or event_ci_upper[i] < 0 else ""
        print(f"  Week {w}: {event_coefs[i]:.2f} [{event_ci_lower[i]:.2f}, {event_ci_upper[i]:.2f}] {sig}")

print("\nPost-treatment weeks:")
for i, w in enumerate(event_weeks):
    if w >= 0:
        sig = "***" if event_ci_lower[i] > 0 or event_ci_upper[i] < 0 else ""
        print(f"  Week {w}: {event_coefs[i]:.2f} [{event_ci_lower[i]:.2f}, {event_ci_upper[i]:.2f}] {sig}")

# Calculate average post-treatment effect
post_coefs = [event_coefs[i] for i, w in enumerate(event_weeks) if w >= 0]
avg_post_effect = np.mean(post_coefs)
print(f"\nAverage post-treatment effect: {avg_post_effect:.2f}")

import pandas as pd

df = pd.read_csv('branches.csv')

# Summary statistics for the board

# Number of branches
total_branches = df['branch_id'].nunique()

print(f"Total branches in study: {total_branches}")
print(f"  - Switched to new procedure: {switched_branches}")
print(f"  - Kept old procedure: {control_branches}")

# Time period
print(f"\nStudy period: Week {df['week'].min()} to Week {df['week'].max()}")
print(f"  - Pre-switch: Weeks 6-14 (9 weeks)")
print(f"  - Post-switch: Weeks 15-22 (8 weeks)")

# Service score statistics
print(f"\nService Score Statistics:")
print(f"  Overall mean: {df['service_score'].mean():.1f}")
print(f"  Overall std dev: {df['service_score'].std():.1f}")

# Pre vs Post for switched branches
switched_pre = df[(df['switched_branch'] == 1) & (df['week'] < 15)]['service_score'].mean()
switched_post = df[(df['switched_branch'] == 1) & (df['week'] >= 15)]['service_score'].mean()
control_pre = df[(df['switched_branch'] == 0) & (df['week'] < 15)]['service_score'].mean()
control_post = df[(df['switched_branch'] == 0) & (df['week'] >= 15)]['service_score'].mean()

print(f"\n  Switched branches:")
print(f"    Pre-switch average: {switched_pre:.1f}")
print(f"    Post-switch average: {switched_post:.1f}")
print(f"    Change: {switched_post - switched_pre:+.1f}")

print(f"\n  Non-switched branches:")
print(f"    Pre-switch average: {control_pre:.1f}")
print(f"    Post-switch average: {control_post:.1f}")
print(f"    Change: {control_post - control_pre:+.1f}")

# DiD estimate
did = (switched_post - switched_pre) - (control_post - control_pre)
print(f"\n  Difference-in-Differences estimate: +{did:.1f} points")
print(f"  Percentage improvement: {(did / switched_pre) * 100:.1f}%")

# Statistical significance
import statsmodels.api as sm
from statsmodels.formula.api import ols

df['treatment'] = df['switched_branch'] * df['after_switch']
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

did_coef = model.params['treatment']
did_pvalue = model.pvalues['treatment']
did_se = model.bse['treatment']
did_ci_lower = did_coef - 1.96 * did_se
did_ci_upper = did_coef + 1.96 * did_se

print(f"\n  Statistical significance:")
print(f"    Coefficient: {did_coef:.2f}")
print(f"    Standard error: {did_se:.2f}")
print(f"    95% Confidence interval: [{did_ci_lower:.2f}, {did_ci_upper:.2f}]")
print(f"    P-value: {did_pvalue:.4f}")
print(f"    Statistically significant: {'Yes' if did_pvalue < 0.05 else 'No'}")

print("\n=== Additional Analysis ===")

# Effect by post-switch week
post_df = df[df['week'] >= 15].copy()
