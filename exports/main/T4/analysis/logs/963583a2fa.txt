import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate summary statistics by group and period
print("=== Service Score by Group and Period ===")
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Calculate the DiD estimate manually
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n=== Manual DiD Calculation ===")
print(f"Control group (non-switched): Before={before_control:.2f}, After={after_control:.2f}, Change={after_control - before_control:.2f}")
print(f"Treatment group (switched): Before={before_treatment:.2f}, After={after_treatment:.2f}, Change={after_treatment - before_treatment:.2f}")
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"DiD Estimate: {did_estimate:.2f}")

# Run DiD regression
print("\n=== DiD Regression (Basic) ===")
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model1.summary())

# Run DiD regression with controls
print("\n=== DiD Regression (With Controls) ===")
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print(model2.summary())

# Check parallel trends - look at pre-treatment trends
print("\n=== Pre-Treatment Trends Check (Weeks 6-14) ===")
pre_data = df[df['week'] < 15]
pre_trends = pre_data.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
print(pre_trends)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Check parallel trends more formally - event study approach
print("=== Event Study: Week-by-week differences ===")
week_means = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
week_diffs = week_means[1] - week_means[0]
print(week_diffs)

# Calculate pre-treatment trend difference
pre_week_diffs = week_diffs[week_diffs.index < 15]
print(f"\nAverage pre-treatment difference (weeks 6-14): {pre_week_diffs.mean():.2f}")
print(f"Std dev of pre-treatment differences: {pre_week_diffs.std():.2f}")

# Run regression with branch fixed effects
print("\n=== DiD with Branch Fixed Effects ===")
# Create branch dummies
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df_with_dummies['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [col for col in df_with_dummies.columns if col.startswith('branch_')]
formula = f"service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index + {' + '.join(branch_cols)}"
model_fe = ols(formula, data=df_with_dummies).fit()
print(f"Treatment effect (with branch FE): {model_fe.params['treatment']:.3f}")
print(f"Std error: {model_fe.bse['treatment']:.3f}")
print(f"P-value: {model_fe.pvalues['treatment']:.3f}")
print(f"95% CI: [{model_fe.conf_int().loc['treatment', 0]:.3f}, {model_fe.conf_int().loc['treatment', 1]:.3f}]")

# Clustered standard errors by branch
print("\n=== DiD with Clustered SE (by branch) ===")
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_clustered.summary().tables[1])

# Create visualization data
plt.figure(figsize=(12, 6))
weeks = sorted(df['week'].unique())
control_means = []
treatment_means = []
for w in weeks:
    control_means.append(df[(df['week'] == w) & (df['switched_branch'] == 0)]['service_score'].mean())
    treatment_means.append(df[(df['week'] == w) & (df['switched_branch'] == 1)]['service_score'].mean())

plt.plot(weeks, control_means, 'o-', label='Control (Old Procedure)', linewidth=2, markersize=6)
plt.plot(weeks, treatment_means, 's-', label='Treatment (New Procedure)', linewidth=2, markersize=6)
plt.axvline(x=14.5, color='red', linestyle='--', label='Switch Point (Week 15)')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Scores by Group Over Time')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('service_scores_over_time.png', dpi=150, bbox_inches='tight')
print("\nPlot saved as service_scores_over_time.png")

# Summary stats for the report
print("\n=== Final Summary Statistics ===")
print(f"Total branches: {df['branch_id'].nunique()}")
print(f"Switched branches: {df['switched_branch'].sum()}")
print(f"Non-switched branches: {(df['switched_branch'] == 0).sum()}")
print(f"Weeks analyzed: {df['week'].min()} to {df['week'].max()}")
print(f"\nTreatment effect estimate: {model_clustered.params['treatment']:.2f} points")
