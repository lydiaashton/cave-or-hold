import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Check parallel trends by looking at week-to-week changes in pre-period
pre_treatment = df[df['week'] < 15].copy()
weekly_avg = pre_treatment.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

print("=== Week-to-week changes in pre-treatment period ===")
print("Control group changes:")
control_changes = weekly_avg[0].diff()
print(control_changes)
print("\nTreatment group changes:")
treated_changes = weekly_avg[1].diff()
print(treated_changes)
print("\nDifference in changes (should be near 0 if parallel):")
diff_changes = treated_changes - control_changes
print(diff_changes)
print(f"\nAverage difference in weekly changes: {diff_changes.mean():.3f}")

# Event study - estimate effect for each week relative to switch
# Create week dummies interacted with treatment
df['relative_week'] = df['week'] - 15  # Week 15 is the switch (0), before is negative

# Create indicators for each week relative to switch, using week 14 as baseline
for w in range(-9, 8):  # weeks 6-22 relative to 15
    if w != -1:  # skip week 14 (baseline)
        df[f'week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
week_cols = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(week_cols)
model_es = smf.ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n=== Event Study Results ===")
print("Coefficients for treatment effects by week (baseline = week 14):")
results = []
for w in range(-9, 8):
    if w != -1:
        col = f'week_{w}'
        coef = model_es.params[col]
        se = model_es.bse[col]
        pval = model_es.pvalues[col]
        results.append((w, coef, se, pval))
        print(f"Week {w+15} (relative {w:+d}): coef={coef:.2f}, SE={se:.2f}, p={pval:.3f}")

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))

weeks_rel = [r[0] for r in results]
coefs = [r[1] for r in results]
ses = [r[2] for r in results]

ax.errorbar(weeks_rel, coefs, yerr=1.96*np.array(ses), fmt='o', capsize=5, 
            label='Treatment Effect', markersize=8, linewidth=2)
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch Point')
ax.set_xlabel('Weeks Relative to Switch (0 = Week 15)', fontsize=12)
ax.set_ylabel('Coefficient (vs Week 14 baseline)', fontsize=12)
ax.set_title('Event Study: Treatment Effect Over Time', fontsize=14)
ax.legend(fontsize=11)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-9, 8))

plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")

pre_treatment_weeks = [f'week_{w}' for w in range(-9, -1)]  # weeks 6-13
print(f"\n=== Pre-trend test ===")
print(f"Testing joint significance of pre-treatment coefficients...")
from statsmodels.stats.anova import anova_lm

    if w != -1:  # skip week 14 (baseline)

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# Create relative week
df['relative_week'] = df['week'] - 15  # Week 15 is the switch (0), before is negative

# Create week dummies for fixed effects
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df = pd.concat([df, week_dummies], axis=1)

# Create indicators for each week relative to switch for treated branches only
# Using week 14 (relative -1) as baseline
relative_dummies = {}
for w in range(-9, 8):  # weeks 6-22 relative to 15
    if w != -1:  # skip week 14 (baseline)
        col_name = f'treat_rel{abs(w)}_{"n" if w < 0 else "p"}' if w != 0 else 'treat_rel0'
        relative_dummies[w] = col_name
        df[col_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Create design matrix with week FE and relative time dummies for treated
week_cols = [c for c in df.columns if c.startswith('week_')]
treat_cols = list(relative_dummies.values())
X = df[['switched_branch'] + week_cols + treat_cols]  # switched_branch captures time-invariant group diff
X = sm.add_constant(X)
y = df['service_score']

# Fit model with cluster-robust SE
model_es = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results (with week fixed effects) ===")
print("Coefficients for treatment effects by week (baseline = week 14):")
results = []
for w in range(-9, 8):
    if w != -1:
        col = relative_dummies[w]
        coef = model_es.params[col]
        se = model_es.bse[col]
        pval = model_es.pvalues[col]
        results.append((w, coef, se, pval))
        actual_week = w + 15
        print(f"Week {actual_week} (relative {w:+d}): coef={coef:.2f}, SE={se:.2f}, p={pval:.3f}")

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))

weeks_rel = [r[0] for r in results]
coefs = [r[1] for r in results]
ses = [r[2] for r in results]

ax.errorbar(weeks_rel, coefs, yerr=1.96*np.array(ses), fmt='o', capsize=5, 
            label='Treatment Effect', markersize=8, linewidth=2, color='blue')

ax.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch Point')
