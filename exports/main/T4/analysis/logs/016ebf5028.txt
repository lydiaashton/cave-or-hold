import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model 1: Simple DiD (no controls)
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Model 1: Simple DiD (no controls) ===")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\n=== Model 2: DiD with controls ===")
print(model2.summary().tables[1])

# Model 3: DiD with branch fixed effects
# This controls for time-invariant branch characteristics
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Build formula with branch fixed effects
branch_cols = [col for col in df_with_dummies.columns if col.startswith('branch')]
formula = 'service_score ~ after_switch + treatment + staffing_index + visits_index + ' + ' + '.join(branch_cols)
model3 = ols(formula, data=df_with_dummies).fit()
print("\n=== Model 3: DiD with Branch Fixed Effects ===")
# Just show the key coefficients
key_vars = ['after_switch', 'treatment', 'staffing_index', 'visits_index']
print(model3.summary().tables[1][model3.summary().tables[1].index.isin(key_vars)])

# Model 4: DiD with branch and week fixed effects
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df_full = pd.concat([df, branch_dummies, week_dummies], axis=1)
week_cols = [col for col in df_full.columns if col.startswith('week')]
formula_full = 'service_score ~ treatment + staffing_index + visits_index + ' + ' + '.join(branch_cols) + ' + ' + ' + '.join(week_cols)
model4 = ols(formula_full, data=df_full).fit()
print("\n=== Model 4: DiD with Branch and Week Fixed Effects ===")
print(model4.summary().tables[1][model4.summary().tables[1].index.isin(['treatment', 'staffing_index', 'visits_index'])])

# Store the key results
print("\n\n=== SUMMARY OF TREATMENT EFFECT ESTIMATES ===")
print(f"Model 1 (Simple DiD): {model1.params['treatment']:.3f} (SE: {model1.bse['treatment']:.3f}, p: {model1.pvalues['treatment']:.4f})")
print(f"Model 2 (With controls): {model2.params['treatment']:.3f} (SE: {model2.bse['treatment']:.3f}, p: {model2.pvalues['treatment']:.4f})")
print(f"Model 3 (Branch FE): {model3.params['treatment']:.3f} (SE: {model3.bse['treatment']:.3f}, p: {model3.pvalues['treatment']:.4f})")
print(f"Model 4 (Branch + Week FE): {model4.params['treatment']:.3f} (SE: {model4.bse['treatment']:.3f}, p: {model4.pvalues['treatment']:.4f})")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import warnings
warnings.filterwarnings('ignore')

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model 1: Simple DiD (no controls)
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Model 1: Simple DiD (clustered SEs) ===")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Model 2: DiD with controls (clustered SEs) ===")
print(model2.summary().tables[1])

# Model 3: Branch fixed effects using demeaning approach
# First, calculate branch means
branch_means = df.groupby('branch_id')[['service_score', 'after_switch', 'treatment', 'staffing_index', 'visits_index']].mean()
branch_means.columns = ['service_score_mean', 'after_switch_mean', 'treatment_mean', 'staffing_index_mean', 'visits_index_mean']

df_fe = df.merge(branch_means, left_on='branch_id', right_index=True)

# Demean the variables
for col in ['service_score', 'after_switch', 'treatment', 'staffing_index', 'visits_index']:
    df_fe[f'{col}_demeaned'] = df_fe[col] - df_fe[f'{col}_mean']

# Run regression on demeaned data (this is equivalent to branch FE)
model3 = ols('service_score_demeaned ~ after_switch_demeaned + treatment_demeaned + staffing_index_demeaned + visits_index_demeaned', data=df_fe).fit(cov_type='cluster', cov_kwds={'groups': df_fe['branch_id']})
print("\n=== Model 3: Branch Fixed Effects (clustered SEs) ===")
print(model3.summary().tables[1])

# Model 4: Branch and week fixed effects
# Demean by branch first, then by week (two-way FE)
week_means = df_fe.groupby('week')[['service_score_demeaned', 'after_switch_demeaned', 'treatment_demeaned', 'staffing_index_demeaned', 'visits_index_demeaned']].mean()
week_means.columns = [f'{col}_week_mean' for col in ['service_score', 'after_switch', 'treatment', 'staffing_index', 'visits_index']]

df_twfe = df_fe.merge(week_means, left_on='week', right_index=True)

for col in ['service_score', 'after_switch', 'treatment', 'staffing_index', 'visits_index']:
    df_twfe[f'{col}_twfe'] = df_twfe[f'{col}_demeaned'] - df_twfe[f'{col}_week_mean']

model4 = ols('service_score_twfe ~ after_switch_twfe + treatment_twfe + staffing_index_twfe + visits_index_twfe', data=df_twfe).fit(cov_type='cluster', cov_kwds={'groups': df_twfe['branch_id']})
print("\n=== Model 4: Branch + Week Fixed Effects (clustered SEs) ===")
print(model4.summary().tables[1])

# Store the key results
print("\n\n=== SUMMARY OF TREATMENT EFFECT ESTIMATES ===")
print(f"Model 1 (Simple DiD): {model1.params['treatment']:.3f} (SE: {model1.bse['treatment']:.3f}, p: {model1.pvalues['treatment']:.4f})")
print(f"Model 2 (With controls): {model2.params['treatment']:.3f} (SE: {model2.bse['treatment']:.3f}, p: {model2.pvalues['treatment']:.4f})")
print(f"Model 3 (Branch FE): {model3.params['treatment_demeaned']:.3f} (SE: {model3.bse['treatment_demeaned']:.3f}, p: {model3.pvalues['treatment_demeaned']:.4f})")
print(f"Model 4 (Branch + Week FE): {model4.params['treatment_twfe']:.3f} (SE: {model4.bse['treatment_twfe']:.3f}, p: {model4.pvalues['treatment_twfe']:.4f})")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import warnings
warnings.filterwarnings('ignore')

df = pd.read_csv('branches.csv')

# Test for parallel trends: check if pre-treatment trends differ
# Create week dummies interacted with treatment for pre-period only
df['week_centered'] = df['week'] - 14  # Center at switch point

# Pre-period only
pre_df = df[df['week'] < 15].copy()
pre_df['treatment'] = pre_df['switched_branch'] * pre_df['after_switch']  # Should be all 0 in pre-period

# Test if there's a differential pre-trend
pre_model = ols('service_score ~ switched_branch + week_centered + switched_branch:week_centered', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print("=== Pre-Treatment Trend Test ===")
print(pre_model.summary().tables[1])
print(f"\nInteraction term (differential pre-trend): coef={pre_model.params['switched_branch:week_centered']:.3f}, p={pre_model.pvalues['switched_branch:week_centered']:.3f}")

# Event study: estimate effect for each post-switch week
# Create week-specific treatment indicators
for week in range(15, 23):
    df[f'treat_week{week}'] = ((df['switched_branch'] == 1) & (df['week'] == week)).astype(int)

# Reference category is week 14 (last pre-switch week)
event_study_vars = [f'treat_week{w}' for w in range(15, 23)]
formula = 'service_score ~ switched_branch + ' + ' + '.join(event_study_vars) + ' + staffing_index + visits_index'

# Add branch fixed effects via demeaning
branch_means = df.groupby('branch_id')[['service_score', 'switched_branch'] + event_study_vars + ['staffing_index', 'visits_index']].mean()
branch_means.columns = [f'{col}_mean' for col in branch_means.columns]

df_es = df.merge(branch_means, left_on='branch_id', right_index=True)
for col in ['service_score', 'switched_branch'] + event_study_vars + ['staffing_index', 'visits_index']:
    df_es[f'{col}_dm'] = df_es[col] - df_es[f'{col}_mean']

es_formula = 'service_score_dm ~ switched_branch_dm + ' + ' + '.join([f'{v}_dm' for v in event_study_vars]) + ' + staffing_index_dm + visits_index_dm'
es_model = ols(es_formula, data=df_es).fit(cov_type='cluster', cov_kwds={'groups': df_es['branch_id']})

print("\n=== Event Study Results (Week-specific treatment effects) ===")
print(f"{'Week':<8} {'Coefficient':<12} {'SE':<10} {'95% CI':<20} {'Significant':<12}")
print("-" * 65)
for week in range(15, 23):
    var_name = f'treat_week{week}_dm'
    coef = es_model.params[var_name]
    se = es_model.bse[var_name]
    ci_low = coef - 1.96 * se
    ci_high = coef + 1.96 * se
    sig = "Yes" if es_model.pvalues[var_name] < 0.05 else "No"
    print(f"{week:<8} {coef:<12.2f} {se:<10.2f} [{ci_low:.1f}, {ci_high:.1f}]  {sig:<12}")

print("\n\n=== KEY STATISTICS FOR BOARD SUMMARY ===")
print(f"Total branches studied: {df['branch_id'].nunique()}")
print(f"  - Switched to new procedure: {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"  - Kept old procedure: {df[df['switched_branch']==0]['branch_id'].nunique()}")
print(f"Study period: Weeks {df['week'].min()} to {df['week'].max()} ({df['week'].max() - df['week'].min() + 1} weeks)")
print(f"Switch occurred: Week 15")
print(f"Pre-switch observations: {df[df['after_switch']==0].shape[0]}")
print(f"Post-switch observations: {df[df['after_switch']==1].shape[0]}")
print(f"\nAverage service score (all data): {df['service_score'].mean():.1f}")
print(f"Average service score (treatment, pre): {df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean():.1f}")
