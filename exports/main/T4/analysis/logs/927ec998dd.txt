import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD: average scores by group and period
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES ANALYSIS")
print("=" * 60)

# Calculate means for the 4 groups
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\nAverage Service Scores:")
print(f"  Control (not switched) - Before: {before_control:.2f}")
print(f"  Control (not switched) - After:  {after_control:.2f}")
print(f"  Treatment (switched)   - Before: {before_treatment:.2f}")
print(f"  Treatment (switched)   - After:  {after_treatment:.2f}")

# Calculate differences
control_diff = after_control - before_control
treatment_diff = after_treatment - before_treatment
did_effect = treatment_diff - control_diff

print(f"\nChanges over time:")
print(f"  Control group change: {control_diff:.2f}")
print(f"  Treatment group change: {treatment_diff:.2f}")
print(f"\n  DiD Effect (Treatment - Control): {did_effect:.2f}")

# Run regression with cluster-robust standard errors
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Get cluster-robust standard errors
model2_robust = model2.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print("\nModel 2 with Cluster-Robust SE (by branch):")
print(model2_robust.summary().tables[1])

# Get the treatment effect and CI

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Full model with cluster-robust SE
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
model_robust = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Access by position (Intercept=0, switched_branch=1, after_switch=2, treatment=3, staffing=4, visits=5)
treatment_coef = model_robust.params[3]
treatment_se = model_robust.bse[3]
treatment_pval = model_robust.pvalues[3]

# 95% CI
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se

print("=" * 60)
print("FINAL ESTIMATES")
print("=" * 60)
print(f"\nTreatment Effect (New Procedure):")
print(f"  Point estimate: {treatment_coef:.2f} points")
print(f"  Standard error: {treatment_se:.2f}")
print(f"  95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"  p-value: {treatment_pval:.3f}")

# Context: what does 1.5 points mean?
overall_mean = df['service_score'].mean()
overall_std = df['service_score'].std()
print(f"\nContext:")
print(f"  Overall mean service score: {overall_mean:.1f}")
print(f"  Overall std dev: {overall_std:.1f}")
print(f"  Effect as % of mean: {treatment_coef/overall_mean*100:.1f}%")
print(f"  Effect as fraction of std dev: {treatment_coef/overall_std:.2f} SD")

# Check if there's any pattern we're missing - maybe look at staffing/visits changes
print("\n" + "=" * 60)
