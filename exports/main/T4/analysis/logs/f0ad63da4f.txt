import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch']  # 1 if branch switched
df['post'] = df['after_switch']  # 1 if after week 15
df['DiD'] = df['treated'] * df['post']  # Interaction term

print("=== Pre/Post Service Scores by Group ===")
pre_scores = df[df['post']==0].groupby('treated')['service_score'].mean()
post_scores = df[df['post']==1].groupby('treated')['service_score'].mean()

print("\nBefore switch (weeks 6-14):")
print(f"  Control (non-switched): {pre_scores[0]:.2f}")
print(f"  Treatment (switched):   {pre_scores[1]:.2f}")
print(f"  Difference:             {pre_scores[1] - pre_scores[0]:.2f}")

print("\nAfter switch (weeks 15-22):")
print(f"  Control (non-switched): {post_scores[0]:.2f}")
print(f"  Treatment (switched):   {post_scores[1]:.2f}")
print(f"  Difference:             {post_scores[1] - post_scores[0]:.2f}")

# Difference-in-Differences estimate
did_estimate = (post_scores[1] - post_scores[0]) - (pre_scores[1] - pre_scores[0])
print(f"\n=== DiD Estimate ===")
print(f"DiD = (Post_T - Post_C) - (Pre_T - Pre_C) = {did_estimate:.3f}")

# Now run regression with branch fixed effects
print("\n=== Regression Analysis ===")

# Model 1: Simple DiD without fixed effects
model1 = ols('service_score ~ treated + post + DiD', data=df).fit()
print("\nModel 1: Simple DiD (no fixed effects)")
print(model1.summary().tables[1])

# Model 2: With branch fixed effects
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df_with_dummies['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Build formula dynamically
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_')]
formula = f"service_score ~ treated + post + DiD + {' + '.join(branch_cols)}"
model2 = ols(formula, data=df_with_dummies).fit()
print("\nModel 2: DiD with Branch Fixed Effects")
print(model2.summary().tables[1].loc[['treated', 'post', 'DiD']])

# Model 3: With branch fixed effects and control variables
formula3 = f"service_score ~ treated + post + DiD + staffing_index + visits_index + {' + '.join(branch_cols)}"
model3 = ols(formula3, data=df_with_dummies).fit()
print("\nModel 3: DiD with Branch FE + Controls")
print(model3.summary().tables[1].loc[['treated', 'post', 'DiD', 'staffing_index', 'visits_index']])
