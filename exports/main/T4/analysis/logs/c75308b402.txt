import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Calculate average service score by week and group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_score']

# Pivot for easier plotting
pivot = weekly_avg.pivot(index='week', columns='switched_branch', values='avg_score')
pivot.columns = ['Control (No Switch)', 'Treatment (Switched)']

# Create the plot
fig, ax = plt.subplots(figsize=(10, 6))

# Plot the trends
ax.plot(pivot.index, pivot['Control (No Switch)'], 'o-', label='Control (No Switch)', linewidth=2, markersize=6)
ax.plot(pivot.index, pivot['Treatment (Switched)'], 's-', label='Treatment (Switched)', linewidth=2, markersize=6)

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (Week 15)')

# Add labels and title
ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Score Trends: Treatment vs Control Groups\n(Vertical line marks the switch at Week 15)', fontsize=14)
ax.legend(loc='best', fontsize=10)
ax.grid(True, alpha=0.3)

# Set x-axis ticks
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("Plot saved as service_score_trends.png")

# Print the weekly averages for reference
print("\n=== Weekly Average Service Scores ===")
print(pivot.round(2))

# Check parallel trends statistically (pre-period only)
pre_period = df[df['week'] < 15]
pre_weekly = pre_period.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()

# Calculate week-over-week changes for each group
control_changes = pre_pivot[0].diff()
treatment_changes = pre_pivot[1].diff()

print("\n=== Week-over-Week Changes (Pre-Period) ===")
changes_df = pd.DataFrame({'Control': control_changes, 'Treatment': treatment_changes})
print(changes_df.round(3))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Test parallel trends: interact treatment with each pre-period week
# This tests whether the treatment and control groups had different trends before the switch
df['week_centered'] = df['week'] - 14  # Center at switch point

# Create week dummies for pre-period trend test
pre_week_dummies = pd.get_dummies(df[df['week'] < 15]['week'], prefix='pre_week')
pre_week_dummies['switched_branch'] = df[df['week'] < 15]['switched_branch']

# Test if pre-period trends differ between groups
pre_period = df[df['week'] < 15].copy()
pre_period['week_num'] = pre_period['week']

# Simple test: regress service score on week, switched_branch, and their interaction
pre_model = ols('service_score ~ week_num + switched_branch + week_num:switched_branch', data=pre_period).fit(cov_type='cluster', cov_kwds={'groups': pre_period['branch_id']})
print("=== Pre-Period Trend Test ===")
print("Testing if treatment and control had different trends before the switch")
print(pre_model.summary())

# The interaction term (week_num:switched_branch) tests parallel trends
# If not significant, parallel trends assumption holds

# Calculate effect size
# Mean service score is about 50, so a 2.13 point increase is about 4.3%
mean_score = df['service_score'].mean()
treatment_effect = 2.13
pct_change = (treatment_effect / mean_score) * 100
print(f"\n=== Effect Size ===")
print(f"Mean service score: {mean_score:.2f}")
print(f"Treatment effect: {treatment_effect:.2f} points")
print(f"Percentage change: {pct_change:.1f}%")

# Confidence interval from the regression
# From earlier: coef = 2.1207, SE = 0.582
# 95% CI: 2.1207 ± 1.96*0.582 = [0.98, 3.26]
ci_lower = 2.1207 - 1.96 * 0.582
ci_upper = 2.1207 + 1.96 * 0.582
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")

# Power analysis / sample size info

print(f"  - Control branches: {df[df['switched_branch']==0]['branch_id'].nunique()}")
print(f"Weeks: {df['week'].min()} to {df['week'].max()}")
print(f"  - Pre-switch weeks: 6-14 (9 weeks)")
