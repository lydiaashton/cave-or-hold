import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model 1: Simple DiD (no controls)
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Model 1: Simple DiD (clustered SEs by branch) ===")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Model 2: DiD with controls (clustered SEs by branch) ===")
print(model2.summary().tables[1])

# Model 3: With branch fixed effects
df_with_fe = df.copy()
df_with_fe['branch_id'] = df_with_fe['branch_id'].astype('category')
model3 = ols('service_score ~ after_switch + treatment + staffing_index + visits_index + C(branch_id)', data=df_with_fe).fit(cov_type='cluster', cov_kwds={'groups': df_with_fe['branch_id']})
print("\n=== Model 3: DiD with branch fixed effects (clustered SEs) ===")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study - create interactions for each post-switch week
for week in range(15, 23):
    df[f'post_week_{week}'] = ((df['week'] == week) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
event_study_vars = ['switched_branch'] + [f'post_week_{w}' for w in range(15, 23)]
formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(week) + staffing_index + visits_index'
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== EVENT STUDY RESULTS ===")
print("Coefficients represent the treatment effect for each post-switch week")
print("(relative to the pre-switch baseline, controlling for week fixed effects)")
print()
for week in range(15, 23):
    coef = event_model.params[f'post_week_{week}']
    se = event_model.bse[f'post_week_{week}']
    ci_low = coef - 1.96 * se
    ci_high = coef + 1.96 * se
    sig = "***" if event_model.pvalues[f'post_week_{week}'] < 0.01 else "**" if event_model.pvalues[f'post_week_{week}'] < 0.05 else "*" if event_model.pvalues[f'post_week_{week}'] < 0.1 else ""
    print(f"Week {week}: {coef:.2f} (SE: {se:.2f}, 95% CI: [{ci_low:.2f}, {ci_high:.2f}]) {sig}")
