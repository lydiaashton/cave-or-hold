import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't switch?
switched_counts = df.groupby('switched_branch')['branch_id'].nunique()
print("Branches by switch status:")
print(switched_counts)
print("\nPercentage switched:", switched_counts[1] / switched_counts.sum() * 100, "%")

# Average service scores before and after by switch status
print("\n\nAverage service scores:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(pivot)

# Calculate the difference-in-differences
before_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n\nBefore/After means:")
print(f"Switched branches - Before: {before_switched:.2f}, After: {after_switched:.2f}, Change: {after_switched - before_switched:.2f}")
print(f"Control branches - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")

diff_in_diff = (after_switched - before_switched) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {diff_in_diff:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# Regression with covariates
model_with_covariates = ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', 
                            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("DiD Regression with Covariates:")
print(model_with_covariates.summary())

# Check effect heterogeneity - does the effect vary by branch characteristics?
# Let's look at the distribution of branch-level effects
print("\n\nBranch-level analysis:")
branch_summary = df.groupby('branch_id').agg({
    'switched_branch': 'first',
    'service_score': ['mean', 'std']
}).round(2)
branch_summary.columns = ['switched', 'avg_score', 'score_std']
print(branch_summary.head(10))

# Calculate branch-level before/after changes
before = df[df['after_switch'] == 0].groupby('branch_id')['service_score'].mean()
after = df[df['after_switch'] == 1].groupby('branch_id')['service_score'].mean()
branch_changes = after - before
branch_changes = branch_changes.reset_index()
branch_changes.columns = ['branch_id', 'score_change']

# Merge with switch status
branch_switch = df.groupby('branch_id')['switched_branch'].first().reset_index()
branch_analysis = branch_changes.merge(branch_switch, on='branch_id')

print("\n\nBranch-level score changes:")
print("Switched branches - mean change:", branch_analysis[branch_analysis['switched_branch'] == 1]['score_change'].mean())
print("Control branches - mean change:", branch_analysis[branch_analysis['switched_branch'] == 0]['score_change'].mean())
print("\nSwitched branches - median change:", branch_analysis[branch_analysis['switched_branch'] == 1]['score_change'].median())

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Get overall service score statistics for context
print("Overall service score statistics:")
print(f"Mean: {df['service_score'].mean():.2f}")
print(f"Std Dev: {df['service_score'].std():.2f}")
print(f"Min: {df['service_score'].min():.2f}")
print(f"Max: {df['service_score'].max():.2f}")

# Effect size in standard deviation units
effect_size = 2.04 / df['service_score'].std()
print(f"\nEffect size (in SD units): {effect_size:.2f}")

# What percentage improvement is this?
baseline = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
pct_improvement = 2.04 / baseline * 100
print(f"Percentage improvement relative to baseline: {pct_improvement:.1f}%")

# Check how many branches benefited vs were hurt
branch_switch = df.groupby('branch_id')['switched_branch'].first().reset_index()
before = df[df['after_switch'] == 0].groupby('branch_id')['service_score'].mean()
after = df[df['after_switch'] == 1].groupby('branch_id')['service_score'].mean()
branch_changes = (after - before).reset_index()
branch_changes.columns = ['branch_id', 'score_change']
branch_analysis = branch_changes.merge(branch_switch, on='branch_id')

switched_branches = branch_analysis[branch_analysis['switched_branch'] == 1]
print(f"\nAmong switched branches:")
print(f"  Improved: {(switched_branches['score_change'] > 0).sum()} ({(switched_branches['score_change'] > 0).mean()*100:.1f}%)")
print(f"  Declined: {(switched_branches['score_change'] < 0).sum()} ({(switched_branches['score_change'] < 0).mean()*100:.1f}%)")

control_branches = branch_analysis[branch_analysis['switched_branch'] == 0]
print(f"\nAmong control branches:")
print(f"  Improved: {(control_branches['score_change'] > 0).sum()} ({(control_branches['score_change'] > 0).mean()*100:.1f}%)")
print(f"  Declined: {(control_branches['score_change'] < 0).sum()} ({(control_branches['score_change'] < 0).mean()*100:.1f}%)")
