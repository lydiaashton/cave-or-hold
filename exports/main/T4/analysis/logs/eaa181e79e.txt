print(summary)

control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_diff = control_after - control_before

treatment_diff = treatment_after - treatment_before

# Difference-in-differences

print(f"Treatment group (switched): Before={treatment_before:.2f}, After={treatment_after:.2f}, Change={treatment_diff:.2f}")
print(f"\nDifference-in-Differences estimate: {did:.2f}")

import pandas as pd

from statsmodels.formula.api import ols

# Test parallel trends: interact switched_branch with week in pre-period only
# If parallel trends hold, these interactions should not be significant
df_pre = df[df['week'] < 15].copy()
df_pre['week_centered'] = df_pre['week'] - df_pre['week'].mean()
df_pre['trend_interaction'] = df_pre['switched_branch'] * df_pre['week_centered']

model_trends = ols('service_score ~ switched_branch + week_centered + trend_interaction', data=df_pre).fit(cov_type='cluster', cov_kwds={'groups': df_pre['branch_id']})
print("=== Parallel Trends Test (Pre-period only) ===")

print(f"  P-value: {model_trends.pvalues['trend_interaction']:.4f}")
print("\nIf p > 0.05, parallel trends assumption is supported (no differential pre-trends)")

# Also check event study style - effect by week in post period
df['post_week'] = df['week'].apply(lambda x: x - 14 if x >= 15 else 0)
df['post_week'] = df['post_week'].astype(int)

# Create dummies for each post week
for w in range(1, 9):
    df[f'post_week_{w}'] = ((df['week'] == 14 + w) & (df['switched_branch'] == 1)).astype(int)

# Regression with post-week dummies
post_week_vars = [f'post_week_{w}' for w in range(1, 9)]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(post_week_vars)
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
