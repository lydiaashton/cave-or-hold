import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD groups
# Treatment = switched_branch, Post = after_switch
# DiD interaction = switched_branch * after_switch

# First, let's see the 2x2 table of average service scores
pivot = df.pivot_table(
    values='service_score',
    index='switched_branch',
    columns='after_switch',
    aggfunc='mean'
)
print("Average Service Scores by Group and Period:")
print(pivot)
print()

# Calculate DiD manually
control_before = pivot.loc[0, 0]  # Non-switched, before
control_after = pivot.loc[0, 1]   # Non-switched, after
treated_before = pivot.loc[1, 0]  # Switched, before
treated_after = pivot.loc[1, 1]   # Switched, after

control_change = control_after - control_before
treated_change = treated_after - treated_before
did_estimate = treated_change - control_change

print("Difference-in-Differences Calculation:")
print(f"Control group (non-switched): {control_before:.2f} -> {control_after:.2f} (change: {control_change:.2f})")
print(f"Treatment group (switched): {treated_before:.2f} -> {treated_after:.2f} (change: {treated_change:.2f})")
print(f"DiD Estimate: {did_estimate:.2f}")
print()

# Check how many branches in each group
print("Branches by group:")
print(df.groupby('switched_branch')['branch_id'].nunique())
print()

# Run regression with standard errors clustered by branch
df['interaction'] = df['switched_branch'] * df['after_switch']

# Simple OLS first
model = ols('service_score ~ switched_branch + after_switch + interaction', data=df).fit()
print("Simple OLS Regression:")
print(model.summary())
print()

# Now let's check parallel trends - compare trends before the switch
pre_switch = df[df['week'] < 15].copy()
post_switch = df[df['week'] >= 15].copy()

print("Pre-switch trends (weeks 6-14):")
pre_pivot = pre_switch.pivot_table(
    values='service_score',
    index='week',
    columns='switched_branch',
    aggfunc='mean'
)
print(pre_pivot)
print()

# Calculate average pre-switch scores by group
pre_avg = pre_switch.groupby('switched_branch')['service_score'].mean()
print("Average pre-switch scores:")
print(pre_avg)
print(f"Difference before switch: {pre_avg[1] - pre_avg[0]:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Clustered standard errors by branch
df['interaction'] = df['switched_branch'] * df['after_switch']

# Regression with cluster-robust standard errors
model_clustered = ols('service_score ~ switched_branch + after_switch + interaction', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Regression with Cluster-Robust SE (by branch):")
print(model_clustered.summary())
print()

# Add controls for staffing and visits
model_controls = ols('service_score ~ switched_branch + after_switch + interaction + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Regression with Controls (clustered SE):")
print(model_controls.summary())
print()

# Check parallel trends more formally - event study approach
# Calculate average service score by week and group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print("Weekly averages by group:")
print(weekly_avg)
print()

# Calculate the difference between treatment and control each week
weekly_diff = weekly_avg[1] - weekly_avg[0]
print("Weekly treatment-control difference:")
print(weekly_diff)
print()

# Plot the trends
plt.figure(figsize=(12, 6))
plt.plot(weekly_avg.index, weekly_avg[0], 'o-', label='Control (Non-Switched)', linewidth=2, markersize=8)
plt.plot(weekly_avg.index, weekly_avg[1], 's-', label='Treatment (Switched)', linewidth=2, markersize=8)
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')
plt.xlabel('Week', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Score Trends: Treatment vs Control Groups', fontsize=14)
plt.legend(fontsize=11)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("Plot saved as service_score_trends.png")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Test parallel trends - compare pre-switch trends
pre_switch = df[df['week'] < 15].copy()

# Regress service score on switched_branch using only pre-switch data
# If parallel trends hold, this coefficient should not be significantly different from baseline
pre_model = ols('service_score ~ switched_branch', data=pre_switch).fit(cov_type='cluster', cov_kwds={'groups': pre_switch['branch_id']})
print("Pre-switch period only (testing baseline difference):")
print(pre_model.summary())
print()

# Test if post-switch difference is significantly different from pre-switch difference
# This is essentially what DiD does, but let's verify

# Calculate effect size relative to baseline
baseline_avg = df['service_score'].mean()
did_estimate = 1.69  # from the model with controls
effect_size_pct = (did_estimate / baseline_avg) * 100
print(f"Baseline average service score: {baseline_avg:.2f}")
print(f"DiD estimate: {did_estimate:.2f}")
print(f"Effect size as % of baseline: {effect_size_pct:.1f}%")
print()

# Check the distribution of service scores
print("Service score distribution:")
print(df['service_score'].describe())
print()

# Check if there's any issue with the data - any branches that switched mid-period?
# All switched branches should have after_switch=0 before week 15 and after_switch=1 from week 15
check = df.groupby('branch_id')['switched_branch'].nunique()
print("Branches with inconsistent switched_branch values:")
print(check[check > 1])
print()

# Summary stats by group

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create a DiD visualization showing the gap between treatment and control
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

# Calculate the gap
gap = weekly_avg[1] - weekly_avg[0]

fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 5))

# Plot 1: Trends
ax1.plot(weekly_avg.index, weekly_avg[0], 'o-', label='Control (Non-Switched)', linewidth=2, markersize=8, color='blue')
ax1.plot(weekly_avg.index, weekly_avg[1], 's-', label='Treatment (Switched)', linewidth=2, markersize=8, color='green')
ax1.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')
ax1.set_xlabel('Week', fontsize=12)
ax1.set_ylabel('Average Service Score', fontsize=12)
ax1.set_title('Service Score Trends by Group', fontsize=14)
ax1.legend(fontsize=10)
ax1.grid(True, alpha=0.3)
ax1.set_xticks(range(6, 23))

# Plot 2: Gap between treatment and control
ax2.bar(gap.index, gap.values, color=['gray' if w < 15 else 'steelblue' for w in gap.index], edgecolor='black')
ax2.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')
ax2.axhline(y=0, color='black', linewidth=1)
ax2.set_xlabel('Week', fontsize=12)
ax2.set_ylabel('Treatment - Control Gap', fontsize=12)
ax2.set_title('Weekly Gap Between Treatment and Control Groups', fontsize=14)
ax2.legend(fontsize=10)
ax2.grid(True, alpha=0.3, axis='y')
ax2.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('did_visualization.png', dpi=150)
print("DiD visualization saved as did_visualization.png")

# Calculate average gap before and after
pre_gap = gap[gap.index < 15].mean()
post_gap = gap[gap.index >= 15].mean()
print(f"\nAverage gap before switch (weeks 6-14): {pre_gap:.2f}")
print(f"Average gap after switch (weeks 15-22): {post_gap:.2f}")
print(f"Change in gap (DiD estimate): {post_gap - pre_gap:.2f}")
