import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*treated + beta2*post + beta3*(treated*post) + error
# beta3 is the DiD estimate

df['treated_post'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()
print("Basic DiD Regression:")
print(model.summary())

# Now with branch fixed effects (more robust)
df['branch_id'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ switched_branch + after_switch + treated_post + C(branch_id)', data=df).fit()
print("\n\nDiD Regression with Branch Fixed Effects:")
print(model_fe.summary())

# With week fixed effects as well
df['week'] = df['week'].astype(str)
model_fe_week = ols('service_score ~ switched_branch + after_switch + treated_post + C(branch_id) + C(week)', data=df).fit()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.multicomp import pairwise_tukeyhsd

df = pd.read_csv('branches.csv')

# Create interaction term
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Run regression with branch and week fixed effects, clustered standard errors
# Use a cleaner specification
df_dummies = pd.get_dummies(df, columns=['branch_id', 'week'], drop_first=True)

# Identify the fixed effect columns
fe_cols = [c for c in df_dummies.columns if c.startswith('branch_id_') or c.startswith('week_')]

# Run the regression
X = df_dummies[['switched_branch', 'after_switch', 'treated_post'] + fe_cols]
y = df_dummies['service_score']

# Add constant
X = sm.add_constant(X, has_constant='add')

model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df_dummies['branch_id']})
print("DiD with Branch and Week FE, Clustered SEs:")
print(model.summary())
