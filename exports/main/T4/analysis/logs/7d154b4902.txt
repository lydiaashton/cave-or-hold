import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Summary by group and period
print("=== Service Score by Group and Period ===")
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='after_switch', 
                       aggfunc=['mean', 'std', 'count'])
print(pivot)

# Calculate the DiD estimate manually
mean_control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
mean_control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
mean_treated_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
mean_treated_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"\nControl (no switch) - Before: {mean_control_before:.2f}, After: {mean_control_after:.2f}, Change: {mean_control_after - mean_control_before:.2f}")
print(f"Treated (switched) - Before: {mean_treated_before:.2f}, After: {mean_treated_after:.2f}, Change: {mean_treated_after - mean_treated_before:.2f}")
print(f"\nDifference-in-Differences estimate: {(mean_treated_after - mean_treated_before) - (mean_control_after - mean_control_before):.2f}")

# Run DiD regression
print("\n=== DiD Regression ===")
model = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
print(model.summary())

# DiD with controls
print("\n=== DiD Regression with Controls ===")
model_controls = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit()

pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_df).fit()

print("Interaction term coefficient:", pre_model.params['switched_branch:week_centered'])

df['did'] = df['switched_branch'] * df['after_switch']

model_clustered = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study with confidence intervals
df['did'] = df['switched_branch'] * df['after_switch']

# Create week dummies relative to switch
df['weeks_relative'] = df['week'] - 15

# Event study regression (omitting week -1 as reference)
event_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
for w in range(-9, 8):
    if w != -1:
        df[f'week_{w}'] = ((df['weeks_relative'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study
event_formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars) + ' + staffing_index + visits_index'
event_model = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the interaction terms
event_results = []
for w in range(-9, 8):
    if w == -1:
        event_results.append({'week': w, 'coef': 0, 'se': 0, 'ci_low': 0, 'ci_high': 0})
    else:
        coef = event_model.params[f'week_{w}']
        se = event_model.bse[f'week_{w}']
        event_results.append({
            'week': w, 
            'coef': coef, 
            'se': se,
            'ci_low': coef - 1.96*se,
            'ci_high': coef + 1.96*se
        })

event_df = pd.DataFrame(event_results)

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
ax.bar(event_df['week'], event_df['coef'], yerr=1.96*event_df['se'], 
       color=['orange' if w < 0 else 'green' for w in event_df['week']], 
       alpha=0.7, edgecolor='black', linewidth=1)
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch Point')
ax.set_xlabel('Weeks Relative to Switch (0 = Week 15)', fontsize=12)
ax.set_ylabel('Service Score Difference (Treated - Control)', fontsize=12)
ax.set_title('Event Study: Effect of New Procedure on Service Scores', fontsize=14, fontweight='bold')
ax.grid(True, alpha=0.3, axis='y')
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("Event study plot saved as 'event_study.png'")

print("\n=== Event Study Coefficients ===")
print(event_df.to_string(index=False))

post_avg_gap = df[df['week'] >= 15].groupby('switched_branch')['service_score'].mean()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create week dummies relative to switch with valid names
df['weeks_relative'] = df['week'] - 15

for w in range(-9, 8):
    if w != -1:  # omit week -1 as reference
        col_name = f'w{abs(w)}p' if w >= 0 else f'w{abs(w)}n'
        df[col_name] = ((df['weeks_relative'] == w) & (df['switched_branch'] == 1)).astype(int)

# Build formula
event_vars = []
for w in range(-9, 8):
    if w != -1:
        col_name = f'w{abs(w)}p' if w >= 0 else f'w{abs(w)}n'
        event_vars.append(col_name)

event_formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars) + ' + staffing_index + visits_index'
event_model = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients
event_results = []
for w in range(-9, 8):
    if w == -1:
        event_results.append({'week': w, 'coef': 0, 'se': 0, 'ci_low': 0, 'ci_high': 0})
    else:
        col_name = f'w{abs(w)}p' if w >= 0 else f'w{abs(w)}n'
        coef = event_model.params[col_name]
        se = event_model.bse[col_name]
        event_results.append({
            'week': w, 
            'coef': coef, 
            'se': se,
            'ci_low': coef - 1.96*se,
            'ci_high': coef + 1.96*se
        })

event_df = pd.DataFrame(event_results)

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
colors = ['orange' if w < 0 else 'green' for w in event_df['week']]
ax.bar(event_df['week'], event_df['coef'], 
       color=colors, alpha=0.7, edgecolor='black', linewidth=1, width=0.8)

# Add error bars
for i, row in event_df.iterrows():
    if row['week'] != -1:  # skip reference
        ax.plot([row['week'], row['week']], [row['ci_low'], row['ci_high']], 
                color='black', linewidth=2, solid_capstyle='round')

ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')
ax.set_xlabel('Weeks Relative to Switch (0 = Week 15)', fontsize=12)
ax.set_ylabel('Service Score Difference (Treated vs Control)', fontsize=12)
ax.set_title('Event Study: Effect of New Procedure on Service Scores', fontsize=14, fontweight='bold')
ax.grid(True, alpha=0.3, axis='y')
ax.legend()
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("Event study plot saved as 'event_study.png'")

print("\n=== Event Study Coefficients ===")
print(event_df.to_string(index=False))

# Key stats
print("\n=== Key Statistics ===")
print(f"DiD Estimate: {event_model.params['after_switch'] + event_model.params.get('w0p', 0):.2f}")

# Simple DiD for summary
df['did'] = df['switched_branch'] * df['after_switch']
did_model = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', 
                data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"\nMain DiD coefficient: {did_model.params['did']:.2f}")
print(f"95% CI: [{did_model.params['did'] - 1.96*did_model.bse['did']:.2f}, {did_model.params['did'] + 1.96*did_model.bse['did']:.2f}]")
print(f"P-value: {did_model.pvalues['did']:.4f}")

# Effect size relative to baseline
baseline = df[df['after_switch']==0]['service_score'].mean()
print(f"\nBaseline average service score: {baseline:.1f}")
print(f"Effect size as % of baseline: {did_model.params['did']/baseline*100:.1f}%")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the main trend visualization
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Left panel: Raw trends
ax1 = axes[0]
for switched in [0, 1]:
    weekly_avg = df[df['switched_branch'] == switched].groupby('week')['service_score'].mean()
    label = 'Switched to New Procedure' if switched == 1 else 'Kept Old Procedure'
    color = 'green' if switched == 1 else 'blue'
    ax1.plot(weekly_avg.index, weekly_avg.values, marker='o', linewidth=2.5, label=label, 
             color=color, markersize=8)

ax1.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')
ax1.set_xlabel('Week', fontsize=12)
ax1.set_ylabel('Average Service Score', fontsize=12)
ax1.set_title('Service Score Trends by Branch Type', fontsize=14, fontweight='bold')
ax1.legend(loc='upper left')
ax1.grid(True, alpha=0.3)
ax1.set_xticks(range(6, 23))

# Right panel: Gap between treated and control
ax2 = axes[1]
gap_by_week = df.groupby('week').apply(
    lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()
)

colors = ['orange' if w < 15 else 'green' for w in gap_by_week.index]
ax2.bar(gap_by_week.index, gap_by_week.values, color=colors, alpha=0.7, edgecolor='black')
ax2.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point')
ax2.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax2.set_xlabel('Week', fontsize=12)
ax2.set_ylabel('Score Gap (Switched - Non-Switched)', fontsize=12)
ax2.set_title('Gap Between Switched and Non-Switched Branches', fontsize=14, fontweight='bold')
ax2.legend()
ax2.grid(True, alpha=0.3, axis='y')
ax2.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('trends_and_gap.png', dpi=150)
print("Visualization saved as 'trends_and_gap.png'")

# Final DiD model with all relevant stats
df['did'] = df['switched_branch'] * df['after_switch']
did_model = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', 
                data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n" + "="*60)
print("FINAL ANALYSIS SUMMARY")

print(f"\nDifference-in-Differences Estimate: {did_coef:.2f} points")

print(f"Statistically Significant: {'Yes' if did_pval < 0.05 else 'No'}")

# Practical significance
baseline = df[df['after_switch']==0]['service_score'].mean()
