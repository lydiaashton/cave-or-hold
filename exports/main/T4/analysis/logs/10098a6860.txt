import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Check if the effect varies over time (event study approach)
# Create week dummies interacted with treatment
df['switched_branch'] = df['switched_branch'].astype(int)
df['after_switch'] = df['after_switch'].astype(int)

# Create week indicators relative to switch (week 15 is the switch)
df['week_relative'] = df['week'] - 15

# Run regression with week fixed effects and treatment interaction
# This is an event study design
df['treatment'] = df['switched_branch'] * df['after_switch']

# Let's do week-by-week effects
week_effects = []
for week in range(6, 23):
    week_dummy = (df['week'] == week).astype(int)
    interaction = week_dummy * df['switched_branch']
    
    # Regress service_score on switched_branch, week dummy, and their interaction
    temp_df = df.copy()
    temp_df['week_dum'] = week_dummy
    temp_df['interaction'] = interaction
    
    model = ols('service_score ~ switched_branch + week_dum + interaction', data=temp_df).fit(cov_type='cluster', cov_kwds={'groups': temp_df['branch_id']})
    
    coef = model.params['interaction']
    se = model.bse['interaction']
    p = model.pvalues['interaction']
    ci_lower = model.conf_int().loc['interaction', 0]
    ci_upper = model.conf_int().loc['interaction', 1]
    
    week_effects.append({
        'week': week,
        'coef': coef,
        'se': se,
        'p_value': p,
        'ci_lower': ci_lower,
        'ci_upper': ci_upper,
        'period': 'pre' if week < 15 else 'post'
    })

week_effects_df = pd.DataFrame(week_effects)
print("Week-by-week treatment effects:")
