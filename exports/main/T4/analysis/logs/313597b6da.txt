import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate weekly means for treatment and control groups
weekly_means = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Create a plot
plt.figure(figsize=(12, 6))

for group in [0, 1]:
    subset = weekly_means[weekly_means['switched_branch'] == group]
    label = 'Switched branches (treatment)' if group == 1 else 'Non-switched branches (control)'
    plt.plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2)

plt.axvline(x=14.5, color='red', linestyle='--', label='Switch point (week 15)')

plt.ylabel('Mean Service Score')
plt.title('Service Scores Over Time: Treatment vs Control Groups')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("Plot saved as service_scores_trend.png")

# Check parallel trends more formally - compare pre-treatment trends
pre_switch = df[df['week'] < 15].copy()

# Run a regression on pre-treatment data to check if trends differ

pre_switch['treatment_time'] = pre_switch['switched_branch'] * pre_switch['time_trend']

model_pre = ols('service_score ~ switched_branch + time_trend + treatment_time + C(branch_id)', data=pre_switch).fit()

# Get confidence interval for the treatment effect
df['treatment'] = df['switched_branch'] * df['after_switch']
model_full = ols('service_score ~ treatment + C(branch_id) + C(week)', data=df).fit()
treatment_coef = model_full.params['treatment']
treatment_se = model_full.bse['treatment']
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se
print(f"\n95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")

# This could help understand the mechanism
print("Changes in staffing_index and visits_index by group:")
for var in ['staffing_index', 'visits_index']:
    control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)][var].mean()
    control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)][var].mean()
    treated_before = df[(df['switched_branch']==1) & (df['after_switch']==0)][var].mean()
    treated_after = df[(df['switched_branch']==1) & (df['after_switch']==1)][var].mean()
    
    control_change = control_after - control_before
    treated_change = treated_after - treated_before
    did = treated_change - control_change
    
    print(f"\n{var}:")
    print(f"  Control: {control_before:.3f} -> {control_after:.3f} (change: {control_change:.3f})")
    print(f"  Treated: {treated_before:.3f} -> {treated_after:.3f} (change: {treated_change:.3f})")
    print(f"  DiD estimate: {did:.3f}")
