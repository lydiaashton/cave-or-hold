import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create week dummies for event study (parallel trends check)
# Use week 14 as reference (last pre-treatment week)
df['week_6'] = (df['week'] == 6).astype(int)
df['week_7'] = (df['week'] == 7).astype(int)
df['week_8'] = (df['week'] == 8).astype(int)
df['week_9'] = (df['week'] == 9).astype(int)
df['week_10'] = (df['week'] == 10).astype(int)
df['week_11'] = (df['week'] == 11).astype(int)
df['week_12'] = (df['week'] == 12).astype(int)
df['week_13'] = (df['week'] == 13).astype(int)
# week 14 is reference
df['week_15'] = (df['week'] == 15).astype(int)
df['week_16'] = (df['week'] == 16).astype(int)
df['week_17'] = (df['week'] == 17).astype(int)
df['week_18'] = (df['week'] == 18).astype(int)
df['week_19'] = (df['week'] == 19).astype(int)
df['week_20'] = (df['week'] == 20).astype(int)
df['week_21'] = (df['week'] == 21).astype(int)
df['week_22'] = (df['week'] == 22).astype(int)

# Event study regression - interact each week dummy with treatment
week_cols = ['week_6', 'week_7', 'week_8', 'week_9', 'week_10', 'week_11', 'week_12', 'week_13',
             'week_15', 'week_16', 'week_17', 'week_18', 'week_19', 'week_20', 'week_21', 'week_22']

# Create interaction terms
for col in week_cols:
    df[f'{col}_treat'] = df[col] * df['switched_branch']

# Run event study regression
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'{col}_treat' for col in week_cols])
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Regression (Parallel Trends Check):")
print("="*80)
# Print only the interaction coefficients
coef_table = model_event.summary2().tables[1]
treat_coefs = coef_table[coef_table.index.str.contains('_treat')]
print(treat_coefs)

# Extract coefficients and confidence intervals for plotting
results = []
for col in week_cols:
    coef_name = f'{col}_treat'
    if coef_name in model_event.params.index:
        coef = model_event.params[coef_name]
        se = model_event.bse[coef_name]
        ci_lower = coef - 1.96 * se
        ci_upper = coef + 1.96 * se
        week_num = int(col.replace('week_', ''))
        results.append({'week': week_num, 'coef': coef, 'se': se, 'ci_lower': ci_lower, 'ci_upper': ci_upper})

event_df = pd.DataFrame(results)
event_df['period'] = event_df['week'].apply(lambda x: 'Pre' if x < 15 else 'Post')
print("\n\nEvent Study Coefficients:")
print(event_df)
