import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model 1: Basic DiD without controls
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Model 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 2: DiD with controls")
print(model2.summary().tables[1])

# Model 3: Add branch fixed effects (using C() for categorical)
model3 = ols('service_score ~ C(branch_id) + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 3: With branch fixed effects")
# Just print the key coefficients
print("\nKey coefficients from Model 3:")
print(model3.params[['after_switch', 'treatment', 'staffing_index', 'visits_index']])
print("\nStandard errors:")
print(model3.bse[['after_switch', 'treatment', 'staffing_index', 'visits_index']])
print("\nP-values:")
print(model3.pvalues[['after_switch', 'treatment', 'staffing_index', 'visits_index']])

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study - check pre-trends more formally
# Create week dummies interacted with treatment
df['week_centered'] = df['week'] - 14  # Center at switch week

# Create indicators for each week relative to switch
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != -1:  # omit week -1 as reference
        df[f'week_{w}'] = ((df['week'] - 14 == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
event_vars = [f'week_{w}' for w in range(-8, 9) if w != -1]
formula = f'service_score ~ C(branch_id) + after_switch + {" + ".join(event_vars)} + staffing_index + visits_index'
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the event study
event_coefs = {}
event_se = {}
for w in range(-8, 9):
    if w == -1:
        event_coefs[w] = 0  # reference category
        event_se[w] = 0
    else:
        var_name = f'week_{w}'
        event_coefs[w] = model_event.params[var_name]
        event_se[w] = model_event.bse[var_name]

# Plot event study
plt.figure(figsize=(10, 6))
weeks = list(event_coefs.keys())
coefs = [event_coefs[w] for w in weeks]
ses = [event_se[w] for w in weeks]

plt.errorbar(weeks, coefs, yerr=[1.96*s for s in ses], fmt='o', capsize=5, linewidth=2)
plt.axhline(y=0, color='black', linestyle='-', linewidth=1)
plt.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch Point')
plt.xlabel('Weeks Relative to Switch (0 = Week 15)')
plt.ylabel('Treatment Effect on Service Score')
plt.title('Event Study: Treatment Effects Over Time')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("Saved event study plot to event_study.png")

# Print pre-trend coefficients
print("\nPre-trend coefficients (weeks before switch):")
for w in range(-8, 0):
    print(f"Week {w}: coef={event_coefs[w]:.2f}, se={event_se[w]:.2f}, p={model_event.pvalues[f'week_{w}']:.3f}")

print("\nPost-trend coefficients (weeks after switch):")
for w in range(0, 9):
    if w != -1:
        print(f"Week {w}: coef={event_coefs[w]:.2f}, se={event_se[w]:.2f}, p={model_event.pvalues[f'week_{w}']:.3f}")
