import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run a basic DiD regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch*after_switch) + error
# beta3 is the DiD estimate

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Basic DiD Regression Results:")
print(model.summary())

# Now let's add branch fixed effects and week fixed effects for a more robust analysis
# This controls for time-invariant branch characteristics and common time trends

df_with_dummies = df.copy()

# Add branch fixed effects
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Add week fixed effects
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, week_dummies], axis=1)

# Build formula with fixed effects
branch_cols = [col for col in df_with_dummies.columns if col.startswith('branch')]
week_cols = [col for col in df_with_dummies.columns if col.startswith('week')]

formula = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(branch_cols) + ' + ' + ' + '.join(week_cols)

model_fe = ols(formula, data=df_with_dummies).fit()
print("\n\nDiD Regression with Branch and Week Fixed Effects:")
print(model_fe.summary())

# Extract just the treatment effect
print(f"\n\nTreatment effect (DiD estimate): {model_fe.params['treatment']:.3f}")
print(f"Standard error: {model_fe.bse['treatment']:.3f}")
print(f"95% CI: [{model_fe.conf_int().loc['treatment'][0]:.3f}, {model_fe.conf_int().loc['treatment'][1]:.3f}]")
print(f"P-value: {model_fe.pvalues['treatment']:.4f}")

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Summary of the analysis
print("=" * 60)
print("SUMMARY OF DIFFERENCE-IN-DIFFERENCES ANALYSIS")
print("=" * 60)

# Basic DiD
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

did_estimate = (treatment_after - treatment_before) - (control_after - control_before)

print(f"\n1. BASIC STATISTICS:")
print(f"   - Total observations: {len(df)}")
print(f"   - Branches: {df['branch_id'].nunique()}")
print(f"   - Control branches (old procedure): {df[df['switched_branch']==0]['branch_id'].nunique()}")
print(f"   - Treatment branches (new procedure): {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"   - Pre-switch weeks: 6-14 (9 weeks)")
print(f"   - Post-switch weeks: 15-22 (8 weeks)")

print(f"\n2. AVERAGE SERVICE SCORES:")
print(f"   Control group: {control_before:.2f} (before) -> {control_after:.2f} (after), change: {control_after - control_before:.2f}")
print(f"   Treatment group: {treatment_before:.2f} (before) -> {treatment_after:.2f} (after), change: {treatment_after - treatment_before:.2f}")

print(f"\n3. DIFFERENCE-IN-DIFFERENCES ESTIMATE:")
print(f"   DiD estimate: {did_estimate:.2f} points")
print(f"   Interpretation: The new procedure is associated with a {did_estimate:.2f} point increase in service scores")
print(f"   Relative to baseline (~50), this is a {did_estimate/50*100:.1f}% improvement")

# Regression results
from statsmodels.formula.api import ols
df['treatment'] = df['switched_branch'] * df['after_switch']
model_fe = ols('service_score ~ C(branch_id) + C(week) + after_switch + treatment', data=df).fit()

print(f"\n4. REGRESSION RESULTS (with branch and week fixed effects):")
print(f"   Treatment effect: {model_fe.params['treatment']:.3f}")
print(f"   Standard error: {model_fe.bse['treatment']:.3f}")
print(f"   95% CI: [{model_fe.conf_int().loc['treatment'][0]:.3f}, {model_fe.conf_int().loc['treatment'][1]:.3f}]")
print(f"   P-value: {model_fe.pvalues['treatment']:.4f}")
print(f"   Statistically significant: {'Yes' if model_fe.pvalues['treatment'] < 0.05 else 'No'}")

# Effect size context
print(f"\n5. CONTEXT:")
print(f"   Average service score across all branches: {df['service_score'].mean():.2f}")
print(f"   Standard deviation of service scores: {df['service_score'].std():.2f}")
print(f"   Effect size in SD units: {did_estimate / df['service_score'].std():.2f}")

# Check if the effect is meaningful
print(f"\n6. PRACTICAL SIGNIFICANCE:")
print(f"   The treatment branches started with lower scores ({treatment_before:.2f} vs {control_before:.2f})")
print(f"   After the switch, treatment branches improved while control branches declined slightly")
print(f"   The gap narrowed from {control_before - treatment_before:.2f} to {control_after - treatment_after:.2f} points")
