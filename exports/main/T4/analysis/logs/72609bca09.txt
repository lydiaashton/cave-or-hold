import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Count branches by treatment status
print("Branches by treatment status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate average service scores by group and period
print("\n\nAverage service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the difference-in-differences manually
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n\nMean service scores:")
print(f"Treatment group - Before: {before_treatment:.2f}, After: {after_treatment:.2f}, Change: {after_treatment - before_treatment:.2f}")
print(f"Control group - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")

did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")

# Weekly trends to check parallel trends assumption
print("\n\nWeekly average service scores by group:")
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(weekly.round(2))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treatment'] * df['post']

# Basic DiD regression
model = ols('service_score ~ treatment + post + DiD', data=df).fit()
print("Basic Difference-in-Differences Regression:")
print(model.summary())

# Now with branch fixed effects (more robust)
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = df.copy()
for col in branch_dummies.columns:
    df_with_fe[col] = branch_dummies[col]

# Regression with branch fixed effects
fe_cols = ['treatment', 'post', 'DiD'] + list(branch_dummies.columns)
X = df_with_fe[fe_cols]
X = sm.add_constant(X)
y = df_with_fe['service_score']

# This might be too many variables, let's use a different approach with formula
print("\n\n" + "="*60)
print("Regression with Branch Fixed Effects:")
print("="*60)

# Use categorical variable for branch fixed effects
model_fe = ols('service_score ~ C(branch_id) + post + DiD', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_fe.summary())

# Extract the DiD coefficient
did_coef = model_fe.params['DiD']
did_se = model_fe.bse['DiD']
did_pvalue = model_fe.pvalues['DiD']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se

print(f"\n\nDiD Coefficient: {did_coef:.3f}")
print(f"Standard Error: {did_se:.3f}")
