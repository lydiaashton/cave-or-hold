import pandas as pd

df = pd.read_csv('branches.csv')

# Count branches by switch status
switch_counts = df.groupby('branch_id')['switched_branch'].first().value_counts()

print(switch_counts)

print(f"Non-switched branches: {switch_counts.get(0, 0)}")

print("Average Service Scores by Group and Period")
print("="*60)

# Before switch (weeks 6-14)
before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

print("\nBEFORE switch (weeks 6-14):")
before_avg = before.groupby('switched_branch')['service_score'].mean()
print(f"  Non-switched branches (control): {before_avg[0]:.2f}")
print(f"  Switched branches (treatment): {before_avg[1]:.2f}")
print(f"  Difference (treatment - control): {before_avg[1] - before_avg[0]:.2f}")

print("\nAFTER switch (weeks 15-22):")
after_avg = after.groupby('switched_branch')['service_score'].mean()
print(f"  Non-switched branches (control): {after_avg[0]:.2f}")
print(f"  Switched branches (treatment): {after_avg[1]:.2f}")
print(f"  Difference (treatment - control): {after_avg[1] - after_avg[0]:.2f}")

# Difference-in-Differences calculation
diff_before = before_avg[1] - before_avg[0]
diff_after = after_avg[1] - after_avg[0]
did_effect = diff_after - diff_before

print("\n" + "="*60)
print("DIFFERENCE-IN-DIFFERENCES ESTIMATE")
print("="*60)
print(f"  Change in control group: {after_avg[0] - before_avg[0]:.2f}")
print(f"  Change in treatment group: {after_avg[1] - before_avg[1]:.2f}")
print(f"  DiD Effect (treatment - control): {did_effect:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a Difference-in-Differences regression
# Model: service_score = β0 + β1*treatment + β2*post + β3*(treatment*post) + controls + ε
# β3 is the DiD effect we care about

df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Basic DiD model
model = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit()
print("="*60)
print("DIFFERENCE-IN-DIFFERENCES REGRESSION")
print("="*60)
print(model.summary())

# Model with branch fixed effects
print("\n" + "="*60)
print("DID WITH BRANCH FIXED EFFECTS")
print("="*60)

# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df, branch_dummies], axis=1)

# Build formula with branch fixed effects
branch_cols = [col for col in df_with_dummies.columns if col.startswith('branch_')]
formula = 'service_score ~ switched_branch + after_switch + treatment_post + ' + ' + '.join(branch_cols)
model_fe = ols(formula, data=df_with_dummies).fit()

print(f"R-squared: {model_fe.rsquared:.4f}")
print(f"\nKey coefficients:")
print(f"  treatment_post (DiD effect): {model_fe.params['treatment_post']:.3f}")
print(f"  Std Error: {model_fe.bse['treatment_post']:.3f}")
print(f"  t-statistic: {model_fe.tvalues['treatment_post']:.3f}")
print(f"  p-value: {model_fe.pvalues['treatment_post']:.6f}")

# 95% confidence interval
ci_lower = model_fe.params['treatment_post'] - 1.96 * model_fe.bse['treatment_post']
ci_upper = model_fe.params['treatment_post'] + 1.96 * model_fe.bse['treatment_post']
print(f"  95% CI: [{ci_lower:.3f}, {ci_upper:.3f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check parallel trends - average service score by week for each group
print("="*60)
print("PARALLEL TRENDS CHECK: Weekly Average Service Scores")
print("="*60)

weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_avg.columns = ['Control (No Switch)', 'Treatment (Switched)']
weekly_avg['Difference'] = weekly_avg['Treatment (Switched)'] - weekly_avg['Control (No Switch)']

print(weekly_avg.to_string())

# Check pre-trend differences (weeks 6-14)
print("\n" + "="*60)
print("PRE-TREND ANALYSIS (Weeks 6-14)")
print("="*60)
pre_trend = weekly_avg[weekly_avg.index < 15]
print(f"Average difference before switch: {pre_trend['Difference'].mean():.3f}")
print(f"Std dev of differences: {pre_trend['Difference'].std():.3f}")

# Run DiD with clustered standard errors by branch
print("\n" + "="*60)
print("DID WITH CLUSTERED STANDARD ERRORS (by branch)")
print("="*60)

df['treatment_post'] = df['switched_branch'] * df['after_switch']
model = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit()

# Get cluster-robust standard errors
cov_clusters = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(cov_clusters.summary())

# Also add controls (staffing_index and visits_index)
print("\n" + "="*60)
print("DID WITH CONTROLS AND CLUSTERED SE")
print("="*60)
model_controls = ols('service_score ~ switched_branch + after_switch + treatment_post + staffing_index + visits_index', data=df).fit()
cov_clusters_ctrl = model_controls.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(cov_clusters_ctrl.summary())

# Create the plot
fig, ax = plt.subplots(figsize=(12, 7))

# Plot the two groups
ax.plot(weekly_avg.index, weekly_avg['Control (No Switch)'], 'o-', 
        label='Control Branches (Old Procedure)', linewidth=2, markersize=8, color='gray')
ax.plot(weekly_avg.index, weekly_avg['Treatment (Switched)'], 's-', 
        label='Treatment Branches (New Procedure)', linewidth=2, markersize=8, color='blue')

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')

# Add shaded region for post-switch period
ax.axvspan(14.5, 22, alpha=0.2, color='green', label='Post-Switch Period')

# Labels and title
ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Score Trends: Treatment vs Control Branches\n(Before and After Procedure Switch)', fontsize=14, fontweight='bold')
ax.legend(loc='lower right', fontsize=10)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

ax.annotate('Pre-Switch Period\n(Weeks 6-14)', xy=(10, 56), fontsize=10, alpha=0.7)
