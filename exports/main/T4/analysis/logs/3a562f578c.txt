import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Check how many branches switched vs didn't
switched_counts = df.groupby('switched_branch')['branch_id'].nunique()
print("Number of branches by group:")
print(switched_counts)
print("\nProportion switched:", switched_counts[1] / switched_counts.sum())

# Check service scores before and after by group
print("\n\nService Score Summary by Group and Period:")
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Calculate the difference-in-differences manually
# Before period (after_switch=0)
before_switched = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()

# After period (after_switch=1)
after_switched = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print("\n\nMean Service Scores:")
print(f"Switched branches - Before: {before_switched:.2f}, After: {after_switched:.2f}, Change: {after_switched - before_switched:.2f}")
print(f"Control branches - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")

# Difference in differences
did = (after_switched - before_switched) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {did:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

# Run DiD regression with branch fixed effects and week fixed effects
# This is the standard approach for DiD with panel data

# First, let's do a simple DiD regression
model_simple = ols('service_score ~ treated + post + treated_post', data=df).fit()
print("Simple DiD Regression (no fixed effects):")
print(model_simple.summary().tables[1])

# Now with branch and week fixed effects
# Create dummy variables for branch and week
df = df.copy()
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)

# Combine with original data
df_fe = pd.concat([df, branch_dummies, week_dummies], axis=1)

# Get all the fixed effect column names
branch_cols = [c for c in df_fe.columns if c.startswith('branch_')]
week_cols = [c for c in df_fe.columns if c.startswith('week_')]

# Build the formula
fe_formula = 'service_score ~ treated + post + treated_post + ' + ' + '.join(branch_cols) + ' + ' + ' + '.join(week_cols)

# This might be too many variables, let's try a different approach using categorical variables
model_fe = ols('service_score ~ treated + post + treated_post + C(branch_id) + C(week)', data=df).fit()
print("\n\nDiD Regression with Branch and Week Fixed Effects:")

# Get the coefficient and confidence interval for treated_post
