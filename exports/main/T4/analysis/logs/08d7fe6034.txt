import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Count switched vs non-switched branches
switched_counts = df.groupby('branch_id')['switched_branch'].first().value_counts()
print("Branches by treatment status:")
print(switched_counts)
print(f"\nTotal: {switched_counts.sum()} branches")
print(f"Switched: {switched_counts.get(1, 0)} ({switched_counts.get(1, 0)/switched_counts.sum()*100:.1f}%)")
print(f"Control: {switched_counts.get(0, 0)} ({switched_counts.get(0, 0)/switched_counts.sum()*100:.1f}%)")

# Check balance - average service scores before the switch (weeks 6-14)
before_data = df[df['week'] < 15]
after_data = df[df['week'] >= 15]

print("\n=== Pre-switch period (weeks 6-14) ===")
pre_scores = before_data.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print(pre_scores)

print("\n=== Post-switch period (weeks 15-22) ===")
post_scores = after_data.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print(post_scores)

# Calculate the difference-in-differences manually
switched_pre = before_data[before_data['switched_branch']==1]['service_score'].mean()
control_pre = before_data[before_data['switched_branch']==0]['service_score'].mean()
switched_post = after_data[after_data['switched_branch']==1]['service_score'].mean()
control_post = after_data[after_data['switched_branch']==0]['service_score'].mean()

print("\n=== Difference-in-Differences Calculation ===")
print(f"Switched branches: Pre={switched_pre:.2f}, Post={switched_post:.2f}, Change={switched_post-switched_pre:.2f}")
print(f"Control branches:  Pre={control_pre:.2f}, Post={control_post:.2f}, Change={control_post-control_pre:.2f}")
did_effect = (switched_post - switched_pre) - (control_post - control_pre)
print(f"\nDiD Estimate (treatment effect): {did_effect:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

# Run DiD regression with cluster-robust standard errors
# Using OLS first, then we'll cluster by branch
model = ols('service_score ~ treated + post + treated_post', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# Now with cluster-robust standard errors by branch
model_clustered = ols('service_score ~ treated + post + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Regression with Cluster-Robust SE (by branch) ===")
print(model_clustered.summary())

# Also check if including controls changes anything
model_controls = ols('service_score ~ treated + post + treated_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Regression with Controls (Cluster-Robust SE) ===")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Additional robustness checks

# 1. Check if the effect is consistent across the post period
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

# 2. Check effect size in context of the scale
print("=== Context for Effect Size ===")
print(f"Overall mean service score: {df['service_score'].mean():.2f}")
print(f"Overall std dev of service score: {df['service_score'].std():.2f}")
print(f"Effect size (2.0 points) as % of mean: {2.0/df['service_score'].mean()*100:.1f}%")
print(f"Effect size as fraction of std dev: {2.0/df['service_score'].std():.2f} SD")

# 3. Check if there's heterogeneity - does the effect vary by branch characteristics?
# For now, let's just look at the distribution of branch-level effects

# Calculate branch-level pre and post averages
branch_level = df.groupby(['branch_id', 'switched_branch']).agg({
    'service_score': ['mean'],
    'week': ['min', 'max']
}).reset_index()
branch_level.columns = ['branch_id', 'switched_branch', 'avg_score', 'week_min', 'week_max']

# Actually let's do pre/post at branch level
branch_pre = df[df['week'] < 15].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_pre.columns = ['branch_id', 'switched_branch', 'pre_score']
branch_post = df[df['week'] >= 15].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_post.columns = ['branch_id', 'switched_branch', 'post_score']

branch_change = branch_pre.merge(branch_post, on=['branch_id', 'switched_branch'])
branch_change['change'] = branch_change['post_score'] - branch_change['pre_score']

print("\n=== Branch-Level Changes ===")
print("Switched branches:")
switched_changes = branch_change[branch_change['switched_branch']==1]['change']
print(f"  Mean change: {switched_changes.mean():.2f}")
print(f"  Std dev: {switched_changes.std():.2f}")
print(f"  Min: {switched_changes.min():.2f}, Max: {switched_changes.max():.2f}")
print(f"  % with positive change: {(switched_changes > 0).mean()*100:.1f}%")

print("\nControl branches:")
control_changes = branch_change[branch_change['switched_branch']==0]['change']
print(f"  Mean change: {control_changes.mean():.2f}")
print(f"  Std dev: {control_changes.std():.2f}")
print(f"  Min: {control_changes.min():.2f}, Max: {control_changes.max():.2f}")
print(f"  % with positive change: {(control_changes > 0).mean()*100:.1f}%")

# 4. Statistical test comparing the changes
from scipy import stats
t_stat, p_val = stats.ttest_ind(switched_changes, control_changes)
print(f"\n=== Two-sample t-test on branch-level changes ===")
print(f"t-statistic: {t_stat:.2f}, p-value: {p_val:.4f}")
