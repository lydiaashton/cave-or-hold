import pandas as pd

df = pd.read_csv('branches.csv')

# Create period variable
df['period'] = df['after_switch'].map({0: 'Pre', 1: 'Post'})

# Average service scores by group and period
print("Average service scores by group and period:")
pivot = df.groupby(['switched_branch', 'period'])['service_score'].mean().unstack()
print(pivot)

# Calculate the differences
print("\nDifference-in-Differences calculation:")
control_pre = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_post = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_pre = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_post = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"Control (non-switched) - Pre: {control_pre:.2f}, Post: {control_post:.2f}, Change: {control_post - control_pre:.2f}")
print(f"Treatment (switched) - Pre: {treatment_pre:.2f}, Post: {treatment_post:.2f}, Change: {treatment_post - treatment_pre:.2f}")
did = (treatment_post - treatment_pre) - (control_post - control_pre)
print(f"\nDifference-in-Differences estimate: {did:.2f}")

import pandas as pd

df = pd.read_csv('branches.csv')

# Overall baseline
overall_mean = df['service_score'].mean()
overall_std = df['service_score'].std()

print(f"Overall service score mean: {overall_mean:.2f}")
print(f"Overall service score std: {overall_std:.2f}")

# Effect size
effect_size = 2.31
effect_size_relative = (effect_size / overall_mean) * 100
print(f"\nTreatment effect: {effect_size:.2f} points")
print(f"Relative to mean: {effect_size_relative:.1f}%")

# How many branches improved?
# Calculate branch-level before/after differences
branch_pre = df[df['after_switch'] == 0].groupby('branch_id')['service_score'].mean()
branch_post = df[df['after_switch'] == 1].groupby('branch_id')['service_score'].mean()
branch_change = branch_post - branch_pre

switched_branches = df[df['switched_branch'] == 1]['branch_id'].unique()
switched_changes = branch_change[switched_branches]

print(f"\nAmong switched branches:")
print(f"  Average change: {switched_changes.mean():.2f}")
print(f"  Std of change: {switched_changes.std():.2f}")
print(f"  % with positive change: {(switched_changes > 0).mean() * 100:.1f}%")
print(f"  % with negative change: {(switched_changes < 0).mean() * 100:.1f}%")

non_switched_branches = df[df['switched_branch'] == 0]['branch_id'].unique()
non_switched_changes = branch_change[non_switched_branches]

print(f"\nAmong non-switched branches:")
print(f"  Average change: {non_switched_changes.mean():.2f}")
print(f"  Std of change: {non_switched_changes.std():.2f}")
print(f"  % with positive change: {(non_switched_changes > 0).mean() * 100:.1f}%")
