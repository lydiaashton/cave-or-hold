import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch'] * df['after_switch']

# Calculate means for each group/time combination
print("=== Service Score Means by Group and Period ===")
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='after_switch', 
                       aggfunc='mean')
print(pivot)

# Calculate the DiD estimate manually
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treated_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treated_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print("\n=== Difference-in-Differences Calculation ===")
print(f"Control (non-switched) branches: {control_before:.2f} -> {control_after:.2f}, change = {control_after - control_before:.2f}")
print(f"Treated (switched) branches: {treated_before:.2f} -> {treated_after:.2f}, change = {treated_after - treated_before:.2f}")
did_effect = (treated_after - treated_before) - (control_after - control_before)
print(f"\nDiD Effect (treatment effect): {did_effect:.2f}")

# Run regression with standard errors
print("\n=== Regression Analysis ===")
model = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
print(model.summary())

# Run regression with cluster-robust standard errors (clustered by branch)
print("\n=== Regression with Cluster-Robust SE (by branch) ===")
model_clustered = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_clustered.summary())

import pandas as pd

df = pd.read_csv('branches.csv')

# Check parallel trends assumption - look at weekly averages before and after
print("=== Weekly Service Scores by Group ===")
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

# Check for any pre-existing trends (weeks 6-14)
print("\n=== Pre-switch trend analysis (weeks 6-14) ===")
pre_switch = df[df['week'] < 15].copy()
pre_switch['week_centered'] = pre_switch['week'] - 10  # center around middle of pre-period

# Regression on pre-period only to check if trends were parallel
pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_switch).fit(cov_type='cluster', cov_kwds={'groups': pre_switch['branch_id']})
print(pre_model.summary())

# Create a simple visualization data
plt.figure(figsize=(10, 6))
for switched in [0, 1]:
    subset = df[df['switched_branch'] == switched]
    weekly_means = subset.groupby('week')['service_score'].mean()
    label = 'Switched branches' if switched == 1 else 'Non-switched branches'
    plt.plot(weekly_means.index, weekly_means.values, marker='o', label=label, linewidth=2)
