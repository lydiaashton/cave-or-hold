# Weeks coverage
print("\nWeeks coverage:")
print(df.groupby('after_switch')['week'].unique())

# Check balance - how many observations per group
print("\nObservations per group:")
print(df.groupby(['switched_branch', 'after_switch']).size())

# Mean service scores by group and period
print("\nMean service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(pivot)

# Calculate the difference-in-differences manually
# Treatment group (switched=1): after - before
# Control group (switched=0): after - before
# DiD = (Treatment_after - Treatment_before) - (Control_after - Control_before)

means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
print("\nMeans table:")
print(means)

treatment_diff = means.loc[1, 1] - means.loc[1, 0]
control_diff = means.loc[0, 1] - means.loc[0, 0]
did_estimate = treatment_diff - control_diff

print(f"\nTreatment group change (after - before): {treatment_diff:.3f}")
print(f"Control group change (after - before): {control_diff:.3f}")
print(f"Difference-in-Differences estimate: {did_estimate:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run the DiD regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + epsilon
# beta3 is the DiD estimate

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Difference-in-Differences Regression Results:")
print(model.summary())

# Also run with robust standard errors (not clustered) for comparison
model_robust = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='HC3')
print("\n\nWith Robust Standard Errors:")
print(model_robust.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check parallel trends - look at weekly averages before the switch
pre_switch = df[df['week'] < 15].copy()

# Calculate weekly averages by group
weekly_avg = pre_switch.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

print("Weekly averages before switch (weeks 6-14):")
print(weekly_avg.pivot(index='week', columns='switched_branch', values='service_score'))

# Run regression with week fixed effects to check pre-trends
pre_switch['treatment'] = pre_switch['switched_branch'] * pre_switch['after_switch']

# Add week as categorical
pre_switch['week_cat'] = pre_switch['week'].astype(str)

# Check if there are differential pre-trends
model_pretrend = ols('service_score ~ switched_branch + C(week) + switched_branch:C(week)', data=pre_switch).fit(cov_type='cluster', cov_kwds={'groups': pre_switch['branch_id']})
print("\n\nPre-trend test (interaction of switched_branch with week):")
print(model_pretrend.summary())

# Now let's control for staffing and visits in the main model
df['treatment'] = df['switched_branch'] * df['after_switch']
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate weekly averages by group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Create the plot
fig, ax = plt.subplots(figsize=(10, 6))

# Plot for control group (switched_branch = 0)
control = weekly_avg[weekly_avg['switched_branch'] == 0]
ax.plot(control['week'], control['service_score'], marker='o', label='Control (Old Procedure)', linewidth=2, markersize=6, color='blue')

# Plot for treatment group (switched_branch = 1)
treatment = weekly_avg[weekly_avg['switched_branch'] == 1]
ax.plot(treatment['week'], treatment['service_score'], marker='s', label='Treatment (New Procedure)', linewidth=2, markersize=6, color='green')

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')

# Labels and title
ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Mean Service Score', fontsize=12)
ax.set_title('Service Scores by Branch Group Over Time', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=11)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("Plot saved as service_scores_trend.png")

# Also print the key numbers for the summary
print("\n" + "="*60)
print("KEY NUMBERS FOR SUMMARY")
print("="*60)

# Overall means
overall_mean = df['service_score'].mean()
print(f"Overall mean service score: {overall_mean:.2f}")

# By group and period
means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean()
print(f"\nControl group (old procedure):")
print(f"  Before switch (weeks 6-14): {means.loc[0, 0]:.2f}")
print(f"  After switch (weeks 15-22): {means.loc[0, 1]:.2f}")
print(f"  Change: {means.loc[0, 1] - means.loc[0, 0]:.2f}")

print(f"\nTreatment group (new procedure):")
print(f"  Before switch (weeks 6-14): {means.loc[1, 0]:.2f}")
print(f"  After switch (weeks 15-22): {means.loc[1, 1]:.2f}")
print(f"  Change: {means.loc[1, 1] - means.loc[1, 0]:.2f}")

print(f"\nDifference-in-Differences estimate: {means.loc[1, 1] - means.loc[1, 0] - (means.loc[0, 1] - means.loc[0, 0]):.2f}")
print(f"  (95% CI approximately: 0.73 to 3.01 based on clustered SE)")
print(f"  p-value: 0.001")
