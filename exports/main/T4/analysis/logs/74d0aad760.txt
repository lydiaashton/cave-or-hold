# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the treatment effect

df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main DiD model with clustered SE
model = ols('service_score ~ switched_branch + after_switch + treatment', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get the treatment effect and confidence interval
treatment_effect = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pvalue = model.pvalues['treatment']
ci_lower = treatment_effect - 1.96 * treatment_se
ci_upper = treatment_effect + 1.96 * treatment_se

print("=== Main Results Summary ===")
print(f"Treatment Effect Estimate: {treatment_effect:.2f} points")
print(f"Standard Error: {treatment_se:.2f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {treatment_pvalue:.3f}")

# Calculate effect size relative to baseline
baseline_score = df[df['after_switch']==0]['service_score'].mean()
print(f"\nBaseline (pre-switch) average service score: {baseline_score:.2f}")
print(f"Effect as % of baseline: {(treatment_effect/baseline_score)*100:.1f}%")

# Also get the covariate-adjusted estimate
model_cov = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
                data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
treatment_effect_cov = model_cov.params['treatment']
treatment_se_cov = model_cov.bse['treatment']
ci_lower_cov = treatment_effect_cov - 1.96 * treatment_se_cov
ci_upper_cov = treatment_effect_cov + 1.96 * treatment_se_cov

print(f"\n=== Covariate-Adjusted Results ===")
print(f"Treatment Effect Estimate: {treatment_effect_cov:.2f} points")
print(f"95% Confidence Interval: [{ci_lower_cov:.2f}, {ci_upper_cov:.2f}]")
print(f"P-value: {model_cov.pvalues['treatment']:.3f}")
