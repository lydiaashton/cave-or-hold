import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# The key coefficient is the interaction between switched_branch and after_switch
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression with branch fixed effects
model = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("DiD Regression with Branch Fixed Effects:")
print(model.summary().tables[1])

# Also run without branch FE for comparison
model_simple = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='HC3')
print("\n\nSimple DiD Regression (no fixed effects):")
print(model_simple.summary().tables[1])

# Check parallel trends - compare trends before the switch
print("\n\n=== Parallel Trends Check ===")
before_data = df[df['after_switch'] == 0]
# Check if there's a differential trend before the switch by week
before_data['week_centered'] = before_data['week'] - before_data['week'].mean()
before_data['treatment_trend'] = before_data['switched_branch'] * before_data['week_centered']

print("\nPre-treatment trend test:")
print(trend_model.summary().tables[1])

model_simple = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='HC3')
