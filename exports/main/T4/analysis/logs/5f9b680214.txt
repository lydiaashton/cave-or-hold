import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimator

model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Also run with branch fixed effects and week fixed effects for more robust estimation
print("\n=== Regression with Branch and Week Fixed Effects ===")
# Create dummies for branch and week
df_model = df.copy()
df_model = pd.get_dummies(df_model, columns=['branch_id', 'week'], drop_first=True)

# This might be too many variables, let's use a different approach with fixed effects
from linearmodels.panel import PanelOLS

# Set up panel data
df_panel = df.copy()
df_panel = df_panel.set_index(['branch_id', 'week'])
df_panel['intercept'] = 1

# DiD with entity (branch) and time (week) fixed effects
fe_model = PanelOLS(df_panel['service_score'], 
                    df_panel[['switched_branch', 'after_switch', 'switched_branch_after']],
                    entity_effects=True, 
                    time_effects=True)
# Need to create the interaction term first
df_panel['switched_branch_after'] = df_panel['switched_branch'] * df_panel['after_switch']

fe_model = PanelOLS(df_panel['service_score'], 
                    df_panel[['switched_branch', 'after_switch', 'switched_branch_after']],
                    entity_effects=True, 
                    time_effects=True)
fe_results = fe_model.fit(cov_type='clustered', cluster_entity=True)
print(fe_results)

from linearmodels.panel import PanelOLS

                    df_panel[['switched_branch', 'after_switch', 'switched_branch_after']],

                    time_effects=True)

# Now let's visualize the trends

weekly_trends = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check pre-trends: compare trends before week 15 for both groups
pre_switch = df[df['week'] < 15].copy()

# Calculate weekly averages before switch
pre_trends = pre_switch.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
pre_trends.columns = ['week', 'switched_branch', 'avg_service_score']

print("=== Pre-Switch Weekly Averages ===")
print(pre_trends)

# Calculate the difference between groups each week before switch
pre_trends_wide = pre_trends.pivot(index='week', columns='switched_branch', values='avg_service_score')
pre_trends_wide['diff'] = pre_trends_wide[1] - pre_trends_wide[0]
print("\n=== Difference (Switched - Control) Before Switch ===")
print(pre_trends_wide)

# Test if pre-trends are parallel using regression
# Regress service_score on week*switched_branch interaction for pre-period only
pre_switch['week_centered'] = pre_switch['week'] - pre_switch['week'].mean()
pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_switch).fit()
print("\n=== Pre-Trends Test (interaction should be ~0 if parallel) ===")
print(pre_model.summary())

# Also check post-switch trends
post_switch = df[df['week'] >= 15].copy()
post_trends = post_switch.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
post_trends.columns = ['week', 'switched_branch', 'avg_service_score']

print("\n=== Post-Switch Weekly Averages ===")
print(post_trends)

# Create a combined plot showing pre and post
fig, ax = plt.subplots(figsize=(12, 7))

for switched in [0, 1]:
    subset = df[df['switched_branch'] == switched].groupby('week')['service_score'].mean()
    label = 'Switched to New Procedure' if switched == 1 else 'Control (Old Procedure)'
    color = 'green' if switched == 1 else 'blue'
    ax.plot(subset.index, subset.values, marker='o', label=label, color=color, linewidth=2, markersize=6)

# Add vertical line at week 15
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')
ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Score Trends: Switched vs Control Branches\n(Vertical line marks when switch occurred)', fontsize=14)
ax.legend(fontsize=11)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_score_trends_full.png', dpi=150)
print("\nFull trend plot saved as service_score_trends_full.png")

from statsmodels.formula.api import ols

# Check if staffing_index and visits_index differ between groups
print("=== Staffing Index by Group and Period ===")
staffing_summary = df.groupby(['switched_branch', 'after_switch'])['staffing_index'].agg(['mean', 'std'])
print(staffing_summary)

print("\n=== Visits Index by Group and Period ===")
visits_summary = df.groupby(['switched_branch', 'after_switch'])['visits_index'].agg(['mean', 'std'])
print(visits_summary)

# Run DiD regression controlling for staffing and visits
print("\n=== DiD Regression with Controls ===")
model_with_controls = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit()
print(model_with_controls.summary())

# Check robustness: what if we look at just the post-period difference?
print("\n=== Simple Post-Period Comparison ===")
post_only = df[df['after_switch'] == 1]
post_comparison = post_only.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print(post_comparison)

# T-test for post-period difference
from scipy import stats
post_switched = post_only[post_only['switched_branch']==1]['service_score']
post_control = post_only[post_only['switched_branch']==0]['service_score']
