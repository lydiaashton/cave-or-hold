import pandas as pd

print("="*60)

treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate

df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Basic DiD Regression Results:")
print("="*60)
print(model.summary())

# Now with cluster-robust standard errors (clustered by branch)

print("\n\nDiD Regression with Clustered Standard Errors (by branch):")

# Also control for staffing and visits
model_full = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
