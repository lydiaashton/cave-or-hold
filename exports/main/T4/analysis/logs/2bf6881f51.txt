df['DiD'] = df['treated'] * df['post']

# Calculate group means
print("=== SERVICE SCORE MEANS ===\n")

# Before period (after_switch = 0)
before_switched = df[(df['after_switch'] == 0) & (df['switched_branch'] == 1)]['service_score'].mean()
before_control = df[(df['after_switch'] == 0) & (df['switched_branch'] == 0)]['service_score'].mean()

# After period (after_switch = 1)

print("\n=== DiD REGRESSION ===\n")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster

df = pd.read_csv('branches.csv')

# Create DiD variables
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# DiD regression with cluster-robust standard errors (clustered by branch)
print("=== DiD REGRESSION WITH CLUSTER-ROBUST SE (by branch) ===\n")
model = ols('service_score ~ treated + post + DiD', data=df).fit()
cluster_robust_se = cov_cluster(model, df['branch_id'])
print("Coefficients:")
print(model.params)
print("\nCluster-robust standard errors:")
print(np.sqrt(np.diag(cluster_robust_se)))
print("\nCluster-robust p-values:")
t_stats = model.params / np.sqrt(np.diag(cluster_robust_se))
from scipy import stats
p_values = 2 * (1 - stats.t.cdf(np.abs(t_stats), df=len(df)-4))
print(p_values)

# DiD with controls
print("\n=== DiD REGRESSION WITH CONTROLS ===\n")
model_controls = ols('service_score ~ treated + post + DiD + staffing_index + visits_index', data=df).fit()
cluster_robust_se_controls = cov_cluster(model_controls, df['branch_id'])
print(model_controls.summary())
print("\nCluster-robust SE for DiD coefficient:", np.sqrt(cluster_robust_se_controls.loc['DiD', 'DiD']))
t_stat = model_controls.params['DiD'] / np.sqrt(cluster_robust_se_controls.loc['DiD', 'DiD'])
p_val = 2 * (1 - stats.t.cdf(np.abs(t_stat), df=len(df)-6))
print("Cluster-robust t-stat:", t_stat)
print("Cluster-robust p-value:", p_val)

# Branch-level analysis (aggregate to branch level, before vs after)
print("\n=== BRANCH-LEVEL ANALYSIS ===\n")
branch_avg = df.groupby(['branch_id', 'switched_branch', 'after_switch'])['service_score'].mean().reset_index()
print("Branch-level data (first 10 rows):")
print(branch_avg.head(10))

# Calculate branch-level before/after differences
before = branch_avg[branch_avg['after_switch'] == 0][['branch_id', 'switched_branch', 'service_score']].rename(columns={'service_score': 'before_score'})
after = branch_avg[branch_avg['after_switch'] == 1][['branch_id', 'switched_branch', 'service_score']].rename(columns={'service_score': 'after_score'})
branch_diff = before.merge(after, on=['branch_id', 'switched_branch'])
branch_diff['change'] = branch_diff['after_score'] - branch_diff['before_score']

print("\nAverage change for switched branches:", branch_diff[branch_diff['switched_branch'] == 1]['change'].mean())
print("Average change for control branches:", branch_diff[branch_diff['switched_branch'] == 0]['change'].mean())
print("DiD at branch level:", branch_diff[branch_diff['switched_branch'] == 1]['change'].mean() - branch_diff[branch_diff['switched_branch'] == 0]['change'].mean())

# T-test on branch-level changes
from scipy import stats
switched_changes = branch_diff[branch_diff['switched_branch'] == 1]['change']
control_changes = branch_diff[branch_diff['switched_branch'] == 0]['change']
t_test = stats.ttest_ind(switched_changes, control_changes)
print("\nTwo-sample t-test on branch-level changes:")
print(f"t-statistic: {t_test.statistic:.3f}, p-value: {t_test.pvalue:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster
from scipy import stats

df = pd.read_csv('branches.csv')

# Create DiD variables
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# DiD with controls and cluster-robust SE
print("=== DiD REGRESSION WITH CONTROLS AND CLUSTER-ROBUST SE ===\n")
model_controls = ols('service_score ~ treated + post + DiD + staffing_index + visits_index', data=df).fit()
cluster_robust_se_controls = cov_cluster(model_controls, df['branch_id'])
se_array = np.sqrt(np.diag(cluster_robust_se_controls))

print("Coefficients and Cluster-Robust SE:")
for i, coef_name in enumerate(model_controls.params.index):
    se = se_array[i]
    t_stat = model_controls.params[coef_name] / se
    p_val = 2 * (1 - stats.t.cdf(np.abs(t_stat), df=len(df)-6))
    print(f"{coef_name}: coef={model_controls.params[coef_name]:.3f}, SE={se:.3f}, t={t_stat:.3f}, p={p_val:.3f}")

# Branch-level analysis
print("\n=== BRANCH-LEVEL ANALYSIS ===\n")
branch_avg = df.groupby(['branch_id', 'switched_branch', 'after_switch'])['service_score'].mean().reset_index()

# Calculate branch-level before/after differences
before = branch_avg[branch_avg['after_switch'] == 0][['branch_id', 'switched_branch', 'service_score']].rename(columns={'service_score': 'before_score'})
after = branch_avg[branch_avg['after_switch'] == 1][['branch_id', 'switched_branch', 'service_score']].rename(columns={'service_score': 'after_score'})
branch_diff = before.merge(after, on=['branch_id', 'switched_branch'])
branch_diff['change'] = branch_diff['after_score'] - branch_diff['before_score']

print(f"Number of switched branches: {(branch_diff['switched_branch'] == 1).sum()}")
print(f"Number of control branches: {(branch_diff['switched_branch'] == 0).sum()}")
print(f"\nAverage change for switched branches: {branch_diff[branch_diff['switched_branch'] == 1]['change'].mean():.2f}")
print(f"Average change for control branches: {branch_diff[branch_diff['switched_branch'] == 0]['change'].mean():.2f}")
print(f"DiD at branch level: {branch_diff[branch_diff['switched_branch'] == 1]['change'].mean() - branch_diff[branch_diff['switched_branch'] == 0]['change'].mean():.2f}")

# T-test on branch-level changes
switched_changes = branch_diff[branch_diff['switched_branch'] == 1]['change']
control_changes = branch_diff[branch_diff['switched_branch'] == 0]['change']
t_test = stats.ttest_ind(switched_changes, control_changes)
print(f"\nTwo-sample t-test on branch-level changes:")
print(f"t-statistic: {t_test.statistic:.3f}, p-value: {t_test.pvalue:.3f}")

# Confidence interval for DiD effect
switched_mean = switched_changes.mean()
control_mean = control_changes.mean()
