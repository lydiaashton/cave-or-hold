import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

# Run difference-in-differences regression
# Basic DiD model
model_basic = ols('service_score ~ treated + post + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Basic Difference-in-Differences Regression ===")
print(model_basic.summary())

# DiD with controls

    group_data = weekly_avg[weekly_avg['switched_branch'] == group]
    label = 'Switched Branches' if group == 1 else 'Non-Switched Branches'
    plt.plot(group_data['week'], group_data['service_score'], marker='o', label=label, linewidth=2, markersize=6)

plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')

plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("Chart saved as service_score_trends.png")

# Also calculate effect size
print("\n=== Effect Size Analysis ===")
# Standard deviation of service scores
std_score = df['service_score'].std()
print(f"Standard deviation of service scores: {std_score:.2f}")

# DiD effect
did_effect = 2.43
print(f"DiD effect: {did_effect:.2f}")
print(f"Effect size (Cohen's d): {did_effect / std_score:.2f}")

# Confidence interval for the DiD effect
print("\n=== 95% Confidence Interval for DiD Effect ===")
from statsmodels.formula.api import ols
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']
model = ols('service_score ~ treated + post + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
coef = model.params['treated_post']
se = model.bse['treated_post']
ci_lower = coef - 1.96 * se
ci_upper = coef + 1.96 * se
print(f"DiD coefficient: {coef:.2f}")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Calculate before/after for each branch
def calc_before(x):
    return x[x <= 14].mean()

def calc_after(x):
    return x[x >= 15].mean()

branch_stats = df.groupby(['branch_id', 'switched_branch']).agg({
    'service_score': ['mean', calc_before, calc_after]
}).reset_index()
branch_stats.columns = ['branch_id', 'switched', 'overall_mean', 'before_mean', 'after_mean']
branch_stats['change'] = branch_stats['after_mean'] - branch_stats['before_mean']

print("\n=== Branch-level Analysis ===")
print("\nSwitched branches (n=20):")
switched_changes = branch_stats[branch_stats['switched'] == 1]['change']
print(f"  Mean change: {switched_changes.mean():.2f}")
print(f"  Std dev: {switched_changes.std():.2f}")
print(f"  Min: {switched_changes.min():.2f}, Max: {switched_changes.max():.2f}")
print(f"  Branches with positive change: {(switched_changes > 0).sum()} / {len(switched_changes)}")

print("\nNon-switched branches (n=14):")
non_switched_changes = branch_stats[branch_stats['switched'] == 0]['change']
print(f"  Mean change: {non_switched_changes.mean():.2f}")
print(f"  Std dev: {non_switched_changes.std():.2f}")
print(f"  Min: {non_switched_changes.min():.2f}, Max: {non_switched_changes.max():.2f}")
print(f"  Branches with positive change: {(non_switched_changes > 0).sum()} / {len(non_switched_changes)}")

# T-test comparing changes between groups
from scipy import stats
t_stat, p_val = stats.ttest_ind(switched_changes, non_switched_changes)
print(f"\nT-test comparing changes: t={t_stat:.2f}, p={p_val:.3f}")
