import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
branches = df.groupby('branch_id')['switched_branch'].first()
print("Branches that switched:", branches.sum())
print("Branches that didn't switch:", len(branches) - branches.sum())
print("Total branches:", len(branches))

# Average service scores by group and period
print("\n=== Average Service Scores ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
pivot.columns = ['Before (weeks 6-14)', 'After (weeks 15-22)']
pivot.index = ['Control (no switch)', 'Treatment (switched)']
print(pivot)

# Calculate the differences
print("\n=== Differences ===")
before_diff = pivot['Before (weeks 6-14)'].iloc[1] - pivot['Before (weeks 6-14)'].iloc[0]
after_diff = pivot['After (weeks 15-22)'].iloc[1] - pivot['After (weeks 15-22)'].iloc[0]
did_estimate = after_diff - before_diff

print(f"Before: Treatment - Control = {before_diff:.3f}")
print(f"After: Treatment - Control = {after_diff:.3f}")
print(f"DiD Estimate (After - Before): {did_estimate:.3f}")

# Run a regression for DiD with standard errors
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Simple DiD regression
model = ols('service_score ~ treatment + post + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Regression Results ===")
print(model.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Check pre-trends: average service score by week for each group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Plot the trends
plt.figure(figsize=(12, 6))
for switched in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == switched]
    label = 'Switched branches' if switched == 1 else 'Non-switched branches'
    plt.plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2)

plt.axvline(x=14.5, color='red', linestyle='--', label='Switch point (week 15)')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Score Trends by Branch Group')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("Plot saved as service_score_trends.png")

# Check pre-trends statistically (weeks 6-14 only)
pre_data = df[df['week'] < 15].copy()
pre_data['week_centered'] = pre_data['week'] - 10  # center around middle of pre-period
pre_model = smf.ols('service_score ~ switched_branch * week_centered', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print("\n=== Pre-Trend Analysis (weeks 6-14) ===")
print(pre_model.summary())

# DiD with controls for staffing and visits
df['did'] = df['switched_branch'] * df['after_switch']
controlled_model = smf.ols('service_score ~ treatment + post + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD with Controls ===")
print(controlled_model.summary())

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['after_switch']

# Main DiD model
main_model = smf.ols('service_score ~ treatment + post + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Effect size in context
overall_mean = df['service_score'].mean()
effect_size = main_model.params['did']
pct_improvement = (effect_size / overall_mean) * 100

print(f"Overall mean service score: {overall_mean:.2f}")
print(f"Treatment effect: {effect_size:.3f}")
print(f"Percentage improvement: {pct_improvement:.1f}%")

# Confidence interval
ci_lower = main_model.conf_int().loc['did', 0]
ci_upper = main_model.conf_int().loc['did', 1]
print(f"95% CI: [{ci_lower:.3f}, {ci_upper:.3f}]")

# Number of observations
print(f"\nTotal observations: {len(df)}")
print(f"Branches: {df['branch_id'].nunique()}")
print(f"Switched branches: {df['switched_branch'].sum()}")
print(f"Control branches: {len(df['branch_id'].unique()) - df['switched_branch'].sum()}")

# Pre vs Post averages for treatment group
treatment_pre = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_post = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
control_pre = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_post = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nTreatment group: {treatment_pre:.2f} (before) -> {treatment_post:.2f} (after), change = {treatment_post - treatment_pre:.2f}")
print(f"Control group: {control_pre:.2f} (before) -> {control_post:.2f} (after), change = {control_post - control_pre:.2f}")
