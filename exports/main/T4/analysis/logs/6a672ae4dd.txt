import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with clustered standard errors
# Basic model: service_score ~ treatment + switched_branch + after_switch
model = ols('service_score ~ treatment + switched_branch + after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Basic DiD Regression ===")
print(model.summary())

# Model with controls
model_controls = ols('service_score ~ treatment + switched_branch + after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n=== DiD Regression with Controls ===")
print(model_controls.summary())

# Check parallel trends - compare trends before the switch
print("\n=== Parallel Trends Check ===")
before_data = df[df['week'] < 15]
print("Average service scores by week before switch:")
pivot_before = before_data.pivot_table('service_score', index='week', columns='switched_branch', aggfunc='mean')
print(pivot_before)

# Calculate week-by-week differences before switch
before_data_copy = before_data.copy()
before_data_copy['group_mean'] = before_data_copy.groupby(['week', 'switched_branch'])['service_score'].transform('mean')
before_data_copy['diff_from_control'] = before_data_copy['service_score'] - before_data_copy.groupby('week')['service_score'].transform(lambda x: x[before_data_copy.loc[x.index, 'switched_branch']==0].mean() if len(x[before_data_copy.loc[x.index, 'switched_branch']==0]) > 0 else np.nan)

# Actually let me do this more simply
print("\nWeek-by-week average service scores:")
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study regression with week fixed effects
# Create dummies for each week relative to treatment, excluding week -1 as reference
df['weeks_relative'] = df['week'] - 15

# Create indicators for each relative week
for wr in range(-9, 8):  # -9 to 7
    df[f'rel_week_{wr}'] = ((df['weeks_relative'] == wr) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (excluding rel_week_-1 as reference)
rel_weeks = [f'rel_week_{wr}' for wr in range(-9, 8) if wr != -1]
formula = f"service_score ~ switched_branch + after_switch + {' + '.join(rel_weeks)}"
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Regression ===")
print("Reference period: week -1 (week 14)")
print("\nCoefficients for treatment group by relative week:")
for wr in range(-9, 8):
    if wr != -1:
        coef = event_model.params[f'rel_week_{wr}']
        se = event_model.bse[f'rel_week_{wr}']
        pval = event_model.pvalues[f'rel_week_{wr}']
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"Week {wr:+d}: {coef:6.2f} (SE: {se:.2f}, p={pval:.3f}) {sig}")

# Test for pre-trend (joint significance of pre-period coefficients)
pre_period_coeffs = [f'rel_week_{wr}' for wr in range(-9, -1)]
from statsmodels.stats.contrast import ContrastResults
# Simple F-test for pre-trends
pre_test = event_model.f_test(pre_period_coeffs)
print(f"\nJoint test for pre-treatment trends (all pre-period coeffs = 0):")
print(f"F-statistic: {pre_test.fvalue:.2f}, p-value: {pre_test.pvalue:.3f}")

# Also check the post-period average effect
post_period_coeffs = [f'rel_week_{wr}' for wr in range(0, 8)]
post_test = event_model.f_test(post_period_coeffs)
print(f"\nJoint test for post-treatment effects (all post-period coeffs = 0):")
print(f"F-statistic: {post_test.fvalue:.2f}, p-value: {post_test.pvalue:.3f}")

# Average post-treatment effect
post_coefs = [event_model.params[f'rel_week_{wr}'] for wr in range(0, 8)]
print(f"\nAverage post-treatment effect: {np.mean(post_coefs):.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study regression with week fixed effects
df['weeks_relative'] = df['week'] - 15

# Create indicators for each relative week with cleaner names
for wr in range(-9, 8):  # -9 to 7
    col_name = f'rel_wk_{abs(wr)}_{"neg" if wr < 0 else "pos"}' if wr != 0 else 'rel_wk_0'
    df[col_name] = ((df['weeks_relative'] == wr) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (excluding week -1 as reference)
# Reference: rel_wk_1_neg (week -1)
rel_weeks = []
for wr in range(-9, 8):
    if wr == -1:
        continue  # skip reference
    elif wr < 0:
        rel_weeks.append(f'rel_wk_{abs(wr)}_neg')
    elif wr == 0:
        rel_weeks.append('rel_wk_0')
    else:
        rel_weeks.append(f'rel_wk_{wr}_pos')

formula = f"service_score ~ switched_branch + after_switch + {' + '.join(rel_weeks)}"
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Regression ===")
print("Reference period: week -1 (week 14)")
print("\nCoefficients for treatment group by relative week:")

# Map back to original week numbers for display
week_mapping = {}
for wr in range(-9, 8):
    if wr == -1:
        continue
    elif wr < 0:
        week_mapping[f'rel_wk_{abs(wr)}_neg'] = wr
    elif wr == 0:
        week_mapping['rel_wk_0'] = wr
    else:
        week_mapping[f'rel_wk_{wr}_pos'] = wr

for col in rel_weeks:
    coef = event_model.params[col]
    se = event_model.bse[col]
    pval = event_model.pvalues[col]
    wr = week_mapping[col]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"Week {wr:+d}: {coef:6.2f} (SE: {se:.2f}, p={pval:.3f}) {sig}")

# Test for pre-trend (joint significance of pre-period coefficients)
pre_period_coeffs = [col for col in rel_weeks if week_mapping[col] < -1]
pre_test = event_model.f_test(pre_period_coeffs)

post_test = event_model.f_test(post_period_coeffs)
print(f"\nJoint test for post-treatment effects (weeks 0 to 7 = 0):")
print(f"F-statistic: {post_test.fvalue:.2f}, p-value: {post_test.pvalue:.3f}")

# Average post-treatment effect

print(f"\nAverage post-treatment effect: {np.mean(post_coefs):.2f}")
