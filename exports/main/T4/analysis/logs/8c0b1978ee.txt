import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load data
df = pd.read_csv('branches.csv')

# Count branches by switch status
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate average service scores by group and period
print("\n\nAverage service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the DiD manually
print("\n\nDifference-in-Differences Calculation:")
# Control group (switched_branch=0)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_diff = control_after - control_before

# Treatment group (switched_branch=1)
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_diff = treat_after - treat_before

# DiD estimate
did_estimate = treat_diff - control_diff

print(f"Control group (no switch): Before={control_before:.2f}, After={control_after:.2f}, Change={control_diff:.2f}")
print(f"Treatment group (switched): Before={treat_before:.2f}, After={treat_after:.2f}, Change={treat_diff:.2f}")
print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")

# Run regression with standard errors
print("\n\n=== Regression Analysis ===")
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print(model.summary())

# Also run with cluster-robust standard errors at branch level
print("\n\n=== Regression with Cluster-Robust SE (by branch) ===")
model_cluster = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_cluster.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create a cleaner dynamic effects plot
post_weeks = [15, 16, 17, 18, 19, 20, 21, 22]
for w in post_weeks:
    df[f'post_week_{w}'] = ((df['week'] == w) & (df['after_switch'] == 1)).astype(int)

dynamic_formula = 'service_score ~ switched_branch + ' + ' + '.join([f'post_week_{w}' for w in post_weeks])
model_dynamic = ols(dynamic_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for post weeks
coefs = []
ses = []
weeks = []
for w in post_weeks:
    coef = model_dynamic.params[f'post_week_{w}']
    se = model_dynamic.bse[f'post_week_{w}']
    coefs.append(coef)
    ses.append(se)
    weeks.append(w)

# Plot dynamic effects
fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(weeks, coefs, yerr=1.96*np.array(ses), fmt='o', capsize=5, 
            label='Effect vs. pre-period baseline', markersize=8, linewidth=2)
ax.axhline(y=0, color='red', linestyle='--', alpha=0.7, label='No effect')
ax.axvline(x=14.5, color='gray', linestyle=':', alpha=0.5)
ax.set_xlabel('Week')
ax.set_ylabel('Service Score Effect (points)')
ax.set_title('Dynamic Treatment Effects: Weekly Impact of New Procedure\n(95% Confidence Intervals)')
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))
plt.tight_layout()
plt.savefig('dynamic_effects.png', dpi=150)
print("Dynamic effects plot saved as 'dynamic_effects.png'")

# Print the dynamic effects table
print("\n=== Dynamic Effects Table ===")
print(f"{'Week':<6} {'Coefficient':<12} {'Std Error':<12} {'95% CI Lower':<14} {'95% CI Upper':<14} {'Significant?':<12}")
print("-" * 80)
for i, w in enumerate(weeks):
    ci_lower = coefs[i] - 1.96 * ses[i]
    ci_upper = coefs[i] + 1.96 * ses[i]
    sig = "Yes" if ci_lower > 0 or ci_upper < 0 else "No"
    print(f"{w:<6} {coefs[i]:<12.2f} {ses[i]:<12.2f} {ci_lower:<14.2f} {ci_upper:<14.2f} {sig:<12}")

# Calculate overall summary statistics for the report
print("\n\n=== Summary Statistics for Report ===")
print(f"Total observations: {len(df)}")
print(f"Number of branches: {df['branch_id'].nunique()}")
print(f"Branches that switched: {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"Branches that didn't switch: {df[df['switched_branch']==0]['branch_id'].nunique()}")
print(f"Weeks of data: {df['week'].min()} to {df['week'].max()}")
print(f"Switch occurred at week: 15")
print(f"\nAverage service score (all): {df['service_score'].mean():.2f}")
print(f"Average service score (switched branches, before): {df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean():.2f}")
print(f"Average service score (switched branches, after): {df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean():.2f}")
print(f"Average service score (non-switched branches, before): {df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean():.2f}")
print(f"Average service score (non-switched branches, after): {df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean():.2f}")

# Main DiD with confidence interval
did_coef = model_dynamic.params['switched_branch:after_switch'] if 'switched_branch:after_switch' in model_dynamic.params else 1.94
# Actually let me recalculate from the simple DiD model
model_simple = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
did_coef = model_simple.params['switched_branch:after_switch']
did_se = model_simple.bse['switched_branch:after_switch']
did_pvalue = model_simple.pvalues['switched_branch:after_switch']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se

print(f"\nMain DiD Estimate: {did_coef:.2f} points")
print(f"Standard Error: {did_se:.2f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {did_pvalue:.3f}")
