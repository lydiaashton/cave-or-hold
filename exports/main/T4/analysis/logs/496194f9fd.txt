import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators
# Week 14 is the last pre-switch week, so we'll use that as reference
df['relative_week'] = df['week'] - 14  # Week 14 becomes 0, week 15 becomes 1, etc.

# Create interaction terms for each week with treatment
# We'll estimate separate effects for each week relative to the switch
event_study_results = []

for week in range(6, 23):
    week_data = df[df['week'] == week].copy()
    week_data['treatment_effect'] = week_data['switched_branch']
    
    # Simple comparison for this week
    if len(week_data) > 0:
        treatment_avg = week_data[week_data['switched_branch']==1]['service_score'].mean()
        control_avg = week_data[week_data['switched_branch']==0]['service_score'].mean()
        diff = treatment_avg - control_avg
        event_study_results.append({
            'week': week,
            'relative_week': week - 14,
            'treatment_avg': treatment_avg,
            'control_avg': control_avg,
            'difference': diff
        })

event_df = pd.DataFrame(event_study_results)
print("Event Study - Weekly Differences (Treatment - Control):")
print(event_df)

# Run a more formal event study regression
# Create dummies for each week interacted with treatment
for week in range(6, 23):
    df[f'week_{week}'] = (df['week'] == week).astype(int)
    df[f'week_{week}_treated'] = df[f'week_{week}'] * df['switched_branch']

# Event study regression (omitting week 14 as reference)
treatment_week_cols = [f'week_{w}_treated' for w in range(6, 14)] + [f'week_{w}_treated' for w in range(15, 23)]
control_cols = [f'week_{w}' for w in range(6, 23)]

# Build formula
formula = 'service_score ~ switched_branch + ' + ' + '.join(control_cols) + ' + ' + ' + '.join(treatment_week_cols)

# This might be too many variables, let's try a simpler approach
# Just run week-by-week DiD estimates with confidence intervals

print("\n=== Week-by-week DiD estimates ===")
week_effects = []
for week in range(6, 23):
    week_data = df[df['week'] == week].copy()
    if len(week_data) >= 4:  # Need enough data
        # Simple regression for this week
        model = smf.ols('service_score ~ switched_branch', data=week_data).fit(cov_type='cluster', cov_kwds={'groups': week_data['branch_id']})
        coef = model.params['switched_branch']
        se = model.bse['switched_branch']
        ci_lower = coef - 1.96 * se
        ci_upper = coef + 1.96 * se
        week_effects.append({
            'week': week,
            'relative_week': week - 14,
            'effect': coef,
            'se': se,
            'ci_lower': ci_lower,
            'ci_upper': ci_upper
        })

effects_df = pd.DataFrame(week_effects)
print(effects_df)

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))

# Plot point estimates
ax.errorbar(effects_df['relative_week'], effects_df['effect'],
