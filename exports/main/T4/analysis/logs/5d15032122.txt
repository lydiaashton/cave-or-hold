import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD with branch fixed effects and cluster-robust SEs
model_fe = ols('service_score ~ after_switch + treatment + staffing_index + visits_index + C(branch_id)', data=df).fit()

# Cluster-robust standard errors
cov_clusters = model_fe.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print("=== Model with Branch FE and Cluster-Robust SEs ===")
print(cov_clusters.summary())
print(f"\nTreatment effect: {cov_clusters.params['treatment']:.3f}")
print(f"Cluster-robust SE: {cov_clusters.bse['treatment']:.3f}")
print(f"95% CI: [{cov_clusters.conf_int().loc['treatment', 0]:.3f}, {cov_clusters.conf_int().loc['treatment', 1]:.3f}]")
print(f"P-value: {cov_clusters.pvalues['treatment']:.3f}")

# Event study: estimate effect for each week relative to switch
# Create week dummies interacted with treatment
df['weeks_relative'] = df['week'] - 14  # Week 14 is the last pre-switch week
df['weeks_relative'] = df['weeks_relative'].astype(int)

# Create dummies for each week relative to switch, interacted with switched_branch
event_study_vars = []
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != -1:  # omit week -1 (week 13) as reference
        var_name = f'rel_week_{w}'
        df[var_name] = ((df['weeks_relative'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(var_name)

# Run event study regression with branch and week fixed effects
event_formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(branch_id) + C(week)'
model_event = ols(event_formula, data=df).fit()
cov_event = model_event.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("\n=== Event Study Results ===")
event_results = []
for w in range(-8, 9):
    if w != -1:
        var_name = f'rel_week_{w}'
        coef = cov_event.params[var_name]
        se = cov_event.bse[var_name]
        event_results.append({'week_relative': w, 'coef': coef, 'se': se})

event_df = pd.DataFrame(event_results)
print(event_df)

# Plot event study
fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(event_df['week_relative'], event_df['coef'], yerr=1.96*event_df['se'], 
            fmt='o', capsize=5, label='Treatment Effect')
ax.axhline(y=0, color='red', linestyle='--', label='No Effect')
ax.axvline(x=-0.5, color='gray', linestyle=':', label='Switch Point')
