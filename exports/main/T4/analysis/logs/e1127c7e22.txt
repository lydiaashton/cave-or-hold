import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the 2x2 table for DiD
print("=== Difference-in-Differences 2x2 Table ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
print(pivot)
print()

# Calculate the DiD estimate manually
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
before_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print("Control (not switched): Before =", round(before_control, 3), "After =", round(after_control, 3))
print("Treatment (switched): Before =", round(before_treatment, 3), "After =", round(after_treatment, 3))
print()

# Changes over time
control_change = after_control - before_control
treatment_change = after_treatment - before_treatment
print("Control change:", round(control_change, 3))
print("Treatment change:", round(treatment_change, 3))
print()

# DiD estimate
did_estimate = treatment_change - control_change
print("DiD Estimate (Treatment effect):", round(did_estimate, 3))
print()

# Now run regression with standard errors
print("=== Regression Analysis ===")

# Basic DiD regression
model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 1: Basic DiD (clustered SE by branch)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 2: DiD with controls (clustered SE by branch)")
print(model2.summary().tables[1])

# DiD with branch fixed effects
print("\n=== Model with Branch Fixed Effects ===")
model3 = ols('service_score ~ switched_branch:after_switch + C(branch_id) + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model3.summary().tables[1])

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study - estimate effect for each week relative to switch
print("=== Event Study Analysis ===")
# Create relative week variable
df['relative_week'] = df['week'] - 15

# Create dummies for each week interacted with treatment
for w in range(6, 23):
    df[f'week_{w}'] = (df['week'] == w).astype(int)
    df[f'week_{w}_treated'] = df[f'week_{w}'] * df['switched_branch']

# Run regression with all week-treatment interactions (week 14 as reference)
week_cols = [f'week_{w}_treated' for w in range(6, 23) if w != 14]
formula = 'service_score ~ ' + ' + '.join(week_cols) + ' + C(branch_id) + C(week)'
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Coefficients (relative to week 14):")
print("(Negative weeks are pre-treatment, week 15+ are post-treatment)")
print()
for w in range(6, 23):
    if w == 14:
        print(f"Week {w}: (reference period)")
        continue
    coef = model_event.params[f'week_{w}_treated']
    se = model_event.bse[f'week_{w}_treated']
    ci_low = coef - 1.96 * se
    ci_high = coef + 1.96 * se
    pval = model_event.pvalues[f'week_{w}_treated']
    sig = "***" if pval < 0.01 else ("**" if pval < 0.05 else ("*" if pval < 0.1 else ""))
    period = "PRE" if w < 15 else "POST"
    print(f"Week {w} ({period}): {coef:.2f} (SE: {se:.2f}) 95% CI: [{ci_low:.2f}, {ci_high:.2f}] {sig}")

print()
print("=== Summary Statistics ===")
print(f"Total observations: {len(df):,}")
print(f"Total branches: {df['branch_id'].nunique()}")
print(f"Switched branches: {df['switched_branch'].sum()} ({df['switched_branch'].mean()*100:.1f}%)")
print(f"Control branches: {len(df[df['switched_branch']==0]['branch_id'].unique())}")
print()
print(f"Average service score (all): {df['service_score'].mean():.2f}")
print(f"Average service score (treatment, before): {df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean():.2f}")
print(f"Average service score (treatment, after): {df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean():.2f}")
print(f"Average service score (control, before): {df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean():.2f}")
print(f"Average service score (control, after): {df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean():.2f}")
