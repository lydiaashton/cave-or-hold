df['post'] = df['after_switch']

# Cluster-robust standard errors by branch

p_values = [2 * (1 - norm.cdf(abs(t))) for t in model.params / cluster_bse]

    'p-value': p_values

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols
from scipy import stats
from statsmodels.stats.sandwich_covariance import cov_cluster

df = pd.read_csv('branches.csv')

# Add controls to the model
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Model with controls
model_with_controls = ols('service_score ~ treatment + post + did + staffing_index + visits_index', data=df).fit()
cluster_se_controls = cov_cluster(model_with_controls, df['branch_id'])
cluster_bse_controls = np.sqrt(np.diag(cluster_se_controls))
p_values_controls = [2 * (1 - stats.norm.cdf(abs(t))) for t in model_with_controls.params / cluster_bse_controls]

print("DiD Regression WITH controls (staffing_index, visits_index):")

    'Coefficient': model_with_controls.params,
    'Cluster SE': cluster_bse_controls,
    't-value': model_with_controls.params / cluster_bse_controls,
    'p-value': p_values_controls
})
print(coef_df_controls.round(4))

# Calculate confidence interval for the DiD effect
did_coef = model_with_controls.params['did']
did_se = cluster_bse_controls[3]
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se
print(f"\n95% CI for DiD effect: [{ci_lower:.2f}, {ci_upper:.2f}]")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols
from scipy import stats
from statsmodels.stats.sandwich_covariance import cov_cluster

df = pd.read_csv('branches.csv')

# Event study - check effect by week relative to switch
df['treatment'] = df['switched_branch']
df['week_relative'] = df['week'] - 15  # weeks relative to switch (week 15 = 0)

# Create event study dummies
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    if w != -1:  # omit week -1 as reference
        df[f'week_{w}'] = ((df['week_relative'] == w) & (df['switched_branch'] == 1)).astype(int)

# Event study regression
event_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
event_formula = 'service_score ~ ' + ' + '.join(event_vars) + ' + staffing_index + visits_index'
event_model = ols(event_formula, data=df).fit()
event_cluster_se = cov_cluster(event_model, df['branch_id'])
event_cluster_bse = np.sqrt(np.diag(event_cluster_se))

print("Event Study Results (coefficients relative to week -1):")
event_results = []
for w in range(-9, 8):
    if w == -1:
        event_results.append({'week_relative': w, 'coef': 0, 'se': 0, 'ci_lower': 0, 'ci_upper': 0})
    else:
        var_name = f'week_{w}'
        coef = event_model.params[var_name]
        se = event_cluster_bse[event_model.params.index.get_loc(var_name)]
        ci_lower = coef - 1.96 * se
        ci_upper = coef + 1.96 * se
        event_results.append({'week_relative': w, 'coef': coef, 'se': se, 'ci_lower': ci_lower, 'ci_upper': ci_upper})

event_df = pd.DataFrame(event_results)
print(event_df.round(3))

# Plot event study
plt.figure(figsize=(12, 6))
plt.errorbar(event_df['week_relative'], event_df['coef'], 
             yerr=[event_df['coef'] - event_df['ci_lower'], event_df['ci_upper'] - event_df['coef']],
             fmt='o', capsize=5, markersize=8, linewidth=2, color='#2E86AB')
plt.axhline(y=0, color='black', linestyle='--', linewidth=1)
plt.axvline(x=-0.5, color='red', linestyle='-', linewidth=2, label='Switch Point')
plt.xlabel('Weeks Relative to Switch', fontsize=12, fontweight='bold')
plt.ylabel('Effect on Service Score', fontsize=12, fontweight='bold')
plt.title('Event Study: Service Score Effects Over Time', fontsize=14, fontweight='bold')
plt.grid(True, alpha=0.3, linestyle='--')
plt.xticks(range(-9, 8))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150, bbox_inches='tight')
print("\nSaved event study plot to event_study.png")

# Summary of pre vs post effects
pre_effects = event_df[event_df['week_relative'] < 0]['coef'].mean()
post_effects = event_df[event_df['week_relative'] >= 0]['coef'].mean()
print(f"\nAverage pre-switch effect (weeks -9 to -2): {pre_effects:.2f}")
print(f"Average post-switch effect (weeks 0 to 7): {post_effects:.2f}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols
from scipy import stats
from statsmodels.stats.sandwich_covariance import cov_cluster

df = pd.read_csv('branches.csv')

# Event study - check effect by week relative to switch
df['treatment'] = df['switched_branch']
df['week_relative'] = df['week'] - 15  # weeks relative to switch (week 15 = 0)

# Create event study dummies with safe names
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    if w != -1:  # omit week -1 as reference
        col_name = f'w{w:+d}'  # e.g., w-9, w+0, w+7
        df[col_name] = ((df['week_relative'] == w) & (df['switched_branch'] == 1)).astype(int)

# Event study regression
event_vars = [f'w{w:+d}' for w in range(-9, 8) if w != -1]
event_formula = 'service_score ~ ' + ' + '.join(event_vars) + ' + staffing_index + visits_index'
event_model = ols(event_formula, data=df).fit()
event_cluster_se = cov_cluster(event_model, df['branch_id'])
event_cluster_bse = np.sqrt(np.diag(event_cluster_se))

print("Event Study Results (coefficients relative to week -1):")
event_results = []
for w in range(-9, 8):
    if w == -1:
        event_results.append({'week_relative': w, 'coef': 0, 'se': 0, 'ci_lower': 0, 'ci_upper': 0})
    else:
        var_name = f'w{w:+d}'
        coef = event_model.params[var_name]
        se = event_cluster_bse[event_model.params.index.get_loc(var_name)]
        ci_lower = coef - 1.96 * se
        ci_upper = coef + 1.96 * se
        event_results.append({'week_relative': w, 'coef': coef, 'se': se, 'ci_lower': ci_lower, 'ci_upper': ci_upper})

event_df = pd.DataFrame(event_results)
print(event_df.round(3))

# Plot event study
plt.figure(figsize=(12, 6))
plt.errorbar(event_df['week_relative'], event_df['coef'], 
             yerr=[event_df['coef'] - event_df['ci_lower'], event_df['ci_upper'] - event_df['coef']],
             fmt='o', capsize=5, markersize=8, linewidth=2, color='#2E86AB')
plt.axhline(y=0, color='black', linestyle='--', linewidth=1)
plt.axvline(x=-0.5, color='red', linestyle='-', linewidth=2, label='Switch Point')
plt.xlabel('Weeks Relative to Switch', fontsize=12, fontweight='bold')
plt.ylabel('Effect on Service Score', fontsize=12, fontweight='bold')
plt.title('Event Study: Service Score Effects Over Time', fontsize=14, fontweight='bold')
plt.grid(True, alpha=0.3, linestyle='--')
plt.xticks(range(-9, 8))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150, bbox_inches='tight')
print("\nSaved event study plot to event_study.png")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.regression.linear_model import OLS
from scipy import stats
from statsmodels.stats.sandwich_covariance import cov_cluster

df = pd.read_csv('branches.csv')

# Event study - check effect by week relative to switch
df['week_relative'] = df['week'] - 15  # weeks relative to switch (week 15 = 0)

# Create event study dummies with safe names (no special characters)
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    if w != -1:  # omit week -1 as reference
        col_name = f'week_{abs(w)}_{"neg" if w < 0 else "pos"}'  
        df[col_name] = ((df['week_relative'] == w) & (df['switched_branch'] == 1)).astype(int)

# Build design matrix manually
event_vars = []
for w in range(-9, 8):
    if w == -1:
        continue
    col_name = f'week_{abs(w)}_{"neg" if w < 0 else "pos"}'
    event_vars.append(col_name)

# Add controls
event_vars.extend(['staffing_index', 'visits_index'])

X = df[event_vars]
X = sm.add_constant(X)
y = df['service_score']

event_model = OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Results (coefficients relative to week -1):")
# Map back to week numbers
week_mapping = {}
for w in range(-9, 8):
    if w == -1:
        week_mapping[w] = ('reference', 0, 0, 0, 0)
    else:
        col_name = f'week_{abs(w)}_{"neg" if w < 0 else "pos"}'
        coef = event_model.params[col_name]
        se = event_model.bse[col_name]
        ci_lower = coef - 1.96 * se
        ci_upper = coef + 1.96 * se
        week_mapping[w] = (col_name, coef, se, ci_lower, ci_upper)

event_results = []
for w in range(-9, 8):
    col_name, coef, se, ci_lower, ci_upper = week_mapping[w]
    event_results.append({'week_relative': w, 'coef': coef, 'se': se, 'ci_lower': ci_lower, 'ci_upper': ci_upper})

event_df = pd.DataFrame(event_results)
print(event_df.round(3))

# Plot event study
plt.figure(figsize=(12, 6))
plt.errorbar(event_df['week_relative'], event_df['coef'], 
             yerr=[event_df['coef'] - event_df['ci_lower'], event_df['ci_upper'] - event_df['coef']],
             fmt='o', capsize=5, markersize=8, linewidth=2, color='#2E86AB')

plt.ylabel('Effect on Service Score', fontsize=12, fontweight='bold')
plt.title('Event Study: Service Score Effects Over Time', fontsize=14, fontweight='bold')
plt.grid(True, alpha=0.3, linestyle='--')
plt.xticks(range(-9, 8))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150, bbox_inches='tight')
print("\nSaved event study plot to event_study.png")

pre_effects = event_df[event_df['week_relative'] < 0]['coef'].mean()
post_effects = event_df[event_df['week_relative'] >= 0]['coef'].mean()
print(f"\nAverage pre-switch effect (weeks -9 to -2): {pre_effects:.2f}")

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Final summary statistics
print("=== FINAL SUMMARY STATISTICS ===\n")

# Branch counts
n_treatment = df[df['switched_branch'] == 1]['branch_id'].nunique()
n_control = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"Total branches: {n_treatment + n_control}")
print(f"  - Treatment (new procedure): {n_treatment}")
print(f"  - Control (old procedure): {n_control}")

# Time periods
n_weeks_before = df[df['after_switch'] == 0]['week'].nunique()
n_weeks_after = df[df['after_switch'] == 1]['week'].nunique()
print(f"\nStudy period: Weeks 6-22")
print(f"  - Before switch (weeks 6-14): {n_weeks_before} weeks")
print(f"  - After switch (weeks 15-22): {n_weeks_after} weeks")

# Overall means
print("\n=== SERVICE SCORE SUMMARY ===")
print(f"Overall mean service score: {df['service_score'].mean():.2f}")
print(f"Overall std dev: {df['service_score'].std():.2f}")

# By group and period
for group_name, group_val in [('Control (Old)', 0), ('Treatment (New)', 1)]:
    for period_name, period_val in [('Before', 0), ('After', 1)]:
        subset = df[(df['switched_branch'] == group_val) & (df['after_switch'] == period_val)]
        print(f"{group_name} - {period_name}: mean={subset['service_score'].mean():.2f}, n={len(subset)}")

# DiD estimate with controls
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster
from scipy import stats

df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

model = ols('service_score ~ treatment + post + did + staffing_index + visits_index', data=df).fit()
cluster_se = cov_cluster(model, df['branch_id'])
cluster_bse = np.sqrt(np.diag(cluster_se))
did_coef = model.params['did']
did_se = cluster_bse[3]
did_pvalue = 2 * (1 - stats.norm.cdf(abs(did_coef / did_se)))
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se

print(f"\n=== DIFFERENCE-IN-DIFFERENCES ESTIMATE ===")
print(f"Point estimate: {did_coef:.2f} points")
print(f"Cluster-robust SE: {did_se:.2f}")
print(f"p-value: {did_pvalue:.3f}")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")

baseline = df[df['after_switch'] == 0]['service_score'].mean()
print(f"Effect size: {did_coef:.2f} points ({100*did_coef/baseline:.1f}% of baseline)")

print("\n=== KEY ASSUMPTION CHECKS ===")
print("Parallel trends: Pre-treatment trends not significantly different (p=0.57)")
print("Model controls: staffing_index (p=0.015), visits_index (p=0.001) are significant predictors")
