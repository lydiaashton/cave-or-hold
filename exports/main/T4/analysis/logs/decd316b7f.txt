import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create treatment interaction term (DiD estimator)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary table: average service scores by group and period
summary_table = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).unstack(level=0)
print("=== Service Scores by Group and Period ===")
print(summary_table)
print()

# Calculate the DiD manually
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
pre_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
post_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("=== Difference-in-Differences Calculation ===")
print(f"Control group (non-switched): Pre={pre_control:.2f}, Post={post_control:.2f}, Change={post_control - pre_control:.2f}")
print(f"Treatment group (switched): Pre={pre_treatment:.2f}, Post={post_treatment:.2f}, Change={post_treatment - pre_treatment:.2f}")
did_estimate = (post_treatment - pre_treatment) - (post_control - pre_control)
print(f"DiD Estimate (Treatment Effect): {did_estimate:.2f}")
print()

# Run regression with branch fixed effects
print("=== Regression Analysis ===")

# Model 1: Basic DiD with branch fixed effects
model1 = ols('service_score ~ treatment + after_switch + switched_branch + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 1: Basic DiD with Branch Fixed Effects")
print(model1.summary().tables[1].iloc[:5])

# Model 2: DiD with controls (staffing and visits)
model2 = ols('service_score ~ treatment + after_switch + switched_branch + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 2: DiD with Branch FE and Controls")
print(model2.summary().tables[1].iloc[:6])

# Get the treatment effect coefficient
treatment_effect = model2.params['treatment']
treatment_se = model2.bse['treatment']
treatment_pvalue = model2.pvalues['treatment']
print(f"\nTreatment Effect: {treatment_effect:.3f} (SE: {treatment_se:.3f}, p-value: {treatment_pvalue:.3f})")

print(model1.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate average service score by week and group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_score']

# Pivot for easier plotting
pivot_data = weekly_avg.pivot(index='week', columns='switched_branch', values='avg_score')
pivot_data.columns = ['Control (No Switch)', 'Treatment (Switched)']

print("=== Weekly Average Service Scores ===")
print(pivot_data)
print()

fig, ax = plt.subplots(figsize=(10, 6))

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Scores by Week: Treatment vs Control Groups', fontsize=14)
ax.legend(fontsize=11)
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("Plot saved as 'service_scores_trend.png'")

# Calculate week-by-week differences
pivot_data['Difference'] = pivot_data['Treatment (Switched)'] - pivot_data['Control (No Switch)']
print("\n=== Week-by-Week Differences (Treatment - Control) ===")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate average service score by week and group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_score']

# Pivot for easier plotting
pivot_data = weekly_avg.pivot(index='week', columns='switched_branch', values='avg_score')
pivot_data.columns = ['Control', 'Treatment']

# Calculate the difference
pivot_data['Diff'] = pivot_data['Treatment'] - pivot_data['Control']

# Create event study plot
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 5))

# Plot 1: Raw scores over time
ax1.plot(pivot_data.index, pivot_data['Control'], 'o-', label='Control (No Switch)', linewidth=2, markersize=8, color='blue')
ax1.plot(pivot_data.index, pivot_data['Treatment'], 's-', label='Treatment (Switched)', linewidth=2, markersize=8, color='green')
ax1.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')
ax1.set_xlabel('Week', fontsize=12)
ax1.set_ylabel('Average Service Score', fontsize=12)
ax1.set_title('Figure 1: Service Scores Over Time', fontsize=14)
ax1.legend(fontsize=11)
ax1.grid(True, alpha=0.3)

# Plot 2: Difference over time (event study style)
weeks = pivot_data.index.tolist()
diffs = pivot_data['Diff'].tolist()
ax2.bar(weeks, diffs, color=['gray' if w < 15 else 'steelblue' for w in weeks], edgecolor='black')
ax2.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax2.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point')
ax2.set_xlabel('Week', fontsize=12)
ax2.set_ylabel('Treatment - Control Difference', fontsize=12)
ax2.set_title('Figure 2: Gap Between Groups Over Time', fontsize=14)
ax2.legend(fontsize=11)
ax2.grid(True, alpha=0.3, axis='y')
ax2.set_xticks(weeks)

plt.tight_layout()
plt.savefig('event_study_plot.png', dpi=150)
print("Event study plot saved as 'event_study_plot.png'")

control_pre_model = ols('service_score ~ week_centered', data=control_pre).fit()
print(f"Control group pre-switch weekly trend: {control_pre_model.params['week_centered']:.3f} points/week")

# Post-period trend (weeks 15-22)
post_data = df[df['week'] >= 15].copy()

# Treatment group post-trend
