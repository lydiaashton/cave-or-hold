import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch']  # 1 if switched branch
df['post'] = df['after_switch']  # 1 if after week 15
df['DiD'] = df['treated'] * df['post']  # Interaction term

# Summary table: Average service scores by group and period
print("=" * 60)
print("AVERAGE SERVICE SCORES BY GROUP AND PERIOD")
print("=" * 60)

# Before switch (after_switch = 0)
before = df[df['after_switch'] == 0]
before_switched = before[before['switched_branch'] == 1]['service_score'].mean()
before_control = before[before['switched_branch'] == 0]['service_score'].mean()

# After switch (after_switch = 1)
after = df[df['after_switch'] == 1]
after_switched = after[after['switched_branch'] == 1]['service_score'].mean()
after_control = after[after['switched_branch'] == 0]['service_score'].mean()

print(f"\nBefore Switch (Weeks 6-14):")
print(f"  Switched branches (treatment): {before_switched:.2f}")
print(f"  Non-switched branches (control): {before_control:.2f}")
print(f"  Difference: {before_switched - before_control:.2f}")

print(f"\nAfter Switch (Weeks 15-22):")
print(f"  Switched branches (treatment): {after_switched:.2f}")
print(f"  Non-switched branches (control): {after_control:.2f}")
print(f"  Difference: {after_switched - after_control:.2f}")

# Difference-in-Differences
did_effect = (after_switched - before_switched) - (after_control - before_control)
print(f"\nDifference-in-Differences Estimate: {did_effect:.2f}")
print(f"  (Change in treatment) - (Change in control)")
print(f"  = ({after_switched:.2f} - {before_switched:.2f}) - ({after_control:.2f} - {before_control:.2f})")
print(f"  = {after_switched - before_switched:.2f} - {after_control - before_control:.2f}")

# Now run regression with standard errors
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD model
model1 = ols('service_score ~ treated + post + DiD', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 1: Basic DiD (clustered SE by branch)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ treated + post + DiD + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 2: DiD with controls (clustered SE by branch)")
print(model2.summary().tables[1])

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Check if effect varies by time since switch
df['weeks_since_switch'] = df['week'] - 14  # 0 = week 14 (last before), 1 = week 15 (first after)
df['post_week'] = df['week'].apply(lambda x: f'W{x}' if x >= 15 else None)

# Event study style: interact treated with each post week
print("=" * 60)
print("EFFECT BY WEEKS AFTER SWITCH")
print("=" * 60)

# Create dummies for each post-switch week
post_weeks = [15, 16, 17, 18, 19, 20, 21, 22]
for w in post_weeks:
    df[f'post_w{w}'] = (df['week'] == w).astype(int)
    df[f'DiD_w{w}'] = df['treated'] * df[f'post_w{w}']

# Run regression with week-specific effects
week_vars = ' + '.join([f'DiD_w{w}' for w in post_weeks])
formula = f'service_score ~ treated + {" + ".join([f"post_w{w}" for w in post_weeks])} + {week_vars}'
model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nWeek-specific DiD effects (relative to pre-switch baseline):")
for w in post_weeks:
    coef = model.params[f'DiD_w{w}']
    se = model.bse[f'DiD_w{w}']
    pval = model.pvalues[f'DiD_w{w}']
    print(f"  Week {w}: {coef:.2f} (SE: {se:.2f}, p={pval:.3f})")

# Also check: how many branches in each group?
print("\n" + "=" * 60)
print("BRANCH COUNTS")
print("=" * 60)
switched_branches = df[df['switched_branch']==1]['branch_id'].nunique()
control_branches = df[df['switched_branch']==0]['branch_id'].nunique()
print(f"Switched branches (treatment): {switched_branches}")
print(f"Non-switched branches (control): {control_branches}")
print(f"Total branches: {switched_branches + control_branches}")

# Confidence interval for the main effect
main_model = ols('service_score ~ treated + post + DiD', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
did_coef = main_model.params['DiD']
did_se = main_model.bse['DiD']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se
print(f"\n" + "=" * 60)
print("FINAL ESTIMATE SUMMARY")
print("=" * 60)
print(f"DiD coefficient: {did_coef:.2f}")
print(f"Standard error: {did_se:.2f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {main_model.pvalues['DiD']:.4f}")
