import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treated'] * df['post']

# Basic DiD table: mean service scores by group and period
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES: SERVICE SCORE SUMMARY")
print("=" * 60)

# Calculate means for each group-period combination
summary = df.groupby(['treated', 'post'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("\nMean Service Scores by Group and Period:")
print(summary)

# Calculate the DiD manually
control_before = df[(df['treated'] == 0) & (df['post'] == 0)]['service_score'].mean()
control_after = df[(df['treated'] == 0) & (df['post'] == 1)]['service_score'].mean()
treated_before = df[(df['treated'] == 1) & (df['post'] == 0)]['service_score'].mean()
treated_after = df[(df['treated'] == 1) & (df['post'] == 1)]['service_score'].mean()

control_change = control_after - control_before
treated_change = treated_after - treated_before
did_effect = treated_change - control_change

print("\n" + "=" * 60)
print("DId CALCULATION:")
print("=" * 60)
print(f"Control group (no switch): Before={control_before:.2f}, After={control_after:.2f}, Change={control_change:.2f}")
print(f"Treated group (switched):  Before={treated_before:.2f}, After={treated_after:.2f}, Change={treated_change:.2f}")
print(f"\nDifference-in-Differences estimate: {did_effect:.2f}")

# Run regression DiD
print("\n" + "=" * 60)
print("REGRESSION RESULTS")
print("=" * 60)

# Basic DiD regression
model1 = ols('service_score ~ treated + post + did', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ treated + post + did + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Cluster-robust standard errors by branch
model3 = ols('service_score ~ treated + post + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 3: DiD with controls (cluster-robust SE by branch)")
print(model3.summary().tables[1])

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check parallel trends: average service score by week for treated vs control (before period only)
print("=" * 60)
print("PARALLEL TRENDS CHECK (Pre-treatment period: weeks 6-14)")
print("=" * 60)

pre_period = df[df['week'] <= 14]
weekly_means = pre_period.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print("\nWeekly means before the switch:")
print(weekly_means)

# Calculate week-by-week differences
weekly_means['diff'] = weekly_means[1] - weekly_means[0]
print("\nDifference (Treated - Control) by week:")
print(weekly_means['diff'])

# Test for pre-trend: regress service_score on week*treated interaction for pre-period only
pre_period = pre_period.copy()
pre_period['week_centered'] = pre_period['week'] - pre_period['week'].mean()
pre_period['interaction'] = pre_period['week_centered'] * pre_period['switched_branch']

import statsmodels.api as sm
from statsmodels.formula.api import ols

pre_trend_model = ols('service_score ~ switched_branch + week_centered + interaction', data=pre_period).fit(cov_type='cluster', cov_kwds={'groups': pre_period['branch_id']})
print("\nPre-trend test (interaction term should be insignificant):")
print(pre_trend_model.summary().tables[1])

# Event study: effect by week relative to switch
print("\n" + "=" * 60)
print("EVENT STUDY: Weekly effects")
print("=" * 60)

df['relative_week'] = df['week'] - 14  # week 14 is the last pre-switch week
df['post_week'] = (df['week'] >= 15).astype(int)

# Create dummies for each post-switch week interacted with treatment
for w in range(15, 23):
    df[f'post_w{w}'] = ((df['week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
event_vars = ' + '.join([f'post_w{w}' for w in range(15, 23)])
event_formula = f'service_score ~ switched_branch + C(week) + {event_vars}'
event_model = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nEvent study coefficients (post-switch weeks):")
for w in range(15, 23):
    coef = event_model.params[f'post_w{w}']
    se = event_model.bse[f'post_w{w}']
    print(f"Week {w}: {coef:.2f} (SE: {se:.2f})")

# Visualize the trends
plt.figure(figsize=(12, 6))

# Get weekly averages for both groups
weekly_data = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

plt.plot(weekly_data.index, weekly_data[0], 'o-', label='Control (No Switch)', linewidth=2, markersize=8)
plt.plot(weekly_data.index, weekly_data[1], 's-', label='Treated (Switched)', linewidth=2, markersize=8)

plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch implemented')
plt.xlabel('Week', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Scores Over Time: Treated vs Control Branches', fontsize=14)
plt.legend(fontsize=11)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("\nPlot saved as 'service_scores_trend.png'")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Additional robustness checks
print("=" * 60)
print("ADDITIONAL ANALYSIS")
print("=" * 60)

# Check if effect varies by branch characteristics (if any)
# Look at distribution of effects
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treated'] * df['post']

# Branch-level analysis: calculate before/after for each branch
branch_summary = df.groupby(['branch_id', 'switched_branch']).agg({
    'service_score': ['mean'],
    'week': ['min', 'max']
}).reset_index()

# Calculate branch-level before and after means
branch_before = df[df['after_switch'] == 0].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_before.columns = ['branch_id', 'switched_branch', 'score_before']

branch_after = df[df['after_switch'] == 1].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_after.columns = ['branch_id', 'switched_branch', 'score_after']

branch_level = branch_before.merge(branch_after, on=['branch_id', 'switched_branch'])
branch_level['change'] = branch_level['score_after'] - branch_level['score_before']

print("\nBranch-level changes:")
print(branch_level['change'].describe())

print("\nBy group:")
print(branch_level.groupby('switched_branch')['change'].describe())

# Calculate effect size (Cohen's d)
pooled_std = np.sqrt(((df[df['switched_branch']==1]['service_score'].std()**2) * (len(df[df['switched_branch']==1])-1) + 
                      (df[df['switched_branch']==0]['service_score'].std()**2) * (len(df[df['switched_branch']==0])-1)) / 
                     (len(df) - 2))

did_effect = 1.876  # from earlier regression
effect_size = did_effect / pooled_std
print(f"\nPooled standard deviation: {pooled_std:.2f}")
print(f"Effect size (Cohen's d): {effect_size:.2f}")

# Confidence interval for the DiD effect
model = ols('service_score ~ treated + post + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
did_coef = model.params['did']
did_se = model.bse['did']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se
print(f"\n95% Confidence Interval for DiD effect: [{ci_lower:.2f}, {ci_upper:.2f}]")

# Check if staffing or visits changed differentially
print("\n" + "=" * 60)
print("DID ON OTHER OUTCOMES (to check for spillovers)")
print("=" * 60)

model_staffing = ols('staffing_index ~ treated + post + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nEffect on staffing_index:")
print(f"  DiD coefficient: {model_staffing.params['did']:.3f} (p={model_staffing.pvalues['did']:.3f})")

model_visits = ols('visits_index ~ treated + post + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nEffect on visits_index:")
print(f"  DiD coefficient: {model_visits.params['did']:.3f} (p={model_visits.pvalues['did']:.3f})")
