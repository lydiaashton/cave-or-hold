import pandas as pd

from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Count branches by switch status
switch_counts = df.groupby('branch_id')['switched_branch'].first().value_counts()
print("Branches by switch status:")
print(switch_counts)
print(f"\nTotal branches: {len(df.groupby('branch_id'))}")
print(f"Switched: {switch_counts.get(1, 0)} ({100*switch_counts.get(1, 0)/len(df.groupby('branch_id')):.1f}%)")
print(f"Not switched: {switch_counts.get(0, 0)} ({100*switch_counts.get(0, 0)/len(df.groupby('branch_id')):.1f}%)")

# Average service scores by group and period
print("\n\nAverage service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the DiD manually
before_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nSwitched branches: Before={before_switched:.2f}, After={after_switched:.2f}, Change={after_switched - before_switched:.2f}")
print(f"Control branches: Before={before_control:.2f}, After={after_control:.2f}, Change={after_control - before_control:.2f}")
did_effect = (after_switched - before_switched) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {did_effect:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Basic DiD Regression ===")
print(model1.summary().tables[1])

# DiD with branch fixed effects
model2 = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit()
print("\n=== DiD with Branch Fixed Effects ===")
print(model2.summary().tables[1])

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create event time relative to switch (week 15)
# For switched branches: event_time = week - 15
# For control branches: we'll use the same scale but they never "switch"
df['event_time'] = df['week'] - 15

# For event study, we need to interact treatment status with each week
# Create week dummies interacted with treatment
weeks = sorted(df['week'].unique())
for w in weeks:
    df[f'week_{w}'] = (df['week'] == w).astype(int)
    df[f'treat_week_{w}'] = df['switched_branch'] * (df['week'] == w).astype(int)

# Event study regression (omit week 14 as reference)
event_vars = '+'.join([f'treat_week_{w}' for w in weeks if w != 14])
formula = f'service_score ~ switched_branch + C(week) + {event_vars}'
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract treatment effects by week
treat_coefs = {}
treat_se = {}
for w in weeks:
    if w != 14:
        coef_name = f'treat_week_{w}'
        treat_coefs[w] = event_model.params[coef_name]
        treat_se[w] = event_model.bse[coef_name]
    else:
        treat_coefs[w] = 0  # reference period
        treat_se[w] = 0

print("Event study coefficients (treatment effect by week):")
for w in sorted(treat_coefs.keys()):
    ci_low = treat_coefs[w] - 1.96 * treat_se[w]
    ci_high = treat_coefs[w] + 1.96 * treat_se[w]
    sig = "***" if treat_se[w] > 0 and abs(treat_coefs[w]/treat_se[w]) > 2.58 else "**" if treat_se[w] > 0 and abs(treat_coefs[w]/treat_se[w]) > 1.96 else "*" if treat_se[w] > 0 and abs(treat_coefs[w]/treat_se[w]) > 1.64 else ""
    print(f"Week {w}: {treat_coefs[w]:.2f} (SE: {treat_se[w]:.2f}) 95% CI: [{ci_low:.2f}, {ci_high:.2f}] {sig}")

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
ax.errorbar(list(treat_coefs.keys()), list(treat_coefs.values()), 
            yerr=[1.96 * treat_se[w] for w in treat_coefs.keys()],

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Let's do a cleaner analysis
# 1. Main DiD result
print("=== MAIN ANALYSIS ===\n")

model_main = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Main DiD Model (clustered by branch):")
print(f"  Treatment effect: {model_main.params['treatment']:.2f}")
print(f"  Standard error: {model_main.bse['treatment']:.2f}")
print(f"  95% CI: [{model_main.params['treatment'] - 1.96*model_main.bse['treatment']:.2f}, {model_main.params['treatment'] + 1.96*model_main.bse['treatment']:.2f}]")
print(f"  P-value: {model_main.pvalues['treatment']:.4f}")

# 2. With controls
model_ctrl = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"\nWith controls (staffing, visits):")
print(f"  Treatment effect: {model_ctrl.params['treatment']:.2f}")
print(f"  Standard error: {model_ctrl.bse['treatment']:.2f}")
print(f"  P-value: {model_ctrl.pvalues['treatment']:.4f}")

# 3. Pre-trends check - compare pre-period trends
pre_df = df[df['week'] < 15].copy()
# Check if treatment group was improving faster than control before the switch
pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()
pre_model = ols('service_score ~ switched_branch + week_centered + switched_branch:week_centered', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print(f"\nPre-trends test (interaction = differential trend before switch):")
print(f"  Differential trend coefficient: {pre_model.params['switched_branch:week_centered']:.3f}")
print(f"  P-value: {pre_model.pvalues['switched_branch:week_centered']:.4f}")
if pre_model.pvalues['switched_branch:week_centered'] < 0.05:
    print("  WARNING: Pre-trends are NOT parallel - treatment group was trending differently before the switch")
else:
    print("  Pre-trends appear parallel")

# 4. Effect size interpretation
baseline = df[df['after_switch'] == 0]['service_score'].mean()
effect = model_main.params['treatment']
print(f"\n=== EFFECT SIZE ===")
print(f"  Baseline average service score: {baseline:.2f}")
print(f"  Treatment effect: {effect:.2f} points")
print(f"  Relative improvement: {100*effect/baseline:.1f}%")
print(f"  Typical branch SD: {df['service_score'].std():.2f}")
print(f"  Effect in SD units: {effect/df['service_score'].std():.2f} SD")
