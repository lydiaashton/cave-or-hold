import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model 1: Simple DiD without controls
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Model 1: Simple DiD (no controls) ===")
print(model1.summary().tables[1])

# Model 2: DiD with controls (staffing and visits)
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\n=== Model 2: DiD with controls ===")
print(model2.summary().tables[1])

# Model 3: With branch fixed effects (using dummy variables for branches)
# This is more appropriate for panel data
print("\n=== Model 3: DiD with Branch Fixed Effects ===")
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
X = pd.concat([df[['switched_branch', 'after_switch', 'treatment', 'staffing_index', 'visits_index']], branch_dummies], axis=1)
X = sm.add_constant(X)
y = df['service_score']
model3 = sm.OLS(y, X).fit()
print(f"Treatment effect (treatment coefficient): {model3.params['treatment']:.4f}")
print(f"Standard error: {model3.bse['treatment']:.4f}")
print(f"95% CI: [{model3.conf_int().loc['treatment'][0]:.4f}, {model3.conf_int().loc['treatment'][1]:.4f}]")
print(f"p-value: {model3.pvalues['treatment']:.6f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create week dummies relative to switch (week 15)
# Let's do an event study approach
df['weeks_relative'] = df['week'] - 15  # Week 15 is the switch week (0)

# Create indicators for each week relative to switch, interacted with treatment
# We'll use week 14 as the reference (last week before switch)
print("=== Event Study Analysis ===")
print("Checking treatment effect by week relative to switch")

# Create week dummies
week_dummies = pd.get_dummies(df['weeks_relative'], prefix='week_rel')
df = pd.concat([df, week_dummies], axis=1)

# Create interaction terms with treatment
for col in week_dummies.columns:
    df[f'{col}_treated'] = df[col] * df['switched_branch']

# Run regression with week 14 as reference (drop it)
# We want to see if the treatment effect varies by week
treatment_week_cols = [col for col in df.columns if 'week_rel' in col and '_treated' in col and 'week_rel_14' not in col]
control_week_cols = [col for col in df.columns if 'week_rel' in col and '_treated' not in col and 'week_rel_14' not in col]

# Build the formula
formula = 'service_score ~ switched_branch + ' + ' + '.join(control_week_cols) + ' + ' + ' + '.join(treatment_week_cols)

model_event = ols(formula, data=df).fit()

# Extract the treatment effects by week
print("\nTreatment effects by week (relative to week 14 baseline):")
results = []
for col in treatment_week_cols:
    if col in model_event.params:
        coef = model_event.params[col]
        pval = model_event.pvalues[col]
        week_num = int(col.replace('week_rel_', '').replace('_treated', ''))
        results.append({'week_relative': week_num, 'effect': coef, 'p_value': pval})

results_df = pd.DataFrame(results).sort_values('week_relative')
print(results_df.to_string(index=False))

# Separate pre and post effects
pre_effects = results_df[results_df['week_relative'] < 0]['effect']
post_effects = results_df[results_df['week_relative'] >= 0]['effect']

print(f"\nAverage pre-switch treatment effect: {pre_effects.mean():.2f}")
print(f"Average post-switch treatment effect: {post_effects.mean():.2f}")
print(f"Estimated causal effect (post - pre): {post_effects.mean() - pre_effects.mean():.2f}")

import pandas as pd

df['treatment_interaction'] = df['switched_branch'] * df['post_period']

# Use regular OLS first
model = ols('service_score ~ switched_branch + post_period + treatment_interaction', data=df).fit()

print("=== Standard DiD Results ===")
print(f"Treatment coefficient: {model.params['treatment_interaction']:.4f}")
print(f"Standard error: {model.bse['treatment_interaction']:.4f}")
print(f"95% CI: [{model.conf_int().loc['treatment_interaction'][0]:.4f}, {model.conf_int().loc['treatment_interaction'][1]:.4f}]")
print(f"p-value: {model.pvalues['treatment_interaction']:.6f}")

# Now let's manually calculate cluster-robust SE
# Using the sandwich estimator approach
def cluster_robust_se(model, cluster_col):
    """Calculate cluster-robust standard errors"""
    clusters = model.model.data.orig_exog.index  # This won't work directly
    # Let's use a simpler approach with the score residuals
    
    # Get residuals and design matrix
    residuals = model.resid
    X = model.model.exog
    
    # Get cluster identifiers from the original data
    cluster_ids = df['branch_id'].values
    
    # Calculate cluster-robust variance
    unique_clusters = np.unique(cluster_ids)
    n_clusters = len(unique_clusters)
    n = len(residuals)
    k = X.shape[1]
    
    # Score contributions by cluster
    scores_by_cluster = {}
    for i, cluster in enumerate(unique_clusters):
        mask = cluster_ids == cluster
        if mask.sum() > 0:
            scores_by_cluster[cluster] = X[mask].T @ residuals[mask]
    
    # Outer product of scores
    meat = np.zeros((k, k))
    for cluster in unique_clusters:
        s = scores_by_cluster[cluster]
        meat += np.outer(s, s)
    
    # Bread (inverse of Hessian)
    bread = np.linalg.inv(X.T @ X)
    
    # Cluster-robust variance estimator
    df_correction = n_clusters / (n_clusters - 1) * (n - 1) / (n - k)
    vcov_cluster = df_correction * bread @ meat @ bread
    
    # Standard errors
    se_cluster = np.sqrt(np.diag(vcov_cluster))
    
    return se_cluster

cluster_se = cluster_robust_se(model, 'branch_id')
