import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Count branches by treatment status
print("Branches by treatment status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate pre/post means for treatment and control groups
print("\n=== Service Score Averages ===")
print("\nPre-switch (weeks 6-14):")
pre = df[df['after_switch'] == 0].groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print(pre)

print("\nPost-switch (weeks 15-22):")
post = df[df['after_switch'] == 1].groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print(post)

# Calculate the differences
print("\n=== Difference-in-Differences Calculation ===")
control_pre = df[(df['after_switch'] == 0) & (df['switched_branch'] == 0)]['service_score'].mean()
control_post = df[(df['after_switch'] == 1) & (df['switched_branch'] == 0)]['service_score'].mean()
treatment_pre = df[(df['after_switch'] == 0) & (df['switched_branch'] == 1)]['service_score'].mean()
treatment_post = df[(df['after_switch'] == 1) & (df['switched_branch'] == 1)]['service_score'].mean()

control_diff = control_post - control_pre
treatment_diff = treatment_post - treatment_pre
did_estimate = treatment_diff - control_diff

print(f"Control group (no switch): {control_pre:.2f} -> {control_post:.2f} (change: {control_diff:.2f})")
print(f"Treatment group (switched): {treatment_pre:.2f} -> {treatment_post:.2f} (change: {treatment_diff:.2f})")
print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")

# Run DiD regression
print("\n=== Regression Analysis ===")
df['interaction'] = df['switched_branch'] * df['after_switch']

# Basic DiD model
model = ols('service_score ~ switched_branch + after_switch + interaction', data=df).fit()
print("\nBasic DiD Model:")
print(model.summary())

# DiD with controls
model_controls = ols('service_score ~ switched_branch + after_switch + interaction + staffing_index + visits_index', data=df).fit()
print("\nDiD Model with Controls:")
print(model_controls.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the interaction term
df['interaction'] = df['switched_branch'] * df['after_switch']

# Run DiD with branch fixed effects using a simpler approach
print("=== DiD with Branch Fixed Effects ===")

# Use C() notation for categorical variables in the formula
model_fe = ols('service_score ~ C(branch_id) + after_switch + interaction', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Just show the key coefficients
print("\nKey coefficients (with branch fixed effects, clustered SEs):")
key_vars = ['after_switch', 'interaction']
for var in key_vars:
    coef = model_fe.params[var]
    se = model_fe.bse[var]
    t = model_fe.tvalues[var]
    p = model_fe.pvalues[var]
    print(f"{var}: coef={coef:.3f}, SE={se:.3f}, t={t:.2f}, p={p:.4f}")

# Also check if there's any pre-trend difference
print("\n=== Pre-Trend Check (weeks 6-14 only) ===")
pre_data = df[df['week'] < 15].copy()
pre_data['week_centered'] = pre_data['week'] - 10  # center around middle of pre-period
pre_data['interaction_trend'] = pre_data['week_centered'] * pre_data['switched_branch']

pre_model = ols('service_score ~ switched_branch + week_centered + interaction_trend', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print(f"\nPre-trend interaction coefficient:")
print(f"interaction_trend: coef={pre_model.params['interaction_trend']:.3f}, SE={pre_model.bse['interaction_trend']:.3f}, p={pre_model.pvalues['interaction_trend']:.4f}")

# Effect size relative to baseline
baseline = df['service_score'].mean()
effect = model_fe.params['interaction']
print(f"\n=== Effect Size ===")
print(f"Baseline service score: {baseline:.2f}")
print(f"Estimated treatment effect: {effect:.2f} points")
print(f"Effect as % of baseline: {effect/baseline*100:.1f}%")

# Confidence interval
ci_lower = effect - 1.96 * model_fe.bse['interaction']
ci_upper = effect + 1.96 * model_fe.bse['interaction']
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
