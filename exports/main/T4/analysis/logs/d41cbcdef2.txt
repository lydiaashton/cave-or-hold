after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()

# Create week dummies and interact with treatment for pre-period only

# Create week indicators for pre-period (weeks 6-14)

    df[f'week_{week}'] = (df['week'] == week).astype(int)
    df[f'week_{week}_treatment'] = df[f'week_{week}'] * df['switched_branch']

import pandas as pd

# Test for parallel pre-trends only for weeks immediately before switch (11-14)
df['treatment_interaction'] = df['switched_branch'] * df['after_switch']

# Create week indicators for weeks 11-14 (week 10 as baseline for this test)
for week in range(11, 15):
    df[f'week_{week}'] = (df['week'] == week).astype(int)
    df[f'week_{week}_treatment'] = df[f'week_{week}'] * df['switched_branch']

# Regression with recent pre-trend interactions
pre_trend_vars = '+'.join([f'week_{w}_treatment' for w in range(11, 15)])
pre_trend_formula = f'service_score ~ switched_branch + after_switch + treatment_interaction + {pre_trend_vars}'

model_recent_pre_trends = ols(pre_trend_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Recent pre-trends test (weeks 11-14, with week 10 as baseline):")
print(model_recent_pre_trends.summary())

# Joint test of recent pre-trend coefficients
recent_pre_trend_coefs = [f'week_{w}_treatment' for w in range(11, 15)]
print(f"\n\nJoint test for recent pre-trends (H0: all coefficients = 0):")
joint_test = model_recent_pre_trends.f_test(recent_pre_trend_coefs)

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment_interaction'] = df['switched_branch'] * df['after_switch']

# Main model
model = ols('service_score ~ switched_branch + after_switch + treatment_interaction', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get the treatment effect and confidence interval
treatment_effect = model.params['treatment_interaction']
ci_lower = model.conf_int().loc['treatment_interaction', 0]
ci_upper = model.conf_int().loc['treatment_interaction', 1]
p_value = model.pvalues['treatment_interaction']

print("MAIN RESULTS SUMMARY")
print("=" * 50)
print(f"Treatment effect estimate: {treatment_effect:.2f} points")
print(f"95% Confidence interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {p_value:.4f}")
print(f"Statistically significant: {'Yes' if p_value < 0.05 else 'No'}")

# Calculate effect size relative to baseline
baseline_score = df[df['after_switch'] == 0]['service_score'].mean()
effect_size_pct = (treatment_effect / baseline_score) * 100
print(f"\nBaseline average service score: {baseline_score:.2f}")
print(f"Effect size: {effect_size_pct:.1f}% of baseline")

# Also check the standard deviation to understand magnitude
sd_score = df['service_score'].std()
effect_size_sd = treatment_effect / sd_score
print(f"Standard deviation of service scores: {sd_score:.2f}")
print(f"Effect size in SD units: {effect_size_sd:.2f} SD")

# Practical interpretation
print("\n" + "=" * 50)
print("PRACTICAL INTERPRETATION")
print("=" * 50)
print(f"The new procedure is associated with an increase of about {treatment_effect:.1f} points")
print(f"in service scores. Given that scores average around 50, this represents")
print(f"approximately a {effect_size_pct:.1f}% improvement.")
