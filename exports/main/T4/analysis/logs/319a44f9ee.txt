import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create relative week variable (weeks relative to switch at week 15)
df['relative_week'] = df['week'] - 15

# Event study regression - interact treatment with each week
# Use week 14 as reference (last pre-treatment week)
df['week_cat'] = df['week'].astype(str)

# Create event study dummies
event_study_df = df.copy()
for w in range(6, 23):
    if w != 14:  # reference period
        event_study_df[f'week_{w}'] = ((event_study_df['week'] == w) & (event_study_df['switched_branch'] == 1)).astype(int)

# Build formula for event study
week_vars = [f'week_{w}' for w in range(6, 23) if w != 14]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(week_vars) + ' + staffing_index + visits_index'

event_model = ols(formula, data=event_study_df).fit(cov_type='cluster', cov_kwds={'groups': event_study_df['branch_id']})

print("Event Study Regression Results:")
print("Coefficients for treatment group by week (relative to week 14):")
for w in range(6, 23):
    if w != 14:
        coef = event_model.params[f'week_{w}']
        se = event_model.bse[f'week_{w}']
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        period = "PRE" if w < 15 else "POST"
        print(f"Week {w} ({period}): coef={coef:.2f}, SE={se:.2f}, 95% CI=[{ci_low:.2f}, {ci_high:.2f}]")

# Plot the event study coefficients
pre_coefs = []
pre_ci_low = []
pre_ci_high = []
post_coefs = []
post_ci_low = []
post_ci_high = []
weeks = []

for w in range(6, 23):
    if w != 14:
        coef = event_model.params[f'week_{w}']
        se = event_model.bse[f'week_{w}']
        weeks.append(w)
        if w < 15:
            pre_coefs.append(coef)
            pre_ci_low.append(coef - 1.96 * se)
            pre_ci_high.append(coef + 1.96 * se)
        else:
            post_coefs.append(coef)
            post_ci_low.append(coef - 1.96 * se)
            post_ci_high.append(coef + 1.96 * se)

# Create plot
fig, ax = plt.subplots(figsize=(10, 6))
ax.axhline(y=0, color='black', linestyle='--', linewidth=1)
ax.axvline(x=14.5, color='red', linestyle='-', linewidth=2, label='Switch (week 15)')

# Plot pre-treatment
pre_weeks = [w for w in range(6, 15)]
ax.errorbar(pre_weeks, pre_coefs, yerr=[np.array(pre_coefs) - np.array(pre_ci_low), np.array(pre_ci_high) - np.array(pre_coefs)], 
            fmt='o', color='blue', label='Pre-treatment', capsize=5)

# Plot post-treatment
post_weeks = [w for w in range(15, 23)]
ax.errorbar(post_weeks, post_coefs, yerr=[np.array(post_coefs) - np.array(post_ci_low), np.array(post_ci_high) - np.array(post_coefs)], 
            fmt='o', color='green', label='Post-treatment', capsize=5)

ax.set_xlabel('Week')
ax.set_ylabel('Effect on Service Score (relative to week 14)')
ax.set_title('Event Study: Treatment Effect by Week')
ax.legend()
ax.grid(True, alpha=0.3)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Let me do a cleaner event study
# Create relative week (0 = week 15, the switch week)
df['rel_week'] = df['week'] - 15

# Create dummies for each relative week, omitting week -1 (week 14) as reference
event_df = df.copy()
for rw in range(-9, 8):  # weeks -9 to +7 (week 6 to 22, excluding -1)
    if rw != -1:
        event_df[f'rel_{rw}'] = ((event_df['rel_week'] == rw) & (event_df['switched_branch'] == 1)).astype(int)

# Build formula
rel_vars = [f'rel_{rw}' for rw in range(-9, 8) if rw != -1]
formula = 'service_score ~ switched_branch + ' + ' + '.join(rel_vars) + ' + staffing_index + visits_index'

event_model = ols(formula, data=event_df).fit(cov_type='cluster', cov_kwds={'groups': event_df['branch_id']})

print("Event Study (relative to week 14):")
print("=" * 60)

# Store results for plotting
results = []
for rw in range(-9, 8):
    if rw != -1:
        coef = event_model.params[f'rel_{rw}']
        se = event_model.bse[f'rel_{rw}']
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        pval = 2 * (1 - sm.stats.norm.cdf(abs(coef / se)))
        period = "PRE" if rw < 0 else "POST"
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"Relative week {rw:+d} (week {rw+15}) [{period}]: {coef:6.2f} (SE={se:.2f}) 95%CI=[{ci_low:.2f}, {ci_high:.2f}] {sig}")
        results.append((rw, coef, se, ci_low, ci_high, period))

# Plot
fig, ax = plt.subplots(figsize=(12, 6))
ax.axhline(y=0, color='black', linestyle='--', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='-', linewidth=2, label='Switch point')

post_results = [r for r in results if r[5] == "POST"]

if pre_results:
    pre_rw = [r[0] for r in pre_results]
    pre_coef = [r[1] for r in pre_results]
    pre_yerr = [[r[1] - r[3] for r in pre_results], [r[4] - r[1] for r in pre_results]]

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create relative week (0 = week 15, the switch week)
df['rel_week'] = df['week'] - 15

# Create dummies for each relative week, omitting week -1 (week 14) as reference
# Use w_ prefix to avoid formula issues with negative numbers
event_df = df.copy()
for rw in range(-9, 8):  # weeks -9 to +7 (week 6 to 22, excluding -1)
    if rw != -1:
        col_name = f'w{rw}'
        event_df[col_name] = ((event_df['rel_week'] == rw) & (event_df['switched_branch'] == 1)).astype(int)

# Build formula
rel_vars = [f'w{rw}' for rw in range(-9, 8) if rw != -1]
formula = 'service_score ~ switched_branch + ' + ' + '.join(rel_vars) + ' + staffing_index + visits_index'

event_model = ols(formula, data=event_df).fit(cov_type='cluster', cov_kwds={'groups': event_df['branch_id']})

print("Event Study (relative to week 14):")
print("=" * 70)

# Store results for plotting
results = []
for rw in range(-9, 8):
    if rw != -1:
        coef = event_model.params[f'w{rw}']
        se = event_model.bse[f'w{rw}']
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        pval = 2 * (1 - sm.stats.norm.cdf(abs(coef / se)))
        period = "PRE" if rw < 0 else "POST"
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"Relative week {rw:+d} (week {rw+15}) [{period}]: {coef:6.2f} (SE={se:.2f}) 95%CI=[{ci_low:.2f}, {ci_high:.2f}] {sig}")
        results.append((rw, coef, se, ci_low, ci_high, period))

# Plot
fig, ax = plt.subplots(figsize=(12, 6))
ax.axhline(y=0, color='black', linestyle='--', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='-', linewidth=2, label='Switch point')

pre_results = [r for r in results if r[5] == "PRE"]
post_results = [r for r in results if r[5] == "POST"]

if pre_results:
    pre_rw = [r[0] for r in pre_results]
    pre_coef = [r[1] for r in pre_results]
    pre_yerr = [[r[1] - r[3] for r in pre_results], [r[4] - r[1] for r in pre_results]]
    ax.errorbar(pre_rw, pre_coef, yerr=pre_yerr, fmt='o', color='blue', label='Pre-treatment', capsize=5, markersize=8)

if post_results:
    post_rw = [r[0] for r in post_results]
    post_coef = [r[1] for r in post_results]
    post_yerr = [[r[1] - r[3] for r in post_results], [r[4] - r[1] for r in post_results]]
    ax.errorbar(post_rw, post_coef, yerr=post_yerr, fmt='o', color='green', label='Post-treatment', capsize=5, markersize=8)

ax.set_xlabel('Relative Week (0 = week 15, switch week)')
ax.set_ylabel('Effect on Service Score (treatment vs control, relative to week 14)')
ax.set_title('Event Study: Treatment Effect Dynamics')
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-9, 8))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)

import pandas as pd
import numpy as np
import statsmodels.api as sm
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create relative week (0 = week 15, the switch week)
df['rel_week'] = df['week'] - 15

# Create dummies for each relative week, omitting week -1 (week 14) as reference
# Use Q() notation to handle special characters in variable names
event_df = df.copy()
for rw in range(-9, 8):  # weeks -9 to +7 (week 6 to 22, excluding -1)
    if rw != -1:
        col_name = f'w{rw}'
        event_df[col_name] = ((event_df['rel_week'] == rw) & (event_df['switched_branch'] == 1)).astype(int)

# Build formula using Q() notation for variables with special characters
rel_vars = [f"Q('w{rw}')" for rw in range(-9, 8) if rw != -1]
formula = "service_score ~ switched_branch + " + " + ".join(rel_vars) + " + staffing_index + visits_index"

print("Formula:", formula[:200], "...")

event_model = sm.OLS.from_formula(formula, data=event_df).fit(cov_type='cluster', cov_kwds={'groups': event_df['branch_id']})

print("\nEvent Study (relative to week 14):")
print("=" * 70)

# Store results for plotting
results = []
for rw in range(-9, 8):
    if rw != -1:
        coef = event_model.params[f'w{rw}']
        se = event_model.bse[f'w{rw}']
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        pval = 2 * (1 - sm.stats.norm.cdf(abs(coef / se)))
        period = "PRE" if rw < 0 else "POST"
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"Relative week {rw:+d} (week {rw+15}) [{period}]: {coef:6.2f} (SE={se:.2f}) 95%CI=[{ci_low:.2f}, {ci_high:.2f}] {sig}")
        results.append((rw, coef, se, ci_low, ci_high, period))
