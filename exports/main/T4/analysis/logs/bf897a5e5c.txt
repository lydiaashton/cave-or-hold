import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with and without controls
print("=== DiD Regression (No Controls) ===")
model1 = ols('service_score ~ switched_branch + after_switch + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model1.summary().tables[1])

print("\n=== DiD Regression (With Controls) ===")
model2 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model2.summary().tables[1])

# Check parallel trends more formally - event study approach
print("\n=== Event Study: Week-by-week effects ===")
# Create week dummies interacted with treated
df['week_centered'] = df['week'] - 15  # Center at switch week

# Create indicators for each week relative to switch
for w in range(-9, 8):  # weeks -9 to +7 relative to week 15
    df[f'week_{w}'] = ((df['week'] - 15) == w).astype(int)

# Run event study regression (excluding week -1 as reference)
week_cols = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'switched_branch * {col}' for col in week_cols]) + ' + staffing_index + visits_index'
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nEvent study coefficients (interaction of treated x week dummies):")
for w in range(-9, 8):
    if w != -1:
        coef_name = f'switched_branch * week_{w}'
        if coef_name in model_event.params.index:
            coef = model_event.params[coef_name]
            se = model_event.bse[coef_name]
            print(f"Week {15+w}: coef={coef:.3f}, SE={se:.3f}")

# Post-switch trends
print("\n=== Post-Switch Week-by-week (Treated - Control difference) ===")
