import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check how many branches switched vs didn't
branches = df.groupby('branch_id')['switched_branch'].first()
print("Branches that switched:", branches.sum())
print("Branches that didn't switch:", len(branches) - branches.sum())
print("Total branches:", len(branches))

# Check balance - average service score by group before and after
print("\n=== Service Score by Group and Period ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the DiD estimate manually
# Treatment group (switched=1): before and after
# Control group (switched=0): before and after

treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print(f"\nTreatment (switched) before: {treatment_before:.2f}")
print(f"Treatment (switched) after: {treatment_after:.2f}")
print(f"Control (not switched) before: {control_before:.2f}")
print(f"Control (not switched) after: {control_after:.2f}")

treatment_change = treatment_after - treatment_before
control_change = control_after - control_before
did_estimate = treatment_change - control_change

print(f"\nTreatment change: {treatment_change:.2f}")
print(f"Control change: {control_change:.2f}")
print(f"DiD estimate (treatment effect): {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Model 1: Basic DiD ===")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Model 2: DiD with Controls ===")
print(model2.summary().tables[1])

# Get the treatment effect estimate and confidence interval
treatment_effect = model2.params['treatment']
treatment_se = model2.bse['treatment']
treatment_ci_lower = treatment_effect - 1.96 * treatment_se
treatment_ci_upper = treatment_effect + 1.96 * treatment_se
treatment_pvalue = model2.pvalues['treatment']

print(f"\n=== Treatment Effect Summary ===")
print(f"Estimate: {treatment_effect:.3f}")
print(f"Std Error: {treatment_se:.3f}")
print(f"95% CI: [{treatment_ci_lower:.3f}, {treatment_ci_upper:.3f}]")
print(f"P-value: {treatment_pvalue:.4f}")
