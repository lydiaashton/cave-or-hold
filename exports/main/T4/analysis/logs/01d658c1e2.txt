import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['period'] = df['week'].apply(lambda x: 'After' if x >= 15 else 'Before')

# Create interaction term for DiD
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

# Run DiD regression
# Basic DiD model
model1 = ols('service_score ~ treated + post + treated_post', data=df).fit()
print("=== Basic DiD Regression ===")
print(model1.summary())

# DiD with branch fixed effects
model2 = ols('service_score ~ treated + post + treated_post + C(branch_id)', data=df).fit()
print("\n\n=== DiD with Branch Fixed Effects ===")
print(model2.summary())

# DiD with branch fixed effects and control variables
model3 = ols('service_score ~ treated + post + treated_post + C(branch_id) + staffing_index + visits_index', data=df).fit()
print("\n\n=== DiD with Branch FE and Controls ===")

# Use week 14 as reference (last pre-switch week)

    df[f'week_{w}'] = (df['weeks_relative'] == w).astype(int)

# Run event study regression with branch fixed effects
# Only include weeks where we have data
event_study_vars = [f'week_{w}' for w in range(-8, 8) if f'week_{w}' in df.columns]
# week_-9 is the reference category

formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(branch_id)'
# Note: we need to be careful about collinearity with branch FE and treatment indicator
