import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = β0 + β1*switched_branch + β2*after_switch + β3*(switched_branch * after_switch) + ε
# β3 is our DiD estimator

df['interaction'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + interaction', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# Now let's add branch fixed effects to control for time-invariant branch characteristics
# This is a more robust specification
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df_with_dummies['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Build formula with branch fixed effects
branch_cols = [col for col in df_with_dummies.columns if col.startswith('branch_')]
formula = 'service_score ~ switched_branch + after_switch + interaction + ' + ' + '.join(branch_cols)
model_fe = ols(formula, data=df_with_dummies).fit()
print("\n=== DiD Regression with Branch Fixed Effects ===")
# Just show the key coefficients
print(f"Interaction coefficient (DiD estimate): {model_fe.params['interaction']:.3f}")
print(f"Standard error: {model_fe.bse['interaction']:.3f}")
print(f"95% CI: [{model_fe.conf_int().loc['interaction'][0]:.3f}, {model_fe.conf_int().loc['interaction'][1]:.3f}]")
print(f"P-value: {model_fe.pvalues['interaction']:.4f}")

# Also add week fixed effects for even more robustness
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df_full = pd.concat([df, branch_dummies, week_dummies], axis=1)
branch_cols = [col for col in df_full.columns if col.startswith('branch_')]
week_cols = [col for col in df_full.columns if col.startswith('week_')]
formula_full = 'service_score ~ switched_branch + after_switch + interaction + ' + ' + '.join(branch_cols) + ' + ' + ' + '.join(week_cols)
model_full = ols(formula_full, data=df_full).fit()
print(f"\n=== DiD Regression with Branch AND Week Fixed Effects ===")
print(f"Interaction coefficient (DiD estimate): {model_full.params['interaction']:.3f}")
print(f"Standard error: {model_full.bse['interaction']:.3f}")
print(f"95% CI: [{model_full.conf_int().loc['interaction'][0]:.3f}, {model_full.conf_int().loc['interaction'][1]:.3f}]")
print(f"P-value: {model_full.pvalues['interaction']:.4f}")
