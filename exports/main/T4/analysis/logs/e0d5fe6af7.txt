import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD table: average service scores by group and period
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES SUMMARY TABLE")
print("=" * 60)

# Before period (weeks 6-14)
before = df[df['after_switch'] == 0]
# After period (weeks 15-22)
after = df[df['after_switch'] == 1]

# Control group (switched_branch = 0)
control_before = before[before['switched_branch'] == 0]['service_score'].mean()
control_after = after[after['switched_branch'] == 0]['service_score'].mean()
control_diff = control_after - control_before

# Treatment group (switched_branch = 1)
treat_before = before[before['switched_branch'] == 1]['service_score'].mean()
treat_after = after[after['switched_branch'] == 1]['service_score'].mean()
treat_diff = treat_after - treat_before

# DiD estimate
did_estimate = treat_diff - control_diff

print(f"\nControl Group (No Switch):")
print(f"  Before (weeks 6-14):  {control_before:.2f}")
print(f"  After (weeks 15-22):  {control_after:.2f}")
print(f"  Change:               {control_diff:.2f}")

print(f"\nTreatment Group (Switched):")
print(f"  Before (weeks 6-14):  {treat_before:.2f}")
print(f"  After (weeks 15-22):  {treat_after:.2f}")
print(f"  Change:               {treat_diff:.2f}")

print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")
print(f"  (Treatment change - Control change)")

# Now run regression with standard errors
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Model 1: Basic DiD
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 1: Basic DiD (clustered SE by branch)")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 2: DiD with staffing and visits controls (clustered SE by branch)")
print(model2.summary().tables[1])

# Check parallel trends - plot average scores by week for each group
print("\n" + "=" * 60)
print("PARALLEL TRENDS CHECK - Weekly Averages")
print("=" * 60)

weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_avg.columns = ['Control (No Switch)', 'Treatment (Switched)']
weekly_avg['Diff'] = weekly_avg['Treatment (Switched)'] - weekly_avg['Control (No Switch)']

print("\nWeekly averages by group:")
print(weekly_avg.round(2))

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the parallel trends plot
plt.figure(figsize=(12, 6))

weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

plt.plot(weekly_avg.index, weekly_avg[0], 'o-', label='Control (No Switch)', linewidth=2, markersize=6, color='gray')
plt.plot(weekly_avg.index, weekly_avg[1], 's-', label='Treatment (Switched)', linewidth=2, markersize=6, color='blue')

# Add vertical line at switch point
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')

plt.xlabel('Week', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Scores by Week: Treatment vs Control Groups', fontsize=14, fontweight='bold')
plt.legend(fontsize=11)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.ylim(45, 60)

# Add annotation
plt.annotate('New Procedure\nImplemented', xy=(15, 57), fontsize=10, 
             arrowprops=dict(arrowstyle='->', color='red'))

plt.tight_layout()
plt.savefig('service_scores_trends.png', dpi=150, bbox_inches='tight')
print("Plot saved as 'service_scores_trends.png'")

# Additional analysis: effect size relative to baseline
baseline_score = df['service_score'].mean()
did_effect = 1.95
percent_improvement = (did_effect / baseline_score) * 100

print(f"\n" + "=" * 60)
print("EFFECT SIZE SUMMARY")
print("=" * 60)
print(f"Baseline average service score: {baseline_score:.2f}")
print(f"DiD estimated effect: +{did_effect:.2f} points")
print(f"Percentage improvement: {percent_improvement:.1f}%")

# Check robustness: any branches that switched but show negative effects?
branch_effects = df.groupby(['branch_id', 'switched_branch']).agg({
    'service_score': ['mean', 'std', 'count']
}).reset_index()

print(f"\nNumber of switched branches: {len(switched_branches)}")
print(f"Average score for switched branches: {switched_branches['mean_score'].mean():.2f}")
