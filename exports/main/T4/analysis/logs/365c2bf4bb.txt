df['treated_post'] = df['treated'] * df['post']

# Basic DiD regression
print("=== Basic DiD Regression ===")
model1 = ols('service_score ~ treated + post + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model1.summary())
