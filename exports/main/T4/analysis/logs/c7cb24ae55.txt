import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Basic DiD Regression (no clustering):")
print(model.summary())

# Now with branch fixed effects and clustering
# Add branch fixed effects
df = df.copy()
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [col for col in df_with_fe.columns if col.startswith('branch_')]
formula = 'service_score ~ after_switch + treatment + ' + ' + '.join(branch_cols)

model_fe = ols(formula, data=df_with_fe).fit(cov_type='cluster', cov_kwds={'groups': df_with_fe['branch_id']})
print("\n\nDiD Regression with Branch Fixed Effects (clustered SEs):")
print(model_fe.summary())

# Also control for staffing and visits
formula_controlled = 'service_score ~ after_switch + treatment + staffing_index + visits_index + ' + ' + '.join(branch_cols)
model_controlled = ols(formula_controlled, data=df_with_fe).fit(cov_type='cluster', cov_kwds={'groups': df_with_fe['branch_id']})
print("\n\nDiD Regression with Controls (clustered SEs):")
print(model_controlled.summary())

diff = treatment['change'].mean() - control['change'].mean()
pooled_std = np.sqrt((treatment['change'].var() * (len(treatment)-1) + control['change'].var() * (len(control)-1)) / (len(treatment) + len(control) - 2))
cohens_d = diff / pooled_std
