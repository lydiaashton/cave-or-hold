import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimator

df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Also run with cluster-robust standard errors by branch
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Regression with Cluster-Robust SE (by branch) ===")
print(model_clustered.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Final DiD model with cluster-robust SE
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get the treatment effect and confidence interval
treatment_effect = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pvalue = model.pvalues['treatment']
ci_lower = treatment_effect - 1.96 * treatment_se
ci_upper = treatment_effect + 1.96 * treatment_se

print("=== Final Results Summary ===")
print(f"Treatment effect (DiD estimate): {treatment_effect:.2f} points")
print(f"Standard error: {treatment_se:.2f}")
print(f"95% Confidence interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {treatment_pvalue:.4f}")

# Calculate percentage improvement relative to baseline
baseline_score = df[df['after_switch'] == 0]['service_score'].mean()
pct_improvement = (treatment_effect / baseline_score) * 100
print(f"\nBaseline average service score: {baseline_score:.1f}")
print(f"Percentage improvement: {pct_improvement:.1f}%")

# Also check the effect size in terms of standard deviations
overall_std = df['service_score'].std()
effect_size_std = treatment_effect / overall_std
print(f"Overall service score std dev: {overall_std:.2f}")
print(f"Effect size (in std dev units): {effect_size_std:.2f}")

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Check if effect varies by branch characteristics
# Calculate branch-level before/after differences
branch_before = df[df['after_switch'] == 0].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_before.columns = ['branch_id', 'switched_branch', 'score_before']

branch_after = df[df['after_switch'] == 1].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_after.columns = ['branch_id', 'switched_branch', 'score_after']

branch_changes = branch_before.merge(branch_after, on=['branch_id', 'switched_branch'])
branch_changes['change'] = branch_changes['score_after'] - branch_changes['score_before']
