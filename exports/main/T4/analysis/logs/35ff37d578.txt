# Simple DiD with cluster-robust SEs
print("=== DiD with Cluster-Robust Standard Errors ===")
model_cluster = ols('service_score ~ switched_branch + after_switch + DiD', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_cluster.summary().tables[1])

# DiD with controls and cluster-robust SEs
print("\n=== DiD with Controls and Cluster-Robust SEs ===")

import pandas as pd

import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the visualization
fig, ax = plt.subplots(figsize=(10, 6))

weekly_means = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

# Plot both groups
ax.plot(weekly_means.index, weekly_means[0], 'o-', label='Non-Switched Branches (Old Procedure)', linewidth=2, markersize=8, color='gray')
ax.plot(weekly_means.index, weekly_means[1], 's-', label='Switched Branches (New Procedure)', linewidth=2, markersize=8, color='blue')

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Implemented (Week 15)')

# Add annotation for the effect
ax.annotate('New Procedure Effect:\n+2.4 points', xy=(18, 53), xytext=(16, 55),
            arrowprops=dict(arrowstyle='->', color='green', lw=2),
            fontsize=11, color='green', fontweight='bold',
            bbox=dict(boxstyle='round', facecolor='white', edgecolor='green', lw=1))

ax.set_ylabel('Service Score', fontsize=12)
ax.set_title('Library Service Scores: Before and After Procedure Change', fontsize=14, fontweight='bold')
ax.legend(loc='upper left', fontsize=10)
ax.set_xticks(range(6, 23))
ax.set_xlim(5.5, 22.5)
ax.set_ylim(42, 60)

from statsmodels.formula.api import ols
model = ols('service_score ~ switched_branch + after_switch + DiD', data=df.assign(DiD=df['switched_branch']*df['after_switch'])).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
did_coef = model.params['DiD']
did_se = model.bse['DiD']

print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {model.pvalues['DiD']:.6f}")
