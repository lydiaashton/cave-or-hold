import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# DiD with branch fixed effects
print("\n=== DiD with Branch Fixed Effects ===")
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = df.copy()
for col in branch_dummies.columns:
    df_with_fe[col] = branch_dummies[col]

# Build formula with fixed effects
fe_cols = branch_dummies.columns.tolist()
formula = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(fe_cols)
model_fe = ols(formula, data=df_with_fe).fit()
print(model_fe.summary())

# Cluster-robust standard errors at branch level
print("\n=== DiD with Branch Fixed Effects (Cluster-Robust SE) ===")
model_fe_cluster = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
cov_type = 'cluster'
cov_kwds = {'groups': df['branch_id']}
