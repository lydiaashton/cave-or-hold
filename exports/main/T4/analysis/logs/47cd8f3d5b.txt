import statsmodels.formula.api as smf

# Load the data
df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Summary statistics by group and period
print("=== Service Score Summary ===")
print("\nBy switched_branch and after_switch:")
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Calculate the DiD manually
print("\n=== Difference-in-Differences Calculation ===")

# Treatment group (switched=1)
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treatment_diff = treatment_after - treatment_before

# Control group (switched=0)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_diff = control_after - control_before

# DiD estimate
did_estimate = treatment_diff - control_diff

print(f"\nTreatment group (switched branches):")
print(f"  Before: {treatment_before:.2f}")
print(f"  After:  {treatment_after:.2f}")
print(f"  Change: {treatment_diff:.2f}")

print(f"\nControl group (non-switched branches):")
print(f"  Before: {control_before:.2f}")
print(f"  After:  {control_after:.2f}")
print(f"  Change: {control_diff:.2f}")

print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
import statsmodels.formula.api as smf

# Load the data
df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Run regression with cluster-robust standard errors (cluster by branch)
print("=== Regression Analysis ===")

# Model 1: Basic DiD
model1 = smf.ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = smf.ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Get the DiD coefficient and confidence interval
did_coef = model2.params['treated_post']
did_se = model2.bse['treated_post']
did_pvalue = model2.pvalues['treated_post']
did_ci_lower = did_coef - 1.96 * did_se
did_ci_upper = did_coef + 1.96 * did_se

print(f"\n=== Key Result ===")
print(f"DiD coefficient (treated_post): {did_coef:.3f}")
print(f"Standard error: {did_se:.3f}")
print(f"95% CI: [{did_ci_lower:.3f}, {did_ci_upper:.3f}]")
print(f"P-value: {did_pvalue:.4f}")

effect_size_pct = (1.783 / baseline) * 100

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

# Load the data
df = pd.read_csv('branches.csv')

# Check parallel trends assumption: difference between groups in pre-period
print("=== Parallel Trends Check ===")
pre_data = df[df['week'] < 15]

# Average difference between treatment and control in pre-period
pre_diff = pre_data.groupby('switched_branch')['service_score'].mean()
print(f"Pre-period average (treatment): {pre_diff[1]:.2f}")
print(f"Pre-period average (control): {pre_diff[0]:.2f}")
print(f"Pre-period difference: {pre_diff[1] - pre_diff[0]:.2f}")

# Run regression on pre-period only to check if there's a time trend difference
pre_data = pre_data.copy()
pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()
model_pre = smf.ols('service_score ~ switched_branch * week_centered', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print("\nPre-period trend test (interaction of switched_branch with week):")
print(model_pre.summary().tables[1])

# Also check balance on control variables
print("\n=== Control Variable Balance ===")
balance = df.groupby('switched_branch')[['staffing_index', 'visits_index']].mean()
print(balance)

# Test if control variables differ significantly
model_staffing = smf.ols('staffing_index ~ switched_branch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
model_visits = smf.ols('visits_index ~ switched_branch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"\nStaffing index difference p-value: {model_staffing.pvalues['switched_branch']:.4f}")
print(f"Visits index difference p-value: {model_visits.pvalues['switched_branch']:.4f}")
