import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check how many branches switched vs didn't
branch_switch = df.groupby('branch_id')['switched_branch'].first()
print("Branches that switched:", (branch_switch == 1).sum())
print("Branches that didn't switch:", (branch_switch == 0).sum())

# Calculate average service scores by group and period
print("\n=== Average Service Scores ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
print(pivot)

# Calculate the differences
print("\n=== Differences ===")
diff_before = pivot[0] - pivot[0]  # Control group before
diff_after = pivot[1] - pivot[0]   # Treatment - Control after
print("Control group (didn't switch):")
print(f"  Before: {pivot.loc[0, 0]:.2f}")
print(f"  After: {pivot.loc[0, 1]:.2f}")
print(f"  Change: {pivot.loc[0, 1] - pivot.loc[0, 0]:.2f}")
print("\nTreatment group (switched):")
print(f"  Before: {pivot.loc[1, 0]:.2f}")
print(f"  After: {pivot.loc[1, 1]:.2f}")
print(f"  Change: {pivot.loc[1, 1] - pivot.loc[1, 0]:.2f}")

# Difference-in-differences estimate
did_estimate = (pivot.loc[1, 1] - pivot.loc[1, 0]) - (pivot.loc[0, 1] - pivot.loc[0, 0])
print(f"\n=== DiD Estimate: {did_estimate:.2f} ===")

# Run DiD regression with standard errors clustered by branch
print("\n=== DiD Regression ===")
df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple OLS first
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model.summary())

# Now with cluster-robust standard errors
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Clustered Standard Errors (by branch) ===")
print(model_clustered.summary())

import pandas as pd

import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create treatment variable
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD with controls
print("=== DiD Regression with Controls ===")
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_controls.summary())

# Check pre-trends more formally (weeks 6-14 only)
print("\n=== Pre-Trend Analysis (Weeks 6-14) ===")
pre_data = df[df['week'] < 15].copy()
pre_model = ols('service_score ~ switched_branch + week', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print(pre_model.summary())

# Event study - effect by week relative to switch
print("\n=== Event Study Analysis ===")
df['weeks_relative'] = df['week'] - 15  # Week 15 is the switch week (0)
# Create week dummies for event study
event_data = df.copy()
event_data['week_cat'] = event_data['week'].astype(str)

# Run regression with week fixed effects interacted with treatment
event_model = ols('service_score ~ switched_branch + C(week) + switched_branch:C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract the interaction coefficients (treatment effect by week)
interaction_coefs = []
for week in range(6, 23):
    coef_name = f'switched_branch:C(week)[{week}]'
    if coef_name in event_model.params.index:
        interaction_coefs.append({
            'week': week,
            'coef': event_model.params[coef_name],
            'se': event_model.bse[coef_name],
            'pvalue': event_model.pvalues[coef_name]
        })

interaction_df = pd.DataFrame(interaction_coefs)
print(interaction_df)

# Calculate average pre and post effects
pre_effects = interaction_df[interaction_df['week'] < 15]['coef'].mean()
post_effects = interaction_df[interaction_df['week'] >= 15]['coef'].mean()
print(f"\nAverage pre-switch effect: {pre_effects:.2f}")
print(f"Average post-switch effect: {post_effects:.2f}")
print(f"Difference: {post_effects - pre_effects:.2f}")

treatment_change = treatment_after - treatment_before
