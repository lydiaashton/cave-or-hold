print("\n=== Average Service Scores ===")

                       index='switched_branch', 
                       columns='period',

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DID
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

# Run DID regression
model = ols('service_score ~ treated + post + treated_post', data=df).fit()
print("=== Simple DID Regression ===")
print(model.summary())

# Now with branch fixed effects
print("\n=== DID with Branch Fixed Effects ===")
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ treated_post + post + C(branch_id_str)', data=df).fit()
print(model_fe.summary())

# With branch and week fixed effects
print("\n=== DID with Branch and Week Fixed Effects ===")
model_fefe = ols('service_score ~ treated_post + C(branch_id_str) + C(week)', data=df).fit()
print(model_fefe.summary())

# Robust standard errors
print("\n=== DID with Branch/Week FE and Robust SE ===")
model_robust = ols('service_score ~ treated_post + C(branch_id_str) + C(week)', data=df).fit()
robust_results = model_robust.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(robust_results.summary())

import pandas as pd

df = pd.read_csv('branches.csv')

# Correct branch counts
switched_branches = df.groupby('branch_id')['switched_branch'].first()
print(f"Total branches: {len(switched_branches)}")
print(f"Branches that switched: {switched_branches.sum()}")
print(f"Branches that didn't switch: {len(switched_branches) - switched_branches.sum()}")

# Check if pre-trends are parallel
print("\n=== Pre-Trend Parallel Check ===")
pre_data = df[df['week'] < 15]
pre_avg = pre_data.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print("Weekly averages before switch:")
print(pre_avg.round(2))

# Difference between groups each week
pre_avg['diff'] = pre_avg[1] - pre_avg[0]
print("\nDifference (Treatment - Control) each week:")
print(pre_avg['diff'].round(2))
print(f"\nAverage pre-period difference: {pre_avg['diff'].mean():.2f}")
print(f"Std dev of pre-period difference: {pre_avg['diff'].std():.2f}")
