import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch fixed effects
# Model: service_score = β0 + β1*after_switch + β2*switched_branch + β3*(after_switch*switched_branch) + ε
# With branch fixed effects to control for time-invariant branch characteristics

# First, simple DiD regression
model1 = ols('service_score ~ after_switch + switched_branch + treatment', data=df).fit()
print("Simple DiD Regression (no fixed effects):")
print(model1.summary())

# Now with branch fixed effects
df_with_dummies = pd.get_dummies(df, columns=['branch_id'], drop_first=True)
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_id_')]

# Build formula
formula = 'service_score ~ after_switch + switched_branch + treatment + ' + ' + '.join(branch_cols)
model2 = ols(formula, data=df_with_dummies).fit()
print("\n\nDiD Regression with Branch Fixed Effects:")
print(model2.summary())

# Extract the treatment effect
treatment_effect = model2.params['treatment']
treatment_se = model2.bse['treatment']
treatment_pvalue = model2.pvalues['treatment']
print(f"\n\nTreatment Effect: {treatment_effect:.3f} (SE: {treatment_se:.3f}, p-value: {treatment_pvalue:.4f})")

weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
print("Weekly averages by treatment status:")
print(weekly_avg.pivot(index='week', columns='switched_branch', values='service_score'))

# Run regression with week fixed effects and treatment interaction
# This allows us to test parallel trends
df['post'] = df['after_switch']
df['treated'] = df['switched_branch']
df['did'] = df['post'] * df['treated']

# Use cluster-robust standard errors at branch level
model = ols('service_score ~ post + treated + did', data=df).fit()
# Get cluster-robust SEs
cov_cluster = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print("\n\nDiD with Cluster-Robust SEs (clustered by branch):")
print(cov_cluster.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study design - create relative week indicators
df['relative_week'] = df['week'] - 14

# Create event study dummies with valid column names
event_weeks = [-8, -7, -6, -5, -4, -3, -2, -1, 1, 2, 3, 4, 5, 6, 7, 8]
for w in event_weeks:
    col_name = f'w{abs(w)}n' if w < 0 else f'w{w}p'
    df[col_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
event_cols = [f'w{abs(w)}n' if w < 0 else f'w{w}p' for w in event_weeks]
formula = 'service_score ~ ' + ' + '.join(event_cols) + ' + switched_branch + C(week)'

model_event = ols(formula, data=df).fit()
cov_event = model_event.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Get parameter names from original model
param_names = model_event.params.index.tolist()
params = cov_event.params
bse = cov_event.bse
pvalues = cov_event.pvalues

# Create a mapping
param_dict = {name: {'coef': params[i], 'se': bse[i], 'pval': pvalues[i]} 
              for i, name in enumerate(param_names)}

print("Event Study Coefficients (relative to week 14):")
print("-" * 70)
results = []
for w in event_weeks:
    col_name = f'w{abs(w)}n' if w < 0 else f'w{w}p'
    coef = param_dict[col_name]['coef']
    se = param_dict[col_name]['se']
    pval = param_dict[col_name]['pval']
    ci_low = coef - 1.96 * se
    ci_high = coef + 1.96 * se
    results.append({'week': w, 'coef': coef, 'se': se, 'pval': pval, 'ci_low': ci_low, 'ci_high': ci_high})
    marker = '*' if pval < 0.1 else ' '
    print(f"Week {w:+3d}: {coef:6.2f} (SE: {se:.2f}), 95% CI: [{ci_low:6.2f}, {ci_high:6.2f}] {marker}")

# Plot event study
results_df = pd.DataFrame(results)

fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(results_df['week'], results_df['coef'], 
            yerr=[results_df['coef'] - results_df['ci_low'], results_df['ci_high'] - results_df['coef']],
            fmt='o', capsize=5, markersize=8, linewidth=2)
ax.axhline(y=0, color='red', linestyle='--', linewidth=1.5)
ax.axvline(x=0.5, color='gray', linestyle=':', linewidth=2, label='Switch point')
ax.set_xlabel('Weeks Relative to Switch (0 = Week 14, reference)', fontsize=12)
ax.set_ylabel('Coefficient (Treatment vs Control)', fontsize=12)
ax.set_title('Event Study: Treatment Effect Dynamics', fontsize=14, fontweight='bold')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-8, 9))
ax.legend()

plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nSaved event study plot to event_study.png")

# Pre-trend test: joint test that all pre-period coefficients are zero
pre_weeks = [-8, -7, -6, -5, -4, -3, -2, -1]
pre_constraints = []
for w in pre_weeks:
    col_name = f'w{abs(w)}n'
    pre_constraints.append(f'{col_name} = 0')
pre_test = cov_event.f_test(pre_constraints)
print(f"\nJoint test of pre-trend coefficients = 0: F={pre_test.statistic:.2f}, p-value={pre_test.pvalue:.3f}")

# Post-period average effect
post_weeks = [1, 2, 3, 4, 5, 6, 7, 8]
post_coefs = []
for w in post_weeks:
    col_name = f'w{w}p'
    post_coefs.append(param_dict[col_name]['coef'])
avg_post_effect = np.mean(post_coefs)
print(f"Average post-switch effect: {avg_post_effect:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Final DiD regression with cluster-robust SEs
df['post'] = df['after_switch']
df['treated'] = df['switched_branch']
df['did'] = df['post'] * df['treated']

model = ols('service_score ~ post + treated + did', data=df).fit()
cov_robust = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("=" * 70)
print("FINAL ANALYSIS SUMMARY")
print("=" * 70)

# Extract key results
did_coef = cov_robust.params['did']
did_se = cov_robust.bse['did']
did_pval = cov_robust.pvalues['did']
did_ci_low = did_coef - 1.96 * did_se
did_ci_high = did_coef + 1.96 * did_se

print(f"\nDIFFERENCE-IN-DIFFERENCES ESTIMATE:")
print(f"  Treatment effect: {did_coef:.2f} points")
print(f"  Standard error: {did_se:.2f}")
print(f"  95% Confidence interval: [{did_ci_low:.2f}, {did_ci_high:.2f}]")
print(f"  p-value: {did_pval:.3f}")
print(f"  Statistically significant at 5% level: {'Yes' if did_pval < 0.05 else 'No'}")
print(f"  Statistically significant at 10% level: {'Yes' if did_pval < 0.10 else 'No'}")

# Effect size
pre_control_mean = df[(df['switched_branch']==0) & (df['week']<15)]['service_score'].mean()
effect_pct = did_coef / pre_control_mean * 100
print(f"\nEFFECT SIZE:")
print(f"  As percentage of pre-switch control mean: {effect_pct:.1f}%")

# Power consideration
print(f"\nSAMPLE SIZE:")
print(f"  Total observations: {len(df)}")
print(f"  Treatment branches: {df['switched_branch'].sum() // 8}")  # 8 post weeks
print(f"  Control branches: {(1 - df['switched_branch']).sum() // 8}")

# Check if effect is meaningful in practical terms
print(f"\nPRACTICAL SIGNIFICANCE:")
print(f"  Average service score (all branches): {df['service_score'].mean():.1f}")
print(f"  Service score std deviation: {df['service_score'].std():.1f}")
print(f"  Effect as fraction of 1 SD: {did_coef / df['service_score'].std():.2f} SD")

# Check robustness - add controls
print("\n" + "=" * 70)
print("ROBUSTNESS CHECK: Adding staffing and visits controls")
print("=" * 70)
model_controls = ols('service_score ~ post + treated + did + staffing_index + visits_index', data=df).fit()
cov_controls = model_controls.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(f"DiD with controls: {cov_controls.params['did']:.2f} (SE: {cov_controls.bse['did']:.2f}, p={cov_controls.pvalues['did']:.3f})")

TITLE: Evaluation of New Branch Stock Rotation and Shelving Procedure

--------------------------------------------------------------------------------
BOTTOM LINE
--------------------------------------------------------------------------------

The new procedure shows a POSITIVE but NOT STATISTICALLY SIGNIFICANT effect on 
service scores. Branches that adopted the new procedure saw service scores 
increase by approximately 1.9 points more than branches that kept the old 
procedure. However, this difference could plausibly be due to random variation.

RECOMMENDATION: Consider a PHASED ROLLOUT with continued monitoring rather than 
immediate system-wide implementation. The evidence is suggestive but not 
conclusive.

--------------------------------------------------------------------------------
KEY FINDINGS
--------------------------------------------------------------------------------

1. TREATMENT EFFECT ESTIMATE
   • Difference-in-differences estimate: +1.86 service score points
   • 95% Confidence interval: [-0.78, +4.49]
   • p-value: 0.176 (not statistically significant at conventional levels)
   
   Interpretation: We estimate the new procedure improved service scores by 
   about 1.9 points, but we cannot rule out that the true effect is zero or 
   even slightly negative.

2. EFFECT SIZE (PRACTICAL SIGNIFICANCE)
   • 3.8% increase relative to baseline control branch scores
   • Equivalent to 0.15 standard deviations of service scores
   • For context: Average service score is ~50 points
   
   Interpretation: If real, the effect is modest but meaningful—roughly the 
   difference between a "good" and "very good" rating.

3. PARALLEL TRENDS VALIDATION
   • Pre-trend test: p = 0.806 (no evidence of differential pre-trends)
   • Event study shows treatment and control branches followed similar 
     trajectories before the switch
   • This supports the validity of our causal inference approach

4. ROBUSTNESS CHECKS
   • Adding controls for staffing and visitor volume: effect increases to 
     +2.10 points (p = 0.109)
   • Effect remains positive across specifications
   • No evidence the result is driven by confounding factors

5. SAMPLE CHARACTERISTICS
   • 34 branches total (20 switched, 14 did not)
   • 17 weeks of data (weeks 6-22)
   • Switch occurred at week 15
   • 578 total branch-week observations

--------------------------------------------------------------------------------
WHAT THE DATA SHOWS VISUALLY
--------------------------------------------------------------------------------

[See attached charts: service_score_trends.png and event_study.png]

• Trend chart: Both groups show upward trajectories, with treatment branches 
  maintaining their historical advantage and possibly widening the gap slightly 
  after the switch.
  
• Event study: All pre-switch coefficients are close to zero and statistically 
  indistinguishable from each other. Post-switch coefficients are uniformly 
  positive but individually not significant.

--------------------------------------------------------------------------------
LIMITATIONS
--------------------------------------------------------------------------------

1. STATISTICAL POWER: With only 34 branches, the study may be underpowered to 
   detect modest effects. A 1.9-point effect would require ~80 branches to 
   detect at 80% power.

2. SHORT TIME HORIZON: Only 8 weeks of post-switch data. Effects may evolve 
   over time (e.g., staff learning curve, initial disruption).

3. GENERALIZABILITY: Switching branches may differ systematically from 
   non-switching branches (though fixed effects and pre-trend tests address 
   this concern).

4. OUTCOME MEASURE: Service score is a composite index. We did not analyze 
   component measures or other outcomes (e.g., patron satisfaction, efficiency).

--------------------------------------------------------------------------------
NEXT STEPS
--------------------------------------------------------------------------------

RECOMMENDED ACTIONS:

1. CONTINUE MONITORING
   • Extend data collection for at least 12 more weeks
   • Track whether effects persist, grow, or diminish over time

2. EXPAND TO ADDITIONAL BRANCHES (PHASED ROLLOUT)
   • Randomly assign 10-15 additional branches to adopt the procedure
   • This would increase statistical power and provide stronger evidence
   • Maintain a control group for continued comparison

3. INVESTIGATE MECHANISMS
   • Survey staff at switched branches about implementation experience
   • Analyze whether effects vary by branch size, location, or baseline 
     performance
   • Examine component measures within the service score index

4. COST-BENEFIT ANALYSIS
   • Quantify implementation costs (training, time, materials)
   • Compare against the estimated benefit of ~2-point service score increase
   • Consider whether benefits justify system-wide rollout even with 
     uncertainty

5. DECISION TIMELINE
   • Re-evaluate in 3 months with additional data
   • If effects remain positive but inconclusive, consider full rollout with 
     ongoing evaluation
   • If effects diminish or turn negative, halt expansion
