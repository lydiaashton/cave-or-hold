import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Calculate means for the 4 groups
print("=== Service Score Means by Group ===")
group_means = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(group_means)

# Calculate DiD manually
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
pre_treated = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
post_treated = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n=== DiD Calculation ===")
print(f"Control (non-switched) - Pre: {pre_control:.2f}, Post: {post_control:.2f}, Change: {post_control - pre_control:.2f}")
print(f"Treated (switched) - Pre: {pre_treated:.2f}, Post: {post_treated:.2f}, Change: {post_treated - pre_treated:.2f}")
did_estimate = (post_treated - pre_treated) - (post_control - pre_control)
print(f"\nDiD Estimate: {did_estimate:.2f}")

# Run DiD regression
print("\n=== DiD Regression ===")
model = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()
print(model.summary())

# Check parallel trends - compare trends before the switch
print("\n=== Parallel Trends Check (Pre-period only) ===")
pre_period = df[df['after_switch'] == 0]
pre_model = ols('service_score ~ switched_branch + C(week)', data=pre_period).fit()
print(pre_model.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with controls and cluster-robust SEs
print("=== DiD Regression with Controls ===")
model_full = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', data=df).fit()
# Cluster-robust standard errors by branch
cov_cluster = model_full.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(cov_cluster.summary())

# Event study / dynamic DiD to check parallel trends and see effect over time
print("\n=== Event Study (Dynamic DiD) ===")
df['relative_week'] = df['week'] - 14  # week 14 is the last pre-switch week
df['relative_week'] = df['relative_week'].apply(lambda x: max(-8, min(8, x)))  # cap at -8 to 8

# Create dummies for each relative week interacted with treatment
event_study_vars = []
for w in range(-8, 9):
    if w != -1:  # omit week -1 as reference
        df[f'rel_week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(f'rel_week_{w}')

# Also include week fixed effects and switched_branch main effect
formula = 'service_score ~ switched_branch + C(week) + ' + ' + '.join(event_study_vars)
event_model = ols(formula, data=df).fit()
cov_cluster_event = event_model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Extract coefficients for the event study
event_coefs = []
event_se = []
event_weeks = []
for w in range(-8, 9):
    if w != -1:
        var_name = f'rel_week_{w}'
        if var_name in cov_cluster_event.params.index:
            event_coefs.append(cov_cluster_event.params[var_name])
            event_se.append(cov_cluster_event.bse[var_name])
            event_weeks.append(w)

print("Week (relative to switch) | Coefficient | Std Error")
print("-" * 50)
for w, coef, se in zip(event_weeks, event_coefs, event_se):
    print(f"{w:>+3d} (week {w+14:2d})              | {coef:>+8.2f}   | {se:8.2f}")

# Plot the event study
plt.figure(figsize=(10, 6))
plt.errorbar(event_weeks, event_coefs, yerr=[1.96*s for s in event_se], fmt='o', capsize=5)
plt.axhline(y=0, color='red', linestyle='--')
plt.axvline(x=0, color='gray', linestyle='-', alpha=0.5, label='Switch occurs')
plt.xlabel('Relative Week (0 = first week after switch)')
plt.ylabel('Effect on Service Score')
plt.title('Event Study: Effect of New Procedure on Service Score')
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")

# Summary statistics by branch type
print("\n=== Branch Summary ===")
branch_summary = df.groupby('switched_branch').agg({
    'branch_id': 'nunique',
    'service_score': ['mean', 'std'],
    'staffing_index': 'mean',
    'visits_index': 'mean'
}).round(2)
print(branch_summary)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with controls and cluster-robust SEs
print("=== DiD Regression with Controls (Cluster-Robust SEs) ===")
model_full = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', data=df).fit()
cov_cluster = model_full.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(cov_cluster.summary())

# Event study / dynamic DiD
print("\n=== Event Study (Dynamic DiD) ===")
df['relative_week'] = df['week'] - 14  # week 14 is the last pre-switch week

# Create dummies for each relative week interacted with treatment
event_data = []
for w in range(-8, 9):
    if w != -1:  # omit week -1 as reference
        df[f'w{w+10:02d}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_data.append((w, f'w{w+10:02d}'))

# Build formula
event_vars = [v for _, v in event_data]
formula = 'service_score ~ switched_branch + C(week) + ' + ' + '.join(event_vars)
event_model = ols(formula, data=df).fit()
cov_cluster_event = event_model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Extract coefficients
event_coefs = []
event_se = []
week_labels = []
for w, var_name in event_data:
    week_labels.append(w)
    event_coefs.append(cov_cluster_event.params[var_name])
    event_se.append(cov_cluster_event.bse[var_name])

print("Relative Week | Coefficient | Std Error | 95% CI")
print("-" * 55)
for w, coef, se in zip(week_labels, event_coefs, event_se):
    ci_low = coef - 1.96 * se
    ci_high = coef + 1.96 * se
    t_stat = coef / se if se > 0 else 0
    sig = "***" if abs(t_stat) > 2.58 else "**" if abs(t_stat) > 1.96 else "*" if abs(t_stat) > 1.64 else ""
    print(f"{w:>+11d}   | {coef:>+10.2f}  | {se:8.2f}  | [{ci_low:6.2f}, {ci_high:6.2f}] {sig}")

# Plot the event study
plt.figure(figsize=(12, 6))
plt.errorbar(week_labels, event_coefs, yerr=[1.96*s for s in event_se], fmt='o', capsize=5, markersize=8)
plt.axhline(y=0, color='red', linestyle='--', linewidth=2)
plt.axvline(x=0, color='gray', linestyle='-', alpha=0.5, label='Switch occurs (week 15)')
plt.xlabel('Relative Week (0 = first week after switch, -1 = last week before)')
plt.ylabel('Effect on Service Score (vs. control branches)')
plt.title('Event Study: Effect of New Procedure on Service Score\n(Week -1 is reference period)')
plt.grid(True, alpha=0.3)
plt.xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")

# Check pre-trends more formally (weeks -8 to -2)
print("\n=== Pre-Trend Coefficients (weeks -8 to -2) ===")
print("Pre-trend coefficients (should be close to 0 if parallel trends hold):")
for i, (w, var_name) in enumerate(event_data):
    if w < -1:
        coef = cov_cluster_event.params[var_name]
        se = cov_cluster_event.bse[var_name]
        t_stat = coef / se if se > 0 else 0
        pval = 2 * (1 - sm.stats.norm.cdf(abs(t_stat)))
        print(f"  Week {w:>+3d}: {coef:6.2f} (SE: {se:6.2f}, t: {t_stat:5.2f}, p: {pval:.3f})")

# Visualization of average scores over time by group
print("\n=== Average Service Score by Week and Group ===")
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack('switched_branch')
print(weekly_avg)

# Plot time series
plt.figure(figsize=(12, 6))
weeks = range(6, 23)
control_scores = [weekly_avg[0].get(w, np.nan) for w in weeks]
treated_scores = [weekly_avg[1].get(w, np.nan) for w in weeks]

plt.plot(weeks, control_scores, 'o-', label='Control (Old Procedure)', linewidth=2, markersize=8)
plt.plot(weeks, treated_scores, 's-', label='Treated (New Procedure)', linewidth=2, markersize=8)
plt.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch point')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Score Trends: Treatment vs. Control Groups')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('time_series.png', dpi=150)
print("\nTime series plot saved as 'time_series.png'")

# Calculate average post-period effect
print("\n=== Post-Switch Effects ===")
post_coefs = [coef for w, coef, se in zip(week_labels, event_coefs, event_se) if w >= 0]
post_count = len(post_coefs)
avg_post_effect = np.mean(post_coefs)
print(f"Number of post-switch weeks analyzed: {post_count}")
print(f"Average post-switch effect: {avg_post_effect:.2f} points")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt
from scipy import stats

df = pd.read_csv('branches.csv')
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Main DiD estimate with controls
print("=== Main DiD Regression (with controls, cluster-robust SEs) ===")
model_full = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', data=df).fit()
cov_cluster = model_full.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Access params by index position
param_names = model_full.params.index.tolist()
treated_post_idx = param_names.index('treated_post')

did_coef = cov_cluster.params[treated_post_idx]
did_se = cov_cluster.bse[treated_post_idx]
did_t = did_coef / did_se
did_pval = 2 * (1 - stats.norm.cdf(abs(did_t)))

print(f"DiD coefficient (treated_post): {did_coef:.2f}")
print(f"Standard error: {did_se:.2f}")
print(f"95% CI: [{did_coef - 1.96*did_se:.2f}, {did_coef + 1.96*did_se:.2f}]")
print(f"t-statistic: {did_t:.2f}")
print(f"p-value: {did_pval:.3f}")

# Summary of key findings
print("\n" + "="*60)
print("SUMMARY OF KEY FINDINGS")
print("="*60)

# Calculate basic DiD
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
pre_treated = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
post_treated = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
simple_did = (post_treated - pre_treated) - (post_control - pre_control)
