import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check how many branches in each group
branches_per_group = df.groupby('switched_branch')['branch_id'].nunique()
print("Branches per group:")
print(branches_per_group)
print(f"\nTotal branches: {branches_per_group.sum()}")

# Create treatment interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary: Average service scores by group and period
print("\n" + "="*60)
print("SERVICE SCORES BY GROUP AND PERIOD")
print("="*60)

summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(summary)

# Calculate the DiD manually
control_pre = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_post = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treatment_pre = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_post = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print("\n" + "-"*60)
print(f"Control group (old procedure):")
print(f"  Pre-switch average: {control_pre:.2f}")
print(f"  Post-switch average: {control_post:.2f}")
print(f"  Change: {control_post - control_pre:.2f}")
print(f"\nTreatment group (new procedure):")
print(f"  Pre-switch average: {treatment_pre:.2f}")
print(f"  Post-switch average: {treatment_post:.2f}")
print(f"  Change: {treatment_post - treatment_pre:.2f}")
print(f"\nDifference-in-Differences estimate:")
did = (treatment_post - treatment_pre) - (control_post - control_pre)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with cluster-robust standard errors
print("="*60)
print("DIFFERENCE-IN-DIFFERENCES REGRESSION ANALYSIS")
print("="*60)

# Model 1: Basic DiD
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 1: Basic DiD (with cluster-robust SEs by branch)")
print(model1.summary().tables[1])

# Model 2: DiD with control variables
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n" + "="*60)
print("Model 2: DiD with control variables (staffing_index, visits_index)")
print(model2.summary().tables[1])

# Model 3: With branch fixed effects
# Create branch dummies
df_with_fe = df.copy()
model3 = ols('service_score ~ after_switch + treatment + staffing_index + visits_index + C(branch_id)', data=df_with_fe).fit(cov_type='cluster', cov_kwds={'groups': df_with_fe['branch_id']})
print("\n" + "="*60)
print("Model 3: DiD with branch fixed effects")
print(model3.summary().tables[1].iloc[-5:])  # Just show the key coefficients

# Extract the treatment effect
treatment_effect = model3.params['treatment']
treatment_se = model3.bse['treatment']
treatment_pval = model3.pvalues['treatment']
print(f"\nTreatment effect (with branch FE): {treatment_effect:.3f}")
print(f"Standard error: {treatment_se:.3f}")
print(f"P-value: {treatment_pval:.4f}")
print(f"95% CI: [{treatment_effect - 1.96*treatment_se:.2f}, {treatment_effect + 1.96*treatment_se:.2f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Branch-level analysis: calculate each branch's before/after change
print("="*60)
print("BRANCH-LEVEL ANALYSIS")
print("="*60)

branch_stats = df.groupby(['branch_id', 'switched_branch']).agg({
    'service_score': ['mean', 'std', 'count'],
    'after_switch': 'mean'  # This tells us proportion of post-period observations
}).reset_index()

# Calculate before and after for each branch
branch_before = df[df['after_switch']==0].groupby('branch_id')['service_score'].mean()
branch_after = df[df['after_switch']==1].groupby('branch_id')['service_score'].mean()

branch_changes = pd.DataFrame({
    'before': branch_before,
    'after': branch_after
})
branch_changes['change'] = branch_changes['after'] - branch_changes['before']
branch_changes['switched'] = df.groupby('branch_id')['switched_branch'].first()

print("\nBranch-level changes summary:")
print(branch_changes.groupby('switched')['change'].describe())

# How many treatment branches improved vs declined?
treatment_changes = branch_changes[branch_changes['switched']==1]['change']
print(f"\nTreatment branches (n={len(treatment_changes)}):")
print(f"  Improved (positive change): {(treatment_changes > 0).sum()} ({100*(treatment_changes > 0).mean():.1f}%)")
print(f"  Declined (negative change): {(treatment_changes < 0).sum()} ({100*(treatment_changes < 0).mean():.1f}%)")
print(f"  Mean change: {treatment_changes.mean():.2f}")
print(f"  Median change: {treatment_changes.median():.2f}")

control_changes = branch_changes[branch_changes['switched']==0]['change']
print(f"\nControl branches (n={len(control_changes)}):")
print(f"  Improved (positive change): {(control_changes > 0).sum()} ({100*(control_changes > 0).mean():.1f}%)")
print(f"  Declined (negative change): {(control_changes < 0).sum()} ({100*(control_changes < 0).mean():.1f}%)")
print(f"  Mean change: {control_changes.mean():.2f}")
print(f"  Median change: {control_changes.median():.2f}")

# Effect size in context
avg_score = df['service_score'].mean()
treatment_effect = 2.50
print(f"\n" + "="*60)
print("EFFECT SIZE IN CONTEXT")
print("="*60)
print(f"Overall average service score: {avg_score:.1f}")
print(f"Estimated treatment effect: {treatment_effect:.2f} points")
print(f"Relative improvement: {100*treatment_effect/avg_score:.1f}%")

# Standardized effect size (Cohen's d approximation)
pooled_std = np.sqrt(((treatment_changes.std()**2 * len(treatment_changes)) + 
                      (control_changes.std()**2 * len(control_changes))) / 
                     (len(treatment_changes) + len(control_changes) - 2))
cohens_d = treatment_effect / pooled_std
print(f"Pooled standard deviation: {pooled_std:.2f}")
print(f"Standardized effect size (Cohen's d): {cohens_d:.2f}")
