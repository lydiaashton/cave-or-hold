import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic summary: mean service scores by group and period
print("=" * 60)
print("MEAN SERVICE SCORES BY GROUP AND PERIOD")
print("=" * 60)

pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(pivot)

# Calculate the DiD manually
print("\n" + "=" * 60)
print("DIFFERENCE-IN-DIFFERENCES CALCULATION")
print("=" * 60)

# Before period (after_switch=0)
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
before_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()

# After period (after_switch=1)
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
after_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"Control group (no switch): Before={before_control:.2f}, After={after_control:.2f}, Change={after_control-before_control:.2f}")
print(f"Treatment group (switched): Before={before_treatment:.2f}, After={after_treatment:.2f}, Change={after_treatment-before_treatment:.2f}")

did = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {did:.2f}")

# Check pre-trends: compare trends before week 15
print("\n" + "=" * 60)
print("PRE-TREND CHECK: MEAN SCORES BY WEEK")
print("=" * 60)

pre_trend = df[df['week'] < 15].groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
print(pre_trend)

# Run DiD regression
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model.summary())

# DiD with branch fixed effects
print("\n" + "=" * 60)
print("DID WITH BRANCH FIXED EFFECTS")
print("=" * 60)

# Create branch dummies
df_with_fe = df.copy()
branch_dummies = pd.get_dummies(df_with_fe['branch_id'], prefix='branch', drop_first=True)
df_fe = pd.concat([df_with_fe.reset_index(drop=True), branch_dummies.reset_index(drop=True)], axis=1)

# Regression with branch FE
fe_cols = [c for c in df_fe.columns if c.startswith('branch_')]
X = df_fe[['switched_branch', 'after_switch', 'treatment'] + fe_cols]
X = sm.add_constant(X)
y = df_fe['service_score']

model_fe = sm.OLS(y, X).fit()
print(f"Treatment effect coefficient: {model_fe.params['treatment']:.4f}")
print(f"Standard error: {model_fe.bse['treatment']:.4f}")
print(f"P-value: {model_fe.pvalues['treatment']:.4f}")
print(f"95% CI: [{model_fe.conf_int().loc['treatment', 0]:.4f}, {model_fe.conf_int().loc['treatment', 1]:.4f}]")

# Check how many branches switched
print("\n" + "=" * 60)
print("BRANCH COUNTS")
print("=" * 60)
print(f"Total branches: {df['branch_id'].nunique()}")
print(f"Switched branches: {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"Non-switched branches: {df[df['switched_branch']==0]['branch_id'].nunique()}")
