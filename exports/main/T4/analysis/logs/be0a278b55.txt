import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run the basic DiD regression
print("=== Basic Difference-in-Differences Regression ===")
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model1.summary())

# Run regression with branch fixed effects
print("\n=== DiD with Branch Fixed Effects ===")
df['branch_id'] = df['branch_id'].astype(str)
model2 = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit()
print(model2.summary())

# Run regression with branch and week fixed effects

import pandas as pd
import numpy as np

# Load the data
df = pd.read_csv('branches.csv')

# Summary statistics
print("=== Summary Statistics ===")
print(f"Total observations: {len(df)}")
print(f"Number of branches: {df['branch_id'].nunique()}")
print(f"Number of weeks: {df['week'].nunique()}")
print(f"Branches that switched: {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"Branches that didn't switch: {df[df['switched_branch']==0]['branch_id'].nunique()}")

# Overall service score stats
print(f"\nOverall mean service score: {df['service_score'].mean():.2f}")
print(f"Overall SD service score: {df['service_score'].std():.2f}")

# Effect size in terms of SD
overall_sd = df['service_score'].std()
did_estimate = 1.81
effect_size = did_estimate / overall_sd
print(f"\nEffect size (in SD units): {effect_size:.3f}")

# Calculate the percentage improvement relative to baseline
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
print(f"\nBaseline (treatment group before): {treatment_before:.2f}")
print(f"Percentage improvement: {(did_estimate / treatment_before) * 100:.1f}%")

# Check if there's any heterogeneity - do some branches benefit more?
print("\n=== Branch-level Changes ===")
branch_changes = df.groupby(['branch_id', 'switched_branch'])['service_score'].mean().unstack()
branch_changes['change'] = branch_changes[1] - branch_changes[0]
print(branch_changes.head(10))

# How many switched branches improved?
switched_branches = df[df['switched_branch']==1]['branch_id'].unique()
improved = 0
for b in switched_branches:
    before = df[(df['branch_id']==b) & (df['after_switch']==0)]['service_score'].mean()
    after = df[(df['branch_id']==b) & (df['after_switch']==1)]['service_score'].mean()
    if after > before:
        improved += 1
print(f"\nOf {len(switched_branches)} switched branches, {improved} improved ({improved/len(switched_branches)*100:.0f}%)")
