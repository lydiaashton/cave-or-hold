import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate average service scores by week for each group
weekly_trends = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Create the plot
fig, ax = plt.subplots(figsize=(10, 6))

# Plot control group (didn't switch)
control = weekly_trends[weekly_trends['switched_branch'] == 0]
ax.plot(control['week'], control['service_score'], marker='o', label='Control (No Switch)', linewidth=2, markersize=6)

# Plot treatment group (switched)
treated = weekly_trends[weekly_trends['switched_branch'] == 1]
ax.plot(treated['week'], treated['service_score'], marker='s', label='Treatment (Switched)', linewidth=2, markersize=6)

# Add vertical line at switch point (week 15)
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Score Trends: Treatment vs Control Groups', fontsize=14, fontweight='bold')
ax.legend(fontsize=11)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("Plot saved as service_score_trends.png")

# Also calculate the gap between groups over time
weekly_trends_pivot = weekly_trends.pivot(index='week', columns='switched_branch', values='service_score')
weekly_trends_pivot['gap'] = weekly_trends_pivot[1] - weekly_trends_pivot[0]
print("\nGap between treatment and control by week:")
print(weekly_trends_pivot)
