import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster
from scipy import stats

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main DiD model
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
cluster_cov = cov_cluster(model, df['branch_id'])
cluster_se = np.sqrt(np.diag(cluster_cov))

# Calculate 95% confidence interval for treatment effect
treatment_coef = model.params['treatment']
treatment_se = cluster_se[3]
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se

print("=== FINAL RESULTS FOR BOARD SUMMARY ===")
print(f"\nTreatment Effect (Difference-in-Differences): {treatment_coef:.2f} points")
print(f"Standard Error (clustered by branch): {treatment_se:.2f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
