import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Check how many branches switched vs didn't
branches = df[['branch_id', 'switched_branch']].drop_duplicates()
print("Branches by switch status:")
print(branches['switched_branch'].value_counts())
print(f"\nTotal branches: {len(branches)}")
print(f"Switched: {branches['switched_branch'].sum()} ({branches['switched_branch'].mean()*100:.1f}%)")

# Check data completeness - do all branches have all weeks?
weeks_per_branch = df.groupby('branch_id')['week'].count()
print(f"\nWeeks per branch - min: {weeks_per_branch.min()}, max: {weeks_per_branch.max()}, mean: {weeks_per_branch.mean():.1f}")

# Check for missing weeks
print("\nMissing weeks per branch (should be 17 weeks total):")
missing = 17 - weeks_per_branch
print(f"Branches with missing weeks: {(missing > 0).sum()}")

# Average service score by group and period
df['period'] = df['after_switch'].map({0: 'Before', 1: 'After'})
df['group'] = df['switched_branch'].map({0: 'Control (Old)', 1: 'Treatment (New)'})

print("\n\nAverage Service Score by Group and Period:")
pivot = df.groupby(['group', 'period'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the difference-in-differences manually
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n\nDiD Calculation:")
print(f"Control (Old procedure): Before={before_control:.2f}, After={after_control:.2f}, Change={after_control - before_control:.2f}")
print(f"Treatment (New procedure): Before={before_treatment:.2f}, After={after_treatment:.2f}, Change={after_treatment - before_treatment:.2f}")

import pandas as pd
import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# Basic DiD: service_score ~ switched_branch * after_switch
model_did = smf.ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("Basic DiD Regression:")
print(model_did.summary())

# DiD with branch fixed effects
model_fe = smf.ols('service_score ~ switched_branch * after_switch + C(branch_id)', data=df).fit()
print("\n\nDiD with Branch Fixed Effects:")
print(model_fe.summary())

# DiD with branch fixed effects and controls
model_full = smf.ols('service_score ~ switched_branch * after_switch + C(branch_id) + staffing_index + visits_index', data=df).fit()
print("\n\nDiD with Branch FE and Controls:")
print(model_full.summary())

# Get the coefficient and confidence interval for the interaction term
