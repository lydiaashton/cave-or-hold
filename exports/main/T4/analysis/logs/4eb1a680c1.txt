# Summary: Average service scores by group and period
print("=== Average Service Scores ===")

before = df[df['after_switch'] == 0]
print(before.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

print("\nAfter switch (weeks 15-22):")
after = df[df['after_switch'] == 1]

print(f"DiD estimate: {did_estimate:.4f}")

# Run DiD regression
print("\n=== DiD Regression (Simple) ===")
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study - create week dummies interacted with treatment
df = df.copy()
df['week_dummy'] = df['week'].astype(str)

# Create event time relative to switch (week 15 = 0)
df['event_time'] = df['week'] - 15

# Create event study dummies (omitting week 14 as reference)
for week in range(6, 23):
    if week != 14:  # Reference period
        df[f'week_{week}'] = (df['week'] == week).astype(int)
        df[f'week_{week}_treat'] = df[f'week_{week}'] * df['switched_branch']

# Run event study regression
event_vars = [f'week_{week}' for week in range(6, 23) if week != 14]
event_treat_vars = [f'week_{week}_treat' for week in range(6, 23) if week != 14]

formula = 'service_score ~ switched_branch + ' + ' + '.join(event_vars) + ' + ' + ' + '.join(event_treat_vars)
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract treatment effects by week
print("=== Event Study Results (Treatment Effects by Week) ===")
print("(Week 14 is the reference period)")
print()

results = []
for week in range(6, 23):
    if week == 14:
        results.append((week, 0, 0, 0, 0))  # Reference
    else:
        coef_name = f'week_{week}_treat'
        if coef_name in model_event.params:
            coef = model_event.params[coef_name]
            se = model_event.bse[coef_name]
            ci_low = coef - 1.96 * se
            ci_high = coef + 1.96 * se
            results.append((week, coef, se, ci_low, ci_high))

event_df = pd.DataFrame(results, columns=['week', 'coef', 'se', 'ci_low', 'ci_high'])
event_df['period'] = event_df['week'].apply(lambda x: 'Pre' if x < 15 else ('Switch' if x == 15 else 'Post'))
print(event_df.to_string(index=False))

# Calculate average pre and post effects
pre_effects = event_df[event_df['week'] < 15]['coef'].mean()
post_effects = event_df[event_df['week'] >= 15]['coef'].mean()
print(f"\nAverage pre-switch effect (weeks 6-14): {pre_effects:.3f}")
print(f"Average post-switch effect (weeks 15-22): {post_effects:.3f}")

# Create event study plot
fig, ax = plt.subplots(figsize=(12, 6))

ax.errorbar(event_df['week'], event_df['coef'], yerr=1.96*event_df['se'], 
            fmt='o', capsize=5, label='Treatment Effect', markersize=8)
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (Week 15)')
ax.axvspan(15, 22, alpha=0.1, color='green', label='Post-Switch Period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Treatment Effect (vs Week 14)', fontsize=12)
ax.set_title('Event Study: Treatment Effects Over Time', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=10)
ax.grid(True, alpha=0.3, axis='y')
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Better event study: interact treatment with each week
# This estimates the treatment effect for each week relative to the control group

# Create week dummies and interactions
for week in range(6, 23):
    df[f'is_week_{week}'] = (df['week'] == week).astype(int)
    df[f'treat_week_{week}'] = df['switched_branch'] * df[f'is_week_{week}']

# Run regression with week fixed effects and treatment-by-week interactions
# Omit week 6 as reference for identification
week_dummies = [f'is_week_{w}' for w in range(7, 23)]
treat_weeks = [f'treat_week_{w}' for w in range(6, 23)]

formula = 'service_score ~ switched_branch + ' + ' + '.join(week_dummies) + ' + ' + ' + '.join(treat_weeks)
model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract treatment effects
print("=== Event Study: Treatment Effect by Week ===")
print("(Effects relative to control group in each week)")
print()

results = []
for week in range(6, 23):
    coef_name = f'treat_week_{week}'
    coef = model.params[coef_name]
    se = model.bse[coef_name]
    ci_low = coef - 1.96 * se
    ci_high = coef + 1.96 * se
    period = 'Pre' if week < 15 else ('Switch' if week == 15 else 'Post')
    results.append((week, coef, se, ci_low, ci_high, period))

event_df = pd.DataFrame(results, columns=['week', 'coef', 'se', 'ci_low', 'ci_high', 'period'])
print(event_df.to_string(index=False))

# Pre vs Post comparison
pre_avg = event_df[event_df['period'] == 'Pre']['coef'].mean()
post_avg = event_df[event_df['period'] == 'Post']['coef'].mean()
print(f"\nAverage pre-switch treatment effect: {pre_avg:.3f}")

            fmt='o', capsize=5, color=colors, markersize=8, linewidth=2)
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch (Week 15)')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Treatment Effect (Treatment - Control)', fontsize=12)
ax.set_title('Event Study: Weekly Treatment Effects', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=10)

ax.set_xticks(range(6, 23))

plt.tight_layout()

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create treatment variable
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate weekly averages by group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_avg.columns = ['Control', 'Treatment']
weekly_avg['Diff'] = weekly_avg['Treatment'] - weekly_avg['Control']

# Create a clean visualization
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Left panel: Trends over time
ax1 = axes[0]
ax1.plot(weekly_avg.index, weekly_avg['Control'], 'o-', label='Control (Old Procedure)', 
         linewidth=2.5, markersize=7, color='#2E86AB')
ax1.plot(weekly_avg.index, weekly_avg['Treatment'], 's-', label='Treatment (New Procedure)', 
         linewidth=2.5, markersize=7, color='#A23B72')
ax1.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch Point (Week 15)')
ax1.axvspan(15, 22.5, alpha=0.15, color='green', label='Post-Switch Period')
ax1.set_xlabel('Week', fontsize=12, fontweight='bold')
ax1.set_ylabel('Average Service Score', fontsize=12, fontweight='bold')
ax1.set_title('Service Score Trends by Group', fontsize=14, fontweight='bold')
ax1.legend(loc='upper left', fontsize=10)
ax1.grid(True, alpha=0.3)
ax1.set_xticks(range(6, 23))

# Right panel: Difference between groups over time
ax2 = axes[1]
colors = ['steelblue' if w < 15 else 'forestgreen' for w in weekly_avg.index]
ax2.bar(weekly_avg.index, weekly_avg['Diff'], color=colors, edgecolor='black', linewidth=0.5)
ax2.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax2.axvline(x=14.5, color='gray', linestyle='--', linewidth=2)
ax2.set_xlabel('Week', fontsize=12, fontweight='bold')
ax2.set_ylabel('Treatment - Control (points)', fontsize=12, fontweight='bold')
ax2.set_title('Gap Between Treatment and Control Groups', fontsize=14, fontweight='bold')
ax2.grid(True, alpha=0.3, axis='y')
ax2.set_xticks(range(6, 23))

# Add average lines
pre_avg_diff = weekly_avg[weekly_avg.index < 15]['Diff'].mean()
post_avg_diff = weekly_avg[weekly_avg.index >= 15]['Diff'].mean()
ax2.axhline(y=pre_avg_diff, color='steelblue', linestyle='--', linewidth=2, alpha=0.7, label=f'Pre Avg: {pre_avg_diff:.1f}')
ax2.axhline(y=post_avg_diff, color='forestgreen', linestyle='--', linewidth=2, alpha=0.7, label=f'Post Avg: {post_avg_diff:.1f}')
ax2.legend(loc='upper left', fontsize=10)

plt.tight_layout()
plt.savefig('analysis_summary.png', dpi=150, bbox_inches='tight')
print("Summary plot saved as 'analysis_summary.png'")

# Print key statistics for the report
print("\n" + "="*60)
print("KEY STATISTICS FOR BOARD SUMMARY")
print("="*60)

n_total = df['branch_id'].nunique()

print(f"\nTotal branches: {n_total}")
print(f"  - Treatment (new procedure): {n_treatment}")
print(f"  - Control (old procedure): {n_control}")

print(f"\nStudy period: Weeks 6-22 (17 weeks)")
print(f"  - Pre-switch: Weeks 6-14 (9 weeks)")
print(f"  - Post-switch: Weeks 15-22 (8 weeks)")

# Service score improvements
before_control = df[(df['after_switch'] == 0) & (df['switched_branch'] == 0)]['service_score'].mean()
before_treatment = df[(df['after_switch'] == 0) & (df['switched_branch'] == 1)]['service_score'].mean()
after_control = df[(df['after_switch'] == 1) & (df['switched_branch'] == 0)]['service_score'].mean()
