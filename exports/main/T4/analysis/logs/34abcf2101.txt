import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimator

df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# DiD with controls (staffing and visits)
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\n=== DiD Regression with Controls ===")
print(model_controls.summary())

# Cluster-robust standard errors by branch
model_clustered = model_controls.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print("\n=== DiD with Clustered SE (by branch) ===")
print(model_clustered.summary())

# Calculate confidence interval for the treatment effect
import statsmodels.api as sm
from statsmodels.formula.api import ols

df['treatment'] = df['switched_branch'] * df['after_switch']
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()

# Calculate percentage improvement relative to baseline

pct_improvement = (treatment_coef / baseline) * 100
print(f"\nBaseline (switched branches, before): {baseline:.2f}")
print(f"Percentage improvement: {pct_improvement:.1f}%")

import pandas as pd

import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run the model with clustered SE
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
model_clustered = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Extract treatment effect (index 3 in the array)
treatment_coef = model_clustered.params[3]
treatment_se = model_clustered.bse[3]
treatment_pvalue = model_clustered.pvalues[3]
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se

print("=== Treatment Effect Summary ===")
print(f"Estimated effect: {treatment_coef:.2f} points")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {treatment_pvalue:.4f}")
print(f"Statistically significant: {'Yes' if treatment_pvalue < 0.05 else 'No'}")

# Calculate percentage improvement relative to baseline
baseline = df[(df['after_switch'] == 0) & (df['switched_branch'] == 1)]['service_score'].mean()
pct_improvement = (treatment_coef / baseline) * 100
print(f"\nBaseline (switched branches, before): {baseline:.2f}")
print(f"Percentage improvement: {pct_improvement:.1f}%")

# Create visualization
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
control_scores = weekly_avg.iloc[:, 0].values
treatment_scores = weekly_avg.iloc[:, 1].values
weeks = weekly_avg.index.values

plt.figure(figsize=(12, 6))
plt.plot(weeks, control_scores, 'o-', label='Control (No Switch)', linewidth=2, markersize=8)
plt.plot(weeks, treatment_scores, 's-', label='Treatment (Switched)', linewidth=2, markersize=8)

plt.xlabel('Week', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Scores: Treatment vs Control Groups\nBefore and After Procedure Change', fontsize=14)
plt.legend(fontsize=11)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()

import pandas as pd
import numpy as np

# Load the data
df = pd.read_csv('branches.csv')

# Check effect by post-switch period (early vs late)
df['treatment'] = df['switched_branch'] * df['after_switch']
df['weeks_after'] = df['week'] - 14  # weeks since switch
df['weeks_after'] = df['weeks_after'].clip(lower=0)  # 0 for pre-period

# Split post-period into early (weeks 15-18) and late (weeks 19-22)
df['post_early'] = ((df['week'] >= 15) & (df['week'] <= 18)).astype(int)
df['post_late'] = ((df['week'] >= 19) & (df['week'] <= 22)).astype(int)
df['treat_early'] = df['switched_branch'] * df['post_early']
df['treat_late'] = df['switched_branch'] * df['post_late']

import statsmodels.api as sm
from statsmodels.formula.api import ols

# Check if effect grows over time
model_time = ols('service_score ~ switched_branch + post_early + post_late + treat_early + treat_late + staffing_index + visits_index', data=df).fit()
model_time_clustered = model_time.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("=== Effect by Post-Switch Period ===")
print(f"Early post-period effect (weeks 15-18): {model_time_clustered.params[3]:.2f} (p={model_time_clustered.pvalues[3]:.3f})")
print(f"Late post-period effect (weeks 19-22): {model_time_clustered.params[4]:.2f} (p={model_time_clustered.pvalues[4]:.3f})")

# Check distribution of branch-level effects
# Calculate average before and after for each branch
branch_avg = df.groupby(['branch_id', 'switched_branch']).apply(
    lambda x: pd.Series({
        'before': x[x['after_switch']==0]['service_score'].mean(),
        'after': x[x['after_switch']==1]['service_score'].mean(),
        'change': x[x['after_switch']==1]['service_score'].mean() - x[x['after_switch']==0]['service_score'].mean()
    })
).reset_index()
