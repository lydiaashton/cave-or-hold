import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['did_interaction'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + did_interaction', data=df).fit()
print("Basic DiD Regression:")
print(model1.summary())

# DiD with branch fixed effects
model2 = ols('service_score ~ switched_branch + after_switch + did_interaction + C(branch_id)', data=df).fit()

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['did_interaction'] = df['switched_branch'] * df['after_switch']

# Final model with controls
model = ols('service_score ~ after_switch + did_interaction + C(branch_id) + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract key statistics
did_coef = model.params['did_interaction']
did_se = model.bse['did_interaction']
did_pvalue = model.pvalues['did_interaction']
did_ci_lower = did_coef - 1.96 * did_se
did_ci_upper = did_coef + 1.96 * did_se

print("="*60)
print("FINAL ANALYSIS SUMMARY")
print("="*60)
