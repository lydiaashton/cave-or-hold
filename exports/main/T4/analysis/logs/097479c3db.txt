import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model 1: Simple DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Model 1: Simple DiD (no controls)")
print(model1.summary())

# Model 2: DiD with control variables
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\n\nModel 2: DiD with controls")
print(model2.summary())

# Model 3: DiD with branch fixed effects
# Create dummy variables for branches
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Build formula with branch fixed effects
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch')]
formula = 'service_score ~ after_switch + treatment + staffing_index + visits_index + ' + ' + '.join(branch_cols)
model3 = ols(formula, data=df_with_dummies).fit()
print("\n\nModel 3: DiD with branch fixed effects and controls")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks since switch, with week 14 as reference)
# Week 15 is the first week after switch, so relative_week = week - 14
df['relative_week'] = df['week'] - 14

# Create interaction terms for each week with treatment
# We'll use week 14 (relative_week=0) as the reference period
for w in range(-8, 9):  # weeks -8 to 8 relative to switch (week 14 is 0)
    if w != 0:  # skip reference week
        df[f'treat_week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression with week 14 as reference
week_cols = [f'treat_week_{w}' for w in range(-8, 9) if w != 0]
formula = 'service_score ~ ' + ' + '.join(week_cols) + ' + staffing_index + visits_index'
# Add branch fixed effects
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_es = pd.concat([df, branch_dummies], axis=1)
branch_cols = [c for c in df_es.columns if c.startswith('branch')]
formula = 'service_score ~ ' + ' + '.join(week_cols) + ' + staffing_index + visits_index + ' + ' + '.join(branch_cols)

model_es = ols(formula, data=df_es).fit()

# Extract coefficients and confidence intervals
results = []
for w in range(-8, 9):
    if w == 0:
        results.append({'relative_week': w, 'coef': 0, 'ci_lower': 0, 'ci_upper': 0})
    else:
        col = f'treat_week_{w}'
        coef = model_es.params[col]
        ci = model_es.conf_int().loc[col]
        results.append({'relative_week': w, 'coef': coef, 'ci_lower': ci[0], 'ci_upper': ci[1]})

results_df = pd.DataFrame(results)
print("Event Study Results (relative to week 14):")
print(results_df)

# Check if pre-treatment coefficients are jointly zero
pre_treat_cols = [f'treat_week_{w}' for w in range(-8, 0)]
# F-test for joint significance
hypotheses = [f"{col} = 0" for col in pre_treat_cols]
f_test = model_es.f_test(hypotheses)
print(f"\nF-test for pre-treatment coefficients jointly zero: F={f_test.fvalue:.2f}, p={f_test.pvalue:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Simpler approach: create relative week and run regression without so many dummies
# Let's just check pre vs post trends more simply

# Create period indicators
df['pre_period'] = (df['week'] < 15).astype(int)
df['post_period'] = (df['week'] >= 15).astype(int)

# Check means by period and group
print("Service scores by group and period:")
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='pre_period', 
                       aggfunc=['mean', 'std', 'count'])
print(pivot)

# Run DiD with clustered standard errors at branch level
df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple model
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()

# Get cluster-robust standard errors
from statsmodels.stats.sandwich_covariance import cov_cluster
cov_clustered = cov_cluster(model, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))
t_stats = model.params / se_clustered
from scipy import stats
p_values_clustered = 2 * (1 - stats.t.cdf(np.abs(t_stats), len(df['branch_id']) - 1))

print("\n\nCoefficients with cluster-robust SE (clustered by branch):")
results_df = pd.DataFrame({
    'coef': model.params,
    'clustered_se': se_clustered,
    't_stat': t_stats,
    'p_value': p_values_clustered
})
print(results_df)

# Effect size relative to baseline
baseline = df[df['after_switch']==0]['service_score'].mean()
treatment_effect = model.params['treatment']
print(f"\n\nBaseline service score (pre-switch avg): {baseline:.2f}")
print(f"Treatment effect: {treatment_effect:.2f} points")
print(f"Treatment effect as % of baseline: {treatment_effect/baseline*100:.1f}%")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check if effect varies by branch characteristics
# First, let's see the effect for each switching branch
df['treatment'] = df['switched_branch'] * df['after_switch']

# Get branch-level effects by comparing pre/post for switching branches
branch_effects = []
for branch_id in df[df['switched_branch']==1]['branch_id'].unique():
    branch_data = df[df['branch_id']==branch_id]
    pre_mean = branch_data[branch_data['week']<15]['service_score'].mean()
    post_mean = branch_data[branch_data['week']>=15]['service_score'].mean()
    branch_effect = post_mean - pre_mean
    branch_effects.append({'branch_id': branch_id, 'pre': pre_mean, 'post': post_mean, 'change': branch_effect})

branch_effects_df = pd.DataFrame(branch_effects)
print("Branch-level changes for switching branches:")
print(branch_effects_df.describe())

# Compare to control branches
control_changes = []
for branch_id in df[df['switched_branch']==0]['branch_id'].unique():
    branch_data = df[df['branch_id']==branch_id]
    pre_mean = branch_data[branch_data['week']<15]['service_score'].mean()
    post_mean = branch_data[branch_data['week']>=15]['service_score'].mean()
    change = post_mean - pre_mean
    control_changes.append({'branch_id': branch_id, 'pre': pre_mean, 'post': post_mean, 'change': change})

control_df = pd.DataFrame(control_changes)
print("\nBranch-level changes for control branches:")
print(control_df.describe())

# Test if the average change differs between groups
from scipy import stats
t_stat, p_val = stats.ttest_ind(branch_effects_df['change'], control_df['change'])
print(f"\nT-test comparing changes: t={t_stat:.2f}, p={p_val:.4f}")

# Check consistency over post period
print("\n\nTreatment effect by post-switch week:")
post_weekly = df[df['week']>=15].groupby('week').apply(
    lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()
)
print(post_weekly)
