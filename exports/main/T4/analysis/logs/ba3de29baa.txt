# After period: after_switch = 1

before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()

after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n=== Difference-in-Differences Calculation ===")
print(f"Control (no switch) - Before: {before_control:.2f}")

print(f"  Control change: {after_control - before_control:.2f}")
print()
print(f"Treatment (switched) - Before: {before_treatment:.2f}")
print(f"Treatment (switched) - After: {after_treatment:.2f}")
print(f"  Treatment change: {after_treatment - before_treatment:.2f}")
print()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run DiD regression with standard errors
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is our treatment effect

df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# Now with branch fixed effects (more robust)
# This controls for time-invariant branch characteristics
print("\n=== DiD with Branch Fixed Effects ===")
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Build formula with branch dummies
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_')]
formula = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(branch_cols)
model_fe = ols(formula, data=df_with_dummies).fit()
print(model_fe.summary())

# Cluster-robust standard errors at branch level
print("\n=== DiD with Branch Fixed Effects (Cluster-Robust SEs) ===")
model_fe_cluster = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
# Use cluster-robust standard errors
cov_type = 'cluster'
cov_kwds = {'groups': df['branch_id']}
model_fe_cluster = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type=cov_type, cov_kwds=cov_kwds)
print(model_fe_cluster.summary())
