treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

# Control group (switched=0)

control_diff = control_after - control_before

# DiD estimate
did_estimate = treatment_diff - control_diff

print(f"Control group - Before: {control_before:.2f}, After: {control_after:.2f}, Change: {control_diff:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Add branch fixed effects model for robustness
# This controls for time-invariant branch characteristics
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [col for col in df_with_dummies.columns if col.startswith('branch_')]
formula = 'service_score ~ switched_branch * after_switch + staffing_index + visits_index + ' + ' + '.join(branch_cols)

# This might be computationally intensive, let's use a different approach
# Use the linearmodels package for fixed effects, or just use statsmodels with absorbed fixed effects

# Actually, let's do a simpler approach: branch fixed effects regression
# We can use the within transformation or just include branch as a categorical variable

# Let's try with C(branch_id) for fixed effects
model_fe = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== DiD with Branch Fixed Effects ===")
print(model_fe.summary().tables[1])

# Also check post-period only to see sustained effect
post_period = df[df['week'] >= 15]
