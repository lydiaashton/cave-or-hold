import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

# === Difference-in-Differences Summary Table ===
print("=== DiD Summary Table ===")
before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

# Average service scores by group and period
switched_before = before[before['switched_branch'] == 1]['service_score'].mean()
switched_after = after[after['switched_branch'] == 1]['service_score'].mean()
control_before = before[before['switched_branch'] == 0]['service_score'].mean()
control_after = after[after['switched_branch'] == 0]['service_score'].mean()

print(f"Switched branches - Before: {switched_before:.2f}, After: {switched_after:.2f}, Change: {switched_after - switched_before:.2f}")
print(f"Control branches - Before: {control_before:.2f}, After: {control_after:.2f}, Change: {control_after - control_before:.2f}")
print(f"\nDifference-in-Differences: {(switched_after - switched_before) - (control_after - control_before):.2f}")

# === Regression Analysis ===
print("\n=== Regression Analysis (DiD) ===")

# Simple DiD regression
model = ols('service_score ~ treated + post + treated_post', data=df).fit()
print(model.summary())

# DiD with branch fixed effects
print("\n=== DiD with Branch Fixed Effects ===")
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ treated_post + C(branch_id_str)', data=df).fit()
print(model_fe.summary())

# DiD with branch and week fixed effects
print("\n=== DiD with Branch and Week Fixed Effects ===")
model_fe_week = ols('service_score ~ treated_post + C(branch_id_str) + C(week)', data=df).fit()
print(model_fe_week.summary())

# Check if staffing and visits explain the effect
print("\n=== DiD with Controls ===")
model_controls = ols('service_score ~ treated_post + C(branch_id_str) + C(week) + staffing_index + visits_index', data=df).fit()
print(model_controls.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treated_post'] = df['switched_branch'] * df['after_switch']

# === Key Regression Results ===
print("=== Key Coefficients Summary ===")

# Simple DiD
model_simple = ols('service_score ~ treated + post + treated_post', data=df).fit()
print(f"\nSimple DiD - treated_post coefficient: {model_simple.params['treated_post']:.3f} (p={model_simple.pvalues['treated_post']:.4f})")

# With branch FE
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ treated_post + C(branch_id_str)', data=df).fit()
print(f"Branch FE - treated_post coefficient: {model_fe.params['treated_post']:.3f} (p={model_fe.pvalues['treated_post']:.4f})")

# With branch and week FE
model_fe_week = ols('service_score ~ treated_post + C(branch_id_str) + C(week)', data=df).fit()
print(f"Branch + Week FE - treated_post coefficient: {model_fe_week.params['treated_post']:.3f} (p={model_fe_week.pvalues['treated_post']:.4f})")

# With controls
model_controls = ols('service_score ~ treated_post + C(branch_id_str) + C(week) + staffing_index + visits_index', data=df).fit()
print(f"With Controls - treated_post coefficient: {model_controls.params['treated_post']:.3f} (p={model_controls.pvalues['treated_post']:.4f})")

# Get confidence interval
ci = model_fe_week.conf_int().loc['treated_post']
print(f"\n95% CI for main estimate: [{ci[0]:.3f}, {ci[1]:.3f}]")

# === Check Parallel Trends ===
print("\n=== Parallel Trends Check (Pre-treatment trends) ===")
pre_data = df[df['week'] < 15]

# Average service score by week and treatment group
pre_trends = pre_data.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
pre_trends.columns = ['Control', 'Switched']
pre_trends['Diff'] = pre_trends['Switched'] - pre_trends['Control']
print(pre_trends)

# === Plot trends ===
fig, ax = plt.subplots(figsize=(10, 6))

# Plot average service scores by week for both groups
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

df['branch_id_str'] = df['branch_id'].astype(str)

# === Key Regression Results ===
print("=== Key Coefficients Summary ===")

# Simple DiD
model_simple = ols('service_score ~ treated + post + treated_post', data=df).fit()
print(f"\nSimple DiD - treated_post coefficient: {model_simple.params['treated_post']:.3f} (p={model_simple.pvalues['treated_post']:.4f})")

# With branch FE
model_fe = ols('service_score ~ treated_post + C(branch_id_str)', data=df).fit()
print(f"Branch FE - treated_post coefficient: {model_fe.params['treated_post']:.3f} (p={model_fe.pvalues['treated_post']:.4f})")

# With branch and week FE
model_fe_week = ols('service_score ~ treated_post + C(branch_id_str) + C(week)', data=df).fit()
print(f"Branch + Week FE - treated_post coefficient: {model_fe_week.params['treated_post']:.3f} (p={model_fe_week.pvalues['treated_post']:.4f})")

import pandas as pd

# Create a DiD visualization
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Left panel: Before/After comparison
before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

groups = ['Control\n(No Switch)', 'Switched\nBranches']
before_means = [before[before['switched_branch'] == 0]['service_score'].mean(),
                before[before['switched_branch'] == 1]['service_score'].mean()]
after_means = [after[after['switched_branch'] == 0]['service_score'].mean(),
               after[after['switched_branch'] == 1]['service_score'].mean()]

x = np.arange(len(groups))
width = 0.35

axes[0].bar(x - width/2, before_means, width, label='Before (Weeks 6-14)', color='#1f77b4', alpha=0.8)
axes[0].bar(x + width/2, after_means, width, label='After (Weeks 15-22)', color='#2ca02c', alpha=0.8)
axes[0].set_ylabel('Average Service Score')
axes[0].set_title('Service Scores: Before vs After Switch', fontsize=12, fontweight='bold')
axes[0].set_xticks(x)
axes[0].set_xticklabels(groups)
axes[0].legend()
axes[0].grid(True, alpha=0.3, axis='y')

# Add value labels
for i, (b, a) in enumerate(zip(before_means, after_means)):
    axes[0].text(i - width/2, b + 0.5, f'{b:.1f}', ha='center', fontsize=10)
    axes[0].text(i + width/2, a + 0.5, f'{a:.1f}', ha='center', fontsize=10)

# Right panel: Difference-in-Differences illustration
# Show the changes
control_change = after_means[0] - before_means[0]
switched_change = after_means[1] - before_means[1]
did_effect = switched_change - control_change

axes[1].bar(['Control\nChange', 'Switched\nChange', 'DiD\nEffect'], 
            [control_change, switched_change, did_effect],
            color=['#1f77b4', '#2ca02c', '#d62728'], alpha=0.8)
axes[1].set_ylabel('Change in Service Score')
axes[1].set_title('Difference-in-Differences Estimate', fontsize=12, fontweight='bold')
axes[1].axhline(y=0, color='black', linewidth=0.5)
axes[1].grid(True, alpha=0.3, axis='y')

# Add value labels
for i, val in enumerate([control_change, switched_change, did_effect]):
    axes[1].text(i, val + 0.2 if val > 0 else val - 0.5, f'{val:.2f}', ha='center', fontsize=11, fontweight='bold')

plt.tight_layout()
plt.savefig('did_visualization.png', dpi=150)
print("Saved DiD visualization to did_visualization.png")

print(f"Number of branches: {df['branch_id'].nunique()}")
