import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Create treatment group indicator (interaction term for DiD)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary table: mean service scores by group and period
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("Service Score by Group and Period:")
print(summary)
print()

# Calculate DiD manually
# Control group (switched_branch=0)
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
control_change = control_after - control_before

# Treatment group (switched_branch=1)
treat_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treat_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
treat_change = treat_after - treat_before

# Difference-in-Differences
did = treat_change - control_change

print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES CALCULATION")
print("=" * 60)
print(f"\nControl Group (no switch):")
print(f"  Before (weeks 6-14): {control_before:.2f}")
print(f"  After (weeks 15-22): {control_after:.2f}")
print(f"  Change: {control_change:.2f}")
print(f"\nTreatment Group (switched):")
print(f"  Before (weeks 6-14): {treat_before:.2f}")
print(f"  After (weeks 15-22): {treat_after:.2f}")
print(f"  Change: {treat_change:.2f}")
print(f"\nDifference-in-Differences Estimate: {did:.2f}")
print()

# Check pre-trends: compare means before the switch
print("=" * 60)
print("PRE-TREND CHECK (weeks 6-14)")
print("=" * 60)
pre_trends = df[df['after_switch'] == 0].groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
print("\nWeekly means before switch:")
print(pre_trends.pivot(index='week', columns='switched_branch', values='service_score').round(2))

# Test if pre-treatment means are different
pre_treat = df[(df['after_switch'] == 0) & (df['switched_branch'] == 1)]['service_score']
pre_control = df[(df['after_switch'] == 0) & (df['switched_branch'] == 0)]['service_score']
from scipy import stats
t_stat, p_val = stats.ttest_ind(pre_treat, pre_control)
print(f"\nPre-treatment difference test: t={t_stat:.2f}, p={p_val:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression
print("=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Model 1: Basic DiD
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\n" + "=" * 60)
print("Model 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Model 3: With branch fixed effects (using dummy variables)
# This controls for time-invariant branch characteristics
print("\n" + "=" * 60)
print("Model 3: DiD with Branch Fixed Effects")
print("=" * 60)

# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df, branch_dummies], axis=1)

# Build formula with branch FEs
branch_cols = [col for col in df_with_fe.columns if col.startswith('branch_')]
formula = f'service_score ~ after_switch + treatment + staffing_index + visits_index + {" + ".join(branch_cols)}'
model3 = ols(formula, data=df_with_fe).fit()

# Extract key coefficients
print(f"\nTreatment effect (with branch FE): {model3.params['treatment']:.3f}")
print(f"Standard error: {model3.bse['treatment']:.3f}")
print(f"95% CI: [{model3.conf_int().loc['treatment'][0]:.3f}, {model3.conf_int().loc['treatment'][1]:.3f}]")
print(f"p-value: {model3.pvalues['treatment']:.4f}")

# Cluster-robust standard errors by branch
print("\n" + "=" * 60)
print("Model 4: Clustered SEs by branch")
print("=" * 60)
model4 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model4.summary().tables[1])
