import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# First, let's look at the 2x2 table for DiD
print("=== Difference-in-Differences Summary ===\n")

# Calculate means for each group
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("Control Group (did not switch):")
print(f"  Before switch: {before_control:.2f}")
print(f"  After switch:  {after_control:.2f}")
print(f"  Change:        {after_control - before_control:.2f}")

print("\nTreatment Group (switched):")
print(f"  Before switch: {before_treatment:.2f}")
print(f"  After switch:  {after_treatment:.2f}")
print(f"  Change:        {after_treatment - before_treatment:.2f}")

print("\nDifference-in-Differences estimate:")
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"  DiD = {did_estimate:.2f}")

# Now run the regression
print("\n=== Regression Analysis ===\n")

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
print("Model 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with controls (staffing_index, visits_index)")

print("\n=== Clustered Standard Errors (by branch) ===")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']

# Robustness check: event study / dynamic effects
# Create relative week indicators
df['relative_week'] = df['week'] - 15  # Week 15 is the switch, so 0

# Create dummies for each relative week (excluding the reference period)
# Let's use week 14 as reference (last pre-switch week)
print("=== Event Study / Dynamic Effects ===")
# Create interaction terms for each week with treatment
for w in range(6, 23):
    df[f'week_{w}'] = (df['week'] == w).astype(int)
    df[f'did_week_{w}'] = ((df['week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (using week 14 as reference)
event_study_vars = '+'.join([f'did_week_{w}' for w in range(6, 23) if w != 14])
formula = f'service_score ~ switched_branch + C(week) + staffing_index + visits_index + {event_study_vars}'
model_es = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract the dynamic DiD coefficients
print("\nDynamic DiD coefficients (relative to week 14):")
did_coefs = {}
for w in range(6, 23):
    if w != 14:
        coef_name = f'did_week_{w}'
        if coef_name in model_es.params.index:
            did_coefs[w] = {
                'coef': model_es.params[coef_name],
                'pvalue': model_es.pvalues[coef_name]
            }
            print(f"Week {w}: coef={model_es.params[coef_name]:.2f}, p={model_es.pvalues[coef_name]:.3f}")

# Check pre-trends (weeks 6-14, excluding reference week 14)
print("\n=== Pre-Trend Test ===")
pre_weeks = [w for w in range(6, 14)]
pre_trend_pvalues = [did_coefs[w]['pvalue'] for w in pre_weeks if w in did_coefs]
print(f"Pre-trend weeks (6-13):")
for w in pre_weeks:
    if w in did_coefs:
        print(f"  Week {w}: p={did_coefs[w]['pvalue']:.3f}")

# Post-treatment weeks
print(f"\nPost-treatment weeks (15-22):")
for w in range(15, 23):
    if w in did_coefs:
        print(f"  Week {w}: coef={did_coefs[w]['coef']:.2f}, p={did_coefs[w]['pvalue']:.3f}")
