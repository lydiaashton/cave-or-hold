import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# 1. Calculate average service scores by group and period
print("=" * 60)
print("AVERAGE SERVICE SCORES BY GROUP AND PERIOD")
print("=" * 60)

# Before switch (after_switch=0)
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
before_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()

# After switch (after_switch=1)
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
after_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"\nControl branches (old procedure):")
print(f"  Before switch: {before_control:.2f}")
print(f"  After switch:  {after_control:.2f}")
print(f"  Change:        {after_control - before_control:.2f}")

print(f"\nTreatment branches (new procedure):")
print(f"  Before switch: {before_treatment:.2f}")
print(f"  After switch:  {after_treatment:.2f}")
print(f"  Change:        {after_treatment - before_treatment:.2f}")

# Difference-in-differences estimate
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")

# 2. Run DiD regression
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD model
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Clustered standard errors by branch
print("\n" + "=" * 60)
print("CLUSTERED STANDARD ERRORS (by branch)")
print("=" * 60)

from statsmodels.stats.sandwich_covariance import cov_cluster

# Re-fit with clustered SE
model1_cluster = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 1 with clustered SE:")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Robustness check: Include branch fixed effects
print("=" * 60)
print("ROBUSTNESS: BRANCH FIXED EFFECTS MODEL")
print("=" * 60)

# Create branch dummies
df_with_fe = df.copy()
branch_dummies = pd.get_dummies(df_with_fe['branch_id'], prefix='branch', drop_first=True)
df_fe = pd.concat([df_with_fe, branch_dummies], axis=1)

# Model with branch fixed effects (absorbs switched_branch)
fe_cols = ['after_switch', 'treatment', 'staffing_index', 'visits_index'] + [c for c in df_fe.columns if c.startswith('branch_')]
formula = 'service_score ~ ' + ' + '.join(fe_cols)
model_fe = ols(formula, data=df_fe).fit(cov_type='cluster', cov_kwds={'groups': df_fe['branch_id']})

print("\nModel with Branch Fixed Effects:")
print("Treatment coefficient:", model_fe.params['treatment'])
print("Std Error:", model_fe.bse['treatment'])
print("P-value:", model_fe.pvalues['treatment'])
print(f"95% CI: [{model_fe.conf_int().loc['treatment'][0]:.2f}, {model_fe.conf_int().loc['treatment'][1]:.2f}]")

# Check if effect varies by post-switch week
print("\n" + "=" * 60)
print("EFFECT BY POST-SWITCH WEEK")
print("=" * 60)

# Calculate DiD for each post-switch week
post_weeks = df[df['week'] >= 15]['week'].unique()
baseline_control = df[(df['switched_branch']==0) & (df['week']<15)]['service_score'].mean()
baseline_treatment = df[(df['switched_branch']==1) & (df['week']<15)]['service_score'].mean()
baseline_diff = baseline_treatment - baseline_control

print(f"Baseline difference (treatment - control): {baseline_diff:.2f}")
print("\nWeek | Treatment avg | Control avg | Diff from baseline | DiD estimate")
print("-" * 70)

for week in sorted(post_weeks):
    week_control = df[(df['switched_branch']==0) & (df['week']==week)]['service_score'].mean()
    week_treatment = df[(df['switched_branch']==1) & (df['week']==week)]['service_score'].mean()
    week_diff = week_treatment - week_control
    did_weekly = week_diff - baseline_diff
    print(f"{week:4d} | {week_treatment:13.2f} | {week_control:11.2f} | {week_diff - baseline_diff:17.2f} | {did_weekly:13.2f}")

# Summary statistics for the effect
post_data = df[df['week'] >= 15]
weekly_did = post_data.groupby('week').apply(
    lambda x: (x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()) - baseline_diff
)
print(f"\nAverage weekly DiD estimate (post-switch): {weekly_did.mean():.2f}")
print(f"Std dev of weekly estimates: {weekly_did.std():.2f}")
print(f"Min: {weekly_did.min():.2f}, Max: {weekly_did.max():.2f}")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Robustness check: Include branch fixed effects
print("=" * 60)
print("ROBUSTNESS: BRANCH FIXED EFFECTS MODEL")
print("=" * 60)

# Model with branch fixed effects using C() notation
model_fe = ols('service_score ~ C(branch_id) + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nModel with Branch Fixed Effects:")
print("Treatment coefficient:", model_fe.params['treatment'])
print("Std Error:", model_fe.bse['treatment'])
print("P-value:", model_fe.pvalues['treatment'])
print(f"95% CI: [{model_fe.conf_int().loc['treatment'][0]:.2f}, {model_fe.conf_int().loc['treatment'][1]:.2f}]")

# Check if effect varies by post-switch week
print("\n" + "=" * 60)
print("EFFECT BY POST-SWITCH WEEK")
print("=" * 60)

# Calculate DiD for each post-switch week
post_weeks = df[df['week'] >= 15]['week'].unique()
baseline_control = df[(df['switched_branch']==0) & (df['week']<15)]['service_score'].mean()
baseline_treatment = df[(df['switched_branch']==1) & (df['week']<15)]['service_score'].mean()
baseline_diff = baseline_treatment - baseline_control

print(f"Baseline difference (treatment - control): {baseline_diff:.2f}")
print("\nWeek | Treatment avg | Control avg | Diff from baseline | DiD estimate")
print("-" * 70)

for week in sorted(post_weeks):
    week_control = df[(df['switched_branch']==0) & (df['week']==week)]['service_score'].mean()
    week_treatment = df[(df['switched_branch']==1) & (df['week']==week)]['service_score'].mean()
    week_diff = week_treatment - week_control
    did_weekly = week_diff - baseline_diff
    print(f"{week:4d} | {week_treatment:13.2f} | {week_control:11.2f} | {week_diff - baseline_diff:17.2f} | {did_weekly:13.2f}")

# Summary statistics for the effect
post_data = df[df['week'] >= 15]
weekly_did = post_data.groupby('week').apply(
    lambda x: (x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()) - baseline_diff
)
print(f"\nAverage weekly DiD estimate (post-switch): {weekly_did.mean():.2f}")
print(f"Std dev of weekly estimates: {weekly_did.std():.2f}")
print(f"Min: {weekly_did.min():.2f}, Max: {weekly_did.max():.2f}")

# Count how many weeks show positive effect
positive_weeks = (weekly_did > 0).sum()
print(f"Weeks with positive effect: {positive_weeks} of {len(weekly_did)}")
