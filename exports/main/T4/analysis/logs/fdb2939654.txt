import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary: how many branches switched?
print("Number of branches that switched:", df[df['switched_branch']==1]['branch_id'].nunique())
print("Number of branches that did not switch:", df[df['switched_branch']==0]['branch_id'].nunique())

# Average service scores by group and period
print("\n=== Average Service Scores ===")
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='after_switch', 
                       aggfunc='mean')
print(pivot)

# Calculate the DiD estimate manually
before_switched = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_switched = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print("\n=== Difference-in-Differences Calculation ===")
print(f"Switched branches - Before: {before_switched:.2f}, After: {after_switched:.2f}, Change: {after_switched - before_switched:.2f}")
print(f"Control branches - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")
did_estimate = (after_switched - before_switched) - (after_control - before_control)
print(f"\nDiD Estimate (treatment effect): {did_estimate:.2f}")

# Check pre-trends (weeks 6-14)
print("\n=== Pre-Trend Check (Weeks 6-14) ===")
pre_trend = df[df['week'] <= 14].groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

import pandas as pd
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Get the main result with confidence interval
model = smf.ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_ci_lower = treatment_coef - 1.96 * treatment_se
treatment_ci_upper = treatment_coef + 1.96 * treatment_se

print("=== MAIN RESULT ===")
print(f"Treatment Effect: {treatment_coef:.2f} points")
print(f"95% Confidence Interval: [{treatment_ci_lower:.2f}, {treatment_ci_upper:.2f}]")
print(f"p-value: {model.pvalues['treatment']:.4f}")

# Calculate percentage improvement
avg_score = df['service_score'].mean()
pct_improvement = (treatment_coef / avg_score) * 100
print(f"\nAverage service score across all observations: {avg_score:.2f}")
print(f"Percentage improvement: {pct_improvement:.1f}%")

# Check effect size in terms of standard deviation
sd_score = df['service_score'].std()
effect_size = treatment_coef / sd_score
print(f"Standard deviation of service scores: {sd_score:.2f}")
print(f"Effect size (Cohen's d): {effect_size:.2f}")

# Count observations
print(f"\nTotal observations: {len(df)}")
print(f"Branches: {df['branch_id'].nunique()}")
print(f"Switched branches: {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"Control branches: {df[df['switched_branch']==0]['branch_id'].nunique()}")
