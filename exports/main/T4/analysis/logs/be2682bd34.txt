import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# The issue is that switched_branch is time-invariant, so it gets absorbed by branch FE
# But the interaction switched_branch * after_switch varies over time, so it should be estimable

# Let's create the interaction term explicitly
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run with branch FE - the treatment variable is what we care about
# We need to be careful about specification

# Approach 1: Branch FE only, with treatment indicator
formula1 = 'service_score ~ treatment + after_switch + C(branch_id)'
model1 = ols(formula1, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Model 1: Branch FE with Treatment Indicator ===")
print(f"Treatment effect: {model1.params['treatment']:.3f} (SE: {model1.bse['treatment']:.3f}, p={model1.pvalues['treatment']:.4f})")

# Approach 2: Add week FE as well
formula2 = 'service_score ~ treatment + C(branch_id) + C(week)'
model2 = ols(formula2, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Model 2: Branch + Week FE with Treatment Indicator ===")
print(f"Treatment effect: {model2.params['treatment']:.3f} (SE: {model2.bse['treatment']:.3f}, p={model2.pvalues['treatment']:.4f})")

# Approach 3: Event study - look at effects by week relative to switch
df['relative_week'] = df['week'] - 15  # 0 = first week after switch, -1 = last week before

# Create indicators for each relative week (excluding the reference period)
# Use week -1 as reference (last week before switch)
for rw in range(-9, 8):  # weeks -9 to -1 (pre), 0 to 7 (post)
    if rw != -1:  # reference category
        df[f'rel_week_{rw}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Run event study with branch and week FE
event_vars = [f'rel_week_{rw}' for rw in range(-9, 8) if rw != -1]
event_formula = f"service_score ~ {' + '.join(event_vars)} + C(branch_id) + C(week)"
model_event = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n=== Event Study Results (relative to week -1, treatment group only) ===")
print(f"{'Week':<10} {'Coefficient':>12} {'Std Err':>10} {'P-value':>10}")
print("-" * 45)
for rw in range(-9, 8):
    if rw != -1:
        var = f'rel_week_{rw}'
        coef = model_event.params[var]
        se = model_event.bse[var]

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create cleaner event study variables with safe names
df['relative_week'] = df['week'] - 15

# Use safe variable names (no minus signs)
event_cols = {}
for rw in range(-9, 8):
    if rw != -1:  # reference
        col_name = f'rw{rw+10}'  # shift by 10 to make all positive: -9->1, -1->9, 0->10, 7->17
        event_cols[rw] = col_name
        df[col_name] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Run event study
event_vars = list(event_cols.values())
event_formula = f"service_score ~ {' + '.join(event_vars)} + C(branch_id) + C(week)"
model_event = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract results for plotting
results = []
for rw, col in event_cols.items():
    results.append({
        'relative_week': rw,
        'coef': model_event.params[col],
        'se': model_event.bse[col],
        'pvalue': model_event.pvalues[col],
        'period': 'Pre' if rw < 0 else 'Post'
    })

event_df = pd.DataFrame(results)
event_df = event_df.sort_values('relative_week')

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch Point')

# Pre-period
pre_df = event_df[event_df['relative_week'] < 0]
ax.errorbar(pre_df['relative_week'], pre_df['coef'], yerr=1.96*pre_df['se'], 
            fmt='o', color='blue', capsize=5, label='Pre-Switch', markersize=8)

# Post-period
post_df = event_df[event_df['relative_week'] >= 0]
ax.errorbar(post_df['relative_week'], post_df['coef'], yerr=1.96*post_df['se'], 
            fmt='o', color='green', capsize=5, label='Post-Switch', markersize=8)

ax.set_xlabel('Weeks Relative to Switch (0 = First Week After)', fontsize=12)
ax.set_ylabel('Effect on Service Score (vs Week -1)', fontsize=12)
ax.set_title('Event Study: Treatment Effect Over Time', fontsize=14, fontweight='bold')
ax.legend(fontsize=11)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-9, 8))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("Event study plot saved as event_study.png")

print("\n=== Event Study Coefficients ===")
print(event_df.to_string(index=False))

# Summary stats
print("\n=== SUMMARY STATISTICS ===")
print(f"Average pre-switch coefficient: {pre_df['coef'].mean():.2f}")
print(f"Average post-switch coefficient: {post_df['coef'].mean():.2f}")
print(f"Post-switch SE (average): {post_df['se'].mean():.3f}")
print(f"95% CI for average post-switch effect: [{post_df['coef'].mean() - 1.96*post_df['se'].mean():.2f}, {post_df['coef'].mean() + 1.96*post_df['se'].mean():.2f}]")

# Also print the main DiD estimate from the simpler model
df['treatment'] = df['switched_branch'] * df['after_switch']
model_simple = ols('service_score ~ treatment + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"\nMain DiD estimate (with branch + week FE): {model_simple.params['treatment']:.3f}")
print(f"Standard error: {model_simple.bse['treatment']:.3f}")
print(f"95% CI: [{model_simple.params['treatment'] - 1.96*model_simple.bse['treatment']:.2f}, {model_simple.params['treatment'] + 1.96*model_simple.bse['treatment']:.2f}]")
print(f"P-value: {model_simple.pvalues['treatment']:.6f}")
