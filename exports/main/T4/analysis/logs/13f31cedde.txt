import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Calculate mean service scores by group and period
print("=== Mean Service Scores by Group and Period ===")
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Calculate the DiD estimate manually
before_treated = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treated = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nTreated group: Before = {before_treated:.2f}, After = {after_treated:.2f}, Change = {after_treated - before_treated:.2f}")
print(f"Control group: Before = {before_control:.2f}, After = {after_control:.2f}, Change = {after_control - before_control:.2f}")
print(f"\nDifference-in-Differences estimate: {(after_treated - before_treated) - (after_control - before_control):.2f}")

# Run DiD regression
print("\n=== DiD Regression (Basic) ===")
model_basic = ols('service_score ~ switched_branch + after_switch + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_basic.summary())

# Run DiD regression with controls
print("\n=== DiD Regression (With Controls) ===")
model_controls = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_controls.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create relative week variable (weeks since switch)
# Switch happened at week 15, so week 15 is relative week 0
df['relative_week'] = df['week'] - 15

# Create event study dummies (omitting week -1 as reference)
# Weeks: -9 to -1 (before), 0 to 7 (after)
for w in range(-9, 8):
    if w != -1:  # omit week -1 as reference
        df[f'week_{w}'] = (df['relative_week'] == w).astype(int)

# Event study regression
event_study_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ switched_branch + ' + ' + '.join(event_study_vars)

# Run event study with controls
model_event = ols(formula + ' + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Regression ===")
print(model_event.summary())

# Extract coefficients for the event study
print("\n=== Event Study Coefficients ===")
coef_dict = {}
for w in range(-9, 8):
    if w != -1:
        coef = model_event.params[f'week_{w}']
        se = model_event.bse[f'week_{w}']
        coef_dict[w] = (coef, se)
        print(f"Week {w:+d}: {coef:.2f} (SE: {se:.2f})")
    else:
        print(f"Week {w:+d}: [Reference]")

# Check pre-trend significance (joint test for weeks -9 to -2)
pre_trend_vars = [f'week_{w}' for w in range(-9, -1)]
print(f"\n=== Pre-Trend Joint Test (Weeks -9 to -2) ===")
from statsmodels.stats.contrast import ContrastMatrix
# Simple approach: test if all pre-trend coefficients are jointly zero
hypotheses = [f"{var} = 0" for var in pre_trend_vars]
f_test = model_event.f_test(hypotheses)
print(f"F-statistic: {f_test.fvalue:.2f}, p-value: {f_test.pvalue:.4f}")

import numpy as np

# Load the data

formula = 'service_score ~ switched_branch + ' + ' + '.join(event_study_vars)

# Run event study with controls
model_event = ols(formula + ' + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
