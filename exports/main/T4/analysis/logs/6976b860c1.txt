import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check how many branches switched vs didn't
branches = df.groupby('branch_id')['switched_branch'].first()
print("Branches that switched:", branches.sum())
print("Branches that didn't switch:", len(branches) - branches.sum())
print("\nProportion switched:", branches.mean())

# Create a summary table of service scores by group and period
df['period'] = df['after_switch'].map({0: 'Before', 1: 'After'})
df['group'] = df['switched_branch'].map({0: 'Control', 1: 'Treatment'})

summary = df.groupby(['group', 'period'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("\nService Score Summary by Group and Period:")
print(summary)

# Calculate the difference-in-differences manually
before_control = df[(df['group'] == 'Control') & (df['period'] == 'Before')]['service_score'].mean()
after_control = df[(df['group'] == 'Control') & (df['period'] == 'After')]['service_score'].mean()
before_treatment = df[(df['group'] == 'Treatment') & (df['period'] == 'Before')]['service_score'].mean()
after_treatment = df[(df['group'] == 'Treatment') & (df['period'] == 'After')]['service_score'].mean()

print("\nMean Service Scores:")
print(f"Control - Before: {before_control:.2f}")
print(f"Control - After: {after_control:.2f}")
print(f"  Control change: {after_control - before_control:.2f}")
print(f"\nTreatment - Before: {before_treatment:.2f}")
print(f"Treatment - After: {after_treatment:.2f}")
print(f"  Treatment change: {after_treatment - before_treatment:.2f}")

did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate weekly averages by group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg['group'] = weekly_avg['switched_branch'].map({0: 'Control (No Switch)', 1: 'Treatment (Switched)'})

# Create the plot
fig, ax = plt.subplots(figsize=(12, 6))

for group in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == group]
    label = 'Treatment (Switched)' if group == 1 else 'Control (No Switch)'
    color = '#2E86AB' if group == 1 else '#A23B72'
    ax.plot(subset['week'], subset['service_score'], marker='o', linewidth=2, markersize=6, 
            label=label, color=color)

# Add vertical line at week 15 (switch point)
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch Week (15)')
ax.axvspan(14.5, 22, alpha=0.1, color='gray', label='After Switch Period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Score Trends: Treatment vs Control Groups\n(Vertical line marks the switch at Week 15)', fontsize=14)
ax.legend(loc='upper left')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("Plot saved as 'service_score_trends.png'")

# Check parallel trends statistically - compare trends before week 15
pre_switch = df[df['week'] < 15].copy()
pre_switch['week_centered'] = pre_switch['week'] - pre_switch['week'].mean()

# Test if pre-trends are parallel
model_pretrend = ols('service_score ~ switched_branch * week_centered', data=pre_switch).fit(cov_type='cluster', cov_kwds={'groups': pre_switch['branch_id']})
print("\nPre-Trend Parallel Test (interaction of switched_branch with week trend before switch):")
print(f"Interaction coefficient: {model_pretrend.params['switched_branch:week_centered']:.4f}")
print(f"p-value: {model_pretrend.pvalues['switched_branch:week_centered']:.4f}")
print("(Non-significant p-value supports parallel trends assumption)")

# Also show weekly DiD estimates
print("\n\nWeekly averages by group:")
print(weekly_avg.pivot(index='week', columns='switched_branch', values='service_score').round(2))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Add branch fixed effects using C(branch_id)
# This controls for time-invariant differences between branches
model_fe = ols('service_score ~ C(branch_id) + switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("DiD with Branch Fixed Effects:")
print(model_fe.summary())

# The key coefficient is still the interaction term
print(f"\nKey coefficient (switched_branch:after_switch): {model_fe.params['switched_branch:after_switch']:.3f}")
print(f"SE: {model_fe.bse['switched_branch:after_switch']:.3f}")
print(f"95% CI: [{model_fe.conf_int().loc['switched_branch:after_switch'][0]:.3f}, {model_fe.conf_int().loc['switched_branch:after_switch'][1]:.3f}]")

# Alternative: event study approach - look at week-specific effects relative to week 14 (last pre-switch week)
df['week_relative'] = df['week'] - 14  # Week 14 is baseline (0)

# Create week dummies interacted with treatment
# First, let's see if we can estimate week-by-week treatment effects
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=False)
df_with_weeks = pd.concat([df, week_dummies], axis=1)

# For event study, we need to interact each week with treatment status
# Let's do a simpler approach: estimate the effect for each post-switch week separately
print("\n\nEvent Study Analysis:")
for week_num in range(15, 23):
    df_temp = df.copy()
    df_temp['post_specific'] = (df_temp['week'] == week_num).astype(int)
    df_temp['treatment_post'] = df_temp['switched_branch'] * df_temp['post_specific']
    
    model_week = ols('service_score ~ C(branch_id) + switched_branch + post_specific + treatment_post + staffing_index + visits_index', 
                     data=df_temp).fit(cov_type='cluster', cov_kwds={'groups': df_temp['branch_id']})
    
    coef = model_week.params['treatment_post']
    se = model_week.bse['treatment_post']
    pval = model_week.pvalues['treatment_post']
    print(f"Week {week_num}: Effect = {coef:.2f}, SE = {se:.2f}, p = {pval:.3f}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators and interact with treatment
# Week 14 is the last pre-switch week, use as reference
df['week_rel'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create indicators for each week relative to switch
# We'll estimate effects for each week, using week 14 as baseline
event_study_results = []

for w in range(-8, 9):  # Week 6 to 22, relative to week 14
    if w == 0:  # Skip reference week
        continue
    
    df_temp = df.copy()
    df_temp['week_indicator'] = (df_temp['week_rel'] == w).astype(int)
    df_temp['treatment_week'] = df_temp['switched_branch'] * df_temp['week_indicator']
    
    # Run regression with branch and week FEs
    model = ols('service_score ~ C(branch_id) + C(week) + switched_branch + week_indicator + treatment_week + staffing_index + visits_index', 
                data=df_temp).fit(cov_type='cluster', cov_kwds={'groups': df_temp['branch_id']})
    
    coef = model.params.get('treatment_week', np.nan)
    se = model.bse.get('treatment_week', np.nan)
    
    event_study_results.append({
        'week': 14 + w,
        'week_rel': w,
        'period': 'Before' if w < 0 else 'After',
        'coef': coef,
        'se': se,
        'ci_lower': coef - 1.96 * se if not np.isnan(se) else np.nan,
        'ci_upper': coef + 1.96 * se if not np.isnan(se) else np.nan
    })

event_df = pd.DataFrame(event_study_results)
print("Event Study Results:")
print(event_df.round(3))

# Plot the event study
fig, ax = plt.subplots(figsize=(12, 6))

# Plot pre-period
pre = event_df[event_df['period'] == 'Before']
ax.errorbar(pre['week'], pre['coef'], yerr=[pre['coef'] - pre['ci_lower'], pre['ci_upper'] - pre['coef']], 
            fmt='o', color='#A23B72', label='Before Switch', capsize=5, linewidth=2, markersize=8)

# Plot post-period
post = event_df[event_df['period'] == 'After']
ax.errorbar(post['week'], post['coef'], yerr=[post['coef'] - post['ci_lower'], post['ci_upper'] - post['coef']], 
            fmt='o', color='#2E86AB', label='After Switch', capsize=5, linewidth=2, markersize=8)

ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch Point')
ax.axhline(y=0, color='black', linestyle='-', linewidth=1, alpha=0.5)
ax.axhline(y=1.87, color='green', linestyle=':', linewidth=2, label='Average Treatment Effect (1.87)')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Treatment Effect (vs Control)', fontsize=12)
ax.set_title('Event Study: Weekly Treatment Effects on Service Score\n(Reference: Week 14, baseline difference between groups)', fontsize=14)
ax.legend(loc='upper left')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\n\nEvent study plot saved as 'event_study.png'")

# Calculate average pre and post effects
print("\n\nAverage effect before switch (weeks 6-14):", event_df[event_df['period'] == 'Before']['coef'].mean())
print("Average effect after switch (weeks 15-22):", event_df[event_df['period'] == 'After']['coef'].mean())
