import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks since switch, with week 14 as reference)
# Week 15 is the first "after" week, so relative_week = week - 14
df['relative_week'] = df['week'] - 14

# Event study: estimate effect for each week relative to switch
# Using week 14 (relative_week=0) as the reference period
# We'll estimate separate treatment effects for each week

# Create interaction dummies for each week
event_study_results = []
for w in range(-8, 9):  # weeks -8 to +8 relative to switch (week 14 is 0)
    if w == 0:  # reference week
        continue
    df[f'rel_week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run regression with all relative week interactions
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'rel_week_{w}' for w in range(-8, 9) if w != 0])
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results ===")
# Extract coefficients for the relative week terms
coef_dict = {}
for w in range(-8, 9):
    if w == 0:
        coef_dict[w] = 0  # reference
        continue
    coef_name = f'rel_week_{w}'
    if coef_name in event_model.params:
        coef_dict[w] = event_model.params[coef_name]

print("\nRelative Week Coefficients (treatment effect by week):")
for w in sorted(coef_dict.keys()):
    print(f"Week {w}: {coef_dict[w]:.3f}")

# Plot event study
plt.figure(figsize=(10, 6))
weeks = list(coef_dict.keys())
coefs = [coef_dict[w] for w in weeks]
plt.plot(weeks, coefs, 'o-', linewidth=2, markersize=8)
plt.axhline(y=0, color='black', linestyle='-', linewidth=1)
plt.axvline(x=0.5, color='red', linestyle='--', linewidth=2, label='Switch Week')
plt.xlabel('Relative Week (0 = Week 14, pre-switch reference)', fontsize=12)
plt.ylabel('Treatment Effect (Service Score Points)', fontsize=12)
plt.title('Event Study: Treatment Effect by Week', fontsize=14)
plt.grid(True, alpha=0.3)
plt.legend()
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")

# Get confidence intervals for the event study
print("\n=== Event Study with Confidence Intervals ===")
for w in sorted(coef_dict.keys()):
    if w == 0:
        continue
    coef_name = f'rel_week_{w}'
    if coef_name in event_model.params:
        coef = event_model.params[coef_name]
        se = event_model.bse[coef_name]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        print(f"Week {w}: {coef:.3f} (95% CI: {ci_low:.3f}, {ci_high:.3f})")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks since switch, with week 14 as reference)
df['relative_week'] = df['week'] - 14

# Create interaction dummies for each week with safe names
for w in range(-8, 9):
    if w == 0:  # reference week
        continue
    safe_name = f'rel_week_p{w}' if w >= 0 else f'rel_week_m{abs(w)}'
    df[safe_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Build formula
terms = []
for w in range(-8, 9):
    if w == 0:
        continue
    safe_name = f'rel_week_p{w}' if w >= 0 else f'rel_week_m{abs(w)}'
    terms.append(safe_name)

formula = 'service_score ~ switched_branch + ' + ' + '.join(terms)
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results ===")

# Extract coefficients and map back to relative weeks
coef_dict = {}
for w in range(-8, 9):
    if w == 0:
        coef_dict[w] = 0  # reference
        continue
    safe_name = f'rel_week_p{w}' if w >= 0 else f'rel_week_m{abs(w)}'
    coef_dict[w] = event_model.params[safe_name]

print("\nRelative Week Coefficients (treatment effect by week):")
for w in sorted(coef_dict.keys()):
    print(f"Week {w}: {coef_dict[w]:.3f}")

# Plot event study
plt.figure(figsize=(10, 6))
weeks = list(coef_dict.keys())
coefs = [coef_dict[w] for w in weeks]

# Get confidence intervals
ci_low = []
ci_high = []
for w in weeks:
    if w == 0:
        ci_low.append(0)
        ci_high.append(0)
        continue
    safe_name = f'rel_week_p{w}' if w >= 0 else f'rel_week_m{abs(w)}'
    coef = event_model.params[safe_name]
    se = event_model.bse[safe_name]
    ci_low.append(coef - 1.96 * se)
    ci_high.append(coef + 1.96 * se)

plt.plot(weeks, coefs, 'o-', linewidth=2, markersize=8, label='Treatment Effect')
plt.fill_between(weeks, ci_low, ci_high, alpha=0.3, label='95% CI')
plt.axhline(y=0, color='black', linestyle='-', linewidth=1)
plt.axvline(x=0.5, color='red', linestyle='--', linewidth=2, label='Switch Week')
plt.xlabel('Relative Week (0 = Week 14, pre-switch reference)', fontsize=12)
plt.ylabel('Treatment Effect (Service Score Points)', fontsize=12)
plt.title('Event Study: Treatment Effect by Week', fontsize=14)
plt.grid(True, alpha=0.3)
plt.legend()
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")

# Get confidence intervals for the event study
print("\n=== Event Study with Confidence Intervals ===")
for w in sorted(coef_dict.keys()):
    if w == 0:
        print(f"Week {w}: 0.000 (reference)")

model_week_fe = ols('service_score ~ switched_branch + treatment + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== DiD with Week Fixed Effects ===")

print("\n=== DiD with Branch and Week Fixed Effects ===")
# Use a within transformation approach or just include branch dummies
# For 709 branches, this is feasible
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)

# Build formula with branch dummies
branch_cols = [c for c in df_with_branch_fe.columns if c.startswith('branch_')]

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Final preferred model with week fixed effects and controls
model_final = ols('service_score ~ switched_branch + treatment + C(week) + staffing_index + visits_index', 
                  data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Final Model (Week FE + Controls) ===")
print(model_final.summary())

# Calculate practical significance
# Average service score is about 50, so a 2 point increase is about 4%
print("\n=== Practical Significance ===")
avg_score = df['service_score'].mean()
treatment_effect = model_final.params['treatment']

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create a clean trend plot
weekly_means = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
weekly_means.columns = ['Control', 'Treatment']

# Calculate counterfactual for treatment group (what they would have scored without treatment)
# Using control group's post-switch change
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
control_change = control_after - control_before

treated_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
counterfactual_after = treated_before + control_change

print("=== Counterfactual Analysis ===")
print(f"Control before: {control_before:.2f}")
print(f"Control after: {control_after:.2f}")
print(f"Control change: {control_change:.2f}")
print(f"Treated before: {treated_before:.2f}")
print(f"Treated actual after: {df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean():.2f}")
print(f"Treated counterfactual after: {counterfactual_after:.2f}")
print(f"Estimated treatment effect: {df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean() - counterfactual_after:.2f}")

# Create visualization
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Plot 1: Raw trends
ax1 = axes[0]
ax1.plot(weekly_means.index, weekly_means['Control'], 'o-', label='Control (No Switch)', linewidth=2, markersize=5, color='steelblue')
ax1.plot(weekly_means.index, weekly_means['Treatment'], 's-', label='Treatment (Switched)', linewidth=2, markersize=5, color='darkorange')
ax1.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (Week 15)')
ax1.set_xlabel('Week', fontsize=11)
ax1.set_ylabel('Service Score', fontsize=11)
ax1.set_title('Service Scores Over Time', fontsize=13, fontweight='bold')
ax1.legend(fontsize=10)
ax1.grid(True, alpha=0.3)
ax1.set_xticks(range(6, 23))

# Plot 2: Difference between groups
ax2 = axes[1]
diff = weekly_means['Treatment'] - weekly_means['Control']
ax2.plot(diff.index, diff.values, 'o-', linewidth=2, markersize=5, color='green')
ax2.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax2.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (Week 15)')
ax2.fill_between([6, 14], -1, 3, alpha=0.1, color='gray', label='Pre-switch period')
ax2.fill_between([15, 22], -1, 3, alpha=0.1, color='lightblue', label='Post-switch period')
ax2.set_xlabel('Week', fontsize=11)
ax2.set_ylabel('Treatment - Control (points)', fontsize=11)
ax2.set_title('Gap Between Treatment and Control Groups', fontsize=13, fontweight='bold')
ax2.legend(fontsize=10)
ax2.grid(True, alpha=0.3)
ax2.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('analysis_summary.png', dpi=150)
print("\nSummary visualization saved as 'analysis_summary.png'")
