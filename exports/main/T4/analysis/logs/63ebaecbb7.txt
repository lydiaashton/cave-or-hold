import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Count branches by switch status
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate average service scores by group and period
print("\n\nAverage service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the difference-in-differences manually
print("\n\nDifference-in-Differences Calculation:")
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
before_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"Control (no switch) - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")
print(f"Treatment (switched) - Before: {before_treatment:.2f}, After: {after_treatment:.2f}, Change: {after_treatment - before_treatment:.2f}")
did_effect = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences Effect: {did_effect:.2f}")

# Weekly trends by group
print("\n\nWeekly average service scores by group:")
weekly = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(0).round(2)
print(weekly)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch fixed effects
# Model: service_score = β0 + β1*after_switch + β2*switched_branch + β3*(after_switch * switched_branch) + ε
model = ols('service_score ~ after_switch + switched_branch + treatment', data=df).fit()
print("Basic DiD Regression (no fixed effects):")
print(model.summary())

# Now with branch fixed effects (more robust)
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df_with_dummies['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Get all branch dummy column names
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_')]

# Build formula
formula = 'service_score ~ after_switch + switched_branch + treatment + ' + ' + '.join(branch_cols)
model_fe = ols(formula, data=df_with_dummies).fit()
print("\n\nDiD Regression with Branch Fixed Effects:")
print(model_fe.summary().tables[1].iloc[:10])  # Just first 10 rows for brevity

# Extract the treatment effect
treatment_effect = model_fe.params['treatment']
treatment_se = model_fe.bse['treatment']
treatment_pvalue = model_fe.pvalues['treatment']
ci_lower = treatment_effect - 1.96 * treatment_se
ci_upper = treatment_effect + 1.96 * treatment_se

print(f"\n\nTreatment Effect (DiD coefficient): {treatment_effect:.3f}")
print(f"Standard Error: {treatment_se:.3f}")
print(f"95% CI: [{ci_lower:.3f}, {ci_upper:.3f}]")
print(f"P-value: {treatment_pvalue:.6f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with cluster-robust standard errors
model = ols('service_score ~ after_switch + switched_branch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("DiD Regression with Cluster-Robust Standard Errors:")
print(model.summary())

# Extract the treatment effect
treatment_effect = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pvalue = model.pvalues['treatment']
ci_lower = treatment_effect - 1.96 * treatment_se
ci_upper = treatment_effect + 1.96 * treatment_se

print(f"\n\nTreatment Effect (DiD coefficient): {treatment_effect:.3f}")
print(f"Standard Error: {treatment_se:.3f}")
print(f"95% CI: [{ci_lower:.3f}, {ci_upper:.3f}]")
print(f"P-value: {treatment_pvalue:.6f}")

# Check parallel trends - compare pre-treatment trends
print("\n\n=== Parallel Trends Check ===")
pre_data = df[df['week'] < 15].copy()
pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()

# Regress service_score on week, switched_branch, and their interaction (pre-period only)
pre_model = ols('service_score ~ week_centered * switched_branch', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print("\nPre-treatment trend comparison:")
print(pre_model.summary().tables[1])

# Also check week-by-week differences before treatment
print("\n\nAverage difference (treatment - control) by week (pre-period only):")
pre_pivot = pre_data.groupby('week').apply(lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean())
print(pre_pivot)

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create visualization of trends
plt.figure(figsize=(12, 6))

# Calculate weekly averages by group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(0)

weeks = weekly_avg.index
control = weekly_avg[0].values
treatment = weekly_avg[1].values

plt.plot(weeks, control, 'o-', label='Control (No Switch)', linewidth=2, markersize=8)
plt.plot(weeks, treatment, 's-', label='Treatment (Switched)', linewidth=2, markersize=8)

# Add vertical line at switch point
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (Week 15)')

plt.xlabel('Week', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Score Trends: Treatment vs Control Branches', fontsize=14, fontweight='bold')
plt.legend(fontsize=11)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("Saved service_score_trends.png")

# Also check if staffing or visits might be confounders
print("\n=== Checking potential confounders ===")

# DiD for staffing_index
df['treatment'] = df['switched_branch'] * df['after_switch']
staffing_model = ols('staffing_index ~ after_switch + switched_branch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"\nStaffing Index DiD Effect: {staffing_model.params['treatment']:.3f} (p={staffing_model.pvalues['treatment']:.3f})")

# DiD for visits_index
visits_model = ols('visits_index ~ after_switch + switched_branch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"Visits Index DiD Effect: {visits_model.params['treatment']:.3f} (p={visits_model.pvalues['treatment']:.3f})")

# Add controls to main model
from statsmodels.formula.api import ols
full_model = ols('service_score ~ after_switch + switched_branch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"\nFull Model (with controls) Treatment Effect: {full_model.params['treatment']:.3f} (p={full_model.pvalues['treatment']:.3f})")
print(full_model.summary().tables[1])
