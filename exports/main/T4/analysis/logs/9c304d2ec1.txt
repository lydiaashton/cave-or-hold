df['treated'] = df['switched_branch'] * df['after_switch']

import pandas as pd
import statsmodels.formula.api as smf
import numpy as np

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# Main DiD model with controls
model = smf.ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', data=df).fit()
model_clustered = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Convert to Series for easier access
params = pd.Series(model_clustered.params, index=model_clustered.params.index)
bse = pd.Series(model_clustered.bse, index=model_clustered.bse.index)
pvalues = pd.Series(model_clustered.pvalues, index=model_clustered.pvalues.index)

# Extract key results
treated_coef = params['treated']
treated_se = bse['treated']
treated_pvalue = pvalues['treated']

# Calculate 95% CI
treated_ci_lower = treated_coef - 1.96 * treated_se
treated_ci_upper = treated_coef + 1.96 * treated_se

print("=" * 60)
print("KEY RESULTS FOR BOARD SUMMARY")
print("=" * 60)
print(f"\nTreatment Effect (New Procedure):")
print(f"  Point estimate: {treated_coef:.2f} points")
print(f"  95% Confidence Interval: [{treated_ci_lower:.2f}, {treated_ci_upper:.2f}]")
print(f"  P-value: {treated_pvalue:.4f}")
print(f"  Statistically significant: {'Yes' if treated_pvalue < 0.05 else 'No'}")

# Calculate effect size relative to baseline
baseline_score = df[df['after_switch'] == 0]['service_score'].mean()
relative_effect = (treated_coef / baseline_score) * 100
print(f"\n  Baseline average score: {baseline_score:.1f}")
