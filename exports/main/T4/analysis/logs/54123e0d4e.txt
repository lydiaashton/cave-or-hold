import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create treatment interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate mean service scores by group and period
print("=== Mean Service Scores by Group and Period ===")
group_means = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(group_means)

# Calculate the DiD estimate manually
print("\n=== Difference-in-Differences Calculation ===")
# Control group (switched_branch=0)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_diff = control_after - control_before

# Treatment group (switched_branch=1)
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_diff = treat_after - treat_before

did_estimate = treat_diff - control_diff

print(f"Control group: Before={control_before:.2f}, After={control_after:.2f}, Change={control_diff:.2f}")
print(f"Treatment group: Before={treat_before:.2f}, After={treat_after:.2f}, Change={treat_diff:.2f}")
print(f"\nDiD Estimate (Treatment effect): {did_estimate:.3f}")

# Run DiD regression
print("\n=== DiD Regression ===")
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model.summary())

# Check parallel trends - look at weekly trends before the switch
print("\n=== Weekly Trends Before Switch (Weeks 6-14) ===")
pre_switch = df[df['week'] < 15].copy()
weekly_trends = pre_switch.groupby(['switched_branch', 'week'])['service_score'].mean().unstack()
print(weekly_trends)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD with branch fixed effects
print("=== DiD Regression with Branch Fixed Effects ===")
# Add branch as categorical
df['branch_id'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit()
# Note: switched_branch will be collinear with branch fixed effects, so it will be dropped
print(model_fe.summary())

# Let's run it without switched_branch since it's absorbed by branch FE
model_fe2 = ols('service_score ~ after_switch + treatment + C(branch_id)', data=df).fit()
print("\n=== DiD with Branch FE (switched_branch absorbed) ===")
print(model_fe2.summary())

# Check parallel trends visually - plot weekly averages
print("\n=== Weekly Averages by Group ===")
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
print(weekly_avg)

# Create a plot
plt.figure(figsize=(10, 6))
for switched in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == switched]
    label = 'Switched Branches (Treatment)' if switched == 1 else 'Non-Switched Branches (Control)'
    plt.plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2)

plt.axvline(x=14.5, color='red', linestyle='--', label='Switch Point (Week 15)')
plt.xlabel('Week')
plt.ylabel('Mean Service Score')
plt.title('Service Scores Over Time: Treatment vs Control Groups')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('service_scores_trend.png', dpi=150)
print("\nPlot saved as service_scores_trend.png")

# Calculate effect size relative to baseline
baseline = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
effect_size_pct = (1.9218 / baseline) * 100
print(f"\n=== Effect Size ===")
print(f"Baseline (treatment group before switch): {baseline:.2f}")
print(f"DiD Effect: 1.92 points")
print(f"Effect as % of baseline: {effect_size_pct:.1f}%")

# Check if staffing and visits changed differently for treatment vs control
print("\n=== Checking Other Metrics (Staffing & Visits) ===")
for metric in ['staffing_index', 'visits_index']:
    model = ols(f'{metric} ~ switched_branch + after_switch + treatment', data=df).fit()
    print(f"\n{metric}:")
    print(f"  Treatment effect coefficient: {model.params['treatment']:.4f} (p={model.pvalues['treatment']:.4f})")

from statsmodels.formula.api import ols
import matplotlib

# Run cleaner DiD with branch fixed effects
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ after_switch + treatment + C(branch_id_str)', data=df).fit()

# Get cluster-robust standard errors at branch level
model_fe_cluster = ols('service_score ~ after_switch + treatment + C(branch_id_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== DiD with Branch Fixed Effects (Cluster-Robust SEs) ===")
print(model_fe_cluster.summary().tables[1])

# Extract key coefficient
treatment_coef = model_fe_cluster.params['treatment']
treatment_se = model_fe_cluster.bse['treatment']

treatment_ci_low = treatment_coef - 1.96 * treatment_se
treatment_ci_high = treatment_coef + 1.96 * treatment_se

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple DiD with cluster-robust SEs (no branch FE to avoid numerical issues)
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(
    cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Difference-in-Differences Regression (Cluster-Robust SEs) ===")
print(model.summary().tables[1])

treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_p = model.pvalues['treatment']
treatment_ci_low = treatment_coef - 1.96 * treatment_se
treatment_ci_high = treatment_coef + 1.96 * treatment_se

print(f"\n=== KEY RESULTS ===")
print(f"Treatment Effect: {treatment_coef:.3f} points")
print(f"Standard Error: {treatment_se:.3f}")
print(f"95% CI: [{treatment_ci_low:.3f}, {treatment_ci_high:.3f}]")
print(f"P-value: {treatment_p:.6f}")
print(f"Statistically significant: {'Yes' if treatment_p < 0.05 else 'No'}")

# Calculate effect size
baseline = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
effect_size_pct = (treatment_coef / baseline) * 100
print(f"\nBaseline (treatment group pre-switch): {baseline:.2f}")
print(f"Effect as % of baseline: {effect_size_pct:.1f}%")

# Check parallel trends with simple pre-period comparison
print("\n=== Parallel Trends Check (Pre-Period Only) ===")
pre_data = df[df['week'] < 15].copy()
pre_model = ols('service_score ~ switched_branch + C(week)', data=pre_data).fit(
    cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print(f"Pre-period treatment vs control difference: {pre_model.params['switched_branch']:.3f} (p={pre_model.pvalues['switched_branch']:.3f})")

# Event study - simplified version
print("\n=== Event Study (Weekly Treatment Effects) ===")
df['week_rel'] = df['week'] - 14  # relative to week 14

# Create week dummies for treatment group
event_data = df.copy()
for w in sorted(df['week'].unique()):
    if w != 13:  # omit week 13 as reference
        event_data[f'treat_week{w}'] = ((event_data['week'] == w) & (event_data['switched_branch'] == 1)).astype(int)

week_vars = [f'treat_week{w}' for w in sorted(df['week'].unique()) if w != 13]
formula = 'service_score ~ ' + ' + '.join(week_vars) + ' + C(week)'
event_model = ols(formula, data=event_data).fit(cov_type='cluster', cov_kwds={'groups': event_data['branch_id']})

print("Week | Effect | SE | 95% CI | Period")
print("-" * 50)
for w in sorted(df['week'].unique()):
    if w != 13:
        var = f'treat_week{w}'
        coef = event_model.params[var]
        se = event_model.bse[var]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        period = "Pre" if w < 15 else "POST"
        sig = "***" if event_model.pvalues[var] < 0.01 else "**" if event_model.pvalues[var] < 0.05 else "*" if event_model.pvalues[var] < 0.1 else ""
        print(f"{w:4} | {coef:6.2f} | {se:5.2f} | [{ci_low:5.2f}, {ci_high:5.2f}] | {period} {sig}")

# Create visualization
plt.figure(figsize=(12, 6))

cis_high = []

for w in sorted(df['week'].unique()):

             fmt='o', capsize=5, markersize=8, label='Treatment Effect vs Week 13')

plt.ylabel('Service Score Effect')
plt.title('Event Study: Weekly Treatment Effects on Service Scores')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(sorted(df['week'].unique()))
