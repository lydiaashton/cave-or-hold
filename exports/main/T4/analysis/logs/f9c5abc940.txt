import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate

df['treatment_interaction'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment_interaction', data=df).fit()
print("Basic DiD Regression (no clustering):")
print(model.summary())

# Now with cluster-robust standard errors at branch level
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment_interaction', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression (clustered by branch):")
print(model_clustered.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Calculate mean service score by week and treatment status
weekly_means = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_means['group'] = weekly_means['switched_branch'].map({0: 'Control (Old Procedure)', 1: 'Treatment (New Procedure)'})

# Plot the trends
plt.figure(figsize=(10, 6))
for group in [0, 1]:
    subset = weekly_means[weekly_means['switched_branch'] == group]
    label = 'Treatment (New Procedure)' if group == 1 else 'Control (Old Procedure)'
    plt.plot(subset['week'], subset['service_score'], marker='o', linewidth=2, label=label)

plt.axvline(x=14.5, color='red', linestyle='--', label='Switch Week (15)')
plt.xlabel('Week')
plt.ylabel('Mean Service Score')
plt.title('Service Score Trends by Group Over Time')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('trends.png', dpi=150)
print("Plot saved as trends.png")

# Also check parallel trends more formally - compare trends in pre-period
pre_period = df[df['week'] < 15]
print("\n\nPre-period trends (weeks 6-14):")
print(pre_period.groupby(['week', 'switched_branch'])['service_score'].mean().unstack())

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term
df['treatment_interaction'] = df['switched_branch'] * df['after_switch']

# Run the main DiD model with controls and clustered SEs
model = ols('service_score ~ switched_branch + after_switch + treatment_interaction + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("="*80)
print("FINAL DIFFERENCE-IN-DIFFERENCES ANALYSIS")
print("="*80)
print("\nModel: service_score ~ switched_branch + after_switch + treatment_interaction + staffing_index + visits_index")
print("Standard errors clustered by branch")
print("\n" + "="*80)
print(model.summary().tables[1])
print("="*80)

# Extract key results
did_coef = model.params['treatment_interaction']
did_se = model.bse['treatment_interaction']
did_pvalue = model.pvalues['treatment_interaction']
did_ci_lower = model.conf_int().loc['treatment_interaction', 0]
did_ci_upper = model.conf_int().loc['treatment_interaction', 1]

print(f"\n\nKEY RESULT:")
print(f"  Treatment effect (DiD estimate): {did_coef:.2f} points")
print(f"  95% Confidence Interval: [{did_ci_lower:.2f}, {did_ci_upper:.2f}]")
print(f"  P-value: {did_pvalue:.4f}")
print(f"  Statistically significant: {'Yes' if did_pvalue < 0.05 else 'No'}")

# Calculate effect size relative to baseline
baseline_score = df[df['after_switch']==0]['service_score'].mean()
print(f"\n  Baseline mean service score: {baseline_score:.2f}")
print(f"  Effect as % of baseline: {(did_coef/baseline_score)*100:.1f}%")
