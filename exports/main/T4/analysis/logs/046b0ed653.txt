# Calculate average service scores by group and period
print("\n\nAverage service scores by group and period:")
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='period', 
                       aggfunc='mean')

pivot.index = ['Control (old procedure)', 'Treatment (new procedure)']
print(pivot)

control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()

treatment_change = treatment_after - treatment_before
did_estimate = treatment_change - control_change

print(f"Treatment group change: {treatment_change:.3f}")
print(f"Difference-in-Differences estimate: {did_estimate:.3f}")

print("\n\nWeekly average service scores by group:")

import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Check parallel trends - compare weekly differences between treatment and control before the switch
print("Pre-treatment trends (weeks 6-14):")
pre_df = df[df['week'] <= 14].copy()
weekly_pre = pre_df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_pre['diff'] = weekly_pre[1] - weekly_pre[0]
print(weekly_pre)

# Event study - create week dummies relative to switch
df['weeks_to_switch'] = df['week'] - 15  # week 15 is the switch week (0)

# Create dummies for each week relative to switch (use week 14 as reference)
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    df[f'week_{w}'] = ((df['weeks_to_switch'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (excluding week -1 as reference)
week_cols = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = f"service_score ~ C(branch_id) + after_switch + {' + '.join(week_cols)}"
event_model = smf.ols(formula, data=df).fit()

print("\n\nEvent Study Coefficients (relative to week 14):")
event_results = []
for w in range(-9, 8):
    if w == -1:
        event_results.append({'week': w, 'coef': 0, 'std_err': 0, 'ci_low': 0, 'ci_high': 0})
    else:
        coef = event_model.params.get(f'week_{w}', 0)
        se = event_model.bse.get(f'week_{w}', 0)
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        event_results.append({'week': w, 'coef': coef, 'std_err': se, 'ci_low': ci_low, 'ci_high': ci_high})

event_df = pd.DataFrame(event_results)
print(event_df)

# Calculate average pre-trend and post-treatment effects
pre_trend_avg = event_df[event_df['week'] < -1]['coef'].mean()
post_treatment_avg = event_df[event_df['week'] >= 0]['coef'].mean()
print(f"\nAverage pre-treatment coefficient (weeks 6-13): {pre_trend_avg:.3f}")
print(f"Average post-treatment coefficient (weeks 15-22): {post_treatment_avg:.3f}")
