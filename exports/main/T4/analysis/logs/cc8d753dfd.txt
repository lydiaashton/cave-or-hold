import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run the DiD regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is our treatment effect

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Also run with robust standard errors clustered by branch
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Regression with Clustered Standard Errors (by branch) ===")
print(model_clustered.summary())

# Check the treatment effect coefficient
treatment_effect = model_clustered.params['treatment']
treatment_se = model_clustered.bse['treatment']
treatment_pvalue = model_clustered.pvalues['treatment']

print(f"\nTreatment effect estimate: {treatment_effect:.2f}")
print(f"Standard error: {treatment_se:.2f}")
print(f"95% CI: [{treatment_effect - 1.96*treatment_se:.2f}, {treatment_effect + 1.96*treatment_se:.2f}]")
print(f"P-value: {treatment_pvalue:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Let's look at the change from pre to post for each group more carefully
print("=== Pre vs Post Analysis ===")

# Pre period (weeks 6-14)
pre = df[df['week'] < 15]
# Post period (weeks 15-22)
post = df[df['week'] >= 15]

# For switched branches
switched_pre = pre[pre['switched_branch'] == 1]['service_score']
switched_post = post[post['switched_branch'] == 1]['service_score']

# For control branches
control_pre = pre[pre['switched_branch'] == 0]['service_score']
control_post = post[post['switched_branch'] == 0]['service_score']

print(f"Switched branches - Pre: mean={switched_pre.mean():.2f}, n={len(switched_pre)}")
print(f"Switched branches - Post: mean={switched_post.mean():.2f}, n={len(switched_post)}")
print(f"  Change: {switched_post.mean() - switched_pre.mean():.2f}")

print(f"\nControl branches - Pre: mean={control_pre.mean():.2f}, n={len(control_pre)}")
print(f"Control branches - Post: mean={control_post.mean():.2f}, n={len(control_post)}")
print(f"  Change: {control_post.mean() - control_pre.mean():.2f}")

print(f"\nDiD estimate: {(switched_post.mean() - switched_pre.mean()) - (control_post.mean() - control_pre.mean()):.2f}")

# Let's also do a t-test on the treatment effect
# We can look at branch-level changes
branch_changes = df.groupby(['branch_id', 'switched_branch']).apply(
    lambda x: pd.Series({
        'pre_mean': x[x['week'] < 15]['service_score'].mean(),
        'post_mean': x[x['week'] >= 15]['service_score'].mean(),
        'change': x[x['week'] >= 15]['service_score'].mean() - x[x['week'] < 15]['service_score'].mean()
    })
).reset_index()

print("\n=== Branch-Level Changes ===")
print(branch_changes.groupby('switched_branch')['change'].agg(['mean', 'std', 'count']))

# T-test comparing changes between groups
from scipy import stats
switched_changes = branch_changes[branch_changes['switched_branch'] == 1]['change']
control_changes = branch_changes[branch_changes['switched_branch'] == 0]['change']

t_stat, p_val = stats.ttest_ind(switched_changes, control_changes, equal_var=False)
print(f"\nTwo-sample t-test on branch-level changes:")
print(f"  t-statistic: {t_stat:.3f}")
print(f"  p-value: {p_val:.4f}")
print(f"  Mean change (switched): {switched_changes.mean():.2f}")
print(f"  Mean change (control): {control_changes.mean():.2f}")
print(f"  Difference: {switched_changes.mean() - control_changes.mean():.2f}")
