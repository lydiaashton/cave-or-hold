import pandas as pd
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Calculate means by group and period
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).reset_index()
print("=== Group Means by Period ===")
print(summary)

# Calculate the DiD estimate manually
# Treatment before
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
# Treatment after
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
# Control before
ctrl_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
# Control after
ctrl_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print("\n=== Manual DiD Calculation ===")
print(f"Treatment Before: {treat_before:.2f}")
print(f"Treatment After: {treat_after:.2f}")
print(f"Control Before: {ctrl_before:.2f}")
print(f"Control After: {ctrl_after:.2f}")

treatment_change = treat_after - treat_before
control_change = ctrl_after - ctrl_before
did_estimate = treatment_change - control_change

print(f"\nTreatment change: {treatment_change:.2f}")
print(f"Control change: {control_change:.2f}")
print(f"DiD Estimate (Treatment - Control): {did_estimate:.2f}")

# Create a visualization
plt.figure(figsize=(10, 6))

# Calculate weekly averages by group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()

# Plot
for group in [0, 1]:
    group_data = weekly_avg[weekly_avg['switched_branch'] == group]
    label = 'Control (Old Procedure)' if group == 0 else 'Treatment (New Procedure)'
    plt.plot(group_data['week'], group_data['service_score'], marker='o', linewidth=2, label=label)

plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')
plt.xlabel('Week', fontsize=12)
plt.ylabel('Service Score', fontsize=12)
plt.title('Weekly Service Scores by Group', fontsize=14)
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('service_scores_by_week.png', dpi=150)
print("\nPlot saved as service_scores_by_week.png")

import pandas as pd
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Check pre-trends: compare trends before week 15
pre_data = df[df['week'] < 15]
weekly_pre = pre_data.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()

# Check post-trends
post_data = df[df['week'] >= 15]
weekly_post = post_data.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()

# Create a pre-trend visualization
plt.figure(figsize=(12, 6))

for group in [0, 1]:
    group_data = weekly_pre[weekly_pre['switched_branch'] == group]
    label = 'Control (Old Procedure)' if group == 0 else 'Treatment (New Procedure)'
    plt.plot(group_data['week'], group_data['service_score'], marker='o', linewidth=2, label=label)

plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')
plt.xlabel('Week', fontsize=12)
plt.ylabel('Service Score', fontsize=12)
plt.title('Pre-Switch Trends (Weeks 6-14)', fontsize=14)
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 15))
plt.tight_layout()
plt.savefig('pre_trends.png', dpi=150)
print("Pre-trends plot saved")

# Let's also run a regression with week fixed effects to control for time trends
df['week_str'] = df['week'].astype(str)
model_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD with Week Fixed Effects ===")
print(model_fe.summary())

# Also check if there's heterogeneity - maybe some branches benefited more than others
print("\n=== Treatment Effect by Branch (Exploratory) ===")
treatment_branches = df[df['switched_branch']==1]['branch_id'].unique()
branch_effects = []
for branch in treatment_branches:
    branch_data = df[df['branch_id']==branch]
    before = branch_data[branch_data['after_switch']==0]['service_score'].mean()
    after = branch_data[branch_data['after_switch']==1]['service_score'].mean()
    branch_effects.append({'branch_id': branch, 'change': after - before})

branch_effects_df = pd.DataFrame(branch_effects)
print(branch_effects_df.describe())
print("\nBranches with positive change:", (branch_effects_df['change'] > 0).sum())
print("Branches with negative change:", (branch_effects_df['change'] < 0).sum())
