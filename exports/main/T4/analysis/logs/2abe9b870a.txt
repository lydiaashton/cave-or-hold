import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression without controls
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== DiD Regression (No Controls) ===")
print(model1.summary().tables[1])

# Run DiD regression with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\n=== DiD Regression (With Controls) ===")
print(model2.summary().tables[1])

# Clustered standard errors by branch
from statsmodels.stats.sandwich_covariance import cov_cluster
model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
cov_clustered = cov_cluster(model3, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))
print("\n=== Coefficients with Clustered SE (by branch) ===")
coef_df = pd.DataFrame({
    'coef': model3.params,
    'clustered_se': se_clustered,
    't_stat': model3.params / se_clustered,
    'p_value': 2 * (1 - sm.stats.norm.cdf(np.abs(model3.params / se_clustered)))
})
print(coef_df)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster
from scipy import stats

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with controls
model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()

se_clustered = np.sqrt(np.diag(cov_clustered))

    'clustered_se': se_clustered,

})
print(coef_df.round(4))

# Also get R-squared
print(f"\nR-squared: {model3.rsquared:.4f}")
print(f"Adjusted R-squared: {model3.rsquared_adj:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check parallel trends: compare trends in pre-period (weeks 6-14)
pre_period = df[df['week'] < 15].copy()

# Run regression on pre-period only to check if trends were different
# Create a continuous week variable centered
pre_period['week_centered'] = pre_period['week'] - 10  # center around middle of pre-period

# Test for differential pre-trends
model_pre = ols('service_score ~ switched_branch * week_centered', data=pre_period).fit()
print("=== Pre-Trend Test (Weeks 6-14 only) ===")
print(model_pre.summary().tables[1])

# The interaction term tests whether the slopes were different before treatment
# If not significant, parallel trends assumption is supported

# Also let's look at average pre-period scores by group
print("\n=== Average Pre-Period Scores (Weeks 6-14) ===")
pre_avg = pre_period.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print(pre_avg)

# Event study: estimate effect for each week relative to switch
df['relative_week'] = df['week'] - 14  # week 14 is last pre-switch week, so this makes week 15 = 1
df['post_only'] = (df['relative_week'] > 0).astype(int)

# Create dummies for each post-switch week
post_weeks = df[df['relative_week'] > 0]['relative_week'].unique()
print(f"\nPost-switch weeks: {sorted(post_weeks)}")

# Let's do a simpler event study - average effect in early vs late post period
df['early_post'] = ((df['relative_week'] > 0) & (df['relative_week'] <= 4)).astype(int)
df['late_post'] = (df['relative_week'] > 4).astype(int)

model_event = ols('service_score ~ switched_branch + after_switch + switched_branch * early_post + switched_branch * late_post + staffing_index + visits_index', data=df).fit()
print("\n=== Event Study: Early vs Late Post-Period Effects ===")

# Pre vs Post for each group
print("\n=== SERVICE SCORES BY GROUP AND PERIOD ===")
for switch in [0, 1]:
    group_name = "Control (No Switch)" if switch == 0 else "Treatment (Switched)"
    pre = df[(df['switched_branch']==switch) & (df['after_switch']==0)]['service_score']
    post = df[(df['switched_branch']==switch) & (df['after_switch']==1)]['service_score']
    print(f"\n{group_name}:")

    print(f"  Post-switch (weeks 15-22): mean={post.mean():.2f}, std={post.std():.2f}, n={len(post)}")

print("P-value: 0.094 (marginally significant at 10% level)")
print("95% CI approximately: -0.38 to +4.79 points")

ci_lower = coef - 1.96 * se
ci_upper = coef + 1.96 * se

# Effect size relative to baseline
baseline = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
effect_size_pct = (2.2074 / baseline) * 100
print(f"\nEffect size: {effect_size_pct:.1f}% improvement relative to treatment group baseline")
