import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with cluster-robust standard errors
# Model: service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*treatment + error
# beta3 is the DiD estimate

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

# Get cluster-robust standard errors (clustered by branch)
cov_cluster = model.get_robustcov_results(cov_type='cluster', clusters=df['branch_id'])

print("=== DIFFERENCE-IN-DIFFERENCES REGRESSION ===")
print(cov_cluster.summary())

print("\n\n=== KEY COEFFICIENTS ===")
print(f"Intercept (control before): {cov_cluster.params['Intercept']:.3f}")
print(f"switched_branch (treatment vs control before): {cov_cluster.params['switched_branch']:.3f}")
print(f"after_switch (control after vs before): {cov_cluster.params['after_switch']:.3f}")
print(f"treatment (DiD effect): {cov_cluster.params['treatment']:.3f}")
print(f"\nTreatment effect p-value: {cov_cluster.pvalues['treatment']:.6f}")
print(f"Treatment effect 95% CI: [{cov_cluster.conf_int().loc['treatment', 0]:.3f}, {cov_cluster.conf_int().loc['treatment', 1]:.3f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
cov = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("=" * 60)
print("FINAL ANALYSIS SUMMARY")
print("=" * 60)

print("\n1. DATA OVERVIEW")
print("-" * 40)
print(f"Total observations: {len(df):,}")
print(f"Number of branches: {df['branch_id'].nunique()}")
print(f"  - Switched to new procedure: {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"  - Kept old procedure: {df[df['switched_branch']==0]['branch_id'].nunique()}")
print(f"Weeks observed: {df['week'].min()} to {df['week'].max()}")
print(f"Switch occurred: Week 15")

print("\n2. SERVICE SCORE SUMMARY")
print("-" * 40)
pre = df[df['week'] <= 14]
post = df[df['week'] >= 15]

print("Before switch (weeks 6-14):")
print(f"  Control group mean: {pre[pre['switched_branch']==0]['service_score'].mean():.2f}")
print(f"  Treatment group mean: {pre[pre['switched_branch']==1]['service_score'].mean():.2f}")
print(f"  Difference: {pre[pre['switched_branch']==1]['service_score'].mean() - pre[pre['switched_branch']==0]['service_score'].mean():.2f}")

print("\nAfter switch (weeks 15-22):")
print(f"  Control group mean: {post[post['switched_branch']==0]['service_score'].mean():.2f}")
print(f"  Treatment group mean: {post[post['switched_branch']==1]['service_score'].mean():.2f}")
print(f"  Difference: {post[post['switched_branch']==1]['service_score'].mean() - post[post['switched_branch']==0]['service_score'].mean():.2f}")

print("\n3. DIFFERENCE-IN-DIFFERENCES ESTIMATE")
print("-" * 40)
control_change = post[post['switched_branch']==0]['service_score'].mean() - pre[pre['switched_branch']==0]['service_score'].mean()
treatment_change = post[post['switched_branch']==1]['service_score'].mean() - pre[pre['switched_branch']==1]['service_score'].mean()
did = treatment_change - control_change

print(f"Control group change: {control_change:+.2f}")
print(f"Treatment group change: {treatment_change:+.2f}")
print(f"DiD estimate (treatment effect): {did:+.2f}")
print(f"\nRegression estimate: {cov.params[3]:.2f}")
print(f"95% CI: [{cov.conf_int()[3, 0]:.2f}, {cov.conf_int()[3, 1]:.2f}]")
print(f"P-value: {cov.pvalues[3]:.2e}")

print("\n4. PRE-TREND ANALYSIS")
print("-" * 40)
# Check if trends were parallel before the switch
before_data = df[df['week'] <= 14].copy()
before_data['week_c'] = before_data['week'] - 10
model_pre = ols('service_score ~ switched_branch * week_c', data=before_data).fit()
cov_pre = model_pre.get_robustcov_results(cov_type='cluster', groups=before_data['branch_id'])

print(f"Pre-period trend interaction coefficient: {cov_pre.params[3]:.3f}")
print(f"P-value: {cov_pre.pvalues[3]:.2e}")
print("Note: Significant interaction suggests different pre-trends between groups")

print("\n5. WEEKLY DIFFERENCES (Treatment - Control)")
print("-" * 40)
weekly = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
weekly_diff = weekly[1] - weekly[0]
for week in weekly_diff.index:
    marker = " <-- SWITCH" if week == 15 else ""
    print(f"  Week {week:2d}: {weekly_diff[week]:6.2f}{marker}")

print("\n" + "=" * 60)
