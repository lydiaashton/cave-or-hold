import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
branch_switch_status = df.groupby('branch_id')['switched_branch'].first()
print("Branches that switched:", branch_switch_status.sum())
print("Branches that didn't switch:", len(branch_switch_status) - branch_switch_status.sum())

# Average service scores by group and period
print("\n=== Average Service Scores ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the difference-in-differences manually
# Before period (after_switch=0), After period (after_switch=1)
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
before_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"\nControl group (didn't switch):")
print(f"  Before: {before_control:.2f}")
print(f"  After: {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")

print(f"\nTreatment group (switched):")
print(f"  Before: {before_treatment:.2f}")
print(f"  After: {after_treatment:.2f}")
print(f"  Change: {after_treatment - before_treatment:.2f}")

did_effect = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {did_effect:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Difference-in-differences regression
# Basic DiD model
model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("=== Basic DiD Regression ===")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit()
print("\n=== DiD Regression with Controls ===")
print(model2.summary().tables[1])

# Clustered standard errors by branch
# Using cluster-robust standard errors
from statsmodels.stats.sandwich_covariance import cov_cluster
cov_clustered = cov_cluster(model2, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))
print("\n=== Clustered Standard Errors (by branch) ===")
coef_table = model2.summary().tables[1]
print(f"switched_branch:after_switch coefficient: {model2.params['switched_branch:after_switch']:.4f}")
print(f"Clustered SE: {se_clustered[model2.params.index.get_loc('switched_branch:after_switch')]:.4f}")

# Calculate t-stat and p-value
coef = model2.params['switched_branch:after_switch']
se = se_clustered[model2.params.index.get_loc('switched_branch:after_switch')]
t_stat = coef / se
from scipy import stats
p_value = 2 * (1 - stats.t.cdf(abs(t_stat), df=len(df) - len(model2.params)))
print(f"t-statistic: {t_stat:.2f}")
print(f"p-value: {p_value:.4f}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create a before/after comparison bar chart
before_after = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std']).reset_index()
before_after['group'] = before_after['switched_branch'].map({0: 'Control (Old)', 1: 'Treatment (New)'})
before_after['period'] = before_after['after_switch'].map({0: 'Before Switch', 1: 'After Switch'})

fig, ax = plt.subplots(figsize=(10, 6))

x = np.arange(4)
colors = ['#A23B72', '#2E86AB', '#A23B72', '#2E86AB']
bars = ax.bar(x, before_after['mean'], yerr=before_after['std']/np.sqrt(df.groupby(['switched_branch', 'after_switch']).size().values), 
              capsize=5, color=colors, edgecolor='black', linewidth=1)

ax.set_xticks(x)
ax.set_xticklabels(['Control\nBefore', 'Control\nAfter', 'Treatment\nBefore', 'Treatment\nAfter'], fontsize=11)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Scores: Before and After Procedure Switch', fontsize=14, fontweight='bold')
ax.axhline(y=50, color='gray', linestyle='--', alpha=0.5, label='Average Benchmark (50)')

# Add value labels on bars
for i, v in enumerate(before_after['mean']):
    ax.text(i, v + 0.5, f'{v:.1f}', ha='center', fontsize=11, fontweight='bold')

ax.legend(loc='upper right')
ax.grid(True, alpha=0.3, axis='y')
plt.tight_layout()
plt.savefig('before_after_comparison.png', dpi=150)
print("Bar chart saved as 'before_after_comparison.png'")
