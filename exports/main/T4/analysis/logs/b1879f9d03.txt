import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
switched_counts = df.groupby('branch_id')['switched_branch'].first().value_counts()
print("Branches by switch status:")
print(switched_counts)
print(f"\nTotal branches: {df['branch_id'].nunique()}")
print(f"Switched branches: {switched_counts.get(1, 0)}")
print(f"Non-switched branches: {switched_counts.get(0, 0)}")

# Average service scores by group and period
print("\n\n=== Average Service Scores ===")
before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

print("\nBefore switch (weeks 6-14):")
print(before.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

print("\nAfter switch (weeks 15-22):")
print(after.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

# Calculate the differences
print("\n\n=== Difference-in-Differences Calculation ===")
# Control group (didn't switch)
control_before = before[before['switched_branch'] == 0]['service_score'].mean()
control_after = after[after['switched_branch'] == 0]['service_score'].mean()
control_diff = control_after - control_before

# Treatment group (switched)
treat_before = before[before['switched_branch'] == 1]['service_score'].mean()
treat_after = after[after['switched_branch'] == 1]['service_score'].mean()
treat_diff = treat_after - treat_before

print(f"Control group (no switch): {control_before:.2f} -> {control_after:.2f} (change: {control_diff:.2f})")
print(f"Treatment group (switched): {treat_before:.2f} -> {treat_after:.2f} (change: {treat_diff:.2f})")
print(f"\nDifference-in-Differences estimate: {treat_diff - control_diff:.2f}")

# Weekly trends to visualize
print("\n\n=== Weekly Average Service Scores ===")
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(weekly_avg)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate

df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# Now with branch fixed effects (to control for time-invariant branch characteristics)
print("\n\n=== DiD with Branch Fixed Effects ===")
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [col for col in df_with_fe.columns if col.startswith('branch_')]
formula = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(branch_cols)
model_fe = ols(formula, data=df_with_fe).fit()
print(f"R-squared: {model_fe.rsquared:.4f}")
print(f"\nKey coefficients:")
print(model_fe.params[['switched_branch', 'after_switch', 'treatment']])
print(f"\nStandard errors:")
print(model_fe.bse[['switched_branch', 'after_switch', 'treatment']])
print(f"\nP-values:")
print(model_fe.pvalues[['switched_branch', 'after_switch', 'treatment']])

# Cluster-robust standard errors by branch
print("\n\n=== DiD with Cluster-Robust SE (by branch) ===")
model_cluster = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_cluster.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Event study approach: create week dummies interacted with treatment
print("=== Event Study Analysis ===")
# Create relative week indicators (weeks 6-14 are before, 15-22 are after)
# Use week 14 as the reference period (just before switch)
df['relative_week'] = df['week'] - 14  # week 14 = 0, week 15 = 1, etc.

# Create indicators for each week (excluding week 14 as reference)
for w in range(6, 23):
    if w != 14:
        df[f'week_{w}'] = (df['week'] == w).astype(int)
        df[f'week_{w}_treat'] = df[f'week_{w}'] * df['switched_branch']

# Build event study model (weeks 6-13 and 15-22, with week 14 as reference)
week_cols = [f'week_{w}_treat' for w in range(6, 23) if w != 14]
formula = 'service_score ~ switched_branch + ' + ' + '.join(week_cols)
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event study coefficients (treatment effect by week, relative to week 14):")
results = []
for w in range(6, 23):
    if w == 14:
        results.append((w, 0.0, 0.0, 1.0, 'reference'))
    else:
        coef = event_model.params[f'week_{w}_treat']
        se = event_model.bse[f'week_{w}_treat']
        pval = event_model.pvalues[f'week_{w}_treat']
        results.append((w, coef, se, pval, 'significant' if pval < 0.05 else 'not sig'))

for r in results:
    print(f"Week {r[0]:2d}: coef={r[1]:6.2f}, SE={r[2]:.2f}, p={r[3]:.4f} ({r[4]})")

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
weeks = [r[0] for r in results]

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Simpler approach: directly model the gap with week fixed effects
# Create week dummies and interact with treatment
print("=== Cleaner Event Study ===")

# Create relative week from switch (week 15 is first post-switch week)
df['weeks_from_switch'] = df['week'] - 14  # week 14 = 0 (last pre), week 15 = 1 (first post)

# Create indicators for each week relative to switch, interacted with treatment
# Use week 0 (week 14) as reference
for w in range(-8, 9):  # week 6 to 22 relative to week 14
    if w != 0:
        df[f'rel_week_{w}'] = (df['weeks_from_switch'] == w).astype(int)
        df[f'rel_week_{w}_treat'] = df[f'rel_week_{w}'] * df['switched_branch']

# Model: service_score ~ week FE + treatment interactions (relative to week 0)
rel_week_cols = [f'rel_week_{w}_treat' for w in range(-8, 9) if w != 0]
week_fe = [f'rel_week_{w}' for w in range(-8, 9) if w != 0]
formula = 'service_score ~ switched_branch + ' + ' + '.join(week_fe) + ' + ' + ' + '.join(rel_week_cols)
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract and display coefficients
print("\nTreatment effects by week (relative to week 14 baseline):")
results = []
for w in range(-8, 9):
    if w == 0:
        results.append((w + 14, 0.0, 0.0, 'reference'))
    else:
        coef = event_model.params[f'rel_week_{w}_treat']
        se = event_model.bse[f'rel_week_{w}_treat']
        pval = event_model.pvalues[f'rel_week_{w}_treat']
        results.append((w + 14, coef, se, 'sig' if pval < 0.05 else 'ns'))
        print(f"Week {w + 14:2d} (rel {w:+2d}): coef = {coef:6.2f} (SE={se:.2f}, {pval:.4f})")

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
weeks = [r[0] for r in results]
coefs = [r[1] for r in results]
ses = [r[2] for r in results]

colors = ['blue' if w < 14 else 'green' for w in weeks]
ax.errorbar(weeks, coefs, yerr=1.96*np.array(ses), fmt='o', capsize=4, linewidth=2, markersize=8, color=colors)
ax.axhline(y=0, color='gray', linestyle='-', linewidth=1)
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch implemented')
ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Treatment-Control Gap (relative to week 14)', fontsize=12)
ax.set_title('Event Study: Evolution of Treatment-Control Gap', fontsize=14)
ax.legend(fontsize=11)
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study_clean.png', dpi=150)
print("\nEvent study plot saved as event_study_clean.png")

# Calculate average pre and post effects
pre_effects = [r[1] for r in results if r[0] < 14]
post_effects = [r[1] for r in results if r[0] > 14]
print(f"\nAverage pre-switch effect (weeks 6-13): {np.mean(pre_effects):.2f}")
print(f"Average post-switch effect (weeks 15-22): {np.mean(post_effects):.2f}")
print(f"Change in effect after switch: {np.mean(post_effects) - np.mean(pre_effects):.2f}")

# Let's also look at the raw weekly gaps again to confirm
print("\n\n=== Raw Weekly Gaps (Treatment - Control) ===")
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly['gap'] = weekly[1] - weekly[0]

print(f"Change from pre to post: {post_gaps - pre_gaps:.2f}")

import pandas as pd
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the main visualization
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Left panel: Service scores over time
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weeks = weekly.index

ax1 = axes[0]
ax1.plot(weeks, weekly[0], 'o-', label='Control Branches (No Switch)', linewidth=2.5, markersize=6, color='#2E86AB')
ax1.plot(weeks, weekly[1], 's-', label='Treatment Branches (Switched)', linewidth=2.5, markersize=6, color='#A23B72')
ax1.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Procedure Change')
ax1.axvspan(15, 22, alpha=0.1, color='green', label='Post-Change Period')
ax1.set_xlabel('Week', fontsize=12)
ax1.set_ylabel('Average Service Score', fontsize=12)
ax1.set_title('Service Scores by Branch Group', fontsize=14, fontweight='bold')
ax1.legend(fontsize=10, loc='upper left')
ax1.grid(True, alpha=0.3)
ax1.set_xlim(5.5, 22.5)

# Right panel: Treatment-Control gap over time
ax2 = axes[1]
gap = weekly[1] - weekly[0]
ax2.plot(weeks, gap, 'd-', linewidth=2.5, markersize=8, color='#06A77D')
ax2.axvline(x=14.5, color='gray', linestyle='--', linewidth=2)

# Add pre-trend line
pre_weeks = weeks[weeks < 14]
pre_gaps = gap[weeks < 14]
slope, intercept = np.polyfit(pre_weeks.values, pre_gaps.values, 1)
extrapolated = slope * weeks.values + intercept
ax2.plot(weeks, extrapolated, 'r--', linewidth=2, label='Extrapolated Pre-Trend')

ax2.fill_between(weeks, extrapolated - 0.5, extrapolated + 0.5, alpha=0.2, color='red')
ax2.axvspan(15, 22, alpha=0.1, color='green')
ax2.set_xlabel('Week', fontsize=12)
ax2.set_ylabel('Treatment - Control Gap', fontsize=12)
ax2.set_title('Evolution of Performance Gap', fontsize=14, fontweight='bold')
ax2.legend(fontsize=10)
ax2.grid(True, alpha=0.3)
ax2.set_xlim(5.5, 22.5)

plt.tight_layout()
plt.savefig('analysis_summary.png', dpi=150, bbox_inches='tight')
print("Summary visualization saved as analysis_summary.png")
