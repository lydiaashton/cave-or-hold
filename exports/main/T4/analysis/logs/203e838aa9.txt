import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Calculate average service score by week and switch status
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_score']

# Pivot for easier plotting
pivot_weekly = weekly_avg.pivot(index='week', columns='switched_branch', values='avg_score')
pivot_weekly.columns = ['Control (No Switch)', 'Treatment (Switched)']

print("Weekly averages by group:")
print(pivot_weekly)

# Check pre-trends statistically (weeks 6-14)
pre_data = df[df['week'] < 15].copy()
pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()

# Test for differential pre-trends
model_pretrend = ols('service_score ~ switched_branch * week_centered', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print("\n\nPre-trend test (interaction of switch status with time):")
print(model_pretrend.summary())

# Create the plot
plt.figure(figsize=(10, 6))
plt.plot(pivot_weekly.index, pivot_weekly['Control (No Switch)'], 'o-', label='Control (No Switch)', linewidth=2, markersize=8)
plt.plot(pivot_weekly.index, pivot_weekly['Treatment (Switched)'], 's-', label='Treatment (Switched)', linewidth=2, markersize=8)
plt.axvline(x=14.5, color='red', linestyle='--', label='Switch occurred (week 15)')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Scores by Week: Treatment vs Control Groups')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('service_scores_trend.png', dpi=150, bbox_inches='tight')
print("\nPlot saved as service_scores_trend.png")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['did'] = df['switched_branch'] * df['after_switch']

# Main model with controls
model = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Main DiD Results:")
print(f"DiD Coefficient: {model.params['did']:.2f}")
print(f"Standard Error: {model.bse['did']:.2f}")
print(f"95% CI: [{model.conf_int().loc['did', 0]:.2f}, {model.conf_int().loc['did', 1]:.2f}]")
print(f"P-value: {model.pvalues['did']:.3f}")

# Calculate effect size relative to baseline
baseline = df[df['after_switch']==0]['service_score'].mean()
effect_size_pct = (model.params['did'] / baseline) * 100
print(f"\nBaseline average service score: {baseline:.2f}")
print(f"Effect size: {effect_size_pct:.1f}% increase")

# Check if effect is consistent across post-period weeks
post_data = df[df['after_switch']==1].copy()
post_data['weeks_since_switch'] = post_data['week'] - 14  # weeks since switch (week 15 = 1, etc.)

# Check effect by early vs late post-period
post_data['late_post'] = (post_data['weeks_since_switch'] > 4).astype(int)  # weeks 19-22 vs 15-18

model_late = ols('service_score ~ switched_branch + after_switch + did + late_post + switched_branch:late_post', 
                 data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nEffect persistence check:")
print(model_late.summary().tables[1])

# Summary stats for the board
print("\n\n=== SUMMARY STATISTICS FOR BOARD ===")
print(f"Total branches: {df['branch_id'].nunique()}")
print(f"Branches that switched: {df['switched_branch'].sum()}")

print(f"\nPre-switch average (switched branches): {df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean():.1f}")
print(f"Post-switch average (switched branches): {df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean():.1f}")
print(f"Pre-switch average (control branches): {df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean():.1f}")
print(f"Post-switch average (control branches): {df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean():.1f}")
