import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate mean service scores by group and period
print("=== Mean Service Scores by Group and Period ===")
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='after_switch', 
                       aggfunc='mean')
print(pivot)

# Calculate the DiD estimate manually
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"\nControl (non-switched) branches:")
print(f"  Before: {control_before:.3f}")
print(f"  After: {control_after:.3f}")
print(f"  Change: {control_after - control_before:.3f}")

print(f"\nTreatment (switched) branches:")
print(f"  Before: {treatment_before:.3f}")
print(f"  After: {treatment_after:.3f}")
print(f"  Change: {treatment_after - treatment_before:.3f}")

did_estimate = (treatment_after - treatment_before) - (control_after - control_before)
print(f"\nDifference-in-Differences estimate: {did_estimate:.3f}")

# Run regression
print("\n=== Regression Analysis ===")

# Basic DiD model
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nBasic DiD Model:")
print(model1.summary())

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\nDiD Model with Controls:")
print(model2.summary())

# Cluster-robust standard errors by branch
model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nDiD Model with Controls (Cluster-robust SE by branch):")
print(model3.summary())

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Test parallel trends statistically - compare trends before week 15
pre_switch = df[df['week'] < 15].copy()
pre_switch['week_centered'] = pre_switch['week'] - pre_switch['week'].mean()

# Regression with week interaction for pre-period only
model_pre = ols('service_score ~ switched_branch * week_centered', data=pre_switch).fit(cov_type='cluster', cov_kwds={'groups': pre_switch['branch_id']})
print("=== Parallel Trends Test (Pre-switch period) ===")
print("Testing if trends were parallel before the switch...")
print(model_pre.summary())

# Also test with week as categorical to see if there are any pre-trends
pre_switch['week_cat'] = pre_switch['week'].astype('category')

# Compare average differences in pre-period

print(f"\nAverage pre-switch difference: {pre_diff['diff'].mean():.3f}")
print(f"Std dev of pre-switch differences: {pre_diff['diff'].std():.3f}")

# Post-switch differences
post_switch = df[df['week'] >= 15].copy()
post_diff = post_switch.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
post_diff['diff'] = post_diff[1] - post_diff[0]
print("\n=== Post-switch weekly differences ===")
print(post_diff[['diff']])
print(f"\nAverage post-switch difference: {post_diff['diff'].mean():.3f}")
print(f"Std dev of post-switch differences: {post_diff['diff'].std():.3f}")

# Event study - estimate effect for each week relative to switch
df['relative_week'] = df['week'] - 15  # week 15 is period 0
df['relative_week'] = df['relative_week'].apply(lambda x: max(x, -9))  # cap at -9 for week 6

# Create dummies for each relative week
event_study_df = df.copy()
for w in range(-9, 8):  # weeks -9 to +7
    event_study_df[f'week_{w}'] = (event_study_df['relative_week'] == w).astype(int)

# Run event study regression (omitting week -1 as reference)
event_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'switched_branch * {v}' for v in event_vars])
# Simplified: just interact treatment with week dummies
event_formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join([f'treatment * week_{w}' for w in range(-9, 8) if w != -1])

# Actually, let's do a simpler event study
event_study = df.copy()
event_study['treatment'] = event_study['switched_branch'] * event_study['after_switch']

# Create week dummies interacted with treatment
for w in range(-9, 8):
    event_study[f'treat_week_{w}'] = ((event_study['relative_week'] == w) & (event_study['switched_branch'] == 1)).astype(int)

# Reference period is week -1
treat_weeks = [f'treat_week_{w}' for w in range(-9, 8) if w != -1]
event_model = ols(f'service_score ~ switched_branch + after_switch + {" + ".join(treat_weeks)}', 
                  data=event_study).fit(cov_type='cluster', cov_kwds={'groups': event_study['branch_id']})

print("\n=== Event Study Results ===")
print("Coefficients for treatment effects by week (reference: week -1):")
for w in range(-9, 8):
    if w != -1:
        coef = event_model.params[f'treat_week_{w}']
        se = event_model.bse[f'treat_week_{w}']
        pval = event_model.pvalues[f'treat_week_{w}']
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"Week {w:+d}: {coef:6.3f} (SE: {se:.3f}, p={pval:.3f}) {sig}")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Simple event study - create relative week and interact with treatment
df['relative_week'] = df['week'] - 15  # week 15 is period 0

# Create treatment indicator
df['treatment'] = df['switched_branch'] * df['after_switch']

# Create dummies for each relative week (for treatment group only)
event_data = df.copy()
for w in range(-9, 8):  # weeks -9 to +7
    col_name = f'tw{abs(w)}{"m" if w < 0 else "p" if w > 0 else "0"}'
    event_data[col_name] = ((event_data['relative_week'] == w) & (event_data['switched_branch'] == 1)).astype(int)

# Reference period is week -1 (just before switch)
# Build formula
treat_cols = [f'tw{abs(w)}{"m" if w < 0 else "p" if w > 0 else "0"}' for w in range(-9, 8) if w != -1]
formula = f'service_score ~ switched_branch + after_switch + {" + ".join(treat_cols)}'

event_model = ols(formula, data=event_data).fit(cov_type='cluster', cov_kwds={'groups': event_data['branch_id']})

print("=== Event Study Results ===")
print("Coefficients for treatment effects by week (reference: week -1):")

# Map back to week numbers
week_map = {}
for w in range(-9, 8):
    if w != -1:
        col_name = f'tw{abs(w)}{"m" if w < 0 else "p" if w > 0 else "0"}'
        week_map[col_name] = w

results = []
for col, w in week_map.items():
    coef = event_model.params[col]
    se = event_model.bse[col]
    pval = event_model.pvalues[col]
    results.append((w, coef, se, pval))

results.sort(key=lambda x: x[0])

for w, coef, se, pval in results:
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"Week {w:+d}: {coef:6.3f} (SE: {se:.3f}, p={pval:.3f}) {sig}")

# Plot event study
plt.figure(figsize=(12, 6))
weeks = [w for w, _, _, _ in results]
coefs = [coef for _, coef, _, _ in results]
ses = [se for _, _, se, _ in results]

# Add reference point at week -1
weeks_plot = [-1] + weeks
coefs_plot = [0] + coefs
ses_plot = [0] + ses

plt.title('Event Study: Treatment Effect by Week', fontsize=14, fontweight='bold')
plt.xticks(range(-9, 8))
plt.grid(True, alpha=0.3)
plt.legend()
plt.tight_layout()
plt.savefig('event_study_plot.png', dpi=150)

print(f"Total observations: {len(df)}")
print(f"Branches: {df['branch_id'].nunique()}")
print(f"Switched branches: {df['switched_branch'].sum()} ({df['switched_branch'].mean()*100:.1f}%)")

import pandas as pd

import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Check if the effect varies over time (immediate vs sustained)
df['weeks_since_switch'] = df['week'] - 15
df['weeks_since_switch'] = df['weeks_since_switch'].clip(lower=0)  # 0 for pre-period

# Create early vs late post-period
df['late_post'] = ((df['week'] >= 18) & (df['after_switch'] == 1)).astype(int)
df['treatment_early'] = ((df['week'].between(15, 17)) & (df['switched_branch'] == 1)).astype(int)
df['treatment_late'] = ((df['week'] >= 18) & (df['switched_branch'] == 1)).astype(int)

model_time = ols('service_score ~ treatment_early + treatment_late + C(branch_id) + C(week)', 
                 data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Effect by Time Period ===")
print(f"Early post-period (weeks 15-17): {model_time.params['treatment_early']:.3f} (p={model_time.pvalues['treatment_early']:.3f})")
print(f"Late post-period (weeks 18-22): {model_time.params['treatment_late']:.3f} (p={model_time.pvalues['treatment_late']:.3f})")

# Robustness: exclude first week after switch (adjustment period)
df_robust = df[df['week'] != 15].copy()
df_robust['treatment'] = df_robust['switched_branch'] * df_robust['after_switch']
model_robust = ols('service_score ~ treatment + C(branch_id) + C(week)', 
                   data=df_robust).fit(cov_type='cluster', cov_kwds={'groups': df_robust['branch_id']})
print(f"\n=== Robustness (excluding week 15) ===")
print(f"Treatment effect: {model_robust.params['treatment']:.3f} (p={model_robust.pvalues['treatment']:.3f})")

# Check for any pre-trends with week dummies
pre_df = df[df['week'] < 15].copy()
pre_df['treatment'] = pre_df['switched_branch']  # In pre-period, this is just the group indicator

# Test if treatment group had different levels (they did - about 1.4 points higher)
# But the key is whether the TREND was different
pre_df['week_num'] = pre_df['week']
model_pretrend = ols('service_score ~ switched_branch + week_num + switched_branch:week_num + C(branch_id)', 
                     data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print(f"\n=== Pre-Trend Test ===")
print(f"Interaction (different trends): coef={model_pretrend.params['switched_branch:week_num']:.3f}, p={model_pretrend.pvalues['switched_branch:week_num']:.3f}")

# Final summary table
print("\n" + "="*60)
