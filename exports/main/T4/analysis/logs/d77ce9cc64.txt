import pandas as pd

# Load the data
df = pd.read_csv('branches.csv')

# Create relative week variable (weeks relative to switch at week 15)
df['relative_week'] = df['week'] - 15

# Event study: interact each week with treatment status
# Week 14 (relative_week = -1) will be the reference period
df['week_cat'] = df['week'].astype('category')
df['treatment_week'] = df['switched_branch'] * df['week'].astype(str)

# Create event study model
event_study = ols('service_score ~ C(week_cat) + C(week_cat):switched_branch + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract the interaction coefficients (treatment effect by week)
interaction_coefs = {}
interaction_ci = {}
for param in event_study.params.index:
    if 'week_cat' in param and 'switched_branch' in param:
        week_num = int(param.split('[')[1].split(']')[0].split(':')[0].replace('T.', ''))
        interaction_coefs[week_num] = event_study.params[param]
        ci = event_study.conf_int().loc[param]
        interaction_ci[week_num] = (ci[0], ci[1])

# Sort by week
weeks = sorted(interaction_coefs.keys())
coefs = [interaction_coefs[w] for w in weeks]
ci_lower = [interaction_ci[w][0] for w in weeks]
ci_upper = [interaction_ci[w][1] for w in weeks]

# Create event study plot
fig, ax = plt.subplots(figsize=(12, 6))
