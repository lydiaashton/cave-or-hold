import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch*after_switch) + error
# beta3 is the DiD estimator - the effect of the treatment

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Difference-in-Differences Regression Results:")
print("=" * 60)
print(model.summary())

# Also run with cluster-robust standard errors by branch
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nWith Cluster-Robust Standard Errors (by branch):")
print("=" * 60)
print(model_clustered.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate mean service score by week and switch status
weekly_means = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_means.columns = ['week', 'switched_branch', 'mean_score']

# Create the plot
fig, ax = plt.subplots(figsize=(12, 6))

# Plot for switched branches (treatment)
treatment = weekly_means[weekly_means['switched_branch'] == 1]
ax.plot(treatment['week'], treatment['mean_score'], 'b-o', label='Switched Branches (Treatment)', linewidth=2, markersize=6)

# Plot for non-switched branches (control)
control = weekly_means[weekly_means['switched_branch'] == 0]
ax.plot(control['week'], control['mean_score'], 'r-s', label='Non-Switched Branches (Control)', linewidth=2, markersize=6)

# Add vertical line at week 15 (switch point)
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch Implemented (Week 15)')

# Add labels and title
ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Mean Service Score', fontsize=12)
ax.set_title('Service Scores Over Time: Switched vs Non-Switched Branches', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=10)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

# Set y-axis to start from a reasonable value
ax.set_ylim(40, 60)

plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)

# Also print the weekly means for reference
print("\nWeekly mean service scores:")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main DiD model with cluster-robust SE
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract key results
treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pvalue = model.pvalues['treatment']
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se

# Calculate effect size (Cohen's d approximation)
# Using pooled standard deviation
pooled_std = df['service_score'].std()
effect_size = treatment_coef / pooled_std

print("=" * 70)
print("SUMMARY STATISTICS FOR BOARD REPORT")
print("=" * 70)

print(f"\nTreatment Effect Estimate: {treatment_coef:.2f} points")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {treatment_pvalue:.3f}")
print(f"Effect Size (Cohen's d): {effect_size:.3f}")

print("\n\nGroup Means:")
group_means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
group_means.columns = ['Before (Weeks 6-14)', 'After (Weeks 15-22)']
group_means.index = ['Non-Switched (Control)', 'Switched (Treatment)']
print(group_means)

print("\n\nChanges by Group:")
changes = group_means['After (Weeks 15-22)'] - group_means['Before (Weeks 6-14)']
print(changes)
print(f"\nDifference-in-Differences: {changes['Switched (Treatment)'] - changes['Non-Switched (Control)']:.2f} points")

print("\n\nSample Sizes:")
print(df.groupby(['switched_branch', 'after_switch']).size().unstack())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate mean and CI by week and switch status
weekly_stats = df.groupby(['week', 'switched_branch'])['service_score'].agg(['mean', 'std', 'count']).reset_index()
weekly_stats['se'] = weekly_stats['std'] / np.sqrt(weekly_stats['count'])
weekly_stats['ci_lower'] = weekly_stats['mean'] - 1.96 * weekly_stats['se']
weekly_stats['ci_upper'] = weekly_stats['mean'] + 1.96 * weekly_stats['se']

# Create figure with two panels
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 5))

# Panel 1: Time trends with confidence intervals
treatment = weekly_stats[weekly_stats['switched_branch'] == 1]
control = weekly_stats[weekly_stats['switched_branch'] == 0]

ax1.plot(treatment['week'], treatment['mean'], 'b-o', label='Switched Branches', linewidth=2, markersize=6)
ax1.fill_between(treatment['week'], treatment['ci_lower'], treatment['ci_upper'], alpha=0.2, color='blue')

ax1.plot(control['week'], control['mean'], 'r-s', label='Non-Switched Branches', linewidth=2, markersize=6)
ax1.fill_between(control['week'], control['ci_lower'], control['ci_upper'], alpha=0.2, color='red')

ax1.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch Point (Week 15)')
ax1.set_xlabel('Week', fontsize=11)
ax1.set_ylabel('Mean Service Score', fontsize=11)
ax1.set_title('Service Scores Over Time\n(with 95% Confidence Intervals)', fontsize=12, fontweight='bold')
ax1.legend(loc='best', fontsize=9)
ax1.grid(True, alpha=0.3)
ax1.set_xticks(range(6, 23))
ax1.set_ylim(42, 58)

# Panel 2: Before/After comparison
before_after_data = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).reset_index()
before_after_data['se'] = before_after_data['std'] / np.sqrt(before_after_data['count'])

x_positions = [0, 1, 3, 4]  # Gap between before/after groups
colors = ['red', 'red', 'blue', 'blue']
labels = ['Control\nBefore', 'Control\nAfter', 'Treatment\nBefore', 'Treatment\nAfter']

bars = ax2.bar(x_positions, before_after_data['mean'], color=['#d62728', '#d62728', '#1f77b4', '#1f77b4'], 
               yerr=1.96 * before_after_data['se'], capsize=5, alpha=0.8, edgecolor='black')

ax2.set_xticks([0.5, 3.5])
ax2.set_xticklabels(['Non-Switched\n(Control)', 'Switched\n(Treatment)'])
ax2.set_ylabel('Mean Service Score', fontsize=11)
ax2.set_title('Before vs After Comparison\n(with 95% Confidence Intervals)', fontsize=12, fontweight='bold')
ax2.grid(True, alpha=0.3, axis='y')
ax2.set_ylim(42, 58)

# Add value labels on bars
for i, (x, mean) in enumerate(zip(x_positions, before_after_data['mean'])):
    ax2.text(x, mean + 1, f'{mean:.1f}', ha='center', va='bottom', fontsize=10)

plt.tight_layout()
plt.savefig('board_visualization.png', dpi=150, bbox_inches='tight')
print("Visualization saved as board_visualization.png")
