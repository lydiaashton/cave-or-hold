import pandas as pd
import numpy as np
import statsmodels.api as sm
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate summary statistics by group and period
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("Service Score by Group and Period:")
print(summary)
print()

# Calculate the DiD estimate manually
# Before period (after_switch=0), After period (after_switch=1)
before_switched = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_switched = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print("Mean service scores:")
print(f"Switched branches - Before: {before_switched:.2f}, After: {after_switched:.2f}, Change: {after_switched - before_switched:.2f}")
print(f"Control branches - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")
print(f"\nDifference-in-Differences estimate: {(after_switched - before_switched) - (after_control - before_control):.2f}")
print()

# Run regression with DiD specification
# Basic DiD: service_score = beta0 + beta1*switched + beta2*after + beta3*(switched*after) + error
X = df[['switched_branch', 'after_switch', 'treatment']]
X = sm.add_constant(X)
y = df['service_score']

model = sm.OLS(y, X).fit()
print("Basic DiD Regression:")
print(model.summary())

import numpy as np
import statsmodels.api as sm
from statsmodels.stats.sandwich_covariance import cov_cluster

df['treatment'] = df['switched_branch'] * df['after_switch']

# Run regression with cluster-robust standard errors (clustered by branch)
X = df[['switched_branch', 'after_switch', 'treatment']]
X = sm.add_constant(X)
y = df['service_score']

model = sm.OLS(y, X).fit()
# Cluster-robust standard errors
cov_clustered = cov_cluster(model, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))

ci_upper = model.params + 1.96 * se_clustered

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from scipy import stats
import statsmodels.api as sm
from statsmodels.stats.sandwich_covariance import cov_cluster

df = pd.read_csv('branches.csv')

# Event study: create week dummies relative to switch
# Week 14 is the last pre-switch week, so we'll use it as reference
df['relative_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create indicators for each week (excluding week 14 as reference)
weeks = list(range(6, 23))  # weeks 6-22
pre_weeks = [w for w in weeks if w < 14]
post_weeks = [w for w in weeks if w >= 15]

# Create interaction terms for each week with treatment
for w in weeks:
    if w != 14:  # Skip reference week
        df[f'week_{w}'] = (df['week'] == w).astype(int)
        df[f'treatment_week_{w}'] = df['switched_branch'] * (df['week'] == w).astype(int)

# Run event study regression
event_vars = [f'treatment_week_{w}' for w in weeks if w != 14]
control_vars = [f'week_{w}' for w in weeks if w != 14]
all_vars = event_vars + control_vars + ['switched_branch']

X_event = df[all_vars]
X_event = sm.add_constant(X_event)
y = df['service_score']

model_event = sm.OLS(y, X_event).fit()
cov_event = cov_cluster(model_event, df['branch_id'])
se_event = np.sqrt(np.diag(cov_event))

# Extract treatment effects for each week
print("Event Study Results (treatment effects by week):")
print(f"{'Week':<10} {'Relative':<10} {'Coef':>10} {'SE':>10} {'95% CI':>20}")
print("-" * 55)

event_results = []
for w in weeks:
    if w == 14:
        event_results.append({'week': w, 'relative': 0, 'coef': 0, 'se': 0, 'ci_lower': 0, 'ci_upper': 0})
        print(f"{w:<10} {0:<10} {'(ref)':>10} {'-':>10} {'-':>20}")
    else:
        var_name = f'treatment_week_{w}'
        idx = list(X_event.columns).index(var_name)
        coef = model_event.params.values[idx]
        se = se_event[idx]
        ci_lower = coef - 1.96 * se
        ci_upper = coef + 1.96 * se
        event_results.append({'week': w, 'relative': w - 14, 'coef': coef, 'se': se, 'ci_lower': ci_lower, 'ci_upper': ci_upper})
        print(f"{w:<10} {w-14:<10} {coef:>10.2f} {se:>10.2f} [{ci_lower:.2f}, {ci_upper:.2f}]")

# Plot event study
event_df = pd.DataFrame(event_results)
event_df = event_df.sort_values('relative')

fig, ax = plt.subplots(figsize=(10, 6))

ax.set_xlabel('Weeks Relative to Switch (0 = Week 14, reference)', fontsize=12)
ax.set_ylabel('Treatment Effect on Service Score', fontsize=12)
ax.set_title('Event Study: Treatment Effect by Week', fontsize=14)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-8, 9))

plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
