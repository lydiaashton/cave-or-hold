import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Number of branches that switched vs didn't
switched_branches = df[df['switched_branch']==1]['branch_id'].nunique()
control_branches = df[df['switched_branch']==0]['branch_id'].nunique()
print(f"Branches that switched: {switched_branches}")
print(f"Branches that didn't switch (control): {control_branches}")

# Check pre/post service scores by group
print("\n=== Service Score Summary ===")
print("\nBefore switch (weeks 6-14):")
before = df[df['after_switch']==0]
print(before.groupby('switched_branch')['service_score'].describe())

print("\nAfter switch (weeks 15-22):")
after = df[df['after_switch']==1]
print(after.groupby('switched_branch')['service_score'].describe())

# Calculate mean scores by group and period
print("\n=== Mean Service Scores ===")
means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean()
print(means)

# Calculate the difference-in-differences
print("\n=== Difference-in-Differences Calculation ===")
# Control group change
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_change = control_after - control_before

# Treatment group change
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_change = treat_after - treat_before

# DiD estimate
did = treat_change - control_change

print(f"Control group - Before: {control_before:.2f}, After: {control_after:.2f}, Change: {control_change:.2f}")
print(f"Treatment group - Before: {treat_before:.2f}, After: {treat_after:.2f}, Change: {treat_change:.2f}")
print(f"\nDifference-in-Differences estimate: {did:.2f}")

import pandas as pd
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate mean service scores by week and treatment status
weekly_means = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_counts = df.groupby(['week', 'switched_branch'])['service_score'].count().unstack()

# Create a more detailed plot
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Plot 1: Raw trends
ax1 = axes[0]
ax1.plot(weekly_means.index, weekly_means[0], 'o-', label='Control (No Switch)', linewidth=2.5, markersize=8, color='#2E86AB')
ax1.plot(weekly_means.index, weekly_means[1], 's-', label='Treatment (Switched)', linewidth=2.5, markersize=8, color='#A23B72')
ax1.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')
ax1.axvspan(15, 22, alpha=0.1, color='red', label='Post-Switch Period')
ax1.set_xlabel('Week', fontsize=12)
ax1.set_ylabel('Mean Service Score', fontsize=12)
ax1.set_title('Service Score Trends by Group', fontsize=14, fontweight='bold')
ax1.legend(fontsize=10, loc='upper left')
ax1.grid(True, alpha=0.3)
ax1.set_xticks(range(6, 23))

# Plot 2: Difference between treatment and control over time
ax2 = axes[1]
diff = weekly_means[1] - weekly_means[0]
ax2.bar(diff.index, diff.values, color=['gray' if w < 15 else '#A23B72' for w in diff.index], alpha=0.7)
ax2.axhline(y=0, color='black', linewidth=1)
ax2.axvline(x=14.5, color='red', linestyle='--', linewidth=2)
ax2.set_xlabel('Week', fontsize=12)
ax2.set_ylabel('Treatment - Control (points)', fontsize=12)
ax2.set_title('Difference in Service Scores Over Time', fontsize=14, fontweight='bold')

ax2.set_xticks(range(6, 23))
