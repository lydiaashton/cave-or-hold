import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Average service scores by group and period
print("\n\nAverage service scores:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
print(pivot)

# Calculate the difference-in-differences manually
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n\nControl (no switch): Before =", control_before, "After =", control_after, "Change =", control_after - control_before)
print("Treatment (switched): Before =", treatment_before, "After =", treatment_after, "Change =", treatment_after - treatment_before)

did_estimate = (treatment_after - treatment_before) - (control_after - control_before)
print("\nDifference-in-Differences estimate:", did_estimate)

# Check parallel trends - look at weekly averages for both groups
print("\n\nWeekly averages by group:")
weekly = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
weekly.columns = ['Control', 'Treatment']
print(weekly)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['did'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Basic DiD Regression (clustered by branch):")
print(model1.summary())

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with controls (clustered by branch):")
print(model2.summary())

# Check parallel trends - regression with week fixed effects interacted with treatment
# This tests whether pre-trends differ between groups
df['pre_period'] = (df['week'] < 15).astype(int)
df['treatment_pre'] = df['switched_branch'] * df['pre_period']

# Test if there's a differential trend before the switch
model3 = ols('service_score ~ switched_branch + week + treatment_pre', data=df[df['week'] < 15]).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nPre-trend test (weeks 6-14 only):")
print(model3.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Pre-trend test - simpler approach without clustering issues
# Just test if there's differential trend in pre-period
pre_data = df[df['week'] < 15].copy()
pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()
pre_data['treatment_trend'] = pre_data['switched_branch'] * pre_data['week_centered']

model_pre = ols('service_score ~ switched_branch + week_centered + treatment_trend', data=pre_data).fit()
print("Pre-trend test (weeks 6-14):")
print(model_pre.summary())

# Create visualization data
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
weekly_avg['group'] = weekly_avg['switched_branch'].map({0: 'Control (No Switch)', 1: 'Treatment (Switched)'})

# Plot
fig, ax = plt.subplots(figsize=(10, 6))
for group in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == group]
    label = 'Control (No Switch)' if group == 0 else 'Treatment (Switched)'
    ax.plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2, markersize=6)

ax.axvline(x=14.5, color='red', linestyle='--', label='Switch (Week 15)', linewidth=2)
ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Scores by Group Over Time', fontsize=14)
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("\nPlot saved as service_scores_trend.png")

# Summary statistics for the report
print("\n\n=== SUMMARY STATISTICS ===")
print(f"Total observations: {len(df)}")
print(f"Total branches: {df['branch_id'].nunique()}")
print(f"Switched branches: {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"Control branches: {df[df['switched_branch']==0]['branch_id'].nunique()}")
print(f"Pre-switch weeks: 6-14 (9 weeks)")
print(f"Post-switch weeks: 15-22 (8 weeks)")

# Effect size in context
overall_mean = df['service_score'].mean()
print(f"\nOverall mean service score: {overall_mean:.2f}")
did_effect = 2.48
print(f"DiD effect: {did_effect:.2f} points")
print(f"Effect as % of mean: {did_effect/overall_mean*100:.1f}%")
