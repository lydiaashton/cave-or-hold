import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check parallel trends - compare trends before week 15
print("=== Parallel Trends Check (Pre-Period: Weeks 6-14) ===")
pre_df = df[df['week'] < 15].copy()

# Run a regression with interaction to test if trends differed before the switch
# service_score ~ switched_branch * week (for pre-period only)
pre_model = ols('service_score ~ switched_branch * week', data=pre_df).fit()
print("\nPre-period trend regression:")
print(pre_model.summary().tables[1])

# Also check visually - average by week for each group
print("\n\nPre-period weekly averages:")
pre_weekly = pre_df.groupby(['switched_branch', 'week'])['service_score'].mean()
for group in [0, 1]:
    group_data = pre_weekly.xs(group, level='switched_branch')
    print(f"\nGroup {group} ( {'Control' if group==0 else 'Treatment'} ):")
    print(group_data)

# Now run the main DiD regression
print("\n\n=== Main Difference-in-Differences Analysis ===")
# Basic DiD: service_score ~ switched_branch * after_switch
did_model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("\nBasic DiD Regression:")
print(did_model.summary().tables[1])

# DiD with branch fixed effects
print("\n\n=== DiD with Branch Fixed Effects ===")
df_fe = df.copy()
df_fe['branch_id'] = df_fe['branch_id'].astype('category')
fe_model = ols('service_score ~ switched_branch * after_switch + C(branch_id)', data=df_fe).fit()
print(fe_model.summary().tables[1])

# DiD with branch and week fixed effects
print("\n\n=== DiD with Branch and Week Fixed Effects ===")
fe_week_model = ols('service_score ~ switched_branch * after_switch + C(branch_id) + C(week)', data=df_fe).fit()
print(fe_week_model.summary().tables[1])

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create treatment indicator (interaction term)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD with branch and week fixed effects using a different approach
# First, let's use a simpler model specification
print("=== DiD Regression with Branch and Week Fixed Effects ===")

# Create dummies manually to avoid the singular matrix issue
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)

X = pd.concat([df[['switched_branch', 'after_switch', 'treatment']], 
               branch_dummies, week_dummies], axis=1)
y = df['service_score']

# Add constant
X = sm.add_constant(X)

# Fit the model
model = sm.OLS(y, X).fit()

# Get the treatment effect
treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pval = model.pvalues['treatment']
treatment_ci = model.conf_int().loc['treatment']

print(f"\nTreatment Effect: {treatment_coef:.3f}")
print(f"Standard Error: {treatment_se:.3f}")
print(f"95% CI: [{treatment_ci[0]:.3f}, {treatment_ci[1]:.3f}]")
print(f"P-value: {treatment_pval:.6f}")

# Clustered standard errors by branch
print("\n\n=== With Clustered Standard Errors (by branch) ===")
model_cluster = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
treatment_coef_c = model_cluster.params['treatment']
treatment_se_c = model_cluster.bse['treatment']
treatment_pval_c = model_cluster.pvalues['treatment']
treatment_ci_c = model_cluster.conf_int().loc['treatment']

print(f"\nTreatment Effect: {treatment_coef_c:.3f}")
print(f"Clustered SE: {treatment_se_c:.3f}")
print(f"95% CI: [{treatment_ci_c[0]:.3f}, {treatment_ci_c[1]:.3f}]")
print(f"P-value: {treatment_pval_c:.6f}")

# Effect size relative to mean
mean_score = df['service_score'].mean()
print(f"\n\n=== Effect Size ===")
print(f"Mean service score: {mean_score:.2f}")
print(f"Treatment effect: {treatment_coef_c:.3f}")
print(f"Effect as % of mean: {treatment_coef_c/mean_score*100:.2f}%")
