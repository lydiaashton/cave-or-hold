import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create relative time indicators (weeks relative to switch at week 15)
# Week 14 is the last pre-switch week, so we'll use that as reference
df['relative_week'] = df['week'] - 14  # Week 14 becomes 0, week 15 becomes 1, etc.

# Create event study dummies for each week (excluding reference week 14)
# We interact these with switched_branch to get dynamic treatment effects
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != 0:  # Skip reference week
        df[f'week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression with branch and week fixed effects
# Reference category is week 0 (week 14, the last pre-switch week)
event_study_vars = [f'week_{w}' for w in range(-8, 9) if w != 0]
formula = f'service_score ~ {" + ".join(event_study_vars)} + staffing_index + visits_index + C(branch_id) + C(week)'

model_es = ols(formula, data=df).fit()

# Extract coefficients and standard errors for the event study terms
coef_names = model_es.params.index.tolist()
event_coefs = []
event_se = []
event_weeks = []

for w in range(-8, 9):
    if w == 0:
        event_coefs.append(0)  # Reference category
        event_se.append(0)
        event_weeks.append(w)
    else:
        var_name = f'week_{w}'
        if var_name in coef_names:
            idx = coef_names.index(var_name)
            event_coefs.append(model_es.params[var_name])
            event_se.append(model_es.bse[var_name])
            event_weeks.append(w)

# Create event study plot
plt.figure(figsize=(12, 6))
plt.errorbar(event_weeks, event_coefs, yerr=[1.96*s for s in event_se], 
             fmt='o', capsize=5, markersize=8, linewidth=2)
plt.axhline(y=0, color='black', linestyle='-', linewidth=1)
plt.axvline(x=0.5, color='red', linestyle='--', linewidth=2, label='Switch occurs')
plt.xlabel('Weeks Relative to Switch (0 = Week 14, last pre-switch week)')
plt.ylabel('Coefficient (Effect on Service Score)')
plt.title('Event Study: Dynamic Treatment Effects on Service Score')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study_plot.png', dpi=150)
print("Event study plot saved as event_study_plot.png")

# Print the coefficients
print("\nEvent Study Coefficients:")
print("=" * 60)
print(f"{'Week Relative':<15} {'Coefficient':<12} {'SE':<10} {'95% CI Lower':<15} {'95% CI Upper':<15}")
print("=" * 60)
for i, w in enumerate(event_weeks):
    if w == 0:
        print(f"{w:+d} (ref)      {'0.000':<12} {'-':<10} {'-':<15} {'-':<15}")
    else:
        ci_lower = event_coefs[i] - 1.96 * event_se[i]
        ci_upper = event_coefs[i] + 1.96 * event_se[i]
        print(f"{w:+d}            {event_coefs[i]:<12.3f} {event_se[i]:<10.3f} {ci_lower:<15.3f} {ci_upper:<15.3f}")

# Test for pre-trend: are pre-switch coefficients jointly zero?
pre_trend_vars = [f'week_{w}' for w in range(-8, 0)]

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create relative time indicators (weeks relative to switch at week 15)
# Week 14 is the last pre-switch week, so we'll use that as reference
df['relative_week'] = df['week'] - 14  # Week 14 becomes 0, week 15 becomes 1, etc.

# Create event study dummies with valid variable names
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != 0:  # Skip reference week
        df[f'w{w:+d}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression with branch and week fixed effects
event_study_vars = [f'w{w:+d}' for w in range(-8, 9) if w != 0]
formula = f'service_score ~ {" + ".join(event_study_vars)} + staffing_index + visits_index + C(branch_id) + C(week)'

model_es = ols(formula, data=df).fit()

# Extract coefficients and standard errors for the event study terms
coef_names = model_es.params.index.tolist()
event_coefs = []
event_se = []
event_weeks = []

for w in range(-8, 9):
    if w == 0:
        event_coefs.append(0)  # Reference category
        event_se.append(0)
        event_weeks.append(w)
    else:
        var_name = f'w{w:+d}'
        if var_name in coef_names:
            idx = coef_names.index(var_name)
            event_coefs.append(model_es.params[var_name])
            event_se.append(model_es.bse[var_name])
            event_weeks.append(w)

# Create event study plot
plt.figure(figsize=(12, 6))
plt.errorbar(event_weeks, event_coefs, yerr=[1.96*s for s in event_se], 
             fmt='o', capsize=5, markersize=8, linewidth=2)
plt.axhline(y=0, color='black', linestyle='-', linewidth=1)
plt.axvline(x=0.5, color='red', linestyle='--', linewidth=2, label='Switch occurs')
plt.xlabel('Weeks Relative to Switch (0 = Week 14, last pre-switch week)')
plt.ylabel('Coefficient (Effect on Service Score)')
plt.title('Event Study: Dynamic Treatment Effects on Service Score')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study_plot.png', dpi=150)

    if w == 0:

        ci_lower = event_coefs[i] - 1.96 * event_se[i]
        ci_upper = event_coefs[i] + 1.96 * event_se[i]

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create relative time indicators (weeks relative to switch at week 15)
# Week 14 is the last pre-switch week, so we'll use that as reference
df['relative_week'] = df['week'] - 14  # Week 14 becomes 0, week 15 becomes 1, etc.

# Create event study dummies with valid variable names (no + signs)
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != 0:  # Skip reference week
        # Use 'rel' prefix and handle negative sign
        var_name = f'rel{w}' if w >= 0 else f'rel_neg{-w}'
        df[var_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Map week numbers to variable names
week_to_var = {}
for w in range(-8, 9):
    if w != 0:
        week_to_var[w] = f'rel{w}' if w >= 0 else f'rel_neg{-w}'

# Run event study regression with branch and week fixed effects
event_study_vars = list(week_to_var.values())
formula = f'service_score ~ {" + ".join(event_study_vars)} + staffing_index + visits_index + C(branch_id) + C(week)'

model_es = ols(formula, data=df).fit()

# Extract coefficients and standard errors for the event study terms
coef_names = model_es.params.index.tolist()
event_coefs = []
event_se = []
event_weeks = []

for w in range(-8, 9):
    if w == 0:
        event_coefs.append(0)  # Reference category
        event_se.append(0)
        event_weeks.append(w)
    else:
        var_name = week_to_var[w]
        if var_name in coef_names:
            event_coefs.append(model_es.params[var_name])
            event_se.append(model_es.bse[var_name])
            event_weeks.append(w)

# Create event study plot
plt.figure(figsize=(12, 6))
plt.errorbar(event_weeks, event_coefs, yerr=[1.96*s for s in event_se], 
             fmt='o', capsize=5, markersize=8, linewidth=2)
plt.axhline(y=0, color='black', linestyle='-', linewidth=1)
plt.axvline(x=0.5, color='red', linestyle='--', linewidth=2, label='Switch occurs')
plt.xlabel('Weeks Relative to Switch (0 = Week 14, last pre-switch week)')
plt.ylabel('Coefficient (Effect on Service Score)')
plt.title('Event Study: Dynamic Treatment Effects on Service Score')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study_plot.png', dpi=150)
print("Event study plot saved as event_study_plot.png")

# Print the coefficients
print("\nEvent Study Coefficients:")
print("=" * 70)
print(f"{'Week Relative':<15} {'Coefficient':<12} {'SE':<10} {'95% CI Lower':<15} {'95% CI Upper':<15}")
print("=" * 70)
for i, w in enumerate(event_weeks):
    if w == 0:
        print(f"{w:+d} (ref)      {'0.000':<12} {'-':<10} {'-':<15} {'-':<15}")
    else:
        ci_lower = event_coefs[i] - 1.96 * event_se[i]
        ci_upper = event_coefs[i] + 1.96 * event_se[i]

pre_trend_vars = [week_to_var[w] for w in range(-8, 0)]

print(f"\nPre-trend coefficients (weeks -8 to -1):")

print(f"Range of pre-trend coefficients: [{min(pre_trend_coefs):.3f}, {max(pre_trend_coefs):.3f}]")
