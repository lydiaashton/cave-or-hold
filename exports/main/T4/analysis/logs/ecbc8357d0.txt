control_change = control_after - control_before
treatment_change = treatment_after - treatment_before
did_estimate = treatment_change - control_change

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Get the treatment effect from the model
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
model_cov = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
effect = model_cov.params['treatment']
pvalue = model_cov.pvalues['treatment']

baseline = df['service_score'].mean()
print(f"=== Effect Size ===")
print(f"Baseline average service score: {baseline:.2f}")
print(f"Estimated treatment effect: {effect:.2f}")
print(f"Effect as % of baseline: {effect/baseline*100:.1f}%")
print(f"P-value: {pvalue:.3f}")
print()

# Create visualization of trends
plt.figure(figsize=(12, 6))

weekly = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack()
weeks = weekly.columns.tolist()

# Control group (old procedure)
plt.plot(weeks, weekly.loc[0].values, 'o-', label='Control (Old Procedure)', linewidth=2, markersize=6, color='blue')
# Treatment group (new procedure)
plt.plot(weeks, weekly.loc[1].values, 's-', label='Treatment (New Procedure)', linewidth=2, markersize=6, color='green')
