import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check pre-treatment trends by week
print("Pre-treatment trends (weeks 6-14):")
pre_trend = df[df['week'] < 15].groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
print(pre_trend)

# Plot the trends
plt.figure(figsize=(12, 6))
for switched in [0, 1]:
    group_data = df[df['switched_branch'] == switched].groupby('week')['service_score'].mean()
    label = 'Switched (New Procedure)' if switched == 1 else 'Control (Old Procedure)'
    plt.plot(group_data.index, group_data.values, marker='o', label=label, linewidth=2)

plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Score Trends: Treatment vs Control Groups')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('trends_plot.png', dpi=150)
print("\nPlot saved as trends_plot.png")

# Run DiD regression
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\n=== Basic DiD Regression ===")
print(model1.summary().tables[1])

# DiD with controls

print("\n\n=== DiD Regression with Controls ===")
print(model2.summary().tables[1])

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Test for parallel trends - check if pre-treatment trends differ
# Create week dummies interacted with treatment for pre-period only
df_pre = df[df['week'] < 15].copy()
df_pre['week_cat'] = df_pre['week'].astype(str)

# Check if there's a differential trend in the pre-period
# Regress service_score on week, switched_branch, and their interaction in pre-period
df_pre['week_centered'] = df_pre['week'] - df_pre['week'].mean()
pre_trend_model = ols('service_score ~ switched_branch * week_centered', data=df_pre).fit(cov_type='cluster', cov_kwds={'groups': df_pre['branch_id']})
print("=== Pre-treatment Trend Test ===")
print(pre_trend_model.summary().tables[1])

# More robust: DiD with branch fixed effects
# Create branch dummies
df['branch_id_str'] = df['branch_id'].astype(str)

# Use linearmodels for fixed effects if available, otherwise use dummy variables
try:
    from linearmodels.panel import PanelOLS
    
    # Prepare panel data
    df_panel = df.set_index(['branch_id', 'week'])
    
    # DiD with branch fixed effects
    exog = sm.add_constant(df_panel[['after_switch', 'treatment', 'staffing_index', 'visits_index']])
    mod = PanelOLS(df_panel['service_score'], exog, entity_effects=True)
    fe_results = mod.fit(cov_type='clustered', cluster_entity=True)
    print("\n\n=== Fixed Effects DiD (with branch FE) ===")
    print(fe_results)
except ImportError:
    print("\nlinearmodels not available, using dummy variables approach")
    # Create branch dummies (this might be heavy with 709 branches)
    branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
    df_with_dummies = pd.concat([df, branch_dummies], axis=1)
    
    # Build formula with all branch dummies
    branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_')]
    formula = 'service_score ~ after_switch + treatment + staffing_index + visits_index + ' + ' + '.join(branch_cols)
    
    fe_model = ols(formula, data=df_with_dummies).fit(cov_type='cluster', cov_kwds={'groups': df_with_dummies['branch_id']})
    print("\n\n=== Fixed Effects DiD (with branch FE) ===")
    print("Treatment coefficient:", fe_model.params['treatment'])
    print("Treatment std err:", fe_model.bse['treatment'])
    print("Treatment p-value:", fe_model.pvalues['treatment'])

# Event study - check effects by week relative to switch
df['weeks_relative'] = df['week'] - 15  # 0 is the switch week

# Average effect by relative week for treated vs control
event_study = df.groupby(['switched_branch', 'weeks_relative'])['service_score'].mean().unstack(level=0)
print("\n\n=== Event Study: Average Scores by Week Relative to Switch ===")
print(event_study)

# Plot event study

for switched in [0, 1]:

    label = 'Switched (New Procedure)' if switched == 1 else 'Control (Old Procedure)'
    plt.plot(group_data.index, group_data.values, marker='o', label=label, linewidth=2)

plt.axvline(x=-0.5, color='red', linestyle='--', label='Switch Point')
plt.xlabel('Weeks Relative to Switch (0 = Week 15)')
plt.ylabel('Average Service Score')

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create treatment variable
df['treatment'] = df['switched_branch'] * df['after_switch']

# Test for parallel trends - check if pre-treatment trends differ
df_pre = df[df['week'] < 15].copy()
df_pre['week_centered'] = df_pre['week'] - df_pre['week'].mean()
pre_trend_model = ols('service_score ~ switched_branch * week_centered', data=df_pre).fit(cov_type='cluster', cov_kwds={'groups': df_pre['branch_id']})
print("=== Pre-treatment Trend Test ===")
print(pre_trend_model.summary().tables[1])
print("\nNote: The interaction term (switched_branch:week_centered) tests if pre-trends differ.")
print("A significant coefficient suggests the parallel trends assumption may be violated.")

# More robust: DiD with branch fixed effects using dummy variables
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_')]
formula = 'service_score ~ after_switch + treatment + staffing_index + visits_index + ' + ' + '.join(branch_cols)

fe_model = ols(formula, data=df_with_dummies).fit(cov_type='cluster', cov_kwds={'groups': df_with_dummies['branch_id']})
print("\n\n=== Fixed Effects DiD (with branch FE) ===")
print(f"Treatment coefficient: {fe_model.params['treatment']:.4f}")
print(f"Treatment std err: {fe_model.bse['treatment']:.4f}")
print(f"Treatment p-value: {fe_model.pvalues['treatment']:.6f}")
print(f"95% CI: [{fe_model.conf_int().loc['treatment'][0]:.4f}, {fe_model.conf_int().loc['treatment'][1]:.4f}]")

# Event study - check effects by week relative to switch
df['weeks_relative'] = df['week'] - 15  # 0 is the switch week

# Plot event study
plt.figure(figsize=(12, 6))
for switched in [0, 1]:
    group_data = df[df['switched_branch'] == switched].groupby('weeks_relative')['service_score'].mean()
    label = 'Switched (New Procedure)' if switched == 1 else 'Control (Old Procedure)'
    plt.plot(group_data.index, group_data.values, marker='o', label=label, linewidth=2)

plt.axvline(x=-0.5, color='red', linestyle='--', label='Switch Point')
plt.xlabel('Weeks Relative to Switch (0 = Week 15)')
plt.ylabel('Average Service Score')
plt.title('Event Study: Service Scores Around Switch')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('event_study_plot.png', dpi=150)
print("\nEvent study plot saved as event_study_plot.png")

# Calculate effect size in practical terms
avg_score = df['service_score'].mean()
treatment_effect = fe_model.params['treatment']
pct_change = (treatment_effect / avg_score) * 100
print(f"\n=== Practical Significance ===")
print(f"Average service score across all observations: {avg_score:.2f}")
print(f"Estimated treatment effect: {treatment_effect:.2f} points")
print(f"Percentage improvement: {pct_change:.1f}%")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.regression.linear_model import OLS
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Fixed effects via within transformation (demeaning by branch)
# Calculate branch means
branch_means = df.groupby('branch_id')[['service_score', 'after_switch', 'treatment', 'staffing_index', 'visits_index']].transform('mean')

# Demean the variables
df_demeaned = df[['service_score', 'after_switch', 'treatment', 'staffing_index', 'visits_index']] - branch_means

# Run regression on demeaned data (no intercept needed since data is demeaned)
X = df_demeaned[['after_switch', 'treatment', 'staffing_index', 'visits_index']]
X = sm.add_constant(X, has_constant='skip')  # Skip adding constant since data is demeaned
y = df_demeaned['service_score']

fe_model = OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Fixed Effects DiD (within transformation) ===")
print(fe_model.summary().tables[1])

# Also run the simpler DiD without FE for comparison
simple_model = sm.OLS(df['service_score'], sm.add_constant(df[['switched_branch', 'after_switch', 'treatment', 'staffing_index', 'visits_index']])).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\n=== Simple DiD with Controls (no FE) ===")
print(simple_model.summary().tables[1])

# Event study plot
df['weeks_relative'] = df['week'] - 15

plt.figure(figsize=(12, 6))
for switched in [0, 1]:
    group_data = df[df['switched_branch'] == switched].groupby('weeks_relative')['service_score'].mean()
    label = 'Switched (New Procedure)' if switched == 1 else 'Control (Old Procedure)'
    plt.plot(group_data.index, group_data.values, marker='o', label=label, linewidth=2, markersize=8)

plt.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')
plt.xlabel('Weeks Relative to Switch (0 = Week 15)', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Score Trends: Treatment vs Control Groups', fontsize=14, fontweight='bold')
plt.legend(fontsize=11)
plt.grid(True, alpha=0.3)
plt.xticks(range(-9, 8))
plt.tight_layout()
plt.savefig('event_study_plot.png', dpi=150)
print("\nEvent study plot saved as event_study_plot.png")

# Summary statistics for the report
print("\n\n=== Summary Statistics ===")
print(f"Total observations: {len(df):,}")
print(f"Branches that switched: {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"Branches that did not switch: {df[df['switched_branch']==0]['branch_id'].nunique()}")
print(f"Weeks of data: {df['week'].min()} to {df['week'].max()}")
print(f"\nTreatment effect estimate (FE model): {fe_model.params['treatment']:.2f} points")
print(f"95% CI: [{fe_model.conf_int().loc['treatment'][0]:.2f}, {fe_model.conf_int().loc['treatment'][1]:.2f}]")
print(f"P-value: {fe_model.pvalues['treatment']:.6f}")

ax1 = axes[0]
for switched in [0, 1]:
