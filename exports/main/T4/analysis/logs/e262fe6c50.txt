import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# Create a treatment group indicator (interaction of switched_branch and after_switch)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate average service scores by group and period
print("=" * 60)
print("SERVICE SCORE ANALYSIS")
print("=" * 60)

# Before switch (after_switch = 0)
before_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score']
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score']

# After switch (after_switch = 1)
after_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score']
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score']

print("\n1. AVERAGE SERVICE SCORES BY GROUP AND PERIOD")
print("-" * 60)
print(f"Before switch (weeks 6-14):")
print(f"  Switched branches (treatment): {before_switched.mean():.2f} (n={len(before_switched)})")
print(f"  Non-switched branches (control): {before_control.mean():.2f} (n={len(before_control)})")
print(f"  Difference: {before_switched.mean() - before_control.mean():.2f}")

print(f"\nAfter switch (weeks 15-22):")
print(f"  Switched branches (treatment): {after_switched.mean():.2f} (n={len(after_switched)})")
print(f"  Non-switched branches (control): {after_control.mean():.2f} (n={len(after_control)})")
print(f"  Difference: {after_switched.mean() - after_control.mean():.2f}")

# Difference-in-differences calculation
print("\n2. DIFFERENCE-IN-DIFFERENCES ANALYSIS")
print("-" * 60)
change_treatment = after_switched.mean() - before_switched.mean()
change_control = after_control.mean() - before_control.mean()
did_estimate = change_treatment - change_control

print(f"Change in switched branches: {change_treatment:.2f}")
print(f"Change in non-switched branches: {change_control:.2f}")
print(f"Difference-in-Differences estimate: {did_estimate:.2f}")

# Statistical test - t-test on the DiD
print("\n3. STATISTICAL SIGNIFICANCE")
print("-" * 60)

# Run a regression-style DiD test
# Create the regression variables
df['const'] = 1
df['post'] = df['after_switch']
df['treated'] = df['switched_branch']
df['did'] = df['treated'] * df['post']

# Simple DiD regression using OLS
import statsmodels.api as sm

X = df[['const', 'post', 'treated', 'did']]
y = df['service_score']

model = sm.OLS(y, X).fit()
print(model.summary())

# Also check branch-level aggregation to avoid serial correlation concerns
print("\n4. BRANCH-LEVEL AGGREGATED ANALYSIS")
print("-" * 60)

# Aggregate to branch level (before and after averages)
branch_agg = df.groupby(['branch_id', 'switched_branch', 'after_switch'])['service_score'].mean().reset_index()
branch_agg = branch_agg.pivot(index=['branch_id', 'switched_branch'], columns='after_switch', values='service_score').reset_index()
branch_agg.columns = ['branch_id', 'switched_branch', 'before_score', 'after_score']
branch_agg['change'] = branch_agg['after_score'] - branch_agg['before_score']

print("Change in service score by branch type:")
print(branch_agg.groupby('switched_branch')['change'].describe())

# T-test on the changes
switched_change = branch_agg[branch_agg['switched_branch'] == 1]['change']
control_change = branch_agg[branch_agg['switched_branch'] == 0]['change']

t_stat, p_val = stats.ttest_ind(switched_change, control_change)
print(f"\nT-test comparing changes:")
print(f"  Switched branches mean change: {switched_change.mean():.2f}")
print(f"  Control branches mean change: {control_change.mean():.2f}")
print(f"  Difference: {switched_change.mean() - control_change.mean():.2f}")
print(f"  T-statistic: {t_stat:.3f}")
print(f"  P-value: {p_val:.4f}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# Recreate treatment variable
df['treatment'] = df['switched_branch'] * df['after_switch']

# Full model with controls
X = df[['after_switch', 'switched_branch', 'treatment', 'staffing_index', 'visits_index']]
X = sm.add_constant(X)
y = df['service_score']

model_controls = sm.OLS(y, X).fit()
print("REGRESSION WITH CONTROLS")
print("=" * 60)
print(model_controls.summary())

# Let's also create a visualization
print("\n" + "=" * 60)
print("VISUALIZATION DATA")
print("=" * 60)

# Weekly averages for plotting
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()

# Create a simple text-based visualization
print("\nWeekly Service Scores by Group:")
print("-" * 60)
for week in range(6, 23):
    week_data = weekly_avg[weekly_avg['week'] == week]
    control = week_data[week_data['switched_branch'] == 0]['service_score'].values[0]
    treatment = week_data[week_data['switched_branch'] == 1]['service_score'].values[0]
    switch_marker = " <-- SWITCH" if week == 15 else ""
    print(f"Week {week:2d}: Control={control:5.1f}  Treatment={treatment:5.1f}  Diff={treatment-control:6.1f}{switch_marker}")

# Confidence interval for the DiD estimate using cluster-robust SE
print("\n" + "=" * 60)
print("CLUSTER-ROBUST STANDARD ERRORS (by branch)")
print("=" * 60)

# Use statsmodels with cluster-robust SE
from statsmodels.stats.sandwich_covariance import cov_cluster

model = sm.OLS(y, sm.add_constant(df[['after_switch', 'switched_branch', 'treatment']]))
result = model.fit()
cov_clustered = cov_cluster(result, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))

print("\nCoefficients with cluster-robust SE:")
print(f"  Intercept:     {result.params['const']:.2f} (SE: {se_clustered[0]:.2f})")
print(f"  after_switch:  {result.params['after_switch']:.2f} (SE: {se_clustered[1]:.2f})")
print(f"  switched:      {result.params['switched_branch']:.2f} (SE: {se_clustered[2]:.2f})")
print(f"  DiD (treatment): {result.params['treatment']:.2f} (SE: {se_clustered[3]:.2f})")

# Calculate t-stat and p-value for DiD
did_coef = result.params['treatment']
did_se = se_clustered[3]
did_t = did_coef / did_se
from scipy import stats
did_p = 2 * (1 - stats.t.cdf(abs(did_t), df=33))  # 34 branches - 1

print(f"\nDiD Effect: {did_coef:.2f}")
print(f"Cluster-robust SE: {did_se:.2f}")
print(f"T-statistic: {did_t:.2f}")
print(f"P-value (df=33): {did_p:.4f}")
print(f"95% CI: [{did_coef - 1.96*did_se:.2f}, {did_coef + 1.96*did_se:.2f}]")
