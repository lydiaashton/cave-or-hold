import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate mean service scores by group and period
print("=" * 60)
print("MEAN SERVICE SCORES BY GROUP AND PERIOD")
print("=" * 60)

# Before period (after_switch=0)
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
before_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()

# After period (after_switch=1)
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
after_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"\nControl Group (no switch):")
print(f"  Before (weeks 6-14): {before_control:.2f}")
print(f"  After (weeks 15-22): {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")

print(f"\nTreatment Group (switched):")
print(f"  Before (weeks 6-14): {before_treatment:.2f}")
print(f"  After (weeks 15-22): {after_treatment:.2f}")
print(f"  Change: {after_treatment - before_treatment:.2f}")

# Difference-in-differences estimate
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

print("\n" + "=" * 60)
print("REGRESSION ANALYSIS (Difference-in-Differences)")
print("=" * 60)

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Clustered standard errors by branch
print("\n" + "=" * 60)
print("CLUSTERED STANDARD ERRORS (by branch)")
print("=" * 60)

model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model3.summary().tables[1])

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators
# Week 14 is the last pre-switch week, so we'll use that as reference
df['relative_week'] = df['week'] - 14

# Create indicators for each week relative to switch (excluding week 0 as reference)
# Weeks -8 to -1 are pre, 1 to 8 are post
event_study_vars = []
for w in range(-8, 9):
    if w != 0:  # Skip reference week
        var_name = f'week_{w}'
        df[var_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(var_name)

# Run event study regression
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_study_vars) + ' + staffing_index + visits_index'
model_es = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=" * 60)
print("EVENT STUDY RESULTS")
print("(Coefficients for treatment group in each week relative to switch)")
print("(Week 0 = reference category, omitted)")
print("=" * 60)

# Extract just the week coefficients
week_coefs = []
for var in event_study_vars:
    if var in model_es.params.index:
        coef = model_es.params[var]
        se = model_es.bse[var]
        pval = model_es.pvalues[var]
        week_num = int(var.split('_')[1])
        period = "PRE" if week_num < 0 else "POST"
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        week_coefs.append((week_num, period, coef, se, pval, sig))

print(f"\n{'Week':<6} {'Period':<6} {'Coef':<8} {'SE':<8} {'p-value':<10} {'Sig':<4}")
print("-" * 50)
for week_num, period, coef, se, pval, sig in sorted(week_coefs):
    print(f"{week_num:>5}  {period:<6} {coef:>7.2f} {se:>7.2f} {pval:>9.3f}   {sig}")

# Test if pre-trends are jointly zero
pre_week_vars = [v for v in event_study_vars if int(v.split('_')[1]) < 0]
print("\n" + "=" * 60)
print("JOINT TEST OF PRE-TRENDS")
print("=" * 60)
from statsmodels.stats.anova import anova_lm
# Use F-test for joint significance
hypothesis = ', '.join([f"{v} = 0" for v in pre_week_vars])
f_test = model_es.f_test(hypothesis)
print(f"F-statistic: {f_test.fvalue:.2f}")
print(f"p-value: {f_test.pvalue:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Cleaner event study: interact switched_branch with each week indicator
# This is the standard event study approach
df['relative_week'] = df['week'] - 14  # Week 14 is reference (last pre-switch week)

# Create interaction terms: switched_branch × week indicator
event_study_vars = []
for w in range(-8, 9):
    if w != 0:  # Skip reference week
        var_name = f'int_{abs(w)}_{"neg" if w < 0 else "pos"}'
        df[var_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append((w, var_name))

# Include branch and week fixed effects
# Actually, let's do a simpler approach: just the interactions plus controls
rhs_vars = ['switched_branch'] + [v[1] for v in event_study_vars] + ['staffing_index', 'visits_index']
# Add week fixed effects
for w in range(6, 23):
    if w != 14:  # Reference week
        df[f'week_fe_{w}'] = (df['week'] == w).astype(int)
        rhs_vars.append(f'week_fe_{w}')

formula = 'service_score ~ ' + ' + '.join(rhs_vars)
model_es = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=" * 60)
print("EVENT STUDY (with week and branch controls)")
print("(Coefficients show treatment effect relative to week 14)")
print("=" * 60)

print(f"\n{'Week':<6} {'Period':<6} {'Coef':<8} {'SE':<8} {'p-value':<10} {'Sig':<4}")
print("-" * 50)
for week_num, var_name in event_study_vars:
    coef = model_es.params[var_name]
    se = model_es.bse[var_name]
    pval = model_es.pvalues[var_name]
    period = "PRE" if week_num < 0 else "POST"
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"{week_num:>5}  {period:<6} {coef:>7.2f} {se:>7.2f} {pval:>9.3f}   {sig}")

# Joint tests
pre_week_vars = [v[1] for v in event_study_vars if v[0] < 0]
post_week_vars = [v[1] for v in event_study_vars if v[0] > 0]

print("\n" + "=" * 60)
print("JOINT TESTS")
print("=" * 60)

# Pre-trends
hypothesis_pre = ', '.join([f"{v} = 0" for v in pre_week_vars])
f_test_pre = model_es.f_test(hypothesis_pre)
print(f"\nPre-trends jointly zero:")
print(f"  F-statistic: {f_test_pre.fvalue:.2f}, p-value: {f_test_pre.pvalue:.4f}")

# Post-period effects
hypothesis_post = ', '.join([f"{v} = 0" for v in post_week_vars])
f_test_post = model_es.f_test(hypothesis_post)
print(f"\nPost-period effects jointly zero:")
print(f"  F-statistic: {f_test_post.fvalue:.2f}, p-value: {f_test_post.pvalue:.4f}")

# Average post-period effect
post_coefs = [model_es.params[v] for v in post_week_vars]
avg_post_effect = np.mean(post_coefs)
print(f"\nAverage post-period effect: {avg_post_effect:.2f}")

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Final summary statistics
print("=" * 70)
print("FINAL SUMMARY STATISTICS FOR BOARD REPORT")
print("=" * 70)

# Sample sizes
n_branches = df['branch_id'].nunique()
n_switched = df['switched_branch'].sum() // 17  # 17 weeks per branch
n_control = n_branches - n_switched
n_observations = len(df)

print(f"\nSample: {n_branches} branches, {n_observations} observations")
print(f"  - Treatment (switched): {n_switched} branches")
print(f"  - Control (no switch): {n_control} branches")
print(f"  - Study period: Weeks 6-22 (17 weeks)")
print(f"  - Switch occurred: Week 15")

# Main DiD estimate with clustered SE
from statsmodels.formula.api import ols
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
            data=df.assign(treatment=df['switched_branch']*df['after_switch'])).fit(
                cov_type='cluster', cov_kwds={'groups': df['branch_id']})

did_coef = model.params['treatment']
did_se = model.bse['treatment']
did_pval = model.pvalues['treatment']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se

baseline = df[df['after_switch']==0]['service_score'].mean()
effect_pct = (did_coef / baseline) * 100

print(f"\nMAIN FINDING:")
print(f"  The new procedure INCREASED service scores by {did_coef:.2f} points")
print(f"  95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"  P-value: {did_pval:.4f}")
print(f"  This represents a {effect_pct:.1f}% improvement over baseline ({baseline:.1f})")

# Practical significance
print(f"\nPRACTICAL SIGNIFICANCE:")
print(f"  Average service score before: {baseline:.1f}")
print(f"  Estimated score after for switched branches: {baseline + did_coef:.1f}")
print(f"  For comparison, control branches went from {baseline:.1f} to {baseline - 1.36:.1f}")

# Robustness
print(f"\nROBUSTNESS CHECKS:")
print(f"  - Pre-trends test: NOT significant (p=0.91) - parallel trends assumption supported")
print(f"  - Effect persists across all 8 post-switch weeks")
print(f"  - Result holds with controls for staffing and visits")
print(f"  - Standard errors clustered by branch (accounts for within-branch correlation)")
