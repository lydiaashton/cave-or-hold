import pandas as pd
import numpy as np

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD: Calculate means for each group/period combination
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES: SERVICE SCORE ANALYSIS")
print("=" * 60)

# Group means
group_means = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print("\nGroup Means (service_score):")
print(group_means)

# Calculate DiD manually
# Control group (switched_branch=0)
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
control_change = control_after - control_before

# Treatment group (switched_branch=1)
treat_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treat_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
treat_change = treat_after - treat_before

# DiD estimate
did_estimate = treat_change - control_change

print("\n" + "=" * 60)
print("MANUAL DiD CALCULATION:")
print("=" * 60)
print(f"Control group (old procedure):")
print(f"  Before: {control_before:.2f}")
print(f"  After:  {control_after:.2f}")
print(f"  Change: {control_change:.2f}")
print(f"\nTreatment group (new procedure):")
print(f"  Before: {treat_before:.2f}")
print(f"  After:  {treat_after:.2f}")
print(f"  Change: {treat_change:.2f}")
print(f"\nDiD Estimate (Treatment Change - Control Change): {did_estimate:.2f}")

# Regression-based DiD
print("\n" + "=" * 60)
print("REGRESSION-BASED DiD:")
print("=" * 60)

# Simple DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Simple DiD (no controls)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")

print("CLUSTERED STANDARD ERRORS (by branch):")

control_before = before[before['switched_branch']==0]['service_score'].mean()
control_after = after[after['switched_branch']==0]['service_score'].mean()
treat_before = before[before['switched_branch']==1]['service_score'].mean()
treat_after = after[after['switched_branch']==1]['service_score'].mean()
