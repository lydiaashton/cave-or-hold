import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the visualization
plt.figure(figsize=(12, 6))

viz_data = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Plot for control group
control_data = viz_data[viz_data['switched_branch'] == 0]
plt.plot(control_data['week'], control_data['service_score'], 'o-', label='Control (No Switch)', linewidth=2, markersize=8, color='blue')

# Plot for treatment group
treatment_data = viz_data[viz_data['switched_branch'] == 1]
plt.plot(treatment_data['week'], treatment_data['service_score'], 's-', label='Treatment (Switched)', linewidth=2, markersize=8, color='red')

# Add vertical line at switch point
plt.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch Point (Week 15)')

plt.xlabel('Week', fontsize=12)
plt.ylabel('Mean Service Score', fontsize=12)
plt.title('Service Scores Over Time: Treatment vs Control Groups', fontsize=14, fontweight='bold')
plt.legend(fontsize=11)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_scores_over_time.png', dpi=150)
print("Plot saved as 'service_scores_over_time.png'")

# Run DID with covariates
print("\n=== DID with Covariates ===")
model_covariates = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_covariates.summary().tables[1])

did_coef = model_covariates.params['switched_branch:after_switch']
did_se = model_covariates.bse['switched_branch:after_switch']
did_pvalue = model_covariates.pvalues['switched_branch:after_switch']

print(f"\n=== Key Result (with covariates) ===")
print(f"DID Estimate: {did_coef:.3f}")
print(f"Standard Error: {did_se:.3f}")
print(f"95% CI: [{did_coef - 1.96*did_se:.3f}, {did_coef + 1.96*did_se:.3f}]")
print(f"P-value: {did_pvalue:.4f}")
print(f"Statistically significant: {did_pvalue < 0.05}")

# Effect size relative to baseline
baseline = df['service_score'].mean()
print(f"\n=== Effect Size ===")
print(f"Baseline average service score: {baseline:.2f}")
print(f"DID estimate: {did_coef:.2f}")
print(f"Relative effect: {did_coef/baseline*100:.1f}%")
