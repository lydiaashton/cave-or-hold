import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch'] * df['after_switch']

# Calculate means for the 4 groups
print("=== Mean Service Scores by Group ===")
group_means = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(group_means)

# Calculate DiD manually
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n=== Difference-in-Differences Calculation ===")
print(f"Control group (non-switched): Before={before_control:.2f}, After={after_control:.2f}, Change={after_control - before_control:.2f}")
print(f"Treatment group (switched): Before={before_treatment:.2f}, After={after_treatment:.2f}, Change={after_treatment - before_treatment:.2f}")
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDiD Estimate (treatment effect): {did_estimate:.2f}")

# Run regression with cluster-robust standard errors
print("\n=== Regression Analysis ===")

# Basic DiD model
model1 = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary())

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary())

# Cluster-robust standard errors by branch
model3 = ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 3: DiD with controls and cluster-robust SE (by branch)")
print(model3.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check parallel trends - plot average service scores by week for each group
print("=== Parallel Trends Check ===")
weekly_means = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(weekly_means)

# Plot the trends
plt.figure(figsize=(12, 6))
plt.plot(weekly_means.index, weekly_means[0], 'o-', label='Non-Switched Branches', linewidth=2, markersize=8)
plt.plot(weekly_means.index, weekly_means[1], 's-', label='Switched Branches', linewidth=2, markersize=8)
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Score Trends: Switched vs Non-Switched Branches')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('trends_plot.png', dpi=150)
print("\nPlot saved as trends_plot.png")

# Event study / dynamic DiD - estimate effect for each week after switch
print("\n=== Event Study Analysis ===")
df['week_relative'] = df['week'] - 15  # weeks relative to switch (negative = before, positive = after)

# Create week dummies interacted with treatment
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    df[f'week_{w}'] = ((df['week_relative'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run regression with all week dummies (omitting week -1 as reference)
week_cols = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(week_cols) + ' + staffing_index + visits_index'
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event study coefficients (relative to week -1):")
for w in range(-9, 8):
    if w != -1:
        col = f'week_{w}'
        coef = model_event.params[col]
        se = model_event.bse[col]
        pval = model_event.pvalues[col]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"Week {w}: coef={coef:.2f}, SE={se:.2f}, 95% CI=[{ci_low:.2f}, {ci_high:.2f}], p={pval:.3f} {sig}")

# Confidence interval for the main DiD estimate
model_main = ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
did_coef = model_main.params['treated']
did_se = model_main.bse['treated']
did_ci_low = did_coef - 1.96 * did_se
did_ci_high = did_coef + 1.96 * did_se
print(f"\n=== Main DiD Estimate with 95% CI ===")
print(f"Effect: {did_coef:.2f} points")
print(f"95% Confidence Interval: [{did_ci_low:.2f}, {did_ci_high:.2f}]")
print(f"p-value: {model_main.pvalues['treated']:.3f}")
