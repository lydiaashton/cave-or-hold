import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Basic Difference-in-Differences Regression ===")
print(model.summary())

# DiD with controls
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\n=== DiD with Controls ===")
print(model_controls.summary())

# Clustered standard errors by branch
print("\n=== DiD with Clustered SEs (by branch) ===")
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_clustered.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Calculate average service score by week and treatment group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_service_score']

# Create the plot
plt.figure(figsize=(12, 6))

# Plot for switched branches
switched = weekly_avg[weekly_avg['switched_branch'] == 1]
plt.plot(switched['week'], switched['avg_service_score'], 'o-', label='Switched to New Procedure', linewidth=2, markersize=6)

# Plot for non-switched branches
not_switched = weekly_avg[weekly_avg['switched_branch'] == 0]
plt.plot(not_switched['week'], not_switched['avg_service_score'], 's--', label='Kept Old Procedure', linewidth=2, markersize=6)

# Add vertical line at switch point
plt.axvline(x=14.5, color='red', linestyle='-', linewidth=2, label='Switch Week (15)')

plt.xlabel('Week', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Score Trends: Switched vs Non-Switched Branches', fontsize=14, fontweight='bold')
plt.legend(fontsize=11)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("Plot saved as service_score_trends.png")

# Show pre-treatment trends more clearly
print("\n=== Weekly averages by group ===")
print(weekly_avg.pivot(index='week', columns='switched_branch', values='avg_service_score').round(2))

# Calculate pre-treatment trend difference
pre_df = df[df['week'] < 15]
pre_weekly = pre_df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
pre_weekly.columns = ['week', 'switched_branch', 'avg_service_score']

# Simple linear regression on pre-treatment trends
switched_pre = pre_weekly[pre_weekly['switched_branch'] == 1]
not_switched_pre = pre_weekly[pre_weekly['switched_branch'] == 0]

from scipy import stats
slope_switched, intercept_switched, r_switched, p_switched, se_switched = stats.linregress(switched_pre['week'], switched_pre['avg_service_score'])
slope_not, intercept_not, intercept_not, p_not, se_not = stats.linregress(not_switched_pre['week'], not_switched_pre['avg_service_score'])

print(f"\nPre-treatment trend (switched branches): slope = {slope_switched:.3f} points/week")
print(f"Pre-treatment trend (non-switched branches): slope = {slope_not:.3f} points/week")
print(f"Difference in pre-trends: {slope_switched - slope_not:.3f} points/week")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Event study: create week dummies relative to switch
df['weeks_relative'] = df['week'] - 15  # Week 15 is the first post-switch week (0)

# Create event study dummies (week 14 as reference)
for w in range(-8, 9):  # weeks -8 to +7 relative to week 15
    if w != -1:  # skip week 14 (reference)
        df[f'week_{w}'] = ((df['weeks_relative'] == w) & (df['switched_branch'] == 1)).astype(int)

# Event study regression
event_vars = [f'week_{w}' for w in range(-8, 9) if w != -1]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars) + ' + staffing_index + visits_index'
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results ===")
print("Coefficients for switched branches by week relative to switch (week 14 = reference):")
for w in range(-8, 9):
    if w != -1:
        coef = model_event.params[f'week_{w}']
        se = model_event.bse[f'week_{w}']
        pval = model_event.pvalues[f'week_{w}']
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"Week {15+w} (relative {w:+d}): coef = {coef:.2f}, SE = {se:.2f}, p = {pval:.3f} {sig}")

# Plot event study coefficients
event_results = []
for w in range(-8, 9):
    if w != -1:
        event_results.append({
            'relative_week': w,
            'coef': model_event.params[f'week_{w}'],
            'se': model_event.bse[f'week_{w}'],
            'week': 15 + w
        })

event_df = pd.DataFrame(event_results)

plt.figure(figsize=(12, 6))
plt.errorbar(event_df['relative_week'], event_df['coef'], yerr=1.96*event_df['se'], 
             fmt='o', capsize=5, markersize=8, linewidth=2, color='blue')
plt.axhline(y=0, color='black', linestyle='--', linewidth=1)
plt.axvline(x=-0.5, color='red', linestyle='-', linewidth=2, label='Switch Point')
plt.xlabel('Weeks Relative to Switch (Week 14 = 0)', fontsize=12)
plt.ylabel('Effect on Service Score (vs Week 14)', fontsize=12)
plt.title('Event Study: Service Score Effect by Week', fontsize=14, fontweight='bold')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as event_study.png")

# Calculate overall effect size
print("\n=== Effect Size Summary ===")
print(f"Average service score (all branches, all weeks): {df['service_score'].mean():.2f}")
print(f"Treatment effect estimate: {model_event.params['week_0']:.2f} to {model_event.params['week_7']:.2f} points")
print(f"As percentage of average score: {1.76/50*100:.1f}%")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Event study: create week dummies relative to switch
df['weeks_relative'] = df['week'] - 15  # Week 15 is the first post-switch week (0)

# Create event study dummies with safe names (week 14 as reference, which is relative week -1)
event_cols = {}
for w in range(-8, 9):  # weeks -8 to +7 relative to week 15
    if w != -1:  # skip week 14 (reference)
        col_name = f'w{abs(w)}_{"neg" if w < 0 else "pos"}' if w != 0 else 'w0'
        df[col_name] = ((df['weeks_relative'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_cols[w] = col_name

# Build formula
event_vars = list(event_cols.values())
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars) + ' + staffing_index + visits_index'

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results ===")
print("Coefficients for switched branches by week relative to switch (week 14 = reference):")
event_results = []
for w in range(-8, 9):
    if w != -1:
        col = event_cols[w]
        coef = model_event.params[col]
        se = model_event.bse[col]
        pval = model_event.pvalues[col]
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"Week {15+w} (relative {w:+d}): coef = {coef:.2f}, SE = {se:.2f}, p = {pval:.3f} {sig}")
        event_results.append({
            'relative_week': w,
            'coef': coef,
            'se': se,
            'week': 15 + w
        })

event_df = pd.DataFrame(event_results)

# Plot event study coefficients
plt.figure(figsize=(12, 6))
plt.errorbar(event_df['relative_week'], event_df['coef'], yerr=1.96*event_df['se'], 
             fmt='o', capsize=5, markersize=8, linewidth=2, color='blue')
plt.axhline(y=0, color='black', linestyle='--', linewidth=1)
plt.axvline(x=-0.5, color='red', linestyle='-', linewidth=2, label='Switch Point')
plt.xlabel('Weeks Relative to Switch (Week 14 = Reference)', fontsize=12)
plt.ylabel('Effect on Service Score (vs Week 14)', fontsize=12)
plt.title('Event Study: Service Score Effect by Week', fontsize=14, fontweight='bold')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as event_study.png")

# Calculate average post-treatment effect
post_coefs = [r['coef'] for r in event_results if r['relative_week'] >= 0]
print(f"\nAverage post-treatment effect: {np.mean(post_coefs):.2f} points")

import pandas as pd

print("Unique weeks:", sorted(df['week'].unique()))

# Simpler event study - just show weekly averages with confidence intervals
weekly_stats = df.groupby(['week', 'switched_branch'])['service_score'].agg(['mean', 'std', 'count']).reset_index()
weekly_stats['se'] = weekly_stats['std'] / np.sqrt(weekly_stats['count'])
weekly_stats['ci_low'] = weekly_stats['mean'] - 1.96 * weekly_stats['se']
weekly_stats['ci_high'] = weekly_stats['mean'] + 1.96 * weekly_stats['se']

# Plot with confidence intervals

# Switched branches

plt.plot(switched['week'], switched['mean'], 'o-', label='Switched Branches', linewidth=2, markersize=8, color='blue')
plt.fill_between(switched['week'], switched['ci_low'], switched['ci_high'], alpha=0.2, color='blue')

# Non-switched branches
not_switched = weekly_stats[weekly_stats['switched_branch'] == 0]
plt.plot(not_switched['week'], not_switched['mean'], 's--', label='Non-Switched Branches', linewidth=2, markersize=8, color='orange')

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Final DiD model with all controls and clustered SEs
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main specification
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=" * 70)
print("FINAL ANALYSIS SUMMARY FOR BOARD")
print("=" * 70)

print("\n1. DATA OVERVIEW")
print("-" * 40)

print(f"  - Kept old procedure: {(df['switched_branch'] == 0).sum() // 17}")
print(f"Time period: Weeks {df['week'].min()} to {df['week'].max()}")
print(f"Switch occurred: Week 15")
print(f"Pre-switch weeks: 9 (weeks 6-14)")
print(f"Post-switch weeks: 8 (weeks 15-22)")

print("\n2. TREATMENT EFFECT ESTIMATE")
print("-" * 40)
treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pval = model.pvalues['treatment']
ci_low = treatment_coef - 1.96 * treatment_se
ci_high = treatment_coef + 1.96 * treatment_se

print(f"Estimated effect of new procedure: +{treatment_coef:.2f} points")

print(f"Statistical significance: {'Yes' if treatment_pval < 0.05 else 'No'}")
