import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Check group sizes
print("Branches by group:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate pre/post averages by group
print("\n\nAverage service scores by group and period:")
pre_post = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pre_post)

# Calculate the difference-in-differences manually
# Treatment group (switched_branch=1)
treatment_pre = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_post = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treatment_change = treatment_post - treatment_pre

# Control group (switched_branch=0)
control_pre = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_post = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_change = control_post - control_pre

# Difference-in-differences
did_estimate = treatment_change - control_change

print("\n\nDifference-in-Differences Calculation:")
print(f"Treatment group (switched): Pre={treatment_pre:.2f}, Post={treatment_post:.2f}, Change={treatment_change:.2f}")
print(f"Control group (not switched): Pre={control_pre:.2f}, Post={control_post:.2f}, Change={control_change:.2f}")
print(f"\nDiD Estimate (Treatment Change - Control Change): {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate

model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("DiD Regression Results:")
print(model.summary())

# Also run with branch fixed effects for more robust estimate
# This controls for time-invariant branch characteristics
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ switched_branch * after_switch + C(branch_id_str)', data=df).fit()
print("\n\nDiD Regression with Branch Fixed Effects:")
print(model_fe.summary())

# Check pre-treatment trends (parallel trends assumption)
# Compare trends in weeks 6-14 between treatment and control
pre_data = df[df['week'] < 15].copy()
pre_data['relative_week'] = pre_data['week'] - 6  # weeks 0-8 relative to start

# Calculate average service score by week and group
weekly_avg = pre_data.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
print("\n\nPre-treatment weekly averages:")
print(weekly_avg)

# Simple trend analysis for pre-period
pre_trend_model = ols('service_score ~ week * switched_branch', data=pre_data).fit()
print("\n\nPre-treatment trend test (interaction term shows if trends differ):")
print(pre_trend_model.summary())
