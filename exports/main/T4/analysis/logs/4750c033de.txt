import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate group means
print("=" * 60)
print("SERVICE SCORE AVERAGES BY GROUP AND PERIOD")
print("=" * 60)

# Before period (after_switch = 0)
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()

# After period (after_switch = 1)
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nControl Group (old procedure):")
print(f"  Before (weeks 6-14): {before_control:.2f}")
print(f"  After (weeks 15-22): {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")

print(f"\nTreatment Group (new procedure):")
print(f"  Before (weeks 6-14): {before_treatment:.2f}")
print(f"  After (weeks 15-22): {after_treatment:.2f}")
print(f"  Change: {after_treatment - before_treatment:.2f}")

# Difference-in-differences estimate
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

# Run DiD regression
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD model
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Clustered standard errors by branch
print("\n" + "=" * 60)
print("CLUSTERED STANDARD ERRORS (by branch)")
print("=" * 60)

model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create treatment interaction
df['treatment'] = df['switched_branch'] * df['after_switch']

# DiD with branch fixed effects
print("=" * 60)
print("DID WITH BRANCH FIXED EFFECTS")
print("=" * 60)

# Convert branch_id to categorical for fixed effects
df['branch_id_cat'] = df['branch_id'].astype('category')

# Model with branch fixed effects (absorbs switched_branch)
model_fe = ols('service_score ~ C(branch_id_cat) + after_switch + treatment + staffing_index + visits_index', 
               data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract just the treatment effect

print(f"Standard error: {model_fe.bse['treatment']:.3f}")
print(f"P-value: {model_fe.pvalues['treatment']:.4f}")
print(f"95% CI: [{model_fe.conf_int().loc['treatment'][0]:.3f}, {model_fe.conf_int().loc['treatment'][1]:.3f}]")

# Also run without controls to see raw effect
model_fe_simple = ols('service_score ~ C(branch_id_cat) + after_switch + treatment', 
                      data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study - create week dummies interacted with treatment
# Week 14 as reference (last week before switch)
df['week_14'] = (df['week'] == 14).astype(int)

# Create interactions for each week with switched_branch
week_cols = []
for w in range(6, 23):
    if w != 14:  # reference week
        col = f'week_{w}'
        df[col] = (df['week'] == w).astype(int)
        df[f'{col}_treat'] = df[col] * df['switched_branch']
        week_cols.append(f'{col}_treat')

# Event study regression
print("=" * 60)
print("EVENT STUDY - WEEKLY TREATMENT EFFECTS")
print("=" * 60)

# Build formula
formula = 'service_score ~ C(branch_id) + ' + ' + '.join(week_cols) + ' + staffing_index + visits_index'
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract treatment effects by week
print("\nTreatment effect relative to week 14 (reference):")
print("-" * 50)
results = []
for w in range(6, 23):
    if w != 14:
        coef = model_event.params[f'week_{w}_treat']
        se = model_event.bse[f'week_{w}_treat']
        pval = model_event.pvalues[f'week_{w}_treat']
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        period = "Pre" if w < 15 else "Post"
        results.append({
            'week': w,
            'period': period,
            'coef': coef,
            'se': se,
            'pval': pval,
            'ci_low': ci_low,
            'ci_high': ci_high
        })
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"Week {w:2d} ({period}): {coef:6.2f} (SE: {se:.2f}) [{ci_low:.2f}, {ci_high:.2f}] {sig}")

results_df = pd.DataFrame(results)

# Average pre and post effects
pre_effects = results_df[results_df['period'] == 'Pre']['coef']
post_effects = results_df[results_df['period'] == 'Post']['coef']

print(f"\nAverage pre-switch effect (weeks 6-13): {pre_effects.mean():.2f}")
print(f"Average post-switch effect (weeks 15-22): {post_effects.mean():.2f}")
print(f"Difference (DiD estimate): {post_effects.mean() - pre_effects.mean():.2f}")

# Save event study results
results_df.to_csv('event_study_results.csv', index=False)
print("\nEvent study results saved to event_study_results.csv")
