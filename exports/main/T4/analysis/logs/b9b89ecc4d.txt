import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many unique branches?
print("Unique branches:", df['branch_id'].nunique())

# How many branches switched vs didn't?
branches = df.groupby('branch_id')['switched_branch'].first()
print("\nBranches that switched:", branches.sum())
print("Branches that didn't switch:", len(branches) - branches.sum())

# Check weeks per branch
weeks_per_branch = df.groupby('branch_id')['week'].nunique()
print("\nWeeks per branch (should be 17 weeks, 6-22):")
print(weeks_per_branch.describe())

# Create a before/after comparison
df['period'] = df['after_switch'].map({0: 'Before', 1: 'After'})

# Average service scores by group and period
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print("\n\nService scores by group and period:")
print(summary)

# Calculate the change for each group
before_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n\nMean service scores:")
print(f"Switched branches - Before: {before_switched:.2f}, After: {after_switched:.2f}, Change: {after_switched - before_switched:.2f}")
print(f"Control branches - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")

# Difference-in-differences estimate
diff_in_diff = (after_switched - before_switched) - (after_control - before_control)
print(f"\nDifference-in-differences estimate: {diff_in_diff:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

# Run a difference-in-differences regression
# This controls for branch fixed effects and time trends
model = ols('service_score ~ treated + post + treated_post', data=df).fit()
print("Simple DiD Regression:")
print(model.summary())

print("\n" + "="*60)
print("\nKey coefficient (treated_post):", model.params['treated_post'])
print("Standard error:", model.bse['treated_post'])
print("95% CI:", model.conf_int().loc['treated_post'].values)
print("P-value:", model.pvalues['treated_post'])

# Now let's add branch fixed effects
print("\n\n" + "="*60)
print("DiD with Branch Fixed Effects:")
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ treated_post + C(branch_id_str)', data=df).fit()
print(model_fe.summary())

print("\nKey coefficient (treated_post):", model_fe.params['treated_post'])
print("Standard error:", model_fe.bse['treated_post'])
print("95% CI:", model_fe.conf_int().loc['treated_post'].values)
print("P-value:", model_fe.pvalues['treated_post'])

# Add week fixed effects too

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check if effect varies by branch characteristics
# First, let's see if there's variation across branches
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Branch-level analysis: calculate the change for each branch
branch_summary = df.groupby(['branch_id', 'switched_branch']).apply(
    lambda x: pd.Series({
        'before_mean': x[x['after_switch'] == 0]['service_score'].mean(),
        'after_mean': x[x['after_switch'] == 1]['service_score'].mean(),
        'change': x[x['after_switch'] == 1]['service_score'].mean() - x[x['after_switch'] == 0]['service_score'].mean()
    })
).reset_index()

# Compare changes between switched and non-switched branches
switched_changes = branch_summary[branch_summary['switched_branch'] == 1]['change']
control_changes = branch_summary[branch_summary['switched_branch'] == 0]['change']

print("Branch-level changes in service score:")
print(f"\nSwitched branches (n={len(switched_changes)}):")
print(f"  Mean change: {switched_changes.mean():.2f}")
print(f"  Std dev: {switched_changes.std():.2f}")
print(f"  Median: {switched_changes.median():.2f}")
print(f"  Min: {switched_changes.min():.2f}, Max: {switched_changes.max():.2f}")
print(f"  % with positive change: {(switched_changes > 0).mean()*100:.1f}%")

print(f"\nControl branches (n={len(control_changes)}):")
print(f"  Mean change: {control_changes.mean():.2f}")
print(f"  Std dev: {control_changes.std():.2f}")
print(f"  Median: {control_changes.median():.2f}")
print(f"  Min: {control_changes.min():.2f}, Max: {control_changes.max():.2f}")
print(f"  % with positive change: {(control_changes > 0).mean()*100:.1f}%")

# Statistical test
from scipy import stats
t_stat, p_val = stats.ttest_ind(switched_changes, control_changes)
print(f"\nTwo-sample t-test comparing changes:")
print(f"  t-statistic: {t_stat:.2f}")
print(f"  p-value: {p_val:.2e}")

# Effect size (Cohen's d)
pooled_std = np.sqrt((switched_changes.std()**2 + control_changes.std()**2) / 2)
cohens_d = (switched_changes.mean() - control_changes.mean()) / pooled_std
print(f"  Effect size (Cohen's d): {cohens_d:.2f}")

# Distribution of effects
print("\n\nDistribution of treatment effects (switched - control changes):")
print(f"  25th percentile: {np.percentile(switched_changes - control_changes.mean(), 25):.2f}")
print(f"  50th percentile: {np.percentile(switched_changes - control_changes.mean(), 50):.2f}")
print(f"  75th percentile: {np.percentile(switched_changes - control_changes.mean(), 75):.2f}")

# Check if baseline performance moderates the effect
branch_summary['baseline'] = branch_summary['before_mean']
branch_summary['treated'] = branch_summary['switched_branch']
branch_summary['effect'] = branch_summary['change'] - control_changes.mean()

# For treated branches only, correlate baseline with effect
treated_branches = branch_summary[branch_summary['switched_branch'] == 1]
corr = treated_branches['baseline'].corr(treated_branches['change'])
print(f"\n\nCorrelation between baseline score and change for treated branches: {corr:.2f}")
