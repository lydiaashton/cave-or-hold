import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Create treatment group indicator (interaction of switched_branch and after_switch)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate averages for the 2x2 table
print("=== AVERAGE SERVICE SCORES ===\n")

# Before switch (after_switch = 0)
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()

# After switch (after_switch = 1)
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("Control Group (did not switch):")
print(f"  Before: {before_control:.2f}")
print(f"  After:  {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")
print()
print("Treatment Group (switched):")
print(f"  Before: {before_treatment:.2f}")
print(f"  After:  {after_treatment:.2f}")
print(f"  Change: {after_treatment - before_treatment:.2f}")
print()

# Difference-in-differences
diff_treatment = after_treatment - before_treatment
diff_control = after_control - before_control
did_estimate = diff_treatment - diff_control

print("=== DIFFERENCE-IN-DIFFERENCES ===")
print(f"Treatment group change: {diff_treatment:.2f}")
print(f"Control group change:   {diff_control:.2f}")
print(f"DiD Estimate:           {did_estimate:.2f}")
print()

# Count branches in each group
print("=== GROUP SIZES ===")
switched_branches = df['branch_id'][df['switched_branch'] == 1].nunique()
control_branches = df['branch_id'][df['switched_branch'] == 0].nunique()
print(f"Switched branches: {switched_branches}")
print(f"Control branches:  {control_branches}")
print(f"Total branches:    {switched_branches + control_branches}")

import pandas as pd
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate weekly averages by group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg['switched_branch'] = weekly_avg['switched_branch'].map({0: 'Control (Old)', 1: 'Treatment (New)'})

# Create the plot
fig, ax = plt.subplots(figsize=(10, 6))

for group in weekly_avg['switched_branch'].unique():
    subset = weekly_avg[weekly_avg['switched_branch'] == group]
    ax.plot(subset['week'], subset['service_score'], marker='o', linewidth=2, label=group)

# Add vertical line at switch point
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch (week 15)')
ax.axvspan(15, 22, alpha=0.1, color='green', label='Post-switch period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Scores by Week: Treatment vs Control Groups', fontsize=14)
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("Plot saved as service_scores_trend.png")

# Also calculate effect size relative to baseline
baseline_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
did_effect = 1.75
effect_size_pct = (did_effect / baseline_control) * 100
print(f"\nEffect size: {did_effect:.2f} points ({effect_size_pct:.1f}% of baseline)")
