import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
print("=== Simple Difference-in-Differences Regression ===")
print(model.summary())

# DiD with controls
model_controls = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit()
print("\n=== DiD Regression with Controls ===")
print(model_controls.summary())

# Clustered standard errors by branch
print("\n=== DiD with Clustered SE (by branch) ===")
model_clustered = ols('service_score ~ switched_branch + after_switch + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_clustered.summary())

# Calculate the DiD estimate manually
pre_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
post_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

did_estimate = (post_treatment - pre_treatment) - (post_control - pre_control)
print(f"\n=== Manual DiD Calculation ===")
print(f"Treatment Pre: {pre_treatment:.2f}, Post: {post_treatment:.2f}, Change: {post_treatment - pre_treatment:.2f}")
print(f"Control Pre: {pre_control:.2f}, Post: {post_control:.2f}, Change: {post_control - pre_control:.2f}")
print(f"DiD Estimate: {did_estimate:.2f}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check parallel trends - average service score by week and group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_score']

# Pivot for easier plotting
pivot = weekly_avg.pivot(index='week', columns='switched_branch', values='avg_score')
pivot.columns = ['Control', 'Treatment']
pivot['Difference'] = pivot['Treatment'] - pivot['Control']

print("=== Weekly Averages by Group ===")
print(pivot)

# Check pre-trends statistically (weeks 6-14)
pre_df = df[df['week'] < 15].copy()
pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()

# Test for differential pre-trends
pre_trend_model = ols('service_score ~ switched_branch * week_centered', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print("\n=== Pre-Trend Test (interaction with time) ===")
print(pre_trend_model.summary())

# Event study - relative week effects
df['relative_week'] = df['week'] - 15  # week 15 is the switch week (0)

# Create event study dummies
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    df[f'rel_week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Event study regression (omitting week -1 as reference)
event_study_vars = [f'rel_week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_study_vars)
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n=== Event Study Coefficients ===")
for w in range(-9, 8):
    if w != -1:
        coef = event_model.params[f'rel_week_{w}']
        se = event_model.bse[f'rel_week_{w}']
        print(f"Week {w}: coef={coef:.2f}, se={se:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']

# Main DiD with clustered SE
model = ols('service_score ~ switched_branch + after_switch + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== FINAL DiD MODEL (Clustered SE) ===")
print(model.summary())

# Get the coefficient and confidence interval
did_coef = model.params['did']
did_se = model.bse['did']
did_ci_lower = did_coef - 1.96 * did_se
did_ci_upper = did_coef + 1.96 * did_se
did_pvalue = model.pvalues['did']

print(f"\n=== KEY RESULTS ===")
print(f"DiD Coefficient: {did_coef:.2f}")
print(f"Standard Error: {did_se:.2f}")
print(f"95% CI: [{did_ci_lower:.2f}, {did_ci_upper:.2f}]")
print(f"P-value: {did_pvalue:.3f}")

# Effect size relative to baseline
baseline = df[df['after_switch'] == 0]['service_score'].mean()
effect_pct = (did_coef / baseline) * 100
print(f"\nBaseline average service score: {baseline:.2f}")
print(f"Effect as % of baseline: {effect_pct:.1f}%")

print(f"DiD Coefficient: {model_controls.params['did']:.2f}")
