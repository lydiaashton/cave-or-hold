import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Difference-in-Differences summary table
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES SUMMARY")
print("=" * 60)

# Before period (weeks 6-14)
before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

# Group means
switched_before = before[before['switched_branch'] == 1]['service_score'].mean()
switched_after = after[after['switched_branch'] == 1]['service_score'].mean()
control_before = before[before['switched_branch'] == 0]['service_score'].mean()
control_after = after[after['switched_branch'] == 0]['service_score'].mean()

print("\nAverage Service Scores by Group and Period:")
print("-" * 50)
print(f"                    Before      After       Change")
print(f"Switched Branches:  {switched_before:.2f}      {switched_after:.2f}      {switched_after - switched_before:.2f}")
print(f"Control Branches:   {control_before:.2f}      {control_after:.2f}      {control_after - control_before:.2f}")
print("-" * 50)
print(f"DiD Estimate:       {(switched_after - switched_before) - (control_after - control_before):.2f}")

# Run DiD regression
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model.summary())

# Check parallel trends - compare trends before the switch
print("\n" + "=" * 60)
print("PARALLEL TRENDS CHECK (Pre-switch weeks 6-14)")
print("=" * 60)

pre_switch = df[df['week'] < 15].copy()
pre_switch['week_centered'] = pre_switch['week'] - pre_switch['week'].mean()

# Check if trends were similar before the switch
pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_switch).fit()
print(pre_model.summary())

# Weekly averages by group
print("\n" + "=" * 60)
print("WEEKLY AVERAGES BY GROUP")
print("=" * 60)

weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_avg.columns = ['Control', 'Switched']
weekly_avg['Difference'] = weekly_avg['Switched'] - weekly_avg['Control']
print(weekly_avg.round(2))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# DiD with cluster-robust standard errors
print("=" * 60)
print("DID WITH CLUSTER-ROBUST STANDARD ERRORS")
print("(clustered by branch_id)")
print("=" * 60)

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model.summary())

# Event study - effect by week relative to switch
print("\n" + "=" * 60)
print("EVENT STUDY ANALYSIS")
print("=" * 60)

df['weeks_from_switch'] = df['week'] - 15  # Week 15 is switch week
df['weeks_from_switch'] = df['weeks_from_switch'].clip(-9, 7)  # Cap for cleaner analysis

# Create dummies for each week relative to switch
for w in range(-9, 8):
    df[f'week_{w}'] = (df['weeks_from_switch'] == w).astype(int)

# Run event study regression (omit week -1 as reference)
event_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'switched_branch * {v}' for v in event_vars])
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract treatment effects for each week
print("\nTreatment effect by week (relative to week 14 as baseline):")
print("-" * 50)
for w in range(-9, 8):
    if w == -1:
        continue
    coef_name = f'switched_branch:week_{w}'
    if coef_name in event_model.params:
        coef = event_model.params[coef_name]
        se = event_model.bse[coef_name]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        sig = "***" if event_model.pvalues[coef_name] < 0.01 else "**" if event_model.pvalues[coef_name] < 0.05 else "*" if event_model.pvalues[coef_name] < 0.1 else ""
        print(f"Week {15+w:2d} (t={w:+2d}): {coef:6.2f}  [{ci_low:6.2f}, {ci_high:6.2f}] {sig}")

# Create visualization
fig, ax = plt.subplots(figsize=(12, 6))

# Plot weekly averages
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
ax.plot(weekly_avg.index, weekly_avg[0], 'o-', label='Control Branches (Old Procedure)', linewidth=2, markersize=8)
ax.plot(weekly_avg.index, weekly_avg[1], 's-', label='Switched Branches (New Procedure)', linewidth=2, markersize=8)

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')
ax.axvspan(14.5, 22, alpha=0.1, color='red', label='Post-Switch Period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Scores by Branch Type Over Time', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=10)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("\nChart saved as 'service_scores_trend.png'")

# Additional analysis: check if staffing or visits explain the effect
print("\n" + "=" * 60)
print("CONTROLLING FOR STAFFING AND VISITS")
print("=" * 60)

full_model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
                 data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(full_model.summary())

# Pre-post comparison with confidence intervals

before = df[df['after_switch'] == 0]

# Switched branches
switched_before = before[before['switched_branch'] == 1]['service_score']

# Control branches
control_before = before[before['switched_branch'] == 0]['service_score']
control_after = after[after['switched_branch'] == 0]['service_score']

def ci(series, conf=0.95):
    n = len(series)
    mean = series.mean()
    se = series.std() / np.sqrt(n)
    margin = 1.96 * se
    return mean, mean - margin, mean + margin

print("\nService Score Means (95% CI):")
print("-" * 60)
m, l, h = ci(switched_before)
print(f"Switched Branches - Before: {m:.2f} [{l:.2f}, {h:.2f}]")
m, l, h = ci(switched_after)
print(f"Switched Branches - After:  {m:.2f} [{l:.2f}, {h:.2f}]")
print(f"  Change: {switched_after.mean() - switched_before.mean():.2f}")
print()
m, l, h = ci(control_before)
