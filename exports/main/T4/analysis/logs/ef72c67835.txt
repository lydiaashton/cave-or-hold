import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Create treatment groups
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate means by group and period
print("=== Service Score Means ===")
print("\nBy switched status and period:")
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='after_switch', 
                       aggfunc='mean')
print(pivot)

# Calculate differences
print("\n=== Differences ===")
switched_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
switched_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
nonswitched_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
nonswitched_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print(f"Switched branches: Before={switched_before:.2f}, After={switched_after:.2f}, Change={switched_after-switched_before:.2f}")
print(f"Non-switched branches: Before={nonswitched_before:.2f}, After={nonswitched_after:.2f}, Change={nonswitched_after-nonswitched_before:.2f}")

diff_in_diff = (switched_after - switched_before) - (nonswitched_after - nonswitched_before)
print(f"\nDifference-in-Differences estimate: {diff_in_diff:.2f}")

# Let's also look at weekly trends
print("\n=== Weekly Trends ===")
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
print(weekly_avg)

import pandas as pd
import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Difference-in-differences regression
# Basic DiD model
model = smf.ols('service_score ~ switched_branch * after_switch', data=df)
results = model.fit()
print("=== Basic DiD Regression ===")
print(results.summary())

# With branch fixed effects
print("\n=== DiD with Branch Fixed Effects ===")
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = smf.ols('service_score ~ switched_branch * after_switch + C(branch_id)', data=df)
results_fe = model_fe.fit()
print(results_fe.summary())

# With branch and week fixed effects
print("\n=== DiD with Branch and Week Fixed Effects ===")
model_fefe = smf.ols('service_score ~ switched_branch * after_switch + C(branch_id) + C(week)', data=df)
results_fefe = model_fefe.fit()
print(results_fefe.summary())

# With controls
print("\n=== DiD with Branch, Week FE and Controls ===")
model_controls = smf.ols('service_score ~ switched_branch * after_switch + C(branch_id) + C(week) + staffing_index + visits_index', data=df)
results_controls = model_controls.fit()
print(results_controls.summary())

import pandas as pd
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Test for differential pre-trends
print("=== Pre-trend test ===")
pre_df = df[df['week'] < 15].copy()
pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()
model_pre = smf.ols('service_score ~ switched_branch * week_centered + C(branch_id)', data=pre_df)
results_pre = model_pre.fit()
print("Interaction (switched_branch:week_centered) coefficient:", results_pre.params['switched_branch:week_centered'])
print("P-value:", results_pre.pvalues['switched_branch:week_centered'])

# Also check if there's a significant difference in pre-period averages
print("\n=== Pre-period average difference ===")
pre_avg_switched = pre_df[pre_df['switched_branch']==1]['service_score'].mean()
pre_avg_nonswitched = pre_df[pre_df['switched_branch']==0]['service_score'].mean()
print(f"Switched branches pre-average: {pre_avg_switched:.2f}")
print(f"Non-switched branches pre-average: {pre_avg_nonswitched:.2f}")
print(f"Difference: {pre_avg_switched - pre_avg_nonswitched:.2f}")

# Post-period
post_df = df[df['week'] >= 15]
post_avg_switched = post_df[post_df['switched_branch']==1]['service_score'].mean()
post_avg_nonswitched = post_df[post_df['switched_branch']==0]['service_score'].mean()
print(f"\nSwitched branches post-average: {post_avg_switched:.2f}")
print(f"Non-switched branches post-average: {post_avg_nonswitched:.2f}")
print(f"Difference: {post_avg_switched - post_avg_nonswitched:.2f}")

# Event study - estimate effect for each week relative to switch
print("\n=== Event Study ===")
df['relative_week'] = df['week'] - 15  # week 15 is period 0
# Create indicators for each relative week
for rw in range(-9, 8):  # weeks -9 to +7 relative to switch
    df[f'rel_week_{rw}'] = (df['relative_week'] == rw).astype(int)

# Reference category is week -1 (just before switch)
event_vars = [f'rel_week_{rw}' for rw in range(-9, 8) if rw != -1]
formula = 'service_score ~ ' + ' + '.join([f'{v} * switched_branch' for v in event_vars]) + ' + C(branch_id) + C(week)'
print("Formula:", formula[:200], "...")

model_event = smf.ols(formula, data=df)
results_event = model_event.fit()

# Extract interaction coefficients
print("\nEvent study coefficients (relative to week -1):")
for rw in range(-9, 8):
    if rw != -1:
        coef_name = f'rel_week_{rw}:switched_branch'
        if coef_name in results_event.params:
            coef = results_event.params[coef_name]
            pval = results_event.pvalues[coef_name]
            ci = results_event.conf_int().loc[coef_name]
            sig = '*' if pval < 0.05 else ''
            print(f"Week {rw:+d}: {coef:6.2f} (p={pval:.3f}) [{ci[0]:.2f}, {ci[1]:.2f}] {sig}")

import pandas as pd
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Simpler event study - create week indicators and interact with treatment
df['relative_week'] = df['week'] - 15

# Create cleaner variable names
df['rw'] = df['relative_week'].apply(lambda x: f"rw{int(x)+10}")  # shift to make all positive

# Create dummies for each relative week
for rw in df['relative_week'].unique():
    df[f'd_{int(rw)+10}'] = (df['relative_week'] == rw).astype(int)

# Reference week is -1 (week 14), which is d_9
# We'll estimate effects for all other weeks relative to week 14
event_weeks = [int(rw)+10 for rw in df['relative_week'].unique() if rw != -1]
print("Event weeks to estimate:", event_weeks)

# Build formula
event_terms = [f'd_{w}:switched_branch' for w in event_weeks]
formula = 'service_score ~ ' + ' + '.join(event_terms) + ' + C(branch_id) + C(week)'

model_event = smf.ols(formula, data=df)
results_event = model_event.fit()

print("\n=== Event Study Results ===")
print("Coefficients represent difference relative to week 14 (pre-switch reference)")
print("\nPre-switch weeks (should be ~0 if parallel trends hold):")
for rw in range(-9, 0):  # weeks -9 to -1
    if rw != -1:  # skip reference
        coef_name = f'd_{int(rw)+10}:switched_branch'
        if coef_name in results_event.params:
            coef = results_event.params[coef_name]
            pval = results_event.pvalues[coef_name]
            ci = results_event.conf_int().loc[coef_name]
            sig = '*' if pval < 0.05 else ''
            print(f"Week {15+rw} (rel {rw:+d}): {coef:6.2f} (p={pval:.3f}) {sig}")

print("\nPost-switch weeks (treatment effects):")
for rw in range(0, 8):  # weeks 0 to 7
    coef_name = f'd_{int(rw)+10}:switched_branch'
    if coef_name in results_event.params:
        coef = results_event.params[coef_name]
        pval = results_event.pvalues[coef_name]
        ci = results_event.conf_int().loc[coef_name]
        sig = '*' if pval < 0.05 else ''
        print(f"Week {15+rw} (rel {rw:+d}): {coef:6.2f} (p={pval:.3f}) {sig}")
