# Basic DiD: service_score ~ switched_branch * after_switch

print(model.summary().tables[1])

# Add branch fixed effects (more robust)
# This controls for time-invariant branch characteristics

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate weekly averages by group
weekly = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
weekly.columns = ['Control', 'Treatment']
weekly['Gap'] = weekly['Treatment'] - weekly['Control']

# Create figure with two panels
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 5))

# Panel 1: Service scores over time
ax1.plot(weekly.index, weekly['Control'], 'o-', label='Control (Old Procedure)', 
         linewidth=2, markersize=6, color='steelblue')
ax1.plot(weekly.index, weekly['Treatment'], 's-', label='Treatment (New Procedure)', 
         linewidth=2, markersize=6, color='coral')
ax1.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (Week 15)')
ax1.set_xlabel('Week', fontsize=12)
ax1.set_ylabel('Average Service Score', fontsize=12)
ax1.set_title('Service Scores Over Time', fontsize=14)
ax1.legend(fontsize=10)
ax1.grid(True, alpha=0.3)
ax1.set_xticks(range(6, 23))

# Panel 2: Gap between treatment and control
ax2.bar(weekly.index, weekly['Gap'], color='gray', alpha=0.7)
ax2.axvline(x=14.5, color='red', linestyle='--', linewidth=2)
ax2.axhline(y=0, color='black', linewidth=0.5)
ax2.set_xlabel('Week', fontsize=12)
ax2.set_ylabel('Treatment - Control Gap', fontsize=12)
ax2.set_title('Gap Between Groups Over Time', fontsize=14)
ax2.grid(True, alpha=0.3, axis='y')
ax2.set_xticks(range(6, 23))

# Add average gap lines
avg_gap_pre = weekly[weekly.index < 15]['Gap'].mean()
avg_gap_post = weekly[weekly.index >= 15]['Gap'].mean()
ax2.axhline(y=avg_gap_pre, color='blue', linestyle=':', linewidth=2, label=f'Pre avg: {avg_gap_pre:.1f}')
ax2.axhline(y=avg_gap_post, color='green', linestyle=':', linewidth=2, label=f'Post avg: {avg_gap_post:.1f}')
ax2.legend(fontsize=10)

plt.tight_layout()
plt.savefig('analysis_figures.png', dpi=150)
print("Figures saved as analysis_figures.png")

# Print key statistics for the report
print("\n" + "="*60)
print("KEY STATISTICS FOR BOARD SUMMARY")
print("="*60)
print(f"\nTreatment effect estimate (DiD): +2.09 service score points")
print(f"95% Confidence interval: [1.74, 2.44]")
print(f"Statistical significance: p < 0.001")
print(f"\nContext: Average service score is ~50")
print(f"Effect size: 2.09/50 = {2.09/50*100:.1f}% improvement")
print(f"\nCAVEAT: Treatment group was already improving faster before the switch")
print(f"Pre-treatment trend difference: +0.27 points/week")
print(f"This may inflate the estimated treatment effect")
