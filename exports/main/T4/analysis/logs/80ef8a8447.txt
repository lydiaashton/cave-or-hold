import pandas as pd
import numpy as np
import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create the interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model1 = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Basic DiD Regression ===")
print(model1.summary())

# DiD regression with controls
model2 = smf.ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\n=== DiD Regression with Controls ===")
print(model2.summary())

# Clustered standard errors by branch (important for panel data)
model3 = smf.ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Regression with Clustered SE (by branch) ===")

ax1 = axes[0]
for switched in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == switched]
    label = 'Switched to New Procedure' if switched == 1 else 'Kept Old Procedure'
    ax1.plot(subset['week'], subset['service_score'], marker='o', linewidth=2, label=label, markersize=6)

ax1.set_xticks(range(6, 23))
ax1.grid(True, alpha=0.3)

before_switched = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score']
