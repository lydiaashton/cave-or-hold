import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check how many branches switched vs didn't
branches = df.groupby('branch_id')['switched_branch'].first()
print("Branches that switched:", branches.sum())
print("Branches that didn't switch:", len(branches) - branches.sum())

# Calculate average service scores by group and period
print("\n=== Average Service Scores ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
print(pivot)

# Calculate the differences
print("\n=== Differences ===")
diff_switched = pivot.loc[1, 1] - pivot.loc[1, 0]
diff_control = pivot.loc[0, 1] - pivot.loc[0, 0]
did_estimate = diff_switched - diff_control
print(f"Change for switched branches: {diff_switched:.3f}")
print(f"Change for control branches: {diff_control:.3f}")
print(f"Difference-in-Differences estimate: {did_estimate:.3f}")

# Run a DiD regression
print("\n=== DiD Regression ===")
df['treatment'] = df['switched_branch'] * df['after_switch']
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model.summary())

# Check parallel trends - look at pre-treatment trends
print("\n=== Pre-treatment trends (weeks 6-14) ===")
pre_data = df[df['week'] < 15]
pre_pivot = pre_data.groupby(['switched_branch', 'week'])['service_score'].mean().unstack()
print(pre_pivot)

# Calculate weekly differences before treatment

pre_weekly = pre_data.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

print("=== DiD Regression with Cluster-Robust SE ===")
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_clustered.summary())

# Check parallel trends more formally - event study
print("\n=== Event Study Analysis ===")
# Create week dummies interacted with treatment group
df['week_centered'] = df['week'] - 15  # Center at switch week

# Create indicators for each week relative to switch
for w in range(-9, 8):  # weeks -9 to +7 relative to week 15
    df[f'week_{w}'] = ((df['week'] - 15) == w).astype(int)

# Run event study regression (omitting week -1 as reference)
event_study_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ switched_branch + ' + ' + '.join(event_study_vars)
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event study coefficients (relative to week 14):")
for w in range(-9, 8):
    if w != -1:
        coef = event_model.params[f'week_{w}']
        se = event_model.bse[f'week_{w}']
        print(f"Week {15+w}: {coef:.3f} (SE: {se:.3f})")

print("\n=== Creating visualization ===")

ax.legend()
ax.grid(True, alpha=0.3)

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
from scipy import stats

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Get the treatment effect from the regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se
print(f"95% CI for treatment effect: [{ci_lower:.3f}, {ci_upper:.3f}]")
print(f"P-value: {model.pvalues['treatment']:.3f}")

# Power analysis - what effect size could we detect?
n_treated = 20
n_control = 14
weeks_pre = 9
weeks_post = 8
obs_per_branch = weeks_pre + weeks_post

# Calculate standard deviation of service scores
sd = df['service_score'].std()
print(f"\nStandard deviation of service scores: {sd:.3f}")

# Standard error of the DiD estimator (approximate)
# SE ≈ sqrt(2) * sd / sqrt(n * T) where n is branches per group and T is periods
# More precisely for unbalanced design:
se_approx = sd * np.sqrt(1/(n_treated * weeks_post) + 1/(n_treated * weeks_pre) + 
                          1/(n_control * weeks_post) + 1/(n_control * weeks_pre))
print(f"Approximate SE: {se_approx:.3f}")

# Minimum detectable effect (80% power, alpha=0.05)
mde = 2.8 * se_approx  # Rough approximation
print(f"Approximate minimum detectable effect (80% power): {mde:.3f}")

# Check if there's any relationship with staffing or visits
print("\n=== Correlations ===")
print(df[['service_score', 'staffing_index', 'visits_index']].corr())

# Look at the post-treatment period more carefully
post_data = df[df['week'] >= 15]
print("\n=== Post-treatment period summary ===")
print(post_data.groupby('switched_branch')['service_score'].describe())
