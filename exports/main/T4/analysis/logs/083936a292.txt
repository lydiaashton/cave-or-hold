import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['DiD'] = df['switched_branch'] * df['after_switch']

print("=== Pre vs Post Comparison ===")
print("\nMean service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate DiD manually
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

control_change = control_after - control_before
treatment_change = treatment_after - treatment_before
did_estimate = treatment_change - control_change

print(f"\n=== Simple DiD Calculation ===")
print(f"Control (non-switched) before: {control_before:.2f}")
print(f"Control (non-switched) after: {control_after:.2f}")
print(f"Control change: {control_change:.2f}")
print(f"\nTreatment (switched) before: {treatment_before:.2f}")
print(f"Treatment (switched) after: {treatment_after:.2f}")
print(f"Treatment change: {treatment_change:.2f}")
print(f"\nDiD Estimate (treatment effect): {did_estimate:.2f}")

# Run regression with controls
print("\n=== Regression Analysis ===")

# Model 1: Simple DiD
model1 = ols('service_score ~ switched_branch + after_switch + DiD', data=df).fit()
print("\nModel 1: Simple DiD (no controls)")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + DiD + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Model 3: With branch fixed effects
model3 = ols('service_score ~ after_switch + DiD + staffing_index + visits_index + C(branch_id)', data=df).fit()
print("\nModel 3: DiD with branch fixed effects and controls")
print(model3.summary().tables[1])
