import pandas as pd
import numpy as np
from scipy import stats

# Load the data
df = pd.read_csv('branches.csv')

# Calculate average service scores by group and period
print("=== Average Service Scores ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
pivot.columns = ['Before (weeks 6-14)', 'After (weeks 15-22)']
pivot.index = ['Control (no switch)', 'Treatment (switched)']
print(pivot)

# Calculate the differences
print("\n=== Differences ===")
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

control_diff = control_after - control_before
treatment_diff = treatment_after - treatment_before
did_estimate = treatment_diff - control_diff

print(f"Control group change (after - before): {control_diff:.3f}")
print(f"Treatment group change (after - before): {treatment_diff:.3f}")
print(f"Difference-in-Differences estimate: {did_estimate:.3f}")

# Now let's run a regression to get proper standard errors
# DiD regression: service_score = beta0 + beta1*treatment + beta2*after + beta3*(treatment*after) + error
df['treatment_after'] = df['switched_branch'] * df['after_switch']

# Simple OLS regression
import statsmodels.api as sm

X = df[['switched_branch', 'after_switch', 'treatment_after']]
X = sm.add_constant(X)
y = df['service_score']

model = sm.OLS(y, X).fit()
print("\n=== DiD Regression Results ===")
print(model.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term
df['treatment_after'] = df['switched_branch'] * df['after_switch']

# Run regression with controls (staffing_index and visits_index)
X = df[['switched_branch', 'after_switch', 'treatment_after', 'staffing_index', 'visits_index']]
X = sm.add_constant(X)
y = df['service_score']

# Cluster-robust standard errors
model_with_controls = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== DiD Regression with Controls (Clustered SEs) ===")
print(model_with_controls.summary())

# Let's also check effect sizes in context
print("\n=== Effect Size Context ===")
print(f"Overall mean service score: {df['service_score'].mean():.2f}")
print(f"Overall std dev of service score: {df['service_score'].std():.2f}")
print(f"DiD estimate: {model_with_controls.coef['treatment_after']:.3f}")
print(f"DiD estimate as % of mean: {model_with_controls.coef['treatment_after'] / df['service_score'].mean() * 100:.2f}%")
print(f"DiD estimate in std dev units: {model_with_controls.coef['treatment_after'] / df['service_score'].std():.3f}")

# Confidence interval
ci = model_with_controls.conf_int().loc['treatment_after']
print(f"\n95% CI for treatment effect: [{ci[0]:.3f}, {ci[1]:.3f}]")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Calculate weekly averages by group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack()
weekly_avg = weekly_avg.T  # weeks as rows
weekly_avg.columns = ['Control (no switch)', 'Treatment (switched)']
weekly_avg.index.name = 'Week'

# Create the plot
fig, ax = plt.subplots(figsize=(12, 6))

# Plot the trends
ax.plot(weekly_avg.index, weekly_avg['Control (no switch)'], 'o-', label='Control (no switch)', linewidth=2, markersize=8)
ax.plot(weekly_avg.index, weekly_avg['Treatment (switched)'], 's-', label='Treatment (switched)', linewidth=2, markersize=8)

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch implemented (week 15)')

# Add shading for before/after periods
ax.axvspan(6, 14.5, alpha=0.2, color='green', label='Before switch')
ax.axvspan(14.5, 22, alpha=0.2, color='blue', label='After switch')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Score Trends: Treatment vs Control Branches\n(Difference-in-Differences Analysis)', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=10)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("Plot saved as 'service_score_trends.png'")

# Print the data for the plot
print("\n=== Weekly Average Service Scores ===")
print(weekly_avg.round(2))
