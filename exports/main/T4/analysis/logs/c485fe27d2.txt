# Load the data

# Create the DiD interaction term
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Calculate pre/post means by treatment status
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES: PRE/POST MEANS")
print("=" * 60)

pre_means = df[df['after_switch'] == 0].groupby('switched_branch')['service_score'].mean()

print(f"  Treatment group (switched_branch=1): {post_means[1]:.2f}")
print(f"  Difference (Treatment - Control): {post_means[1] - post_means[0]:.2f}")

# Calculate DiD estimate
did_estimate = (post_means[1] - post_means[0]) - (pre_means[1] - pre_means[0])

# Run regression

print("REGRESSION RESULTS")

model1 = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit()

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create week dummies relative to switch (week 15 is the switch)
# Week 14 will be the reference (last pre-switch week)
df['relative_week'] = df['week'] - 14  # week 14 = 0, week 15 = 1, etc.

# Create interaction terms for each week with treatment
# We'll use week 14 (relative_week=0) as reference
for w in range(-8, 9):  # weeks 6-22 relative to week 14
    if w != 0:  # skip reference week
        df[f'week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
week_cols = [f'week_{w}' for w in range(-8, 9) if w != 0]
formula = 'service_score ~ switched_branch + ' + ' + '.join(week_cols) + ' + staffing_index + visits_index'

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the week interactions
coef_dict = {}
for w in range(-8, 9):
    if w == 0:
        coef_dict[w] = 0  # reference
    else:
        col = f'week_{w}'
        coef_dict[w] = model_event.params[col]

# Get confidence intervals
ci_dict = {}
for w in range(-8, 9):
    if w == 0:
        ci_dict[w] = (0, 0)
    else:
        col = f'week_{w}'
        ci = model_event.conf_int().loc[col]
        ci_dict[w] = (ci[0], ci[1])

print("=" * 60)
print("EVENT STUDY: EFFECT BY WEEK RELATIVE TO SWITCH")
print("=" * 60)
print("\n(Week 0 = week 14, the last pre-switch week, is the reference)")
print("\nWeek  RelWeek  Coef    95% CI")
print("-" * 45)
for w in range(-8, 9):
    week_num = w + 14
    period = "Pre" if w < 0 else ("Switch" if w == 0 else "Post")
    ci = ci_dict[w]
    print(f"{week_num:2d}    {w:3d}      {coef_dict[w]:6.2f}   [{ci[0]:6.2f}, {ci[1]:6.2f}]  {period}")

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))

weeks = list(coef_dict.keys())
coefs = [coef_dict[w] for w in weeks]
ci_lower = [ci_dict[w][0] for w in weeks]
ci_upper = [ci_dict[w][1] for w in weeks]

ax.errorbar(weeks, coefs, yerr=[[(c - coef_dict[w]) for w, c in zip(weeks, ci_lower)], 
                                 [(c - coef_dict[w]) for w, c in zip(weeks, ci_upper)]],

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create week dummies relative to switch (week 15 is the switch)
# Week 14 will be the reference (last pre-switch week)
df['relative_week'] = df['week'] - 14  # week 14 = 0, week 15 = 1, etc.

# Create interaction terms for each week with treatment (avoid negative numbers in column names)
for w in range(-8, 9):  # weeks 6-22 relative to week 14
    if w != 0:  # skip reference week
        col_name = f'w{w:+d}'  # e.g., w-8, w-1, w+1, w+8
        df[col_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
week_cols = [f'w{w:+d}' for w in range(-8, 9) if w != 0]
formula = 'service_score ~ switched_branch + ' + ' + '.join(week_cols) + ' + staffing_index + visits_index'

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the week interactions
coef_dict = {}
for w in range(-8, 9):
    if w == 0:
        coef_dict[w] = 0  # reference
    else:
        col = f'w{w:+d}'
        coef_dict[w] = model_event.params[col]

# Get confidence intervals
ci_dict = {}
for w in range(-8, 9):
    if w == 0:
        ci_dict[w] = (0, 0)
    else:
        col = f'w{w:+d}'
        ci = model_event.conf_int().loc[col]
        ci_dict[w] = (ci[0], ci[1])

print("=" * 60)
print("EVENT STUDY: EFFECT BY WEEK RELATIVE TO SWITCH")
print("=" * 60)
print("\n(Week 0 = week 14, the last pre-switch week, is the reference)")
print("\nWeek  RelWeek  Coef    95% CI")
print("-" * 45)
for w in range(-8, 9):
    week_num = w + 14
    period = "Pre" if w < 0 else ("Ref" if w == 0 else "Post")
    ci = ci_dict[w]
    print(f"{week_num:2d}    {w:3d}      {coef_dict[w]:6.2f}   [{ci[0]:6.2f}, {ci[1]:6.2f}]  {period}")

# Plot event study

weeks = list(coef_dict.keys())
coefs = [coef_dict[w] for w in weeks]

# Calculate error bars
yerr_lower = [coef_dict[w] - ci_dict[w][0] for w in weeks]

import pandas as pd
import numpy as np
from statsmodels.api import OLS, add_constant
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create week dummies relative to switch (week 15 is the switch)
# Week 14 will be the reference (last pre-switch week)
df['relative_week'] = df['week'] - 14  # week 14 = 0, week 15 = 1, etc.

# Create interaction terms for each week with treatment (use simple names)
week_dummies = {}
for w in range(-8, 9):  # weeks 6-22 relative to week 14
    if w != 0:  # skip reference week
        col_name = f'week{abs(w)}_{"neg" if w < 0 else "pos"}'
        week_dummies[w] = col_name
        df[col_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Build design matrix
X_cols = ['switched_branch', 'staffing_index', 'visits_index'] + list(week_dummies.values())
X = df[X_cols]
X = add_constant(X)
y = df['service_score']

# Fit model with cluster-robust SE
model = OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the week interactions
coef_dict = {}
ci_dict = {}
for w in range(-8, 9):
    if w == 0:
        coef_dict[w] = 0  # reference
        ci_dict[w] = (0, 0)
    else:
        col = week_dummies[w]
        coef_dict[w] = model.params[col]
        ci = model.conf_int().loc[col]
        ci_dict[w] = (ci[0], ci[1])

print("=" * 60)
print("EVENT STUDY: EFFECT BY WEEK RELATIVE TO SWITCH")
print("=" * 60)
print("\n(Week 0 = week 14, the last pre-switch week, is the reference)")
print("\nWeek  RelWeek  Coef    95% CI")
print("-" * 45)
for w in range(-8, 9):
    week_num = w + 14
    period = "Pre" if w < 0 else ("Ref" if w == 0 else "Post")
    ci = ci_dict[w]
    print(f"{week_num:2d}    {w:3d}      {coef_dict[w]:6.2f}   [{ci[0]:6.2f}, {ci[1]:6.2f}]  {period}")

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))

weeks = list(coef_dict.keys())
coefs = [coef_dict[w] for w in weeks]

# Calculate error bars
yerr_lower = [coef_dict[w] - ci_dict[w][0] for w in weeks]
yerr_upper = [ci_dict[w][1] - coef_dict[w] for w in weeks]
yerr = [yerr_lower, yerr_upper]

ax.errorbar(weeks, coefs, yerr=yerr, fmt='o', capsize=5, linewidth=2, markersize=8, color='blue', label='Treatment Effect')

ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch Point')

ax.set_xlabel('Weeks Relative to Switch (0 = Week 14)', fontsize=12)
ax.set_ylabel('Effect on Service Score', fontsize=12)
ax.set_title('Event Study: Treatment Effect by Week', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=11)
ax.grid(True, alpha=0.3)
ax.set_xticks(weeks)

plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\n\nEvent study plot saved as 'event_study.png'")

# Print model summary
print("\n" + "=" * 60)
print("FULL MODEL SUMMARY")
print("=" * 60)
print(model.summary())
