import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# === 1. Simple DiD Table ===
print("=== Difference-in-Differences Summary ===")

# Mean service scores by group and period
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
pivot.columns = ['Pre (weeks 6-14)', 'Post (weeks 15-22)']
pivot.index = ['Control (no switch)', 'Treatment (switched)']
print("\nMean Service Scores:")
print(pivot)

# Calculate the DiD estimate manually
control_pre = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_post = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treat_pre = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_post = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

control_change = control_post - control_pre
treat_change = treat_post - treat_pre
did_estimate = treat_change - control_change

print(f"\nControl group change: {control_change:.3f}")
print(f"Treatment group change: {treat_change:.3f}")
print(f"DiD Estimate (treatment effect): {did_estimate:.3f}")

# === 2. Regression Analysis ===
print("\n\n=== Regression Analysis ===")

# Model 1: Basic DiD
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# Model 2: With controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Model 3: With branch fixed effects
print("\nModel 3: DiD with branch fixed effects")
model3 = ols('service_score ~ after_switch + treatment + staffing_index + visits_index + C(branch_id)', data=df).fit()
print(model3.summary().tables[1])

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model with branch fixed effects
model_fe = ols('service_score ~ after_switch + treatment + staffing_index + visits_index + C(branch_id)', data=df).fit()

treatment_coef = model_fe.params['treatment']
treatment_se = model_fe.bse['treatment']
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se

print("=== Final Treatment Effect Estimate ===")
print(f"Coefficient: {treatment_coef:.3f}")
print(f"Standard Error: {treatment_se:.3f}")
print(f"95% CI: [{ci_lower:.3f}, {ci_upper:.3f}]")
print(f"P-value: {model_fe.pvalues['treatment']:.4f}")

# Effect size
std_score = df['service_score'].std()
effect_size = treatment_coef / std_score
print(f"\nEffect size: {effect_size:.3f} standard deviations")

# Also get robust standard errors (clustered by branch)
print("\n=== Clustered Standard Errors (by branch) ===")
model_fe_cluster = model_fe.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(f"Clustered SE: {model_fe_cluster.bse['treatment']:.3f}")
ci_lower_cl = treatment_coef - 1.96 * model_fe_cluster.bse['treatment']

print(f"95% CI (clustered): [{ci_lower_cl:.3f}, {ci_upper_cl:.3f}]")
print(f"P-value (clustered): {model_fe_cluster.pvalues['treatment']:.4f}")
