import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run DiD regression with cluster-robust standard errors
print("=== DiD Regression Analysis ===")

# Basic DiD model
model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary())

# DiD with controls
model2 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit()
print("\n\nModel 2: DiD with controls (staffing_index, visits_index)")
print(model2.summary())

# Cluster-robust standard errors by branch
print("\n\n=== Cluster-Robust Standard Errors (by branch) ===")
cov_clusters = model2.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(cov_clusters.summary())

# Extract the key coefficient
did_coef = cov_clusters.params['switched_branch:after_switch']
did_se = cov_clusters.bse['switched_branch:after_switch']
did_pvalue = cov_clusters.pvalues['switched_branch:after_switch']
print(f"\nDiD coefficient: {did_coef:.3f}")
print(f"Standard error: {did_se:.3f}")
print(f"P-value: {did_pvalue:.4f}")
print(f"95% CI: [{did_coef - 1.96*did_se:.3f}, {did_coef + 1.96*did_se:.3f}]")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study regression - estimate effect for each post-switch week
# Using week 14 (relative week -1) as reference
df['relative_week'] = df['week'] - 14

# Create dummy variables for each week relative to switch
post_weeks = [0, 1, 2, 3, 4, 5, 6, 7, 8]  # weeks 15-22 relative to week 14
pre_weeks = [-8, -7, -6, -5, -4, -3, -2, -1]  # weeks 6-13 relative to week 14

# Create indicator variables
for w in post_weeks:
    df[f'post_w{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (omitting pre week -1 as reference)
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'post_w{w}' for w in post_weeks])
model_es = ols(formula, data=df).fit()
cov_es = model_es.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("=== Event Study Regression Results ===")
print(cov_es.summary())

# Extract post-switch effects
print("\n=== Post-Switch Week Effects (vs pre-switch baseline) ===")
for w in post_weeks:
    coef = cov_es.params[f'post_w{w}']
    se = cov_es.bse[f'post_w{w}']
    pval = cov_es.pvalues[f'post_w{w}']
    ci_low = coef - 1.96 * se
    ci_high = coef + 1.96 * se
    print(f"Week {14+w}: Effect = {coef:.2f}, SE = {se:.2f}, 95% CI = [{ci_low:.2f}, {ci_high:.2f}], p = {pval:.4f}")

# Calculate average post-switch effect
post_effects = [cov_es.params[f'post_w{w}'] for w in post_weeks]
avg_effect = np.mean(post_effects)
print(f"\nAverage post-switch effect: {avg_effect:.2f}")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study regression - estimate effect for each post-switch week
df['relative_week'] = df['week'] - 14

# Create dummy variables for each week relative to switch
post_weeks = [0, 1, 2, 3, 4, 5, 6, 7, 8]  # weeks 15-22 relative to week 14

for w in post_weeks:
    df[f'post_w{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'post_w{w}' for w in post_weeks])
model_es = ols(formula, data=df).fit()
cov_es = model_es.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Extract post-switch effects
print("=== Post-Switch Week Effects (vs pre-switch baseline) ===")
results = []
for w in post_weeks:
    coef = cov_es.params[f'post_w{w}']
    se = cov_es.bse[f'post_w{w}']
    pval = cov_es.pvalues[f'post_w{w}']
    ci_low = coef - 1.96 * se
    ci_high = coef + 1.96 * se
    results.append({'week': 14+w, 'effect': coef, 'se': se, 'pval': pval, 'ci_low': ci_low, 'ci_high': ci_high})
    print(f"Week {14+w}: Effect = {coef:.2f}, SE = {se:.2f}, 95% CI = [{ci_low:.2f}, {ci_high:.2f}], p = {pval:.4f}")

# Calculate average post-switch effect

print(f"\nAverage post-switch effect: {avg_effect:.2f}")
