import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch*after_switch) + error
# beta3 is the DiD estimator - the effect of the treatment

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Extract the DiD coefficient
did_effect = model.params['treatment']
did_pvalue = model.pvalues['treatment']
did_confint = model.conf_int().loc['treatment']

print(f"\n=== DiD Effect Estimate ===")
print(f"Effect of new procedure: {did_effect:.3f} points")
print(f"95% CI: [{did_confint[0]:.3f}, {did_confint[1]:.3f}]")
print(f"P-value: {did_pvalue:.6f}")
print(f"Statistically significant at 5% level: {did_pvalue < 0.05}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Check parallel trends - compare trends before the switch
print("=== Parallel Trends Check (Pre-Switch Period) ===")
pre_switch = df[df['week'] < 15].copy()

# Calculate average service score by week for each group
weekly_avg = pre_switch.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print("\nWeekly averages before switch:")
print(weekly_avg)
print(f"\nDifference (treatment - control) by week:")
weekly_diff = weekly_avg[1] - weekly_avg[0]
print(weekly_diff)

# Regression with branch fixed effects
print("\n=== Regression with Branch Fixed Effects ===")
# Create branch dummies
df_with_fe = df.copy()
branch_dummies = pd.get_dummies(df_with_fe['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df_with_fe, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [col for col in df_with_fe.columns if col.startswith('branch')]
formula = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(branch_cols)

model_fe = ols(formula, data=df_with_fe).fit()
print(f"DiD coefficient with branch FE: {model_fe.params['treatment']:.3f}")
print(f"P-value: {model_fe.pvalues['treatment']:.6f}")

# Regression with cluster-robust standard errors
print("\n=== Regression with Cluster-Robust SE (by branch) ===")
model_cluster = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_cluster.summary().tables[1])

did_effect_cluster = model_cluster.params['treatment']
did_pvalue_cluster = model_cluster.pvalues['treatment']
did_confint_cluster = model_cluster.conf_int().loc['treatment']

print(f"\n=== Clustered DiD Effect Estimate ===")
print(f"Effect of new procedure: {did_effect_cluster:.3f} points")
print(f"95% CI: [{did_confint_cluster[0]:.3f}, {did_confint_cluster[1]:.3f}]")
print(f"P-value: {did_pvalue_cluster:.6f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Check if staffing and visits differ between groups
print("=== Covariate Balance Check ===")
print("\nStaffing Index by group:")
print(df.groupby(['switched_branch', 'after_switch'])['staffing_index'].mean())

print("\nVisits Index by group:")
print(df.groupby(['switched_branch', 'after_switch'])['visits_index'].mean())

# DiD with controls
print("\n=== DiD with Controls (staffing and visits) ===")
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"DiD coefficient with controls: {model_controls.params['treatment']:.3f}")
print(f"P-value: {model_controls.pvalues['treatment']:.6f}")
print(f"95% CI: [{model_controls.conf_int().loc['treatment'][0]:.3f}, {model_controls.conf_int().loc['treatment'][1]:.3f}]")

# Effect size relative to baseline
baseline = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
effect_size_pct = (model_controls.params['treatment'] / baseline) * 100
print(f"\nEffect size: {model_controls.params['treatment']:.2f} points ({effect_size_pct:.1f}% of baseline)")

# Create a visualization data summary
print("\n=== Summary Table for Report ===")
summary = df.groupby(['switched_branch', 'after_switch']).agg({
    'service_score': ['mean', 'std', 'count']
}).round(2)
print(summary)

import pandas as pd

# Create plot data - average service score by week for each group

# Plot the two groups
ax.plot(weekly_avg.index, weekly_avg[0], 'o-', label='Control (Old Procedure)', linewidth=2, markersize=6, color='#2E86AB')
ax.plot(weekly_avg.index, weekly_avg[1], 's-', label='Treatment (New Procedure)', linewidth=2, markersize=6, color='#A23B72')

# Set x-axis ticks
