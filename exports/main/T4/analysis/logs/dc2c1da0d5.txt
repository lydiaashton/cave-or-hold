import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Check if pre-trends are parallel by regressing on interaction of week and treatment
df['week_centered'] = df['week'] - 10  # Center around middle of pre-period
df['pre_period'] = (df['week'] < 15).astype(int)
df['week_x_treatment_pre'] = df['week_centered'] * df['switched_branch'] * df['pre_period']

# Test for parallel pre-trends
pre_trend_model = ols('service_score ~ switched_branch + week_centered + week_x_treatment_pre', 
                      data=df[df['week'] < 15]).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Pre-Trend Test (Interaction of Week and Treatment in Pre-Period) ===")
print(pre_trend_model.summary())

# Also run an event study style analysis - effect by week relative to switch
df['weeks_relative'] = df['week'] - 15  # Week 15 is the switch (0), before is negative, after is positive

# Create dummies for each week relative to switch, omitting week -1 as reference
week_dummies = pd.get_dummies(df['weeks_relative'], prefix='week_rel')
df_with_dummies = pd.concat([df, week_dummies], axis=1)

# Build regression formula
week_cols = [col for col in week_dummies.columns if col != 'week_rel_-1']  # Omit week -1 as reference
formula = 'service_score ~ switched_branch + ' + ' + '.join(week_cols)

event_study = ols(formula, data=df_with_dummies).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Event Study (Week-by-Week Effects) ===")
# Print just the week coefficients
week_coefs = event_study.params[week_cols]
week_se = event_study.bse[week_cols]
week_results = pd.DataFrame({'coef': week_coefs, 'se': week_se})
week_results['t_stat'] = week_results['coef'] / week_results['se']
week_results['p_value'] = 2 * (1 - sm.distributions.norm.cdf(abs(week_results['t_stat'])))
print(week_results.round(4))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Test for parallel pre-trends - simpler approach
# Just check if the slope of service scores over time differs between groups before the switch
pre_data = df[df['week'] < 15].copy()

# Simple regression with interaction
pre_trend_model = ols('service_score ~ switched_branch * week', data=pre_data).fit()
print("=== Pre-Trend Test (Interaction of Week and Treatment in Pre-Period) ===")
print(pre_trend_model.summary())

# Extract the interaction coefficient - this tests if trends differ
interaction_coef = pre_trend_model.params['switched_branch:week']
interaction_pval = pre_trend_model.pvalues['switched_branch:week']
print(f"\nPre-trend interaction coefficient: {interaction_coef:.4f}, p-value: {interaction_pval:.4f}")
print("If p > 0.05, we cannot reject parallel trends assumption.")

# Now let's do the event study with a simpler approach
# Calculate the difference in means for each week
weekly_diff = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_diff['diff'] = weekly_diff[1] - weekly_diff[0]
print("\n=== Weekly Differences (Treatment - Control) ===")
print(weekly_diff.round(2))

# Calculate average pre and post differences
pre_diff = weekly_diff[weekly_diff.index < 15]['diff'].mean()
post_diff = weekly_diff[weekly_diff.index >= 15]['diff'].mean()
print(f"\nAverage pre-switch difference: {pre_diff:.2f}")
print(f"Average post-switch difference: {post_diff:.2f}")
print(f"DiD estimate (post - pre): {post_diff - pre_diff:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create DiD term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD with covariates (staffing_index and visits_index)
model_with_covariates = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
                            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== DiD Regression with Covariates ===")
print(model_with_covariates.summary())

# Calculate effect size relative to baseline
baseline_mean = df[df['after_switch'] == 0]['service_score'].mean()
treatment_effect = model_with_covariates.params['treatment']
effect_size_pct = (treatment_effect / baseline_mean) * 100
print(f"\nBaseline mean service score: {baseline_mean:.2f}")
print(f"Treatment effect: {treatment_effect:.2f}")
print(f"Effect size: {effect_size_pct:.2f}%")

# Check robustness - what if we only look at branches with complete data?
# Count observations per branch
branch_counts = df.groupby('branch_id').size()
complete_branches = branch_counts[branch_counts == 17].index  # 17 weeks of data
df_complete = df[df['branch_id'].isin(complete_branches)]
print(f"\nBranches with complete data: {len(complete_branches)} out of {df['branch_id'].nunique()}")

# Run DiD on complete cases
model_complete = ols('service_score ~ switched_branch + after_switch + treatment', 
                     data=df_complete).fit(cov_type='cluster', cov_kwds={'groups': df_complete['branch_id']})
print("\n=== DiD on Complete Cases Only ===")
print(model_complete.summary())

import pandas as pd

import matplotlib
matplotlib.use('Agg')

# Load the data
df = pd.read_csv('branches.csv')

# Create a summary visualization
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Plot 1: Trends over time
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_score']
pivot = weekly_avg.pivot(index='week', columns='switched_branch', values='avg_score')

ax1 = axes[0]
ax1.plot(pivot.index, pivot[0], 'o-', label='Control (Old Procedure)', linewidth=2, markersize=6, color='blue')
ax1.plot(pivot.index, pivot[1], 's-', label='Treatment (New Procedure)', linewidth=2, markersize=6, color='green')
ax1.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Week (15)')
ax1.set_xlabel('Week', fontsize=12)
ax1.set_ylabel('Average Service Score', fontsize=12)
ax1.set_title('Service Score Trends by Group', fontsize=14)
ax1.legend(loc='best', fontsize=10)
ax1.grid(True, alpha=0.3)
ax1.set_xticks(range(6, 23))

# Plot 2: Before/After comparison
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

ax2 = axes[1]
x = np.array([0, 1, 2, 3])
bars = ax2.bar(x, [before_control, after_control, before_treatment, after_treatment], 
               color=['lightblue', 'blue', 'lightgreen', 'green'], edgecolor='black')
ax2.set_xticks(x)
ax2.set_xticklabels(['Control\nBefore', 'Control\nAfter', 'Treatment\nBefore', 'Treatment\nAfter'], fontsize=10)
ax2.set_ylabel('Average Service Score', fontsize=12)
ax2.set_title('Before/After Comparison by Group', fontsize=14)
ax2.grid(True, alpha=0.3, axis='y')

# Add value labels on bars
for i, v in enumerate([before_control, after_control, before_treatment, after_treatment]):
    ax2.text(i, v + 0.3, f'{v:.1f}', ha='center', fontsize=10)
