import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Check how many branches switched vs didn't
print("=== Branch Groups ===")
switched_branches = df[df['switched_branch'] == 1]['branch_id'].nunique()
control_branches = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"Switched branches (treatment): {switched_branches}")
print(f"Control branches: {control_branches}")

# Check data completeness per branch
print("\n=== Data Completeness ===")
weeks_per_branch = df.groupby('branch_id')['week'].count()
print(f"Weeks per branch (min, max): {weeks_per_branch.min()}, {weeks_per_branch.max()}")
print(f"Expected weeks per branch: 17 (weeks 6-22)")

# Check if all branches have all weeks
incomplete = weeks_per_branch[weeks_per_branch < 17]
if len(incomplete) > 0:
    print(f"\nBranches with incomplete data: {len(incomplete)}")
    print(incomplete)

# Calculate pre and post averages by group
print("\n=== Pre/Post Averages by Group ===")
df['period'] = np.where(df['after_switch'] == 0, 'Pre', 'Post')

pre_post = df.groupby(['switched_branch', 'period'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pre_post)

# Calculate the difference-in-differences
print("\n=== Difference-in-Differences Calculation ===")
# Treatment group (switched_branch=1)
treatment_pre = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_post = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_change = treatment_post - treatment_pre

# Control group (switched_branch=0)
control_pre = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_post = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
control_change = control_post - control_pre

print(f"Treatment group: Pre={treatment_pre:.2f}, Post={treatment_post:.2f}, Change={treatment_change:.2f}")
print(f"Control group: Pre={control_pre:.2f}, Post={control_post:.2f}, Change={control_change:.2f}")
print(f"\nDifference-in-Differences (DiD) estimate: {treatment_change - control_change:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate

df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== DiD Regression Results ===")
print(model.summary())

# Also run with cluster-robust standard errors by branch
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Regression with Clustered SE (by branch) ===")
print(model_clustered.summary())

# Check parallel trends - compare pre-period trends
print("\n=== Pre-Period Trends Check ===")
pre_data = df[df['after_switch'] == 0]
pre_trend = ols('service_score ~ week + switched_branch + week:switched_branch', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print(pre_trend.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create trend data by week and group
trend_data = df.groupby(['week', 'switched_branch'])['service_score'].agg(['mean', 'std', 'count']).reset_index()
trend_data['se'] = trend_data['std'] / np.sqrt(trend_data['count'])

# Plot the trends
fig, ax = plt.subplots(figsize=(10, 6))

# Treatment group (switched)
treatment = trend_data[trend_data['switched_branch'] == 1]
ax.plot(treatment['week'], treatment['mean'], 'b-o', label='Switched Branches (Treatment)', linewidth=2, markersize=6)
ax.fill_between(treatment['week'], 
                treatment['mean'] - 1.96*treatment['se'], 
                treatment['mean'] + 1.96*treatment['se'], 
                alpha=0.2, color='blue')

# Control group (not switched)
control = trend_data[trend_data['switched_branch'] == 0]
ax.plot(control['week'], control['mean'], 'r-s', label='Non-Switched Branches (Control)', linewidth=2, markersize=6)
ax.fill_between(control['week'], 
                control['mean'] - 1.96*control['se'], 
                control['mean'] + 1.96*control['se'], 
                alpha=0.2, color='red')

# Add vertical line at switch point
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch Point (Week 15)')
ax.axvspan(14.5, 22, alpha=0.1, color='green', label='Post-Switch Period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Service Score', fontsize=12)
ax.set_title('Service Score Trends: Switched vs Non-Switched Branches', fontsize=14, fontweight='bold')
ax.legend(loc='upper left')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
