# Load the data
df = pd.read_csv('branches.csv')

# Difference-in-differences calculation
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"  Change: {treatment_after - treatment_before:.3f}")
print(f"\nDifference-in-Differences estimate: {(treatment_after - treatment_before) - (control_after - control_before):.3f}")

# Run a regression with DiD

# Load the data

# Event study / dynamic DiD - check effects by week relative to switch
# Create relative week variable (weeks 6-14 are -9 to -1, weeks 15-22 are 0 to 7)
df['relative_week'] = df['week'] - 15

# Create dummies for each week relative to switch, interacted with treatment
# Use week 14 (relative_week = -1) as reference
for w in range(-9, 8):
    if w != -1:  # skip reference week
        df[f'week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
event_study_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_study_vars)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create relative week variable (weeks 6-14 are -9 to -1, weeks 15-22 are 0 to 7)
df['relative_week'] = df['week'] - 15

# Create dummies for each week relative to switch, interacted with treatment
# Use week 14 (relative_week = -1) as reference
# Rename to avoid negative sign issues: w01, w02, ..., w09 for negative, w10, w11, ... for positive
for w in range(-9, 8):
    if w != -1:  # skip reference week
        if w < 0:
            var_name = f'w{abs(w):02d}'  # w01, w02, ..., w09 for weeks -9 to -1
        else:
            var_name = f'p{w:02d}'  # p00, p01, ..., p07 for weeks 0 to 7
        df[var_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Build formula
event_study_vars = []
for w in range(-9, 8):
    if w != -1:
        if w < 0:
            event_study_vars.append(f'w{abs(w):02d}')
        else:
            event_study_vars.append(f'p{w:02d}')

formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_study_vars)
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the interaction terms
coeffs = []
weeks = []
se_list = []
for w in range(-9, 8):
    if w == -1:
        coeffs.append(0)  # reference category
        weeks.append(w)
        se_list.append(0)
    else:
        if w < 0:
            var_name = f'w{abs(w):02d}'
        else:
            var_name = f'p{w:02d}'
        coeffs.append(model_event.params[var_name])
        weeks.append(w)
        se_list.append(model_event.bse[var_name])

# Plot event study with confidence intervals
fig, ax = plt.subplots(figsize=(12, 6))
ax.axvline(x=-0.5, color='red', linestyle='--', label='Switch point')
ax.axhline(y=0, color='gray', linestyle='-', alpha=0.5)

# Plot coefficients with CI
lower = [c - 1.96*s for c, s in zip(coeffs, se_list)]
upper = [c + 1.96*s for c, s in zip(coeffs, se_list)]
ax.fill_between(weeks, lower, upper, alpha=0.3, color='blue')
ax.plot(weeks, coeffs, marker='o', linestyle='-', linewidth=2, markersize=8, color='blue', label='Treatment Effect')
ax.set_xlabel('Weeks Relative to Switch (0 = first week after switch)')
ax.set_ylabel('Coefficient (vs. control group)')
ax.set_title('Event Study: Service Score Effects by Week')
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-9, 8))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("Event study plot saved as event_study.png")

# Print the coefficients
print("=== Event Study Coefficients ===")
for w, c, s in zip(weeks, coeffs, se_list):
    print(f"Week {w:+3d}: {c:6.3f} (SE: {s:.3f})")

# Check if pre-trends are jointly significant (weeks -9 to -2)
pre_trend_vars = []
for w in range(-9, -1):
    pre_trend_vars.append(f'w{abs(w):02d}')

# Get the F-test for pre-trend coefficients
R = np.zeros((len(pre_trend_vars), len(model_event.params)))
for i, var in enumerate(pre_trend_vars):
    if var in model_event.params.index:
        R[i, model_event.params.index.get_loc(var)] = 1

test = model_event.f_test(R)
print(f"\n=== Pre-trend Joint Test ===")
print(f"F-statistic: {test.fvalue:.3f}")
print(f"P-value: {test.pvalue:.4f}")
print(f"Pre-trends jointly significant: {test.pvalue < 0.05}")

import pandas as pd

from statsmodels.formula.api import ols

# The treatment effect

print(f"Percentage improvement: {treatment_effect/baseline_avg*100:.1f}%")

                  data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

df['treat_high_vis'] = df['treatment'] * df['high_visits']
df['treat_low_vis'] = df['treatment'] * (1 - df['high_visits'])

                data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Effect by Visit Level ===")
print(f"High visit branches: {model_vis.params['treat_high_vis']:.3f} (p={model_vis.pvalues['treat_high_vis']:.4f})")
print(f"Low visit branches: {model_vis.params['treat_low_vis']:.3f} (p={model_vis.pvalues['treat_low_vis']:.4f})")

fig, axes = plt.subplots(1, 2, figsize=(14, 5))

import pandas as pd

from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')

# Load the data
df = pd.read_csv('branches.csv')

# Re-examine the pre-trends more carefully
# The event study showed significant pre-trend violations
# Let's look at this more carefully

print("=== Re-examining Pre-Trend Assumption ===\n")

# Calculate weekly averages for both groups
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly['diff'] = weekly[1] - weekly[0]

print("Weekly averages and differences:")
print(weekly)

# Pre-switch weeks (6-14)
pre_weeks = weekly[weekly.index < 15]
print(f"\nPre-switch period (weeks 6-14):")
print(f"  Average difference (treatment - control): {pre_weeks['diff'].mean():.3f}")
print(f"  Std dev of differences: {pre_weeks['diff'].std():.3f}")
print(f"  Trend in differences (slope): {np.polyfit(range(len(pre_weeks)), pre_weeks['diff'].values, 1)[0]:.3f}")

# Post-switch weeks (15-22)
post_weeks = weekly[weekly.index >= 15]
print(f"\nPost-switch period (weeks 15-22):")
print(f"  Average difference (treatment - control): {post_weeks['diff'].mean():.3f}")
print(f"  Std dev of differences: {post_weeks['diff'].std():.3f}")

# The change in the gap
pre_avg_diff = pre_weeks['diff'].mean()
post_avg_diff = post_weeks['diff'].mean()
print(f"\n  Change in gap (post - pre): {post_avg_diff - pre_avg_diff:.3f}")

# Key concern: pre-trends were NOT parallel
print("\n" + "="*60)
print("CRITICAL CONCERN: Pre-Trend Violation")
print("="*60)
print("""
The event study analysis showed that pre-trends were NOT parallel:
- Week -7: Treatment group was +6.2 points ahead (p<0.001)
- Week -2: Treatment group was +5.7 points ahead (p<0.001)
- Joint pre-trend test: F=186.2, p<0.001 (HIGHLY SIGNIFICANT)

This violates the key assumption for difference-in-differences.
The treatment and control groups were on DIFFERENT trajectories
even BEFORE the switch occurred.

This means the 1.89 point 'effect' may be overstated because:
1. Treatment branches may have been systematically different
2. The groups may have been converging/diverging naturally
3. Selection bias: branches chosen to switch may differ systematically
""")

# Using only the last 3 pre-weeks as baseline (weeks 12-14)
recent_pre = df[(df['week'] >= 12) & (df['week'] < 15)]
recent_post = df[df['week'] >= 15]

recent_pre_avg = recent_pre.groupby('switched_branch')['service_score'].mean()
recent_post_avg = recent_post.groupby('switched_branch')['service_score'].mean()

print("\n=== More Conservative Estimate (Recent Pre-Period) ===")
print(f"Using only weeks 12-14 as baseline:")

conservative_did = (recent_post_avg[1] - recent_pre_avg[1]) - (recent_post_avg[0] - recent_pre_avg[0])
print(f"  DiD estimate: {conservative_did:.3f}")

# Also check: what if we control for branch fixed effects properly?
print("\n=== Branch Fixed Effects Model ===")
# Use within-branch variation only
df['post'] = df['after_switch']
df['treat_post'] = df['switched_branch'] * df['after_switch']

# This model uses only within-branch variation
model_fe = ols('service_score ~ post + treat_post + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Note: switched_branch drops out due to collinearity with fixed effects
print(f"Treatment coefficient (with branch FE): {model_fe.params['treat_post']:.3f}")
