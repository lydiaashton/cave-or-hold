import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators
# Week 14 is the last pre-switch week, use as reference
df['relative_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create dummies for each relative week (excluding week 0 as reference)
for rw in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if rw != 0:
        df[f'rel_week_{rw}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Event study regression
event_study_vars = [f'rel_week_{rw}' for rw in range(-8, 9) if rw != 0]
formula = 'service_score ~ switched_branch + C(week) + ' + ' + '.join(event_study_vars)

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results ===")
print("Coefficients show treatment effect relative to week 14 (reference period)")
print("\nRelative Week | Coefficient | Std Error | p-value")
print("-" * 55)

results = []
for rw in range(-8, 9):
    if rw == 0:
        results.append((rw, 0, 0, 1.0))  # Reference period
    else:
        var_name = f'rel_week_{rw}'
        coef = model_event.params[var_name]
        se = model_event.bse[var_name]
        pval = model_event.pvalues[var_name]
        results.append((rw, coef, se, pval))
        marker = "*" if pval < 0.05 else ""
        print(f"{rw:13d} | {coef:11.3f} | {se:9.3f} | {pval:.3f} {marker}")

# Plot event study
fig, ax = plt.subplots(figsize=(14, 7))

rws = [r[0] for r in results]
coefs = [r[1] for r in results]
ses = [r[2] for r in results]

# Calculate confidence intervals
ci_lower = [coefs[i] - 1.96 * ses[i] for i in range(len(coefs))]
ci_upper = [coefs[i] + 1.96 * ses[i] for i in range(len(coefs))]

ax.errorbar(rws, coefs, yerr=[ [coefs[i]-ci_lower[i] for i in range(len(coefs))], [ci_upper[i]-coefs[i] for i in range(len(coefs))] ], 
            fmt='o', capsize=5, markersize=8, linewidth=2, color='blue', label='Treatment Effect')

ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch Point')
ax.set_xlabel('Relative Week (0 = Week 14, last pre-switch week)')
ax.set_ylabel('Service Score Effect (relative to control)')
ax.set_title('Event Study: Dynamic Treatment Effects on Service Score', fontweight='bold', fontsize=14)
ax.legend(loc='best')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")

# Summary statistics for post-period effects
post_effects = [r[1] for r in results if r[0] > 0]
print(f"\nAverage post-switch effect: {np.mean(post_effects):.3f}")
print(f"Range of post-switch effects: [{min(post_effects):.3f}, {max(post_effects):.3f}]")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators
# Week 14 is the last pre-switch week, use as reference
df['relative_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create dummies for each relative week (excluding week 0 as reference)
# Use 'w' prefix to avoid issues with negative numbers in variable names
for rw in range(-8, 9):
    if rw != 0:
        df[f'w{abs(rw)}_{"neg" if rw < 0 else "pos"}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Build formula
event_study_vars = []
for rw in range(-8, 9):
    if rw < 0:
        event_study_vars.append(f'w{abs(rw)}_neg')
    elif rw > 0:
        event_study_vars.append(f'w{rw}_pos')

formula = 'service_score ~ switched_branch + C(week) + ' + ' + '.join(event_study_vars)

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results ===")
print("Coefficients show treatment effect relative to week 14 (reference period)")
print("\nRelative Week | Coefficient | Std Error | p-value")
print("-" * 55)

results = []
for rw in range(-8, 9):
    if rw == 0:
        results.append((rw, 0, 0, 1.0))  # Reference period
    else:
        if rw < 0:
            var_name = f'w{abs(rw)}_neg'
        else:
            var_name = f'w{rw}_pos'
        coef = model_event.params[var_name]
        se = model_event.bse[var_name]
        pval = model_event.pvalues[var_name]
        results.append((rw, coef, se, pval))
        marker = "*" if pval < 0.05 else ""
        print(f"{rw:13d} | {coef:11.3f} | {se:9.3f} | {pval:.3f} {marker}")

# Plot event study
fig, ax = plt.subplots(figsize=(14, 7))

rws = [r[0] for r in results]
coefs = [r[1] for r in results]
ses = [r[2] for r in results]

# Calculate confidence intervals
ci_lower = [coefs[i] - 1.96 * ses[i] for i in range(len(coefs))]
ci_upper = [coefs[i] + 1.96 * ses[i] for i in range(len(coefs))]

ax.errorbar(rws, coefs, yerr=[ [coefs[i]-ci_lower[i] for i in range(len(coefs))], [ci_upper[i]-coefs[i] for i in range(len(coefs))] ], 
            fmt='o', capsize=5, markersize=8, linewidth=2, color='blue', label='Treatment Effect')

ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch Point')
ax.set_xlabel('Relative Week (0 = Week 14, last pre-switch week)')
ax.set_ylabel('Service Score Effect (relative to control)')
ax.set_title('Event Study: Dynamic Treatment Effects on Service Score', fontweight='bold', fontsize=14)
ax.legend(loc='best')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")

# Summary statistics for post-period effects
post_effects = [r[1] for r in results if r[0] > 0]
print(f"\nAverage post-switch effect: {np.mean(post_effects):.3f}")
print(f"Range of post-switch effects: [{min(post_effects):.3f}, {max(post_effects):.3f}]")

# Count how many post-period effects are statistically significant
sig_count = sum(1 for r in results if r[0] > 0 and r[3] < 0.05)
print(f"Statistically significant post-period effects (p<0.05): {sig_count} out of {len(post_effects)}")

print(f"\n=== PRACTICAL SIGNIFICANCE ===")

from scipy import stats
coef = model.params['switched_branch:after_switch']
se = model.bse['switched_branch:after_switch']
prob_positive = 1 - stats.norm.cdf(0, coef, se)
print(f"\nProbability effect is positive (Bayesian approximation): {prob_positive:.1%}")

print(f"\n=== WHAT COULD WE DETECT? ===")
n_branches = 34
n_weeks = 17
n_post = 8

n_treat = 20
