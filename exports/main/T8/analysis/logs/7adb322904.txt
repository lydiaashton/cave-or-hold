import pandas as pd

pre_means = df[df['after_switch'] == 0].groupby('switched_branch')['service_score'].mean()

post_means = df[df['after_switch'] == 1].groupby('switched_branch')['service_score'].mean()

treatment_pre = pre_means[1]

treatment_diff = treatment_post - treatment_pre
did_estimate = treatment_diff - control_diff

print(f"Treatment group change (post - pre): {treatment_diff:.3f}")

import pandas as pd
import statsmodels.api as sm
from statsmodels.formula.api import ols
import numpy as np

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Basic DiD regression
print("=== Difference-in-Differences Regression ===")
print("\nModel 1: Basic DiD (no controls)")
model1 = ols('service_score ~ treatment + post + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model1.summary().tables[1])

print("\nModel 2: DiD with covariates (staffing_index, visits_index)")
model2 = ols('service_score ~ treatment + post + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model2.summary().tables[1])

# Check pre-trends more formally - test if trends differ in pre-period
print("\n=== Pre-trends Test ===")
pre_df = df[df['after_switch'] == 0].copy()
pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()
pre_df['treatment_week_interaction'] = pre_df['switched_branch'] * pre_df['week_centered']

pre_model = ols('service_score ~ switched_branch + week_centered + treatment_week_interaction', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print(pre_model.summary().tables[1])
print(f"\nPre-trend interaction p-value: {pre_model.pvalues['treatment_week_interaction']:.4f}")

import pandas as pd
import matplotlib.pyplot as plt
import numpy as np

df = pd.read_csv('branches.csv')

# Create plot of average service scores over time by group
plt.figure(figsize=(12, 6))

# Calculate weekly averages by group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)

weeks = weekly_avg.index
control = weekly_avg[0].values
treatment = weekly_avg[1].values

plt.plot(weeks, control, 'o-', label='Control (Old Procedure)', linewidth=2, markersize=8, color='#2E86AB')
plt.plot(weeks, treatment, 's-', label='Treatment (New Procedure)', linewidth=2, markersize=8, color='#A23B72')

# Add vertical line at switch point
plt.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch occurred')

# Add shaded region for post-period
plt.axvspan(14.5, 22, alpha=0.2, color='green', label='Post-switch period')

plt.xlabel('Week', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Scores Over Time: Control vs Treatment Branches', fontsize=14, fontweight='bold')
plt.legend(loc='best', fontsize=10)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.ylim(44, 56)

plt.tight_layout()
plt.savefig('service_scores_over_time.png', dpi=150)
print("Plot saved as service_scores_over_time.png")

# Create a summary table
print("\n=== Summary Table ===")
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).unstack(level=0)
print(summary)

# Calculate effect size relative to baseline
baseline = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
