import pandas as pd
import numpy as np
import statsmodels.api as sm
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create relative week indicators (week 14 is the last pre-switch week, so it's the reference)
# Week 15 is the first post-switch week
df['relative_week'] = df['week'] - 14  # week 14 becomes 0, week 15 becomes 1, etc.

# Create indicators for each relative week (excluding week 0 as reference)
# We'll use weeks -8 to -1 as pre, and 1 to 8 as post
for rw in range(-8, 9):
    if rw != 0:
        df[f'rw_{rw}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
rw_cols = [f'rw_{rw}' for rw in range(-8, 9) if rw != 0]
X_event = df[rw_cols + ['switched_branch']]  # include switched_branch main effect
X_event = sm.add_constant(X_event)
y = df['service_score']

model_event = sm.OLS(y, X_event).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the relative week interactions
coef_dict = {}
se_dict = {}
for rw in range(-8, 9):
    if rw == 0:
        coef_dict[rw] = 0  # reference period
        se_dict[rw] = 0
    else:
        col = f'rw_{rw}'
        coef_dict[rw] = model_event.params[col]
        se_dict[rw] = model_event.bse[col]

event_results = pd.DataFrame({
    'relative_week': list(range(-8, 9)),
    'coef': [coef_dict[rw] for rw in range(-8, 9)],
    'se': [se_dict[rw] for rw in range(-8, 9)]
})
event_results['ci_lower'] = event_results['coef'] - 1.96 * event_results['se']
event_results['ci_upper'] = event_results['coef'] + 1.96 * event_results['se']

print("Event Study Results (coefficients relative to week 14):")
print(event_results.to_string(index=False))

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))

ax.errorbar(event_results['relative_week'], event_results['coef'], 
            yerr=1.96 * event_results['se'], fmt='o', capsize=5, 
            color='blue', markersize=8, linewidth=2, label='Effect estimate')

ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=0.5, color='red', linestyle='--', linewidth=2, label='Switch implemented')
ax.axvspan(0.5, 8.5, alpha=0.1, color='green', label='Post-switch period')

ax.set_xlabel('Weeks Relative to Switch (0 = Week 14, last pre-switch week)', fontsize=12)
ax.set_ylabel('Service Score Effect (Treatment vs Control)', fontsize=12)
ax.set_title('Event Study: Dynamic Effects of New Shelving Procedure', fontsize=14, fontweight='bold')
ax.legend(loc='best')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-8, 9))

plt.tight_layout()
plt.savefig('event_study_plot.png', dpi=150)

df = pd.read_csv('branches.csv')

# Two-way fixed effects model (branch + week fixed effects)
# This controls for time-invariant branch characteristics and common time shocks

# Create dummy variables for branches and weeks

# Treatment indicator
df['treatment'] = df['switched_branch'] * df['after_switch']

# Combine all regressors
X_fe = pd.concat([df[['treatment', 'staffing_index', 'visits_index']], branch_dummies, week_dummies], axis=1)
y = df['service_score']

# This will be a large regression, let's use a more efficient approach
# Actually, let's use the linearmodels package for fixed effects if available, otherwise do it manually

try:
    from linearmodels.panel import PanelOLS
    
    # Set up panel data
    df_panel = df.copy()
    df_panel = df_panel.set_index(['branch_id', 'week'])
    
    # TWFE model
    mod = PanelOLS(df_panel['service_score'],

import pandas as pd

df = pd.read_csv('branches.csv')

# Clean heterogeneous DiD analysis
# Step 1: Get branch pre-scores
branch_pre = df[df['after_switch'] == 0].groupby('branch_id')['service_score'].mean().reset_index()
branch_pre.columns = ['branch_id', 'pre_score']
median_pre = branch_pre['pre_score'].median()

df = df.merge(branch_pre, on='branch_id')
df['high_baseline'] = (df['pre_score'] > median_pre).astype(int)

# Step 2: Run separate DiD regressions for each group
print("=" * 70)
print("CORRECTED HETEROGENEOUS EFFECTS ANALYSIS")
print("=" * 70)

results = {}
for group_name, group_df in [('HIGH baseline', df[df['high_baseline'] == 1]), 
                              ('LOW baseline', df[df['high_baseline'] == 0])]:
    
    group_df = group_df.copy()
    group_df['treatment'] = group_df['switched_branch'] * group_df['after_switch']
    
    X = group_df[['switched_branch', 'after_switch', 'treatment', 'staffing_index', 'visits_index']]
    X = sm.add_constant(X)
    y = group_df['service_score']
    
    model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': group_df['branch_id']})
    treatment_effect = model.params['treatment']
    se = model.bse['treatment']
    pval = model.pvalues['treatment']
    ci_lower = treatment_effect - 1.96 * se
    ci_upper = treatment_effect + 1.96 * se
    
    results[group_name] = {
        'effect': treatment_effect,
        'se': se,
        'pval': pval,
        'ci': (ci_lower, ci_upper),
        'n_branches': group_df['branch_id'].nunique()
    }
    
    print(f"\n{group_name} (n={results[group_name]['n_branches']} branches):")
    print(f"  Treatment effect: {treatment_effect:.2f} points")
    print(f"  95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
    print(f"  p-value: {pval:.4f}")

# Test if effects are different
print("\n" + "=" * 70)
print("TEST: Are the effects different between groups?")
print("=" * 70)

effect_diff = results['HIGH baseline']['effect'] - results['LOW baseline']['effect']
# Approximate SE for difference (assuming independence)
se_diff = np.sqrt(results['HIGH baseline']['se']**2 + results['LOW baseline']['se']**2)
z_stat = effect_diff / se_diff
p_diff = 2 * (1 - 0.5 * (1 + np.math.erf(abs(z_stat) / np.sqrt(2))))

print(f"Difference in effects: {effect_diff:.2f} points")
print(f"SE of difference: {se_diff:.2f}")
print(f"p-value for difference: {p_diff:.4f}")

if p_diff > 0.05:
    print("\n✓ Effects are NOT significantly different between groups")
else:
    print("\n⚠ Effects ARE significantly different between groups")
