import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
switched_counts = df.groupby('switched_branch')['branch_id'].nunique()
print("Branches by switch status:")
print(switched_counts)
print(f"\nTotal branches: {df['branch_id'].nunique()}")
print(f"Switched: {switched_counts.get(1, 0)}")
print(f"Not switched: {switched_counts.get(0, 0)}")

# Check data completeness - does every branch have all weeks?
weeks_per_branch = df.groupby('branch_id')['week'].nunique()
print(f"\nWeeks per branch - min: {weeks_per_branch.min()}, max: {weeks_per_branch.max()}")

# Average service score by group and period
print("\n\nAverage service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(pivot)

# Calculate the difference-in-differences manually
# Before period: after_switch = 0
# After period: after_switch = 1

before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nControl group (no switch): Before={before_control:.2f}, After={after_control:.2f}, Change={after_control - before_control:.2f}")
print(f"Treatment group (switched): Before={before_treatment:.2f}, After={after_treatment:.2f}, Change={after_treatment - before_treatment:.2f}")

did_effect = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {did_effect:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run a regression with the DiD specification
# Basic DiD model
model_basic = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Basic DiD Regression:")
print(model_basic.summary())

# Now with branch fixed effects
df['branch_id'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ after_switch + treatment + C(branch_id)', data=df).fit()
print("\n\nDiD with Branch Fixed Effects:")
print(model_fe.summary())

# With branch and week fixed effects
df['week'] = df['week'].astype(str)
model_fe_week = ols('service_score ~ after_switch + treatment + C(branch_id) + C(week)', data=df).fit()
print("\n\nDiD with Branch and Week Fixed Effects:")
print(model_fe_week.summary())

# Get the coefficient and confidence interval for treatment
coef = model_fe_week.params['treatment']
ci = model_fe_week.conf_int().loc['treatment']
print(f"\n\nTreatment effect: {coef:.3f}")
print(f"95% CI: [{ci[0]:.3f}, {ci[1]:.3f}]")
print(f"P-value: {model_fe_week.pvalues['treatment']:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['week_num'] = df['week'].astype(int)
df['treatment'] = df['switched_branch'] * df['after_switch']
df['branch_id_str'] = df['branch_id'].astype(str)
df['week_str'] = df['week'].astype(str)

# Main model
model = ols('service_score ~ after_switch + treatment + C(branch_id_str) + C(week_str)', data=df).fit()

print("="*60)
print("CRITICAL RE-EVALUATION")
print("="*60)

# Effect size context
mean_score = df['service_score'].mean()
std_score = df['service_score'].std()
