import pandas as pd

df = pd.read_csv('branches.csv')

print("\nPivot table (mean service scores):")

print("\n--- Differences ---")

before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()

print(f"Before difference (switched - control): {before_switched - before_control:.2f}")

import numpy as np

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression
# service_score = beta0 + beta1*switched + beta2*after + beta3*(switched*after) + error
# beta3 is the DiD estimate
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Difference-in-Differences Regression Results:")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run regression with branch fixed effects
# This controls for time-invariant branch characteristics
df['branch_cat'] = df['branch_id'].astype('category')
df['week_cat'] = df['week'].astype('category')

# Model with branch and week fixed effects
model_fe = ols('service_score ~ treatment + C(branch_cat) + C(week_cat)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Regression with Branch and Week Fixed Effects:")
print("=" * 70)
print(model_fe.summary())

# Extract just the treatment effect
treatment_coef = model_fe.params['treatment']
treatment_se = model_fe.bse['treatment']
treatment_pvalue = model_fe.pvalues['treatment']
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se

print(f"\n\n=== KEY RESULT ===")
print(f"Treatment Effect (New Procedure): {treatment_coef:.2f} points")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {treatment_pvalue:.3f}")
