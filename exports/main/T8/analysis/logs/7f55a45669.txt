import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run DiD regression with cluster-robust standard errors
# Basic DiD model
model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("=== Model 1: Basic DiD ===")
print(model1.summary())

# With cluster-robust SEs at branch level
print("\n=== Model 1 with Cluster-Robust SEs (by branch) ===")
cov_clusters = sm.cov_type.clustered
cov_kwds = {'groups': df['branch_id']}
model1_robust = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds=cov_kwds)
print(model1_robust.summary())

# Add controls for staffing and visits
model2 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds=cov_kwds)
print("\n=== Model 2: DiD with Controls (staffing, visits) ===")
print(model2.summary())

# Check parallel trends more formally - interact switched_branch with week dummies for pre-period
print("\n=== Parallel Trends Test (Pre-period only, weeks 6-14) ===")
pre_df = df[df['week'] < 15].copy()
pre_df['week_cat'] = pre_df['week'].astype(str)
# Use week 6 as reference
model_trend = ols('service_score ~ switched_branch * C(week_cat)', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print(model_trend.summary())

# Look at the interaction terms specifically
