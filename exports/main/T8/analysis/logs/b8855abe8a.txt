import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treated'] * df['post']

# Calculate group means
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES ANALYSIS")
print("=" * 60)

# Group means before and after
before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

print("\n1. SERVICE SCORES BY GROUP AND PERIOD")
print("-" * 60)

# Switched branches
switched_before = before[before['switched_branch'] == 1]['service_score'].mean()
switched_after = after[after['switched_branch'] == 1]['service_score'].mean()
switched_change = switched_after - switched_before

# Non-switched branches (control)
control_before = before[before['switched_branch'] == 0]['service_score'].mean()
control_after = after[after['switched_branch'] == 0]['service_score'].mean()
control_change = control_after - control_before

print(f"\nSwitched branches (treatment group):")
print(f"  Before switch (weeks 6-14): {switched_before:.2f}")
print(f"  After switch (weeks 15-22): {switched_after:.2f}")
print(f"  Change: {switched_change:+.2f}")

print(f"\nNon-switched branches (control group):")
print(f"  Before switch (weeks 6-14): {control_before:.2f}")
print(f"  After switch (weeks 15-22): {control_after:.2f}")
print(f"  Change: {control_change:+.2f}")

# DiD estimate
did_estimate = switched_change - control_change
print(f"\n{'=' * 60}")
print(f"DIFFERENCE-IN-DIFFERENCES ESTIMATE: {did_estimate:+.2f}")
print(f"{'=' * 60}")

# Run regression with cluster-robust standard errors
print("\n2. REGRESSION ANALYSIS")
print("-" * 60)

# Basic DiD regression
model = ols('service_score ~ treated + post + did', data=df).fit()
print("\nBasic DiD Regression (without clustering):")
print(model.summary().tables[1])

# Regression with branch fixed effects and cluster-robust SEs
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ treated + post + did + C(branch_id_str)', data=df).fit()
print("\nDiD Regression with Branch Fixed Effects:")
print(model_fe.summary().tables[1])

# Cluster-robust standard errors
cov_type = 'cluster'
cov_kwds = {'groups': df['branch_id']}
model_cluster = ols('service_score ~ treated + post + did + C(branch_id_str)', data=df).fit(cov_type=cov_type, cov_kwds=cov_kwds)
print("\nDiD Regression with Branch Fixed Effects (Cluster-Robust SEs):")
print(model_cluster.summary().tables[1])
