final_model = smf.ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Final DiD Model Results (with controls, clustered SEs):")
print("=" * 70)
print(f"Treatment Effect (new procedure): {final_model.params['treatment']:.2f} points")
print(f"Standard Error: {final_model.bse['treatment']:.2f}")
print(f"95% CI: [{final_model.conf_int().loc['treatment'][0]:.2f}, {final_model.conf_int().loc['treatment'][1]:.2f}]")
print(f"P-value: {final_model.pvalues['treatment']:.4f}")

# Calculate effect size relative to baseline
baseline_mean = df[df['after_switch'] == 0]['service_score'].mean()
effect_size_pct = (final_model.params['treatment'] / baseline_mean) * 100
print(f"\nBaseline average service score: {baseline_mean:.1f}")
print(f"Effect as % of baseline: {effect_size_pct:.1f}%")

# Also check if the effect is practically significant
print(f"\nStandard deviation of service scores: {df['service_score'].std():.1f}")
print(f"Effect size in SD units: {final_model.params['treatment'] / df['service_score'].std():.2f} SD")

# Summary table for the board
print("\n\n" + "=" * 70)
