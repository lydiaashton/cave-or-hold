import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Run Difference-in-Differences regression
# Basic DiD model
model1 = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit()
print("=== Basic DiD Model ===")
print(model1.summary())

print("\n\n=== DiD with Branch Fixed Effects ===")
# Model with branch fixed effects (controls for time-invariant branch characteristics)
model2 = ols('service_score ~ after_switch + treatment_post + C(branch_id)', data=df).fit()
print(model2.summary())

print("\n\n=== DiD with Branch and Week Fixed Effects ===")
# Model with both branch and week fixed effects
model3 = ols('service_score ~ after_switch + treatment_post + C(branch_id) + C(week)', data=df).fit()
print(model3.summary())

print("\n\n=== DiD with Controls and Fixed Effects ===")
# Full model with staffing and visits as controls
model4 = ols('service_score ~ after_switch + treatment_post + staffing_index + visits_index + C(branch_id) + C(week)', data=df).fit()
print(model4.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.diagnostic import het_breuschpagan

df = pd.read_csv('branches.csv')
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Run the full model with cluster-robust standard errors
model = ols('service_score ~ after_switch + treatment_post + staffing_index + visits_index + C(branch_id) + C(week)', data=df).fit()

# Get cluster-robust standard errors
from statsmodels.stats.sandwich_covariance import cov_cluster
cluster_cov = cov_cluster(model, df['branch_id'])
cluster_se = np.sqrt(np.diag(cluster_cov))

print("=== Full Model with Cluster-Robust SE (clustered by branch) ===")
print(f"\nTreatment effect (treatment_post): {model.params['treatment_post']:.3f}")
print(f"Cluster-robust SE: {cluster_se[model.params.index.get_loc('treatment_post')]:.3f}")

# Calculate t-stat and p-value
treatment_idx = model.params.index.get_loc('treatment_post')
t_stat = model.params['treatment_post'] / cluster_se[treatment_idx]
from scipy import stats
p_value = 2 * (1 - stats.t.cdf(abs(t_stat), df=len(df) - len(model.params)))
print(f"t-statistic: {t_stat:.3f}")
print(f"p-value: {p_value:.4f}")

# 95% CI
ci_lower = model.params['treatment_post'] - 1.96 * cluster_se[treatment_idx]
ci_upper = model.params['treatment_post'] + 1.96 * cluster_se[treatment_idx]
print(f"95% CI: [{ci_lower:.3f}, {ci_upper:.3f}]")

print("\n\n=== All Key Coefficients ===")
for var in ['after_switch', 'treatment_post', 'staffing_index', 'visits_index']:
    idx = model.params.index.get_loc(var)
    se = cluster_se[idx]
    coef = model.params[var]
    t = coef / se
    p = 2 * (1 - stats.t.cdf(abs(t), df=len(df) - len(model.params)))
    print(f"{var}: coef={coef:.3f}, SE={se:.3f}, t={t:.3f}, p={p:.4f}")
