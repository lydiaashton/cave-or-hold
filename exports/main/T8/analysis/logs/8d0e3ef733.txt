import pandas as pd

import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary table: Average service scores by group and period
print("=" * 60)
print("AVERAGE SERVICE SCORES BY GROUP AND PERIOD")
print("=" * 60)

# Before switch (after_switch = 0)
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()

# After switch (after_switch = 1)
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nTreatment Group (switched branches):")
print(f"  Before switch: {before_treatment:.2f}")
print(f"  After switch:  {after_treatment:.2f}")
print(f"  Change:        {after_treatment - before_treatment:.2f}")

print(f"  Before switch: {before_control:.2f}")
print(f"  After switch:  {after_control:.2f}")
print(f"  Change:        {after_control - before_control:.2f}")

# Difference-in-differences
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

# Run DiD regression
print("\n" + "=" * 60)
print("DIFFERENCE-IN-DIFFERENCES REGRESSION")
print("=" * 60)

# Basic DiD model
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Event study with proper treatment variable
df_event = df.copy()
for w in range(6, 23):
    df_event[f'week_{w}'] = (df_event['week'] == w).astype(int)
    df_event[f'treat_week_{w}'] = df_event['switched_branch'] * df_event[f'week_{w}']

# Using week 14 as reference (last week before switch)
week_cols = [f'week_{w}' for w in range(6, 15)] + [f'week_{w}' for w in range(16, 23)]
treat_week_cols = [f'treat_week_{w}' for w in range(6, 15)] + [f'treat_week_{w}' for w in range(16, 23)]

formula = 'service_score ~ switched_branch + ' + ' + '.join(week_cols) + ' + ' + ' + '.join(treat_week_cols)
model_event = ols(formula, data=df_event).fit(cov_type='cluster', cov_kwds={'groups': df_event['branch_id']})

# Extract treatment effects by week
print("Treatment effects by week (relative to week 14, clustered SE):")
pre_switch_effects = []
post_switch_effects = []
for w in range(6, 15):
    coef = model_event.params[f'treat_week_{w}']
    se = model_event.bse[f'treat_week_{w}']
    pre_switch_effects.append((w, coef, se))
    print(f"  Week {w}: {coef:.2f} (SE: {se:.2f})")
print("  Week 14: 0.00 (reference)")
for w in range(16, 23):
    coef = model_event.params[f'treat_week_{w}']
    se = model_event.bse[f'treat_week_{w}']
    post_switch_effects.append((w, coef, se))
    print(f"  Week {w}: {coef:.2f} (SE: {se:.2f})")

# Plot event study
fig, ax = plt.subplots(figsize=(10, 6))
weeks_pre = [w for w, _, _ in pre_switch_effects]
coefs_pre = [coef for _, coef, _ in pre_switch_effects]
se_pre = [se for _, _, se in pre_switch_effects]

weeks_post = [w for w, _, _ in post_switch_effects]
coefs_post = [coef for _, coef, _ in post_switch_effects]
se_post = [se for _, _, se in post_switch_effects]

# Plot pre-switch
ax.errorbar(weeks_pre, coefs_pre, yerr=[1.96*s for s in se_pre], fmt='o', capsize=4, label='Before Switch', color='blue')
# Plot post-switch
ax.errorbar(weeks_post, coefs_post, yerr=[1.96*s for s in se_post], fmt='s', capsize=4, label='After Switch', color='green')
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point')
ax.set_xlabel('Week')
ax.set_ylabel('Treatment Effect (vs Week 14)')
ax.set_title('Event Study: Treatment Effect on Service Score by Week')
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study_plot.png', dpi=150)
print("\nEvent study plot saved as event_study_plot.png")

# Check if pre-trends are statistically different from zero
print("\n" + "=" * 60)
print("PRE-TREND TEST (weeks 6-13 vs week 14)")
print("=" * 60)

# Test if average pre-treatment effect is zero
pre_coefs = [coef for _, coef, _ in pre_switch_effects]
pre_ses = [se for _, _, se in pre_switch_effects]
# Simple test: are any pre-trends significantly different from 0?
significant_pre = [(w, coef, se) for (w, coef, se) in pre_switch_effects if abs(coef/se) > 1.96]
print(f"Pre-switch weeks with statistically significant effects (p<0.05): {len(significant_pre)}")
for w, coef, se in significant_pre:
    print(f"  Week {w}: {coef:.2f} (SE: {se:.2f}), t={coef/se:.2f}")

# Also check post-switch significance
print("\nPost-switch weeks with statistically significant effects (p<0.05):")
significant_post = [(w, coef, se) for (w, coef, se) in post_switch_effects if abs(coef/se) > 1.96]
print(f"  Count: {len(significant_post)}")
for w, coef, se in significant_post:
    print(f"  Week {w}: {coef:.2f} (SE: {se:.2f}), t={coef/se:.2f}")

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Final summary statistics
print("=" * 60)
print("FINAL SUMMARY STATISTICS")
print("=" * 60)

# Number of branches
total_branches = df['branch_id'].nunique()
switched_branches = df[df['switched_branch'] == 1]['branch_id'].nunique()
control_branches = df[df['switched_branch'] == 0]['branch_id'].nunique()

print(f"\nTotal branches: {total_branches}")
print(f"Switched branches (treatment): {switched_branches}")
print(f"Non-switched branches (control): {control_branches}")

# Service score summary
print(f"\nOverall service score:")
print(f"  Mean: {df['service_score'].mean():.2f}")
print(f"  Std Dev: {df['service_score'].std():.2f}")

# DiD estimate with cluster-robust SE
from statsmodels.formula.api import ols
df['treatment'] = df['switched_branch'] * df['after_switch']
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

did_coef = model.params['treatment']
did_se = model.bse['treatment']
did_pvalue = model.pvalues['treatment']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se

print(f"\nDifference-in-Differences Estimate:")
print(f"  Coefficient: {did_coef:.2f}")
print(f"  Cluster-robust SE: {did_se:.2f}")
print(f"  95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"  P-value: {did_pvalue:.4f}")

# Effect size relative to mean
mean_score = df['service_score'].mean()
effect_pct = (did_coef / mean_score) * 100
print(f"\nEffect size: {effect_pct:.1f}% of average service score")

# Practical significance
print(f"\nInterpretation:")
print(f"  The new procedure is associated with a {did_coef:.1f} point increase in service scores.")
print(f"  Given the average score is about {mean_score:.0f}, this represents approximately a {effect_pct:.0f}% improvement.")
