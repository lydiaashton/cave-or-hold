import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Create treatment and period indicators
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Classic 2x2 DiD table
print("\n=== Difference-in-Differences 2x2 Table ===")
print("\nMean service scores:")

# Treatment group (switched_branch=1)
treat_pre = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_post = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

# Control group (switched_branch=0)
ctrl_pre = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
ctrl_post = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print(f"Treatment (switched) branches:")
print(f"  Pre-switch (weeks 6-14): {treat_pre:.2f}")
print(f"  Post-switch (weeks 15-22): {treat_post:.2f}")
print(f"  Change: {treat_post - treat_pre:.2f}")

print(f"\nControl (non-switched) branches:")
print(f"  Pre-switch (weeks 6-14): {ctrl_pre:.2f}")
print(f"  Post-switch (weeks 15-22): {ctrl_post:.2f}")
print(f"  Change: {ctrl_post - ctrl_pre:.2f}")

did_estimate = (treat_post - treat_pre) - (ctrl_post - ctrl_pre)
print(f"\n=== DiD Estimate: {did_estimate:.2f} ===")
print("(Treatment change minus Control change)")

# Now run regression
print("\n=== Regression Analysis ===")

# Model 1: Simple DiD
model1 = ols('service_score ~ treatment + post + did', data=df).fit()
print("\nModel 1: Simple DiD (no controls)")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ treatment + post + did + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Model 3: With branch fixed effects
model3 = ols('service_score ~ treatment + post + did + staffing_index + visits_index + C(branch_id)', data=df).fit()
print("\nModel 3: DiD with branch fixed effects")
# Just show the key coefficients
print("\nKey coefficients:")
print(model3.params[['did', 'staffing_index', 'visits_index']])
print("\nStandard errors:")
print(model3.bse[['did', 'staffing_index', 'visits_index']])

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check parallel trends - plot average service scores by week for each group
print("=== Parallel Trends Check ===")
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(weekly_avg)

# Calculate pre-trend differences
print("\nPre-switch weekly averages (weeks 6-14):")
pre_data = df[df['week'] < 15]
pre_weekly = pre_data.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(pre_weekly)

# Difference between treatment and control each week pre-switch
pre_diff = pre_weekly[1] - pre_weekly[0]
print("\nTreatment - Control difference (pre-switch weeks):")
print(pre_diff)
print(f"\nAverage pre-trend difference: {pre_diff.mean():.2f}")
print(f"Pre-trend slope (regression on week): {np.polyfit(pre_diff.index, pre_diff.values, 1)[0]:.3f}")

# Now run fixed effects model properly with cluster-robust SE
print("\n=== Fixed Effects Model with Cluster-Robust SE ===")

# Create dummies for branch fixed effects
df_fe = df.copy()
branch_dummies = pd.get_dummies(df_fe['branch_id'], prefix='branch', drop_first=True)
df_fe = pd.concat([df_fe, branch_dummies], axis=1)

# Variables for regression
X = df_fe[['treatment', 'post', 'did', 'staffing_index', 'visits_index'] + [c for c in df_fe.columns if c.startswith('branch')]]
X = sm.add_constant(X)
y = df_fe['service_score']

model_fe = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df_fe['branch_id']})
print("\nKey coefficients with cluster-robust SE (clustered by branch):")
print(f"DID coefficient: {model_fe.params['did']:.3f}")
print(f"Standard error: {model_fe.bse['did']:.3f}")
print(f"t-statistic: {model_fe.tvalues['did']:.3f}")
print(f"p-value: {model_fe.pvalues['did']:.3f}")
print(f"95% CI: [{model_fe.conf_int().loc['did', 0]:.3f}, {model_fe.conf_int().loc['did', 1]:.3f}]")

# Also check effect size relative to baseline
baseline = df['service_score'].mean()
print(f"\nEffect size: {model_fe.params['did']:.2f} points")
print(f"Baseline average service score: {baseline:.2f}")
print(f"Effect as % of baseline: {100 * model_fe.params['did'] / baseline:.1f}%")

# Create visualization data
plt.figure(figsize=(10, 6))
weeks = sorted(df['week'].unique())
for switched in [0, 1]:
    subset = df[df['switched_branch'] == switched]
    weekly_means = subset.groupby('week')['service_score'].mean()
    label = 'Switched branches (treatment)' if switched == 1 else 'Non-switched branches (control)'
    plt.plot(weeks, weekly_means, marker='o', label=label, linewidth=2)

plt.axvline(x=14.5, color='red', linestyle='--', label='Switch implemented (week 15)')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Scores by Branch Type Over Time')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('service_scores_over_time.png', dpi=150)
print("\nPlot saved as 'service_scores_over_time.png'")
