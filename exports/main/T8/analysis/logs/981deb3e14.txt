import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Check treatment/control balance
print("Treatment vs Control branches:")
treatment_branches = df[df['switched_branch'] == 1]['branch_id'].nunique()
control_branches = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"  Treatment (switched): {treatment_branches} branches")
print(f"  Control (not switched): {control_branches} branches")
print(f"  Total: {treatment_branches + control_branches} branches")
print(f"  Treatment proportion: {treatment_branches/(treatment_branches + control_branches):.1%}")

# Check observations per group
print("\nObservations per group:")
print(df.groupby('switched_branch').size())

# Check weeks per group
print("\nWeeks coverage by group:")
print(df.groupby(['switched_branch', 'after_switch']).size())

# Average service scores by group and period
print("\nAverage service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the difference-in-differences manually
print("\n--- Simple Difference-in-Differences Calculation ---")
# Treatment before
t_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
# Treatment after
t_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
# Control before
c_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
# Control after
c_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"Treatment (switched) branches:")
print(f"  Before switch: {t_before:.2f}")
print(f"  After switch: {t_after:.2f}")
print(f"  Change: {t_after - t_before:.2f}")

print(f"  After switch: {c_after:.2f}")
print(f"  Change: {c_after - c_before:.2f}")

did_effect = (t_after - t_before) - (c_after - c_before)
print(f"\nDifference-in-Differences estimate: {did_effect:.2f}")
print(f"  (Treatment change) - (Control change) = {t_after - t_before:.2f} - {c_after - c_before:.2f} = {did_effect:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model 1: Simple DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Model 1: Simple Difference-in-Differences ===")
print(model1.summary())

# Model 2: DiD with control variables
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Model 2: DiD with Controls ===")
print(model2.summary())

# Model 3: With branch fixed effects
print("\n=== Model 3: DiD with Branch Fixed Effects ===")
# Create branch dummies
df_with_fe = df.copy()
model3 = ols('service_score ~ after_switch + treatment + staffing_index + visits_index + C(branch_id)', data=df_with_fe).fit(cov_type='cluster', cov_kwds={'groups': df_with_fe['branch_id']})
print(model3.summary())
