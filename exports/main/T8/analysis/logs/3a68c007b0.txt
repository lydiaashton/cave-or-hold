import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Summary table: mean service scores by group and period
print("=== Mean Service Scores by Group and Period ===\n")

# Treatment group (switched)
treatment_pre = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_post = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

# Control group (not switched)
control_pre = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_post = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"Treatment (switched) branches:")
print(f"  Pre-switch (weeks 6-14): {treatment_pre:.2f}")
print(f"  Post-switch (weeks 15-22): {treatment_post:.2f}")
print(f"  Change: {treatment_post - treatment_pre:.2f}")
print()
print(f"Control (not switched) branches:")
print(f"  Pre-switch (weeks 6-14): {control_pre:.2f}")
print(f"  Post-switch (weeks 15-22): {control_post:.2f}")
print(f"  Change: {control_post - control_pre:.2f}")
print()
print(f"=== Difference-in-Differences Estimate ===")
print(f"DiD = (Treatment Post - Treatment Pre) - (Control Post - Control Pre)")
did = (treatment_post - treatment_pre) - (control_post - control_pre)
print(f"DiD = {did:.4f}")

# Now run the regression
print("\n=== Regression Results ===")
model = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit()
print(model.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Check parallel trends: compare trends in pre-period
print("=== Parallel Trends Check (Pre-Period Only) ===\n")

pre_data = df[df['after_switch'] == 0].copy()

# Regress service_score on week, switched_branch, and their interaction (pre-period only)
# If parallel trends hold, the interaction should be insignificant
pre_model = ols('service_score ~ C(week) + switched_branch + switched_branch:C(week)', data=pre_data).fit()
print("Pre-period trend comparison:")
print(pre_model.summary().tables[1])

# Also check week-by-week means
print("\n=== Week-by-Week Mean Service Scores ===")
weekly_means = pre_data.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_means.columns = ['Control', 'Treatment']
weekly_means['Difference'] = weekly_means['Treatment'] - weekly_means['Control']
print(weekly_means)

# Now run DiD with branch fixed effects
print("\n=== DiD with Branch Fixed Effects ===")
# Convert branch_id to categorical for fixed effects
df['branch_id_cat'] = df['branch_id'].astype('category')

# This will be a large model, so let's use a different approach
# We'll demean the data within branches
df_demeaned = df.copy()
branch_means = df.groupby('branch_id')['service_score'].transform('mean')
df_demeaned['service_score_demeaned'] = df['service_score'] - branch_means

# Also demean the predictors
for col in ['switched_branch', 'after_switch', 'treatment_post']:
    col_means = df.groupby('branch_id')[col].transform('mean')
    df_demeaned[f'{col}_demeaned'] = df[col] - col_means

# Actually, switched_branch is constant within branch, so it gets absorbed
# Let's just run with after_switch and treatment_post, clustering SE by branch
fe_model = ols('service_score ~ C(branch_id) + after_switch + treatment_post', data=df).fit()
print(fe_model.summary().tables[1].iloc[-3:])  # Just the key coefficients

import pandas as pd
import numpy as np

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create event time relative to switch (week 15)
# Weeks 6-14 are pre (-9 to -1), weeks 15-22 are post (0 to 7)
df['event_time'] = df['week'] - 15

# Create event study dummies (omitting week -1 as reference)
for t in range(-9, 8):
    if t != -1:  # omit week -1 as reference
        df[f'event_{t}'] = ((df['event_time'] == t) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression with branch fixed effects
event_vars = [f'event_{t}' for t in range(-9, 8) if t != -1]
formula = 'service_score ~ C(branch_id) + ' + ' + '.join(event_vars)
event_model = ols(formula, data=df).fit()

# Get clustered SE
event_clustered = event_model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("=== Event Study Coefficients (relative to week -1) ===")
param_names = event_model.params.index.tolist()
results = []
for t in range(-9, 8):
    if t != -1:
        var_name = f'event_{t}'
        idx = param_names.index(var_name)
        coef = event_clustered.params[idx]
        se = event_clustered.bse[idx]
        pval = event_clustered.pvalues[idx]
        results.append((t, coef, se, pval))
        period = "Pre" if t < 0 else "Post"
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"Week {t:+d} ({period}): coef = {coef:.3f}, SE = {se:.3f}, p = {pval:.4f} {sig}")
    else:
        print(f"Week {t:+d} (Pre): [REFERENCE]")

# Save the event study results for plotting
event_results_df = pd.DataFrame(results, columns=['event_time', 'coef', 'se', 'pvalue'])
event_results_df['period'] = event_results_df['event_time'].apply(lambda x: 'Pre' if x < 0 else 'Post')
