import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study / dynamic DiD - check effects by week relative to switch
# Create relative week variable (week 15 = 0, week 14 = -1, etc.)
df['relative_week'] = df['week'] - 15

# Run event study regression with week fixed effects
# We'll interact treatment with each post-switch week
df['post_week'] = (df['week'] >= 15).astype(int)
df['treatment_post'] = df['switched_branch'] * df['post_week']

# More detailed: create dummies for each week and interact with treatment
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=False)
df_with_dummies = pd.concat([df, week_dummies], axis=1)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Clean DiD with week fixed effects (remove after_switch since it's collinear with week FEs)
model_week_fe = ols('service_score ~ switched_branch + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Model with Week FEs (no after_switch):")
print(model_week_fe.summary())

# Better: Run DiD without week FEs but get proper SE
model_basic = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
coef = model_basic.params['switched_branch:after_switch']
se = model_basic.bse['switched_branch:after_switch']
ci_lower = coef - 1.96 * se
ci_upper = coef + 1.96 * se
print(f"\n=== MAIN RESULT ===")

print(f"Standard error: {se:.2f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"p-value: {model_basic.pvalues['switched_branch:after_switch']:.4f}")

# Effect size
baseline_score = df[df['after_switch'] == 0]['service_score'].mean()
print(f"\nBaseline average score (pre-switch): {baseline_score:.2f}")
print(f"Effect as % of baseline: {coef/baseline_score*100:.1f}%")
