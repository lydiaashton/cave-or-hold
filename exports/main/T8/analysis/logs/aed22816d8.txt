import pandas as pd
import numpy as np
import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = β0 + β1*switched_branch + β2*after_switch + β3*(switched_branch * after_switch) + ε
# β3 is the DiD estimate

df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Simple DiD Regression:")
print(model.summary())

# Now with branch fixed effects and week fixed effects
# This controls for time-invariant branch characteristics and common time trends
model_fe = smf.ols('service_score ~ switched_branch * after_switch + C(branch_id) + C(week)', data=df).fit()
print("\n\nDiD Regression with Branch and Week Fixed Effects:")
print(model_fe.summary())

# Clustered standard errors by branch
# Since we can't easily do clustered SEs with statsmodels formula, let's do it manually
from statsmodels.regression.linear_model import OLS
from statsmodels.tools import add_constant

# Create dummy variables
df_fe = df.copy()
df_fe = pd.get_dummies(df_fe, columns=['branch_id', 'week'], drop_first=True)

X = df_fe[['switched_branch', 'after_switch', 'treatment'] + [c for c in df_fe.columns if c.startswith('branch_id_') or c.startswith('week_')]]
X = add_constant(X)
y = df_fe['service_score']

model_cluster = OLS(y, X).fit()
print("\n\nCoefficients from FE model:")
print(model_cluster.params[['switched_branch', 'after_switch', 'treatment']])
