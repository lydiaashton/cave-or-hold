import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Check how many branches switched vs didn't
print("=== Treatment/Control Group Sizes ===")
switched_counts = df.groupby('switched_branch')['branch_id'].nunique()
print(switched_counts)
print(f"\nTotal branches: {switched_counts.sum()}")
print(f"Switched: {switched_counts[1]} ({switched_counts[1]/switched_counts.sum()*100:.1f}%)")
print(f"Control: {switched_counts[0]} ({switched_counts[0]/switched_counts.sum()*100:.1f}%)")

# Check balance of pre-switch service scores
print("\n=== Pre-Switch Service Scores (Weeks 6-14) ===")
pre_data = df[df['after_switch'] == 0]
pre_by_group = pre_data.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print(pre_by_group)

# Check post-switch service scores
print("\n=== Post-Switch Service Scores (Weeks 15-22) ===")
post_data = df[df['after_switch'] == 1]
post_by_group = post_data.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print(post_by_group)

# Simple difference-in-differences calculation
print("\n=== Simple DiD Calculation ===")
# Treatment group: before and after
treat_before = pre_data[pre_data['switched_branch'] == 1]['service_score'].mean()
treat_after = post_data[post_data['switched_branch'] == 1]['service_score'].mean()
control_before = pre_data[pre_data['switched_branch'] == 0]['service_score'].mean()
control_after = post_data[post_data['switched_branch'] == 0]['service_score'].mean()

print(f"Treatment group - Before: {treat_before:.2f}, After: {treat_after:.2f}, Change: {treat_after - treat_before:.2f}")
print(f"Control group - Before: {control_before:.2f}, After: {control_after:.2f}, Change: {control_after - control_before:.2f}")
did_estimate = (treat_after - treat_before) - (control_after - control_before)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
print("=== Difference-in-Differences Regression ===")
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model.summary())

# DiD with branch fixed effects

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks relative to switch at week 15)
df['relative_week'] = df['week'] - 15

# Create event study dummies (omitting week -1 as reference)
for w in range(-9, 8):  # weeks -9 to -1 (pre), 0 to 7 (post)
    if w != -1:  # omit reference category
        df[f'week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Event study regression
event_study_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = f"service_score ~ switched_branch + after_switch + {' + '.join(event_study_vars)}"

model_es = ols(formula, data=df).fit()
cluster_robust_es = model_es.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("=== Event Study Regression (Pre-Trends Test) ===")
print(f"Reference period: Week -1 (week 14)")
print(f"\nPre-treatment coefficients (should be close to 0 if parallel trends hold):")
for w in range(-9, 0):
    if w != -1:
        coef = cluster_robust_es.params[f'week_{w}']
        se = cluster_robust_es.bse[f'week_{w}']
        pval = cluster_robust_es.pvalues[f'week_{w}']
        ci_low = cluster_robust_es.conf_int().loc[f'week_{w}'][0]
        ci_high = cluster_robust_es.conf_int().loc[f'week_{w}'][1]
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"Week {w}: coef={coef:.2f}, SE={se:.2f}, 95% CI=[{ci_low:.2f}, {ci_high:.2f}] {sig}")

print(f"\nPost-treatment coefficients (treatment effects):")

# Omitting week -1 as reference
for w in range(-9, 8):
    if w != -1:
        if w < 0:
            col_name = f'pre_{abs(w)}'  # pre_9, pre_8, ..., pre_2
        else:
            col_name = f'post_{w}'  # post_0, post_1, ..., post_7

    pval = cluster_robust_es.pvalues[var]
