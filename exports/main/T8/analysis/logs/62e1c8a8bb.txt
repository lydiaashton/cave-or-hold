import pandas as pd
import numpy as np
import statsmodels.api as sm
import statsmodels.formula.api as smf
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch'] * df['after_switch']

# Summary table: average service scores by group and period
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES SUMMARY TABLE")
print("=" * 60)

# Before period (weeks 6-14)
before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

# Control group (switched_branch=0)
control_before = before[before['switched_branch'] == 0]['service_score'].mean()
control_after = after[after['switched_branch'] == 0]['service_score'].mean()

# Treatment group (switched_branch=1)
treat_before = before[before['switched_branch'] == 1]['service_score'].mean()
treat_after = after[after['switched_branch'] == 1]['service_score'].mean()

print(f"\nControl Group (did not switch):")
print(f"  Before (weeks 6-14): {control_before:.2f}")
print(f"  After (weeks 15-22): {control_after:.2f}")
print(f"  Change: {control_after - control_before:.2f}")

print(f"\nTreatment Group (switched):")
print(f"  Before (weeks 6-14): {treat_before:.2f}")
print(f"  After (weeks 15-22): {treat_after:.2f}")
print(f"  Change: {treat_after - treat_before:.2f}")

# DiD estimate
did_estimate = (treat_after - treat_before) - (control_after - control_before)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

# Run DiD regression
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD model
model1 = smf.ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
print("\nBasic DiD Model (no controls):")
print(model1.summary().tables[1])

# DiD with controls
model2 = smf.ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', data=df).fit()
print("\nDiD Model with controls (staffing_index, visits_index):")
print(model2.summary().tables[1])

# Cluster-robust standard errors by branch
model3 = smf.ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nDiD Model with cluster-robust SE (by branch):")
print(model3.summary().tables[1])

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Test parallel trends: interact switched_branch with week dummies for pre-period only
# This tests whether the trends were different before the treatment
pre_df = df[df['week'] < 15].copy()

# Create week dummies (relative to week 6 as baseline)
for w in range(7, 15):
    pre_df[f'week_{w}'] = (pre_df['week'] == w).astype(int)

# Test if treatment group had different weekly trends before the switch
# We'll use week 6 as baseline
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'week_{w} * switched_branch' for w in range(7, 15)])
model_trends = smf.ols(formula, data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})

print("Parallel Trends Test (pre-period interaction terms):")
print("=" * 60)
# Show just the interaction terms
params = model_trends.params
pvalues = model_trends.pvalues

print("\nInteraction terms (week X switched_branch) - should be insignificant if parallel trends hold:")
for w in range(7, 15):
    coef = params[f'switched_branch:week_{w}']
    pval = pvalues[f'switched_branch:week_{w}']
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Week {w}: coef={coef:.3f}, p={pval:.3f} {sig}")

# Also check overall F-test for all interaction terms
print("\nF-test for all pre-trend interactions:")
f_test = model_trends.f_test([f'switched_branch:week_{w}' for w in range(7, 15)])
print(f"  F-statistic: {f_test.statistic[0][0]:.2f}, p-value: {f_test.pvalue:.3f}")

print("\n" + "=" * 60)
