print("Average service scores by group and period:")
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='period', 
                       aggfunc='mean')
pivot.index = ['Control (old)', 'Treatment (new)']
print(pivot)

# Calculate the differences
print("\n\nDifference-in-Differences calculation:")
control_before = df[(df['switched_branch'] == 0) & (df['week'] < 15)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['week'] >= 15)]['service_score'].mean()
treatment_before = df[(df['switched_branch'] == 1) & (df['week'] < 15)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['week'] >= 15)]['service_score'].mean()

print(f"Treatment group: Before={treatment_before:.2f}, After={treatment_after:.2f}, Change={treatment_after - treatment_before:.2f}")

did_effect = (treatment_after - treatment_before) - (control_after - control_before)
print(f"\nDifference-in-Differences estimate: {did_effect:.2f}")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks since switch, with week 14 as reference)
# Week 14 is the last week before switch, so we'll use it as baseline
df['relative_week'] = df['week'] - 14  # week 14 becomes 0, week 15 becomes 1, etc.

# Create dummies for each week relative to switch, excluding week 0 (baseline)
# We'll interact these with switched_branch
weeks = sorted(df['relative_week'].unique())
print("Relative weeks:", weeks)

# Create event study model
# We need to create dummies for each relative week (except baseline week 0)
# and interact with switched_branch

# For simplicity, let's create the dummies
for w in weeks:
    if w != 0:  # Skip baseline
        df[f'week_{w}'] = (df['relative_week'] == w).astype(int)

# Build formula for event study
# Include main effects for switched_branch and each week dummy, plus interactions
week_cols = [f'week_{w}' for w in weeks if w != 0]
interaction_cols = [f'switched_branch:week_{w}' for w in weeks if w != 0]

# Main model: service_score ~ switched_branch + week_dummies + switched_branch*week_dummies
# We include switched_branch main effect, week dummies, and their interactions
formula_parts = ['switched_branch'] + week_cols + [f'switched_branch * week_{w}' for w in weeks if w != 0]
# Actually, let's simplify - we want the interaction terms
formula = 'service_score ~ switched_branch + ' + ' + '.join(week_cols) + ' + ' + ' + '.join([f'switched_branch:week_{w}' for w in weeks if w != 0])

event_study = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract the interaction coefficients (the treatment effects for each week)
print("\nEvent Study Results (treatment effect by week relative to switch):")
print("="*80)

results = []
for w in weeks:
    if w == 0:
        results.append({'relative_week': w, 'coef': 0, 'se': 0, 'pvalue': np.nan, 'label': 'Baseline (week 14)'})
    else:
        coef_name = f'switched_branch:week_{w}'
        if coef_name in event_study.params:
            coef = event_study.params[coef_name]
            se = event_study.bse[coef_name]
            pval = event_study.pvalues[coef_name]
            results.append({'relative_week': w, 'coef': coef, 'se': se, 'pvalue': pval, 'label': f'Week {14+w}'})

results_df = pd.DataFrame(results)
print(results_df.to_string(index=False))

# Plot the event study
plt.figure(figsize=(12, 6))
plt.errorbar(results_df['relative_week'], results_df['coef'], yerr=1.96*results_df['se'], 
             fmt='o', capsize=5, markersize=8, linewidth=2)
plt.axhline(y=0, color='black', linestyle='-', linewidth=1)
plt.axvline(x=0.5, color='red', linestyle='--', linewidth=2, label='Switch occurs')
plt.xlabel('Weeks Relative to Switch (0 = week 14, baseline)')
plt.ylabel('Treatment Effect on Service Score')
plt.title('Event Study: Treatment Effect by Week')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as event_study.png")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run the main DiD model again with cluster-robust SEs
model = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Main DiD Results (clustered SEs):")
print(f"  Treatment effect (switched_branch:after_switch): {model.params['switched_branch:after_switch']:.2f}")
print(f"  Standard error: {model.bse['switched_branch:after_switch']:.2f}")
print(f"  p-value: {model.pvalues['switched_branch:after_switch']:.3f}")
print(f"  95% CI: [{model.conf_int().loc['switched_branch:after_switch'][0]:.2f}, {model.conf_int().loc['switched_branch:after_switch'][1]:.2f}]")

# Calculate effect size relative to baseline
baseline = df[df['week'] < 15]['service_score'].mean()
effect_size_pct = (model.params['switched_branch:after_switch'] / baseline) * 100
print(f"\n  Effect as % of baseline: {effect_size_pct:.1f}%")

# Power analysis - what sample size would we need for significance?
# Current effect / SE = t-statistic

print(f"\n  Current t-statistic: {t_stat:.2f}")

print(f"  Would need approximately {(1.96/t_stat)**2:.1f}x more data for significance at this effect size")
