import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run DiD regression with branch fixed effects
# Create branch dummies for fixed effects
df = df.copy()

# Basic DiD regression
model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("=== Model 1: Basic DiD (no fixed effects) ===")
print(model1.summary().tables[1])

# DiD with branch fixed effects
df['branch_id_str'] = df['branch_id'].astype(str)
model2 = ols('service_score ~ switched_branch * after_switch + C(branch_id_str)', data=df).fit()
print("\n=== Model 2: DiD with Branch Fixed Effects ===")
print(model2.summary().tables[1])

# DiD with branch FE and week FE
model3 = ols('service_score ~ switched_branch * after_switch + C(branch_id_str) + C(week)', data=df).fit()
print("\n=== Model 3: DiD with Branch + Week Fixed Effects ===")
print(model3.summary().tables[1])

# DiD with controls
model4 = ols('service_score ~ switched_branch * after_switch + C(branch_id_str) + C(week) + staffing_index + visits_index', data=df).fit()
