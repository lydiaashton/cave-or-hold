import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run a basic DiD regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimator - the causal effect of the treatment

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Basic Difference-in-Differences Regression:")
print(model.summary())

# Also run with cluster-robust standard errors at the branch level
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with Clustered Standard Errors (by branch):")
print(model_clustered.summary())

from statsmodels.formula.api import ols

matplotlib.use('Agg')

# Load the data
df = pd.read_csv('branches.csv')

# Create relative week variable (weeks since switch, with week 14 as reference)
df['relative_week'] = df['week'] - 14  # week 14 is the last pre-switch week

# Event study: create dummies for each week interacted with treatment
# We'll use week 14 (relative_week = 0) as the reference
weeks = df['relative_week'].unique()
print("Relative weeks:", sorted(weeks))

# Create the event study regression
# We need to be careful about the reference category
event_study_formula = 'service_score ~ switched_branch + ' + ' + '.join([f'switched_branch * C(week)' for week in range(6, 23) if week != 14]) + ' + C(week)'

# Actually, let's do this more carefully with a cleaner approach
df['week_cat'] = df['week'].astype('category')

# Event study with week 14 as reference
event_vars = []
for w in range(6, 23):
    if w != 14:
        event_vars.append(f'switched_branch * (week == {w})')

formula = 'service_score ~ switched_branch + ' + ' + '.join([f'C(week)' for w in range(6, 23) if w != 14]) + ' + ' + ' + '.join([f'switched_branch * (week == {w})' for w in range(6, 23) if w != 14])

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract the interaction coefficients (treatment effects by week)
coef_dict = {}
for w in range(6, 23):
    if w == 14:
        coef_dict[w] = 0  # reference category
    else:
        coef_name = f'switched_branch * (week == {w})'
        if coef_name in model_event.params.index:
            coef_dict[w] = model_event.params[coef_name]

print("\nEvent study coefficients (treatment effect by week):")
for w in sorted(coef_dict.keys()):
    print(f"Week {w}: {coef_dict[w]:.2f}")

# Plot the event study

import pandas as pd
import numpy as np

# Load the data
df = pd.read_csv('branches.csv')

# Calculate the weekly gap between treatment and control
weekly = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
weekly['gap'] = weekly[1] - weekly[0]

# Pre-period trend (weeks 6-14, that's 9 weeks, 8 intervals)
pre_weeks = list(range(6, 15))
pre_gaps = weekly.loc[pre_weeks, 'gap']
pre_trend = (pre_gaps.iloc[-1] - pre_gaps.iloc[0]) / (len(pre_weeks) - 1)

# Post-period trend (weeks 15-22, that's 8 weeks, 7 intervals)
post_weeks = list(range(15, 23))
post_gaps = weekly.loc[post_weeks, 'gap']
post_trend = (post_gaps.iloc[-1] - post_gaps.iloc[0]) / (len(post_weeks) - 1)

print("=== CRITICAL RE-ANALYSIS ===\n")
print(f"Gap in week 6 (first pre): {weekly.loc[6, 'gap']:.2f} points")
print(f"Gap in week 14 (last pre): {weekly.loc[14, 'gap']:.2f} points")
print(f"Gap in week 15 (first post): {weekly.loc[15, 'gap']:.2f} points")
print(f"Gap in week 22 (last post): {weekly.loc[22, 'gap']:.2f} points")
print()
print(f"Pre-period trend: gap widened by {pre_trend:.3f} points per week")
print(f"Post-period trend: gap widened by {post_trend:.3f} points per week")
print()
print(f"Total gap growth in pre-period (8 weeks): {pre_gaps.iloc[-1] - pre_gaps.iloc[0]:.2f} points")
print(f"Total gap growth in post-period (7 weeks): {post_gaps.iloc[-1] - post_gaps.iloc[0]:.2f} points")
print()

# What would we expect post-switch if the pre-trend continued?
expected_week22_gap = weekly.loc[6, 'gap'] + pre_trend * 16  # 16 weeks from week 6 to 22
actual_week22_gap = weekly.loc[22, 'gap']
print(f"Expected gap at week 22 if pre-trend continued: {expected_week22_gap:.2f}")
print(f"Actual gap at week 22: {actual_week22_gap:.2f}")
print(f"Difference (potential treatment effect): {actual_week22_gap - expected_week22_gap:.2f}")
print()

# Alternative: compare the average gap change
pre_avg_gap = pre_gaps.mean()
post_avg_gap = post_gaps.mean()
simple_did = post_avg_gap - pre_avg_gap
print(f"Simple DiD (post avg gap - pre avg gap): {simple_did:.2f} points")
print()

# But if pre-trend was 0.29/week, over 8 weeks that's 2.32 points of "artificial" DiD
print(f"Pre-trend bias concern: {pre_trend:.2f} points/week × 8 weeks = ~{pre_trend * 8:.2f} points")
print(f"This could explain most or all of the observed {simple_did:.2f} point effect")
