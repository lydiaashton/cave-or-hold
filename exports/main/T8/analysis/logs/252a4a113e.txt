import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch and week fixed effects
print("=== Difference-in-Differences Regression ===")
print("\nModel 1: Simple DiD (no fixed effects)")
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model1.summary().tables[1])

print("\n\nModel 2: With Branch Fixed Effects")
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df, branch_dummies], axis=1)
fe_cols = [c for c in df_with_fe.columns if c.startswith('branch_')]
formula2 = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(fe_cols)
model2 = ols(formula2, data=df_with_fe).fit()
print(f"R-squared: {model2.rsquared:.4f}")
print(f"\nKey coefficients:")
print(model2.params[['switched_branch', 'after_switch', 'treatment']].round(4))
print(f"\nTreatment effect (DiD): {model2.params['treatment']:.3f}")
print(f"95% CI: [{model2.conf_int().loc['treatment', 0]:.3f}, {model2.conf_int().loc['treatment', 1]:.3f}]")
print(f"P-value: {model2.pvalues['treatment']:.6f}")

print("\n\nModel 3: With Branch and Week Fixed Effects")
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df_full_fe = pd.concat([df.reset_index(drop=True), branch_dummies.reset_index(drop=True), week_dummies.reset_index(drop=True)], axis=1)
week_cols = [c for c in df_full_fe.columns if c.startswith('week_')]
formula3 = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(fe_cols + week_cols)
model3 = ols(formula3, data=df_full_fe).fit()
print(f"R-squared: {model3.rsquared:.4f}")
print(f"\nTreatment effect (DiD): {model3.params['treatment']:.3f}")
print(f"95% CI: [{model3.conf_int().loc['treatment', 0]:.3f}, {model3.conf_int().loc['treatment', 1]:.3f}]")
print(f"P-value: {model3.pvalues['treatment']:.6f}")

# Cluster-robust standard errors at branch level
print("\n\nModel 4: With Cluster-Robust SE (clustered by branch)")
model4 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
cov_cluster = model4.get_robustcov_results(cov_type='cluster', clusters=df['branch_id'])
print(f"Treatment effect: {cov_cluster.params['treatment']:.3f}")
print(f"Cluster-robust SE: {cov_cluster.bse['treatment']:.3f}")
print(f"95% CI: [{cov_cluster.conf_int().loc['treatment', 0]:.3f}, {cov_cluster.conf_int().loc['treatment', 1]:.3f}]")
print(f"P-value: {cov_cluster.pvalues['treatment']:.6f}")
