import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks since switch)
# Week 15 is the first post-switch week, so relative_week = week - 14
df['relative_week'] = df['week'] - 14

# Create event study dummies (omit week -1 as reference, which is week 13)
# Actually, let's use week 14 (relative_week = 0) as the reference since it's the last pre-period week
# Or better: use the first pre-period week as reference

# Let's create dummies for each week, using week 14 as reference (last pre-period week)
for w in range(6, 23):
    if w != 14:  # omit week 14 as reference
        df[f'week_{w}'] = (df['week'] == w).astype(int)

# Event study regression with interactions
# We'll interact each week dummy with switched_branch
event_study_vars = [f'week_{w}' for w in range(6, 23) if w != 14]
interaction_terms = [f'switched_branch:week_{w}' for w in range(6, 23) if w != 14]

formula = 'service_score ~ switched_branch + ' + ' + '.join(event_study_vars) + ' + ' + ' + '.join(interaction_terms)

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract the interaction coefficients (these are the event study estimates)
print("=== Event Study Results (coefficients for switched_branch * week interactions) ===")
print("(Reference period: week 14, the last week before switch)")
print("\nWeek | Coef | Std Err | 95% CI")
print("-" * 50)

results = []
for w in range(6, 23):
    if w == 14:
        results.append((w, 0.0, 0.0, 0.0, 0.0, "REF"))
        continue
    coef_name = f'switched_branch:week_{w}'
    if coef_name in model_event.params.index:
        coef = model_event.params[coef_name]
        se = model_event.bse[coef_name]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        results.append((w, coef, se, ci_low, ci_high, ""))
        marker = "***" if abs(coef/se) > 2.58 else "**" if abs(coef/se) > 1.96 else "*" if abs(coef/se) > 1.64 else ""
        print(f"{w:4} | {coef:6.2f} | {se:7.3f} | [{ci_low:6.2f}, {ci_high:6.2f}] {marker}")

# Save event study results for plotting
event_df = pd.DataFrame(results, columns=['week', 'coef', 'se', 'ci_low', 'ci_high', 'sig'])
event_df.to_csv('event_study_results.csv', index=False)
print("\nEvent study results saved to event_study_results.csv")
