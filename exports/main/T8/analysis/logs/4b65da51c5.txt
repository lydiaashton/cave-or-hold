# Run DiD regression
# Model 1: Basic DiD without controls

# Model 2: DiD with cluster-robust standard errors (clustered by branch)
model1_cluster = smf.ols('service_score ~ switched_branch + after_switch + switched_branch:after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Model 3: DiD with control variables
model2 = smf.ols('service_score ~ switched_branch + after_switch + switched_branch:after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
