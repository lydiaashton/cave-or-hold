import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create treatment interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary: average service scores by group and period
print("=== Average Service Scores ===")
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Calculate the differences
print("\n=== Difference-in-Differences Calculation ===")
# Control group (switched_branch=0)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_diff = control_after - control_before

# Treatment group (switched_branch=1)
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_diff = treat_after - treat_before

# DiD estimate
did_estimate = treat_diff - control_diff

print(f"Control group (no switch): Before={control_before:.2f}, After={control_after:.2f}, Change={control_diff:.2f}")
print(f"Treatment group (switched): Before={treat_before:.2f}, After={treat_after:.2f}, Change={treat_diff:.2f}")
print(f"\nDifference-in-Differences estimate: {did_estimate:.3f}")

# Now run regression
print("\n=== Regression Analysis ===")

# Basic DiD model
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Model 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# Model with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with controls")
print(model2.summary().tables[1])

# Cluster-robust standard errors by branch
model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 3: DiD with controls (cluster-robust SE by branch)")
print(model3.summary().tables[1])
