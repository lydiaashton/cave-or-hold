import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch'] * df['after_switch']

# === Difference-in-Differences Summary Table ===
print("=== Service Score by Group and Period ===")
pivot = df.pivot_table(
    values='service_score',
    index='switched_branch',
    columns='after_switch',
    aggfunc='mean'
)
pivot.columns = ['Before (Weeks 6-14)', 'After (Weeks 15-22)']
pivot.index = ['Control (Old Procedure)', 'Treatment (New Procedure)']
print(pivot.round(2))

# Calculate DiD estimate manually
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

control_change = control_after - control_before
treatment_change = treatment_after - treatment_before
did_estimate = treatment_change - control_change

print(f"\n=== Manual DiD Calculation ===")
print(f"Control group: {control_before:.2f} -> {control_after:.2f} (change: {control_change:.2f})")
print(f"Treatment group: {treatment_before:.2f} -> {treatment_after:.2f} (change: {treatment_change:.2f})")
print(f"DiD Estimate (Treatment Effect): {did_estimate:.2f}")

# === Regression Analysis ===
print("\n=== Regression Analysis (DiD) ===")

# Model 1: Basic DiD
model1 = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
print("\nModel 1: Basic DiD")
print(model1.summary().tables[1])

# Model 2: With controls (staffing and visits)
model2 = ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with Controls")
print(model2.summary().tables[1])

# Model 3: With branch fixed effects
model3 = ols('service_score ~ after_switch + treated + staffing_index + visits_index + C(branch_id)', data=df).fit()
print("\nModel 3: DiD with Branch Fixed Effects")
# Just show the key coefficients
coefs = model3.params[['after_switch', 'treated']]
print(f"\nKey coefficients:")
print(coefs)

# Get standard errors and p-values for treated effect
print(f"\nTreated effect: {model3.params['treated']:.3f}, SE: {model3.bse['treated']:.3f}, p-value: {model3.pvalues['treated']:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# === Look at the distribution of effects ===
# Calculate branch-level before/after changes
branch_stats = df.groupby(['branch_id', 'switched_branch']).agg({
    'service_score': ['mean', 'std', 'count']
}).reset_index()
branch_stats.columns = ['branch_id', 'switched_branch', 'mean_score', 'std_score', 'n_obs']

# Calculate before and after for each branch
before = df[df['after_switch']==0].groupby('branch_id')['service_score'].mean().reset_index()
before.columns = ['branch_id', 'before_score']

after = df[df['after_switch']==1].groupby('branch_id')['service_score'].mean().reset_index()
after.columns = ['branch_id', 'after_score']

branch_changes = before.merge(after, on='branch_id')
branch_changes['change'] = branch_changes['after_score'] - branch_changes['before_score']
branch_changes = branch_changes.merge(df[['branch_id', 'switched_branch']].drop_duplicates(), on='branch_id')

print("=== Branch-Level Changes ===")
print(f"\nControl branches (n={len(branch_changes[branch_changes['switched_branch']==0])}):")
control_changes = branch_changes[branch_changes['switched_branch']==0]['change']
print(f"  Mean change: {control_changes.mean():.2f}")
print(f"  Std dev: {control_changes.std():.2f}")
print(f"  Min: {control_changes.min():.2f}, Max: {control_changes.max():.2f}")

print(f"\nTreatment branches (n={len(branch_changes[branch_changes['switched_branch']==1])}):")
treatment_changes = branch_changes[branch_changes['switched_branch']==1]['change']
print(f"  Mean change: {treatment_changes.mean():.2f}")
print(f"  Std dev: {treatment_changes.std():.2f}")
print(f"  Min: {treatment_changes.min():.2f}, Max: {treatment_changes.max():.2f}")

print(f"\nDifference in changes (DiD): {treatment_changes.mean() - control_changes.mean():.2f}")

# === Event study: effect by week after switch ===
df['weeks_since_switch'] = df['week'] - 15
df['weeks_since_switch'] = df['weeks_since_switch'].clip(-9, 7)  # cap at bounds

# Create week dummies interacted with treatment
event_study = df.copy()
event_dummies = pd.get_dummies(event_study['weeks_since_switch'], prefix='week_rel')

                   data=df).fit(cov_type='HC3')

one_sided_pval = 1 - stats.norm.cdf(effect/se)
