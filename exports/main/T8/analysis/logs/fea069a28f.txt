    label = 'Treatment (Switched)' if group == 1 else 'Control (Not Switched)'

pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()

# Test if trends are parallel before treatment

print("\nPre-period interaction coefficient (tests parallel trends):")
print(f"Coefficient: {pre_model.params['switched_branch:week_centered']:.4f}")
print(f"P-value: {pre_model.pvalues['switched_branch:week_centered']:.4f}")

# Calculate effect sizes
print("\n=== Effect Size Summary ===")
control_pre = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()

import pandas as pd

df = pd.read_csv('branches.csv')

# Additional robustness checks

# 1. Check if results hold without controls
print("=== Robustness Check 1: Without Controls ===")
model_no_controls = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"DiD Coefficient: {model_no_controls.params['switched_branch:after_switch']:.2f}")
print(f"P-value: {model_no_controls.pvalues['switched_branch:after_switch']:.3f}")
print(f"95% CI: [{model_no_controls.conf_int().loc['switched_branch:after_switch'][0]:.2f}, {model_no_controls.conf_int().loc['switched_branch:after_switch'][1]:.2f}]")

# 2. Check individual branch fixed effects model
print("\n=== Robustness Check 2: With Branch Fixed Effects ===")
# Add branch fixed effects
df_with_fe = df.copy()
branch_dummies = pd.get_dummies(df_with_fe['branch_id'], prefix='branch', drop_first=True)
df_fe = pd.concat([df_with_fe, branch_dummies], axis=1)

# Build formula with branch FEs
fe_cols = [c for c in df_fe.columns if c.startswith('branch')]
formula = f"service_score ~ switched_branch * after_switch + staffing_index + visits_index + {' + '.join(fe_cols)}"
model_fe = ols(formula, data=df_fe).fit(cov_type='cluster', cov_kwds={'groups': df_fe['branch_id']})
print(f"DiD Coefficient: {model_fe.params['switched_branch:after_switch']:.2f}")
print(f"P-value: {model_fe.pvalues['switched_branch:after_switch']:.3f}")

# 3. Check event study - effect by week
