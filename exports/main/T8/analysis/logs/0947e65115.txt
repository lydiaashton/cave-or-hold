import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# === GROUP SUMMARY ===
print("=== BRANCH GROUPS ===")
branch_summary = df.groupby('branch_id')['switched_branch'].first().value_counts()
print(f"Branches that switched: {branch_summary.get(1, 0)}")
print(f"Branches that didn't switch: {branch_summary.get(0, 0)}")

# === SERVICE SCORE BY GROUP AND PERIOD ===
print("\n=== SERVICE SCORE BY GROUP AND PERIOD ===")
df['group_period'] = df['switched_branch'].astype(str) + '_' + df['after_switch'].astype(str)
score_summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(score_summary)

# Calculate the difference-in-differences manually
print("\n=== MANUAL DIFFERENCE-IN-DIFFERENCES ===")
# Control group (switched_branch=0)
control_pre = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_post = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_diff = control_post - control_pre

# Treatment group (switched_branch=1)
treat_pre = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_post = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_diff = treat_post - treat_pre

did_effect = treat_diff - control_diff

print(f"Control group (non-switched):")
print(f"  Pre-switch avg: {control_pre:.3f}")
print(f"  Post-switch avg: {control_post:.3f}")
print(f"  Change: {control_diff:.3f}")
print(f"\nTreatment group (switched):")
print(f"  Pre-switch avg: {treat_pre:.3f}")
print(f"  Post-switch avg: {treat_post:.3f}")
print(f"  Change: {treat_diff:.3f}")
print(f"\nDifference-in-Differences estimate: {did_effect:.3f}")

# === WEEKLY TRENDS ===
print("\n=== WEEKLY SERVICE SCORE TRENDS ===")
weekly_trends = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
print(weekly_trends.pivot(index='week', columns='switched_branch', values='service_score').round(2))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# === REGRESSION ANALYSIS ===
print("=== REGRESSION ANALYSIS ===")

# Model 1: Basic DiD (no fixed effects)
model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("\n--- Model 1: Basic DiD ---")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit()
print("\n--- Model 2: DiD with Controls ---")
print(model2.summary().tables[1])

# Model 3: DiD with branch fixed effects
# Create branch dummies
df_with_fe = df.copy()
branch_dummies = pd.get_dummies(df_with_fe['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df_with_fe, branch_dummies], axis=1)

# Build formula for fixed effects model
fe_cols = [c for c in df_with_fe.columns if c.startswith('branch_')]
formula = 'service_score ~ switched_branch * after_switch + staffing_index + visits_index + ' + ' + '.join(fe_cols)
model3 = ols(formula, data=df_with_fe).fit()
print("\n--- Model 3: DiD with Branch Fixed Effects ---")
print(f"R-squared: {model3.rsquared:.4f}")
print(f"DiD coefficient (switched_branch:after_switch): {model3.params['switched_branch:after_switch']:.4f}")
print(f"DiD std error: {model3.bse['switched_branch:after_switch']:.4f}")
print(f"DiD p-value: {model3.pvalues['switched_branch:after_switch']:.6f}")
print(f"95% CI: [{model3.conf_int().loc['switched_branch:after_switch', 0]:.4f}, {model3.conf_int().loc['switched_branch:after_switch', 1]:.4f}]")

# === PARALLEL TRENDS CHECK ===
print("\n=== PARALLEL TRENDS CHECK (Pre-period only) ===")
pre_period = df[df['after_switch'] == 0].copy()
pre_period['week_centered'] = pre_period['week'] - pre_period['week'].mean()

# Check if trends were similar before the switch
pre_trend_model = ols('service_score ~ switched_branch * week_centered', data=pre_period).fit()
print("\nPre-period trend interaction:")
print(f"  switched_branch:week_centered coefficient: {pre_trend_model.params['switched_branch:week_centered']:.4f}")
print(f"  p-value: {pre_trend_model.pvalues['switched_branch:week_centered']:.4f}")

# Also check week-by-week differences in pre-period
print("\nPre-period weekly averages:")
pre_weekly = pre_period.groupby(['switched_branch', 'week'])['service_score'].mean().unstack()
pre_weekly['diff'] = pre_weekly[1] - pre_weekly[0]
print(pre_weekly)
