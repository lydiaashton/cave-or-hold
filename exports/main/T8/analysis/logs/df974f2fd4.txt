import pandas as pd
import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch*after_switch) + error
# beta3 is the DiD estimate - the effect of the new procedure

model = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

print(post_avg)

# Calculate the gap each week post-switch
print("\n\nPost-switch gap (treatment - control):")
for week in range(15, 23):
    control = df[(df['switched_branch'] == 0) & (df['week'] == week)]['service_score'].mean()
    treatment = df[(df['switched_branch'] == 1) & (df['week'] == week)]['service_score'].mean()

import pandas as pd
import matplotlib.pyplot as plt
import numpy as np

df = pd.read_csv('branches.csv')

# Calculate weekly averages by group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Check pre-trends more carefully
print("=== PRE-TREND ANALYSIS (Weeks 6-14) ===\n")

pre_switch = df[df['week'] < 15]

# Weekly averages
control_weekly = pre_switch[pre_switch['switched_branch'] == 0].groupby('week')['service_score'].mean()
treatment_weekly = pre_switch[pre_switch['switched_branch'] == 1].groupby('week')['service_score'].mean()

print("Weekly averages before switch:")
print("\nWeek | Control | Treatment | Gap (T-C)")
print("-" * 40)
for week in range(6, 15):
    c = control_weekly.get(week, np.nan)
    t = treatment_weekly.get(week, np.nan)
    gap = t - c
    print(f"{week:5} | {c:7.2f} | {t:9.2f} | {gap:7.2f}")

# Linear trend in the gap before switch
print("\n\n=== Trend in Pre-Switch Gap ===")
gap_data = []
for week in range(6, 15):
    c = control_weekly.get(week, np.nan)
    t = treatment_weekly.get(week, np.nan)
    gap_data.append({'week': week, 'gap': t - c})

gap_df = pd.DataFrame(gap_data)
# Fit a linear trend
from scipy import stats
slope, intercept, r_value, p_value, std_err = stats.linregress(gap_df['week'], gap_df['gap'])
print(f"Pre-switch gap trend: {slope:.3f} points per week (p={p_value:.4f})")
print(f"This means the gap was GROWING by {slope*9:.2f} points over the 9-week pre-period")

# What would we expect post-switch if pre-trend continued?
expected_gap_week22 = intercept + slope * 22
actual_gap_week22 = df[(df['week'] == 22) & (df['switched_branch'] == 1)]['service_score'].mean() - \
                    df[(df['week'] == 22) & (df['switched_branch'] == 0)]['service_score'].mean()
print(f"\nExpected gap at week 22 (if pre-trend continued): {expected_gap_week22:.2f}")
print(f"Actual gap at week 22: {actual_gap_week22:.2f}")
print(f"Difference (potential treatment effect): {actual_gap_week22 - expected_gap_week22:.2f}")

# Alternative: compare average post-switch gap to average pre-switch gap
pre_switch_avg_gap = gap_df['gap'].mean()
post_switch = df[df['week'] >= 15]
post_gap_data = []
for week in range(15, 23):
    c = post_switch[post_switch['switched_branch'] == 0].groupby('week')['service_score'].mean().get(week, np.nan)
    t = post_switch[post_switch['switched_branch'] == 1].groupby('week')['service_score'].mean().get(week, np.nan)
    post_gap_data.append(t - c)
post_switch_avg_gap = np.mean(post_gap_data)

print(f"\nAverage pre-switch gap (weeks 6-14): {pre_switch_avg_gap:.2f}")
print(f"Average post-switch gap (weeks 15-22): {post_switch_avg_gap:.2f}")
print(f"Raw gap increase: {post_switch_avg_gap - pre_switch_avg_gap:.2f}")

# But account for the trend - what was the gap at week 14 vs average of weeks 15-22?
gap_week14 = gap_df[gap_df['week'] == 14]['gap'].values[0]
print(f"\nGap at week 14 (last pre-switch week): {gap_week14:.2f}")
print(f"Average gap weeks 15-22: {post_switch_avg_gap:.2f}")
print(f"Gap increase from week 14 baseline: {post_switch_avg_gap - gap_week14:.2f}")
