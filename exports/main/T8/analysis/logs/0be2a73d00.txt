import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Main DiD regression
print("=== Main Difference-in-Differences Regression ===")
model_did = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_did.summary())

# DiD with controls
print("\n=== DiD Regression with Controls ===")
model_controls = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_controls.summary())

# Branch fixed effects model (more robust)
print("\n=== DiD with Branch Fixed Effects ===")
# Create branch dummies
df_with_fe = df.copy()
df_with_fe = pd.get_dummies(df_with_fe, columns=['branch_id'], drop_first=True)

# Build formula with all branch dummies
branch_cols = [c for c in df_with_fe.columns if c.startswith('branch_id_')]
formula = 'service_score ~ switched_branch * after_switch + ' + ' + '.join(branch_cols)
model_fe = ols(formula, data=df_with_fe).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"R-squared: {model_fe.rsquared:.3f}")
print(f"\nTreatment effect (switched_branch:after_switch): {model_fe.params['switched_branch:after_switch']:.3f}")
print(f"Std Error: {model_fe.bse['switched_branch:after_switch']:.3f}")
print(f"P-value: {model_fe.pvalues['switched_branch:after_switch']:.4f}")
print(f"95% CI: [{model_fe.conf_int().loc['switched_branch:after_switch'][0]:.3f}, {model_fe.conf_int().loc['switched_branch:after_switch'][1]:.3f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study - create relative week indicators
# Week 14 is the last pre-treatment week, so we'll use it as reference
df = df.copy()
df['relative_week'] = df['week'] - 14  # week 14 becomes 0

# Create event study dummies (excluding reference week 14)
event_weeks = sorted(df['relative_week'].unique())
print("Relative weeks:", event_weeks)

# Create dummies for each relative week (excluding week 0 which is reference)
for w in event_weeks:
    if w != 0:  # Skip reference week
        df[f'rel_week_{w}'] = (df['relative_week'] == w).astype(int)

# Run event study regression with treatment interaction
# Only include treatment branches for the interaction terms
event_cols = [f'rel_week_{w}' for w in event_weeks if w != 0]
formula = 'service_score ~ ' + ' + '.join([f'{c} * switched_branch' for c in event_cols]) + ' + switched_branch'
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the interaction terms
print("\n=== Event Study Coefficients (Treatment Effect by Week) ===")
results = []
for w in event_weeks:
    if w == 0:
        results.append((w, 0, 0, 0))  # Reference week
    else:
        coef_name = f'rel_week_{w}:switched_branch'
        if coef_name in model_event.params:
            coef = model_event.params[coef_name]
            se = model_event.bse[coef_name]
            results.append((w, coef, se, model_event.pvalues[coef_name]))

results_df = pd.DataFrame(results, columns=['relative_week', 'coef', 'se', 'pvalue'])
results_df = results_df.sort_values('relative_week')
print(results_df)

# Plot event study
fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(results_df['relative_week'], results_df['coef'], yerr=1.96*results_df['se'], 
            fmt='o', capsize=5, label='Treatment Effect')
ax.axhline(y=0, color='red', linestyle='--', label='No Effect')
ax.axvline(x=0.5, color='green', linestyle='-', alpha=0.5, label='Switch Point')
ax.set_xlabel('Relative Week (0 = Week 14, last pre-switch week)')
ax.set_ylabel('Service Score Effect')
ax.set_title('Event Study: Treatment Effect Over Time')
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved to event_study.png")

import pandas as pd
import numpy as np
import statsmodels.api as sm
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Simpler event study - create treatment*post interaction for each week
df = df.copy()
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Create week-specific treatment effects
for w in range(6, 23):
    df[f'week_{w}'] = (df['week'] == w).astype(int)
    df[f'treat_week_{w}'] = df['switched_branch'] * df[f'week_{w}']

# Run regression with week fixed effects and treatment-by-week interactions
# Use week 14 as reference (last pre-treatment week)
week_dummies = [f'week_{w}' for w in range(6, 23) if w != 14]
treat_week_dummies = [f'treat_week_{w}' for w in range(6, 23) if w != 14]

formula = 'service_score ~ switched_branch + ' + ' + '.join(week_dummies) + ' + ' + ' + '.join(treat_week_dummies)
model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract treatment effects by week
print("=== Event Study: Treatment Effect by Week ===")
results = []
for w in range(6, 23):
    if w == 14:
        results.append({'week': w, 'relative_week': w-14, 'coef': 0, 'se': 0, 'pvalue': np.nan, 'ci_lower': 0, 'ci_upper': 0})
    else:
        coef_name = f'treat_week_{w}'
        coef = model.params[coef_name]
        se = model.bse[coef_name]
        ci = model.conf_int().loc[coef_name]
        results.append({
            'week': w, 
            'relative_week': w-14, 
            'coef': coef, 
            'se': se, 
            'pvalue': model.pvalues[coef_name],
            'ci_lower': ci[0],
            'ci_upper': ci[1]
        })

results_df = pd.DataFrame(results)
results_df = results_df.sort_values('relative_week')
print(results_df.to_string(index=False))

# Plot
fig, ax = plt.subplots(figsize=(10, 6))
pre_df = results_df[results_df['relative_week'] < 0]
post_df = results_df[results_df['relative_week'] >= 0]

ax.errorbar(pre_df['relative_week'], pre_df['coef'], yerr=1.96*pre_df['se'], 
            fmt='o', capsize=5, label='Pre-Switch', color='blue', alpha=0.7)
ax.errorbar(post_df['relative_week'], post_df['coef'], yerr=1.96*post_df['se'], 
            fmt='o', capsize=5, label='Post-Switch', color='green', alpha=0.7)
ax.axhline(y=0, color='red', linestyle='--', label='No Effect')

ax.set_ylabel('Service Score Effect (vs. Control)')
ax.set_title('Event Study: Treatment Effect Over Time')

print("\nEvent study plot saved to event_study.png")

print("\n=== Pre-Treatment Effects Test (weeks 6-13) ===")
