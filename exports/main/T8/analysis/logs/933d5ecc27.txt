print(pivot)

# Calculate the difference-in-differences manually
# Pre-period means
pre_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()

# Post-period means
post_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

# DiD estimate
did_estimate = (post_treatment - pre_treatment) - (post_control - pre_control)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with robust standard errors
# Model 1: Basic DiD
model1 = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit(cov_type='HC1')
print("Model 1: Basic DiD")
print(model1.summary())

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment_post + staffing_index + visits_index', data=df).fit(cov_type='HC1')
print("\n\nModel 2: DiD with controls")
print(model2.summary())

# Model 3: DiD with branch fixed effects
# This controls for time-invariant branch characteristics
model3 = ols('service_score ~ after_switch + treatment_post + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='HC1')
print("\n\nModel 3: DiD with branch fixed effects")
