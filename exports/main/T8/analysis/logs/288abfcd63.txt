import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load data
df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Basic DiD: Calculate means by group and period
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES: MEAN SERVICE SCORES")
print("=" * 60)

# Pre-period (after_switch = 0)
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
pre_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()

# Post-period (after_switch = 1)
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
post_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nControl Group (switched_branch=0):")
print(f"  Pre-period mean: {pre_control:.2f}")
print(f"  Post-period mean: {post_control:.2f}")
print(f"  Change: {post_control - pre_control:.2f}")

print(f"\nTreatment Group (switched_branch=1):")
print(f"  Pre-period mean: {pre_treatment:.2f}")
print(f"  Post-period mean: {post_treatment:.2f}")
print(f"  Change: {post_treatment - pre_treatment:.2f}")

# DiD estimate
did_estimate = (post_treatment - pre_treatment) - (post_control - pre_control)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

# Run regression
print("\n" + "=" * 60)
print("REGRESSION RESULTS")
print("=" * 60)

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment_post + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Clustered standard errors by branch
print("\n" + "=" * 60)
print("CLUSTERED STANDARD ERRORS (by branch)")
print("=" * 60)

# Use cluster-robust standard errors
model1_cluster = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 1 with clustered SE:")
print(model1_cluster.summary().tables[1])

model2_cluster = ols('service_score ~ switched_branch + after_switch + treatment_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 2 with clustered SE:")
print(model2_cluster.summary().tables[1])

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load data
df = pd.read_csv('branches.csv')

# Event study with cleaner specification
df['relative_week'] = df['week'] - 14  # Week 14 = 0 (last pre-period week)

# Create dummy variables for each week
for w in df['relative_week'].unique():
    df[f'week_{w}'] = (df['relative_week'] == w).astype(int)

# Create interaction terms
for w in df['relative_week'].unique():
    df[f'treat_week_{w}'] = df['switched_branch'] * df[f'week_{w}']

# Run event study (omit week 0 as reference)
pre_weeks = [-8, -7, -6, -5, -4, -3, -2, -1]  # Weeks 6-13
post_weeks = [1, 2, 3, 4, 5, 6, 7, 8]  # Weeks 15-22

# Build formula
pre_terms = ' + '.join([f'treat_week_{w}' for w in pre_weeks])
post_terms = ' + '.join([f'treat_week_{w}' for w in post_weeks])
week_terms = ' + '.join([f'week_{w}' for w in pre_weeks + post_weeks])

formula = f'service_score ~ switched_branch + {week_terms} + {pre_terms} + {post_terms}'

model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=" * 70)
print("EVENT STUDY: Treatment Effect by Week (Reference: Week 14)")
print("=" * 70)

results = []
print("\nPRE-PERIOD (should be ~0 if parallel trends hold):")
for w in pre_weeks:
    coef_name = f'treat_week_{w}'
    if coef_name in model.params.index:
        coef = model.params[coef_name]
        pval = model.pvalues[coef_name]
        ci = model.conf_int().loc[coef_name]
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"  Week {14+w}: {coef:+.2f}  [{ci[0]:+.2f}, {ci[1]:+.2f}]  p={pval:.3f} {sig}")
        results.append({'week': 14+w, 'coef': coef, 'ci_low': ci[0], 'ci_high': ci[1], 'period': 'pre'})

print("\nPOST-PERIOD (treatment effect):")
for w in post_weeks:
    coef_name = f'treat_week_{w}'
    if coef_name in model.params.index:
        coef = model.params[coef_name]
        pval = model.pvalues[coef_name]
        ci = model.conf_int().loc[coef_name]
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"  Week {14+w}: {coef:+.2f}  [{ci[0]:+.2f}, {ci[1]:+.2f}]  p={pval:.3f} {sig}")
        results.append({'week': 14+w, 'ci_low': ci[0], 'ci_high': ci[1], 'coef': coef, 'period': 'post'})

# Joint tests
print("\n" + "=" * 70)
print("JOINT SIGNIFICANCE TESTS")
print("=" * 70)

pre_coefs = [f'treat_week_{w}' for w in pre_weeks if f'treat_week_{w}' in model.params.index]
post_coefs = [f'treat_week_{w}' for w in post_weeks if f'treat_week_{w}' in model.params.index]

if pre_coefs:
    pre_test = model.f_test(pre_coefs)
    print(f"\nPre-period effects jointly significant: {pre_test.pvalue < 0.05} (p={pre_test.pvalue:.4f})")

if post_coefs:
    post_test = model.f_test(post_coefs)
    print(f"Post-period effects jointly significant: {post_test.pvalue < 0.05} (p={post_test.pvalue:.4f})")

# Create event study plot
results_df = pd.DataFrame(results)

fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(results_df['week'], results_df['coef'], 
            yerr=[results_df['coef'] - results_df['ci_low'], results_df['ci_high'] - results_df['coef']],
            fmt='o', capsize=5, label='Treatment Effect')
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point')
ax.set_xlabel('Week')
ax.set_ylabel('Treatment Effect (vs Control)')
ax.set_title('Event Study: Treatment Effect by Week\n(Reference: Week 14 = 0)')
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150, bbox_inches='tight')
print("\nSaved event study plot to event_study.png")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load data
df = pd.read_csv('branches.csv')

# Use C() notation for categorical variable
# This creates an event study with week 0 (week 14) as reference
formula = 'service_score ~ C(relative_week) * switched_branch'

model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=" * 70)
print("EVENT STUDY: Treatment Effect by Week")
print("=" * 70)
print("Reference period: Week 14 (last week before switch)")
print()

# Extract interaction terms (these are the treatment effects for each week)
pre_weeks = [-8, -7, -6, -5, -4, -3, -2, -1]  # Weeks 6-13
post_weeks = [1, 2, 3, 4, 5, 6, 7, 8]  # Weeks 15-22

print("PRE-PERIOD (should be ~0 if parallel trends hold):")
for w in pre_weeks:
    coef_name = f'C(relative_week)[{w}]:switched_branch'
    if coef_name in model.params.index:
        coef = model.params[coef_name]
        pval = model.pvalues[coef_name]
        ci = model.conf_int().loc[coef_name]
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"  Week {14+w}: {coef:+.2f}  [{ci[0]:+.2f}, {ci[1]:+.2f}]  p={pval:.3f} {sig}")

for w in post_weeks:

        ci = model.conf_int().loc[coef_name]
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""

# Joint tests
print("\n" + "=" * 70)
print("JOINT SIGNIFICANCE TESTS")
print("=" * 70)

import pandas as pd
import numpy as np
import statsmodels.api as sm

# Load data
df = pd.read_csv('branches.csv')

# Create relative week (week 14 = 0, the last pre-period week)
df['relative_week'] = df['week'] - 14

# Create week dummies with safe names (excluding week 0 as reference)
for w in range(-8, 9):
    if w != 0:
        safe_name = f'w_{abs(w)}_{"n" if w < 0 else "p"}'
        df[safe_name] = (df['relative_week'] == w).astype(int)
        df[f't_{abs(w)}_{"n" if w < 0 else "p"}'] = df['switched_branch'] * df[safe_name]

# Get all predictor columns
week_cols = [c for c in df.columns if c.startswith('w_')]
treat_cols = [c for c in df.columns if c.startswith('t_')]

# Build design matrix
X = df[['switched_branch'] + week_cols + treat_cols]
X = sm.add_constant(X)
y = df['service_score']

# Fit model with clustered SE
model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=" * 70)
print("EVENT STUDY: Treatment Effect by Week")
print("=" * 70)
print("Reference period: Week 14 (relative_week = 0)")
print()

# Map safe names back to actual weeks
def safe_to_week(col):
    parts = col.split('_')
    num = int(parts[1])
    sign = -1 if parts[2] == 'n' else 1
    return sign * num

pre_weeks = [-8, -7, -6, -5, -4, -3, -2, -1]
post_weeks = [1, 2, 3, 4, 5, 6, 7, 8]

print("PRE-PERIOD (should be ~0 if parallel trends hold):")
pre_coef_names = []
for w in pre_weeks:
    safe = f't_{abs(w)}_{"n" if w < 0 else "p"}'
    pre_coef_names.append(safe)
    coef = model.params[safe]
    pval = model.pvalues[safe]
    ci = model.conf_int().loc[safe]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Week {14+w}: {coef:+.2f}  [{ci[0]:+.2f}, {ci[1]:+.2f}]  p={pval:.3f} {sig}")

print("\nPOST-PERIOD (treatment effect):")
post_coef_names = []
post_effects = []
for w in post_weeks:
    safe = f't_{abs(w)}_{"n" if w < 0 else "p"}'
    post_coef_names.append(safe)
    coef = model.params[safe]
    pval = model.pvalues[safe]
    ci = model.conf_int().loc[safe]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Week {14+w}: {coef:+.2f}  [{ci[0]:+.2f}, {ci[1]:+.2f}]  p={pval:.3f} {sig}")
    post_effects.append(coef)

# Joint tests
print("\n" + "=" * 70)
print("JOINT SIGNIFICANCE TESTS")
print("=" * 70)

pre_test = model.f_test(pre_coef_names)
print(f"\nPre-period effects jointly significant: {pre_test.pvalue < 0.05} (p={pre_test.pvalue:.4f})")

post_test = model.f_test(post_coef_names)
print(f"Post-period effects jointly significant: {post_test.pvalue < 0.05} (p={post_test.pvalue:.4f})")

# Average post-period effect
avg_post_effect = np.mean(post_effects)
print(f"\nAverage post-period treatment effect: {avg_post_effect:.2f} points")

# Compare to simple DiD
simple_did = 2.1042
print(f"Simple DiD estimate: {simple_did:.2f} points")

# Key insight: look at pre-trend coefficients
print("\n" + "=" * 70)
print("KEY CONCERN: PRE-TREND COEFFICIENTS")
print("=" * 70)
print("\nIf parallel trends held, pre-period coefficients should be near zero.")
print("Instead, we see:")
pre_effect_values = [model.params[c] for c in pre_coef_names]
print(f"  Range: {min(pre_effect_values):.2f} to {max(pre_effect_values):.2f}")
print(f"  Mean: {np.mean(pre_effect_values):.2f}")
print(f"  Last pre-period week (week 13): {model.params['t_1_n']:.2f} (p={model.pvalues['t_1_n']:.3f})")
