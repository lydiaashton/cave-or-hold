import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Check parallel trends - compare pre-treatment trends
pre_data = df[df['week'] < 15].copy()

# Average service score by week and treatment group
pre_trends = pre_data.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print("Pre-treatment trends (average service score by week):")
print(pre_trends)

# Calculate week-over-week changes for each group
print("\nWeek-over-week changes in pre-treatment period:")
for group in [0, 1]:
    group_data = pre_trends[group]
    changes = group_data.diff()
    print(f"\nSwitched={group}:")
    print(changes)

# Simple DiD regression
print("\n" + "="*60)
print("DIFFERENCE-IN-DIFFERENCES ANALYSIS")
print("="*60)

# Model 1: Basic DiD (no controls)
model1 = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with controls (staffing_index, visits_index)")
print(model2.summary().tables[1])

# Model 3: DiD with branch fixed effects
model3 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index + C(branch_id)', data=df).fit()
print("\nModel 3: DiD with branch fixed effects")
# Just show the DiD coefficient
did_coef = model3.params['did']
did_pval = model3.pvalues['did']
print(f"DID coefficient: {did_coef:.4f} (p-value: {did_pval:.4f})")

# Clustered standard errors by branch (more appropriate for panel data)
print("\n" + "="*60)
print("CLUSTERED STANDARD ERRORS (by branch)")
print("="*60)

# Use cluster-robust standard errors
model1_cluster = model1.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print("\nModel 1 with clustered SE:")
print(model1_cluster.summary().tables[1])

model2_cluster = model2.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print("\nModel 2 with clustered SE:")
print(model2_cluster.summary().tables[1])

for group in [0, 1]:
    group_data = pre_df[pre_df['switched_branch'] == group].copy()

    week_coef = model_cluster.params['week_numeric']
    week_pval = model_cluster.pvalues['week_numeric']
    print(f"\nSwitched={group}: Weekly trend = {week_coef:+.3f} points/week (p={week_pval:.4f})")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

print("="*70)
print("CRITICAL RE-EXAMINATION")
print("="*70)

pre_df = df[df['week'] < 15].copy()

# Average service score by week for each group
pre_trends = pre_df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print("\nPre-treatment average service scores by week:")
print(pre_trends)

# Calculate the gap between treatment and control each week
gaps = pre_trends[1] - pre_trends[0]
print("\nTreatment-Control gap by week (pre-treatment):")
print(gaps)

# Is the gap increasing over time?
print(f"\nGap at week 6: {gaps.iloc[0]:.2f}")
print(f"Gap at week 14: {gaps.iloc[-1]:.2f}")
print(f"Change in gap over pre-period: {gaps.iloc[-1] - gaps.iloc[0]:.2f} points")

# Linear trend of the gap
gap_df = gaps.reset_index()
gap_df.columns = ['week', 'gap']
trend_model = ols('gap ~ week', data=gap_df).fit()
print(f"\nTrend in gap: {trend_model.params['week']:.3f} points per week")
print(f"P-value: {trend_model.pvalues['week']:.4f}")

# Post-treatment gaps
print("\n" + "="*70)
print("POST-TREATMENT GAPS")
print("="*70)

post_df = df[df['week'] >= 15].copy()
post_trends = post_df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
post_gaps = post_trends[1] - post_trends[0]
print("\nTreatment-Control gap by week (post-treatment):")
print(post_gaps)
print(f"\nAverage post-treatment gap: {post_gaps.mean():.2f}")
print(f"Average pre-treatment gap: {gaps.mean():.2f}")
print(f"DiD estimate: {post_gaps.mean() - gaps.mean():.2f}")

# The key question: is the post-treatment gap larger than what pre-trends would predict?
print("\n" + "="*70)
print("KEY QUESTION: IS THERE A BREAK AT WEEK 15?")
print("="*70)

# If pre-trend continued, what would we expect?
pre_trend_per_week = trend_model.params['week']  # ~0.23
weeks_from_14 = np.array([1, 2, 3, 4, 5, 6, 7, 8])  # weeks 15-22
expected_gap = gaps.iloc[-1] + pre_trend_per_week * weeks_from_14
actual_gap = post_gaps.values

print(f"\nPre-trend: {pre_trend_per_week:.3f} points/week")
print(f"\nIf pre-trend continued, expected gaps:")
for i, (exp, act) in enumerate(zip(expected_gap, actual_gap)):
    week = 15 + i
    print(f"  Week {week}: Expected {exp:.2f}, Actual {act:.2f}, Difference {act-exp:+.2f}")

# Average difference between actual and expected
avg_difference = (actual_gap - expected_gap).mean()
print(f"\nAverage actual vs expected: {avg_difference:+.2f} points")

# This is the effect beyond what pre-trends would predict
print("\n" + "="*70)
print("ADJUSTED EFFECT ESTIMATE")
print("="*70)
print(f"""
Raw DiD estimate: 1.89 points
Pre-trend artifact (0.23 × 8 weeks): ~1.84 points
Adjusted estimate: ~0.05 points

However, looking at actual vs expected:
Average difference: {avg_difference:+.2f} points

This suggests the true effect may be smaller than the raw DiD,
but still potentially positive.
""")

# Let's also check if there's heterogeneity - maybe some branches benefited more
print("\n" + "="*70)
print("HETEROGENEITY CHECK")
print("="*70)

early_pre_gap = df[df['week'].isin([11, 12, 13, 14])][df[df['week'].isin([11, 12, 13, 14])]['switched_branch']==1]['service_score'].mean() - df[df['week'].isin([11, 12, 13, 14])][df[df['week'].isin([11, 12, 13, 14])]['switched_branch']==0]['service_score'].mean()

early_did = early_post_gap - early_pre_gap
