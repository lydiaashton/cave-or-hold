before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()

print(f"Treatment (switched) - Before: {before_treatment:.2f}, After: {after_treatment:.2f}")

control_change = after_control - before_control
treatment_change = after_treatment - before_treatment

print(f"\nControl group change: {control_change:.2f}")
print(f"Treatment group change: {treatment_change:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*treatment + beta2*post + beta3*(treatment*post) + error
# beta3 is the DiD estimate

df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# Now with branch fixed effects (more robust)
print("\n\n=== DiD with Branch Fixed Effects ===")
df_with_dummies = pd.get_dummies(df, columns=['branch_id'], drop_first=True)
fixed_effects_cols = [col for col in df_with_dummies.columns if col.startswith('branch_id_')]

# Build formula
formula = 'service_score ~ switched_branch + after_switch + treatment_post + ' + ' + '.join(fixed_effects_cols)
model_fe = ols(formula, data=df_with_dummies).fit()
print(model_fe.summary())

# Also check with standard errors clustered by branch
print("\n\n=== DiD with Branch Fixed Effects (Clustered SE) ===")
model_fe_cluster = ols('service_score ~ switched_branch + after_switch + treatment_post + C(branch_id)', data=df).fit()
cov_cluster = sm.stats.get_robustcov_results(model_fe_cluster, 'cluster', groups=df['branch_id'])
print(cov_cluster.summary())

import numpy as np

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# The key issue: switched_branch is time-invariant, so it's absorbed by branch FE
# We need to focus on the interaction term (treatment_post)

# Run regression with branch FE but without the main treatment effect (it's collinear)
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Model with branch FE - switched_branch will be dropped due to collinearity
model_fe = ols('service_score ~ after_switch + treatment_post + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== DiD with Branch Fixed Effects (Clustered SE by Branch) ===")
print(model_fe.summary())

# Also check week fixed effects to control for time trends
print("\n\n=== DiD with Branch and Week Fixed Effects ===")
model_fe_week = ols('service_score ~ treatment_post + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_fe_week.summary())
