df['treatment'] = df['switched_branch'] * df['after_switch']

# Model 1: Basic DiD
print("=== Model 1: Basic DiD ===")
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(f"Treatment effect: {model1.params['treatment']:.3f}, p-value: {model1.pvalues['treatment']:.3f}")

# Model 2: DiD with controls
print("\n=== Model 2: DiD with Controls ===")
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print(f"Treatment effect: {model2.params['treatment']:.3f}, p-value: {model2.pvalues['treatment']:.3f}")
print(model2.summary())

# Model 3: Branch fixed effects (absorbing switched_branch since it's time-invariant)
print("\n=== Model 3: Branch Fixed Effects ===")
# Use demeaning approach or just include branch dummies without switched_branch
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_fe = pd.concat([df, branch_dummies], axis=1)
branch_cols = [col for col in df_fe.columns if col.startswith('branch_')]
formula_fe = 'service_score ~ after_switch + treatment + staffing_index + visits_index + ' + ' + '.join(branch_cols)
model3 = ols(formula_fe, data=df_fe).fit()
print(f"Treatment effect: {model3.params['treatment']:.3f}, p-value: {model3.pvalues['treatment']:.3f}")

# Model 4: Clustered standard errors by branch
print("\n=== Model 4: Clustered SEs by Branch ===")
from statsmodels.stats.sandwich_covariance import cov_cluster
model4 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
cov_clustered = cov_cluster(model4, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))

from scipy import stats

print(f"Probability mass is mostly positive, but uncertainty is substantial")

# Standardized effect size
