import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# Let's do a more formal statistical test
# Create treatment indicator (interaction of switched_branch and after_switch)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple regression-style analysis
# We'll compare the treatment effect using a t-test on the "effect"
# Group by branch to get branch-level before/after changes

branch_data = df.groupby(['branch_id', 'switched_branch']).apply(
    lambda x: pd.Series({
        'before_mean': x[x['after_switch'] == 0]['service_score'].mean(),
        'after_mean': x[x['after_switch'] == 1]['service_score'].mean(),
        'change': x[x['after_switch'] == 1]['service_score'].mean() - x[x['after_switch'] == 0]['service_score'].mean()
    })
).reset_index()

print("Branch-level changes:")
print(branch_data.head(10))

# Compare changes between switched and non-switched branches
switched_changes = branch_data[branch_data['switched_branch'] == 1]['change']
non_switched_changes = branch_data[branch_data['switched_branch'] == 0]['change']

print(f"\nSwitched branches - mean change: {switched_changes.mean():.3f}, std: {switched_changes.std():.3f}, n: {len(switched_changes)}")
print(f"Non-switched branches - mean change: {non_switched_changes.mean():.3f}, std: {non_switched_changes.std():.3f}, n: {len(non_switched_changes)}")

# Two-sample t-test
t_stat, p_value = stats.ttest_ind(switched_changes, non_switched_changes)
print(f"\nTwo-sample t-test comparing changes:")
print(f"  t-statistic: {t_stat:.3f}")
print(f"  p-value: {p_value:.6f}")

# Effect size (Cohen's d)
pooled_std = np.sqrt(((len(switched_changes)-1)*switched_changes.std()**2 + (len(non_switched_changes)-1)*non_switched_changes.std()**2) / (len(switched_changes)+len(non_switched_changes)-2))
cohens_d = (switched_changes.mean() - non_switched_changes.mean()) / pooled_std
print(f"  Cohen's d (effect size): {cohens_d:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# More careful parallel trends check
pre_switch = df[df['week'] <= 14].copy()

# Test for parallel trends more carefully
pre_switch['week_centered'] = pre_switch['week'] - 10

X = pre_switch[['switched_branch', 'week_centered']].copy()
X['interaction'] = X['switched_branch'] * X['week_centered']
X = sm.add_constant(X)
y = pre_switch['service_score']

model_trend = sm.OLS(y, X).fit()

print("=== PARALLEL TRENDS TEST ===")
print("\nPre-switch trend interaction term:")
print(f"  Coefficient: {model_trend.params['interaction']:.3f}")
print(f"  p-value: {model_trend.pvalues['interaction']:.3f}")
print(f"  95% CI: [{model_trend.conf_int().loc['interaction', 0]:.3f}, {model_trend.conf_int().loc['interaction', 1]:.3f}]")

print("\n=== POTENTIAL CONCERNS ===")

# Check if there's a pre-existing level difference
pre_diff = pre_switch.groupby('switched_branch')['service_score'].mean()
print(f"\n1. Pre-existing level difference: {pre_diff[1] - pre_diff[0]:.2f} points")
print(f"   (Switched branches started {pre_diff[1] - pre_diff[0]:.2f} points higher)")

# Check the DiD confidence interval
post_switch = df[df['week'] >= 15].copy()

# Branch-level changes
branch_data = df.groupby(['branch_id', 'switched_branch']).apply(
    lambda x: pd.Series({
        'before_mean': x[x['after_switch'] == 0]['service_score'].mean(),
        'after_mean': x[x['after_switch'] == 1]['service_score'].mean(),
        'change': x[x['after_switch'] == 1]['service_score'].mean() - x[x['after_switch'] == 0]['service_score'].mean()
    })
).reset_index()

switched_changes = branch_data[branch_data['switched_branch'] == 1]['change']
non_switched_changes = branch_data[branch_data['switched_branch'] == 0]['change']

did_effect = switched_changes.mean() - non_switched_changes.mean()
from scipy import stats
t_stat, p_value = stats.ttest_ind(switched_changes, non_switched_changes)

# Confidence interval for DiD
pooled_se = np.sqrt(switched_changes.var()/len(switched_changes) + non_switched_changes.var()/len(non_switched_changes))
ci_lower = did_effect - 1.96 * pooled_se
ci_upper = did_effect + 1.96 * pooled_se

print(f"\n2. DiD Effect Estimate: {did_effect:.2f} points")
print(f"   95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"   p-value: {p_value:.6f}")

# Check if effect is consistent across branches
print(f"\n3. Effect consistency:")
print(f"   % of switched branches with positive change: {(switched_changes > 0).mean()*100:.1f}%")
print(f"   % of non-switched branches with positive change: {(non_switched_changes > 0).mean()*100:.1f}%")

# Check for any obvious data issues
print(f"\n4. Data quality checks:")
print(f"   Service score range: [{df['service_score'].min():.1f}, {df['service_score'].max():.1f}]")
print(f"   Missing values: {df.isnull().sum().sum()}")
