import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Count branches by switch status
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate pre/post means for both groups
print("\n=== Service Score Means ===")
print("\nPre-switch (weeks 6-14):")
pre = df[df['after_switch'] == 0].groupby('switched_branch')['service_score'].mean()
print(pre)

print("\nPost-switch (weeks 15-22):")
post = df[df['after_switch'] == 1].groupby('switched_branch')['service_score'].mean()
print(post)

# Calculate differences
print("\n=== Difference-in-Differences Calculation ===")
control_pre = pre[0]
control_post = post[0]
treatment_pre = pre[1]
treatment_post = post[1]

control_diff = control_post - control_pre
treatment_diff = treatment_post - treatment_pre
did_estimate = treatment_diff - control_diff

print(f"Control group change: {control_diff:.3f}")
print(f"Treatment group change: {treatment_diff:.3f}")
print(f"DiD estimate (treatment effect): {did_estimate:.3f}")

# Run regression with DiD
print("\n=== Regression Analysis ===")
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model.summary())

# Add branch fixed effects
print("\n=== With Branch Fixed Effects ===")
df['branch_id'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit()
print(model_fe.summary())
