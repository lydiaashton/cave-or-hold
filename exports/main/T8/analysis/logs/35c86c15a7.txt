import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is our DiD estimate

model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# Now let's add branch fixed effects and week fixed effects for a more robust analysis
# This controls for time-invariant branch characteristics and common time trends

df['switched_after'] = df['switched_branch'] * df['after_switch']

# Add branch and week as categorical variables
model_fe = ols('service_score ~ switched_after + C(branch_id) + C(week)', data=df).fit()
print("\n=== DiD with Branch and Week Fixed Effects ===")
print(model_fe.summary())

# Let's also check if we should control for staffing and visits
model_controls = ols('service_score ~ switched_after + C(branch_id) + C(week) + staffing_index + visits_index', data=df).fit()
print("\n=== DiD with Branch/Week FE and Controls ===")
print(model_controls.summary())
