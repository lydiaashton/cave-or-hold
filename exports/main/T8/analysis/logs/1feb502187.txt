import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Calculate average service scores by group and period
print("=== Average Service Scores by Group and Period ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the difference-in-differences manually
print("\n=== Difference-in-Differences Calculation ===")

# Before switch (after_switch = 0)
before_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()

# After switch (after_switch = 1)
after_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"Switched branches - Before: {before_switched:.2f}, After: {after_switched:.2f}, Change: {after_switched - before_switched:.2f}")
print(f"Control branches - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")
print(f"\nDifference-in-Differences: {(after_switched - before_switched) - (after_control - before_control):.2f}")

# Run a regression with difference-in-differences
print("\n=== Regression Analysis ===")
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model.summary())

# Regression with branch fixed effects
print("\n=== Regression with Branch Fixed Effects ===")
df['branch_id'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit()
print(model_fe.summary())

# Check for pre-trends more formally
# Compare trends in weeks 6-14 between switched and control
print("=== Pre-Treatment Trend Analysis ===")
pre_data = df[df['week'] < 15].copy()

# Run regression on pre-period only to check if there's a differential trend
pre_data['branch_id'] = pre_data['branch_id'].astype(str)
pre_data['week_num'] = pre_data['week']
model_pre = ols('service_score ~ week_num * switched_branch + C(branch_id)', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})

print(model_pre.bse[['week_num', 'switched_branch', 'week_num:switched_branch']])
print(model_pre.pvalues[['week_num', 'switched_branch', 'week_num:switched_branch']])

# Event study - effect by week relative to switch
print("\n=== Event Study Analysis ===")
df['weeks_to_switch'] = df['week'] - 15  # week 15 is when switch happened
df['weeks_to_switch'] = df['weeks_to_switch'].astype(int)

# Create event study dummies
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    df[f'event_{w}'] = ((df['weeks_to_switch'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (omitting week -1 as reference)
event_vars = [f'event_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ ' + ' + '.join(event_vars) + ' + C(branch_id) + C(week)'
df_es = df.copy()

df_es['week'] = df_es['week'].astype(str)

model_event = ols(formula, data=df_es).fit(cov_type='cluster', cov_kwds={'groups': df_es['branch_id']})

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Pre-treatment trend check - already done, shows no significant differential trend
# week_num:switched_branch p-value = 0.866, so parallel trends assumption appears reasonable

# Simplified event study - group pre and post periods
print("=== Simplified Analysis ===")

# Calculate means for each group-period
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Main DiD estimate with cluster-robust SE
df_model = df.copy()
df_model['branch_id'] = df_model['branch_id'].astype(str)
df_model['week'] = df_model['week'].astype(str)

# Model 1: Basic DiD with branch and week FE
model1 = ols('service_score ~ treatment + C(branch_id) + C(week)', data=df_model).fit(cov_type='cluster', cov_kwds={'groups': df_model['branch_id']})
print("\n=== Model 1: DiD with Branch + Week FE ===")
print(f"Treatment effect: {model1.params['treatment']:.2f}")
print(f"Std Error: {model1.bse['treatment']:.2f}")
print(f"95% CI: [{model1.conf_int().loc['treatment'][0]:.2f}, {model1.conf_int().loc['treatment'][1]:.2f}]")
print(f"P-value: {model1.pvalues['treatment']:.3f}")

# Model 2: With controls
model2 = ols('service_score ~ treatment + staffing_index + visits_index + C(branch_id) + C(week)', data=df_model).fit(cov_type='cluster', cov_kwds={'groups': df_model['branch_id']})
print("\n=== Model 2: DiD with Controls ===")
print(f"Treatment effect: {model2.params['treatment']:.2f}")
print(f"Std Error: {model2.bse['treatment']:.2f}")
print(f"95% CI: [{model2.conf_int().loc['treatment'][0]:.2f}, {model2.conf_int().loc['treatment'][1]:.2f}]")
print(f"P-value: {model2.pvalues['treatment']:.3f}")

# Effect size calculation
avg_score = df['service_score'].mean()
effect_size_pct = model1.params['treatment'] / avg_score * 100
print(f"\n=== Effect Size ===")
print(f"Average service score: {avg_score:.1f}")
print(f"Estimated treatment effect: {model1.params['treatment']:.2f} points")
print(f"Percentage improvement: {effect_size_pct:.1f}%")

# Power analysis - what's the minimum detectable effect?
n_branches = df['branch_id'].nunique()
n_periods = len(df['week'].unique())
n_treated = df['switched_branch'].sum() // n_periods  # approximate
n_control = (len(df) - df['switched_branch'].sum()) // n_periods

print(f"\n=== Sample Size ===")
print(f"Total branches: {n_branches}")
print(f"Treated branches: {df['switched_branch'].sum() // n_periods}")
print(f"Control branches: {(len(df) - df['switched_branch'].sum()) // n_periods}")
print(f"Weeks of data: {n_periods}")
print(f"Total observations: {len(df)}")
