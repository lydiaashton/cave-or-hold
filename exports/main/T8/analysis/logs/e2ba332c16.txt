import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# Run the main model again
model = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Key Statistics ===")
print(f"Treatment effect: {model.params['treated']:.2f}")
print(f"Standard error: {model.bse['treated']:.2f}")
print(f"t-statistic: {model.tvalues['treated']:.2f}")
print(f"P-value: {model.pvalues['treated']:.3f}")
print(f"95% CI: [{model.conf_int().loc['treated', 0]:.2f}, {model.conf_int().loc['treated', 1]:.2f}]")

# What's the probability the effect is positive?
# Using t-distribution
from scipy import stats
t_stat = model.tvalues['treated']
df_degrees = model.df_resid
# One-tailed p-value (probability of effect > 0)
one_tail_p = 1 - stats.t.cdf(0, df_degrees, loc=t_stat)
print(f"\nProbability effect is positive (one-tailed): {one_tail_p:.1%}")

# Power analysis - what effect size could we detect?
n_branches = 34
n_weeks = 17
print(f"\nSample: {n_branches} branches × {n_weeks} weeks = {n_branches*n_weeks} observations")
print(f"Treatment group: 20 branches, Control group: 14 branches")

# Standard deviation of service scores
print(f"\nService score SD: {df['service_score'].std():.2f}")
