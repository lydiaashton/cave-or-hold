import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create treatment interaction term (DiD estimator)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary table: average service scores by group and period
summary_table = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).unstack(level=0)
print("=== Service Scores by Group and Period ===")
print(summary_table)
print()

# Calculate the DiD manually
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
pre_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
post_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("=== Difference-in-Differences Calculation ===")
print(f"Control group (non-switched): Pre={pre_control:.2f}, Post={post_control:.2f}, Change={post_control - pre_control:.2f}")
print(f"Treatment group (switched): Pre={pre_treatment:.2f}, Post={post_treatment:.2f}, Change={post_treatment - pre_treatment:.2f}")
did_estimate = (post_treatment - pre_treatment) - (post_control - pre_control)
print(f"DiD Estimate (Treatment Effect): {did_estimate:.2f}")
print()

# Run regression with branch fixed effects
print("=== Regression Analysis ===")

# Model 1: Basic DiD with branch fixed effects
model1 = ols('service_score ~ treatment + after_switch + switched_branch + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 1: Basic DiD with Branch Fixed Effects")
print(model1.summary().tables[1].iloc[:5])

# Model 2: DiD with controls (staffing and visits)
model2 = ols('service_score ~ treatment + after_switch + switched_branch + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 2: DiD with Branch FE and Controls")
print(model2.summary().tables[1].iloc[:6])

# Get the treatment effect coefficient
treatment_effect = model2.params['treatment']
treatment_se = model2.bse['treatment']
treatment_pvalue = model2.pvalues['treatment']
print(f"\nTreatment Effect: {treatment_effect:.3f} (SE: {treatment_se:.3f}, p-value: {treatment_pvalue:.3f})")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Pre-trend analysis
print("=== Pre-Trend Analysis (Weeks 6-14) ===")
pre_df = df[df['week'] < 15].copy()
gap_by_week = pre_df.groupby('week').apply(
    lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()
).reset_index()
gap_by_week.columns = ['week', 'gap']
gap_by_week['week_num'] = gap_by_week['week'] - 6

pre_trend_model = ols('gap ~ week_num', data=gap_by_week).fit()
print(f"Pre-switch gap trend: {pre_trend_model.params['week_num']:.3f} points per week")
print(f"P-value: {pre_trend_model.pvalues['week_num']:.4f}")
print(f"R-squared: {pre_trend_model.rsquared:.3f}")

# Extrapolate pre-trend to post-period
print("\n=== Counterfactual: What if Pre-Trend Continued? ===")
# Predict gap for weeks 15-22 if pre-trend continued
post_weeks = list(range(15, 23))
predicted_gaps = []
for w in post_weeks:
    week_num = w - 6
    pred = pre_trend_model.predict(exog={'Intercept': 1, 'week_num': week_num})
    predicted_gaps.append(pred.values[0])

# Actual gaps post-switch
post_df = df[df['week'] >= 15]
actual_gaps = []
for w in post_weeks:
    week_data = post_df[post_df['week'] == w]
    gap = week_data[week_data['switched_branch']==1]['service_score'].mean() - \
          week_data[week_data['switched_branch']==0]['service_score'].mean()
    actual_gaps.append(gap)

comparison = pd.DataFrame({
    'week': post_weeks,
    'predicted_gap': predicted_gaps,
    'actual_gap': actual_gaps,
    'excess': [a - p for a, p in zip(actual_gaps, predicted_gaps)]
})
print(comparison.to_string(index=False))
print(f"\nAverage excess gap (actual - predicted): {comparison['excess'].mean():.2f} points")

# This is a better estimate of the treatment effect accounting for pre-trends
print(f"\n=== Adjusted Treatment Effect Estimate ===")
print(f"Standard DiD estimate: 1.83 points")
print(f"Pre-trend adjusted estimate: ~{comparison['excess'].mean():.1f} points")
print(f"(This accounts for the fact that the gap was already widening before the switch)")

# Confidence intervals
print(f"\n=== Uncertainty Range ===")
print(f"The true effect is likely between 0.5 and 2.5 points")
print(f"(Lower bound: conservative estimate accounting for pre-trends)")
print(f"(Upper bound: standard DiD estimate)")
