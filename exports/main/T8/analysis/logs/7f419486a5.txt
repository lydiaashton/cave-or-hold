# Create the interaction term for DiD
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with robust standard errors (clustered by branch)
# Basic DiD model
model_basic = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()

print(model_basic.summary())

model_controls = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', data=df).fit()

print(model_controls.summary())

# Clustered standard errors by branch
model_clustered = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Regression with Clustered SE (by branch) ===")
print(model_clustered.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Calculate weekly averages for treatment and control groups
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_score']

# Pivot for easier plotting
weekly_pivot = weekly_avg.pivot(index='week', columns='switched_branch', values='avg_score')
weekly_pivot.columns = ['Control', 'Treatment']

print("=== Weekly Averages by Group ===")
print(weekly_pivot)

# Calculate the difference between treatment and control each week
weekly_pivot['Difference'] = weekly_pivot['Treatment'] - weekly_pivot['Control']
print("\n=== Weekly Differences (Treatment - Control) ===")
print(weekly_pivot['Difference'])

# Pre-period trend check (weeks 6-14)
pre_period_diff = weekly_pivot.loc[6:14, 'Difference']
print(f"\nPre-period difference trend (weeks 6-14):")
print(pre_period_diff)
print(f"Pre-period mean difference: {pre_period_diff.mean():.2f}")
print(f"Pre-period std of difference: {pre_period_diff.std():.2f}")

# Check if there's a significant pre-trend
from scipy import stats
weeks_pre = pre_period_diff.index.values
diff_pre = pre_period_diff.values
slope, intercept, r_value, p_value, std_err = stats.linregress(weeks_pre, diff_pre)
print(f"\nPre-period trend slope: {slope:.3f} (p-value: {p_value:.3f})")

# Create visualization
plt.figure(figsize=(12, 6))
plt.plot(weekly_pivot.index, weekly_pivot['Control'], 'o-', label='Control (No Switch)', linewidth=2, markersize=6)
plt.plot(weekly_pivot.index, weekly_pivot['Treatment'], 's-', label='Treatment (Switched)', linewidth=2, markersize=6)

plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Score Trends: Treatment vs Control Groups', fontsize=14, fontweight='bold')
plt.legend(loc='best', fontsize=10)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("\nPlot saved as 'service_score_trends.png'")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Run regression with branch fixed effects
# Convert branch_id to categorical
df['branch_id_cat'] = df['branch_id'].astype('category')

# Model with branch fixed effects
model_fe = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== DiD with Branch Fixed Effects (Clustered SE) ===")
print(model_fe.summary())

# Also run with week fixed effects to control for time trends
df['week_cat'] = df['week'].astype('category')
model_fe_week = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD with Branch and Week Fixed Effects (Clustered SE) ===")
print(model_fe_week.summary())

# Extract key coefficient
print("\n=== Key Result ===")
print(f"Treatment effect (treated_post coefficient): {model_fe_week.params['treated_post']:.2f}")
print(f"Standard error: {model_fe_week.bse['treated_post']:.2f}")
print(f"95% CI: [{model_fe_week.conf_int().loc['treated_post', 0]:.2f}, {model_fe_week.conf_int().loc['treated_post', 1]:.2f}]")
print(f"P-value: {model_fe_week.pvalues['treated_post']:.4f}")

import pandas as pd
import numpy as np

# Load the data
df = pd.read_csv('branches.csv')

# Calculate the effect for each branch that switched
# Compare their post vs pre average
branch_effects = []
for branch_id in df['branch_id'].unique():
    branch_data = df[df['branch_id'] == branch_id]
    switched = branch_data['switched_branch'].iloc[0]
    
    pre_avg = branch_data[branch_data['after_switch'] == 0]['service_score'].mean()
    post_avg = branch_data[branch_data['after_switch'] == 1]['service_score'].mean()
    change = post_avg - pre_avg
    
    branch_effects.append({
        'branch_id': branch_id,
        'switched': switched,
        'pre_avg': pre_avg,
        'post_avg': post_avg,
        'change': change
    })

branch_effects_df = pd.DataFrame(branch_effects)

# Compare changes between treatment and control
treatment_changes = branch_effects_df[branch_effects_df['switched'] == 1]['change']
control_changes = branch_effects_df[branch_effects_df['switched'] == 0]['change']

print("=== Distribution of Changes ===")
print(f"\nTreatment branches (n={len(treatment_changes)}):")
print(f"  Mean change: {treatment_changes.mean():.2f}")
print(f"  Std dev: {treatment_changes.std():.2f}")
print(f"  Median: {treatment_changes.median():.2f}")
print(f"  Min: {treatment_changes.min():.2f}")
print(f"  Max: {treatment_changes.max():.2f}")
print(f"  % with positive change: {(treatment_changes > 0).mean()*100:.1f}%")

print(f"\nControl branches (n={len(control_changes)}):")
print(f"  Mean change: {control_changes.mean():.2f}")
print(f"  Std dev: {control_changes.std():.2f}")
print(f"  Median: {control_changes.median():.2f}")
print(f"  Min: {control_changes.min():.2f}")
print(f"  Max: {control_changes.max():.2f}")
print(f"  % with positive change: {(control_changes > 0).mean()*100:.1f}%")

# Difference in mean changes (this is the DiD estimate at branch level)
did_branch_level = treatment_changes.mean() - control_changes.mean()
print(f"\n=== Branch-Level DiD Estimate ===")
print(f"  Difference in mean changes: {did_branch_level:.2f}")

# How many treatment branches improved more than the average control branch?
control_mean_change = control_changes.mean()
treatment_above_control = (treatment_changes > control_mean_change).mean()
print(f"  % of treatment branches with above-average improvement: {treatment_above_control*100:.1f}%")

# Statistical test
from scipy import stats
t_stat, p_val = stats.ttest_ind(treatment_changes, control_changes)
print(f"\n  Two-sample t-test p-value: {p_val:.4f}")

import pandas as pd

# Load the data

model = ols('service_score ~ after_switch + treated_post + staffing_index + visits_index + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

ci_lower = model.conf_int().loc['treated_post', 0]
ci_upper = model.conf_int().loc['treated_post', 1]
print(f"CI excludes zero: {ci_lower > 0 or ci_upper < 0}")

pre_treatment = df[(df['after_switch'] == 0) & (df['switched_branch'] == 1)]['service_score'].mean()
