import pandas as pd
import numpy as np
import statsmodels.api as sm
import statsmodels.formula.api as smf
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = β0 + β1*switched + β2*after + β3*(switched*after) + ε
# β3 is our DiD estimator

df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Simple DiD Regression ===")
print(model.summary())

# Now with branch fixed effects (more robust)
# This controls for time-invariant branch characteristics
model_fe = smf.ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit()
print("\n=== DiD Regression with Branch Fixed Effects ===")
print(model_fe.summary())

# Also with week fixed effects to control for time trends
model_full = smf.ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id) + C(week)', data=df).fit()
print("\n=== DiD Regression with Branch and Week Fixed Effects ===")
print(model_full.summary())

    lambda x: pd.Series({
        'before_mean': x[x['after_switch']==0]['service_score'].mean(),
        'after_mean': x[x['after_switch']==1]['service_score'].mean(),
        'change': x[x['after_switch']==1]['service_score'].mean() - x[x['after_switch']==0]['service_score'].mean()
    })
).reset_index()

print(branch_summary.head())

print(f"\nDifference in changes: {diff_in_changes:.3f}")
