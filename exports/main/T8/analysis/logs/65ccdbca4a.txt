import pandas as pd
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
print("=== Basic DiD Regression ===")
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model1.summary())

# DiD with controls
print("\n=== DiD Regression with Controls ===")
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print(model2.summary())

# Branch fixed effects DiD (more robust)
print("\n=== DiD with Branch Fixed Effects ===")
df_with_dummies = pd.get_dummies(df, columns=['branch_id'], drop_first=True)
control_vars = ['after_switch', 'treatment', 'staffing_index', 'visits_index']
branch_dummies = [c for c in df_with_dummies.columns if c.startswith('branch_id_')]
formula = 'service_score ~ ' + ' + '.join(control_vars + branch_dummies)
model3 = ols(formula, data=df_with_dummies).fit()
print(model3.summary())

# Extract just the treatment effect
print("\n=== Treatment Effect Summary ===")
print(f"Basic DiD coefficient: {model1.params['treatment']:.3f} (SE: {model1.bse['treatment']:.3f})")
print(f"With controls: {model2.params['treatment']:.3f} (SE: {model2.bse['treatment']:.3f})")
print(f"With branch FE: {model3.params['treatment']:.3f} (SE: {model3.bse['treatment']:.3f})")

# Check parallel trends - compare pre-period trends

pre_data = df[df['week'] <= 14].copy()

# Regression on pre-period only with week trend
