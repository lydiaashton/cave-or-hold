import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create a treatment group indicator (interaction term for DiD)
df['treated'] = df['switched_branch'] * df['after_switch']

# Calculate average service scores by group and period
print("=== Average Service Scores ===")
print("\nBy switched_branch and after_switch:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate DiD manually
# Control group (switched_branch=0): before and after
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_change = control_after - control_before

# Treatment group (switched_branch=1): before and after
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_change = treat_after - treat_before

# Difference in Differences
did = treat_change - control_change

print(f"\n=== Difference-in-Differences Calculation ===")
print(f"Control group (non-switched):")
print(f"  Before: {control_before:.2f}")
print(f"  After: {control_after:.2f}")
print(f"  Change: {control_change:.2f}")
print(f"\nTreatment group (switched):")
print(f"  Before: {treat_before:.2f}")
print(f"  After: {treat_after:.2f}")
print(f"  Change: {treat_change:.2f}")
print(f"\nDiD Estimate (Treatment Change - Control Change): {did:.2f}")

# Count branches in each group
print(f"\n=== Branch Counts ===")
switched_count = df['branch_id'][df['switched_branch']==1].nunique()
non_switched_count = df['branch_id'][df['switched_branch']==0].nunique()
print(f"Switched branches: {switched_count}")
print(f"Non-switched branches: {non_switched_count}")

weekly_trends.columns = ['Control (No Switch)', 'Treatment (Switched)']

print("=== Weekly Trends by Group ===")
print(weekly_trends)

# Check pre-trends (weeks 6-14, before switch)
pre_trends = df[df['week'] < 15].groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print("\n=== Pre-Treatment Trends (Weeks 6-14) ===")
print(pre_trends)

# Calculate trend slopes for pre-period
control_pre = df[(df['week'] < 15) & (df['switched_branch']==0)].groupby('week')['service_score'].mean()
treat_pre = df[(df['week'] < 15) & (df['switched_branch']==1)].groupby('week')['service_score'].mean()

control_slope = np.polyfit(control_pre.index, control_pre.values, 1)[0]
treat_slope = np.polyfit(treat_pre.index, treat_pre.values, 1)[0]

print(f"\nPre-treatment trend slope (Control): {control_slope:.3f} points/week")
print(f"Pre-treatment trend slope (Treatment): {treat_slope:.3f} points/week")
print(f"Difference in slopes: {treat_slope - control_slope:.3f} points/week")

plt.figure(figsize=(12, 6))
plt.plot(weekly_trends.index, weekly_trends['Control (No Switch)'], 'o-', label='Control (No Switch)', linewidth=2, markersize=8)
plt.plot(weekly_trends.index, weekly_trends['Treatment (Switched)'], 's-', label='Treatment (Switched)', linewidth=2, markersize=8)

plt.tight_layout()
plt.savefig('trends_plot.png', dpi=150)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term
df['treated'] = df['switched_branch'] * df['after_switch']

# Run regression with branch fixed effects (more robust DiD)
# This controls for time-invariant branch characteristics
df_week = df.copy()

# Add week as categorical for time fixed effects
for w in range(6, 23):
    df_week[f'week_{w}'] = (df_week['week'] == w).astype(int)

# Build formula with branch FE (using C() for categorical)
# We'll use a simpler approach with branch_id as categorical
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_fe = pd.concat([df, branch_dummies], axis=1)

# Build the formula
branch_cols = [c for c in df_fe.columns if c.startswith('branch_')]
formula = f"service_score ~ switched_branch + after_switch + treated + {' + '.join(branch_cols)}"

model_fe = ols(formula, data=df_fe).fit()
cov_cluster_fe = model_fe.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("=== DiD with Branch Fixed Effects ===")
print(f"Treated coefficient: {cov_cluster_fe.params['treated']:.3f}")
print(f"Standard error: {cov_cluster_fe.bse['treated']:.3f}")
print(f"t-statistic: {cov_cluster_fe.tvalues['treated']:.3f}")
print(f"P-value: {cov_cluster_fe.pvalues['treated']:.3f}")
print(f"95% CI: [{cov_cluster_fe.conf_int().loc['treated', 0]:.3f}, {cov_cluster_fe.conf_int().loc['treated', 1]:.3f}]")

# Also add time (week) fixed effects
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df_fe_time = pd.concat([df, branch_dummies, week_dummies], axis=1)

branch_cols = [c for c in df_fe_time.columns if c.startswith('branch_')]
week_cols = [c for c in df_fe_time.columns if c.startswith('week_')]
formula_full = f"service_score ~ switched_branch + after_switch + treated + {' + '.join(branch_cols)} + {' + '.join(week_cols)}"

model_full = ols(formula_full, data=df_fe_time).fit()
cov_cluster_full = model_full.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("\n=== DiD with Branch AND Time Fixed Effects ===")
print(f"Treated coefficient: {cov_cluster_full.params['treated']:.3f}")
print(f"Standard error: {cov_cluster_full.bse['treated']:.3f}")
print(f"t-statistic: {cov_cluster_full.tvalues['treated']:.3f}")
print(f"P-value: {cov_cluster_full.pvalues['treated']:.3f}")
print(f"95% CI: [{cov_cluster_full.conf_int().loc['treated', 0]:.3f}, {cov_cluster_full.conf_int().loc['treated', 1]:.3f}]")

# Summary of all estimates
print("\n=== Summary of All DiD Estimates ===")
estimates = {
    'Basic DiD': (1.91, 1.26, 0.140),
    'With Controls': (2.06, 1.26, 0.110),
    'Branch FE': (cov_cluster_fe.params['treated'], cov_cluster_fe.bse['treated'], cov_cluster_fe.pvalues['treated']),
    'Branch + Time FE': (cov_cluster_full.params['treated'], cov_cluster_full.bse['treated'], cov_cluster_full.pvalues['treated'])
}

for method, (coef, se, pval) in estimates.items():
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.10 else ""
    print(f"{method:25s}: {coef:6.2f} (SE: {se:.2f}, p={pval:.3f}) {sig}")
