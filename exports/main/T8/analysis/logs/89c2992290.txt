import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
switched_counts = df.groupby('branch_id')['switched_branch'].first().value_counts()
print("Branches by switch status:")
print(switched_counts)
print(f"\nTotal branches: {len(df.groupby('branch_id'))}")
print(f"Switched: {switched_counts.get(1, 0)} ({100*switched_counts.get(1, 0)/len(df.groupby('branch_id')):.1f}%)")
print(f"Not switched: {switched_counts.get(0, 0)} ({100*switched_counts.get(0, 0)/len(df.groupby('branch_id')):.1f}%)")

# Average service scores by group and period
print("\n\nAverage service scores:")
df['period'] = df['after_switch'].map({0: 'Before (weeks 6-14)', 1: 'After (weeks 15-22)'})
df['group'] = df['switched_branch'].map({0: 'Control (no switch)', 1: 'Treatment (switched)'})

pivot = df.pivot_table(values='service_score', index='group', columns='period', aggfunc='mean')
print(pivot)

# Calculate differences
print("\n\nDifference-in-Differences calculation:")
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
before_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"Control group - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")
print(f"Treatment group - Before: {before_treatment:.2f}, After: {after_treatment:.2f}, Change: {after_treatment - before_treatment:.2f}")
print(f"\nDiD Estimate (Treatment change - Control change): {(after_treatment - before_treatment) - (after_control - before_control):.2f}")

# Weekly trends to check parallel trends assumption
print("\n\nWeekly average service scores by group:")
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly.columns = ['Control', 'Treatment']
weekly['Diff'] = weekly['Treatment'] - weekly['Control']
print(weekly)

import pandas as pd
import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model1 = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Model 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# DiD with branch fixed effects
model2 = smf.ols('service_score ~ C(branch_id) + after_switch + treatment', data=df).fit()
print("\n\nModel 2: DiD with branch fixed effects")
# Just show the key coefficients
print(f"after_switch coef: {model2.params['after_switch']:.3f}")
print(f"treatment coef: {model2.params['treatment']:.3f}")
print(f"treatment p-value: {model2.pvalues['treatment']:.4f}")

# DiD with branch fixed effects and controls
model3 = smf.ols('service_score ~ C(branch_id) + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\n\nModel 3: DiD with branch FE and controls")
print(f"treatment coef: {model3.params['treatment']:.3f}")
print(f"treatment p-value: {model3.pvalues['treatment']:.4f}")
print(f"treatment 95% CI: [{model3.conf_int().loc['treatment'][0]:.3f}, {model3.conf_int().loc['treatment'][1]:.3f}]")

# Cluster-robust standard errors by branch
model3_robust = smf.ols('service_score ~ C(branch_id) + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nModel 3 with cluster-robust SEs (by branch):")
print(f"treatment coef: {model3_robust.params['treatment']:.3f}")
print(f"treatment p-value: {model3_robust.pvalues['treatment']:.4f}")
print(f"treatment 95% CI: [{model3_robust.conf_int().loc['treatment'][0]:.3f}, {model3_robust.conf_int().loc['treatment'][1]:.3f}]")

# Check pre-trends more formally - interaction of treatment group with week dummies for pre-period
df_pre = df[df['week'] < 15].copy()
df_pre['week_centered'] = df_pre['week'] - 10  # center around middle of pre-period
model_pretrend = smf.ols('service_score ~ C(branch_id) + switched_branch * week_centered', data=df_pre).fit()
print("\n\nPre-trend test (weeks 6-14 only):")
print(f"switched_branch:week_centered coef: {model_pretrend.params['switched_branch:week_centered']:.3f}")
print(f"p-value: {model_pretrend.pvalues['switched_branch:week_centered']:.4f}")

    treatment = week_data[week_data['switched_branch']==1]['service_score'].mean()

print("-" * 50)

METHODOLOGICAL CONCERNS

The standard difference-in-differences approach requires that treatment and
control groups would have followed parallel trends in the absence of treatment.
This assumption is clearly violated here:

  Pre-period differential trend: +0.42 points/week (p < 0.001)
  
This means branches self-selected into treatment based on factors correlated
with improving performance (management quality, staff engagement, resources, etc.)
