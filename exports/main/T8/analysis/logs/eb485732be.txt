import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Simple Difference-in-Differences calculation
# Group means
means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean()
print("Group means:")
print(means)

# DiD estimate
switched_before = means.loc[(1, 0)]
switched_after = means.loc[(1, 1)]
control_before = means.loc[(0, 0)]
control_after = means.loc[(0, 1)]

did_simple = (switched_after - switched_before) - (control_after - control_before)
print(f"\nSimple DiD estimate: {did_simple:.3f}")
print(f"  Switched change: {switched_after - switched_before:.3f}")
print(f"  Control change: {control_after - control_before:.3f}")

# Now run regression with branch fixed effects
# Create interaction term
df['treated'] = df['switched_branch'] * df['after_switch']

# Regression with branch fixed effects
# service_score = beta0 + beta1*after_switch + beta2*switched_branch + beta3*(after_switch*switched_branch) + branch_FE + error
# beta3 is our DiD estimate

# Using OLS with branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
X = pd.concat([df[['after_switch', 'switched_branch', 'treated']], branch_dummies], axis=1)
y = df['service_score']

model = sm.OLS(y, sm.add_constant(X)).fit()
print("\n" + "="*60)
print("Regression with Branch Fixed Effects")
print("="*60)
print(model.summary().tables[1].head(10))

# Get the treated coefficient (DiD estimate)
did_coef = model.params['treated']
did_se = model.bse['treated']
did_pvalue = model.pvalues['treated']
print(f"\nDiD coefficient (treated): {did_coef:.3f}")
print(f"Standard error: {did_se:.3f}")
print(f"P-value: {did_pvalue:.4f}")
print(f"95% CI: [{did_coef - 1.96*did_se:.3f}, {did_coef + 1.96*did_se:.3f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Simple Difference-in-Differences calculation
means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean()
print("Group means:")
print(means)

# DiD estimate
switched_before = means.loc[(1, 0)]
switched_after = means.loc[(1, 1)]
control_before = means.loc[(0, 0)]
control_after = means.loc[(0, 1)]

did_simple = (switched_after - switched_before) - (control_after - control_before)
print(f"\nSimple DiD estimate: {did_simple:.3f}")
print(f"  Switched change: {switched_after - switched_before:.3f}")
print(f"  Control change: {control_after - control_before:.3f}")

# Regression approach - use C() for categorical variables in formula API
# This will handle fixed effects properly
df['treated'] = df['switched_branch'] * df['after_switch']

# Model with branch fixed effects using formula API
# We include after_switch, switched_branch, and their interaction
# Branch fixed effects absorb the switched_branch main effect
model = ols('service_score ~ C(branch_id) + after_switch + treated', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n" + "="*60)
print("Regression with Branch Fixed Effects")
print("(Clustered standard errors by branch)")
print("="*60)
print(model.summary())

# Get the treated coefficient (DiD estimate)
did_coef = model.params['treated']
did_se = model.bse['treated']
did_pvalue = model.pvalues['treated']
print(f"\nDiD coefficient (treated): {did_coef:.3f}")
print(f"Standard error: {did_se:.3f}")
print(f"P-value: {did_pvalue:.4f}")
print(f"95% CI: [{did_coef - 1.96*did_se:.3f}, {did_coef + 1.96*did_se:.3f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Check effect by post-switch week
df['treated'] = df['switched_branch'] * df['after_switch']

# Create week-specific treatment effects (interaction of treated with each post week)
post_weeks = [15, 16, 17, 18, 19, 20, 21, 22]
for week in post_weeks:
    df[f'treated_week{week}'] = ((df['switched_branch'] == 1) & (df['week'] == week)).astype(int)

# Run regression with week-specific effects
covariates = ['C(branch_id)', 'after_switch'] + [f'treated_week{w}' for w in post_weeks]
formula = 'service_score ~ ' + ' + '.join(covariates)
model_weekly = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Week-specific treatment effects:")
print("="*60)
for week in post_weeks:
    coef = model_weekly.params[f'treated_week{week}']
    se = model_weekly.bse[f'treated_week{week}']
    pval = model_weekly.pvalues[f'treated_week{week}']
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"Week {week}: {coef:6.2f} (SE: {se:.2f}, p={pval:.3f}) {sig}")

# Also check if there's a trend in the treatment effect
df['weeks_since_switch'] = np.maximum(0, df['week'] - 14)  # 0 before switch, 1+ after
df['treated_trend'] = df['switched_branch'] * df['weeks_since_switch']

model_trend = ols('service_score ~ C(branch_id) + after_switch + treated + treated_trend', 
                  data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n" + "="*60)
print("Treatment effect trend over time:")
print("="*60)
print(f"Initial effect (treated): {model_trend.params['treated']:.2f} (p={model_trend.pvalues['treated']:.3f})")
print(f"Weekly trend (treated_trend): {model_trend.params['treated_trend']:.3f} (p={model_trend.pvalues['treated_trend']:.3f})")
