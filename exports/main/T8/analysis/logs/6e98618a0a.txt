import pandas as pd
import numpy as np
from scipy import stats
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster

df = pd.read_csv('branches.csv')

# Run DiD regression with controls
model = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit()

# Get clustered standard errors
cov_clustered = cov_cluster(model, df['branch_id'])
clustered_se = np.sqrt(np.diag(cov_clustered))

# Get coefficient for the interaction term (the DiD estimate)
did_coef = model.params['switched_branch:after_switch']
did_se_clustered = clustereded_se[3]  # Interaction term is 4th parameter (after intercept, switched_branch, after_switch)

# Calculate t-statistic and p-value
# Degrees of freedom = number of clusters - 1 = 34 - 1 = 33 (conservative)
# Or more precisely: n_clusters - n_parameters
n_clusters = df['branch_id'].nunique()
n_params = len(model.params)
df_clustering = n_clusters - n_params

t_stat = did_coef / did_se_clustered
p_value = 2 * (1 - stats.t.cdf(abs(t_stat), df_clustering))

print(f"Difference-in-Differences Estimate: {did_coef:.2f}")
print(f"Clustered Standard Error: {did_se_clustered:.2f}")
print(f"t-statistic: {t_stat:.2f}")
print(f"Degrees of freedom (clusters - params): {df_clustering}")
print(f"p-value (two-tailed): {p_value:.3f}")

# 95% confidence interval
ci_lower = did_coef - 1.96 * did_se_clustered
ci_upper = did_coef + 1.96 * did_se_clustered
