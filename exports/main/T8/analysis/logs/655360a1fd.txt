import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create week dummies relative to switch
# Week 14 is the last pre-switch week, so we'll use it as reference
df['relative_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create indicators for each week (excluding reference week 14)
# We'll interact these with switched_branch to see differential effects by week
event_study_vars = []
for w in range(6, 23):
    if w != 14:  # Skip reference week
        df[f'week_{w}'] = (df['week'] == w).astype(int)
        event_study_vars.append(f'week_{w}')

# Build formula for event study
# We want to see the interaction of each week dummy with switched_branch
# This shows how the treatment group differs from control in each week
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'switched_branch:week_{w}' for w in range(6, 23) if w != 14]) + ' + ' + ' + '.join([f'week_{w}' for w in range(6, 23) if w != 14])

print("=== Event Study Regression ===")
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract the interaction coefficients (these are the event study estimates)
print("\nEvent Study Coefficients (Treatment vs Control by Week):")
print("(Week 14 is the reference period)")
print("-" * 60)

coeffs = []
for w in range(6, 23):
    if w == 14:
        coeffs.append((w, 0.0, 0.0, 1.0))  # Reference
    else:
        coef_name = f'switched_branch:week_{w}'
        if coef_name in event_model.params.index:
            coef = event_model.params[coef_name]
            se = event_model.bse[coef_name]
            pval = event_model.pvalues[coef_name]
            coeffs.append((w, coef, se, pval))
            marker = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
            print(f"Week {w:2d}: {coef:7.2f} (SE: {se:5.2f}, p={pval:.3f}) {marker}")
