df = pd.read_csv('branches.csv')

# Create the DiD interaction term

print("DIFFERENCE-IN-DIFFERENCES: Average Service Scores")

before_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()

# After period (after_switch=1)
after_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study regression: estimate effect for each week relative to week 14 (last pre-treatment week)
print("=" * 60)
print("EVENT STUDY REGRESSION")
print("=" * 60)

# Create week dummies interacted with treatment
df['week_14'] = (df['week'] == 14).astype(int)  # Reference category

# Create interaction terms for each week with switched_branch
for week in range(6, 23):
    if week != 14:
        df[f'week_{week}'] = (df['week'] == week).astype(int)
        df[f'did_week_{week}'] = df[f'week_{week}'] * df['switched_branch']

# Build formula
pre_weeks = [f'did_week_{w}' for w in range(6, 14)]
post_weeks = [f'did_week_{w}' for w in range(15, 23)]
week_dummies = [f'week_{w}' for w in range(6, 23) if w != 14]

formula = f"service_score ~ switched_branch + {' + '.join(week_dummies)} + {' + '.join([f'did_week_{w}' for w in range(6, 23) if w != 14])}"

event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nEvent Study Coefficients (relative to week 14):")
print("-" * 60)
print("\nPRE-TREATMENT PERIOD (should be ~0 if parallel trends hold):")
for week in range(6, 14):
    coef = event_model.params[f'did_week_{week}']
    se = event_model.bse[f'did_week_{week}']
    ci_low = coef - 1.96 * se
    ci_high = coef + 1.96 * se
    sig = "***" if event_model.pvalues[f'did_week_{week}'] < 0.01 else "**" if event_model.pvalues[f'did_week_{week}'] < 0.05 else "*" if event_model.pvalues[f'did_week_{week}'] < 0.1 else ""
    print(f"  Week {week}: {coef:6.2f} (SE: {se:.2f}) 95% CI: [{ci_low:.2f}, {ci_high:.2f}] {sig}")

print("\nPOST-TREATMENT PERIOD (treatment effect):")
for week in range(15, 23):
    coef = event_model.params[f'did_week_{week}']
    se = event_model.bse[f'did_week_{week}']
    ci_low = coef - 1.96 * se
    ci_high = coef + 1.96 * se
    sig = "***" if event_model.pvalues[f'did_week_{week}'] < 0.01 else "**" if event_model.pvalues[f'did_week_{week}'] < 0.05 else "*" if event_model.pvalues[f'did_week_{week}'] < 0.1 else ""
    print(f"  Week {week}: {coef:6.2f} (SE: {se:.2f}) 95% CI: [{ci_low:.2f}, {ci_high:.2f}] {sig}")

# Average post-treatment effect
print("\n" + "=" * 60)
print("AVERAGE POST-TREATMENT EFFECT")
print("=" * 60)
post_coefs = [event_model.params[f'did_week_{w}'] for w in range(15, 23)]
