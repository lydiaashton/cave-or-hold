import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch'] * df['after_switch']

# Difference-in-Differences regression
# service_score = beta0 + beta1*switched + beta2*after + beta3*(switched*after) + error
# beta3 is the DiD estimator - the effect of the new procedure

model = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
print("DiD Regression Results:")
print(model.summary())

# Calculate the DiD estimate manually to verify
# Group means
before_switched = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_switched = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print("\n\nGroup Means:")
print(f"Switched branches - Before: {before_switched:.2f}, After: {after_switched:.2f}, Change: {after_switched - before_switched:.2f}")
print(f"Control branches - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")
print(f"\nDiD Estimate: {(after_switched - before_switched) - (after_control - before_control):.2f}")

# Standard errors and confidence intervals
print(f"\nDiD coefficient from regression: {model.params['treated']:.4f}")
print(f"Standard error: {model.bse['treated']:.4f}")
print(f"95% CI: [{model.conf_int().loc['treated'][0]:.4f}, {model.conf_int().loc['treated'][1]:.4f}]")
print(f"P-value: {model.pvalues['treated']:.4f}")

import pandas as pd
import numpy as np

from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Add branch fixed effects
df['branch_id'] = df['branch_id'].astype(str)

# DiD with branch fixed effects
model_fe = ols('service_score ~ switched_branch + after_switch + treated + C(branch_id)', data=df).fit()
print("DiD with Branch Fixed Effects:")
print(model_fe.summary())

# Also try with robust standard errors (clustered by branch)
import statsmodels.api as sm
from statsmodels.regression.linear_model import WLS
