import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run DiD regression with standard errors
# Basic DiD model
model = ols('service_score ~ switched_branch * after_switch', data=df)
results = model.fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== DiD Regression Results ===")
print(results.summary())

# Extract key coefficient
did_coef = results.params['switched_branch:after_switch']
did_pvalue = results.pvalues['switched_branch:after_switch']
did_se = results.bse['switched_branch:after_switch']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se

print(f"\n=== Key DiD Coefficient ===")
print(f"Estimate: {did_coef:.3f}")
print(f"Std Error: {did_se:.3f}")
print(f"95% CI: [{ci_lower:.3f}, {ci_upper:.3f}]")
print(f"P-value: {did_pvalue:.4f}")

# Check parallel trends - compare pre-treatment trends
print(f"\n=== Parallel Trends Check (Pre-treatment weeks 6-14) ===")
pre_data = df[df['week'] < 15]

# Run regression on pre-treatment period only with week interaction
pre_model = ols('service_score ~ switched_branch * week', data=pre_data)
pre_results = pre_model.fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print(pre_results.summary())

# Check if the interaction is significant (it shouldn't be for parallel trends)
pre_interaction_coef = pre_results.params['switched_branch:week']
pre_interaction_pval = pre_results.pvalues['switched_branch:week']
print(f"\nPre-treatment trend difference (switched_branch:week): {pre_interaction_coef:.3f}, p={pre_interaction_pval:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study - check effect by week relative to switch
df['weeks_relative'] = df['week'] - 15  # week 15 is the switch (0), before is negative, after is positive

# Create week dummies interacted with treatment
# Reference week is week 14 (just before switch)
df['pre_reference'] = (df['week'] == 14).astype(int)

# Run event study regression
event_cols = []
for w in range(6, 23):
    if w != 14:  # skip reference week
        col = f'week_{w}'
        df[col] = (df['week'] == w).astype(int)
        df[f'{col}_treat'] = df[col] * df['switched_branch']
        event_cols.append(f'{col}_treat')

# Also include main effects for weeks and treatment
week_dummies = [f'week_{w}' for w in range(6, 23) if w != 14]
formula = 'service_score ~ switched_branch + ' + ' + '.join(week_dummies) + ' + ' + ' + '.join(event_cols)

event_model = ols(formula, data=df)
event_results = event_model.fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract treatment effects by week
print("=== Event Study Results (Treatment Effect by Week) ===")
print("Week 14 is reference (coefficient = 0)")
effects = []
for w in range(6, 23):
    if w == 14:
        effects.append((w, 0, 0, 0, 0))  # reference
    else:
        coef = event_results.params.get(f'week_{w}_treat', 0)
        se = event_results.bse.get(f'week_{w}_treat', 0)
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        effects.append((w, coef, se, ci_low, ci_high))

for w, coef, se, ci_low, ci_high in effects:
    marker = "***" if w >= 15 and ci_low > 0 else ""
    print(f"Week {w:2d}: {coef:6.2f} (SE={se:.2f}) 95%CI=[{ci_low:6.2f}, {ci_high:6.2f}] {marker}")

# Add covariates to main model
print("\n=== DiD with Covariates ===")
model_cov = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df)
results_cov = model_cov.fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(results_cov.summary())

did_cov_coef = results_cov.params['switched_branch:after_switch']
did_cov_pval = results_cov.pvalues['switched_branch:after_switch']
print(f"\nDiD with covariates: {did_cov_coef:.3f}, p={did_cov_pval:.4f}")
