import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run the DiD regression
# service_score = β0 + β1*switched_branch + β2*after_switch + β3*(switched_branch * after_switch) + ε
# β3 is the DiD estimate - the effect of the treatment

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Get the coefficient and confidence interval for the treatment effect
treatment_effect = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pvalue = model.pvalues['treatment']
ci_lower = treatment_effect - 1.96 * treatment_se
ci_upper = treatment_effect + 1.96 * treatment_se

print(f"Estimated effect: {treatment_effect:.3f}")
print(f"Standard error: {treatment_se:.3f}")
print(f"95% CI: [{ci_lower:.3f}, {ci_upper:.3f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run regression with clustered standard errors by branch
print("=== DiD Regression with Clustered Standard Errors ===")
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

# Get clustered standard errors
cov_clustered = cov_cluster(model_clustered, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))

print(f"\nCoefficients with clustered SEs:")
print(f"  Intercept: {model_clustered.params['Intercept']:.3f} (SE: {se_clustered[0]:.3f})")
print(f"  switched_branch: {model_clustered.params['switched_branch']:.3f} (SE: {se_clustered[1]:.3f})")
print(f"  after_switch: {model_clustered.params['after_switch']:.3f} (SE: {se_clustered[2]:.3f})")
print(f"  treatment: {model_clustered.params['treatment']:.3f} (SE: {se_clustered[3]:.3f})")

# Calculate t-statistic and p-value for treatment
treatment_coef = model_clustered.params['treatment']
treatment_se_clustered = se_clustered[3]
t_stat = treatment_coef / treatment_se_clustered
from scipy import stats
p_value_clustered = 2 * (1 - stats.t.cdf(abs(t_stat), df=len(df)-4))

print(f"\nTreatment effect (clustered):")
print(f"  Coefficient: {treatment_coef:.3f}")
print(f"  Clustered SE: {treatment_se_clustered:.3f}")
print(f"  t-statistic: {t_stat:.3f}")
print(f"  P-value: {p_value_clustered:.4f}")
print(f"  95% CI: [{treatment_coef - 1.96*treatment_se_clustered:.3f}, {treatment_coef + 1.96*treatment_se_clustered:.3f}]")

# Now add control variables
print("\n=== DiD Regression with Controls ===")
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
cov_clustered_controls = cov_cluster(model_controls, df['branch_id'])
se_clustered_controls = np.sqrt(np.diag(cov_clustered_controls))

print(model_controls.summary())

treatment_coef_controls = model_controls.params['treatment']
treatment_se_controls = se_clustered_controls[3]
t_stat_controls = treatment_coef_controls / treatment_se_controls
p_value_controls = 2 * (1 - stats.t.cdf(abs(t_stat_controls), df=len(df)-6))
