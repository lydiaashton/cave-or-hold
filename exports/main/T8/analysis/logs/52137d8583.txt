import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Create treatment groups
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary by group and period
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("Service Score by Group and Period:")
print(summary)
print()

# Calculate DiD manually
# Treatment group (switched_branch=1)
treatment_pre = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_post = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

# Control group (switched_branch=0)
control_pre = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_post = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print("Treatment (switched) branches:")
print(f"  Pre-switch average: {treatment_pre:.2f}")
print(f"  Post-switch average: {treatment_post:.2f}")
print(f"  Change: {treatment_post - treatment_pre:.2f}")
print()
print("Control (non-switched) branches:")
print(f"  Pre-switch average: {control_pre:.2f}")
print(f"  Post-switch average: {control_post:.2f}")
print(f"  Change: {control_post - control_pre:.2f}")
print()

# Difference-in-differences
did = (treatment_post - treatment_pre) - (control_post - control_pre)

import pandas as pd
import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create the interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch fixed effects
# Model: service_score ~ switched_branch + after_switch + treatment + branch FE
model = smf.ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df)
results = model.fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Difference-in-Differences Regression Results:")
print("=" * 60)
print(results.summary().tables[1])
print()

# Extract key coefficient
did_coef = results.params['treatment']
did_se = results.bse['treatment']
did_pvalue = results.pvalues['treatment']

print(f"DiD Coefficient: {did_coef:.3f}")
print(f"Standard Error: {did_se:.3f}")
print(f"P-value: {did_pvalue:.4f}")
print(f"95% CI: [{did_coef - 1.96*did_se:.3f}, {did_coef + 1.96*did_se:.3f}]")
