import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
branches = df.groupby('branch_id')['switched_branch'].first()
print("=== Branch Assignment ===")
print(f"Total branches: {len(branches)}")
print(f"Switched branches: {(branches == 1).sum()}")
print(f"Non-switched branches: {(branches == 0).sum()}")

# Check balance - pre-switch service scores by group
print("\n=== Pre-Switch Service Scores (Weeks 6-14) ===")
pre_switch = df[df['after_switch'] == 0]
pre_by_group = pre_switch.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print(pre_by_group)

# Post-switch service scores by group
print("\n=== Post-Switch Service Scores (Weeks 15-22) ===")
post_switch = df[df['after_switch'] == 1]
post_by_group = post_switch.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print(post_by_group)

# Simple difference-in-differences calculation
print("\n=== Simple DiD Calculation ===")
# Treatment group (switched=1)
treatment_pre = pre_switch[pre_switch['switched_branch'] == 1]['service_score'].mean()
treatment_post = post_switch[post_switch['switched_branch'] == 1]['service_score'].mean()
# Control group (switched=0)
control_pre = pre_switch[pre_switch['switched_branch'] == 0]['service_score'].mean()
control_post = post_switch[post_switch['switched_branch'] == 0]['service_score'].mean()

print(f"Treatment group - Pre: {treatment_pre:.2f}, Post: {treatment_post:.2f}, Change: {treatment_post - treatment_pre:.2f}")
print(f"Control group - Pre: {control_pre:.2f}, Post: {control_post:.2f}, Change: {control_post - control_pre:.2f}")
did_effect = (treatment_post - treatment_pre) - (control_post - control_pre)
print(f"\nDifference-in-Differences Estimate: {did_effect:.2f}")

from statsmodels.formula.api import ols

# Basic DiD model: service_score ~ switched_branch * after_switch
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

print("\n=== DiD with Branch Fixed Effects ===")

model_fe = ols('service_score ~ switched_branch * after_switch + C(branch_id_str)', data=df).fit()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from linearmodels.panel import PanelOLS

df = pd.read_csv('branches.csv')

# Set up panel data
df = df.set_index(['branch_id', 'week'])

# Run PanelOLS with entity (branch) fixed effects and clustered standard errors
# The key is that switched_branch is time-invariant, so we can't include it directly
# But we can include the interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model with branch fixed effects
mod = PanelOLS(df['service_score'], 
               df[['treatment', 'staffing_index', 'visits_index']], 
               entity_effects=True)
result = mod.fit(cov_type='clustered', cluster_entity=True)
print("=== PanelOLS with Branch Fixed Effects (Clustered SEs) ===")
print(result)

# Also run without controls for comparison
mod_simple = PanelOLS(df['service_score'], 
                      df[['treatment']], 
                      entity_effects=True)
result_simple = mod_simple.fit(cov_type='clustered', cluster_entity=True)
print("\n=== PanelOLS Simple (Clustered SEs) ===")
print(result_simple)

import pandas as pd
import numpy as np
from linearmodels.panel import PanelOLS

df = pd.read_csv('branches.csv')

# Event study: create week dummies interacted with treatment
# Reference week = 14 (last pre-switch week)
df = df.copy()
df['week_cat'] = df['week'].astype(str)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Create relative week indicators (weeks 6-14 are pre, 15-22 are post)
# Let's do week-by-week effects relative to week 14
df_wide = df.pivot_table(index=['branch_id', 'switched_branch'], 
                         columns='week', 
                         values='service_score').reset_index()

# Calculate average effect in post period for each branch
post_weeks = [15, 16, 17, 18, 19, 20, 21, 22]
pre_weeks = [6, 7, 8, 9, 10, 11, 12, 13, 14]

df_wide['pre_avg'] = df_wide[pre_weeks].mean(axis=1)
df_wide['post_avg'] = df_wide[post_weeks].mean(axis=1)
df_wide['change'] = df_wide['post_avg'] - df_wide['pre_avg']

# Compare changes between treatment and control
treatment_change = df_wide[df_wide['switched_branch'] == 1]['change']
control_change = df_wide[df_wide['switched_branch'] == 0]['change']

print("=== Branch-Level Changes (Post - Pre) ===")
print(f"Treatment branches (n={len(treatment_change)}):")
print(f"  Mean change: {treatment_change.mean():.2f}")
print(f"  Std dev: {treatment_change.std():.2f}")
print(f"\nControl branches (n={len(control_change)}):")
print(f"  Mean change: {control_change.mean():.2f}")
print(f"  Std dev: {control_change.std():.2f}")

# Simple t-test on branch-level changes
from scipy import stats
t_stat, p_val = stats.ttest_ind(treatment_change, control_change)
print(f"\n=== Two-sample t-test on branch-level changes ===")
print(f"T-statistic: {t_stat:.3f}")
print(f"P-value: {p_val:.3f}")
print(f"DiD estimate (treatment - control): {treatment_change.mean() - control_change.mean():.2f}")

print("\n=== Robustness Check: Branch-level DiD Regression ===")
branch_level = df_wide[['switched_branch', 'change']].copy()
from statsmodels.formula.api import ols
branch_model = ols('change ~ switched_branch', data=branch_level).fit()
print(branch_model.summary())
