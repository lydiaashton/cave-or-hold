import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Create a treatment group indicator (interaction of switched_branch and after_switch)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary by group and period
print("=== Service Score Summary by Group and Period ===")
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Calculate the Difference-in-Differences manually
# Group means
mean_control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
mean_control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
mean_treated_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
mean_treated_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n=== Group Means ===")
print(f"Control (non-switched) Before: {mean_control_before:.2f}")
print(f"Control (non-switched) After: {mean_control_after:.2f}")
print(f"Treated (switched) Before: {mean_treated_before:.2f}")
print(f"Treated (switched) After: {mean_treated_after:.2f}")

# Differences
control_diff = mean_control_after - mean_control_before
treated_diff = mean_treated_after - mean_treated_before
did_estimate = treated_diff - control_diff

print(f"\n=== Difference-in-Differences ===")
print(f"Control change (after - before): {control_diff:.2f}")
print(f"Treated change (after - before): {treated_diff:.2f}")
print(f"DiD Estimate (treated change - control change): {did_estimate:.2f}")

# Check pre-trends: compare trends before week 15
print("\n=== Pre-Trend Check (Weeks 6-14) ===")
pre_trends = df[df['week'] < 15].groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
print(pre_trends)
