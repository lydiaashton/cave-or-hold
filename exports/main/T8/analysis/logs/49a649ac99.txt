import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimator - the effect of the new procedure

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Difference-in-Differences Regression Results:")
print("=" * 60)
print(model.summary())

# Get the DiD estimate and confidence interval
did_estimate = model.params['treatment']
did_pvalue = model.pvalues['treatment']
did_ci = model.conf_int().loc['treatment']

print(f"\nDiD Estimate: {did_estimate:.3f}")
print(f"95% CI: [{did_ci[0]:.3f}, {did_ci[1]:.3f}]")
print(f"P-value: {did_pvalue:.4f}")
