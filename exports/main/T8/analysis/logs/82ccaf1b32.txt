import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']

# === Proper Event Study ===
# Create relative week indicators (week 14 is the reference)
df['relative_week'] = df['week'] - 14  # -8 to -1 pre, 1 to 8 post

# Create indicators for each relative week, interacted with treatment
for rw in range(-8, 0):  # Pre-period (reference will be -1)
    df[f'pre_{rw}'] = ((df['relative_week'] == rw) & (df['switched_branch']==1)).astype(int)
for rw in range(1, 9):  # Post-period
    df[f'post_{rw}'] = ((df['relative_week'] == rw) & (df['switched_branch']==1)).astype(int)

# Run event study regression (excluding pre_-1 as reference)
pre_cols = [f'pre_{rw}' for rw in range(-8, -1)]
post_cols = [f'post_{rw}' for rw in range(1, 9)]
formula = f"service_score ~ switched_branch + after_switch + {' + '.join(pre_cols + post_cols)} + staffing_index + visits_index"

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Proper Event Study (Relative to Week 14) ===")
print("\nPre-period coefficients (should be ~0 if parallel trends hold):")
for rw in range(-8, -1):
    coef = model_event.params[f'pre_{rw}']
    se = model_event.bse[f'pre_{rw}']
    ci_low = coef - 1.96*se
    ci_high = coef + 1.96*se
    sig = "***" if model_event.pvalues[f'pre_{rw}'] < 0.01 else "**" if model_event.pvalues[f'pre_{rw}'] < 0.05 else "*" if model_event.pvalues[f'pre_{rw}'] < 0.1 else ""
    print(f"  Week {14+rw} (rel {rw}): {coef:6.2f} (95% CI: [{ci_low:6.2f}, {ci_high:6.2f}]) {sig}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']

# === Proper Event Study ===
# Create relative week indicators (week 14 is the reference)
df['relative_week'] = df['week'] - 14  # -8 to -1 pre, 1 to 8 post

# Create indicators for each relative week, interacted with treatment
# Use 'm' prefix for negative weeks to avoid formula issues
for rw in range(-8, 0):  # Pre-period
    df[f'pre_m{abs(rw)}'] = ((df['relative_week'] == rw) & (df['switched_branch']==1)).astype(int)
for rw in range(1, 9):  # Post-period
    df[f'post_{rw}'] = ((df['relative_week'] == rw) & (df['switched_branch']==1)).astype(int)

# Run event study regression (excluding pre_m1 as reference)
pre_cols = [f'pre_m{rw}' for rw in range(8, 1, -1)]  # pre_m8 to pre_m2
post_cols = [f'post_{rw}' for rw in range(1, 9)]
formula = f"service_score ~ switched_branch + after_switch + {' + '.join(pre_cols + post_cols)} + staffing_index + visits_index"

print("Formula columns:", pre_cols + post_cols)

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n=== Proper Event Study (Relative to Week 14) ===")
print("\nPre-period coefficients (should be ~0 if parallel trends hold):")
for rw in range(-8, -1):
    col = f'pre_m{abs(rw)}'
    coef = model_event.params[col]
    se = model_event.bse[col]
    ci_low = coef - 1.96*se
    ci_high = coef + 1.96*se
    sig = "***" if model_event.pvalues[col] < 0.01 else "**" if model_event.pvalues[col] < 0.05 else "*" if model_event.pvalues[col] < 0.1 else ""
    print(f"  Week {14+rw} (rel {rw}): {coef:6.2f} (95% CI: [{ci_low:6.2f}, {ci_high:6.2f}]) {sig}")

print("\nPost-period coefficients (treatment effects):")
post_effects = []
for rw in range(1, 9):
    col = f'post_{rw}'
    coef = model_event.params[col]
    se = model_event.bse[col]
    ci_low = coef - 1.96*se
    ci_high = coef + 1.96*se
    sig = "***" if model_event.pvalues[col] < 0.01 else "**" if model_event.pvalues[col] < 0.05 else "*" if model_event.pvalues[col] < 0.1 else ""
    print(f"  Week {14+rw} (rel {rw}): {coef:6.2f} (95% CI: [{ci_low:6.2f}, {ci_high:6.2f}]) {sig}")
    post_effects.append(coef)

# === Plot Event Study ===
fig, ax = plt.subplots(figsize=(12, 6))

# Pre-period
pre_weeks = list(range(-8, -1))
pre_coefs = [model_event.params[f'pre_m{abs(rw)}'] for rw in pre_weeks]
pre_ci_low = [model_event.params[f'pre_m{abs(rw)}'] - 1.96*model_event.bse[f'pre_m{abs(rw)}'] for rw in pre_weeks]
pre_ci_high = [model_event.params[f'pre_m{abs(rw)}'] + 1.96*model_event.bse[f'pre_m{abs(rw)}'] for rw in pre_weeks]

post_weeks = list(range(1, 9))
