import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run DiD regression with cluster-robust standard errors
# Basic DiD model
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("Basic DiD Regression (heteroskedasticity-robust SE):")
print(model.get_robustcov_results(cov_type='HC1').summary())

# Now with branch fixed effects and clustering by branch
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = df.copy()
for col in branch_dummies.columns:
    df_with_fe[col] = branch_dummies[col]

# Build formula with branch FE
fe_cols = '+'.join(branch_dummies.columns)
formula = f'service_score ~ switched_branch * after_switch + {fe_cols}'
model_fe = ols(formula, data=df_with_fe).fit()

# Cluster-robust SE by branch
print("\n\nDiD Regression with Branch Fixed Effects (clustered by branch):")
print(model_fe.get_robustcov_results(cov_type='cluster', cov_kwds={'groups': df_with_fe['branch_id']}).summary())

import pandas as pd

from linearmodels.panel import PanelOLS

# Re-run the main DiD model to verify

df_panel = df_panel.set_index(['branch_id', 'week'])
df_panel['treatment'] = df_panel['switched_branch'] * df_panel['after_switch']

mod = PanelOLS(df_panel['service_score'], 
               df_panel[['treatment']], 
               entity_effects=True, time_effects=True)
res = mod.fit(cov_type='clustered', cluster_entity=True)

print("=== MAIN DiD RESULTS ===")
coef = float(res.params['treatment'])
se = float(res.std_errors['treatment'])
pval = float(res.pvalues['treatment'])
ci_lower = coef - 1.96 * se
ci_upper = coef + 1.96 * se

print(f"Coefficient: {coef:.3f}")
print(f"Std Error: {se:.3f}")
print(f"T-statistic: {coef/se:.3f}")
print(f"P-value: {pval:.4f}")
print(f"95% CI: [{ci_lower:.3f}, {ci_upper:.3f}]")

# What does a 2-point change mean practically?
print(f"\n=== PRACTICAL SIGNIFICANCE ===")
print(f"Overall mean service score: {df['service_score'].mean():.1f}")
print(f"Overall std dev: {df['service_score'].std():.1f}")
print(f"Effect size as % of mean: {coef / df['service_score'].mean() * 100:.1f}%")
print(f"Effect size in std dev units (Cohen's d): {coef / df['service_score'].std():.2f}")

# Dynamic effects
df_panel2 = df.copy()
df_panel2 = df_panel2.set_index(['branch_id', 'week'])
df_panel2['week_col'] = df_panel2.index.get_level_values('week')

for w in range(15, 23):
    df_panel2[f'post_week_{w}'] = (df_panel2['switched_branch'] == 1) & (df_panel2['week_col'] == w)

post_cols = [f'post_week_{w}' for w in range(15, 23)]
mod_dynamic = PanelOLS(df_panel2['service_score'], 
                       df_panel2[post_cols], 
                       entity_effects=True, time_effects=True)
res_dynamic = mod_dynamic.fit(cov_type='clustered', cluster_entity=True)

print(f"\n=== WEEKLY EFFECTS ===")
positive_weeks = 0
sig_weeks = 0
for w in range(15, 23):
    w_coef = float(res_dynamic.params[f'post_week_{w}'])
    w_pval = float(res_dynamic.pvalues[f'post_week_{w}'])
    if w_coef > 0:
        positive_weeks += 1
    if w_pval < 0.05:
        sig_weeks += 1
