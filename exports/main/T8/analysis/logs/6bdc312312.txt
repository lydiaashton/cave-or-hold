import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Calculate weekly averages by group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
weekly_avg.columns = ['switched_branch', 'week', 'avg_score']

# Pivot for easier plotting
weekly_pivot = weekly_avg.pivot(index='week', columns='switched_branch', values='avg_score')
weekly_pivot.columns = ['Control (No Switch)', 'Treatment (Switched)']

print("Weekly averages by group:")
print(weekly_pivot)

# Calculate pre-switch trends (weeks 6-14)
pre_switch = weekly_pivot.loc[6:14]
print("\nPre-switch period (weeks 6-14):")
print(pre_switch)

# Simple trend analysis for pre-period
from scipy import stats

control_pre = pre_switch['Control (No Switch)'].values
treatment_pre = pre_switch['Treatment (Switched)'].values
weeks_pre = np.arange(6, 15)

# Linear trend for each group
control_slope, control_intercept, control_r, control_p, control_se = stats.linregress(weeks_pre, control_pre)
treatment_slope, treatment_intercept, treatment_r, treatment_p, treatment_se = stats.linregress(weeks_pre, treatment_pre)

print(f"\nPre-switch trend - Control: slope={control_slope:.3f}, p={control_p:.3f}")
print(f"Pre-switch trend - Treatment: slope={treatment_slope:.3f}, p={treatment_p:.3f}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create a visualization
fig, ax = plt.subplots(figsize=(12, 6))

# Weekly averages by group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Plot
for group, label in [(0, 'Control (No Switch)'), (1, 'Treatment (Switched)')]:
    subset = weekly_avg[weekly_avg['switched_branch'] == group]
    ax.plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2, markersize=6)

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch occurred')
ax.axvspan(14.5, 22, alpha=0.1, color='red', label='Post-switch period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Score Trends: Control vs Treatment Groups', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=10)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("Chart saved as 'service_score_trends.png'")

# Branch fixed effects model (more robust)
# This controls for time-invariant branch characteristics
df['treatment'] = df['switched_branch'] * df['after_switch']
fe_model = ols('service_score ~ treatment + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Branch Fixed Effects Model ===")
print(fe_model.summary())

# Extract just the treatment coefficient
treatment_coef = fe_model.params['treatment']
treatment_se = fe_model.bse['treatment']
treatment_p = fe_model.pvalues['treatment']
print(f"\nTreatment effect: {treatment_coef:.3f} (SE: {treatment_se:.3f}, p-value: {treatment_p:.4f})")
print(f"95% CI: [{treatment_coef - 1.96*treatment_se:.3f}, {treatment_coef + 1.96*treatment_se:.3f}]")
