import pandas as pd
import numpy as np
import statsmodels.api as sm
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch'] * df['after_switch']

# Difference-in-Differences Regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + epsilon
# beta3 is the DiD estimator - the effect of the treatment

X = df[['switched_branch', 'after_switch', 'treated']]
X = sm.add_constant(X)
y = df['service_score']

model = sm.OLS(y, X).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Calculate group means for key periods
print("\n=== Group Means ===")
pre_treatment = df[df['after_switch'] == 0]
post_treatment = df[df['after_switch'] == 1]

print("\nPre-treatment (weeks 6-14):")
print(pre_treatment.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

print("\nPost-treatment (weeks 15-22):")
print(post_treatment.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count']))

# Calculate the DiD manually
control_pre = pre_treatment[pre_treatment['switched_branch'] == 0]['service_score'].mean()
control_post = post_treatment[post_treatment['switched_branch'] == 0]['service_score'].mean()
treated_pre = pre_treatment[pre_treatment['switched_branch'] == 1]['service_score'].mean()
treated_post = post_treatment[post_treatment['switched_branch'] == 1]['service_score'].mean()

control_change = control_post - control_pre
treated_change = treated_post - treated_pre
did_estimate = treated_change - control_change

print(f"\n=== Manual DiD Calculation ===")
print(f"Control group (non-switched): Pre={control_pre:.2f}, Post={control_post:.2f}, Change={control_change:.2f}")
print(f"Treatment group (switched): Pre={treated_pre:.2f}, Post={treated_post:.2f}, Change={treated_change:.2f}")
print(f"DiD Estimate (Treatment Effect): {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# DiD with control variables
print("=== DiD with Control Variables ===")
X = df[['switched_branch', 'after_switch', 'treated', 'staffing_index', 'visits_index']]
X = sm.add_constant(X)
y = df['service_score']

model_controls = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_controls.summary())

# Branch fixed effects model
print("\n=== DiD with Branch Fixed Effects ===")
# Create dummy variables for each branch
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
X_fe = pd.concat([df[['switched_branch', 'after_switch', 'treated', 'staffing_index', 'visits_index']], branch_dummies], axis=1)
X_fe = sm.add_constant(X_fe, has_constant='add')
y = df['service_score']

model_fe = sm.OLS(y, X_fe).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_fe.summary())
print(f"\nTreated coefficient with branch FE: {model_fe.params['treated']:.3f}, p-value: {model_fe.pvalues['treated']:.3f}")

# Event study - effects by week relative to treatment
print("\n=== Event Study Analysis ===")
df['relative_week'] = df['week'] - 15  # Week 15 is the treatment week (0)

# Create week dummies interacted with treatment
event_study_df = df.copy()
for w in range(-9, 8):  # Weeks -9 to +7 relative to treatment
    event_study_df[f'week_{w}'] = ((event_study_df['relative_week'] == w) & (event_study_df['switched_branch'] == 1)).astype(int)

# Use week -1 as reference (just before treatment)
event_cols = [f'week_{w}' for w in range(-9, 8) if w != -1]
X_event = event_study_df[event_cols + ['switched_branch', 'staffing_index', 'visits_index']]
X_event = sm.add_constant(X_event)
y_event = event_study_df['service_score']

model_event = sm.OLS(y_event, X_event).fit(cov_type='cluster', cov_kwds={'groups': event_study_df['branch_id']})

# Extract event study coefficients
event_coefs = []
event_pvals = []
event_weeks = []
for w in range(-9, 8):
    if w == -1:
        event_coefs.append(0)  # Reference period
        event_pvals.append(np.nan)
    else:
        coef = model_event.params.get(f'week_{w}', 0)
        pval = model_event.pvalues.get(f'week_{w}', np.nan)
        event_coefs.append(coef)
        event_pvals.append(pval)
    event_weeks.append(w)

event_results = pd.DataFrame({
    'relative_week': event_weeks,
    'coefficient': event_coefs,
    'p_value': event_pvals
})
print(event_results.to_string(index=False))

# Plot event study
plt.figure(figsize=(12, 6))
plt.axhline(y=0, color='black', linestyle='-', linewidth=1)
plt.axvline(x=-0.5, color='red', linestyle='--', label='Treatment Start')

plt.bar(event_results['relative_week'], event_results['coefficient'], 
        color=['gray' if w < 0 else 'steelblue' for w in event_results['relative_week']],
        alpha=0.7, label='Coefficient')

# Add confidence intervals
for i, row in event_results.iterrows():
    if not np.isnan(row['p_value']):
        se = abs(row['coefficient']) / abs(sm.stats.ztest([row['coefficient']], alternative='two-sided')[0]) if row['coefficient'] != 0 else 0.5
        ci_low = row['coefficient'] - 1.96 * se
        ci_high = row['coefficient'] + 1.96 * se
        plt.plot([row['relative_week'], row['relative_week']], [ci_low, ci_high], 'k-', linewidth=2)

plt.xlabel('Weeks Relative to Treatment (0 = Week 15)')
plt.ylabel('Effect on Service Score')
plt.title('Event Study: Treatment Effect Over Time')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(-9, 8))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")

import pandas as pd
import numpy as np
import statsmodels.api as sm

# Load the data
df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# Main DiD model
X = df[['switched_branch', 'after_switch', 'treated']]
X = sm.add_constant(X)
y = df['service_score']

model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Main DiD Results ===")
print(f"Treatment effect (treated coefficient): {model.params['treated']:.2f}")
print(f"Standard error: {model.bse['treated']:.2f}")
print(f"t-statistic: {model.tvalues['treated']:.2f}")
print(f"p-value: {model.pvalues['treated']:.3f}")

# 95% confidence interval
ci_low = model.params['treated'] - 1.96 * model.bse['treated']
ci_high = model.params['treated'] + 1.96 * model.bse['treated']
print(f"95% Confidence Interval: [{ci_low:.2f}, {ci_high:.2f}]")

# What's the probability the effect is positive?
# Using t-distribution
from scipy import stats
t_stat = model.tvalues['treated']
df_degrees = model.df_resid
prob_positive = 1 - stats.t.cdf(0, df_degrees, loc=model.params['treated'], scale=model.bse['treated'])
print(f"\nProbability effect is positive: {prob_positive*100:.1f}%")

# Power analysis - what effect size could we detect?
print("\n=== Power Considerations ===")
n_branches = 34
n_post_weeks = 8
n_treated = 20
n_control = 14

# Rough power calculation
print(f"Treated branches: {n_treated}")
print(f"Control branches: {n_control}")
print(f"Post-treatment weeks: {n_post_weeks}")
print(f"\nWith this sample size, we have limited power to detect small effects.")
print("A non-significant result does NOT mean there is no effect.")

# Look at the event study more carefully
print("\n=== Event Study - Post-Treatment Weeks Only ===")
df['relative_week'] = df['week'] - 15

# Create cleaner event study
event_df = df.copy()
post_weeks = [0, 1, 2, 3, 4, 5, 6, 7]
for w in post_weeks:
    event_df[f'post_week_{w}'] = ((event_df['relative_week'] == w) & (event_df['switched_branch'] == 1)).astype(int)

# Also include pre-treatment average as control
event_cols = [f'post_week_{w}' for w in post_weeks]
X_event = event_df[event_cols + ['switched_branch']]
X_event = sm.add_constant(X_event)
y_event = event_df['service_score']

model_event = sm.OLS(y_event, X_event).fit(cov_type='cluster', cov_kwds={'groups': event_df['branch_id']})

print("Post-treatment week effects (vs. pre-treatment baseline):")
for w in post_weeks:
    coef = model_event.params[f'post_week_{w}']
    pval = model_event.pvalues[f'post_week_{w}']
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.10 else ""
    print(f"  Week {15+w}: {coef:+.2f} (p={pval:.3f}) {sig}")

# Average post-treatment effect
print("\n=== Average Post-Treatment Effect ===")
post_coefs = [model_event.params[f'post_week_{w}'] for w in post_weeks]
avg_effect = np.mean(post_coefs)
print(f"Average effect across all post-treatment weeks: {avg_effect:.2f} points")

# First 4 weeks vs later weeks
first_4 = np.mean([model_event.params[f'post_week_{w}'] for w in [0,1,2,3]])
later = np.mean([model_event.params[f'post_week_{w}'] for w in [4,5,6,7]])
print(f"Average effect weeks 1-4: {first_4:.2f} points")
print(f"Average effect weeks 5-8: {later:.2f} points")
