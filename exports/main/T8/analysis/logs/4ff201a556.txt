import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create treatment group indicator
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treated'] * df['post']  # Difference-in-differences interaction term

# Calculate mean service scores by group and period
summary = df.groupby(['treated', 'post'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("Service scores by group and period:")
print(summary)

# Calculate the DiD estimate manually
# Control group (treated=0): before vs after
control_before = df[(df['treated']==0) & (df['post']==0)]['service_score'].mean()
control_after = df[(df['treated']==0) & (df['post']==1)]['service_score'].mean()
control_change = control_after - control_before

# Treatment group (treated=1): before vs after
treat_before = df[(df['treated']==1) & (df['post']==0)]['service_score'].mean()
treat_after = df[(df['treated']==1) & (df['post']==1)]['service_score'].mean()
treat_change = treat_after - treat_before

# DiD estimate
did_estimate = treat_change - control_change

print(f"\nControl group (non-switched):")
print(f"  Before (weeks 6-14): {control_before:.2f}")
print(f"  After (weeks 15-22): {control_after:.2f}")
print(f"  Change: {control_change:.2f}")

print(f"\nTreatment group (switched):")
print(f"  Before (weeks 6-14): {treat_before:.2f}")
print(f"  After (weeks 15-22): {treat_after:.2f}")
print(f"  Change: {treat_change:.2f}")

print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run DiD regression with standard errors
# Basic DiD model
model1 = ols('service_score ~ C(switched_branch) * C(after_switch)', data=df).fit()
print("Basic DiD Regression:")
print(model1.summary().tables[1])

# DiD with branch fixed effects
model2 = ols('service_score ~ C(switched_branch) * C(after_switch) + C(branch_id)', data=df).fit()
print("\nDiD with Branch Fixed Effects:")
print(model2.summary().tables[1])

# DiD with branch fixed effects and control variables
model3 = ols('service_score ~ C(switched_branch) * C(after_switch) + C(branch_id) + staffing_index + visits_index', data=df).fit()
print("\nDiD with Branch FE and Controls:")
print(model3.summary().tables[1])

# Cluster-robust standard errors at branch level
model3_clustered = model3.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print("\nDiD with Clustered SE (at branch level):")
print(model3_clustered.summary().tables[1])
