import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch'] * df['after_switch']

# Summary table: mean service scores by group and period
print("=== Service Score Summary ===")
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)
print()

# Calculate DiD manually
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
before_treated = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_treated = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print("=== Difference-in-Differences Calculation ===")
print(f"Control group (no switch): Before={before_control:.2f}, After={after_control:.2f}, Change={after_control-before_control:.2f}")
print(f"Treated group (switched): Before={before_treated:.2f}, After={after_treated:.2f}, Change={after_treated-before_treated:.2f}")
did = (after_treated - before_treated) - (after_control - before_control)
print(f"Difference-in-Differences estimate: {did:.2f}")
print()

# Run regression with robust standard errors
# DiD regression: service_score ~ switched_branch + after_switch + treated
model = ols('service_score ~ C(switched_branch) + C(after_switch) + C(treated)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== DiD Regression Results ===")
print(model.summary())
print()

# Check how many branches switched vs didn't
print("=== Branch Counts ===")
print(df.groupby('switched_branch')['branch_id'].nunique())
print()

# Check balance - are there pre-existing differences?
print("=== Pre-switch comparison (weeks 6-14) ===")
pre_data = df[df['after_switch']==0]
pre_comparison = pre_data.groupby('switched_branch')[['service_score', 'staffing_index', 'visits_index']].mean()
print(pre_comparison)
