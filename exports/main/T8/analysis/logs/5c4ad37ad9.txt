import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# Check if there's heterogeneity - do some branches benefit more than others?
# Calculate branch-level before/after differences
branch_stats = df.groupby(['branch_id', 'switched_branch']).agg({
    'service_score': ['mean', 'std', 'count']
}).reset_index()
branch_stats.columns = ['branch_id', 'switched_branch', 'mean_score', 'std_score', 'n_obs']

# Get before and after for each branch
before_branch = df[df['after_switch'] == 0].groupby('branch_id')['service_score'].mean()
after_branch = df[df['after_switch'] == 1].groupby('branch_id')['service_score'].mean()

branch_changes = pd.DataFrame({
    'branch_id': before_branch.index,
    'before': before_branch.values,
    'after': after_branch.values
})
branch_changes['change'] = branch_changes['after'] - branch_changes['before']
branch_changes['switched'] = df.groupby('branch_id')['switched_branch'].first().values

print("=== Branch-Level Changes ===")
print("\nSwitched branches (n=20):")
switched_changes = branch_changes[branch_changes['switched'] == 1]['change']
print(f"  Mean change: {switched_changes.mean():.2f}")
print(f"  Std dev: {switched_changes.std():.2f}")
print(f"  Min: {switched_changes.min():.2f}, Max: {switched_changes.max():.2f}")
print(f"  Branches with positive change: {(switched_changes > 0).sum()} / {len(switched_changes)}")

print("\nControl branches (n=14):")
control_changes = branch_changes[branch_changes['switched'] == 0]['change']
print(f"  Mean change: {control_changes.mean():.2f}")
print(f"  Std dev: {control_changes.std():.2f}")
print(f"  Min: {control_changes.min():.2f}, Max: {control_changes.max():.2f}")
print(f"  Branches with positive change: {(control_changes > 0).sum()} / {len(control_changes)}")

# T-test on branch-level changes
t_stat, p_val = stats.ttest_ind(switched_changes, control_changes)
print(f"\n=== Two-sample t-test on branch-level changes ===")
print(f"t-statistic: {t_stat:.3f}, p-value: {p_val:.3f}")

# Bootstrap for confidence interval
np.random.seed(42)
bootstrap_effects = []
for i in range(10000):
    # Resample branches with replacement
    switched_sample = switched_changes.sample(n=len(switched_changes), replace=True).mean()
    control_sample = control_changes.sample(n=len(control_changes), replace=True).mean()
    bootstrap_effects.append(switched_sample - control_sample)

bootstrap_effects = np.array(bootstrap_effects)
ci_lower = np.percentile(bootstrap_effects, 2.5)
ci_upper = np.percentile(bootstrap_effects, 97.5)
print(f"\n=== Bootstrap 95% CI (10,000 iterations) ===")
print(f"Mean effect: {bootstrap_effects.mean():.2f}")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"Probability effect > 0: {(bootstrap_effects > 0).mean():.1%}")

import pandas as pd

df = pd.read_csv('branches.csv')

# Calculate effect size in percentage terms
baseline = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
effect = 2.04  # from our DiD estimate
pct_improvement = (effect / baseline) * 100

print(f"Baseline service score (switched branches, pre-period): {baseline:.2f}")
print(f"Estimated treatment effect: {effect:.2f} points")
print(f"Percentage improvement: {pct_improvement:.1f}%")

# Power analysis - what sample size would we need for significance?
import numpy as np
from statsmodels.stats.power import TTestIndPower

# Using branch-level data
switched_changes = df[df['switched_branch'] == 1].groupby('branch_id').apply(
    lambda x: x[x['after_switch'] == 1]['service_score'].mean() - x[x['after_switch'] == 0]['service_score'].mean()
)
control_changes = df[df['switched_branch'] == 0].groupby('branch_id').apply(
    lambda x: x[x['after_switch'] == 1]['service_score'].mean() - x[x['after_switch'] == 0]['service_score'].mean()
)

pooled_std = np.sqrt((switched_changes.var() + control_changes.var()) / 2)
effect_size_cohen = (switched_changes.mean() - control_changes.mean()) / pooled_std

print(f"\nCohen's d (effect size): {effect_size_cohen:.3f}")

power_analysis = TTestIndPower()
# Current sample size
n_current = len(switched_changes)
power_current = power_analysis.power(effect_size=effect_size_cohen, nobs1=n_current, ratio=len(control_changes)/len(switched_changes), alpha=0.05)
print(f"Current statistical power (at alpha=0.05): {power_current:.1%}")

# Sample size needed for 80% power
n_needed = power_analysis.solve_power(effect_size=effect_size_cohen, power=0.8, ratio=len(control_changes)/len(switched_changes), alpha=0.05)
print(f"Branches per group needed for 80% power: {n_needed:.0f}")
