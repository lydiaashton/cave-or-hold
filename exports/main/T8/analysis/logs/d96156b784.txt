import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Check pre-trends: average service score by group and period
print("=== Pre-Trends Check ===")
pre_trends = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(pre_trends)

print("\n=== Average Service Scores by Group and Period ===")
pivot_table = df.pivot_table(values='service_score', 
                              index='switched_branch', 
                              columns='after_switch', 
                              aggfunc='mean')
print(pivot_table)

# Calculate the DiD manually
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n=== Manual DiD Calculation ===")
print(f"Control (non-switched) before: {control_before:.3f}")
print(f"Control (non-switched) after: {control_after:.3f}")
print(f"Control change: {control_after - control_before:.3f}")
print(f"\nTreatment (switched) before: {treatment_before:.3f}")
print(f"Treatment (switched) after: {treatment_after:.3f}")
print(f"Treatment change: {treatment_after - treatment_before:.3f}")

did_estimate = (treatment_after - treatment_before) - (control_after - control_before)
print(f"\nDiD Estimate: {did_estimate:.3f}")

# Run regression with standard errors
print("\n=== Regression Analysis ===")
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model.summary())

# Also run with staffing and visits as controls
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Regression with Controls ===")
print(model_controls.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main regression with cluster-robust standard errors
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract treatment effect
treatment_effect = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pvalue = model.pvalues['treatment']

# 95% confidence interval
ci_lower = treatment_effect - 1.96 * treatment_se
ci_upper = treatment_effect + 1.96 * treatment_se

print("=== Treatment Effect Summary ===")
print(f"Treatment Effect (DiD Estimate): {treatment_effect:.3f}")
print(f"Standard Error: {treatment_se:.3f}")
print(f"95% Confidence Interval: [{ci_lower:.3f}, {ci_upper:.3f}]")
print(f"P-value: {treatment_pvalue:.6f}")
print(f"Statistically Significant: {'Yes' if treatment_pvalue < 0.05 else 'No'}")

# Calculate effect size relative to baseline
baseline_score = df[df['after_switch'] == 0]['service_score'].mean()
effect_size_pct = (treatment_effect / baseline_score) * 100
print(f"\nBaseline Average Score: {baseline_score:.3f}")
print(f"Effect Size (percentage of baseline): {effect_size_pct:.2f}%")

# Number of branches
n_branches = df['branch_id'].nunique()
n_treatment = df[df['switched_branch'] == 1]['branch_id'].nunique()
n_control = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"\nTotal Branches: {n_branches}")
print(f"Treatment Branches: {n_treatment}")
print(f"Control Branches: {n_control}")

# Check post-period effects by week
post_period = df[df['week'] >= 15]
post_weekly = post_period.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
post_weekly['diff'] = post_weekly[1] - post_weekly[0]
print("\n=== Post-Switch Weekly Differences (Treatment - Control) ===")
print(post_weekly[['diff']])
print(f"\nAverage post-switch difference: {post_weekly['diff'].mean():.3f}")
