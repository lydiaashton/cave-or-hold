import pandas as pd

# Average service scores by group and period
print("\n=== Average Service Scores ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the difference-in-differences manually
# Before period averages
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()

# After period averages
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"After - Control group: {after_control:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate

df['treatment_interaction'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment_interaction', data=df).fit()
print("=== Simple DiD Regression ===")
print(model.summary())

# Now let's check parallel trends by looking at week-by-week averages
print("\n=== Week-by-Week Averages by Group ===")
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(weekly_avg)

# Let's also run a regression with branch fixed effects and week fixed effects
# This is more robust
df['week_str'] = df['week'].astype(str)
df['branch_str'] = df['branch_id'].astype(str)

# Regression with fixed effects
model_fe = ols('service_score ~ switched_branch * after_switch + C(branch_id) + C(week)', data=df).fit()
print("\n=== DiD Regression with Branch and Week Fixed Effects ===")
print(model_fe.summary())

# Extract the interaction term coefficient
interaction_coef = model_fe.params['switched_branch:after_switch']
interaction_pvalue = model_fe.pvalues['switched_branch:after_switch']
print(f"\nInteraction coefficient (DiD estimate): {interaction_coef:.3f}")
print(f"P-value: {interaction_pvalue:.4f}")
