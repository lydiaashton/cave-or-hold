model_basic = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Basic Difference-in-Differences Model ===")
print(model_basic.summary())

import statsmodels.api as sm

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Run the main DiD model with cluster-robust SE
df['treatment'] = df['switched_branch'] * df['after_switch']
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract key results
treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pval = model.pvalues['treatment']
treatment_ci_lower = model.conf_int().loc['treatment', 0]
treatment_ci_upper = model.conf_int().loc['treatment', 1]

print("=== Key Results Summary ===")
print(f"Treatment Effect (DiD estimate): {treatment_coef:.2f} points")
print(f"Standard Error: {treatment_se:.2f}")
print(f"95% Confidence Interval: [{treatment_ci_lower:.2f}, {treatment_ci_upper:.2f}]")
print(f"P-value: {treatment_pval:.3f}")

# Calculate pre and post means for each group
pre_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
post_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
pre_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
post_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"\n=== Group Means ===")
print(f"Control Group: Before={pre_control:.2f}, After={post_control:.2f}, Change={post_control-pre_control:.2f}")
print(f"Treatment Group: Before={pre_treatment:.2f}, After={post_treatment:.2f}, Change={post_treatment-pre_treatment:.2f}")
print(f"Difference-in-Differences: {(post_treatment-pre_treatment) - (post_control-pre_control):.2f}")

# Calculate percentage improvement relative to baseline
baseline = pre_treatment
pct_improvement = (treatment_coef / baseline) * 100
print(f"\nPercentage improvement relative to treatment baseline: {pct_improvement:.1f}%")

# Power analysis - what sample size would we need to detect this effect?
# Using the observed variance
residual_var = model.mse_resid
print(f"\nResidual variance: {residual_var:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Key statistics to reconsider
df['treatment'] = df['switched_branch'] * df['after_switch']
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Effect size relative to standard deviation
service_sd = df['service_score'].std()
effect_size = model.params['treatment'] / service_sd

print("=== Reconsidering the Evidence ===")
print(f"Service score standard deviation: {service_sd:.2f}")
print(f"Effect size (Cohen's d): {effect_size:.2f}")
print(f"Treatment coefficient: {model.params['treatment']:.2f}")
print(f"P-value: {model.pvalues['treatment']:.3f}")

# What's the minimum detectable effect with this sample?
# Rough power calculation
n_treatment = 20
n_control = 14
weeks_pre = 9
weeks_post = 8

print(f"\nSample: {n_treatment} treatment branches, {n_control} control branches")
print(f"Pre-period: {weeks_pre} weeks, Post-period: {weeks_post} weeks")

# Probability that effect is actually positive (Bayesian-ish interpretation)
# Using normal approximation
from scipy import stats
coef = model.params['treatment']
se = model.bse['treatment']
prob_positive = 1 - stats.norm.cdf(0, loc=coef, scale=se)
print(f"\nProbability effect is positive (normal approx): {prob_positive:.1%}")

# What would we need for significance at p<0.05?
required_t = 1.96  # approx for large sample
required_se = coef / required_t
print(f"\nFor p<0.05, we'd need SE of: {required_se:.2f} (currently {se:.2f})")
print(f"That's roughly {se/required_se:.1f}x more precise estimates needed")
