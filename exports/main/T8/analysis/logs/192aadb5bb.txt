import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# Create a treatment group indicator (interaction of switched_branch and after_switch)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary by group and period
print("=== Service Score Summary by Group and Period ===\n")

# Before switch (after_switch = 0)
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score']
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score']

# After switch (after_switch = 1)
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score']
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score']

print("Control Group (did not switch):")
print(f"  Before: mean={before_control.mean():.2f}, std={before_control.std():.2f}, n={len(before_control)}")
print(f"  After:  mean={after_control.mean():.2f}, std={after_control.std():.2f}, n={len(after_control)}")
print(f"  Change: {after_control.mean() - before_control.mean():.2f}")

print("\nTreatment Group (switched):")
print(f"  Before: mean={before_treatment.mean():.2f}, std={before_treatment.std():.2f}, n={len(before_treatment)}")
print(f"  After:  mean={after_treatment.mean():.2f}, std={after_treatment.std():.2f}, n={len(after_treatment)}")
print(f"  Change: {after_treatment.mean() - before_treatment.mean():.2f}")

# Difference-in-differences
control_change = after_control.mean() - before_control.mean()
treatment_change = after_treatment.mean() - before_treatment.mean()
did_estimate = treatment_change - control_change

print(f"\n=== Difference-in-Differences Estimate ===")
print(f"Control group change: {control_change:.2f}")
print(f"Treatment group change: {treatment_change:.2f}")
print(f"DiD estimate (treatment effect): {did_estimate:.2f}")

# Check parallel trends assumption - compare pre-treatment trends
print("\n=== Pre-Treatment Trends Check ===")
