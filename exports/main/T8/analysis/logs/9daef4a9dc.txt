import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Count branches by switch status
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())
print()

# Calculate mean service scores by group and period
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("Service scores by group and period:")
print(pivot)
print()

# Calculate the difference-in-differences manually
# Treatment group (switched=1)
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_diff = treatment_after - treatment_before

# Control group (switched=0)
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
control_diff = control_after - control_before

# DiD estimate
did_estimate = treatment_diff - control_diff

print("Difference-in-Differences Calculation:")
print(f"Treatment group (switched): Before={treatment_before:.2f}, After={treatment_after:.2f}, Change={treatment_diff:.2f}")
print(f"Control group (not switched): Before={control_before:.2f}, After={control_after:.2f}, Change={control_diff:.2f}")
print(f"DiD Estimate (Treatment Change - Control Change): {did_estimate:.2f}")
print()

# Run regression with interaction term for formal DiD
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['treatment_post'] = df['treatment'] * df['post']

# Simple DiD regression
model = ols('service_score ~ treatment + post + treatment_post', data=df).fit()
print("DiD Regression Results:")
print(model.summary())
print()

# Check parallel trends - look at pre-treatment trends
print("Parallel Trends Check - Mean service scores by week and group:")
weekly_means = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
print(weekly_means.round(2))

print("Model 1: Simple DiD (with clustered SEs)")
print(model1.params[['treatment', 'post', 'treatment_post']])

print("Model 2: Branch FE (treatment absorbed)")
print("treatment_post coef:", model2.params['treatment_post'])
print("treatment_post p-value:", model2.pvalues['treatment_post'])
print()
