import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD: 2x2 table
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES: 2x2 SUMMARY")
print("=" * 60)

# Calculate means for the 4 groups
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nControl group (no switch):")
print(f"  Before (weeks 6-14): {before_control:.2f}")
print(f"  After (weeks 15-22): {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")

print(f"\nTreatment group (switched):")
print(f"  Before (weeks 6-14): {before_treatment:.2f}")
print(f"  After (weeks 15-22): {after_treatment:.2f}")
print(f"  Change: {after_treatment - before_treatment:.2f}")

# DiD estimate
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")

# Now run regression
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Model 1: Basic DiD
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# Model 2: With branch fixed effects
print("\n" + "=" * 60)

# Create branch dummies

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.regression.linear_model import OLS
from statsmodels.stats.diagnostic import het_breuschpagan
import warnings
warnings.filterwarnings('ignore')

df = pd.read_csv('branches.csv')

# Create DiD variable
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD with branch fixed effects and cluster-robust SEs
print("=" * 60)
print("FIXED EFFECTS MODEL WITH CLUSTER-ROBUST STANDARD ERRORS")
print("=" * 60)

# Demean within branches (fixed effects transformation)
df_sorted = df.sort_values(['branch_id', 'week'])

# Calculate branch means
branch_means = df_sorted.groupby('branch_id')[['service_score', 'switched_branch', 'after_switch', 'treatment', 'staffing_index', 'visits_index']].mean()
branch_means.columns = [c + '_mean' for c in branch_means.columns]

df_fe = df_sorted.merge(branch_means, left_on='branch_id', right_index=True)

# Demeaned variables
df_fe['service_score_d'] = df_fe['service_score'] - df_fe['service_score_mean']
df_fe['switched_branch_d'] = df_fe['switched_branch'] - df_fe['switched_branch_mean']
df_fe['after_switch_d'] = df_fe['after_switch'] - df_fe['after_switch_mean']
df_fe['treatment_d'] = df_fe['treatment'] - df_fe['treatment_mean']
df_fe['staffing_index_d'] = df_fe['staffing_index'] - df_fe['staffing_index_mean']
df_fe['visits_index_d'] = df_fe['visits_index'] - df_fe['visits_index_mean']

# Run FE regression
X = df_fe[['after_switch_d', 'treatment_d']]
X = sm.add_constant(X, has_constant='add')
y = df_fe['service_score_d']

model_fe = OLS(y, X).fit()

# Calculate cluster-robust standard errors
def cluster_robust_se(model, cluster_var):
    """Calculate cluster-robust standard errors"""
    X = model.model.exog
    residuals = model.resid
    n_clusters = len(np.unique(cluster_var))
    n = len(residuals)
    k = X.shape[1]
    
    # Cluster the residuals
    clusters = np.unique(cluster_var)
    score_clusters = []
    for c in clusters:
        mask = cluster_var == c
        score_clusters.append(np.dot(X[mask].T, residuals[mask]))
    
    score_clusters = np.array(score_clusters)
    bread = np.linalg.inv(np.dot(X.T, X))
    meat = np.dot(score_clusters.T, score_clusters)
    
    vcov_cluster = (n_clusters / (n_clusters - 1)) * ((n - 1) / (n - k)) * bread @ meat @ bread
    
    se_cluster = np.sqrt(np.diag(vcov_cluster))
    return se_cluster, vcov_cluster

cluster_se, vcov_cluster = cluster_robust_se(model_fe, df_fe['branch_id'])

print("\nFixed Effects Model (Cluster-robust SEs at branch level):")
print(f"\n{'Variable':<20} {'Coef':>10} {'SE':>10} {'t':>10} {'P>|t|':>10}")
print("-" * 60)
for i, var in enumerate(['const', 'after_switch_d', 'treatment_d']):
