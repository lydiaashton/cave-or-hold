import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched + beta2*after + beta3*(switched*after) + error
# beta3 is the DiD estimate

df['treatment_interaction'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment_interaction', data=df).fit()
print("Basic DiD Regression:")
print(model.summary())

# Now let's add branch fixed effects and week fixed effects for a more robust analysis
# This controls for time-invariant branch characteristics and common time trends

# Create dummy variables for branch and week
df_with_fe = df.copy()
branch_dummies = pd.get_dummies(df_with_fe['branch_id'], prefix='branch', drop_first=True)
week_dummies = pd.get_dummies(df_with_fe['week'], prefix='week', drop_first=True)

# Combine with original data
X = pd.concat([df_with_fe[['switched_branch', 'after_switch', 'treatment_interaction']], 
               branch_dummies, week_dummies], axis=1)
y = df_with_fe['service_score']

# Add constant
X = sm.add_constant(X)

# Fit the model with fixed effects
fe_model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df_with_fe['branch_id']})
print("\n\nDiD Regression with Branch and Week Fixed Effects (clustered SEs):")
print("Treatment interaction coefficient:", fe_model.params['treatment_interaction'])
print("Standard error:", fe_model.bse['treatment_interaction'])
print("P-value:", fe_model.pvalues['treatment_interaction'])
print("95% CI:", fe_model.conf_int().loc['treatment_interaction'])

import pandas as pd
import numpy as np
from scipy import stats

# Load the data
df = pd.read_csv('branches.csv')

# Calculate the key statistics again
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

did_estimate = (treatment_after - treatment_before) - (control_after - control_before)

# Branch-level analysis
branch_before = df[df['after_switch'] == 0].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_before.columns = ['branch_id', 'switched_branch', 'score_before']
branch_after = df[df['after_switch'] == 1].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_after.columns = ['branch_id', 'switched_branch', 'score_after']
branch_level = branch_before.merge(branch_after, on=['branch_id', 'switched_branch'])
branch_level['change'] = branch_level['score_after'] - branch_level['score_before']

treatment_changes = branch_level[branch_level['switched_branch'] == 1]['change']
control_changes = branch_level[branch_level['switched_branch'] == 0]['change']

# Power analysis - what effect size could we detect?
n_treatment = len(treatment_changes)
n_control = len(control_changes)
pooled_std = np.sqrt(((n_treatment - 1) * treatment_changes.std()**2 + (n_control - 1) * control_changes.std()**2) / (n_treatment + n_control - 2))

print("=== STATISTICAL POWER ASSESSMENT ===")
print(f"Treatment branches: {n_treatment}")
print(f"Control branches: {n_control}")
print(f"Pooled std dev of changes: {pooled_std:.2f}")
print(f"Standard error of difference: {pooled_std * np.sqrt(1/n_treatment + 1/n_control):.2f}")

# Minimum detectable effect at 80% power, alpha=0.05
from statsmodels.stats.power import TTestIndPower
power_analysis = TTestIndPower()
effect_size = did_estimate / pooled_std
print(f"\nObserved effect size (Cohen's d): {effect_size:.2f}")

# What's the minimum detectable effect with this sample?
mde = power_analysis.solve_power(effect_size=None, nobs1=n_treatment, ratio=n_control/n_treatment, alpha=0.05, power=0.80)
print(f"Minimum detectable effect (80% power): {mde:.2f} (in std dev units)")
print(f"Minimum detectable effect (points): {mde * pooled_std:.2f}")

# What power do we have to detect the observed effect?
actual_power = power_analysis.solve_power(effect_size=effect_size, nobs1=n_treatment, ratio=n_control/n_treatment, alpha=0.05)
print(f"Statistical power to detect observed effect: {actual_power*100:.0f}%")

# Probability the true effect is positive (Bayesian approximation)
