import pandas as pd

from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create event time relative to switch (week 15)
df['event_time'] = df['week'] - 15

# Event study regression with week fixed effects
# Using week 14 as reference (last pre-switch week)
df['week_fe'] = df['week'].astype(str)

# Create interaction terms for each week with treatment
event_study_df = df.copy()
for week in range(6, 23):
    if week != 14:  # Reference week
        event_study_df[f'week_{week}_treat'] = ((event_study_df['week'] == week) & (event_study_df['switched_branch'] == 1)).astype(int)

# Build formula
control_vars = ['switched_branch', 'after_switch']
week_vars = [f'week_{w}_treat' for w in range(6, 23) if w != 14]
formula = 'service_score ~ ' + ' + '.join(control_vars + week_vars)

# Run event study regression
event_model = ols(formula, data=event_study_df).fit(cov_type='cluster', cov_kwds={'groups': event_study_df['branch_id']})

# Extract coefficients for event study
print("=== Event Study Coefficients ===")
coef_dict = {}
for week in range(6, 23):
    if week == 14:
        coef_dict[week] = 0  # Reference
    else:
        coef_name = f'week_{week}_treat'
        if coef_name in event_model.params.index:
            coef_dict[week] = event_model.params[coef_name]

# Get confidence intervals
ci_dict = {}
for week in range(6, 23):
    if week == 14:
        ci_dict[week] = (0, 0)
    else:
        coef_name = f'week_{week}_treat'
        if coef_name in event_model.conf_int().index:
            ci_dict[week] = event_model.conf_int().loc[coef_name].values

# Print results
print("\nWeek | Coefficient | 95% CI")
print("-" * 40)
for week in sorted(coef_dict.keys()):

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from linearmodels.panel import PanelOLS

# Load the data
df = pd.read_csv('branches.csv')

# Create event time relative to switch (week 15)
df['event_time'] = df['week'] - 15

# Prepare panel data
panel_df = df.copy()
panel_df = panel_df.set_index(['branch_id', 'week'])

# Create treatment interaction for each period
for et in range(-9, 8):
    if et != -1:
        panel_df[f'et_{et}'] = ((panel_df['event_time'] == et) & (panel_df['switched_branch'] == 1)).astype(float)

# Run panel regression
exog_vars = [f'et_{et}' for et in range(-9, 8) if et != -1]
mod = PanelOLS(panel_df['service_score'], panel_df[exog_vars], entity_effects=True, time_effects=True)
panel_results = mod.fit(cov_type='clustered', cluster_entity=True)

# Extract coefficients
ci = panel_results.conf_int()
coef_dict = {-1: 0}
ci_dict = {-1: (0, 0)}
signif_dict = {-1: False}

for et in range(-9, 8):
    if et != -1:
        var_name = f'et_{et}'
        coef_dict[et] = panel_results.params[var_name]
        ci_row = ci.loc[var_name]
        ci_dict[et] = (ci_row['lower'], ci_row['upper'])
        # Check if CI excludes zero
        signif_dict[et] = (ci_dict[et][0] > 0) or (ci_dict[et][1] < 0)

print("=== Event Study with Significance ===")
print("Event Time | Week | Coefficient | 95% CI | Significant")
print("-" * 65)
for et in sorted(coef_dict.keys()):
    week = et + 15
    c = coef_dict[et]
    ci_vals = ci_dict[et]
    sig = "Yes" if signif_dict[et] else "No"
    print(f"{et:10d} | {week:4d} | {c:11.3f} | [{ci_vals[0]:6.3f}, {ci_vals[1]:6.3f}] | {sig}")

# Count significant pre and post periods
pre_significant = sum(1 for et in range(-9, -1) if signif_dict[et])
post_significant = sum(1 for et in range(0, 8) if signif_dict[et])

print(f"\nSignificant pre-switch weeks (6-13): {pre_significant} of 8")
print(f"Significant post-switch weeks (15-22): {post_significant} of 8")

# Simple test: average of pre-period coefficients
