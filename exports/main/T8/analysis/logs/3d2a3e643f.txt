import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD with week fixed effects
# Convert week to categorical
df['week_cat'] = df['week'].astype(str)

# Model with week fixed effects (drop one week as reference)
model_week_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(week_cat)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== DiD with Week Fixed Effects ===")
print(model_week_fe.summary())

# Also run with branch fixed effects for robustness
# This requires a different approach - using branch as categorical
df['branch_cat'] = df['branch_id'].astype(str)

# Model with branch fixed effects (this will be large, so just show key results)
model_branch_fe = ols('service_score ~ after_switch + treatment + C(branch_cat)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD with Branch Fixed Effects ===")
# Just print the key coefficients
print("Key coefficients:")
print(model_branch_fe.params[['after_switch', 'treatment']])
print("\nStandard errors:")
print(model_branch_fe.bse[['after_switch', 'treatment']])
print("\nP-values:")
print(model_branch_fe.pvalues[['after_switch', 'treatment']])

# Event study - check effects by week relative to switch
df['weeks_relative'] = df['week'] - 15  # Week 15 is the switch week (0)
# Create dummies for each week relative to switch
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    df[f'rel_week_{w}'] = (df['weeks_relative'] == w).astype(int)

# Interact with treatment
for w in range(-9, 8):
    df[f'treat_rel_{w}'] = df['switched_branch'] * df[f'rel_week_{w}']

# Run event study regression (omit week -1 as reference)
event_study_vars = ' + '.join([f'treat_rel_{w}' for w in range(-9, 8) if w != -1])
event_formula = f'service_score ~ C(week_cat) + C(branch_cat) + {event_study_vars}'

# This model is too large, let's do a simpler version
event_formula_simple = f'service_score ~ switched_branch + after_switch + {event_study_vars}'
model_event = ols(event_formula_simple, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n=== Event Study Results (Treatment effects by week relative to switch) ===")
# Extract treatment effects for each relative week
event_results = []
for w in range(-9, 8):
    if w != -1:
        coef = model_event.params[f'treat_rel_{w}']
        se = model_event.bse[f'treat_rel_{w}']
        event_results.append({'relative_week': w, 'coef': coef, 'se': se})

event_df = pd.DataFrame(event_results)
event_df['ci_lower'] = event_df['coef'] - 1.96 * event_df['se']
event_df['ci_upper'] = event_df['coef'] + 1.96 * event_df['se']
event_df['significant'] = (event_df['ci_lower'] > 0) | (event_df['ci_upper'] < 0)
print(event_df.to_string(index=False))

for w in range(-9, 8):
    col_name = f'rel_w{abs(w)}_neg' if w < 0 else f'rel_w{w}'
    df[col_name] = (df['weeks_relative'] == w).astype(int)

# Create interaction terms with treatment
treat_cols = []
for w in range(-9, 8):
    if w != -1:  # Omit week -1 as reference

        interact_name = f'treat_{col_name}'
        df[interact_name] = df['switched_branch'] * df[col_name]
        treat_cols.append(interact_name)

df['week_cat'] = df['week'].astype(str)

import pandas as pd

            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main DiD model with controls and cluster-robust SEs
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== FINAL DiD MODEL RESULTS ===")
print(f"Treatment effect coefficient: {model.params['treatment']:.3f}")
print(f"Standard error: {model.bse['treatment']:.3f}")
print(f"95% CI: [{model.params['treatment'] - 1.96 * model.bse['treatment']:.3f}, {model.params['treatment'] + 1.96 * model.bse['treatment']:.3f}]")
print(f"P-value: {model.pvalues['treatment']:.6f}")

# Calculate practical significance
avg_score = df['service_score'].mean()
pct_change = (model.params['treatment'] / avg_score) * 100
print(f"\nAverage service score: {avg_score:.2f}")
print(f"Treatment effect as % of average: {pct_change:.2f}%")

# Create a comprehensive visualization
fig, axes = plt.subplots(2, 2, figsize=(14, 10))

# Plot 1: Trends over time
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
weekly_avg.columns = ['switched', 'week', 'score']
pivot_weekly = weekly_avg.pivot(index='week', columns='switched', values='score')
pivot_weekly.columns = ['Control', 'Treatment']

ax1 = axes[0, 0]
ax1.plot(pivot_weekly.index, pivot_weekly['Control'], 'o-', label='Control (No Switch)', linewidth=2, markersize=6, color='gray')
ax1.plot(pivot_weekly.index, pivot_weekly['Treatment'], 's-', label='Treatment (Switched)', linewidth=2, markersize=6, color='blue')
ax1.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (Week 15)')
ax1.axvspan(15, 22, alpha=0.2, color='green', label='Post-Switch Period')
ax1.set_xlabel('Week', fontsize=11)
ax1.set_ylabel('Average Service Score', fontsize=11)
ax1.set_title('Service Score Trends by Group', fontsize=12, fontweight='bold')
ax1.legend(fontsize=9)
ax1.grid(True, alpha=0.3)
ax1.set_xticks(range(6, 23))

# Plot 2: Pre vs Post comparison
pre_data = df[df['after_switch'] == 0]
post_data = df[df['after_switch'] == 1]

pre_means = pre_data.groupby('switched_branch')['service_score'].mean()
post_means = post_data.groupby('switched_branch')['service_score'].mean()

ax2 = axes[0, 1]
x = np.array([0, 1])
width = 0.35
ax2.bar(x - width/2, [pre_means[0], pre_means[1]], width, label='Pre-Switch', color='lightgray', edgecolor='black')
ax2.bar(x + width/2, [post_means[0], post_means[1]], width, label='Post-Switch', color='steelblue', edgecolor='black')
ax2.set_xlabel('Group', fontsize=11)
ax2.set_ylabel('Average Service Score', fontsize=11)
ax2.set_title('Pre vs Post Comparison', fontsize=12, fontweight='bold')
ax2.set_xticks(x)
ax2.set_xticklabels(['Control', 'Treatment'])
ax2.legend(fontsize=9)
ax2.grid(True, alpha=0.3, axis='y')

# Add DiD annotation
did_effect = (post_means[1] - pre_means[1]) - (post_means[0] - pre_means[0])
ax2.text(0.5, max(post_means.max(), pre_means.max()) + 1, 
         f'DiD Effect: +{did_effect:.2f} pts', 
         fontsize=11, fontweight='bold', ha='center',

# Plot 3: Distribution of scores
ax3 = axes[1, 0]

ax3.boxplot([treatment_pre, treatment_post], labels=['Pre', 'Post'], patch_artist=True,

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main DiD model
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Key statistics
effect = model.params['treatment']
se = model.bse['treatment']
ci_lower = effect - 1.96 * se
ci_upper = effect + 1.96 * se
pvalue = model.pvalues['treatment']

print("="*60)
print("CRITICAL REVIEW OF BOTTOM LINE")
print("="*60)

print(f"\nTreatment Effect: +{effect:.2f} points")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {pvalue:.6f}")

# Check if CI includes 0
if ci_lower > 0:
    print(f"\n✓ Confidence interval does NOT include 0")
    print(f"✓ Effect is statistically significant at 5% level")
else:
    print(f"\n✗ WARNING: Confidence interval includes 0")

# Effect size
avg_score = df['service_score'].mean()
pct = (effect / avg_score) * 100
print(f"\nEffect as % of average score: {pct:.1f}%")

# Pre-trend check
print("\n" + "="*60)
print("PARALLEL TRENDS CHECK")
print("="*60)

# Simple pre-trend: compare trends in pre-period
pre = df[df['week'] < 15].copy()
pre_trend = pre.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
pre_trend.columns = ['switched', 'week', 'score']

# Calculate week-to-week changes for each group
pre_trend_sorted = pre_trend.sort_values(['switched', 'week'])
control_trend = pre_trend_sorted[pre_trend_sorted['switched'] == 0]['score'].diff().dropna()
treatment_trend = pre_trend_sorted[pre_trend_sorted['switched'] == 1]['score'].diff().dropna()

print(f"\nAverage week-to-week change (pre-period):")
print(f"  Control group: {control_trend.mean():.2f}")
print(f"  Treatment group: {treatment_trend.mean():.2f}")
print(f"  Difference in trends: {treatment_trend.mean() - control_trend.mean():.2f}")

# Check if pre-period trends are similar
if abs(treatment_trend.mean() - control_trend.mean()) < 0.5:
    print(f"\n✓ Pre-period trends appear similar (parallel trends plausible)")
else:
    print(f"\n⚠ Pre-period trends differ - interpret with caution")

print("\n" + "="*60)
print("BOTTOM LINE ASSESSMENT")
print("="*60)

if ci_lower > 0 and pvalue < 0.05:
    print("\nCONFIRMED: The evidence supports a positive effect.")
    print(f"- Effect is statistically significant (p < 0.05)")
    print(f"- Lower bound of 95% CI is {ci_lower:.2f} (above 0)")
    print(f"- Direction matches expectation (positive improvement)")
    print("\nRECOMMENDATION: Proceed with rollout, but monitor results.")
else:
    print("\nCAUTION: Evidence is weaker than expected.")
