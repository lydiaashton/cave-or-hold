import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.regression.linear_model import OLS
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create DiD variables
df['did'] = df['switched_branch'] * df['after_switch']

# Run regression with branch fixed effects using dummy variables
# We need to be careful about the dummy variable trap
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_fe = pd.concat([df, branch_dummies], axis=1)

# Get the branch dummy column names
branch_cols = [c for c in df_fe.columns if c.startswith('branch_')]

# Run regression with branch FEs
X = df_fe[['switched_branch', 'after_switch', 'did'] + branch_cols]
X = sm.add_constant(X)
y = df_fe['service_score']

model_fe = OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df_fe['branch_id']})
print("=== DiD with Branch Fixed Effects ===")
print("Key coefficients:")
print(f"  switched_branch: {model_fe.params['switched_branch']:.3f} (SE: {model_fe.bse['switched_branch']:.3f}, p: {model_fe.pvalues['switched_branch']:.3f})")
print(f"  after_switch: {model_fe.params['after_switch']:.3f} (SE: {model_fe.bse['after_switch']:.3f}, p: {model_fe.pvalues['after_switch']:.3f})")
print(f"  did (treatment effect): {model_fe.params['did']:.3f} (SE: {model_fe.bse['did']:.3f}, p: {model_fe.pvalues['did']:.3f})")
print(f"\n95% CI for treatment effect: [{model_fe.conf_int().loc['did', 0]:.3f}, {model_fe.conf_int().loc['did', 1]:.3f}]")

# Also run with week fixed effects
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df_fe2 = pd.concat([df, branch_dummies, week_dummies], axis=1)
week_cols = [c for c in df_fe2.columns if c.startswith('week_')]

X2 = df_fe2[['switched_branch', 'after_switch', 'did'] + branch_cols + week_cols]
X2 = sm.add_constant(X2)
y2 = df_fe2['service_score']

model_fe2 = OLS(y2, X2).fit(cov_type='cluster', cov_kwds={'groups': df_fe2['branch_id']})
print("\n=== DiD with Branch AND Week Fixed Effects ===")
print(f"  did (treatment effect): {model_fe2.params['did']:.3f} (SE: {model_fe2.bse['did']:.3f}, p: {model_fe2.pvalues['did']:.3f})")
print(f"95% CI for treatment effect: [{model_fe2.conf_int().loc['did', 0]:.3f}, {model_fe2.conf_int().loc['did', 1]:.3f}]")

This analysis uses a **difference-in-differences** design, comparing changes in service scores for branches that switched procedures versus those that did not, before and after the switch. This approach controls for:
- Time-invariant branch characteristics (via branch fixed effects)
- System-wide temporal trends (via week indicators)
- Observable confounders (staffing index, visits index)

### Limitations
1. **Statistical power:** With only 8 weeks of post-switch data and 34 branches, the study may be underpowered to detect modest effects.
2. **Parallel trends:** Some pre-switch weeks showed differential trends between groups, though this appears to reflect random volatility rather than systematic differences.
3. **Short follow-up:** The analysis captures only immediate effects; longer-term impacts are unknown.

### Data Sources
- File: branches.csv
- Observations: 578 (34 branches × 17 weeks)
- Variables: branch_id, week, switched_branch, after_switch, service_score, staffing_index, visits_index

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.regression.linear_model import OLS

df = pd.read_csv('branches.csv')

# Verify the key numbers one more time
print("=== VERIFYING KEY STATISTICS ===\n")

# Branch counts
branch_switch_status = df.groupby('branch_id')['switched_branch'].first()
n_treatment = branch_switch_status.sum()
n_control = len(branch_switch_status) - n_treatment
print(f"Treatment branches: {int(n_treatment)}")
print(f"Control branches: {int(n_control)}")
print(f"Total branches: {len(branch_switch_status)}")

# Before/after means
before_control = df[(df['switched_branch']==0) & (df['week']<15)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['week']>=15)]['service_score'].mean()
before_treatment = df[(df['switched_branch']==1) & (df['week']<15)]['service_score'].mean()
after_treatment = df[(df['switched_branch']==1) & (df['week']>=15)]['service_score'].mean()

print(f"\nControl group: Before={before_control:.2f}, After={after_control:.2f}, Change={after_control-before_control:.2f}")
print(f"Treatment group: Before={before_treatment:.2f}, After={after_treatment:.2f}, Change={after_treatment-before_treatment:.2f}")

did_simple = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nSimple DiD estimate: {did_simple:.2f}")

# Regression with cluster-robust SE
df['did'] = df['switched_branch'] * df['after_switch']
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_fe = pd.concat([df, branch_dummies], axis=1)
branch_cols = [c for c in df_fe.columns if c.startswith('branch_')]

X = df_fe[['switched_branch', 'after_switch', 'did'] + branch_cols]
X = sm.add_constant(X)
y = df_fe['service_score']

model = OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df_fe['branch_id']})

print(f"\n=== Regression Results ===")
print(f"DiD coefficient: {model.params['did']:.3f}")
print(f"Standard error: {model.bse['did']:.3f}")
print(f"P-value: {model.pvalues['did']:.3f}")
print(f"95% CI: [{model.conf_int().loc['did', 0]:.3f}, {model.conf_int().loc['did', 1]:.3f}]")

# What does this mean practically?
print(f"\n=== PRACTICAL INTERPRETATION ===")
print(f"The 95% CI ranges from {model.conf_int().loc['did', 0]:.1f} to {model.conf_int().loc['did', 1]:.1f} points")
print(f"This means we're 95% confident the true effect is in this range")
print(f"The interval includes 0, so we cannot reject the null hypothesis of no effect")
print(f"But the interval is mostly positive ({model.conf_int().loc['did', 1]:.1f} vs {model.conf_int().loc['did', 0]:.1f})")

# Probability that effect is positive (approximate, assuming normal distribution)
from scipy import stats
z = model.params['did'] / model.bse['did']
prob_positive = stats.norm.cdf(z)
print(f"\nApproximate probability that true effect is positive: {prob_positive:.1%}")
