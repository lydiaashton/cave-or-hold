import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# ===== Robustness: Check if effect grows over time =====
print("=" * 60)
print("EFFECT EVOLUTION OVER TIME")
print("=" * 60)

# Post-period weekly differences
post_weeks = df[df['after_switch'] == 1].groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
post_weeks['Diff'] = post_weeks[1] - post_weeks[0]
print("\nPost-switch weekly differences:")
print(post_weeks[['Diff']].round(2))

# ===== Alternative specification: Control for branch fixed effects =====
print("\n" + "=" * 60)
print("BRANCH FIXED EFFECTS MODEL")
print("=" * 60)

# Create branch dummies
df_fe = df.copy()
df_fe['branch_cat'] = df_fe['branch_id'].astype('category')

# Model with branch fixed effects
# service_score = branch_FE + after_switch + did + controls
model_fe = ols('service_score ~ C(branch_id) + after_switch + did + staffing_index + visits_index', data=df_fe).fit(cov_type='cluster', cov_kwds={'groups': df_fe['branch_id']})

print("\nBranch Fixed Effects Model (key coefficients):")
key_vars = ['after_switch', 'did', 'staffing_index', 'visits_index']
for var in key_vars:
    coef = model_fe.params[var]
    se = model_fe.bse[var]
    p = 2 * (1 - 0.5 * (1 + np.math.erf(abs(coef/se) / np.sqrt(2))))
    print(f"  {var:20s}: {coef:8.3f} (SE: {se:.3f}, p: {p:.4f})")

# ===== Effect size interpretation =====
print("\n" + "=" * 60)
print("EFFECT SIZE INTERPRETATION")
print("=" * 60)

did_effect = model_fe.params['did']
service_std = df['service_score'].std()
print(f"\nDiD Effect: {did_effect:.2f} points")
print(f"Service Score Std Dev: {service_std:.2f}")
print(f"Effect as % of Std Dev: {did_effect/service_std*100:.1f}%")
print(f"Baseline (control pre): ~49.2")
print(f"Effect as % of baseline: {did_effect/49.2*100:.1f}%")

# ===== Visualization of event study =====
# Extract event study results
event_results = []
df_weekly = df.copy()
for w in range(6, 23):
    df_weekly[f'week_{w}'] = (df_weekly['week'] == w).astype(int)
    df_weekly[f'treat_week_{w}'] = df_weekly['switched_branch'] * df_weekly[f'week_{w}']

treat_cols = [f'treat_week_{w}' for w in range(6, 23) if w != 14]
week_cols = [f'week_{w}' for w in range(6, 23) if w != 14]
formula = 'service_score ~ switched_branch + ' + ' + '.join(treat_cols) + ' + '.join(week_cols)
model_event = ols(formula, data=df_weekly).fit(cov_type='cluster', cov_kwds={'groups': df_weekly['branch_id']})

weeks = []
coefs = []
ses = []
for w in range(6, 23):
    if w == 14:
        weeks.append(w)
        coefs.append(0)
        ses.append(0)
    else:
        coef_name = f'treat_week_{w}'
        weeks.append(w)
        coefs.append(model_event.params[coef_name])
        ses.append(model_event.bse[coef_name])

plt.figure(figsize=(10, 6))
plt.errorbar(weeks, coefs, yerr=[1.96*s for s in ses], fmt='o', capsize=5, 
             color='blue', ecolor='gray', linewidth=2, markersize=8)
plt.axhline(y=0, color='black', linestyle='-', linewidth=1)
