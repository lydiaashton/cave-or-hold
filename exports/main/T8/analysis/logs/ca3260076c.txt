import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create a treatment interaction term (switched_branch * after_switch)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Difference-in-differences summary table
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES SUMMARY")
print("=" * 60)

# Calculate means for each group
before_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nService Score Averages:")
print(f"  Switched branches - Before: {before_switched:.2f}, After: {after_switched:.2f}, Change: {after_switched - before_switched:.2f}")
print(f"  Control branches  - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")

# Difference in differences
diff_switched = after_switched - before_switched
diff_control = after_control - before_control
did_effect = diff_switched - diff_control

print(f"\nDifference-in-Differences Estimate: {did_effect:.2f}")
print(f"  (Change in switched) - (Change in control) = {diff_switched:.2f} - {diff_control:.2f}")

# Run regression with standard errors clustered by branch
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nBasic OLS Regression (no clustering):")
print(model.summary().tables[1])

# Clustered standard errors by branch
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nRegression with Clustered Standard Errors (by branch):")
print(model_clustered.summary().tables[1])

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check pre-trends: compare switched vs control before the switch (weeks 6-14)
print("=" * 60)
print("PRE-TREND ANALYSIS (Weeks 6-14)")
print("=" * 60)

pre_data = df[df['week'] <= 14]
pre_switched = pre_data[pre_data['switched_branch'] == 1]['service_score'].mean()
pre_control = pre_data[pre_data['switched_branch'] == 0]['service_score'].mean()

print(f"Average service score before switch:")
print(f"  Switched branches: {pre_switched:.2f}")
print(f"  Control branches:  {pre_control:.2f}")
print(f"  Difference: {pre_switched - pre_control:.2f}")

# Test if pre-trend difference is significant
pre_model = ols('service_score ~ switched_branch', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print(f"\nPre-trend difference p-value: {pre_model.pvalues['switched_branch']:.3f}")

# Weekly trends visualization data
print("\n" + "=" * 60)
print("WEEKLY AVERAGES BY GROUP")
print("=" * 60)

weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_avg.columns = ['Control', 'Switched']
print(weekly_avg.round(2))

# Create the plot
plt.figure(figsize=(12, 6))
for group in [0, 1]:

plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("\nPlot saved as 'service_score_trends.png'")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Final model with clustered SE
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract key statistics
treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pvalue = model.pvalues['treatment']
treatment_ci_lower = model.conf_int().loc['treatment'][0]
treatment_ci_upper = model.conf_int().loc['treatment'][1]

print("=" * 60)
print("FINAL ANALYSIS SUMMARY")
print("=" * 60)

print(f"\nTreatment Effect Estimate: {treatment_coef:.2f} points")
print(f"95% Confidence Interval: [{treatment_ci_lower:.2f}, {treatment_ci_upper:.2f}]")
print(f"P-value: {treatment_pvalue:.3f}")

# Sample sizes
n_branches = df['branch_id'].nunique()
n_switched = df['switched_branch'].sum() // 17  # approximately
n_control = (1 - df['switched_branch']).sum() // 17
n_switched_actual = df[df['switched_branch'] == 1]['branch_id'].nunique()
n_control_actual = df[df['switched_branch'] == 0]['branch_id'].nunique()

print(f"\nNumber of branches: {n_branches}")
print(f"  Switched branches: {n_switched_actual}")
print(f"  Control branches: {n_control_actual}")
print(f"Total observations: {len(df)}")

# Effect size relative to baseline
baseline_score = df['service_score'].mean()
print(f"\nBaseline average service score: {baseline_score:.2f}")
print(f"Treatment effect as % of baseline: {(treatment_coef/baseline_score)*100:.1f}%")

# Check if effect is practically meaningful
print(f"\nInterpretation:")
print(f"  The new procedure is associated with a {treatment_coef:.2f} point increase")
print(f"  in service scores relative to branches that did not switch.")
print(f"  This effect is {'statistically significant at 5% level' if treatment_pvalue < 0.05 else 'not statistically significant at 5% level (p=' + str(round(treatment_pvalue, 3)) + ')'}")
print(f"  However, it IS marginally significant (p < 0.10)")
