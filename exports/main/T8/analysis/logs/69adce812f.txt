import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
print("=== DIFFERENCE-IN-DIFFERENCES REGRESSION ===")
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model1.summary())

# DiD with controls
print("\n=== DIFFERENCE-IN-DIFFERENCES WITH CONTROLS ===")
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print(model2.summary())

# Clustered standard errors by branch (important for panel data)
print("\n=== CLUSTERED STANDARD ERRORS BY BRANCH ===")
model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model3.summary())

# Extract key coefficient
did_coef = model3.params['treatment']
did_se = model3.bse['treatment']
did_pvalue = model3.pvalues['treatment']

print(f"Standard Error: {did_se:.3f}")
print(f"P-value: {did_pvalue:.4f}")
print(f"95% CI: [{did_coef - 1.96*did_se:.3f}, {did_coef + 1.96*did_se:.3f}]")

model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Calculate probability that effect is positive (Bayesian-ish interpretation)
from scipy import stats
did_coef = model.params['treatment']
did_se = model.bse['treatment']

# Assuming normal distribution, what's the probability the true effect is > 0?
prob_positive = 1 - stats.norm.cdf(0, loc=did_coef, scale=did_se)
print(f"Probability that true effect is positive: {prob_positive:.1%}")

# What about probability effect is > 1 point?
prob_gt_1 = 1 - stats.norm.cdf(1, loc=did_coef, scale=did_se)
print(f"Probability that true effect is > 1 point: {prob_gt_1:.1%}")

# What about probability effect is > 2 points?
prob_gt_2 = 1 - stats.norm.cdf(2, loc=did_coef, scale=did_se)
print(f"Probability that true effect is > 2 points: {prob_gt_2:.1%}")

# Effect size in terms of standard deviations
print(f"\nEffect size relative to score SD: {did_coef / df['service_score'].std():.2f} SD")

# Power analysis - what effect could we detect?
n_treatment = df[df['switched_branch']==1].shape[0]
n_control = df[df['switched_branch']==0].shape[0]
print(f"\nSample sizes: Treatment={n_treatment}, Control={n_control}")

# Minimum detectable effect (rough approximation)
pooled_se = np.sqrt(did_se**2)
mde_80 = 2.8 * pooled_se  # Rough MDE for 80% power
print(f"Approximate minimum detectable effect (80% power): ~{mde_80:.1f} points")
