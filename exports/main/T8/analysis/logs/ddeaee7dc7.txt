import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for difference-in-differences
df['treated'] = df['switched_branch']  # 1 if branch switched
df['post'] = df['after_switch']  # 1 if after week 15
df['treated_post'] = df['treated'] * df['post']

# Run difference-in-differences regression
# service_score = beta0 + beta1*treated + beta2*post + beta3*(treated*post) + error
# beta3 is the DiD estimator - the effect of the new procedure

model = ols('service_score ~ treated + post + treated_post', data=df).fit()
print("Difference-in-Differences Regression Results:")
print("=" * 60)
print(model.summary())

# Extract the key coefficient
did_effect = model.params['treated_post']
did_pvalue = model.pvalues['treated_post']
did_ci = model.conf_int().loc['treated_post']

print(f"\n\nKey Result - Treatment Effect (DiD estimator):")
print(f"  Effect size: {did_effect:.3f}")
print(f"  95% CI: [{did_ci[0]:.3f}, {did_ci[1]:.3f}]")
print(f"  P-value: {did_pvalue:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

# Add controls for staffing and visits
model_with_controls = ols('service_score ~ treated + post + treated_post + staffing_index + visits_index', data=df).fit()
print("DiD Regression with Controls:")
print("=" * 60)
print(model_with_controls.summary())

# Extract the key coefficient
did_effect = model_with_controls.params['treated_post']
did_pvalue = model_with_controls.pvalues['treated_post']
did_ci = model_with_controls.conf_int().loc['treated_post']

print(f"\n\nKey Result with Controls:")
print(f"  Effect size: {did_effect:.3f}")
print(f"  95% CI: [{did_ci[0]:.3f}, {did_ci[1]:.3f}]")
print(f"  P-value: {did_pvalue:.4f}")

# Also check with branch fixed effects (using branch_id as categorical)
print("\n\n" + "=" * 60)
print("DiD with Branch Fixed Effects:")
print("=" * 60)
