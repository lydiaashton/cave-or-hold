import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check parallel trends - average service score by week for each group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
weekly_avg.columns = ['Control (no switch)', 'Treatment (switched)']
print("Weekly average service scores:")
print(weekly_avg)

# Calculate pre-switch trend difference
pre_switch = df[df['week'] < 15]
pre_trend = pre_switch.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)

# Plot the trends
plt.figure(figsize=(12, 6))
plt.plot(weekly_avg.index, weekly_avg['Control (no switch)'], 'o-', label='Control (no switch)', linewidth=2, markersize=6)
plt.plot(weekly_avg.index, weekly_avg['Treatment (switched)'], 's-', label='Treatment (switched)', linewidth=2, markersize=6)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model 1: Basic DiD (no fixed effects)
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='HC1')
print("Model 1: Basic DiD")
print(model1.summary())

# Model 2: DiD with branch fixed effects
# We need to create dummy variables for branches
model2 = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n" + "="*80)
print("Model 2: DiD with Branch Fixed Effects (clustered SEs)")
print(model2.summary())

# Model 3: DiD with branch and week fixed effects
model3 = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n" + "="*80)
print("Model 3: DiD with Branch and Week Fixed Effects (clustered SEs)")
print(model3.summary())

# Model 4: DiD with controls (staffing and visits)
model4 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# For panel data with branch FEs, switched_branch is absorbed (time-invariant)
# So we estimate: service_score = α_i + γ_t + β*treatment + ε
# where α_i are branch FEs and γ_t are week FEs

# Model with branch FE only (dropping switched_branch as it's collinear)
model_branch_fe = ols('service_score ~ after_switch + treatment + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Model: Branch Fixed Effects only")
print("Treatment effect (treatment coefficient):")
print(f"  Coef: {model_branch_fe.params['treatment']:.4f}")
print(f"  Std Err: {model_branch_fe.bse['treatment']:.4f}")
print(f"  t-stat: {model_branch_fe.tvalues['treatment']:.4f}")
print(f"  p-value: {model_branch_fe.pvalues['treatment']:.4f}")
print(f"  95% CI: [{model_branch_fe.conf_int().loc['treatment'][0]:.4f}, {model_branch_fe.conf_int().loc['treatment'][1]:.4f}]")

# Model with branch and week FE
model_full_fe = ols('service_score ~ after_switch + treatment + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel: Branch + Week Fixed Effects")

# Model with controls

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate baseline (pre-switch) average for context
baseline_avg = df[df['week'] < 15]['service_score'].mean()
print(f"Baseline average service score (pre-switch): {baseline_avg:.2f}")
print(f"Treatment effect: ~2.1 points")
print(f"Treatment effect as % of baseline: {2.1/baseline_avg*100:.1f}%")

# Event study - estimate effect for each week relative to switch
# Create week indicators relative to switch (week 15 is the switch)
df['relative_week'] = df['week'] - 15

# Create dummies for each week interacted with treatment
weeks = sorted(df['relative_week'].unique())
print(f"\nRelative weeks: {weeks}")

# For event study, we'll estimate separate treatment effects for each week
# Reference period: week -1 (week 14, just before switch)
event_study_results = []

for w in weeks:
    if w == -1:  # reference period
        continue
    df[f'treat_week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
event_vars = ' + '.join([f'treat_week_{w}' for w in weeks if w != -1])
event_formula = f'service_score ~ {event_vars} + C(branch_id) + C(week)'
event_model = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nEvent Study Results (coefficients relative to week 14):")
for w in weeks:
    if w == -1:
        continue
    var_name = f'treat_week_{w}'
    coef = event_model.params[var_name]
    se = event_model.bse[var_name]
    pval = event_model.pvalues[var_name]
    ci_low, ci_high = event_model.conf_int().loc[var_name]
    period = "Pre" if w < 0 else "Post"
    print(f"  Week {w+15} ({period}): coef={coef:.2f}, SE={se:.2f}, 95%CI=[{ci_low:.2f}, {ci_high:.2f}], p={pval:.3f}")

# Plot event study
plt.figure(figsize=(10, 6))
pre_weeks = [w for w in weeks if w < -1]
post_weeks = [w for w in weeks if w > -1]

pre_coefs = [event_model.params[f'treat_week_{w}'] for w in pre_weeks]
pre_se = [event_model.bse[f'treat_week_{w}'] for w in pre_weeks]
pre_week_labels = [w+15 for w in pre_weeks]

post_coefs = [event_model.params[f'treat_week_{w}'] for w in post_weeks]
post_se = [event_model.bse[f'treat_week_{w}'] for w in post_weeks]
post_week_labels = [w+15 for w in post_weeks]

# Reference line at 0
plt.axhline(y=0, color='black', linestyle='-', linewidth=1)
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (week 15)')

# Pre-treatment coefficients
if pre_weeks:
    plt.errorbar(pre_week_labels, pre_coefs, yerr=[1.96*s for s in pre_se], 
                 fmt='o', color='blue', label='Pre-switch', capsize=5, markersize=8)

# Post-treatment coefficients  
plt.errorbar(post_week_labels, post_coefs, yerr=[1.96*s for s in post_se], 
             fmt='s', color='green', label='Post-switch', capsize=5, markersize=8)

plt.xlabel('Week')
plt.ylabel('Treatment Effect (vs week 14)')
plt.title('Event Study: Treatment Effects by Week')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as event_study.png")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study - estimate effect for each week relative to switch
# Create week indicators relative to switch (week 15 is the switch)
df['relative_week'] = df['week'] - 15

# Create dummies for each week interacted with treatment
# Rename to avoid negative numbers in variable names
weeks = sorted(df['relative_week'].unique())
print(f"Relative weeks: {weeks}")

for w in weeks:
    if w == -1:  # reference period (week 14)
        continue
    # Use w+100 to avoid negative numbers in variable names
    df[f'treat_w{w+100}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
event_vars = ' + '.join([f'treat_w{w+100}' for w in weeks if w != -1])
event_formula = f'service_score ~ {event_vars} + C(branch_id) + C(week)'
event_model = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nEvent Study Results (coefficients relative to week 14):")
pre_effects = []
post_effects = []
for w in weeks:
    if w == -1:
        continue
    var_name = f'treat_w{w+100}'
    coef = event_model.params[var_name]
    se = event_model.bse[var_name]
    pval = event_model.pvalues[var_name]
    ci_low, ci_high = event_model.conf_int().loc[var_name]
    period = "Pre" if w < 0 else "Post"
    print(f"  Week {w+15} ({period}): coef={coef:.2f}, SE={se:.2f}, 95%CI=[{ci_low:.2f}, {ci_high:.2f}], p={pval:.3f}")
    if w < 0:
        pre_effects.append((w+15, coef, se))
    else:
        post_effects.append((w+15, coef, se))

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main DiD model
model = ols('service_score ~ after_switch + treatment + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("MAIN DID RESULTS")
print("=" * 50)
print(f"Treatment coefficient: {model.params['treatment']:.4f}")
print(f"Standard error: {model.bse['treatment']:.4f}")
print(f"95% CI: [{model.conf_int().loc['treatment'][0]:.4f}, {model.conf_int().loc['treatment'][1]:.4f}]")
print(f"p-value: {model.pvalues['treatment']:.4f}")

# Context
baseline = df[df['week'] < 15]['service_score'].mean()
std_dev = df['service_score'].std()
effect_size_cohens_d = model.params['treatment'] / std_dev

print(f"\nCONTEXT")
print("=" * 50)
print(f"Baseline mean service score: {baseline:.2f}")
print(f"Overall standard deviation: {std_dev:.2f}")
print(f"Effect as % of baseline: {model.params['treatment']/baseline*100:.2f}%")
print(f"Effect size (Cohen's d): {effect_size_cohens_d:.3f}")

# What's a meaningful change?
print(f"\nEFFECT SIZE INTERPRETATION")
print("=" * 50)
print(f"Cohen's d of {effect_size_cohens_d:.3f} is typically considered:")
if effect_size_cohens_d < 0.2:
    print("  SMALL effect (below conventional threshold for 'small')")
elif effect_size_cohens_d < 0.5:
    print("  SMALL to MODERATE effect")
elif effect_size_cohens_d < 0.8:
    print("  MODERATE effect")
else:
    print("  LARGE effect")
