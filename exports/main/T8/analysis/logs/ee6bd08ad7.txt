import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = β0 + β1*switched_branch + β2*after_switch + β3*(switched_branch * after_switch) + ε
# β3 is the treatment effect

model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("Difference-in-Differences Regression Results:")
print("="*60)
print(model.summary())

# Also run with branch fixed effects to control for time-invariant branch characteristics
print("\n\n" + "="*60)
print("With Branch Fixed Effects:")
print("="*60)

# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_vars = [col for col in branch_dummies.columns]
formula = f"service_score ~ switched_branch * after_switch + {' + '.join(branch_vars)}"

print(model_fe.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster

df = pd.read_csv('branches.csv')

# Run DiD regression with cluster-robust standard errors
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()

# Cluster-robust standard errors at branch level
cov_clustered = cov_cluster(model, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))

print("Difference-in-Differences with Cluster-Robust Standard Errors:")
print("="*70)
print(f"{'Variable':<30} {'Coefficient':>12} {'Std Err':>12} {'t-stat':>10} {'P-value':>10}")
print("-"*70)

for var in model.params.index:
    coef = model.params[var]
    se = se_clustered[model.params.index.get_loc(var)]
    t_stat = coef / se
    # Approximate p-value using normal distribution
    from scipy import stats
    p_val = 2 * (1 - stats.norm.cdf(abs(t_stat)))
    print(f"{var:<30} {coef:>12.3f} {se:>12.3f} {t_stat:>10.3f} {p_val:>10.4f}")

print("-"*70)

# Key result
did_coef = model.params['switched_branch:after_switch']
did_se = se_clustered[model.params.index.get_loc('switched_branch:after_switch')]
print(f"\nTreatment Effect: {did_coef:.2f} points (SE: {did_se:.3f})")
print(f"95% CI: [{did_coef - 1.96*did_se:.2f}, {did_coef + 1.96*did_se:.2f}]")

# Calculate effect size relative to baseline
baseline_score = df[df['after_switch']==0]['service_score'].mean()
pct_change = (did_coef / baseline_score) * 100
print(f"\nBaseline average score: {baseline_score:.1f}")
print(f"Percentage improvement: {pct_change:.1f}%")
