import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is our treatment effect

df['treatment_interaction'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment_interaction', data=df).fit()
print("=== Difference-in-Differences Regression ===\n")
print(model.summary())

# Get the treatment effect and confidence interval
treatment_effect = model.params['treatment_interaction']
treatment_se = model.bse['treatment_interaction']
treatment_ci_low = treatment_effect - 1.96 * treatment_se
treatment_ci_high = treatment_effect + 1.96 * treatment_se

print(f"\n=== Treatment Effect Summary ===")
print(f"  Estimate: {treatment_effect:.3f}")
print(f"  Std Error: {treatment_se:.3f}")
print(f"  95% CI: [{treatment_ci_low:.3f}, {treatment_ci_high:.3f}]")
print(f"  P-value: {model.pvalues['treatment_interaction']:.4f}")

import pandas as pd

df = pd.read_csv('branches.csv')

# Check parallel trends: compare trends before the switch (weeks 6-14)
print("=== Parallel Trends Check (Pre-Switch Weeks 6-14) ===\n")

pre_switch = df[df['week'] < 15].copy()

# Calculate weekly averages for each group
weekly_avg = pre_switch.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_avg.columns = ['Control', 'Treatment']
weekly_avg['Difference'] = weekly_avg['Treatment'] - weekly_avg['Control']
print("Weekly averages before switch:")
print(weekly_avg)

# Check if the difference is stable over time (parallel trends)
pre_switch['week_centered'] = pre_switch['week'] - pre_switch['week'].mean()
pre_switch['treatment_interaction'] = pre_switch['switched_branch'] * pre_switch['week_centered']

# Test if treatment group had different trend before switch
trend_model = ols('service_score ~ switched_branch + week_centered + treatment_interaction', data=pre_switch).fit()
print(f"\nPre-switch trend difference coefficient: {trend_model.params['treatment_interaction']:.4f}")
print(f"P-value: {trend_model.pvalues['treatment_interaction']:.4f}")
print("(If p > 0.05, parallel trends assumption is reasonable)")

# This controls for time-invariant branch characteristics
df = df.copy()
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df, branch_dummies], axis=1)

# Build formula with branch fixed effects
fe_cols = [c for c in df_with_fe.columns if c.startswith('branch')]
formula = 'service_score ~ switched_branch + after_switch + treatment_interaction + ' + ' + '.join(fe_cols)

import pandas as pd
import numpy as np
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# Manual fixed effects: demean by branch
df = df.copy()

# Calculate branch means
branch_means = df.groupby('branch_id')['service_score'].transform('mean')
df['service_score_demeaned'] = df['service_score'] - branch_means

# Demean the other variables too
df['switched_branch_demeaned'] = df['switched_branch'] - df.groupby('branch_id')['switched_branch'].transform('mean')
df['after_switch_demeaned'] = df['after_switch'] - df.groupby('branch_id')['after_switch'].transform('mean')
df['treatment_interaction'] = df['switched_branch'] * df['after_switch']
df['treatment_interaction_demeaned'] = df['treatment_interaction'] - df.groupby('branch_id')['treatment_interaction'].transform('mean')

# Run regression on demeaned data (this is equivalent to fixed effects)
X = df[['switched_branch_demeaned', 'after_switch_demeaned', 'treatment_interaction_demeaned']]
X = sm.add_constant(X)
y = df['service_score_demeaned']

model_fe = sm.OLS(y, X).fit()

print("=== DiD with Branch Fixed Effects (Within Estimator) ===\n")
print(model_fe.summary())

treatment_effect_fe = model_fe.params['treatment_interaction_demeaned']
treatment_se_fe = model_fe.bse['treatment_interaction_demeaned']
treatment_ci_low_fe = treatment_effect_fe - 1.96 * treatment_se_fe
treatment_ci_high_fe = treatment_effect_fe + 1.96 * treatment_se_fe

print(f"\n=== Treatment Effect Summary (with Branch FE) ===")
print(f"  Estimate: {treatment_effect_fe:.3f}")
print(f"  Std Error: {treatment_se_fe:.3f}")
print(f"  95% CI: [{treatment_ci_low_fe:.3f}, {treatment_ci_high_fe:.3f}]")
print(f"  P-value: {model_fe.pvalues['treatment_interaction_demeaned']:.4f}")
