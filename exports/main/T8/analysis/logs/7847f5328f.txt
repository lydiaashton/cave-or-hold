import pandas as pd
import numpy as np
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# More comprehensive DiD with branch fixed effects
# Create branch dummies
df = df.copy()
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)

# Main variables
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run regression with branch fixed effects
X = df[['switched_branch', 'after_switch', 'treatment']].copy()
X = pd.concat([X, branch_dummies], axis=1)
y = df['service_score']

model_fe = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("DiD with Branch Fixed Effects:")
print("Treatment coefficient:", model_fe.params['treatment'])
print("Treatment p-value:", model_fe.pvalues['treatment'])
print()

# Also add week fixed effects
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
X2 = df[['switched_branch', 'after_switch', 'treatment']].copy()
X2 = pd.concat([X2, branch_dummies, week_dummies], axis=1)

model_fe_week = sm.OLS(y, X2).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("DiD with Branch and Week Fixed Effects:")

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Calculate branch-level changes
branch_pre = df[df['after_switch'] == 0].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_pre.columns = ['branch_id', 'switched_branch', 'pre_score']

branch_post = df[df['after_switch'] == 1].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_post.columns = ['branch_id', 'switched_branch', 'post_score']

branch_change = branch_pre.merge(branch_post, on=['branch_id', 'switched_branch'])
branch_change['change'] = branch_change['post_score'] - branch_change['pre_score']

print("Branch-level changes summary:")
print(branch_change.groupby('switched_branch')['change'].describe())
print()

# Check if there's correlation with baseline performance
branch_change['baseline'] = branch_change['pre_score']
correlation = branch_change['baseline'].corr(branch_change['change'])
print(f"Correlation between baseline score and change: {correlation:.3f}")

# Check if lower-performing branches benefited more
low_baseline = branch_change[branch_change['baseline'] < branch_change['baseline'].median()]
