import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch'] * df['after_switch']

# Basic DiD: 2x2 table
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES ANALYSIS")
print("=" * 60)

# Calculate means for the 4 groups
groups = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean()
print("\nAverage Service Scores by Group:")
print(groups)
print()

# Calculate the DiD estimate manually
control_before = groups.loc[(0, 0)]
control_after = groups.loc[(0, 1)]
treated_before = groups.loc[(1, 0)]
treated_after = groups.loc[(1, 1)]

control_change = control_after - control_before
treated_change = treated_after - treated_before
did_estimate = treated_change - control_change

print(f"Control group (non-switched): {control_before:.2f} -> {control_after:.2f} (change: {control_change:.2f})")
print(f"Treated group (switched): {treated_before:.2f} -> {treated_after:.2f} (change: {treated_change:.2f})")
print(f"\nDifference-in-Differences estimate: {did_estimate:.3f}")
print()

# Run regression with cluster-robust standard errors
print("=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary())

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary())

# Cluster-robust standard errors by branch
model3 = ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 3: DiD with controls (cluster-robust SE by branch)")
print(model3.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Let's look at the raw trends more carefully
print("=" * 60)
print("RAW TRENDS ANALYSIS")
print("=" * 60)

# Average by week and group
trends = df.groupby(['week', 'switched_branch'])['service_score'].agg(['mean', 'std', 'count']).unstack('switched_branch')
print("\nWeekly averages by group:")
print(trends)

# Plot the trends
fig, ax = plt.subplots(figsize=(12, 6))

weeks = range(6, 23)
control_means = [df[(df['week']==w) & (df['switched_branch']==0)]['service_score'].mean() for w in weeks]
treated_means = [df[(df['week']==w) & (df['switched_branch']==1)]['service_score'].mean() for w in weeks]

ax.plot(weeks, control_means, 'o-', label='Control (No Switch)', linewidth=2, markersize=8)
ax.plot(weeks, treated_means, 's-', label='Treated (Switched)', linewidth=2, markersize=8)
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (Week 15)')
ax.set_xlabel('Week')
ax.set_ylabel('Average Service Score')
ax.set_title('Service Score Trends: Control vs Treated Branches')
ax.legend()
ax.grid(True, alpha=0.3)
plt.savefig('trends_plot.png', dpi=150, bbox_inches='tight')
print("\nPlot saved as trends_plot.png")

# Proper event study - interact week dummies with treatment, omit week 14 as baseline
print("\n" + "=" * 60)
print("PROPER EVENT STUDY")
print("=" * 60)

# Create treatment indicator
df['treated_group'] = df['switched_branch']

# Create week dummies
for w in range(6, 23):
    df[f'w{w}'] = (df['week'] == w).astype(int)

# Create interactions (treatment group x week), omit week 14
for w in range(6, 23):
    if w != 14:
        df[f'treated_x_w{w}'] = df['treated_group'] * df[f'w{w}']

# Build formula - include treatment group main effect, week dummies, and interactions
# Omit week 14 as reference
pre_weeks = [f'treated_x_w{w}' for w in range(6, 14)]
post_weeks = [f'treated_x_w{w}' for w in range(15, 23)]
week_dummies_list = [f'w{w}' for w in range(6, 14)] + [f'w{w}' for w in range(15, 23)]

formula = 'service_score ~ treated_group + ' + ' + '.join(week_dummies_list) + ' + ' + ' + '.join(pre_weeks + post_weeks)

model_es = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nEvent Study Coefficients (treated vs control, relative to week 14):")
print(f"{'Week':<6} {'Coefficient':<12} {'Std Error':<12} {'P-value':<10}")
print("-" * 50)

coef_dict = {}
for week in range(6, 23):
    if week == 14:
        coef_dict[week] = {'coef': 0, 'se': 0, 'pvalue': 1, 'note': '(reference)'}
    else:
        coef_name = f'treated_x_w{week}'
        if coef_name in model_es.params:
            coef_dict[week] = {
                'coef': model_es.params[coef_name],
                'se': model_es.bse[coef_name],
                'pvalue': model_es.pvalues[coef_name]
            }

for week in range(6, 23):
    c = coef_dict[week]
    if week == 14:
        print(f"{week:<6} {'(ref)':<12} {'-':<12} {'-':<10}")
    else:
        marker = "***" if c['pvalue'] < 0.01 else ("*" if c['pvalue'] < 0.05 else "")
        print(f"{week:<6} {c['coef']:<12.3f} {c['se']:<12.3f} {c['pvalue']:<10.4f} {marker}")

# Pre-trends test
print("\n" + "=" * 60)
print("PRE-TRENDS F-TEST")
print("=" * 60)

pre_period_coeffs = [f'treated_x_w{w}' for w in range(6, 14)]
