import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Simple DiD Regression:")
print(model.summary())

# Now with branch fixed effects (to control for time-invariant branch characteristics)
# We'll use dummy variables for each branch
print("\n\n" + "="*60)
print("DiD with Branch Fixed Effects:")
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Build formula with branch dummies
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_')]
formula = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(branch_cols)
model_fe = ols(formula, data=df_with_dummies).fit()
print(f"R-squared: {model_fe.rsquared:.4f}")
print(f"\nKey coefficients:")
print(model_fe.params[['switched_branch', 'after_switch', 'treatment']])
print(f"\nStandard errors:")
print(model_fe.bse[['switched_branch', 'after_switch', 'treatment']])
print(f"\nP-values:")
print(model_fe.pvalues[['switched_branch', 'after_switch', 'treatment']])

import pandas as pd
import numpy as np
from linearmodels.panel import PanelOLS

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']
df_panel = df.copy().set_index(['branch_id', 'week'])

# Main DiD result
mod = PanelOLS(df_panel['service_score'], 
               df_panel[['after_switch', 'treatment']], 
               entity_effects=True)
result = mod.fit(cov_type='clustered', cluster_entity=True)

print("MAIN DiD RESULT")
print("="*60)
print(f"Treatment effect: {result.params['treatment']:.3f}")
print(f"Standard error: {result.std_errors['treatment']:.3f}")
print(f"95% CI: [{result.conf_int.iloc[1,0]:.3f}, {result.conf_int.iloc[1,1]:.3f}]")
print(f"P-value: {result.pvalues['treatment']:.4f}")

# Effect size in practical terms
print("\n\nPRACTICAL SIGNIFICANCE")
print("="*60)
overall_mean = df['service_score'].mean()
overall_std = df['service_score'].std()
effect = result.params['treatment']
print(f"Overall mean service score: {overall_mean:.2f}")
print(f"Overall std deviation: {overall_std:.2f}")
print(f"Effect size (Cohen's d): {effect/overall_std:.3f}")
print(f"Effect as % of mean: {effect/overall_mean*100:.1f}%")

# Distribution of scores
print("\n\nSCORE DISTRIBUTION")
print("="*60)
print(f"Min score: {df['service_score'].min():.2f}")
print(f"25th percentile: {df['service_score'].quantile(0.25):.2f}")
print(f"Median: {df['service_score'].median():.2f}")
print(f"75th percentile: {df['service_score'].quantile(0.75):.2f}")
print(f"Max score: {df['service_score'].max():.2f}")

# How many branches actually improved?
print("\n\nBRANCH-LEVEL ANALYSIS")
print("="*60)

# Calculate before/after for each branch
before = df[df['after_switch'] == 0].groupby('branch_id')['service_score'].mean()
after = df[df['after_switch'] == 1].groupby('branch_id')['service_score'].mean()
changes = after - before

# By treatment status
treatment_branches = df[df['switched_branch'] == 1]['branch_id'].unique()
control_branches = df[df['switched_branch'] == 0]['branch_id'].unique()

treatment_changes = changes[changes.index.isin(treatment_branches)]
control_changes = changes[changes.index.isin(control_branches)]

print(f"Treatment branches (n={len(treatment_changes)}):")
print(f"  Mean change: {treatment_changes.mean():.2f}")
print(f"  % that improved: {(treatment_changes > 0).mean()*100:.1f}%")
print(f"  % with meaningful improvement (>1 point): {(treatment_changes > 1).mean()*100:.1f}%")

print(f"\nControl branches (n={len(control_changes)}):")
print(f"  Mean change: {control_changes.mean():.2f}")
print(f"  % that improved: {(control_changes > 0).mean()*100:.1f}%")
print(f"  % with meaningful improvement (>1 point): {(control_changes > 1).mean()*100:.1f}%")

# Check if there's heterogeneity
print("\n\nHETEROGENEITY CHECK")
print("="*60)
treatment_improved = treatment_changes[treatment_changes > 0]
treatment_declined = treatment_changes[treatment_changes <= 0]
print(f"Treatment branches that improved: {len(treatment_improved)} ({len(treatment_improved)/len(treatment_changes)*100:.1f}%)")
print(f"Treatment branches that declined or stayed same: {len(treatment_declined)} ({len(treatment_declined)/len(treatment_changes)*100:.1f}%)")
