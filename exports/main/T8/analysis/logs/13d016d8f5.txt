import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create treatment interaction
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run a branch fixed effects model (more robust to time-invariant differences)
# This absorbs the switched_branch term and focuses on within-branch changes
print("=== Branch Fixed Effects Model ===")
fe_model = ols('service_score ~ after_switch + treatment + staffing_index + visits_index', 
               data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(fe_model.summary())

# Alternative: branch and week fixed effects
print("\n=== Branch + Week Fixed Effects Model ===")
df['week_str'] = df['week'].astype(str)
fe_week_model = ols('service_score ~ treatment + staffing_index + visits_index + C(week_str)', 
                    data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(fe_week_model.summary())

# Event study: estimate effect for each week relative to week 14 (last pre-switch week)
print("\n=== Event Study Analysis ===")
df['relative_week'] = df['week'] - 14  # Week 14 is reference (0)
df['relative_week'] = df['relative_week'].apply(lambda x: max(-8, min(8, x)))  # Cap at -8 to 8

# Create dummies for each relative week interacted with treatment
for rw in range(-8, 9):
    if rw != 0:  # Skip reference week
        df[f'rel_week_{rw}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (excluding reference week 0)
event_vars = '+'.join([f'rel_week_{rw}' for rw in range(-8, 9) if rw != 0])
event_formula = f'service_score ~ {event_vars} + C(week_str)'
event_model = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for treatment weeks
print("\nEvent Study Coefficients (relative to week 14):")
for rw in range(-8, 9):
    if rw != 0:
        coef_name = f'rel_week_{rw}'
        if coef_name in event_model.params:
            coef = event_model.params[coef_name]
            se = event_model.bse[coef_name]
            pval = event_model.pvalues[coef_name]
            sig = '***' if pval < 0.01 else '**' if pval < 0.05 else '*' if pval < 0.1 else ''
            week_label = f"Week {14+rw}"
            period = "PRE" if rw < 0 else "POST"
            print(f"  {week_label} ({period}): {coef:.2f} (SE: {se:.2f}, p: {pval:.3f}) {sig}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create treatment interaction
df['treatment'] = df['switched_branch'] * df['after_switch']

# Event study with renamed variables
df['relative_week'] = df['week'] - 14  # Week 14 is reference (0)

# Create dummies for each relative week interacted with treatment
event_cols = []
for rw in range(-8, 9):
    if rw != 0:  # Skip reference week
        col_name = f'rel_wk_{abs(rw)}_{"neg" if rw < 0 else "pos"}'
        df[col_name] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)
        event_cols.append(col_name)

# Run event study regression
import statsmodels.api as sm
from statsmodels.formula.api import ols

event_formula = f"service_score ~ {' + '.join(event_cols)} + C(week_str)"
event_model = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract and organize coefficients
results = []
for rw in range(-8, 9):
    if rw != 0:
        col_name = f'rel_wk_{abs(rw)}_{"neg" if rw < 0 else "pos"}'
        coef = event_model.params[col_name]
        se = event_model.bse[col_name]
        pval = event_model.pvalues[col_name]
        results.append({
            'relative_week': rw,
            'week': 14 + rw,
            'coef': coef,
            'se': se,
            'pval': pval,
            'period': 'PRE' if rw < 0 else 'POST'
        })

results_df = pd.DataFrame(results)
print("=== Event Study Results ===")
print(results_df.to_string(index=False))

# Plot event study
fig, ax = plt.subplots(figsize=(14, 7))

# Plot coefficients
pre = results_df[results_df['period'] == 'PRE']
post = results_df[results_df['period'] == 'POST']

ax.errorbar(pre['week'], pre['coef'], yerr=1.96*pre['se'], fmt='o', capsize=5, 
            label='Pre-Switch', color='blue', markersize=8, linewidth=2)
ax.errorbar(post['week'], post['coef'], yerr=1.96*post['se'], fmt='s', capsize=5, 
            label='Post-Switch', color='red', markersize=8, linewidth=2)

ax.axvspan(6, 14.5, alpha=0.15, color='blue')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Effect Relative to Week 14 (Treatment vs Control)', fontsize=12)
ax.set_title('Event Study: Treatment Effect by Week\n(Reference: Week 14, Control Group Baseline)', fontsize=14, fontweight='bold')
ax.legend(loc='upper left')

print("\nEvent study plot saved as 'event_study.png'")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Check pre-trends more carefully
print("=== PRE-TREND ANALYSIS ===\n")

# Weekly gaps before switch
pre_data = df[df['week'] < 15]
weekly_gaps_pre = pre_data.groupby('week').apply(
    lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()
)
print("Pre-switch weekly gaps (Treatment - Control):")
for week, gap in weekly_gaps_pre.items():
    print(f"  Week {week}: {gap:.2f}")

print(f"\nAverage pre-switch gap: {weekly_gaps_pre.mean():.2f}")
print(f"Pre-switch gap trend (slope): ", end="")

# Regression on pre-period only to test trend
pre_data = df[df['week'] < 15].copy()
pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()
pre_data['treatment'] = pre_data['switched_branch'] * pre_data['week_centered']

trend_model = ols('service_score ~ switched_branch + week_centered + treatment', data=pre_data).fit(
    cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']}
)
print(f"{trend_model.params['treatment']:.3f} points/week (p={trend_model.pvalues['treatment']:.4f})")

print("\n" + "="*60)
print("KEY CONCERN: Non-parallel pre-trends")
print("="*60)
print(f"""
The treatment branches were already OUTPERFORMING control branches
BEFORE the switch, and the gap was GROWING over time.

Pre-switch gap: averaged {weekly_gaps_pre.mean():.2f} points
Gap growth rate: {trend_model.params['treatment']:.3f} points per week (p={trend_model.pvalues['treatment']:.4f})

This violates the parallel trends assumption required for clean
causal inference. The observed "treatment effect" may be partly
or wholly due to pre-existing differences.""")

# Post-period gaps
print("\n" + "="*60)
print("POST-PERIOD ANALYSIS")
print("="*60)

post_data = df[df['week'] >= 15]
weekly_gaps_post = post_data.groupby('week').apply(
    lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()
)
print("\nPost-switch weekly gaps (Treatment - Control):")
for week, gap in weekly_gaps_post.items():
    print(f"  Week {week}: {gap:.2f}")

print(f"\nAverage post-switch gap: {weekly_gaps_post.mean():.2f}")
print(f"Gap increase from pre to post: {weekly_gaps_post.mean() - weekly_gaps_pre.mean():.2f} points")

# Conservative estimate accounting for pre-trend
print("\n" + "="*60)
print("CONSERVATIVE EFFECT ESTIMATE")
print("="*60)

# If pre-trend continued, what would we expect?
pre_trend_slope = trend_model.params['treatment']
weeks_post = 8  # weeks 15-22
expected_gap_growth = pre_trend_slope * weeks_post / 2  # average over post period
observed_gap_increase = weekly_gaps_post.mean() - weekly_gaps_pre.mean()

print(f"""
If the pre-existing trend continued naturally:
  - Expected gap growth: ~{expected_gap_growth:.2f} points
  
Observed gap growth:
  - Actual gap increase: {observed_gap_increase:.2f} points
  
Conservative treatment effect estimate:
  - {observed_gap_increase - expected_gap_growth:.2f} points (possibly zero or negative)
""")

# Check if post-period gap is significantly different from pre-period
print("\n" + "="*60)
print("STATISTICAL TEST: Did the gap truly change?")
print("="*60)

df['post_period'] = (df['week'] >= 15).astype(int)
df['gap_change'] = df['switched_branch'] * df['post_period']

full_model = ols('service_score ~ switched_branch + post_period + gap_change', data=df).fit(
    cov_type='cluster', cov_kwds={'groups': df['branch_id']}
)
print(f"""
DiD coefficient (gap_change): {full_model.params['gap_change']:.3f}
Standard error: {full_model.bse['gap_change']:.3f}
95% CI: [{full_model.params['gap_change'] - 1.96*full_model.bse['gap_change']:.2f}, {full_model.params['gap_change'] + 1.96*full_model.bse['gap_change']:.2f}]
p-value: {full_model.pvalues['gap_change']:.6f}
""")
