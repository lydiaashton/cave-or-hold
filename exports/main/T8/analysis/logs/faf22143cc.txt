import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks relative to switch at week 15)
df['relative_week'] = df['week'] - 15

# Event study - create dummies for each week relative to treatment
# Reference period: week -1 (week 14)
df['week_6'] = (df['relative_week'] == -9).astype(int)
df['week_7'] = (df['relative_week'] == -8).astype(int)
df['week_8'] = (df['relative_week'] == -7).astype(int)
df['week_9'] = (df['relative_week'] == -6).astype(int)
df['week_10'] = (df['relative_week'] == -5).astype(int)
df['week_11'] = (df['relative_week'] == -4).astype(int)
df['week_12'] = (df['relative_week'] == -3).astype(int)
df['week_13'] = (df['relative_week'] == -2).astype(int)
# week_14 is reference (-1)
df['week_15'] = (df['relative_week'] == 0).astype(int)
df['week_16'] = (df['relative_week'] == 1).astype(int)
df['week_17'] = (df['relative_week'] == 2).astype(int)
df['week_18'] = (df['relative_week'] == 3).astype(int)
df['week_19'] = (df['relative_week'] == 4).astype(int)
df['week_20'] = (df['relative_week'] == 5).astype(int)
df['week_21'] = (df['relative_week'] == 6).astype(int)
df['week_22'] = (df['relative_week'] == 7).astype(int)

# Event study regression with treatment interaction
event_vars = ['week_6', 'week_7', 'week_8', 'week_9', 'week_10', 'week_11', 'week_12', 'week_13',
              'week_15', 'week_16', 'week_17', 'week_18', 'week_19', 'week_20', 'week_21', 'week_22']

formula = 'service_score ~ switched_branch + ' + ' + '.join([f'switched_branch * {v}' for v in event_vars])
# This is getting complex, let me simplify

# Simpler approach: just interact treatment with relative week dummies
event_dummies = ['week_6', 'week_7', 'week_8', 'week_9', 'week_10', 'week_11', 'week_12', 'week_13',
                 'week_15', 'week_16', 'week_17', 'week_18', 'week_19', 'week_20', 'week_21', 'week_22']

# Create interaction terms
for var in event_dummies:
    df[f'{var}_treated'] = df[var] * df['switched_branch']

# Event study formula (excluding reference week 14)
interactions = [f'{var}_treated' for var in event_dummies]
formula = 'service_score ~ switched_branch + ' + ' + '.join(event_dummies) + ' + ' + ' + '.join(interactions)

event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Results (coefficients on interaction terms):")
for var in event_dummies:
    coef_name = f'{var}_treated'
    if coef_name in event_model.params.index:
        coef = event_model.params[coef_name]
        se = event_model.bse[coef_name]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        print(f"Week {var.replace('week_', '')}: coef={coef:.2f}, 95% CI=[{ci_low:.2f}, {ci_high:.2f}]")

# Branch fixed effects model
print("\n\nBranch Fixed Effects Model:")
df['branch_id_str'] = df['branch_id'].astype(str)
fe_model = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index + C(branch_id_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(fe_model.summary().tables[1])

pre_df['week_centered'] = pre_df['week'] - 10  # center around middle of pre-period
