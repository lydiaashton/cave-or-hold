import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Check how many branches switched vs didn't
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Check balance of observations
print("\nObservations by switch status:")
print(df.groupby('switched_branch').size())

# Check service score averages by group and period
print("\nService score by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(pivot)

# Calculate the difference-in-differences manually
# Before period (after_switch=0)
before_treated = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()

# After period (after_switch=1)
after_treated = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print(f"\nBefore - Treated mean: {before_treated:.3f}, Control mean: {before_control:.3f}")
print(f"After - Treated mean: {after_treated:.3f}, Control mean: {after_control:.3f}")

# Difference in differences
diff_treated = after_treated - before_treated
diff_control = after_control - before_control
did = diff_treated - diff_control

print(f"\nChange in treated: {diff_treated:.3f}")
print(f"Change in control: {diff_control:.3f}")
print(f"Difference-in-Differences estimate: {did:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run difference-in-differences regression
# Basic DiD: service_score ~ switched_branch * after_switch
model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Basic DiD Regression (with cluster-robust SEs):")
print(model1.summary())

# Add controls for staffing and visits
model2 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with controls:")
print(model2.summary())

# Add branch fixed effects
# Create branch dummies
df_with_fe = df.copy()
df_with_fe['branch_id'] = df_with_fe['branch_id'].astype('category')

model3 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index + C(branch_id)', data=df_with_fe).fit(cov_type='cluster', cov_kwds={'groups': df_with_fe['branch_id']})
print("\n\nDiD with Branch Fixed Effects:")
print(model3.summary())

# Create week dummies and interactions
df_es = df.copy()
for w in range(6, 23):
    df_es[f'week_{w}'] = (df_es['week'] == w).astype(int)
    df_es[f'switched_x_week_{w}'] = df_es['switched_branch'] * df_es[f'week_{w}']

week_terms = ' + '.join([f'week_{w}' for w in weeks_to_include])

formula = f'service_score ~ {interaction_terms} + {week_terms} + staffing_index + visits_index'
