import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Count branches by treatment status
print("Branches by treatment status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate average service scores by group and period
print("\n\nAverage service scores by group and period:")
df['period'] = np.where(df['after_switch'] == 0, 'Before', 'After')
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='period', 
                       aggfunc='mean')
pivot.columns = ['Before (Weeks 6-14)', 'After (Weeks 15-22)']
pivot['Change'] = pivot['After (Weeks 15-22)'] - pivot['Before (Weeks 6-14)']
print(pivot)

# Calculate the difference-in-differences estimate manually
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treated_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treated_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

control_change = control_after - control_before
treated_change = treated_after - treated_before
did_estimate = treated_change - control_change

print(f"\n\nDifference-in-Differences Calculation:")
print(f"Control group (no switch): {control_before:.2f} -> {control_after:.2f} (change: {control_change:.2f})")
print(f"Treated group (switched): {treated_before:.2f} -> {treated_after:.2f} (change: {treated_change:.2f})")
print(f"DiD Estimate (treatment effect): {did_estimate:.2f}")

# Now run a regression with DiD
print("\n\n=== Regression Analysis ===")

# Basic DiD regression
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("\nBasic DiD Regression:")
print(model.summary())

# DiD with branch fixed effects
print("\n\nDiD with Branch Fixed Effects:")
model_fe = ols('service_score ~ switched_branch * after_switch + C(branch_id)', data=df).fit()
print(model_fe.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: estimate effect for each week relative to switch
# Create week dummies interacted with treatment
df['relative_week'] = df['week'] - 15  # Week 15 is the switch (0), week 14 is -1, etc.

# Create dummies for each week
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=False)
df = pd.concat([df, week_dummies], axis=1)

# Reference week is week 14 (just before switch)
# Run event study regression
week_cols = [col for col in df.columns if col.startswith('week_')]
# Drop week_14 as reference
treatment_cols = [col for col in week_cols if col != 'week_14']

# Create interaction terms
for col in treatment_cols:
    df[f'{col}_treated'] = df[col] * df['switched_branch']

# Event study regression with branch FE would be ideal but let's do a simpler version first
formula = 'service_score ~ switched_branch + ' + ' + '.join(week_cols) + ' + ' + ' + '.join([f'{col}_treated' for col in treatment_cols])
model_event = ols(formula, data=df).fit()

# Extract the interaction coefficients (these are the event study estimates)
event_results = []
for col in treatment_cols:
    coef_name = f'{col}_treated'
    if coef_name in model_event.params:
        week_num = int(col.replace('week_', ''))
        relative = week_num - 14  # relative to week 14
        coef = model_event.params[coef_name]
        se = model_event.bse[coef_name]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        event_results.append({
            'week': week_num,
            'relative_week': relative,
            'coef': coef,
            'se': se,
            'ci_low': ci_low,
            'ci_high': ci_high
        })

event_df = pd.DataFrame(event_results).sort_values('week')
