import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate number of branches in each group
branches_switched = df[df['switched_branch'] == 1]['branch_id'].nunique()
branches_control = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"Branches that switched: {branches_switched}")
print(f"Branches that didn't switch (control): {branches_control}")

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Calculate mean service scores by group and period
print("\n=== Mean Service Scores ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
print(pivot)

# Calculate the DiD estimate manually
pre_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
post_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nTreatment group - Pre: {pre_treatment:.2f}, Post: {post_treatment:.2f}, Change: {post_treatment - pre_treatment:.2f}")
print(f"Control group - Pre: {pre_control:.2f}, Post: {post_control:.2f}, Change: {post_control - pre_control:.2f}")
print(f"\nDifference-in-Differences estimate: {(post_treatment - pre_treatment) - (post_control - pre_control):.2f}")

# Run DiD regression
print("\n=== DiD Regression (Basic) ===")
model1 = ols('service_score ~ switched_branch + after_switch + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model1.summary())

# Run DiD regression with controls
print("\n=== DiD Regression (With Controls) ===")
model2 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model2.summary())

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create week dummies interacted with treatment for pre-trend test
for w in range(6, 15):  # Pre-period weeks
    df[f'pre_week_{w}'] = ((df['week'] == w) & (df['switched_branch'] == 1)).astype(int)

print("=== Pre-trend Test Regression ===")
pre_week_cols = [f'pre_week_{w}' for w in range(6, 15)]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(pre_week_cols)
model_pre = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Show coefficients for pre-trend weeks
print("\nPre-trend week coefficients (testing parallel trends):")
print("(These test whether treatment and control had different trends before the switch)")
for w in range(6, 15):
    coef = model_pre.params[f'pre_week_{w}']
    pval = model_pre.pvalues[f'pre_week_{w}']
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Week {w}: coef={coef:.3f}, p-value={pval:.3f} {sig}")

# Joint test for all pre-trend coefficients
from statsmodels.stats.contrast import ContrastMatrix
import statsmodels.api as sm

# Test if all pre-trend coefficients are jointly zero
R = np.zeros((len(pre_week_cols), len(model_pre.params)))
for i, col in enumerate(pre_week_cols):
    R[i, list(model_pre.params.index).index(col)] = 1

from statsmodels.stats.contrast import WaldTest
wt = model_pre.wald_test(R)
print(f"\nJoint test of all pre-trend coefficients: F-stat={wt.statistic:.2f}, p-value={wt.pvalue:.3f}")

# Also get the main DiD coefficient from this model
print(f"\nMain after_switch coefficient: {model_pre.params['after_switch']:.3f} (p={model_pre.pvalues['after_switch']:.3f})")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Let's do a cleaner event study / dynamic DiD
# Create relative week indicators
df['relative_week'] = df['week'] - 15  # Week 15 is the switch (relative week 0)

# Create indicators for each relative week interacted with treatment
# Using week 14 (relative_week = -1) as the reference period
for rw in range(-9, 8):  # weeks -9 to +7 (week 6 to 22)
    if rw != -1:  # Skip reference period
        df[f'rel_week_{rw}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
rel_week_cols = [f'rel_week_{rw}' for rw in range(-9, 8) if rw != -1]
formula = 'service_score ~ switched_branch + ' + ' + '.join(rel_week_cols)
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study / Dynamic DiD Results ===")
print("(Coefficients show treatment effect relative to week 14 baseline)")
print("\nPre-switch weeks (should be ~0 if parallel trends hold):")
for rw in range(-9, -1):
    coef = model_event.params[f'rel_week_{rw}']
    pval = model_event.pvalues[f'rel_week_{rw}']
    ci_low = model_event.conf_int().loc[f'rel_week_{rw}'][0]
    ci_high = model_event.conf_int().loc[f'rel_week_{rw}'][1]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Relative week {rw:+d} (Week {rw+15}): coef={coef:.2f}, 95% CI=[{ci_low:.2f}, {ci_high:.2f}] {sig}")

print("\nPost-switch weeks (treatment effect):")
for rw in range(0, 8):
    coef = model_event.params[f'rel_week_{rw}']
    pval = model_event.pvalues[f'rel_week_{rw}']
    ci_low = model_event.conf_int().loc[f'rel_week_{rw}'][0]
    ci_high = model_event.conf_int().loc[f'rel_week_{rw}'][1]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Relative week {rw:+d} (Week {rw+15}): coef={coef:.2f}, 95% CI=[{ci_low:.2f}, {ci_high:.2f}] {sig}")

# Calculate average post-treatment effect
post_coefs = [model_event.params[f'rel_week_{rw}'] for rw in range(0, 8)]
print(f"\nAverage post-switch effect: {np.mean(post_coefs):.2f}")

# Create event study plot
fig, ax = plt.subplots(figsize=(12, 6))

rel_weeks = list(range(-9, 8))
coefs = []
ci_lower = []
ci_upper = []

for rw in rel_weeks:
    if rw == -1:
        coefs.append(0)
        ci_lower.append(0)
        ci_upper.append(0)
    else:
        col = f'rel_week_{rw}'
        coefs.append(model_event.params[col])
        ci_lower.append(model_event.conf_int().loc[col][0])
        ci_upper.append(model_event.conf_int().loc[col][1])

ax.plot(rel_weeks, coefs, 'bo-', linewidth=2, markersize=8, label='Treatment Effect')

ax.axvline(x=-0.5, color='red', linestyle='-', linewidth=2, label='Switch Point')
ax.axvspan(0, 7.5, alpha=0.1, color='green', label='Post-Switch Period')

ax.set_xlabel('Relative Week (0 = Switch Week)', fontsize=12)
ax.set_ylabel('Service Score Effect (vs Control)', fontsize=12)
ax.set_title('Event Study: Treatment Effect on Service Scores Over Time', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=10)
ax.grid(True, alpha=0.3)
ax.set_xticks(rel_weeks)

plt.tight_layout()

print("\nEvent study plot saved as 'event_study.png'")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create relative week indicators with valid column names
df['relative_week'] = df['week'] - 15  # Week 15 is the switch (relative week 0)

# Create indicators for each relative week interacted with treatment
# Using week 14 (relative_week = -1) as the reference period
for rw in range(-9, 8):  # weeks -9 to +7 (week 6 to 22)
    if rw != -1:  # Skip reference period
        col_name = f'rel_w{rw:+03d}'  # e.g., rel_w-009, rel_w+007
        df[col_name] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
rel_week_cols = [f'rel_w{rw:+03d}' for rw in range(-9, 8) if rw != -1]
formula = 'service_score ~ switched_branch + ' + ' + '.join(rel_week_cols)
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study / Dynamic DiD Results ===")
print("(Coefficients show treatment effect relative to week 14 baseline)")
print("\nPre-switch weeks (should be ~0 if parallel trends hold):")
pre_coefs = []
for rw in range(-9, -1):
    col = f'rel_w{rw:+03d}'
    coef = model_event.params[col]
    pval = model_event.pvalues[col]
    ci_low = model_event.conf_int().loc[col][0]
    ci_high = model_event.conf_int().loc[col][1]
    pre_coefs.append(coef)
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Relative week {rw:+d} (Week {rw+15}): coef={coef:.2f}, 95% CI=[{ci_low:.2f}, {ci_high:.2f}] {sig}")

print("\nPost-switch weeks (treatment effect):")
post_coefs = []
for rw in range(0, 8):
    col = f'rel_w{rw:+03d}'
    coef = model_event.params[col]
    pval = model_event.pvalues[col]
    ci_low = model_event.conf_int().loc[col][0]
    ci_high = model_event.conf_int().loc[col][1]
    post_coefs.append(coef)
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Relative week {rw:+d} (Week {rw+15}): coef={coef:.2f}, 95% CI=[{ci_low:.2f}, {ci_high:.2f}] {sig}")

print(f"\nAverage pre-switch effect (weeks 6-13): {np.mean(pre_coefs):.2f}")
print(f"Average post-switch effect (weeks 15-22): {np.mean(post_coefs):.2f}")

# Create event study plot
fig, ax = plt.subplots(figsize=(12, 6))

rel_weeks = list(range(-9, 8))
coefs = []
ci_lower = []
ci_upper = []

for rw in rel_weeks:
    if rw == -1:
        coefs.append(0)
        ci_lower.append(0)
        ci_upper.append(0)
    else:
        col = f'rel_w{rw:+03d}'
        coefs.append(model_event.params[col])
        ci_lower.append(model_event.conf_int().loc[col][0])
        ci_upper.append(model_event.conf_int().loc[col][1])

ax.plot(rel_weeks, coefs, 'bo-', linewidth=2, markersize=8, label='Treatment Effect')

ax.legend(loc='best', fontsize=10)
ax.grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")
