import pandas as pd

from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression (no controls)
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

# Cluster-robust standard errors by branch
model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Model with Cluster-Robust SE (by branch) ===")
print(model3.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate average service score by week and treatment group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_score']

# Pivot for easier plotting
weekly_pivot = weekly_avg.pivot(index='week', columns='switched_branch', values='avg_score')
weekly_pivot.columns = ['Control (Old)', 'Treatment (New)']

print("Weekly averages by group:")
print(weekly_pivot)

# Calculate pre-post means for each group
pre_post = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
pre_post.columns = ['Before (Weeks 6-14)', 'After (Weeks 15-22)']
print("\nPre-Post means by group:")
print(pre_post)

# Calculate the DiD manually
control_before = pre_post.loc[0, 'Before (Weeks 6-14)']
control_after = pre_post.loc[0, 'After (Weeks 15-22)']
treatment_before = pre_post.loc[1, 'Before (Weeks 6-14)']
treatment_after = pre_post.loc[1, 'After (Weeks 15-22)']

control_change = control_after - control_before
treatment_change = treatment_after - treatment_before
did_estimate = treatment_change - control_change

print(f"\nControl group change: {control_change:.3f}")
print(f"Treatment group change: {treatment_change:.3f}")
print(f"DiD estimate: {did_estimate:.3f}")

# Plot the trends
plt.figure(figsize=(12, 6))
plt.plot(weekly_pivot.index, weekly_pivot['Control (Old)'], marker='o', label='Control (Old Procedure)', linewidth=2, markersize=8)
plt.plot(weekly_pivot.index, weekly_pivot['Treatment (New)'], marker='s', label='Treatment (New Procedure)', linewidth=2, markersize=8)
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')
plt.xlabel('Week', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Score Trends: Treatment vs Control Groups', fontsize=14, fontweight='bold')
plt.legend(fontsize=11)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("\nPlot saved as service_score_trends.png")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create week dummies to test parallel trends
# Only use pre-treatment period (weeks 6-14)
pre_df = df[df['week'] <= 14].copy()

# Test for differential pre-trends: interact week with treatment in pre-period
# Using week 6 as reference
pre_df['relative_week'] = pre_df['week'] - 6

# Model: test if treatment group had different trend before the switch
model_trends = ols('service_score ~ switched_branch * relative_week', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print("=== Pre-Treatment Trend Test ===")
print(model_trends.summary())

# Also do an event study style analysis
df['event_time'] = df['week'] - 15  # Week 15 is event time 0

# Create event time dummies (relative to week 15)
event_dummies = pd.get_dummies(df['event_time'], prefix='week_rel')
df = pd.concat([df, event_dummies], axis=1)

# Drop the reference category (week -1, just before treatment)
# Run regression with all week dummies except reference
week_cols = [c for c in df.columns if c.startswith('week_rel') and c != 'week_rel_-1']

# Build formula
formula = 'service_score ~ switched_branch + ' + ' + '.join(week_cols)
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n=== Event Study Coefficients (relative to week 14) ===")
event_coefs = [(c, model_event.params[c], model_event.bse[c], model_event.pvalues[c]) 
               for c in week_cols if c.startswith('week_rel')]
event_coefs.sort(key=lambda x: int(x[0].split('_')[-1]))
for coef, est, se, p in event_coefs:
    week_num = int(coef.split('_')[-1]) + 15
    print(f"Week {week_num} (rel {int(coef.split('_')[-1])}): coef={est:.3f}, se={se:.3f}, p={p:.3f}")

# Calculate confidence intervals for the treatment effect
did_model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
did_coef = did_model.params['treatment']
did_se = did_model.bse['treatment']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se
print(f"\n=== Treatment Effect Summary ===")
print(f"Point estimate: {did_coef:.3f}")
print(f"Standard error: {did_se:.3f}")
print(f"95% CI: [{ci_lower:.3f}, {ci_upper:.3f}]")
print(f"P-value: {did_model.pvalues['treatment']:.3f}")

import pandas as pd

import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create event time properly
df['event_time'] = df['week'] - 15  # Week 15 is event time 0

# Create categorical event time
df['event_cat'] = df['event_time'].astype(str)

# Manual event study - create dummies properly
event_dummies = pd.get_dummies(df['event_time'], prefix='e')
df = pd.concat([df, event_dummies], axis=1)

# Reference category: week -1 (just before treatment)
# Get all event time columns except the reference
event_cols = [c for c in df.columns if c.startswith('e_') and c != 'e_-1']

# Run regression manually
X = df[['switched_branch'] + event_cols]
X = sm.add_constant(X)
y = df['service_score']

model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results ===")
print("Coefficients for each week relative to treatment (reference: week -1)")
print()

# Extract and sort event time coefficients
results = []
for col in event_cols:
    week_rel = int(col.split('_')[1])
    results.append((week_rel, model.params[col], model.bse[col], model.pvalues[col]))

results.sort(key=lambda x: x[0])

for week_rel, coef, se, p in results:
    actual_week = week_rel + 15
    sig = "***" if p < 0.01 else "**" if p < 0.05 else "*" if p < 0.1 else ""
    ci_low = coef - 1.96 * se
    ci_high = coef + 1.96 * se
    print(f"Week {actual_week} (t={week_rel:+2d}): coef = {coef:6.2f}, SE = {se:.2f}, 95% CI = [{ci_low:.2f}, {ci_high:.2f}] {sig}")

post_results = [(r[0], r[1], r[2]) for r in results if r[0] >= 0]
