import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch fixed effects
# Model: service_score = beta0 + beta1*treatment + branch_FE + week_FE + error
# We'll cluster standard errors at branch level

# First, a simple DiD without fixed effects
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Simple DiD Regression (clustered SEs):")
print(model1.summary())
print()

# Now with branch fixed effects (using dummy variables)
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df, branch_dummies], axis=1)

# Build formula with branch FEs
branch_cols = [col for col in df_with_fe.columns if col.startswith('branch_')]
formula = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(branch_cols)
model2 = ols(formula, data=df_with_fe).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("DiD Regression with Branch Fixed Effects (clustered SEs):")
print(model2.summary().tables[1])
print()
print("Treatment coefficient:", round(model2.params['treatment'], 3))
print("Treatment p-value:", round(model2.pvalues['treatment'], 4))
print("Treatment 95% CI:", round(model2.conf_int().loc['treatment'][0], 3), "to", round(model2.conf_int().loc['treatment'][1], 3))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Simpler event study: pre vs post with interaction
# Create period indicators
df['pre_early'] = ((df['week'] >= 6) & (df['week'] <= 10)).astype(int)
df['pre_late'] = ((df['week'] >= 11) & (df['week'] <= 14)).astype(int)
df['post_early'] = ((df['week'] >= 15) & (df['week'] <= 18)).astype(int)
df['post_late'] = ((df['week'] >= 19) & (df['week'] <= 22)).astype(int)

# Interactions with treatment
df['pre_early_treat'] = df['pre_early'] * df['switched_branch']
df['pre_late_treat'] = df['pre_late'] * df['switched_branch']
df['post_early_treat'] = df['post_early'] * df['switched_branch']
df['post_late_treat'] = df['post_late'] * df['switched_branch']

# Create branch and week dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='br', drop_first=True)
week_dummies = pd.get_dummies(df['week'], prefix='wk', drop_first=True)
df_fe = pd.concat([df, branch_dummies, week_dummies], axis=1)

branch_cols = [col for col in df_fe.columns if col.startswith('br_')]
week_cols = [col for col in df_fe.columns if col.startswith('wk_')]

# Event study with period interactions (reference: pre_early for control group)
event_vars = ['pre_late_treat', 'post_early_treat', 'post_late_treat']
event_formula = 'service_score ~ ' + ' + '.join(event_vars) + ' + ' + ' + '.join(branch_cols + week_cols)
model_event = ols(event_formula, data=df_fe).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("="*80)
print("EVENT STUDY BY PERIOD")
print("="*80)
print("(Reference: Control group in weeks 6-10)")
print()

    ci_high = model_event.conf_int().loc[var][1]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.10 else ""
    print(f"  {var}: {coef:6.2f} (SE={se:.2f}, p={pval:.3f}) [{ci_low:.2f}, {ci_high:.2f}] {sig}")
