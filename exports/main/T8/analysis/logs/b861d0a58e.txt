import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch fixed effects
# Model: service_score = β0 + β1*after_switch + β2*switched_branch + β3*(after_switch*switched_branch) + ε
# β3 is the DiD estimate

# First, simple OLS without fixed effects
model_simple = ols('service_score ~ after_switch + switched_branch + treatment', data=df).fit()
print("Simple OLS (no fixed effects):")
print(model_simple.summary().tables[1])

# Now with branch fixed effects (using dummy variables for each branch)
# This controls for time-invariant branch characteristics
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df, branch_dummies], axis=1)

# Build formula with branch dummies
branch_cols = [col for col in df_with_dummies.columns if col.startswith('branch_')]
formula = 'service_score ~ after_switch + switched_branch + treatment + ' + ' + '.join(branch_cols)

model_fe = ols(formula, data=df_with_dummies).fit()
print("\n" + "="*60)
print("With Branch Fixed Effects:")
print("="*60)
print(f"Treatment effect (coefficient on interaction): {model_fe.params['treatment']:.4f}")
print(f"Standard error: {model_fe.bse['treatment']:.4f}")
print(f"t-statistic: {model_fe.tvalues['treatment']:.4f}")
print(f"p-value: {model_fe.pvalues['treatment']:.6f}")
print(f"95% CI: [{model_fe.conf_int().loc['treatment', 0]:.4f}, {model_fe.conf_int().loc['treatment', 1]:.4f}]")

# Also cluster standard errors by branch
from statsmodels.stats.api import sandwich_covariance
cluster_se = model_fe.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(f"\nClustered SE (by branch): {cluster_se.bse['treatment']:.4f}")
print(f"Clustered p-value: {cluster_se.pvalues['treatment']:.6f}")
print(f"Clustered 95% CI: [{cluster_se.conf_int().loc['treatment', 0]:.4f}, {cluster_se.conf_int().loc['treatment', 1]:.4f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create relative week variable
df['week_rel'] = df['week'] - 15  # Week 15 = 0 (first post-switch week)

# Create interaction dummies for event study with safe names
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    safe_name = f'd_w{abs(w)}_neg' if w < 0 else f'd_w{w}_pos'
    df[safe_name] = ((df['week_rel'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (omit week -1 as reference)
omit = -1
event_cols = []
col_map = {}  # Map safe names back to week numbers
for w in range(-9, 8):
    if w != omit:
        safe_name = f'd_w{abs(w)}_neg' if w < 0 else f'd_w{w}_pos'
        event_cols.append(safe_name)
        col_map[safe_name] = w

formula = 'service_score ~ ' + ' + '.join(event_cols) + ' + C(branch_id) + C(week)'
event_model = ols(formula, data=df).fit(cov_type='HC3')

# Extract coefficients and confidence intervals
results = []
for safe_name, w in col_map.items():
    coef = event_model.params[safe_name]
    se = event_model.bse[safe_name]
    pval = event_model.pvalues[safe_name]
    results.append({'week_rel': w, 'coef': coef, 'se': se, 'pval': pval})

# Add reference week
results.append({'week_rel': omit, 'coef': 0, 'se': 0, 'pval': 1, 'reference': True})

event_df = pd.DataFrame(results).sort_values('week_rel')
event_df['ci_low'] = event_df['coef'] - 1.96 * event_df['se']
event_df['ci_high'] = event_df['coef'] + 1.96 * event_df['se']

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))

ax.fill_between(event_df['week_rel'], event_df['ci_low'], event_df['ci_high'], alpha=0.3, color='blue')
ax.plot(event_df['week_rel'], event_df['coef'], 'o-', color='blue', linewidth=2, markersize=8)
ax.axhline(y=0, color='black', linestyle='--', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='-', linewidth=2, label='Switch point')

ax.set_xlabel('Weeks Relative to Switch (Week 15 = 0)', fontsize=12)
ax.set_ylabel('Treatment Effect (vs Control)', fontsize=12)
ax.set_title('Event Study: Weekly Treatment Effects on Service Scores', fontsize=14, fontweight='bold')
ax.set_xticks(range(-9, 8))
ax.set_xticklabels([f'W{w+15}' for w in range(-9, 8)])
ax.legend()
ax.grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("Event study chart saved as 'event_study.png'")

print("\n" + "="*70)
print("EVENT STUDY COEFFICIENTS")
print("="*70)
print("\nPre-switch (should be ~0 if parallel trends hold):")
pre_data = event_df[event_df['week_rel'] < 0]
for _, row in pre_data.iterrows():
    if row.get('reference', False):
        print(f"  Week {int(row['week_rel']+15)}: [REFERENCE]")
    else:
        sig = "*" if row['pval'] < 0.05 else ""
        print(f"  Week {int(row['week_rel']+15)}: {row['coef']:.2f} (95% CI: [{row['ci_low']:.2f}, {row['ci_high']:.2f}]){sig}")

print("\nPost-switch (treatment effects):")
post_data = event_df[event_df['week_rel'] >= 0]
for _, row in post_data.iterrows():
    sig = "*" if row['pval'] < 0.05 else ""
    print(f"  Week {int(row['week_rel']+15)}: {row['coef']:.2f} (95% CI: [{row['ci_low']:.2f}, {row['ci_high']:.2f}]){sig}")

# Summary assessment
print("\n" + "="*70)
print("FINAL ASSESSMENT")
print("="*70)

pre_significant = (pre_data['pval'] < 0.05).sum()
post_significant = (post_data['pval'] < 0.05).sum()

print(f"\nPre-switch coefficients significantly different from 0: {pre_significant}/{len(pre_data)}")
print(f"Post-switch coefficients significantly different from 0: {post_significant}/{len(post_data)}")

# Average post-switch effect
avg_post_effect = post_data['coef'].mean()
print(f"\nAverage post-switch effect: {avg_post_effect:.2f} points")

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Quick summary of key metrics
print("="*70)
print("KEY METRICS RECAP")
print("="*70)

# DiD estimate
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

did = (treat_after - treat_before) - (control_after - control_before)

print(f"\nControl group change: {control_after - control_before:.2f} points")
print(f"Treatment group change: {treat_after - treat_before:.2f} points")
print(f"DIFFERENCE-IN-DIFFERENCES: {did:.2f} points")

# Context
overall_mean = df['service_score'].mean()
overall_sd = df['service_score'].std()

print(f"\nOverall mean service score: {overall_mean:.1f}")
print(f"Overall SD: {overall_sd:.1f}")
print(f"Effect as % of mean: {did/overall_mean*100:.1f}%")
print(f"Effect in SD units: {did/overall_sd:.2f} SD")

# How many treatment branches actually improved?
print("\n" + "="*70)
print("BRANCH-LEVEL OUTCOMES")
print("="*70)

branch_summary = df.groupby(['branch_id', 'switched_branch']).apply(
    lambda x: pd.Series({
        'before': x[x['after_switch']==0]['service_score'].mean(),
        'after': x[x['after_switch']==1]['service_score'].mean()
    })
).reset_index()

branch_summary['change'] = branch_summary['after'] - branch_summary['before']
branch_summary['adjusted_change'] = branch_summary['change'] - (control_after - control_before)

treatment_branches = branch_summary[branch_summary['switched_branch']==1]

print(f"  % with negative effect: {(treatment_branches['adjusted_change'] < 0).mean()*100:.1f}%")

print("PRACTICAL SIGNIFICANCE CONSIDERATIONS")
