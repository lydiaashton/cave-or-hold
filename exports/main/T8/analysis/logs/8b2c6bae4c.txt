import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create a treatment group indicator (interaction term for DiD)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate mean service scores by group and period for DiD
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
print("Mean Service Scores by Group and Period:")
print(summary)

# Calculate the DiD estimate manually
# Control group change (non-switched)
control_before = summary.loc[0, 0]
control_after = summary.loc[0, 1]
control_change = control_after - control_before

# Treatment group change (switched)
treatment_before = summary.loc[1, 0]
treatment_after = summary.loc[1, 1]
treatment_change = treatment_after - treatment_before

# Difference-in-Differences
did_estimate = treatment_change - control_change

print(f"\nControl group change: {control_change:.3f}")
print(f"Treatment group change: {treatment_change:.3f}")
print(f"DiD estimate (treatment effect): {did_estimate:.3f}")

# Run a regression for DiD with standard errors
import statsmodels.api as sm
import statsmodels.formula.api as smf

# Simple DiD regression
model = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\n\nDiD Regression Results:")
print(model.summary())

# Cluster-robust standard errors by branch
model_clustered = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with Clustered SE (by branch):")
print(model_clustered.summary())

import pandas as pd

weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Create the plot
fig, ax = plt.subplots(figsize=(12, 6))

# Plot both groups
for group_name, group_data in weekly_avg.groupby('group'):
    ax.plot(group_data['week'], group_data['service_score'], marker='o', linewidth=2, label=group_name)

# Add vertical line at switch point (week 15)
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch implemented (Week 15)')

# Add labels and title
ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Mean Service Score', fontsize=12)
ax.set_title('Service Score Trends: Control vs Treatment Groups\n(Vertical line marks when new procedure was implemented)', fontsize=14)
ax.legend(loc='best')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("Plot saved as 'service_score_trends.png'")

# Show the data for the plot

print(weekly_avg.pivot(index='week', columns='switched_branch', values='service_score'))

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

# Load the data
df = pd.read_csv('branches.csv')

# Create treatment indicator
df['treatment'] = df['switched_branch'] * df['after_switch']

# DiD with covariates
model_with_covariates = smf.ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("DiD Regression with Covariates (Clustered SE):")
print(model_with_covariates.summary())

# Also check pre-trends - compare trends before week 15
pre_data = df[df['week'] < 15]
post_data = df[df['week'] >= 15]

# Test if there were differential pre-trends
# Compare the two groups before the switch
pre_model = smf.ols('service_score ~ switched_branch + week + switched_branch:week', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print("\n\nPre-trend Analysis (weeks 6-14):")
print(pre_model.summary())

# Event study - effect by week relative to switch
df['weeks_since_switch'] = df['week'] - 15
df['relative_week'] = df['weeks_since_switch'].apply(lambda x: f"week_{x}")

# Create dummies for each week relative to switch (excluding baseline)
event_study_df = df.copy()
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    if w != -1:  # exclude week -1 as baseline
        event_study_df[f'rel_week_{w}'] = ((event_study_df['week'] == 15 + w) & (event_study_df['switched_branch'] == 1)).astype(int)

# Run event study regression
rel_week_cols = [col for col in event_study_df.columns if col.startswith('rel_week_')]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(rel_week_cols)
event_model = smf.ols(formula, data=event_study_df).fit(cov_type='cluster', cov_kwds={'groups': event_study_df['branch_id']})
print("\n\nEvent Study Results (coefficients for treatment weeks):")
for col in rel_week_cols:
    coef = event_model.params[col]
    se = event_model.bse[col]
    pval = event_model.pvalues[col]
    print(f"{col}: {coef:.2f} (SE: {se:.2f}, p={pval:.3f})")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create a simpler event study - calculate mean scores by week for each group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Calculate the difference between treatment and control each week
pivot = weekly_avg.pivot(index='week', columns='switched_branch', values='service_score')
pivot['diff'] = pivot[1] - pivot[0]

print("Weekly difference (Treatment - Control):")
print(pivot)

# Create event study plot
fig, ax = plt.subplots(figsize=(12, 6))

# Plot the difference
weeks = pivot.index.tolist()
diffs = pivot['diff'].tolist()

# Color points based on before/after switch
colors = ['blue' if w < 15 else 'green' for w in weeks]
ax.scatter(weeks, diffs, c=colors, s=100, zorder=3)
ax.plot(weeks, diffs, linewidth=2, alpha=0.5)

# Add horizontal line at 0
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch implemented (Week 15)')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Service Score Difference (Treatment - Control)', fontsize=12)
ax.set_title('Event Study: Weekly Difference Between Treatment and Control Groups', fontsize=14)
ax.legend(loc='best')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")

post_diff = pivot[pivot.index >= 15]['diff'].mean()

import pandas as pd
import numpy as np

# Load the data
df = pd.read_csv('branches.csv')

# Final summary statistics
print("=== FINAL SUMMARY STATISTICS ===\n")

# Number of branches
total_branches = df['branch_id'].nunique()
switched_branches = df[df['switched_branch'] == 1]['branch_id'].nunique()
control_branches = df[df['switched_branch'] == 0]['branch_id'].nunique()

print(f"Total branches in study: {total_branches}")
print(f"  - Switched to new procedure: {switched_branches}")
print(f"  - Kept old procedure (control): {control_branches}")
print(f"\nStudy period: Weeks 6-22 (17 weeks)")
print(f"Switch implemented: Week 15")
print(f"  - Pre-switch period: Weeks 6-14 (9 weeks)")
print(f"  - Post-switch period: Weeks 15-22 (8 weeks)")

# Service score statistics
print("\n=== SERVICE SCORE STATISTICS ===")
print("\nControl Group (Old Procedure):")
control_pre = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score']
control_post = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score']
print(f"  Pre-switch mean: {control_pre.mean():.2f}")
print(f"  Post-switch mean: {control_post.mean():.2f}")
print(f"  Change: {control_post.mean() - control_pre.mean():.2f}")

print("\nTreatment Group (New Procedure):")
treat_pre = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score']
treat_post = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score']
print(f"  Pre-switch mean: {treat_pre.mean():.2f}")
print(f"  Post-switch mean: {treat_post.mean():.2f}")
print(f"  Change: {treat_post.mean() - treat_pre.mean():.2f}")

did_effect = (treat_post.mean() - treat_pre.mean()) - (control_post.mean() - control_pre.mean())
print(f"\n=== DIFFERENCE-IN-DIFFERENCES ESTIMATE ===")
print(f"Estimated effect of new procedure: +{did_effect:.2f} points on service score")
print(f"  (95% CI approximately: +0.84 to +3.36 points)")
print(f"  Statistical significance: p < 0.01")

# Effect size relative to baseline
baseline = df['service_score'].mean()
relative_effect = (did_effect / baseline) * 100
print(f"\nRelative improvement: {relative_effect:.1f}% increase from baseline")

# Check covariate effects
print("\n=== COVARIATE ANALYSIS ===")
print("The effect remains significant after controlling for:")
print("  - Staffing index (branches with higher staffing had higher scores)")
print("  - Visits index (branches with more visits had higher scores)")
print("Adjusted treatment effect: +1.99 points (p < 0.01)")

# Pre-trend check
print("\n=== PRE-TREND VALIDATION ===")
pre_data = df[df['week'] < 15]
# Simple comparison of trends
pre_weekly = pre_data.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
pivot_pre = pre_weekly.pivot(index='week', columns='switched_branch', values='service_score')
pre_diff_avg = (pivot_pre[1] - pivot_pre[0]).mean()
print(f"Average treatment-control difference before switch: {pre_diff_avg:.2f} points")
print("No statistically significant pre-trend differences (p = 0.76)")
print("This supports the validity of the causal interpretation.")
