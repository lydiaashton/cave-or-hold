import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Summary table: mean service scores by group and period
summary_table = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("Service Score by Group and Period:")
print(summary_table)
print()

# Calculate DiD manually
# Control group (switched_branch=0)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
# Treatment group (switched_branch=1)
treated_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treated_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print("Control group (non-switched):")
print(f"  Before: {control_before:.2f}")
print(f"  After: {control_after:.2f}")
print(f"  Change: {control_after - control_before:.2f}")
print()
print("Treatment group (switched):")
print(f"  Before: {treated_before:.2f}")
print(f"  After: {treated_after:.2f}")
print(f"  Change: {treated_after - treated_before:.2f}")
print()

did_estimate = (treated_after - treated_before) - (control_after - control_before)
print(f"Difference-in-Differences Estimate: {did_estimate:.2f}")
print()

# Run regression with standard errors
# DiD regression: service_score = β0 + β1*switched_branch + β2*after_switch + β3*(switched_branch*after_switch) + ε
model = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()
print("Regression Results (DiD):")
print(model.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Check parallel trends: compare pre-trends (weeks 6-14) between groups
pre_data = df[df['week'] < 15].copy()

# Test if there's a differential trend before the switch
# Create a continuous week variable centered at week 15
pre_data['week_centered'] = pre_data['week'] - 15
pre_data['switched_week_interaction'] = pre_data['switched_branch'] * pre_data['week_centered']

# Regress on pre-period only to check for differential pre-trends
pre_trend_model = ols('service_score ~ switched_branch + week_centered + switched_week_interaction', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print("Pre-trend test (weeks 6-14 only):")
print(pre_trend_model.summary())
print()

# Event study: create week dummies relative to switch
df['relative_week'] = df['week'] - 15  # week 15 is the switch week (0)

# Create dummies for each relative week, using week -1 as reference
# We'll estimate effects for each week relative to the switch
event_study_df = df.copy()

# Run event study regression
# Create dummy variables for each relative week (excluding reference week -1)
for rw in range(-9, 8):  # weeks -9 to +7 relative to switch
    if rw != -1:  # skip reference week
        event_study_df[f'rel_week_{rw}'] = ((event_study_df['relative_week'] == rw) & (event_study_df['switched_branch'] == 1)).astype(int)

# Build formula
event_vars = [f'rel_week_{rw}' for rw in range(-9, 8) if rw != -1]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)

event_model = ols(formula, data=event_study_df).fit(cov_type='cluster', cov_kwds={'groups': event_study_df['branch_id']})
print("Event Study Results:")
print("Coefficients for treated branches by relative week (reference: week -1):")
for rw in range(-9, 8):
    if rw != -1:
        coef = event_model.params[f'rel_week_{rw}']
        se = event_model.bse[f'rel_week_{rw}']
        pval = event_model.pvalues[f'rel_week_{rw}']
        print(f"  Week {rw}: coef={coef:.2f}, SE={se:.2f}, p={pval:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Re-examine the key statistics
print("=== CRITICAL RE-EXAMINATION ===\n")

# 1. Pre-trend concern from event study
print("1. PRE-TREND CONCERN:")
print("   Event study showed several pre-period weeks with significant differences:")
print("   Weeks -8, -6, -5, -4, -3 all had p < 0.01")
print("   This suggests treated branches were already on different trajectory")
print("   → DiD estimate may be OVERSTATED\n")

# 2. Main DiD with and without controls
model_basic = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
model_controls = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("2. MAIN EFFECT ESTIMATES:")
print(f"   Basic DiD:     coef={model_basic.params['treated_post']:.2f}, p={model_basic.pvalues['treated_post']:.3f}")
print(f"   With controls: coef={model_controls.params['treated_post']:.2f}, p={model_controls.pvalues['treated_post']:.3f}")
print()

# 3. Power analysis - what effect could we detect?
n_treated = 20
n_control = 14
weeks_pre = 9
weeks_post = 8
print("3. STUDY POWER:")
print(f"   Treated branches: {n_treated}")
print(f"   Control branches: {n_control}")
print(f"   Pre-period weeks: {weeks_pre}")
print(f"   Post-period weeks: {weeks_post}")
print(f"   → Limited power to detect small effects\n")

# 4. Practical significance
baseline = df[df['after_switch']==0]['service_score'].mean()
std_dev = df['service_score'].std()
effect = model_basic.params['treated_post']
print("4. PRACTICAL SIGNIFICANCE:")
print(f"   Baseline score: {baseline:.2f}")
print(f"   Std deviation: {std_dev:.2f}")
print(f"   Effect size (Cohen's d): {effect/std_dev:.2f}")
print(f"   → Small effect size\n")

# 5. Cost-benefit consideration
print("5. DECISION FRAMEWORK:")
print("   If procedure is LOW-COST/LOW-RISK:")
print("   → Positive signal (70% vs 50% improved) may justify rollout")
print()
print("   If procedure is HIGH-COST/HIGH-RISK:")
print("   → Insufficient evidence; need more data")
