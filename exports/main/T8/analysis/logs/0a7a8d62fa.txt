import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate

df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Difference-in-Differences Regression Results:")
print(model.summary())

# Also run with cluster-robust standard errors by branch
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

pre_model = ols('service_score ~ switched_branch * week', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})

df = pd.read_csv('branches.csv')

# Create a clearer visualization of the DiD design
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Panel A: Raw trends
ax1 = axes[0]
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
switched = weekly_avg[weekly_avg['switched_branch'] == 1]
control = weekly_avg[weekly_avg['switched_branch'] == 0]

ax1.plot(control['week'], control['service_score'], 'o-', label='Control (Old Procedure)', 
         color='blue', linewidth=2.5, markersize=8, markerfacecolor='white', markeredgewidth=2)
ax1.plot(switched['week'], switched['service_score'], 's-', label='Treatment (New Procedure)', 
         color='green', linewidth=2.5, markersize=8, markerfacecolor='white', markeredgewidth=2)
ax1.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')
ax1.axhline(y=49.41, color='blue', linestyle=':', alpha=0.5)
ax1.axhline(y=47.68, color='green', linestyle=':', alpha=0.5)
ax1.text(6, 49.8, 'Control baseline: 49.4', color='blue', fontsize=9)
ax1.text(6, 47.2, 'Treatment baseline: 47.7', color='green', fontsize=9)

ax1.set_xlabel('Week', fontsize=11)
ax1.set_ylabel('Average Service Score', fontsize=11)
ax1.set_title('Panel A: Raw Service Score Trends', fontsize=12, fontweight='bold')
ax1.legend(loc='lower right', fontsize=9)
ax1.grid(True, alpha=0.3)
ax1.set_xticks(range(6, 23))
ax1.set_ylim(44, 60)

# Panel B: DiD visualization (deviation from pre-period mean)
ax2 = axes[1]
control_pre_mean = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
treatment_pre_mean = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()

weekly_avg_dev = weekly_avg.copy()
weekly_avg_dev['service_score_dev'] = weekly_avg_dev.apply(
    lambda row: row['service_score'] - (control_pre_mean if row['switched_branch']==0 else treatment_pre_mean),
    axis=1
)

switched_dev = weekly_avg_dev[weekly_avg_dev['switched_branch'] == 1]
control_dev = weekly_avg_dev[weekly_avg_dev['switched_branch'] == 0]

ax2.plot(control_dev['week'], control_dev['service_score_dev'], 'o-', label='Control', 
         color='blue', linewidth=2.5, markersize=8, markerfacecolor='white', markeredgewidth=2)
ax2.plot(switched_dev['week'], switched_dev['service_score_dev'], 's-', label='Treatment', 
         color='green', linewidth=2.5, markersize=8, markerfacecolor='white', markeredgewidth=2)
ax2.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')
ax2.axhline(y=0, color='black', linestyle='-', linewidth=1)

ax2.set_xlabel('Week', fontsize=11)
ax2.set_ylabel('Deviation from Pre-Period Mean', fontsize=11)
ax2.set_title('Panel B: Trends Relative to Baseline (DiD View)', fontsize=12, fontweight='bold')
ax2.legend(loc='lower right', fontsize=9)
ax2.grid(True, alpha=0.3)
ax2.set_xticks(range(6, 23))
ax2.set_ylim(-4, 6)

# Add annotation for the DiD effect
ax2.annotate('DiD Effect: ~2.4 points', xy=(18, 3.5), fontsize=10, fontweight='bold',
             arrowprops=dict(arrowstyle='->', color='black'))

plt.tight_layout()
plt.savefig('did_visualization.png', dpi=150, bbox_inches='tight')
print("Visualization saved as did_visualization.png")

# Print key statistics for the summary
print("\n=== KEY STATISTICS FOR SUMMARY ===")
print(f"Pre-period mean (Control): {control_pre_mean:.2f}")
print(f"Pre-period mean (Treatment): {treatment_pre_mean:.2f}")
print(f"Pre-period difference: {treatment_pre_mean - control_pre_mean:.2f} points (treatment LOWER)")
print(f"\nPost-period mean (Control): {df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean():.2f}")
print(f"Post-period mean (Treatment): {df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean():.2f}")
print(f"\nChange for Control: {df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean() - control_pre_mean:.2f}")
print(f"Change for Treatment: {df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean() - treatment_pre_mean:.2f}")
print(f"\nDifference-in-Differences: 2.38 points")
print(f"95% CI: [1.13, 3.64]")
print(f"p-value: < 0.001")
