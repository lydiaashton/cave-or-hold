# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is our DiD estimate

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators
# Week 14 is the last pre-switch week, so we'll use that as reference
df['relative_week'] = df['week'] - 14  # week 14 becomes 0, week 15 becomes 1, etc.

# Create event study dummies (relative weeks -8 to -1 for pre, 1 to 8 for post)
# Reference category: relative_week = -1 (week 13)
df['rel_week_cat'] = df['relative_week'].astype('category')

# Run event study regression
# Omit week -1 as reference
event_study = ols('service_score ~ switched_branch * rel_week_cat', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for interaction terms
print("Event Study Coefficients (Treatment x Relative Week):")
coef_dict = {}
for param in event_study.params.index:
    if 'switched_branch:rel_week_cat' in param:
        coef = event_study.params[param]
        se = event_study.bse[param]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        # Extract the relative week number
        week_num = param.split('[T.')[-1].rstrip(']')
        coef_dict[float(week_num)] = (coef, se, ci_low, ci_high)
        print(f"  Week {week_num}: {coef:.3f} (SE: {se:.3f}), 95% CI: [{ci_low:.3f}, {ci_high:.3f}]")

# Plot event study
plt.figure(figsize=(12, 6))

weeks = sorted(coef_dict.keys())
coefs = [coef_dict[w][0] for w in weeks]
ci_lows = [coef_dict[w][2] for w in weeks]
ci_highs = [coef_dict[w][3] for w in weeks]

plt.errorbar(weeks, coefs, yerr=[ [coefs[i]-ci_lows[i] for i in range(len(weeks))], 
                                   [ci_highs[i]-coefs[i] for i in range(len(weeks))] ],
             fmt='o', capsize=5, label='Treatment Effect', linewidth=2)
plt.axhline(y=0, color='black', linestyle='-', linewidth=1)
plt.axvline(x=0.5, color='red', linestyle='--', linewidth=2, label='Switch implemented')
plt.xlabel('Relative Week (0 = Week 14, 1 = Week 15)', fontsize=12)
plt.ylabel('Service Score Difference (Treatment - Control)', fontsize=12)
plt.title('Event Study: Treatment Effect by Week', fontsize=14, fontweight='bold')
plt.legend(loc='best')
plt.grid(True, alpha=0.3)
plt.xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")
