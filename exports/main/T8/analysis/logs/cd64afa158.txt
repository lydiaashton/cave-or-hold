import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Calculate mean service scores by group and period
print("Mean service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(pivot)

# Calculate the differences
print("\n\nDifference-in-Differences Calculation:")

# Before period (after_switch = 0)
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()

# After period (after_switch = 1)
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"Control group (no switch): Before={before_control:.2f}, After={after_control:.2f}, Change={after_control - before_control:.2f}")
print(f"Treatment group (switched): Before={before_treatment:.2f}, After={after_treatment:.2f}, Change={after_treatment - before_treatment:.2f}")

# DiD estimate
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")

# Also check weekly trends to see if there are pre-trends
print("\n\nWeekly service scores by group:")
weekly = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
print(weekly)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = β0 + β1*switched_branch + β2*after_switch + β3*(switched_branch * after_switch) + ε
# β3 is the DiD estimate

model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("Basic DiD Regression:")
print(model.summary())

# Now add branch fixed effects to control for time-invariant branch characteristics
# This is a more robust specification
print("\n\nDiD with Branch Fixed Effects:")
model_fe = ols('service_score ~ switched_branch * after_switch + C(branch_id)', data=df).fit()
print(model_fe.summary())

# Also add week fixed effects to control for common time trends
print("\n\nDiD with Branch and Week Fixed Effects:")
model_fe_week = ols('service_score ~ switched_branch * after_switch + C(branch_id) + C(week)', data=df).fit()
print(model_fe.summary())
