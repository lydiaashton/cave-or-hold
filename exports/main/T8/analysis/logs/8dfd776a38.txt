import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the 4 groups for DiD analysis
df['group'] = df['switched_branch'].astype(str) + '_' + df['after_switch'].astype(str)

# Calculate means for each group
group_means = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("Service Score by Group:")
print(group_means)

# Calculate the DiD estimate
# Treatment group change: after - before for switched_branch=1
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

treatment_change = treatment_after - treatment_before
control_change = control_after - control_before
did_estimate = treatment_change - control_change

print("\n=== Difference-in-Differences Analysis ===")
print(f"Treatment group (switched): Before={treatment_before:.2f}, After={treatment_after:.2f}, Change={treatment_change:.2f}")
print(f"Control group (not switched): Before={control_before:.2f}, After={control_after:.2f}, Change={control_change:.2f}")
print(f"\nDiD Estimate (Treatment effect): {did_estimate:.2f}")

# Visualize trends by week
plt.figure(figsize=(12, 6))

# Calculate weekly averages for each group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

for switched in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == switched]
    label = 'Switched to New Procedure' if switched == 1 else 'Old Procedure (Control)'
    plt.plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2)

plt.axvline(x=14.5, color='red', linestyle='--', label='Switch Point (Week 15)')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Score Trends by Branch Type')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("\nPlot saved as service_score_trends.png")

import pandas as pd

import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a DiD regression with standard errors
# Basic DiD model
model = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Check parallel trends - compare trends before the switch
print("\n=== Parallel Trends Check (Pre-switch weeks 6-14) ===")
pre_switch = df[df['week'] < 15].copy()

# Calculate weekly means for each group
pre_weekly = pre_switch.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Check if there's a significant difference in trends before the switch

import pandas as pd
import numpy as np

from linearmodels.panel import PanelOLS

df = pd.read_csv('branches.csv')

# Set up panel data
df = df.set_index(['branch_id', 'week'])

# Run fixed effects model with cluster-robust standard errors
fe_model = PanelOLS(df['service_score'], 
                    df[['switched_branch', 'after_switch']], 
                    entity_effects=True,
                    time_effects=False)
fe_result = fe_model.fit(cov_type='clustered', cluster_entity=True)
print("=== Fixed Effects Model (Entity Effects) ===")
print(fe_result)

# Also check if there are heterogeneous effects
# Add staffing and visits as controls
df_reset = df.reset_index()
model_with_controls = sm.OLS.from_formula('service_score ~ switched_branch * after_switch + staffing_index + visits_index', 
                                           data=df_reset).fit(cov_type='cluster', cov_kwds={'groups': df_reset['branch_id']})
print("\n=== DiD with Controls (staffing and visits) ===")
print(model_with_controls.summary().tables[1])

# Check effect heterogeneity - does the effect vary by branch characteristics?
# Calculate branch-level average service score before the switch
branch_avg_before = df_reset[df_reset['after_switch']==0].groupby('branch_id')['service_score'].mean()
branch_avg_before.name = 'avg_score_before'
df_reset = df_reset.merge(branch_avg_before, on='branch_id')

# Split by high/low baseline performers
median_before = branch_avg_before.median()
df_reset['high_baseline'] = (df_reset['avg_score_before'] >= median_before).astype(int)

# Check if effect differs by baseline performance
model_hetero = sm.OLS.from_formula('service_score ~ switched_branch * after_switch * high_baseline', 
                                   data=df_reset).fit(cov_type='cluster', cov_kwds={'groups': df_reset['branch_id']})
print("\n=== Heterogeneity by Baseline Performance ===")
print(model_hetero.summary().tables[1])

import pandas as pd
import numpy as np
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# The key result from the basic DiD regression
print("=== Main DiD Results (from earlier) ===")
print("Treatment effect (switched_branch:after_switch): 1.97 points")
print("Standard error: 0.60")
print("p-value: 0.001")
print("95% CI: [0.80, 3.14]")

# Let's verify with a simpler approach - calculate branch-level changes
print("\n=== Branch-Level Analysis ===")

# Calculate before and after averages for each branch
before = df[df['after_switch']==0].groupby('branch_id')['service_score'].mean()
after = df[df['after_switch']==1].groupby('branch_id')['service_score'].mean()

branch_changes = after - before
branch_changes.name = 'score_change'

# Merge with switch status
switch_status = df.groupby('branch_id')['switched_branch'].first()
branch_data = pd.DataFrame({'score_change': branch_changes, 'switched': switch_status})

# Compare changes between switched and non-switched branches
switched_changes = branch_data[branch_data['switched']==1]['score_change']
control_changes = branch_data[branch_data['switched']==0]['score_change']

print(f"Switched branches (n={len(switched_changes)}): mean change = {switched_changes.mean():.2f}, std = {switched_changes.std():.2f}")
print(f"Control branches (n={len(control_changes)}): mean change = {control_changes.mean():.2f}, std = {control_changes.std():.2f}")

# Two-sample t-test on the changes
from scipy import stats
t_stat, p_val = stats.ttest_ind(switched_changes, control_changes, equal_var=False)
print(f"\nTwo-sample t-test on branch-level changes:")
print(f"t-statistic: {t_stat:.2f}, p-value: {p_val:.4f}")
print(f"Difference in changes: {switched_changes.mean() - control_changes.mean():.2f} points")

# Confidence interval for the difference
diff = switched_changes.mean() - control_changes.mean()
se_diff = np.sqrt(switched_changes.var()/len(switched_changes) + control_changes.var()/len(control_changes))
ci_low = diff - 1.96 * se_diff
ci_high = diff + 1.96 * se_diff
print(f"95% CI: [{ci_low:.2f}, {ci_high:.2f}]")

# Effect size (Cohen's d)
pooled_std = np.sqrt(((len(switched_changes)-1)*switched_changes.var() + (len(control_changes)-1)*control_changes.var()) / (len(switched_changes)+len(control_changes)-2))
cohens_d = diff / pooled_std
print(f"\nEffect size (Cohen's d): {cohens_d:.2f}")
