# Load the data
df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch'] * df['after_switch']

# Summary: average service scores by group and period
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("Service scores by group and period:")
print(summary)

# Calculate the DiD manually
# Treatment group: before and after
treat_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treat_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nTreatment group (switched): Before={treat_before:.2f}, After={treat_after:.2f}, Change={treat_after - treat_before:.2f}")
print(f"Control group (not switched): Before={control_before:.2f}, After={control_after:.2f}, Change={control_after - control_before:.2f}")
print(f"\nDifference-in-Differences estimate: {(treat_after - treat_before) - (control_after - control_before):.2f}")

# Run DiD regression
model = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
print("\n\nDiD Regression (no controls):")
print(model.summary().tables[1])

print("\n\nDiD Regression (with controls):")

# Cluster-robust standard errors by branch

pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})

print("PRE-TREND TEST (critical for DiD validity):")
print(pre_model.summary().tables[1])
print("\n")

# The key coefficient is the interaction term
interaction_coef = pre_model.params['switched_branch:week_centered']
interaction_pval = pre_model.pvalues['switched_branch:week_centered']

print(f"Pre-trend interaction coefficient: {interaction_coef:.3f}")
print(f"P-value: {interaction_pval:.4f}")
print(f"\nInterpretation: Treatment group was improving by {interaction_coef:.2f} points/week MORE than control BEFORE the switch")
print(f"This is {'statistically significant' if interaction_pval < 0.05 else 'NOT statistically significant'}")

# Calculate what portion of the post-switch effect might be due to pre-existing trends
pre_trend_effect = interaction_coef * 8  # 8 weeks post-switch
print(f"\nIf pre-trend continued, expected additional divergence over 8 weeks: {pre_trend_effect:.2f} points")
print(f"Observed DiD effect: 1.84 points")
print(f"Effect potentially attributable to pre-trend: {pre_trend_effect/1.84*100:.1f}%")

# Event study approach - week-by-week differences
weekly_diff = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_diff['diff'] = weekly_diff[1] - weekly_diff[0]
print("\n\nWeek-by-week treatment-control differences:")
print(weekly_diff['diff'].round(2))
