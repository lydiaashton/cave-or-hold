print("DiD with Branch FE, controlling for staffing and visits:")
model_full = ols('service_score ~ after_switch + treatment + staffing_index + visits_index + C(branch_id)', 
                 data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
# Extract just the treatment coefficient
print(f"Treatment coefficient: {model_full.params['treatment']:.3f}")
print(f"Standard error: {model_full.bse['treatment']:.3f}")

# Event study - check effects by week relative to switch
print("\n\nEvent Study Analysis:")

    df[f'week_{w}'] = ((df['weeks_relative'] == w) & (df['switched_branch'] == 1)).astype(int)

event_study_vars = [f'week_{w}' for w in range(-8, 9) if w != -1]
formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(branch_id) + C(week)'
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nEvent study coefficients (relative to week -1):")
for w in range(-8, 9):
    if w != -1:
        var = f'week_{w}'
        coef = model_event.params[var]
        se = model_event.bse[var]
        pval = model_event.pvalues[var]
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"Week {w}: coef={coef:.2f}, SE={se:.2f}, p={pval:.3f} {sig}")

# Calculate average post-treatment effect
post_weeks = [f'week_{w}' for w in range(0, 9)]
post_coefs = [model_event.params[w] for w in post_weeks]
print(f"\nAverage post-treatment effect: {np.mean(post_coefs):.2f}")
