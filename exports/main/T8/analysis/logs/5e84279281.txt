import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated_after'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with cluster-robust standard errors
# Model: service_score = beta0 + beta1*switched + beta2*after + beta3*(switched*after) + error
model = ols('service_score ~ switched_branch + after_switch + treated_after', data=df).fit()

# Get cluster-robust standard errors (clustered by branch)
cov_cluster = model.get_robustcov_results(cov_type='cluster', clusters=df['branch_id'])

print("=== Difference-in-Differences Regression ===")
print(cov_cluster.summary())

# Extract the key coefficient
did_coef = cov_cluster.params['treated_after']
did_se = cov_cluster.bse['treated_after']
did_pvalue = cov_cluster.pvalues['treated_after']
did_ci_low = did_coef - 1.96 * did_se
did_ci_high = did_coef + 1.96 * did_se

print(f"\n=== Key Result ===")
print(f"DiD Coefficient: {did_coef:.3f}")
print(f"Standard Error: {did_se:.3f}")
print(f"P-value: {did_pvalue:.4f}")
print(f"95% CI: [{did_ci_low:.3f}, {did_ci_high:.3f}]")
