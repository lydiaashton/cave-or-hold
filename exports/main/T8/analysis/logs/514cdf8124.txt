import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Run difference-in-differences regression
# Basic DiD model
model = ols('service_score ~ treatment + post + did', data=df).fit()
print("Basic DiD Regression:")
print(model.summary())

print("\n" + "="*60)
print("Key coefficient (did):", model.params['did'])
print("95% CI:", model.conf_int().loc['did'].values)
print("P-value:", model.pvalues['did'])

import pandas as pd

# Model with branch and week fixed effects

import pandas as pd

from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Test for parallel pre-trends: interact treatment with week dummies for pre-period only
df_pre = df[df['week'] < 15].copy()

# Create week dummies and interactions
for w in range(7, 15):  # weeks 7-14 (week 6 is baseline)
    df_pre[f'week_{w}'] = (df_pre['week'] == w).astype(int)
    df_pre[f'treat_week_{w}'] = df_pre['switched_branch'] * df_pre[f'week_{w}']

# Regression with treatment-week interactions for pre-period
pre_trend_vars = '+'.join([f'treat_week_{w}' for w in range(7, 15)])
formula = f'service_score ~ switched_branch + C(week) + {pre_trend_vars}'
model_pre = ols(formula, data=df_pre).fit()

print("Pre-trend test (treatment x week interactions for weeks 7-14):")
print("Testing if treatment branches had different trends before the switch")
print("\nInteraction coefficients (should be insignificant if parallel trends hold):")
for w in range(7, 15):
    coef = model_pre.params[f'treat_week_{w}']
    pval = model_pre.pvalues[f'treat_week_{w}']
    print(f"  Week {w}: coef={coef:.3f}, p={pval:.3f}")

# Joint test for all pre-trend interactions
from statsmodels.stats.anova import anova_lm
# Full model with interactions
formula_full = f'service_score ~ switched_branch + C(week) + {pre_trend_vars}'
# Restricted model without interactions
formula_restricted = 'service_score ~ switched_branch + C(week)'
model_full = ols(formula_full, data=df_pre).fit()
model_restricted = ols(formula_restricted, data=df_pre).fit()

anova_result = anova_lm(model_restricted, model_full)
print("\nJoint F-test for all pre-trend interactions:")
print(anova_result)

# Also check if staffing and visits changed differently for treatment vs control
print("\n\nSecondary outcomes - did staffing/visits change?")

print("\nStaffing Index DiD:")

model_visits = ols('visits_index ~ switched_branch + after_switch + did + C(branch_id) + C(week)', data=df).fit()
print("\nVisits Index DiD:")
