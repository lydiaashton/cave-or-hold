import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate

df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Basic DiD Regression:")
print(model.summary())

# Now add branch fixed effects and week fixed effects for a more robust estimate
# This controls for time-invariant branch characteristics and common time trends

df = df.copy()
df['branch_id'] = df['branch_id'].astype('category')
df['week'] = df['week'].astype('category')

# Regression with branch and week fixed effects
model_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id) + C(week)', data=df).fit()
print("\n\nDiD Regression with Branch and Week Fixed Effects:")
# Just show the key coefficients
print(model_fe.params[['switched_branch', 'after_switch', 'treatment']])
print("\nStandard errors:")
print(model_fe.bse[['switched_branch', 'after_switch', 'treatment']])
print("\nP-values:")
print(model_fe.pvalues[['switched_branch', 'after_switch', 'treatment']])

# Confidence interval for treatment effect
treatment_coef = model_fe.params['treatment']
treatment_se = model_fe.bse['treatment']
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se
print(f"\nTreatment effect: {treatment_coef:.3f}")
print(f"95% CI: [{ci_lower:.3f}, {ci_upper:.3f}]")
