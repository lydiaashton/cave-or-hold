slope, intercept, r_value, p_value, std_err = stats.linregress(pre_weeks, pre_diff)
print(f"Pre-trend slope: {slope:.3f} (p={p_value:.3f})")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators
# Week 14 is the last pre-period week, so we'll use it as reference
df['relative_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create event study dummies (interactions of switched_branch with each week relative to switch)
# We'll omit week 0 (the reference period)
event_study_vars = []
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != 0:  # omit reference
        df[f'event_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(f'event_{w}')

# Build formula
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'C(week)'] + event_study_vars)

# Actually, let's do a cleaner event study with week fixed effects
df['week_cat'] = df['week'].astype(str)

# Event study regression with week FE and branch FE (or at least week FE)
event_vars = [f'event_{w}' for w in range(-8, 9) if w != 0]
formula = f'service_score ~ switched_branch + C(week) + {" + ".join(event_vars)}'

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results ===")
# Extract just the event study coefficients
event_coefs = {}
for w in range(-8, 9):
    if w != 0:
        var_name = f'event_{w}'
        if var_name in model_event.params.index:
            event_coefs[w] = {
                'coef': model_event.params[var_name],
                'std_err': model_event.bse[var_name],
                'pvalue': model_event.pvalues[var_name]
            }

print("Week (relative) | Coefficient | Std Error | p-value")
print("-" * 50)
for w in sorted(event_coefs.keys()):
    c = event_coefs[w]
    print(f"{w:15d} | {c['coef']:11.3f} | {c['std_err']:9.3f} | {c['pvalue']:7.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators
df['relative_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create event study dummies with safe names
event_study_vars = []
for w in range(-8, 9):
    if w != 0:  # omit reference
        safe_name = f'event_w{w:+d}'  # e.g., event_w-1, event_w+1
        df[safe_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(safe_name)

# Build formula with week fixed effects
formula = f'service_score ~ switched_branch + C(week) + {" + ".join(event_study_vars)}'

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results ===")
# Extract just the event study coefficients
event_coefs = {}
for w in range(-8, 9):
    if w != 0:
        var_name = f'event_w{w:+d}'
        if var_name in model_event.params.index:
            event_coefs[w] = {
                'coef': model_event.params[var_name],
                'std_err': model_event.bse[var_name],
                'pvalue': model_event.pvalues[var_name]
            }

print("Week (relative) | Coefficient | Std Error | p-value")
print("-" * 50)
for w in sorted(event_coefs.keys()):
    c = event_coefs[w]
    print(f"{w:15d} | {c['coef']:11.3f} | {c['std_err']:9.3f} | {c['pvalue']:7.3f}")

# Plot event study
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

weeks = sorted(event_coefs.keys())
coefs = [event_coefs[w]['coef'] for w in weeks]
stderrs = [event_coefs[w]['std_err'] for w in weeks]

fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(weeks, coefs, yerr=1.96*np.array(stderrs), fmt='o', capsize=4)

ax.set_title('Event Study: Service Score Effect by Week')
ax.grid(True, alpha=0.3)
plt.savefig('event_study.png', dpi=150, bbox_inches='tight')
print("\nEvent study plot saved as event_study.png")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators
df['relative_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create event study dummies with safe names (no minus signs)
event_study_vars = []
for w in range(-8, 9):
    if w != 0:  # omit reference
        safe_name = f'ew{w+10}'  # shift by 10 to avoid negative: w=-8 -> ew2, w=0 omitted, w=+8 -> ew18
        df[safe_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(safe_name)

# Build formula with week fixed effects
formula = f'service_score ~ switched_branch + C(week) + {" + ".join(event_study_vars)}'

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results ===")
# Extract just the event study coefficients
event_coefs = {}
for w in range(-8, 9):
    if w != 0:
        var_name = f'ew{w+10}'
        if var_name in model_event.params.index:
            event_coefs[w] = {
                'coef': model_event.params[var_name],
                'std_err': model_event.bse[var_name],
                'pvalue': model_event.pvalues[var_name]
            }

print("Week (relative) | Coefficient | Std Error | p-value")
print("-" * 50)
for w in sorted(event_coefs.keys()):
    c = event_coefs[w]
    print(f"{w:15d} | {c['coef']:11.3f} | {c['std_err']:9.3f} | {c['pvalue']:7.3f}")

# Plot event study
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

weeks = sorted(event_coefs.keys())
coefs = [event_coefs[w]['coef'] for w in weeks]
stderrs = [event_coefs[w]['std_err'] for w in weeks]

fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(weeks, coefs, yerr=1.96*np.array(stderrs), fmt='o', capsize=4)
ax.axhline(y=0, color='red', linestyle='--')
ax.axvline(x=0.5, color='gray', linestyle=':', label='Switch point')
ax.set_xlabel('Relative Week (0 = Week 14, reference)')
ax.set_ylabel('Coefficient (vs. Control)')
ax.set_title('Event Study: Service Score Effect by Week')
ax.grid(True, alpha=0.3)
plt.savefig('event_study.png', dpi=150, bbox_inches='tight')
print("\nEvent study plot saved as event_study.png")
