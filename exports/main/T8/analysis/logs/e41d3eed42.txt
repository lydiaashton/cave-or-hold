import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch'] * df['after_switch']

# Summary table: average service scores by group and period
summary_table = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).unstack()
print("=== Service Score Summary by Group and Period ===")
print(summary_table)

# Calculate the DiD estimate manually
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
before_treated = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treated = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n=== Difference-in-Differences Calculation ===")
print(f"Control group (no switch): Before={before_control:.2f}, After={after_control:.2f}, Change={after_control - before_control:.2f}")
print(f"Treated group (switched): Before={before_treated:.2f}, After={after_treated:.2f}, Change={after_treated - before_treated:.2f}")
did_estimate = (after_treated - before_treated) - (after_control - before_control)
print(f"\nDiD Estimate (treatment effect): {did_estimate:.2f}")

# Run regression with branch fixed effects
print("\n=== Regression Analysis ===")

# Model 1: Simple DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
print("\nModel 1: Simple DiD (no fixed effects)")
print(model1.summary().tables[1])

# Model 2: With branch fixed effects
df_with_dummies = pd.get_dummies(df, columns=['branch_id'], drop_first=True)
model2 = ols('service_score ~ switched_branch + after_switch + treated', data=df_with_dummies).fit()
print("\nModel 2: DiD with Branch Fixed Effects")
print(model2.summary().tables[1])

# Model 3: With controls and branch fixed effects
model3 = ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', data=df_with_dummies).fit()
print("\nModel 3: DiD with Branch Fixed Effects and Controls")
print(model3.summary().tables[1])
