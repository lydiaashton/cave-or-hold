import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Calculate mean service scores by group and period
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("Mean Service Scores by Group and Period:")
print(summary)
print("\n")

# Calculate the DiD estimate manually
# Treatment group (switched=1): before and after
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print("Treatment (switched) branches:")
print(f"  Before switch: {treatment_before:.2f}")
print(f"  After switch: {treatment_after:.2f}")
print(f"  Change: {treatment_after - treatment_before:.2f}")
print("\nControl (non-switched) branches:")
print(f"  Before switch: {control_before:.2f}")
print(f"  After switch: {control_after:.2f}")
print(f"  Change: {control_after - control_before:.2f}")
print("\nDifference-in-Differences estimate:")
did_estimate = (treatment_after - treatment_before) - (control_after - control_before)
print(f"  DiD = {did_estimate:.2f}")

# Run regression with DiD
model = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
print("\n\nRegression Results (DiD Model):")
print(model.summary())

# Run regression with cluster-robust standard errors (clustered by branch)

print("\n\nRegression Results (Clustered by Branch):")
print(model_clustered.summary())

pre_weekly_reset = pre_weekly['diff'].reset_index()

trend_model = ols('diff ~ week', data=pre_weekly_reset).fit()
