import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Calculate average service scores by group and period
print("=== Average Service Scores ===")
print("\nBefore switch (weeks 6-14):")
before = df[df['after_switch'] == 0]
print("  Switched branches:", before[before['switched_branch'] == 1]['service_score'].mean())
print("  Control branches:", before[before['switched_branch'] == 0]['service_score'].mean())

print("\nAfter switch (weeks 15-22):")
after = df[df['after_switch'] == 1]
print("  Switched branches:", after[after['switched_branch'] == 1]['service_score'].mean())
print("  Control branches:", after[after['switched_branch'] == 0]['service_score'].mean())

# Calculate the differences
before_switched = before[before['switched_branch'] == 1]['service_score'].mean()
before_control = before[before['switched_branch'] == 0]['service_score'].mean()
after_switched = after[after['switched_branch'] == 1]['service_score'].mean()
after_control = after[after['switched_branch'] == 0]['service_score'].mean()

diff_switched = after_switched - before_switched
diff_control = after_control - before_control
did_effect = diff_switched - diff_control

print("\n=== Difference-in-Differences ===")
print(f"Change in switched branches: {diff_switched:.3f}")
print(f"Change in control branches: {diff_control:.3f}")
print(f"DiD estimate (effect of new procedure): {did_effect:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimator

model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("=== DiD Regression Results ===")
print(model.summary())

# Also run with branch fixed effects for more robustness
print("\n=== DiD with Branch Fixed Effects ===")
# Create branch dummies
df_with_fe = df.copy()
branch_dummies = pd.get_dummies(df_with_fe['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df_with_fe, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [col for col in df_with_fe.columns if col.startswith('branch_')]
formula = 'service_score ~ switched_branch * after_switch + ' + ' + '.join(branch_cols)
model_fe = ols(formula, data=df_with_fe).fit()
print(model_fe.summary())

# Calculate weekly gaps
weekly_gaps = []
for week in range(6, 15):
    week_data = df[df['week'] == week]
    switched_mean = week_data[week_data['switched_branch'] == 1]['service_score'].mean()
    control_mean = week_data[week_data['switched_branch'] == 0]['service_score'].mean()
    weekly_gaps.append({'week': week, 'gap': switched_mean - control_mean})

gaps_df = pd.DataFrame(weekly_gaps)
print(gaps_df)

# Fit a linear trend to pre-switch gaps
model_trend = ols('gap ~ week', data=gaps_df).fit()
print("\n=== Pre-Switch Gap Trend ===")
print(model_trend.summary())
print(f"\nPre-switch gap is increasing by {model_trend.params['week']:.2f} points per week")

# Predicted gap at week 15 if pre-trend continued
predicted_week15 = model_trend.params['Intercept'] + model_trend.params['week'] * 15
print(f"Predicted gap at week 15 (if trend continued): {predicted_week15:.2f}")

# Actual gap at week 15
week15_data = df[df['week'] == 15]

print(f"Actual gap at week 15: {actual_week15:.2f}")
print(f"Jump at switch point: {actual_week15 - predicted_week15:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create a more flexible model that accounts for pre-existing trends
# Include week as a continuous variable and its interaction with switched_branch

print("=== DiD with Linear Time Trend ===")
# This model allows for different trends in each group
model_trend = ols('service_score ~ switched_branch * after_switch + week + switched_branch * week', data=df).fit()
cov_trend = model_trend.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(cov_trend.summary())

print("\n=== Key Coefficients Interpretation ===")
print(f"switched_branch:after_switch = {cov_trend.params['switched_branch:after_switch']:.3f}")
print(f"  (This is the level shift at the switch point, controlling for trends)")
print(f"switched_branch:week = {cov_trend.params['switched_branch:week']:.3f}")
print(f"  (This is the difference in weekly trends between groups)")

# Alternative: Compare post-switch period only, using pre-switch as baseline
print("\n=== Post-Switch Only Analysis (controlling for pre-switch average) ===")
# Calculate each branch's pre-switch average
pre_avg = df[df['week'] < 15].groupby('branch_id')['service_score'].mean().reset_index()
pre_avg.columns = ['branch_id', 'pre_avg_score']

# Merge with post-switch data
post_data = df[df['week'] >= 15].merge(pre_avg, on='branch_id')
model_post = ols('service_score ~ switched_branch + pre_avg_score + week', data=post_data).fit()
cov_post = model_post.get_robustcov_results(cov_type='cluster', groups=post_data['branch_id'])
print(cov_post.summary())
print(f"\nSwitched branch effect (post-switch, controlling for pre-average): {cov_post.params['switched_branch']:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

print("=== DiD with Linear Time Trend ===")
model_trend = ols('service_score ~ switched_branch * after_switch + week + switched_branch * week', data=df).fit()
cov_trend = model_trend.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("\nKey coefficients:")
for name, val in cov_trend.params.items():
    if 'switch' in name.lower() or 'week' in name.lower() or 'Intercept' in name:
        se = cov_trend.bse[name]
        pval = cov_trend.pvalues[name]
        print(f"  {name}: {val:.4f} (SE={se:.4f}, p={pval:.4f})")

# The interaction switched_branch:after_switch tests for a level shift at the switch point
# It is NOT significant (p=0.879), meaning no immediate jump

# Let's also do a simpler comparison: average post-switch scores controlling for branch fixed effects
print("\n=== Branch Fixed Effects Model (within-branch comparison) ===")
# This compares each branch to itself before and after
df['treated'] = df['switched_branch'] * df['after_switch']
model_fe = ols('service_score ~ treated + week + C(branch_id)', data=df).fit()
print(f"Treatment effect (within-branch, controlling for time): {model_fe.params['treated']:.3f}")
print(f"Standard error: {model_fe.bse['treated']:.3f}")
print(f"t-statistic: {model_fe.tvalues['treated']:.2f}")
print(f"p-value: {model_fe.pvalues['treated']:.4f}")

# Alternative: Just compare the change in switched branches vs change in control
print("\n=== Simple Before-After by Group ===")
for group in [0, 1]:
    group_name = "Switched" if group == 1 else "Control"
    before = df[(df['week'] < 15) & (df['switched_branch'] == group)]['service_score'].mean()
    after = df[(df['week'] >= 15) & (df['switched_branch'] == group)]['service_score'].mean()
    print(f"{group_name}: Before={before:.2f}, After={after:.2f}, Change={after-before:.2f}")
