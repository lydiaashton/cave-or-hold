import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create period variable (before/after)
df['period'] = df['after_switch']

# Calculate average service scores by group and period
print("=== Average Service Scores by Group and Period ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(pivot)

# Calculate the DiD manually
print("\n=== Difference-in-Differences Calculation ===")
# Control group (switched_branch=0)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_diff = control_after - control_before

# Treatment group (switched_branch=1)
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treatment_diff = treatment_after - treatment_before

# DiD estimate
did_estimate = treatment_diff - control_diff

print(f"Control group - Before: {control_before:.2f}, After: {control_after:.2f}, Change: {control_diff:.2f}")
print(f"Treatment group - Before: {treatment_before:.2f}, After: {treatment_after:.2f}, Change: {treatment_diff:.2f}")
print(f"\nDiD Estimate (Treatment Effect): {did_estimate:.2f}")

# Now run a regression with standard errors
print("\n=== Regression Analysis ===")
# Basic DiD regression
model = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Full DiD with controls
print("=== Full DiD Model with Controls ===")
model_controls = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', 
                     data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_controls.summary())

# Event study - effect by week relative to switch
print("\n=== Event Study Analysis ===")
df['weeks_relative'] = df['week'] - 15  # Week 15 is the switch (0), before is negative, after is positive

# Create week dummies for event study
week_dummies = pd.get_dummies(df['weeks_relative'], prefix='week_rel')
df_event = pd.concat([df, week_dummies], axis=1)

# Use week -1 (week 14) as reference
reference_week = 'week_rel_-1'
event_weeks = [col for col in week_dummies.columns if col != reference_week]

# Build formula
formula = f"service_score ~ switched_branch + {' + '.join([f'switched_branch:{w}' for w in event_weeks])} + staffing_index + visits_index"
print(f"Formula: {formula}")

# Run event study
event_model = ols(formula, data=df_event).fit(cov_type='cluster', cov_kwds={'groups': df_event['branch_id']})
print("\nEvent Study Results (coefficients for switched_branch:week_rel_X):")
for coef_name in event_model.params.index:
    if 'switched_branch:week_rel' in coef_name:
        coef = event_model.params[coef_name]
        se = event_model.bse[coef_name]
        pval = event_model.pvalues[coef_name]
        print(f"  {coef_name}: {coef:.2f} (SE: {se:.2f}, p: {pval:.3f})")
