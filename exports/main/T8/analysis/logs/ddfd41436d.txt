df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary: Average service scores by group and time period
print("=" * 60)
print("AVERAGE SERVICE SCORES BY GROUP AND TIME PERIOD")
print("=" * 60)

# Before switch (after_switch = 0)
before_switched = df[(df['after_switch'] == 0) & (df['switched_branch'] == 1)]['service_score'].mean()
before_control = df[(df['after_switch'] == 0) & (df['switched_branch'] == 0)]['service_score'].mean()

# After switch (after_switch = 1)
after_switched = df[(df['after_switch'] == 1) & (df['switched_branch'] == 1)]['service_score'].mean()
after_control = df[(df['after_switch'] == 1) & (df['switched_branch'] == 0)]['service_score'].mean()

print(f"\nBEFORE switch (weeks 6-14):")
print(f"  Switched branches (treatment): {before_switched:.2f}")
print(f"  Non-switched branches (control): {before_control:.2f}")
print(f"  Difference: {before_switched - before_control:.2f}")

print(f"\nAFTER switch (weeks 15-22):")
print(f"  Switched branches (treatment): {after_switched:.2f}")
print(f"  Non-switched branches (control): {after_control:.2f}")
print(f"  Difference: {after_switched - after_control:.2f}")

# Difference-in-Differences estimate
did_estimate = (after_switched - before_switched) - (after_control - before_control)
print(f"\nDIFFERENCE-IN-DIFFERENCES ESTIMATE: {did_estimate:.2f}")
print(f"  (Change in treatment) - (Change in control)")
print(f"  = ({after_switched:.2f} - {before_switched:.2f}) - ({after_control:.2f} - {before_control:.2f})")
print(f"  = {after_switched - before_switched:.2f} - {after_control - before_control:.2f}")

# Run regression for DiD
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

print("REGRESSION WITH CONTROLS (staffing_index, visits_index)")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# More detailed analysis
print("=" * 60)
print("DETAILED ANALYSIS")
print("=" * 60)

# Average pre and post for each group
pre_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
post_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nTreatment group: {pre_treatment:.2f} (pre) -> {post_treatment:.2f} (post), change = {post_treatment - pre_treatment:.2f}")
print(f"Control group: {pre_control:.2f} (pre) -> {post_control:.2f} (post), change = {post_control - pre_control:.2f}")
print(f"DiD estimate: {post_treatment - pre_treatment - (post_control - pre_control):.2f}")

# Check staffing and visits patterns
print("\n" + "=" * 60)
print("STAFFING AND VISITS BY GROUP AND TIME")
print("=" * 60)

staffing_pre_treat = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['staffing_index'].mean()
staffing_post_treat = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['staffing_index'].mean()
staffing_pre_ctrl = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['staffing_index'].mean()
staffing_post_ctrl = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['staffing_index'].mean()

print(f"\nStaffing Index:")
print(f"  Treatment: {staffing_pre_treat:.2f} (pre) -> {staffing_post_treat:.2f} (post)")
print(f"  Control: {staffing_pre_ctrl:.2f} (pre) -> {staffing_post_ctrl:.2f} (post)")

visits_pre_treat = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['visits_index'].mean()
visits_post_treat = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['visits_index'].mean()
visits_pre_ctrl = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['visits_index'].mean()
visits_post_ctrl = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['visits_index'].mean()

print(f"\nVisits Index:")
print(f"  Treatment: {visits_pre_treat:.2f} (pre) -> {visits_post_treat:.2f} (post)")
print(f"  Control: {visits_pre_ctrl:.2f} (pre) -> {visits_post_ctrl:.2f} (post)")

# Branch-level analysis
print("\n" + "=" * 60)
print("BRANCH-LEVEL CHANGES")
print("=" * 60)

branch_avg = df.groupby(['branch_id', 'switched_branch', 'after_switch'])['service_score'].mean().unstack(level='after_switch')
branch_avg['change'] = branch_avg[1] - branch_avg[0]

print("\nSwitched branches (treatment):")
switched_branches = branch_avg[branch_avg.index.get_level_values('switched_branch') == 1]
print(f"  Average change: {switched_branches['change'].mean():.2f}")
print(f"  Std dev of change: {switched_branches['change'].std():.2f}")
print(f"  Min change: {switched_branches['change'].min():.2f}")
print(f"  Max change: {switched_branches['change'].max():.2f}")
print(f"  Number of branches with positive change: {(switched_branches['change'] > 0).sum()}")
print(f"  Number of branches with negative change: {(switched_branches['change'] < 0).sum()}")

print("\nNon-switched branches (control):")
control_branches = branch_avg[branch_avg.index.get_level_values('switched_branch') == 0]
print(f"  Average change: {control_branches['change'].mean():.2f}")
print(f"  Std dev of change: {control_branches['change'].std():.2f}")
print(f"  Min change: {control_branches['change'].min():.2f}")
print(f"  Max change: {control_branches['change'].max():.2f}")
print(f"  Number of branches with positive change: {(control_branches['change'] > 0).sum()}")
print(f"  Number of branches with negative change: {(control_branches['change'] < 0).sum()}")

# Power analysis / effect size
print("\n" + "=" * 60)
print("EFFECT SIZE AND PRACTICAL SIGNIFICANCE")
print("=" * 60)

# Overall standard deviation of service scores
overall_sd = df['service_score'].std()
did_estimate = 1.56
print(f"\nOverall SD of service scores: {overall_sd:.2f}")
print(f"DiD estimate: {did_estimate:.2f}")
print(f"Effect size (Cohen's d): {did_estimate / overall_sd:.2f} SD")
print(f"\nInterpretation: The estimated effect is about {did_estimate / overall_sd * 100:.1f}% of one standard deviation.")
print(f"This is typically considered a {'small' if did_estimate / overall_sd < 0.2 else 'medium' if did_estimate / overall_sd < 0.5 else 'large'} effect.")

# Confidence interval
print("\n" + "=" * 60)
print("CONFIDENCE INTERVAL FOR TREATMENT EFFECT")
print("=" * 60)

model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
treatment_coef = model_clustered.params['treatment']
treatment_se = model_clustered.bse['treatment']
ci_lower = treatment_coef - 1.96 * treatment_se
ci_upper = treatment_coef + 1.96 * treatment_se
print(f"\nPoint estimate: {treatment_coef:.2f}")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"\nNote: The CI includes 0, meaning we cannot rule out no effect.")
print(f"The CI also includes negative values, meaning the new procedure could potentially have decreased scores.")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# One-sided test since they had a directional hypothesis
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

treatment_coef = model_clustered.params['treatment']
treatment_se = model_clustered.bse['treatment']
t_value = treatment_coef / treatment_se

# Two-sided p-value from the model
p_two_sided = model_clustered.pvalues['treatment']

# One-sided p-value (since hypothesis was directional: improvement expected)
from scipy import stats
p_one_sided = p_two_sided / 2  # Since coefficient is in expected direction

print("=" * 60)
print("RECONSIDERING THE EVIDENCE")
print("=" * 60)

print(f"\nPoint estimate: +{treatment_coef:.2f} points")
print(f"Standard error: {treatment_se:.2f}")
print(f"t-statistic: {t_value:.2f}")
print(f"\nTwo-sided p-value: {p_two_sided:.3f}")
print(f"One-sided p-value: {p_one_sided:.3f}")
print(f"\n95% CI (two-sided): [{treatment_coef - 1.96*treatment_se:.2f}, {treatment_coef + 1.96*treatment_se:.2f}]")
print(f"90% CI (two-sided): [{treatment_coef - 1.645*treatment_se:.2f}, {treatment_coef + 1.645*treatment_se:.2f}]")

# Probability that effect is positive (assuming normal distribution)
prob_positive = 1 - stats.norm.cdf(0, loc=treatment_coef, scale=treatment_se)
print(f"\nProbability effect is positive: {prob_positive:.1%}")

# What would we need for significance?
print("\n" + "=" * 60)
print("WHAT WOULD CHANGE THE CONCLUSION?")
print("=" * 60)

# Power analysis - how many more weeks would we need?
current_n = 578
current_effect = treatment_coef
current_se = treatment_se

# To get p < 0.05 (two-sided), we need t > 1.96
# SE scales with 1/sqrt(n), so we can calculate needed sample size
needed_se = current_effect / 1.96
needed_n = current_n * (current_se / needed_se) ** 2

print(f"\nTo achieve p < 0.05 (two-sided) with same effect size:")
print(f"  Current SE: {current_se:.2f}")
print(f"  Needed SE: {needed_se:.2f}")
print(f"  Current observations: {current_n}")
print(f"  Needed observations: {needed_n:.0f}")
print(f"  Additional weeks needed: {(needed_n - current_n) / 34:.1f}")

# Also calculate for one-sided test
needed_se_one = current_effect / 1.645
needed_n_one = current_n * (current_se / needed_se_one) ** 2
print(f"\nTo achieve p < 0.05 (one-sided) with same effect size:")
print(f"  Needed observations: {needed_n_one:.0f}")
print(f"  Additional weeks needed: {(needed_n_one - current_n) / 34:.1f}")
