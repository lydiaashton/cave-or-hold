# service_score = β0 + β1*switched_branch + β2*after_switch + β3*(switched_branch * after_switch) + ε
# β3 is the DiD estimator - the effect of the treatment

model = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()

import pandas as pd
import numpy as np

import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Run DiD with cluster-robust standard errors (clustered by branch)
model = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()
cluster_robust_se = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("DiD Regression with Cluster-Robust Standard Errors:")
print("=" * 60)
print(cluster_robust_se.summary())

# Check parallel trends - plot average service scores by week for each group
print("\n\nParallel Trends Check - Mean Service Score by Week and Group:")
print("=" * 60)

weekly_means = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
print(weekly_means.pivot(index='week', columns='switched_branch', values='service_score').round(2))

# Let's also run a regression with week fixed effects to control for time trends
# and branch fixed effects to control for branch-specific characteristics
print("\n\nDiD with Branch and Week Fixed Effects:")
print("=" * 60)

# Create dummy variables for branches and weeks
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df_with_dummies['branch_id'], prefix='branch', drop_first=True)
week_dummies = pd.get_dummies(df_with_dummies['week'], prefix='week', drop_first=True)

# Combine
X = pd.concat([df_with_dummies[['switched_branch', 'after_switch', 'treated_post']], 
               branch_dummies, week_dummies], axis=1)
y = df_with_dummies['service_score']

# Add constant
X = sm.add_constant(X, has_constant='add')

# Fit model
fe_model = sm.OLS(y, X).fit()
# Get cluster-robust SE
fe_cluster = fe_model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Just show the key coefficients
print("Key coefficients (treatment effect):")
print(f"treated_post coefficient: {fe_cluster.params['treated_post']:.3f}")
print(f"treated_post std err: {fe_cluster.bse['treated_post']:.3f}")
print(f"treated_post t-stat: {fe_cluster.tvalues['treated_post']:.3f}")
print(f"treated_post p-value: {fe_cluster.pvalues['treated_post']:.3f}")

# Let's also check if there's a pre-trend difference
print("\n\nPre-trend check (weeks 6-14 only):")
pre_df = df[df['week'] < 15].copy()
pre_model = ols('service_score ~ switched_branch + week', data=pre_df).fit()
pre_cluster = pre_model.get_robustcov_results(cov_type='cluster', groups=pre_df['branch_id'])

    change = after - before
    switched = branch_data['switched_branch'].iloc[0]
    branch_changes.append({'branch_id': bid, 'switched': switched, 'change': change})
