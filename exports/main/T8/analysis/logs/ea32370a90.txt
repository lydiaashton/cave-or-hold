import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Model 1: Basic DiD
print("=== Model 1: Basic Difference-in-Differences ===")
model1 = ols('service_score ~ switched_branch + after_switch + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model1.summary().tables[1])

# Model 2: DiD with controls
print("\n=== Model 2: DiD with Controls (staffing_index, visits_index) ===")
model2 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model2.summary().tables[1])

# Model 3: With branch fixed effects
print("\n=== Model 3: DiD with Branch Fixed Effects ===")
df_with_dummies = pd.get_dummies(df, columns=['branch_id'], drop_first=True)
X = df_with_dummies[['switched_branch', 'after_switch', 'did', 'staffing_index', 'visits_index']]
X = sm.add_constant(X)
# Add branch dummies
for col in df_with_dummies.columns:
    if col.startswith('branch_id_'):
        X[col] = df_with_dummies[col]
y = df_with_dummies['service_score']
model3 = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("DID coefficient:", model3.params['did'], "Std Error:", model3.bse['did'], "p-value:", model3.pvalues['did'])

# Alternative: use formula with C() for fixed effects
print("\n=== Model 3b: Using Formula with Branch FE ===")
model3b = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model3b.summary().tables[1].loc[['did', 'Intercept']])

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the event study data
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].agg(['mean', 'std', 'count']).reset_index()
weekly_pivot = weekly_avg.pivot(index='week', columns='switched_branch', values=['mean', 'std', 'count'])

# Calculate the difference and confidence intervals
diff = weekly_pivot['mean'][1] - weekly_pivot['mean'][0]

# Standard error of difference
se_diff = np.sqrt(
    (weekly_pivot['std'][1]**2 / weekly_pivot['count'][1]) + 
    (weekly_pivot['std'][0]**2 / weekly_pivot['count'][0])
)

ci_lower = diff - 1.96 * se_diff
ci_upper = diff + 1.96 * se_diff

print("=== Weekly Differences with 95% CI ===")
event_study = pd.DataFrame({
    'week': diff.index,
    'diff': diff.values,
    'ci_lower': ci_lower.values,
    'ci_upper': ci_upper.values
})
event_study['post_switch'] = event_study['week'] >= 15
print(event_study.round(2))

# Create a plot
fig, ax = plt.subplots(figsize=(10, 6))
ax.plot(event_study['week'], event_study['diff'], 'o-', label='Difference (Switched - Control)', linewidth=2)
ax.fill_between(event_study['week'], event_study['ci_lower'], event_study['ci_upper'], alpha=0.3, label='95% CI')

weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
weekly_diff = weekly_avg[1] - weekly_avg[0]

for week, diff in weekly_diff.items():
    marker = " <-- SWITCH" if week >= 15 else ""
    print(f"Week {week}: {diff:+.2f}{marker}")

# Calculate pre-trend slope
print("\n=== PRE-TREND ANALYSIS ===\n")
pre_diffs = weekly_diff[weekly_diff.index < 15]
pre_weeks = pre_diffs.index.values
pre_values = pre_diffs.values

# Linear regression on pre-period
slope, intercept = np.polyfit(pre_weeks, pre_values, 1)
print(f"Pre-trend slope: {slope:.2f} points per week")
print(f"If trend continued, expected week 15 diff: {slope * 15 + intercept:.2f}")
print(f"Actual week 15 diff: {weekly_diff[15]:.2f}")

# Calculate what the effect would be if we account for pre-trend
predicted_post = [slope * w + intercept for w in range(15, 23)]
actual_post = weekly_diff[weekly_diff.index >= 15].values
residuals = actual_post - predicted_post
print(f"\nAverage residual (effect beyond pre-trend): {residuals.mean():.2f} points")

# Create visualization
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Left panel: Weekly differences with trend line
ax1 = axes[0]
ax1.plot(pre_weeks, pre_values, 'o-', label='Pre-switch difference', linewidth=2, color='blue')
ax1.plot(weekly_diff[weekly_diff.index >= 15].index, 
         weekly_diff[weekly_diff.index >= 15].values, 
         's-', label='Post-switch difference', linewidth=2, color='green')
ax1.plot(pre_weeks, slope * pre_weeks + intercept, '--', color='blue', alpha=0.5, label='Pre-trend projection')
ax1.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (week 15)')
ax1.axhline(y=0, color='gray', linestyle='-', linewidth=1)
ax1.set_xlabel('Week', fontsize=12)
ax1.set_ylabel('Service Score Difference (Switched - Control)', fontsize=12)
ax1.set_title('Weekly Differences with Pre-Trend', fontsize=14)
ax1.legend()
ax1.grid(True, alpha=0.3)

# Right panel: Residuals from pre-trend
ax2 = axes[1]
post_weeks = weekly_diff[weekly_diff.index >= 15].index
ax2.bar(post_weeks, residuals, color='green', alpha=0.7, label='Effect beyond pre-trend')
ax2.axhline(y=0, color='gray', linestyle='-', linewidth=1)
ax2.set_xlabel('Week (Post-Switch)', fontsize=12)
ax2.set_ylabel('Residual Difference', fontsize=12)

ax2.legend()
ax2.grid(True, alpha=0.3, axis='y')
