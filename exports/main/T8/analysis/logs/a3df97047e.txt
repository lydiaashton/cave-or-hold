import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Count branches by treatment status
print("Branches by treatment status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Calculate average service scores before and after by treatment group
print("\n\nAverage service scores by group and period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the difference-in-differences manually
before_treated = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_treated = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n\nDifference-in-Differences Calculation:")
print(f"Treated - Before: {before_treated:.2f}")
print(f"Treated - After: {after_treated:.2f}")
print(f"Treated Change: {after_treated - before_treated:.2f}")
print(f"\nControl - Before: {before_control:.2f}")
print(f"Control - After: {after_control:.2f}")
print(f"Control Change: {after_control - before_control:.2f}")
print(f"\nDiD Estimate (Treated Change - Control Change): {after_treated - before_treated - (after_control - before_control):.2f}")

# Look at weekly trends
print("\n\nWeekly average service scores by group:")
weekly = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0).round(2)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Run DiD regression
# Model: service_score ~ switched_branch + after_switch + treated_post
model = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()
print("Difference-in-Differences Regression Results:")
print("=" * 60)
print(model.summary())

# Also run with branch fixed effects to control for time-invariant branch characteristics
print("\n\n" + "=" * 60)
print("DiD Regression with Branch Fixed Effects:")
print("=" * 60)
model_fe = ols('service_score ~ switched_branch + after_switch + treated_post + C(branch_id)', data=df).fit()

# Calculate weekly averages and differences
weekly_pre = pre_df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
print("Pre-treatment weekly averages:")
print(weekly_pre)
print("\nDifference (Treated - Control) each week:")
print(weekly_pre[1] - weekly_pre[0])

# Let's also look at post-treatment
post_df = df[df['week'] >= 15].copy()
weekly_post = post_df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
print("\n\nPost-treatment weekly averages:")
print(weekly_post)
print("\nDifference (Treated - Control) each week:")
print(weekly_post[1] - weekly_post[0])

# Create an event study specification
# Create relative time indicators
df['relative_week'] = df['week'] - 15  # weeks relative to switch (week 15 = 0)

# Create event study dummies
for w in range(-9, 8):  # weeks -9 to +7 relative to treatment
    df[f'rel_week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (omitting week -1 as reference)
event_study_vars = [f'rel_week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(branch_id) + C(week)'
event_model = ols(formula, data=df).fit()

print("\n\n" + "=" * 60)
print("Event Study Results (relative to week 14 as baseline):")
print("=" * 60)
event_results = event_model.summary().tables[1]
print(event_results)

# Extract coefficients for plotting
coef_dict = {}
se_dict = {}
for w in range(-9, 8):
    if w == -1:
        coef_dict[w] = 0
        se_dict[w] = 0
    else:
        var_name = f'rel_week_{w}'
        if var_name in event_model.params:
            coef_dict[w] = event_model.params[var_name]
            se_dict[w] = event_model.bse[var_name]

print("\n\nEvent Study Coefficients:")
for w in sorted(coef_dict.keys()):
    print(f"Week {w:+d}: {coef_dict[w]:.2f} (SE: {se_dict[w]:.2f})")
