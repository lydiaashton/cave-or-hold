import pandas as pd
import numpy as np

# Calculate average service scores by group and period
print("=== Average Service Scores ===")
print("\nBefore switch (weeks 6-14):")
before = df[df['after_switch'] == 0]
print("  Switched branches:", before[before['switched_branch'] == 1]['service_score'].mean())
print("  Control branches:", before[before['switched_branch'] == 0]['service_score'].mean())

print("\nAfter switch (weeks 15-22):")
after = df[df['after_switch'] == 1]
print("  Switched branches:", after[after['switched_branch'] == 1]['service_score'].mean())
print("  Control branches:", after[after['switched_branch'] == 0]['service_score'].mean())

# Calculate the differences
before_switched = before[before['switched_branch'] == 1]['service_score'].mean()
before_control = before[before['switched_branch'] == 0]['service_score'].mean()
after_switched = after[after['switched_branch'] == 1]['service_score'].mean()
after_control = after[after['switched_branch'] == 0]['service_score'].mean()

diff_switched = after_switched - before_switched
diff_control = after_control - before_control
did_effect = diff_switched - diff_control

print("\n=== Difference-in-Differences ===")
print(f"Change in switched branches: {diff_switched:.3f}")
print(f"Change in control branches: {diff_control:.3f}")
print(f"DiD estimate (effect of new procedure): {did_effect:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimator

model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("=== DiD Regression Results ===")
print(model.summary())

# Also run with branch fixed effects for more robustness
print("\n=== DiD with Branch Fixed Effects ===")
# Create branch dummies
df_with_fe = df.copy()
branch_dummies = pd.get_dummies(df_with_fe['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df_with_fe, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [col for col in df_with_fe.columns if col.startswith('branch_')]
formula = 'service_score ~ switched_branch * after_switch + ' + ' + '.join(branch_cols)
model_fe = ols(formula, data=df_with_fe).fit()
print(model_fe.summary())

import pandas as pd
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check pre-trends more carefully
print("=== Pre-Switch Trend Analysis ===")
pre_data = df[df['week'] < 15]

# Calculate weekly gaps
pre_gaps = []
for week in range(6, 15):
    week_data = pre_data[pre_data['week'] == week]
    switched_mean = week_data[week_data['switched_branch'] == 1]['service_score'].mean()
    control_mean = week_data[week_data['switched_branch'] == 0]['service_score'].mean()
    pre_gaps.append(switched_mean - control_mean)

print("Pre-switch weekly gaps (Switched - Control):")
for i, week in enumerate(range(6, 15)):
    print(f"  Week {week}: {pre_gaps[i]:.2f}")

# Fit a trend line to pre-switch gaps
from scipy import stats
slope, intercept, r_value, p_value, std_err = stats.linregress(range(6, 15), pre_gaps)
print(f"\nPre-switch trend: slope = {slope:.3f} points/week (p={p_value:.3f})")

# Calculate what we'd expect if pre-trend continued
expected_week22_gap = intercept + slope * 22
actual_week22_gap = 5.98  # from earlier analysis
print(f"\nIf pre-trend continued, expected week 22 gap: {expected_week22_gap:.2f}")
print(f"Actual week 22 gap: {actual_week22_gap:.2f}")
print(f"Additional gap beyond trend: {actual_week22_gap - expected_week22_gap:.2f}")

# Post-switch gaps
print("\n=== Post-Switch Weekly Gaps ===")
post_gaps = []
for week in range(15, 23):
    week_data = df[df['week'] == week]
    switched_mean = week_data[week_data['switched_branch'] == 1]['service_score'].mean()
    control_mean = week_data[week_data['switched_branch'] == 0]['service_score'].mean()
    post_gaps.append(switched_mean - control_mean)
    print(f"  Week {week}: {post_gaps[-1]:.2f}")

print(f"\nAverage pre-switch gap: {np.mean(pre_gaps):.2f}")
print(f"Average post-switch gap: {np.mean(post_gaps):.2f}")
print(f"Simple DiD: {np.mean(post_gaps) - np.mean(pre_gaps):.2f}")

# Adjusted estimate accounting for pre-trend
# If gap was increasing by slope per week, over 8 post-weeks that's 8*slope
trend_adjusted_effect = np.mean(post_gaps) - np.mean(pre_gaps) - (slope * 8 / 2)  # rough adjustment
print(f"\nPre-trend slope suggests gap was already growing by {slope:.2f} points/week")
print(f"This could explain part of the observed effect")
