import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Calculate mean service scores by group and period
print("Mean service scores by group and period:")

print(pivot)

# Calculate the differences
print("\n--- Difference-in-Differences Calculation ---")
# Before period (after_switch = 0)
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()

# After period (after_switch = 1)
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

# DiD estimate

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = β0 + β1*treatment + β2*post + β3*(treatment*post) + ε
# where β3 is the DiD estimate

df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# Add branch fixed effects
print("\n=== DiD with Branch Fixed Effects ===")
# Using branch_id as categorical
model_fe = ols('service_score ~ switched_branch + after_switch + treatment_post + C(branch_id)', data=df).fit()
print(model_fe.summary())

# Cluster-robust standard errors at branch level
print("\n=== DiD with Branch Fixed Effects (Clustered SEs) ===")
model_fe_clustered = ols('service_score ~ switched_branch + after_switch + treatment_post + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_fe_clustered.summary())
