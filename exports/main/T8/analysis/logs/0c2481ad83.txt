# Calculate the difference-in-differences manually
print("\n=== DiD Calculation ===")
# Control group (switched_branch=0)
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with robust standard errors clustered by branch
# First, let's do a simple OLS with cluster-robust SEs
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Simple DiD Regression ===")
print(model.summary())

# Now with cluster-robust standard errors by branch
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Regression with Cluster-Robust SEs ===")
print(model_clustered.summary())

# Add control variables
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Regression with Controls (Cluster-Robust SEs) ===")
print(model_controls.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks since switch)
# Week 15 is when switch happened, so relative_week = week - 15
df['relative_week'] = df['week'] - 15

# Create event study dummies for each week relative to switch
# We'll use week -1 (week 14) as the reference period
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    if w != -1:  # skip reference week
        df[f'week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
event_study_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = f"service_score ~ switched_branch + after_switch + {' + '.join(event_study_vars)}"
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Regression (Dynamic DiD) ===")
print(f"Reference period: week -1 (week 14)")
print("\nCoefficients for treatment weeks:")
for w in range(-9, 8):
    if w != -1:
        coef = model_event.params[f'week_{w}']
        pval = model_event.pvalues[f'week_{w}']
        ci_low = model_event.conf_int().loc[f'week_{w}'][0]
        ci_high = model_event.conf_int().loc[f'week_{w}'][1]
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"Week {w:3d} (actual week {w+15:2d}): coef={coef:6.2f}, p={pval:.3f}, 95% CI=[{ci_low:6.2f}, {ci_high:6.2f}] {sig}")

# Also check if pre-trends are jointly significant
pre_trend_vars = [f'week_{w}' for w in range(-9, -1)]
print("\n=== Joint Test of Pre-Trends ===")
from statsmodels.stats.contrast import ContrastMatrix
# Simple approach: test if all pre-treatment coefficients are zero
hypotheses = [f"{v} = 0" for v in pre_trend_vars]
f_test = model_event.f_test(hypotheses)
print(f"F-statistic: {f_test.fvalue:.2f}, p-value: {f_test.pvalue:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks since switch)
df['relative_week'] = df['week'] - 15

# Create event study dummies with valid variable names
for w in range(-9, 8):
    if w != -1:  # skip reference week
        var_name = f'w{w:+d}'  # e.g., w-9, w+0, w+7
        df[var_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
event_study_vars = [f'w{w:+d}' for w in range(-9, 8) if w != -1]
formula = f"service_score ~ switched_branch + after_switch + {' + '.join(event_study_vars)}"
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Regression (Dynamic DiD) ===")
print(f"Reference period: week -1 (week 14)")
print("\nCoefficients for treatment weeks:")
results = []
for w in range(-9, 8):
    if w != -1:
        var_name = f'w{w:+d}'
        coef = model_event.params[var_name]
        pval = model_event.pvalues[var_name]
        ci_low = model_event.conf_int().loc[var_name][0]
        ci_high = model_event.conf_int().loc[var_name][1]
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        results.append((w, w+15, coef, pval, ci_low, ci_high, sig))
        print(f"Week {w:3d} (actual week {w+15:2d}): coef={coef:6.2f}, p={pval:.3f}, 95% CI=[{ci_low:6.2f}, {ci_high:6.2f}] {sig}")

# Test pre-trends jointly
print("\n=== Joint Test of Pre-Trends ===")
pre_trend_vars = [f'w{w:+d}' for w in range(-9, -1)]
hypotheses = [f"{v} = 0" for v in pre_trend_vars]
f_test = model_event.f_test(hypotheses)
print(f"F-statistic: {f_test.fvalue:.2f}, p-value: {f_test.pvalue:.4f}")

# Save results for plotting
import matplotlib.pyplot as plt

weeks = [r[0] for r in results]
coefs = [r[2] for r in results]
ci_lows = [r[4] for r in results]
ci_highs = [r[5] for r in results]
