import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch and week fixed effects
# Using OLS with categorical variables for fixed effects
model = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id) + C(week)', data=df).fit()

print("Difference-in-Differences Regression Results")
print("=" * 60)
print(model.summary().tables[1])

# Get the treatment effect estimate and confidence interval
treatment_effect = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pvalue = model.pvalues['treatment']
ci_lower = treatment_effect - 1.96 * treatment_se
ci_upper = treatment_effect + 1.96 * treatment_se

print(f"95% Confidence Interval: [{ci_lower:.3f}, {ci_upper:.3f}]")

# Event study design - create relative week indicators with safe names (prefix with 'w')
df['relative_week'] = df['week'] - 15  # week 15 is period 0 (first post period)

# Create indicators for each relative week with safe variable names (w_9, w_8, etc.)
for rw in range(-9, 8):  # weeks -9 to 7 (week 6 to 22)
    if rw < 0:
        safe_name = f'w_m{abs(rw)}'  # w_m9, w_m8, etc. for negative
    else:
        safe_name = f'w_p{rw}'  # w_p0, w_p1, etc. for positive
    df[safe_name] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Event study regression (omitting week -1 as reference, which is w_m1)
event_study_vars = [f'w_m{abs(rw)}' if rw < 0 else f'w_p{rw}' for rw in range(-9, 8) if rw != -1]
formula = 'service_score ~ C(week) + switched_branch + ' + ' + '.join(event_study_vars)

for rw in range(-9, 8):

        coef = model_event.params[var_name]
        se = model_event.bse[var_name]

        event_coefs.append(coef)
        event_se.append(se)
        event_ci_lower.append(coef - 1.96 * se)
        event_ci_upper.append(coef + 1.96 * se)
        event_weeks.append(rw)
