import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Create before/after periods
# Before: weeks 6-14, After: weeks 15-22
df['period'] = np.where(df['week'] < 15, 'before', 'after')

# Calculate average service scores by group and period
summary = df.groupby(['switched_branch', 'period'])['service_score'].agg(['mean', 'std', 'count']).reset_index()
print("Average service scores by group and period:")
print(summary.pivot(index='switched_branch', columns='period', values=['mean', 'std', 'count']))

# Calculate the differences
print("\n--- Difference-in-Differences Calculation ---")

# For switched branches (treatment)
switched_before = df[(df['switched_branch'] == 1) & (df['period'] == 'before')]['service_score'].mean()
switched_after = df[(df['switched_branch'] == 1) & (df['period'] == 'after')]['service_score'].mean()
switched_diff = switched_after - switched_before

# For non-switched branches (control)
control_before = df[(df['switched_branch'] == 0) & (df['period'] == 'before')]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['period'] == 'after')]['service_score'].mean()
control_diff = control_after - control_before

# Difference in differences
did_effect = switched_diff - control_diff

print(f"\nSwitched branches (treatment):")
print(f"  Before: {switched_before:.3f}")

print(f"  Change: {switched_diff:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimator

# Create interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run regression with cluster-robust standard errors
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Difference-in-Differences Regression Results")
print("=" * 60)
print(model.summary())

did_coef = model.params['treatment']
did_se = model.bse['treatment']
did_pvalue = model.pvalues['treatment']
did_ci_low = model.conf_int().loc['treatment', 0]
did_ci_high = model.conf_int().loc['treatment', 1]

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create week-specific treatment indicators (for weeks 15-22)
for week in range(15, 23):
    df[f'treat_week{week}'] = ((df['switched_branch'] == 1) & (df['week'] == week)).astype(int)

# Run regression with week-specific effects (excluding week 14 as reference)
# This is an event study design
week_vars = [f'treat_week{w}' for w in range(15, 23)]
formula = 'service_score ~ switched_branch + week + ' + ' + '.join(week_vars)

event_study = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study: Treatment Effect by Week")
print("=" * 60)
print(f"{'Week':<8} {'Coefficient':<12} {'Std Err':<10} {'p-value':<10} {'95% CI':<20}")
print("-" * 60)

results = []
for week in range(15, 23):
    var = f'treat_week{week}'
    coef = event_study.params[var]
    se = event_study.bse[var]
    pval = event_study.pvalues[var]
    ci_low = event_study.conf_int().loc[var, 0]
    ci_high = event_study.conf_int().loc[var, 1]
    results.append({'week': week, 'coef': coef, 'se': se, 'pval': pval, 'ci_low': ci_low, 'ci_high': ci_high})
    print(f"{week:<8} {coef:<12.3f} {se:<10.3f} {pval:<10.4f} [{ci_low:.2f}, {ci_high:.2f}]")

# Plot the event study
plt.figure(figsize=(10, 6))
weeks = [r['week'] for r in results]
coefs = [r['coef'] for r in results]
ci_lows = [r['ci_low'] for r in results]
ci_highs = [r['ci_high'] for r in results]

plt.errorbar(weeks, coefs, yerr=[[c - l for c, l in zip(coefs, ci_lows)], [h - c for c, h in zip(coefs, ci_highs)]], 
             fmt='o', capsize=5, linewidth=2, markersize=8, label='Treatment Effect')
plt.axhline(y=0, color='red', linestyle='--', linewidth=2, label='No effect')
plt.axvline(x=14.5, color='gray', linestyle=':', linewidth=2, label='Switch implemented')

plt.xlabel('Week', fontsize=12)
plt.ylabel('Treatment Effect on Service Score', fontsize=12)
plt.title('Event Study: Weekly Treatment Effects After Switch', fontsize=14)
plt.legend(fontsize=10)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")
