import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.api import get_cluster_cov
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with controls
print("=== DiD Regression with Controls ===")
model_controls = ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', data=df).fit()
print(model_controls.summary())

# Cluster-robust standard errors by branch
print("\n=== Cluster-Robust Standard Errors (by branch) ===")
cov_clusters = get_cluster_cov(model_controls, cluster=df['branch_id'])
robust_se = np.sqrt(np.diag(cov_clusters))
print(f"\nCoefficients with cluster-robust SE:")
print(f"Intercept: {model_controls.params['Intercept']:.3f} (SE: {robust_se[0]:.3f})")
print(f"switched_branch: {model_controls.params['switched_branch']:.3f} (SE: {robust_se[1]:.3f})")
print(f"after_switch: {model_controls.params['after_switch']:.3f} (SE: {robust_se[2]:.3f})")
print(f"treated (DiD effect): {model_controls.params['treated']:.3f} (SE: {robust_se[3]:.3f})")
print(f"staffing_index: {model_controls.params['staffing_index']:.3f} (SE: {robust_se[4]:.3f})")
print(f"visits_index: {model_controls.params['visits_index']:.3f} (SE: {robust_se[5]:.3f})")

# Calculate t-stat and p-value for treated effect
treated_coef = model_controls.params['treated']
treated_se = robust_se[3]
t_stat = treated_coef / treated_se
from scipy import stats
p_value = 2 * (1 - stats.t.cdf(abs(t_stat), df=len(df) - len(model_controls.params)))
print(f"\nTreated effect: t-stat = {t_stat:.2f}, p-value = {p_value:.3f}")

plt.savefig('trends.png', dpi=150)
print("\nTrend plot saved to trends.png")

print("\n=== Weekly Average Service Scores ===")
