import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Test for pre-trends: interact treatment with week (only pre-period)
# If parallel trends hold, the coefficient should not be significantly different from 0
pre_df = df[df['week'] < 15].copy()
pre_df['week_centered'] = pre_df['week'] - 10  # center around middle of pre-period

# Regress service score on treatment, week, and their interaction (pre-period only)
pre_model = ols('service_score ~ treatment + week_centered + treatment:week_centered + staffing_index + visits_index', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print("Pre-Trend Test (weeks 6-14 only):")
print(pre_model.summary())

# The interaction term tests whether treatment and control had different trends before the switch
print("\n\nInterpretation: The 'treatment:week_centered' coefficient tests parallel trends.")
print("If significantly different from 0, parallel trends assumption is violated.")

# Alternative: Event study approach - estimate effect for each week relative to week 14
df['week_relative'] = df['week'] - 14  # week 14 is the reference (last pre-switch week)

# Create dummies for each week relative to switch
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != -1:  # skip week -1 (week 13) as reference... actually let's use week 14 as reference
        df[f'week_{w}'] = (df['week_relative'] == w).astype(int)

# Actually, let's use week 14 (relative week 0) as reference and estimate effects for all other weeks
# But we need to interact with treatment to see differential effects

# Simpler approach: estimate DiD for each post-switch week separately
print("\n\n=== Week-by-week DiD estimates (post-switch) ===")
results = []
for week in range(15, 23):
    week_data = df[df['week'].isin([14, week])].copy()  # compare week 14 baseline to each post week
    week_data['post'] = (week_data['week'] == week).astype(int)
    week_data['did'] = week_data['treatment'] * week_data['post']
    
    model = ols('service_score ~ treatment + post + did', data=week_data).fit(cov_type='cluster', cov_kwds={'groups': week_data['branch_id']})
    did_coef = model.params['did']
    did_se = model.bse['did']
    did_pval = model.pvalues['did']
    results.append({'week': week, 'did_estimate': did_coef, 'std_err': did_se, 'p_value': did_pval})

results_df = pd.DataFrame(results)
