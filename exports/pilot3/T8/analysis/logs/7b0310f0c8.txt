import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Create a treatment indicator (interaction of switched_branch and after_switch)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate means for the 4 groups in a DiD design
print("\n=== Difference-in-Differences Setup ===")
print("\nMean service scores by group:")

# Before period (after_switch=0), Control (switched_branch=0)
before_control = df[(df['after_switch']==0) & (df['switched_branch']==0)]['service_score'].mean()
# Before period, Treatment (switched_branch=1)
before_treatment = df[(df['after_switch']==0) & (df['switched_branch']==1)]['service_score'].mean()
# After period (after_switch=1), Control (switched_branch=0)
after_control = df[(df['after_switch']==1) & (df['switched_branch']==0)]['service_score'].mean()
# After period, Treatment (switched_branch=1)
after_treatment = df[(df['after_switch']==1) & (df['switched_branch']==1)]['service_score'].mean()

print(f"Before - Control (didn't switch): {before_control:.2f}")
print(f"Before - Treatment (switched):   {before_treatment:.2f}")
print(f"After  - Control (didn't switch): {after_control:.2f}")
print(f"After  - Treatment (switched):    {after_treatment:.2f}")

# Calculate DiD estimate manually
change_control = after_control - before_control
change_treatment = after_treatment - before_treatment
did_estimate = change_treatment - change_control

print(f"\nChange in Control group: {change_control:.2f}")
print(f"Change in Treatment group: {change_treatment:.2f}")
print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")

# Now run a regression with the treatment indicator
print("\n=== Regression Analysis ===")

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nBasic DiD Model:")
print(model.summary())

# Add branch fixed effects and week fixed effects for more robust analysis
df['branch_id'] = df['branch_id'].astype(str)
df['week'] = df['week'].astype(str)

model_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id) + C(week)', data=df).fit()
print("\nDiD Model with Branch and Week Fixed Effects:")
print(model_fe.summary())
