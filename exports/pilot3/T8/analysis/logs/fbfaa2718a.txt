import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate the 2x2 table for DiD
print("=== Difference-in-Differences 2x2 Table ===\n")

# Before period (after_switch = 0)
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()

# After period (after_switch = 1)
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"Control Group (didn't switch):")
print(f"  Before: {before_control:.2f}")
print(f"  After:  {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")
print()
print(f"Treatment Group (switched):")
print(f"  Before: {before_treatment:.2f}")
print(f"  After:  {after_treatment:.2f}")
print(f"  Change: {after_treatment - before_treatment:.2f}")
print()

# DiD estimator
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"Difference-in-Differences Estimate: {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks relative to switch at week 15)
df['relative_week'] = df['week'] - 15

# Create event study dummies (omit week -1 as reference)
for w in range(-9, 8):  # weeks -9 to -1 and 0 to 7
    if w != -1:  # omit reference category
        df[f'week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Event study regression with controls
event_study_vars = ' + '.join([f'week_{w}' for w in range(-9, 8) if w != -1])
formula = f'service_score ~ switched_branch + after_switch + staffing_index + visits_index + C(branch_id) + {event_study_vars}'

model_es = smf.ols(formula, data=df).fit()

print("=== Event Study Coefficients (Treatment x Week relative to switch) ===")
print("(Reference: Week -1, i.e., week 14)")
print()

coeffs = []
for w in range(-9, 8):
    if w != -1:
        var_name = f'week_{w}'
        coef = model_es.params[var_name]
        se = model_es.bse[var_name]
        pval = model_es.pvalues[var_name]
        ci_lower = coef - 1.96 * se
        ci_upper = coef + 1.96 * se

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks relative to switch at week 15)
df['relative_week'] = df['week'] - 15

# Create event study dummies with safe names (omit week -1 as reference)
event_vars = []
for w in range(-9, 8):  # weeks -9 to -1 and 0 to 7
    if w != -1:  # omit reference category
        var_name = f'ew_{abs(w)}_{"neg" if w < 0 else "pos"}'
        df[var_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_vars.append(var_name)

# Event study regression with controls and branch FE
formula = f'service_score ~ switched_branch + after_switch + staffing_index + visits_index + C(branch_id) + {" + ".join(event_vars)}'

model_es = smf.ols(formula, data=df).fit()

print("=== Event Study Coefficients (Treatment x Week relative to switch) ===")
print("(Reference: Week -1, i.e., week 14)")
print()

# Map back to original week numbers
week_mapping = {}
for w in range(-9, 8):
    if w != -1:
        var_name = f'ew_{abs(w)}_{"neg" if w < 0 else "pos"}'
        week_mapping[var_name] = w

results = []
for var_name in event_vars:
    w = week_mapping[var_name]
    coef = model_es.params[var_name]
    se = model_es.bse[var_name]
    pval = model_es.pvalues[var_name]
    ci_lower = coef - 1.96 * se
    ci_upper = coef + 1.96 * se
    results.append((w, coef, se, ci_lower, ci_upper, pval))
    sig = '***' if pval < 0.01 else '**' if pval < 0.05 else '*' if pval < 0.1 else ''
    print(f"Week {w:+d} (actual week {w+15:2d}): {coef:6.2f} (SE: {se:.2f}) 95%CI: [{ci_lower:.2f}, {ci_upper:.2f}] {sig}")

# Check pre-trend significance
print("\n=== Pre-trend Test ===")
pre_pvals = [r[5] for r in results if r[0] < -1]  # weeks -9 to -2
print(f"Pre-trend coefficients p-values: {[f'{p:.3f}' for p in pre_pvals]}")
print(f"All pre-trends insignificant at 5%? {all(p > 0.05 for p in pre_pvals)}")

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

print("=== Key Concern: Parallel Trends Assumption ===\n")

# Pre-period trends by group
pre_period = df[df['week'] <= 14].copy()
pre_period['week_centered'] = pre_period['week'] - 10

control_pre = pre_period[pre_period['switched_branch'] == 0]
treatment_pre = pre_period[pre_period['switched_branch'] == 1]

control_trend = smf.ols('service_score ~ week_centered + staffing_index + visits_index', data=control_pre).fit()
treatment_trend = smf.ols('service_score ~ week_centered + staffing_index + visits_index', data=treatment_pre).fit()

print("PRE-PERIOD TRENDS (weeks 6-14):")
print(f"  Control group:  {control_trend.params['week_centered']:.3f} points/week (p={control_trend.pvalues['week_centered']:.4f})")
print(f"  Treatment group: {treatment_trend.params['week_centered']:.3f} points/week (p={treatment_trend.pvalues['week_centered']:.4f})")
print(f"  → Groups were on DIFFERENT trajectories before the switch")
print(f"  → Control was declining; Treatment was stable")
print()

# Simple DiD (no FE to avoid numerical issues)
did_simple = smf.ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("SIMPLE DiD (no branch fixed effects):")
print(f"  Treatment effect: {did_simple.params['treatment']:.3f} (95% CI: [{did_simple.params['treatment'] - 1.96*did_simple.bse['treatment']:.2f}, {did_simple.params['treatment'] + 1.96*did_simple.bse['treatment']:.2f}])")
print(f"  p-value: {did_simple.pvalues['treatment']:.4f}")
print()

# What if we just compare post-period averages?
post_period = df[df['week'] >= 15]
control_post = post_period[post_period['switched_branch'] == 0]['service_score'].mean()
treatment_post = post_period[post_period['switched_branch'] == 1]['service_score'].mean()

control_pre_avg = pre_period[pre_period['switched_branch'] == 0]['service_score'].mean()
treatment_pre_avg = pre_period[pre_period['switched_branch'] == 1]['service_score'].mean()

print("RAW COMPARISONS:")
print(f"  Control:  {control_pre_avg:.2f} (pre) → {control_post:.2f} (post) = +{control_post - control_pre_avg:.2f}")
print(f"  Treatment: {treatment_pre_avg:.2f} (pre) → {treatment_post:.2f} (post) = +{treatment_post - treatment_pre_avg:.2f}")
print(f"  DiD estimate: {treatment_post - treatment_pre_avg - (control_post - control_pre_avg):.2f}")
print()

# Check: how much of the effect could be due to differential pre-trends?
# If control declined 0.36/week for 8 weeks (weeks 6-14), that's ~2.9 points of decline
# Treatment was flat
# So some of the "gain" might just be control continuing to fall
print("INTERPRETATION CAUTION:")
print(f"  Control group declined ~{abs(control_trend.params['week_centered']):.2f} points/week pre-switch")
print(f"  Over 8 pre-period weeks, that's ~{abs(control_trend.params['week_centered']) * 8:.1f} points of underlying decline")
print(f"  If this trend continued post-switch, it would inflate the apparent treatment effect")
