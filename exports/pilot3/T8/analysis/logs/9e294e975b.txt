# Run DiD regression
# Model 1: Basic DiD

# Check pre-trends statistically (weeks 6-14)

pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()
pre_df['treatment_group'] = pre_df['switched_branch']
pre_df['interaction'] = pre_df['week_centered'] * pre_df['treatment_group']

# Test if trends are parallel in pre-period
pre_trend_model = ols('service_score ~ week_centered + treatment_group + interaction', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print("Pre-trend test (interaction should be non-significant for parallel trends):")
print(pre_trend_model.summary())

# Also run an event study / dynamic DiD to see effects by week
df['relative_week'] = df['week'] - 15  # Week 15 is period 0
df['relative_week'] = df['relative_week'].clip(-9, 7)  # Cap at -9 to 7

# Create dummies for each relative week (excluding week -1 as reference)
for w in range(-9, 8):
    if w != -1:  # Skip week -1 as reference
        df[f'rel_week_{w}'] = (df['relative_week'] == w).astype(int)
        df[f'rel_week_{w}_treat'] = df[f'rel_week_{w}'] * df['switched_branch']

# Run event study regression
event_study_vars = [f'rel_week_{w}_treat' for w in range(-9, 8) if w != -1]
event_study_formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(week) + C(branch_id)'

# Simpler approach - just treatment interactions
event_vars = ' + '.join(event_study_vars)
event_model = ols(f'service_score ~ {event_vars} + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n\nEvent Study Results (coefficients for treatment x relative week):")
for w in range(-9, 8):

        var_name = f'rel_week_{w}_treat'
        if var_name in event_model.params.index:
            coef = event_model.params[var_name]
            se = event_model.bse[var_name]
            pval = event_model.pvalues[var_name]
            sig = '***' if pval < 0.01 else '**' if pval < 0.05 else '*' if pval < 0.1 else ''
            print(f"Week {w:+d} (relative to switch): coef={coef:.2f}, SE={se:.2f}, p={pval:.3f} {sig}")
