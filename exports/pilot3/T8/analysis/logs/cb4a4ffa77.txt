# DiD estimate
did_estimate = treatment_change - control_change

print(f"Control group: {control_before:.2f} -> {control_after:.2f} (change: {control_change:.2f})")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study - create week dummies relative to switch
df['relative_week'] = df['week'] - 14  # week 14 is the last pre-switch week

# Create event study dummies (using week 14 as reference)
for w in range(-8, 9):  # weeks -8 to +8 relative to switch
    if w != 0:  # skip reference week
        df[f'week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Event study regression
event_cols = [f'week_{w}' for w in range(-8, 9) if w != 0]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_cols)
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the interaction terms
coeffs = []
cis_lower = []
cis_upper = []
weeks = []

for w in range(-8, 9):
    if w == 0:
        continue
    col = f'week_{w}'
    if col in event_model.params:
        coef = event_model.params[col]
        ci = event_model.conf_int().loc[col]
        coeffs.append(coef)
        cis_lower.append(ci[0])
        cis_upper.append(ci[1])
        weeks.append(w)

# Plot event study
fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(weeks, coeffs, yerr=[np.array(coeffs) - np.array(cis_lower), 
                                  np.array(cis_upper) - np.array(coeffs)], 
            fmt='o', capsize=4, markersize=6, linewidth=2, color='green')
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=0.5, color='red', linestyle='--', linewidth=2, label='Policy Change')
ax.set_xlabel('Weeks Relative to Switch (0 = Week 14, last pre-switch week)', fontsize=11)
ax.set_ylabel('Coefficient (vs Control Branches)', fontsize=11)
ax.set_title('Event Study: Effect of New Shelving Procedure on Service Scores', fontsize=13, fontweight='bold')
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150, bbox_inches='tight')
print("Event study plot saved as event_study.png")

# Print key coefficients
print("\nEvent Study Coefficients (switched branches vs control):")
print("-" * 50)
for i, w in enumerate(weeks):
    label = "PRE" if w < 0 else "POST"
    sig = "***" if cis_lower[i] > 0 or cis_upper[i] < 0 else ""
    print(f"Week {w:+3d} ({label}): {coeffs[i]:6.2f}  [{cis_lower[i]:6.2f}, {cis_upper[i]:6.2f}] {sig}")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study - create week dummies relative to switch
df['relative_week'] = df['week'] - 14  # week 14 is the last pre-switch week

# Create event study dummies with valid column names (using w prefix)
event_cols = []
for w in range(-8, 9):
    if w != 0:  # skip reference week
        col_name = f'w{abs(w)}_neg' if w < 0 else f'w{w}_pos'
        df[col_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_cols.append(col_name)

# Event study regression
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_cols)
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the interaction terms
coeffs = []
cis_lower = []
cis_upper = []
weeks = []

for w in range(-8, 9):
    if w == 0:
        continue
    col_name = f'w{abs(w)}_neg' if w < 0 else f'w{w}_pos'
    coef = event_model.params[col_name]
    ci = event_model.conf_int().loc[col_name]
    coeffs.append(coef)
    cis_lower.append(ci[0])
    cis_upper.append(ci[1])
    weeks.append(w)

# Plot event study
fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(weeks, coeffs, yerr=[np.array(coeffs) - np.array(cis_lower), 
                                  np.array(cis_upper) - np.array(coeffs)], 
            fmt='o', capsize=4, markersize=6, linewidth=2, color='green')
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=0.5, color='red', linestyle='--', linewidth=2, label='Policy Change')
ax.set_xlabel('Weeks Relative to Switch (0 = Week 14, last pre-switch week)', fontsize=11)
ax.set_ylabel('Coefficient (Switched vs Control)', fontsize=11)
ax.set_title('Event Study: Effect of New Shelving Procedure on Service Scores', fontsize=13, fontweight='bold')
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150, bbox_inches='tight')
print("Event study plot saved as event_study.png")

# Print key coefficients
print("\nEvent Study Coefficients (switched branches vs control, relative to week 14):")
print("-" * 60)
for i, w in enumerate(weeks):
    period = "PRE" if w < 0 else "POST"
    sig = "***" if (cis_lower[i] > 0 or cis_upper[i] < 0) and cis_lower[i] > -0.5 else ""
    print(f"Week {w:+3d} ({period:4s}): {coeffs[i]:6.2f}  [{cis_lower[i]:6.2f}, {cis_upper[i]:6.2f}] {sig}")

# Summary stats
print("\n" + "=" * 60)
print("SUMMARY STATISTICS")
print("=" * 60)
print(f"\nTotal observations: {len(df):,}")
print(f"Branches: {df['branch_id'].nunique()}")
print(f"  - Switched to new procedure: {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"  - Control (old procedure): {df[df['switched_branch']==0]['branch_id'].nunique()}")
print(f"\nStudy period: Weeks {df['week'].min()} to {df['week'].max()}")
print(f"Policy change: Week 15")
print(f"\nMain Result (DiD with cluster-robust SE):")
print(f"  Treatment effect: +{2.30:.2f} service score points")
print(f"  95% CI: [{1.71:.2f}, {2.90:.2f}]")
print(f"  p-value: <0.001")
print(f"\nInterpretation:")
print(f"  - Baseline average service score: ~49")
print(f"  - Treatment effect represents ~{2.30/49*100:.1f}% improvement")
print(f"  - Effect appears to grow over time post-switch")

import pandas as pd

pre_data = df[df['week'] < 15].copy()

# Test for differential pre-trends
pre_data['week_centered'] = pre_data['week'] - 10
pre_model = ols('service_score ~ switched_branch + week_centered + switched_branch:week_centered', 
                data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})

print("PRE-TREND TEST (Critical for DiD validity)")
print("=" * 60)
print(pre_model.summary())

# Extract the interaction term (differential pre-trend)
diff_pretrend = pre_model.params['switched_branch:week_centered']
diff_pretrend_pval = pre_model.pvalues['switched_branch:week_centered']
diff_pretrend_ci = pre_model.conf_int().loc['switched_branch:week_centered']

print(f"\nDifferential pre-trend coefficient: {diff_pretrend:.3f}")
print(f"95% CI: [{diff_pretrend_ci[0]:.3f}, {diff_pretrend_ci[1]:.3f}]")
print(f"p-value: {diff_pretrend_pval:.4f}")

if diff_pretrend_pval < 0.05:
    print("\n⚠️  WARNING: Significant differential pre-trends detected!")
    print("   This violates the parallel trends assumption required for valid DiD.")
else:
    print("\n✓ No significant differential pre-trends detected.")

for w in range(6, 15):
