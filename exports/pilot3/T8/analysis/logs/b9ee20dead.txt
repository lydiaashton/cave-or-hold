import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create treatment/control and before/after groups
df['group'] = df['switched_branch'].map({1: 'Treatment (Switched)', 0: 'Control (Not Switched)'})
df['period'] = df['after_switch'].map({1: 'After', 0: 'Before'})

# Summary statistics by group and period
print("Service Score Summary by Group and Period:")
summary = df.groupby(['group', 'period'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Calculate DiD manually
print("\n" + "="*60)
print("Difference-in-Differences Calculation:")
print("="*60)

# Treatment group
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_diff = treat_after - treat_before

# Control group
ctrl_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
ctrl_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
ctrl_diff = ctrl_after - ctrl_before

# DiD estimate
did = treat_diff - ctrl_diff

print(f"\nTreatment Group (Switched):")
print(f"  Before: {treat_before:.3f}")
print(f"  After:  {treat_after:.3f}")
print(f"  Change: {treat_diff:.3f}")

print(f"\nControl Group (Not Switched):")
print(f"  Before: {ctrl_before:.3f}")
print(f"  After:  {ctrl_after:.3f}")
print(f"  Change: {ctrl_diff:.3f}")

print(f"\nDifference-in-Differences Estimate: {did:.3f}")

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Run DiD regression with branch fixed effects and clustered standard errors
# Model: service_score = beta0 + beta1*treatment + beta2*post + beta3*did + branch_fe + error

# First, simple DiD without fixed effects
model1 = smf.ols('service_score ~ treatment + post + did', data=df).fit()
print("Simple DiD Regression (no fixed effects):")
print(model1.summary())

# Now with branch fixed effects
# We'll use dummy variables for each branch
model2 = smf.ols('service_score ~ treatment + post + did + C(branch_id)', data=df).fit()
print("\n" + "="*60)
print("DiD Regression with Branch Fixed Effects:")
print("="*60)
print(model2.summary())

# Clustered standard errors at branch level
model3 = smf.ols('service_score ~ treatment + post + did + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n" + "="*60)
print("DiD Regression with Branch Fixed Effects (Clustered SE):")
print("="*60)
print(model3.summary())

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# For DiD with branch fixed effects, we need to be careful about collinearity
# Since treatment is constant within branch, we should use:
# service_score = alpha_i + beta*post + gamma*(treatment*post) + error
# where alpha_i are branch fixed effects

# Create the interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Model with branch FE - treatment drops out due to collinearity
model_fe = smf.ols('service_score ~ after_switch + did + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("DiD Regression with Branch Fixed Effects (Clustered SE):")
print("="*60)
# Just show the key coefficients
coef_df = pd.DataFrame({
    'coef': model_fe.params,
    'std_err': model_fe.bse,
    't': model_fe.tvalues,
    'p_value': model_fe.pvalues
})
print(coef_df.loc[['after_switch', 'did']])

# Also run a simpler model to verify
print("\n" + "="*60)
print("Simpler DiD (no branch FE, clustered SE):")
model_simple = smf.ols('service_score ~ switched_branch + after_switch + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_simple.summary().tables[1])
