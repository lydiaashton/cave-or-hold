import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the 2x2 table for difference-in-differences
print("=== Service Score by Group and Period ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(pivot)

# Calculate DiD manually
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
before_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print("\n=== Difference-in-Differences Calculation ===")
print(f"Control (non-switched) before: {before_control:.3f}")
print(f"Control (non-switched) after: {after_control:.3f}")
print(f"Control change: {after_control - before_control:.3f}")
print()
print(f"Treatment (switched) before: {before_treatment:.3f}")
print(f"Treatment (switched) after: {after_treatment:.3f}")
print(f"Treatment change: {after_treatment - before_treatment:.3f}")
print()
did_effect = (after_treatment - before_treatment) - (after_control - before_control)
print(f"DiD Estimate (treatment effect): {did_effect:.3f}")

# Run regression with DiD
print("\n=== Regression Analysis ===")
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model1.summary().tables[1])

# Add branch fixed effects
print("\n=== With Branch Fixed Effects ===")
df['branch_id'] = df['branch_id'].astype('category')
model2 = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit()
print(model2.summary().tables[1])

# Add controls (staffing and visits)
print("\n=== With Branch FE and Controls ===")
model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index + C(branch_id)', data=df).fit()
print(model3.summary().tables[1])

# Cluster-robust standard errors by branch
print("\n=== Cluster-Robust SE (by branch) ===")
model3_cluster = model3.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(model3_cluster.summary().tables[1])
