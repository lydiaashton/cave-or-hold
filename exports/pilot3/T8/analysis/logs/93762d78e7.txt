import pandas as pd
import numpy as np
import statsmodels.api as sm
from linearmodels.panel import PanelOLS

df = pd.read_csv('branches.csv')

# Set up panel data
df = df.set_index(['branch_id', 'week'])

# Create interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model with branch fixed effects
mod = PanelOLS(df['service_score'], 
               df[['treatment', 'staffing_index', 'visits_index']], 
               entity_effects=True)
result = mod.fit(cov_type='clustered', cluster_entity=True)
print("=== PanelOLS with Branch Fixed Effects ===")
print(result)

# Also run a simpler model without controls for comparison
mod_simple = PanelOLS(df['service_score'], 
                      df[['treatment']], 
                      entity_effects=True)
result_simple = mod_simple.fit(cov_type='clustered', cluster_entity=True)
print("\n=== PanelOLS with Branch Fixed Effects (No Controls) ===")
print(result_simple)

# Check robustness - event study style (check each post-switch week)
df_reset = df.reset_index()
df_reset['post_week'] = df_reset['week'] - 14  # weeks since switch
df_reset['post_week'] = df_reset['post_week'].apply(lambda x: max(0, x))

# Average treatment effect by post-switch week
post_analysis = df_reset[df_reset['switched_branch']==1].groupby('post_week')['service_score'].mean()
post_control = df_reset[df_reset['switched_branch']==0].groupby('post_week')['service_score'].mean()
print("\n=== Post-Switch Weekly Comparison ===")
comparison = pd.DataFrame({'Treatment': post_analysis, 'Control': post_control})
