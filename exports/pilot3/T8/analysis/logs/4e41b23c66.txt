import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# === Descriptive Statistics ===
print("=== Mean Service Scores by Group and Period ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the DiD manually
print("\n=== Difference-in-Differences Calculation ===")
# Control group (switched_branch=0)
control_pre = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_post = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_diff = control_post - control_pre

# Treatment group (switched_branch=1)
treat_pre = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_post = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_diff = treat_post - treat_pre

did = treat_diff - control_diff

print(f"Control group (non-switched):")
print(f"  Pre-switch mean: {control_pre:.2f}")
print(f"  Post-switch mean: {control_post:.2f}")
print(f"  Change: {control_diff:.2f}")
print(f"\nTreatment group (switched):")
print(f"  Pre-switch mean: {treat_pre:.2f}")
print(f"  Post-switch mean: {treat_post:.2f}")
print(f"  Change: {treat_diff:.2f}")
print(f"\nDifference-in-Differences estimate: {did:.2f}")
