import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

# Calculate means by group and period
print("=== Mean Service Scores by Group and Period ===")
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='after_switch', 
                       aggfunc='mean')
print(pivot)
print()

# Calculate DiD manually
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
before_treated = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_treated = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print("Control group (no switch):")
print(f"  Before: {before_control:.2f}")
print(f"  After: {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")
print()
print("Treated group (switched):")
print(f"  Before: {before_treated:.2f}")
print(f"  After: {after_treated:.2f}")
print(f"  Change: {after_treated - before_treated:.2f}")
print()

did_estimate = (after_treated - before_treated) - (after_control - before_control)
print(f"Difference-in-Differences Estimate: {did_estimate:.2f}")
print()

# Run regression with standard errors clustered by branch
print("=== Regression Analysis ===")
model = ols('service_score ~ treated + post + treated_post', data=df).fit()

print()

print(model_clustered.summary())

# More thorough diagnostics
print("=== Checking for potential issues ===")
print()

# 1. Check balance in group sizes over time
print("1. Observations per group per period:")
balance = df.groupby(['switched_branch', 'after_switch']).size().unstack()

print()

# 2. Check if there are any branches that switched mid-study (should be none based on design)
print("2. Checking switched_branch consistency within branch:")
branch_switch_check = df.groupby('branch_id')['switched_branch'].nunique()
print(f"Branches with inconsistent switch status: {(branch_switch_check > 1).sum()}")
print()

# 3. Check for any time-varying confounders
print("3. Staffing and visits trends by group and period:")
for var in ['staffing_index', 'visits_index']:
    pivot = df.pivot_table(values=var, index='switched_branch', columns='after_switch', aggfunc='mean')
    print(f"\n{var}:")
    print(pivot)
    # DiD for this variable
    before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)][var].mean()
    after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)][var].mean()
    before_treated = df[(df['switched_branch']==1) & (df['after_switch']==0)][var].mean()
    after_treated = df[(df['switched_branch']==1) & (df['after_switch']==1)][var].mean()
    did = (after_treated - before_treated) - (after_control - before_control)
    print(f"DiD estimate for {var}: {did:.3f}")
