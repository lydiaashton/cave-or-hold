# Load the data
df = pd.read_csv('branches.csv')
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Robustness check: Event study (week-by-week effects)
print("=== Event Study Analysis ===")
# Create week dummies interacted with treatment
df['week_cat'] = df['week'].astype(str)

# Use week 14 as reference (last pre-treatment week)
event_study_formula = 'service_score ~ C(week_cat) * switched_branch + staffing_index + visits_index'
# This is getting complex, let's do a simpler approach

# Alternative: estimate effect for each post-treatment week separately
print("\nPost-treatment week-by-week effects:")
for week in range(15, 23):
    week_df = df[df['week'].isin([14, week])].copy()
    week_df['post'] = (week_df['week'] == week).astype(int)
    week_df['treatment_post'] = week_df['switched_branch'] * week_df['post']
    model = ols('service_score ~ switched_branch + post + treatment_post', data=week_df).fit(cov_type='cluster', cov_kwds={'groups': week_df['branch_id']})
    coef = model.params['treatment_post']
    se = model.bse['treatment_post']
    pval = model.pvalues['treatment_post']
    print(f"Week {week}: coef = {coef:.2f}, SE = {se:.2f}, p = {pval:.3f}")

# Check effect heterogeneity by branch characteristics (if any)
print("\n=== Summary of Main Findings ===")

print(f"Treatment branches: {df[df['switched_branch']==1]['branch_id'].nunique()}")

print(f"Pre-treatment weeks: 6-14 (9 weeks)")
print(f"Post-treatment weeks: 15-22 (8 weeks)")

# Final DiD estimate with confidence interval
final_model = ols('service_score ~ switched_branch + after_switch + treatment_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
did_coef = final_model.params['treatment_post']
did_se = final_model.bse['treatment_post']

did_ci_upper = did_coef + 1.96 * did_se
