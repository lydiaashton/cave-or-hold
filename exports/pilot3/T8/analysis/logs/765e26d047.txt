# Create treatment indicator
df['treated'] = (df['switched_branch'] == 1).astype(int)

# Create week dummies and interactions
# Use week 14 as reference (last pre-treatment week)
for w in range(6, 23):
    df[f'week_d_{w}'] = (df['week'] == w).astype(int)
    if w != 14:  # Reference week
        df[f'treated_week_{w}'] = (df['switched_branch'] == 1) & (df['week'] == w)
        df[f'treated_week_{w}'] = df['treated_week_{w}'].astype(int)

interaction_terms = ' + '.join([f'treated_week_{w}' for w in range(6, 23) if w != 14])

# Don't include treated main effect since it's collinear with week FE
formula = f'service_score ~ {week_terms} + {interaction_terms}'

model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study with Week Fixed Effects (Week 14 as reference) ===")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create treatment indicator
df['treated'] = (df['switched_branch'] == 1).astype(int)

# Create week dummies and interactions
# Use week 14 as reference (last pre-treatment week)
for w in range(6, 23):
    df[f'week_d_{w}'] = (df['week'] == w).astype(int)
    if w != 14:  # Reference week
        df[f'treated_week_{w}'] = ((df['switched_branch'] == 1) & (df['week'] == w)).astype(int)

# Build formula with week fixed effects and treatment-week interactions
# Exclude week 14 as reference
week_terms = ' + '.join([f'week_d_{w}' for w in range(6, 23) if w != 14])
interaction_terms = ' + '.join([f'treated_week_{w}' for w in range(6, 23) if w != 14])

formula = f'service_score ~ {week_terms} + {interaction_terms}'

model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study with Week Fixed Effects (Week 14 as reference) ===")
print("\nPre-treatment coefficients (weeks 6-13, relative to week 14):")
pre_weeks = [6, 7, 8, 9, 10, 11, 12, 13]
for w in pre_weeks:
    col = f'treated_week_{w}'
    coef = model.params[col]
    se = model.bse[col]
    pval = model.pvalues[col]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Week {w}: coef={coef:6.3f}, SE={se:.3f}, p={pval:.3f} {sig}")

print("\nPost-treatment coefficients (weeks 15-22, relative to week 14):")
post_weeks = [15, 16, 17, 18, 19, 20, 21, 22]
for w in post_weeks:
    col = f'treated_week_{w}'
    coef = model.params[col]
    se = model.bse[col]
    pval = model.pvalues[col]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Week {w}: coef={coef:6.3f}, SE={se:.3f}, p={pval:.3f} {sig}")

# Average post-treatment effect
post_coefs = [model.params[f'treated_week_{w}'] for w in post_weeks]
print(f"\nAverage post-treatment effect: {np.mean(post_coefs):.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treated'] = (df['switched_branch'] == 1).astype(int)

# Re-run simple DiD for comparison
df['treatment_interaction'] = df['switched_branch'] * df['after_switch']
model_simple = ols('service_score ~ switched_branch + after_switch + treatment_interaction', 
                   data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Simple DiD Results ===")
print(f"Treatment effect: {model_simple.params['treatment_interaction']:.3f}")
print(f"Standard error: {model_simple.bse['treatment_interaction']:.3f}")
print(f"p-value: {model_simple.pvalues['treatment_interaction']:.3f}")
print(f"95% CI: [{model_simple.conf_int().loc['treatment_interaction'][0]:.3f}, {model_simple.conf_int().loc['treatment_interaction'][1]:.3f}]")

# With covariates
model_cov = ols('service_score ~ switched_branch + after_switch + treatment_interaction + staffing_index + visits_index', 
                data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD with Covariates ===")
print(f"Treatment effect: {model_cov.params['treatment_interaction']:.3f}")
print(f"Standard error: {model_cov.bse['treatment_interaction']:.3f}")
print(f"p-value: {model_cov.pvalues['treatment_interaction']:.3f}")

# Event study plot
# Re-create the event study model
for w in range(6, 23):
    df[f'week_d_{w}'] = (df['week'] == w).astype(int)
    if w != 14:
        df[f'treated_week_{w}'] = ((df['switched_branch'] == 1) & (df['week'] == w)).astype(int)

week_terms = ' + '.join([f'week_d_{w}' for w in range(6, 23) if w != 14])
interaction_terms = ' + '.join([f'treated_week_{w}' for w in range(6, 23) if w != 14])
formula = f'service_score ~ {week_terms} + {interaction_terms}'

model_es = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients and confidence intervals
weeks = list(range(6, 23))
coefs = []
cis_low = []
cis_high = []
for w in weeks:
    if w == 14:
        coefs.append(0)
        cis_low.append(0)
        cis_high.append(0)
    else:
        col = f'treated_week_{w}'
        coefs.append(model_es.params[col])
        ci = model_es.conf_int().loc[col]
        cis_low.append(ci[0])
        cis_high.append(ci[1])

# Plot
fig, ax = plt.subplots(figsize=(10, 6))
ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point')

ax.errorbar(weeks, coefs, yerr=[np.array(coefs) - np.array(cis_low), np.array(cis_high) - np.array(coefs)],
            fmt='o', capsize=5, label='Treatment Effect', markersize=8)

ax.set_xlabel('Week')
ax.set_ylabel('Coefficient (vs Week 14)')
ax.set_title('Event Study: Treatment Effect Over Time')
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)

            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

treated_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
