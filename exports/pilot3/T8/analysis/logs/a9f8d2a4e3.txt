# Calculate the DiD estimate manually
# Treatment group: before and after
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print(f"Control group - Before: {control_before:.2f}, After: {control_after:.2f}, Change: {control_after - control_before:.2f}")
did_estimate = (treatment_after - treatment_before) - (control_after - control_before)
print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create event study variables - relative time to switch
# Week 15 is when switch happened (after_switch=1 starts)
# So weeks 6-14 are before (relative time -9 to -1), weeks 15-22 are after (relative time 0 to 7)
df['relative_week'] = df['week'] - 15

# Create event study dummies (omit one category as reference - let's use week -1, i.e., week 14)
# We'll create indicators for each relative week
event_study_vars = []
for rw in sorted(df['relative_week'].unique()):
    if rw != -1:  # omit week -1 as reference
        var_name = f'rel_week_{rw}'
        df[var_name] = (df['relative_week'] == rw).astype(int)
        event_study_vars.append(var_name)

# Run event study regression with interactions
# We need: switched_branch * each relative week dummy
interaction_terms = [f'switched_branch:{var}' for var in event_study_vars]
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'switched_branch:{var}' for var in event_study_vars])

# Actually, let's do this more carefully with a proper event study
# Create the interaction terms manually
for rw in sorted(df['relative_week'].unique()):
    if rw != -1:
        var_name = f'rel_week_{rw}'
        interact_name = f'treat_rel_{rw}'
        df[interact_name] = df['switched_branch'] * df[var_name]

interact_cols = [f'treat_rel_{rw}' for rw in sorted(df['relative_week'].unique()) if rw != -1]
formula = 'service_score ~ switched_branch + ' + ' + '.join(interact_cols)

model_es = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Regression (relative to week 14 as baseline):")
print("="*70)
# Print just the interaction coefficients
coef_df = model_es.summary2().tables[1]
print(coef_df[coef_df.index.str.contains('treat_rel')])

# Extract coefficients for plotting
results = []
for rw in sorted(df['relative_week'].unique()):
    if rw == -1:
        results.append({'relative_week': rw, 'coef': 0, 'ci_lower': 0, 'ci_upper': 0})  # reference
    else:
        col_name = f'treat_rel_{rw}'
        if col_name in model_es.params:
            coef = model_es.params[col_name]
            ci = model_es.conf_int().loc[col_name]
            results.append({'relative_week': rw, 'coef': coef, 'ci_lower': ci[0], 'ci_upper': ci[1]})

results_df = pd.DataFrame(results)
print("\n\nEvent Study Coefficients:")
print(results_df.sort_values('relative_week').round(3))

# Since switched_branch is collinear with branch FE, we use after_switch and the interaction
# But we need to be careful about the specification

# Let's use a cleaner approach: create a treatment indicator that's 1 only for treated branches in post-period
df['treatment_effect'] = df['switched_branch'] * df['after_switch']

# Model with branch FE

# Add week FE as well (but this will absorb after_switch)
formula_twfe = 'service_score ~ treatment_effect + C(branch_id) + C(week)'
