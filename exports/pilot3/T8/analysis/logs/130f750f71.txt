from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create treatment group indicator and after period indicator
# switched_branch is already the treatment indicator
# after_switch is already the post-period indicator

# Calculate means by group and period
print("=== Service Score Means by Group and Period ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the DiD manually
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nControl (non-switched) branches:")
print(f"  Before: {control_before:.2f}")
print(f"  After: {control_after:.2f}")
print(f"  Change: {control_after - control_before:.2f}")

print(f"\nTreatment (switched) branches:")
print(f"  Before: {treatment_before:.2f}")
print(f"  After: {treatment_after:.2f}")
print(f"  Change: {treatment_after - treatment_before:.2f}")

did_estimate = (treatment_after - treatment_before) - (control_after - control_before)
print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")

# Now run regression with cluster-robust standard errors
print("\n=== Regression Analysis ===")

# Basic DiD model
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("\nBasic DiD Model (no controls):")
print(model.summary())

model_controls = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit()

print(model_controls.summary())

model_cluster = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Branch-level analysis: calculate before/after change for each branch
branch_before = df[df['after_switch'] == 0].groupby('branch_id')['service_score'].mean()
branch_after = df[df['after_switch'] == 1].groupby('branch_id')['service_score'].mean()

branch_changes = pd.DataFrame({
    'before': branch_before,
    'after': branch_after,
    'change': branch_after - branch_before
})

# Merge with switched status
branch_info = df.groupby('branch_id')['switched_branch'].first()
branch_changes['switched'] = branch_info

print("=== Branch-level Changes ===")
print(f"\nSwitched branches (n={branch_changes[branch_changes['switched']==1].shape[0]}):")
switched_changes = branch_changes[branch_changes['switched']==1]['change']
print(f"  Mean change: {switched_changes.mean():.2f}")
print(f"  Std dev: {switched_changes.std():.2f}")
print(f"  Median: {switched_changes.median():.2f}")
print(f"  Min: {switched_changes.min():.2f}, Max: {switched_changes.max():.2f}")
print(f"  % positive: {(switched_changes > 0).mean()*100:.1f}%")

print(f"\nNon-switched branches (n={branch_changes[branch_changes['switched']==0].shape[0]}):")
control_changes = branch_changes[branch_changes['switched']==0]['change']
print(f"  Mean change: {control_changes.mean():.2f}")
print(f"  Std dev: {control_changes.std():.2f}")
print(f"  Median: {control_changes.median():.2f}")
print(f"  Min: {control_changes.min():.2f}, Max: {control_changes.max():.2f}")
print(f"  % positive: {(control_changes > 0).mean()*100:.1f}%")

# T-test comparing changes between groups
from scipy import stats
t_stat, p_val = stats.ttest_ind(switched_changes, control_changes)
print(f"\n=== Two-sample t-test on branch-level changes ===")
print(f"t-statistic: {t_stat:.2f}, p-value: {p_val:.4f}")

# Check if effect varies by branch characteristics
# Add staffing and visits averages
branch_staffing = df.groupby('branch_id')['staffing_index'].mean()
branch_visits = df.groupby('branch_id')['visits_index'].mean()
branch_changes['avg_staffing'] = branch_staffing
branch_changes['avg_visits'] = branch_visits

# Regression of branch-level change on switched status
branch_changes_reset = branch_changes.reset_index()
branch_model = ols('change ~ switched', data=branch_changes_reset).fit()
print("\n=== Branch-level change regression ===")
print(branch_model.summary())

# Heterogeneity: does effect vary by staffing or visits?
branch_model_het = ols('change ~ switched + avg_staffing + avg_visits + switched:avg_staffing + switched:avg_visits', 
                       data=branch_changes_reset).fit()
print("\n=== Heterogeneity analysis ===")
print(branch_model_het.summary())
