import pandas as pd
import numpy as np
from linearmodels.panel import PanelOLS

df = pd.read_csv('branches.csv')
df = df.set_index(['branch_id', 'week'])
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Main model with covariates
model = PanelOLS(df['service_score'], 
                 df[['after_switch', 'treated_post', 'staffing_index', 'visits_index']], 
                 entity_effects=True)
result = model.fit(cov_type='clustered', cluster_entity=True)

# Extract key statistics
coef = result.params['treated_post']
se = result.std_errors['treated_post']
pval = result.pvalues['treated_post']

ci_upper = result.conf_int.loc['treated_post', 'upper']

print("=== KEY RESULTS ===")
print(f"Treatment effect (treated_post coefficient): {coef:.2f}")
print(f"Standard error: {se:.2f}")
print(f"P-value: {pval:.3f}")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")

# Calculate effect size relative to baseline
baseline = df[df['after_switch'] == 0]['service_score'].mean()
print(f"\nBaseline average service score (pre-period): {baseline:.2f}")
print(f"Effect size as % of baseline: {coef/baseline*100:.1f}%")

print("\n=== GROUP STATISTICS ===")
