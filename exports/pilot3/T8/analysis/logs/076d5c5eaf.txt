import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?

print(f"Switched: {switched_counts.get(1, 0)} ({switched_counts.get(1, 0)/switched_counts.sum()*100:.1f}%)")
print(f"Did not switch: {switched_counts.get(0, 0)} ({switched_counts.get(0, 0)/switched_counts.sum()*100:.1f}%)")

print("="*60)

before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

print("\nOverall service scores:")
print(f"Before switch (weeks 6-14): mean = {before['service_score'].mean():.2f}, n = {len(before)}")
print(f"After switch (weeks 15-22): mean = {after['service_score'].mean():.2f}, n = {len(after)}")

print("\n\nBy switch status - BEFORE switch (weeks 6-14):")
before_grouped = before.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print(before_grouped)

print("\nBy switch status - AFTER switch (weeks 15-22):")
after_grouped = after.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print(after_grouped)

# Calculate the change for each group
print("\n\nChange in service scores (After - Before):")
switched_before = before[before['switched_branch'] == 1]['service_score'].mean()
switched_after = after[after['switched_branch'] == 1]['service_score'].mean()
nonswitched_before = before[before['switched_branch'] == 0]['service_score'].mean()
nonswitched_after = after[after['switched_branch'] == 0]['service_score'].mean()

switched_change = switched_after - switched_before
nonswitched_change = nonswitched_after - nonswitched_before
diff_in_diff = switched_change - nonswitched_change

print(f"Non-switched branches: {nonswitched_before:.2f} -> {nonswitched_after:.2f} (change: {nonswitched_change:+.2f})")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Run regression with branch fixed effects using dummy variables
# This is computationally intensive but gives us proper FE estimates
branch_dummies = pd.get_dummies(df['branch_id'], prefix='b', drop_first=True)
df_fe = pd.concat([df[['service_score', 'switched_branch', 'after_switch', 'treated_post']], 
                   branch_dummies], axis=1)

# Build the formula
branch_cols = [c for c in df_fe.columns if c.startswith('b_')]
formula = f"service_score ~ switched_branch + after_switch + treated_post + {' + '.join(branch_cols)}"

# Fit the model
model_fe = ols(formula, data=df_fe).fit()

# Get just the key coefficients
print("Difference-in-Differences with Branch Fixed Effects:")
print("="*70)
print("\nKey coefficients:")
print(model_fe.params[['switched_branch', 'after_switch', 'treated_post']])
print("\nStandard errors:")
print(model_fe.bse[['switched_branch', 'after_switch', 'treated_post']])
print("\nP-values:")
print(model_fe.pvalues[['switched_branch', 'after_switch', 'treated_post']])

# Now cluster by branch for robust standard errors
# Using cluster-robust SEs
model_fe_clustered = model_fe.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print("\n\nWith Cluster-Robust Standard Errors (clustered by branch):")
print("="*70)
print("\nKey coefficients with clustered SEs:")
print(model_fe_clustered.params[['switched_branch', 'after_switch', 'treated_post']])
print("\nClustered standard errors:")
print(model_fe_clustered.bse[['switched_branch', 'after_switch', 'treated_post']])
print("\nClustered p-values:")
print(model_fe_clustered.pvalues[['switched_branch', 'after_switch', 'treated_post']])
print("\n95% CI for treated_post effect:")
ci = model_fe_clustered.conf_int().loc['treated_post']
print(f"[{ci[0]:.3f}, {ci[1]:.3f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Basic DiD model with clustered standard errors
model = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()

# Get cluster-robust standard errors
model_clustered = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("Difference-in-Differences Analysis")
print("="*70)
print("\nModel: service_score ~ switched_branch + after_switch + treated_post")
print("Standard errors clustered by branch")
print("\n" + "="*70)
print(model_clustered.summary().tables[1])

# Extract key results
coef = model_clustered.params['treated_post']
se = model_clustered.bse['treated_post']
pval = model_clustered.pvalues['treated_post']
ci = model_clustered.conf_int().loc['treated_post']

print("\n" + "="*70)
print("KEY RESULT: Effect of New Procedure on Service Score")
print("="*70)
print(f"Estimated effect: +{coef:.2f} points")
print(f"95% Confidence Interval: [{ci[0]:.2f}, {ci[1]:.2f}]")
print(f"P-value: {pval:.4f}")
print(f"Statistically significant: {'Yes' if pval < 0.05 else 'No'}")

# Also check if we should control for staffing and visits
print("\n\n" + "="*70)
print("Model with Controls (staffing_index, visits_index)")
print("="*70)
model_controls = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', data=df).fit()
model_controls_clustered = model_controls.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(model_controls_clustered.summary().tables[1])

coef_ctrl = model_controls_clustered.params['treated_post']
se_ctrl = model_controls_clustered.bse['treated_post']
pval_ctrl = model_controls_clustered.pvalues['treated_post']
ci_ctrl = model_controls_clustered.conf_int().loc['treated_post']

print(f"\nEffect with controls: +{coef_ctrl:.2f} points")
print(f"95% CI: [{ci_ctrl[0]:.2f}, {ci_ctrl[1]:.2f}]")
