import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Calculate averages for the 2x2 table
print("=== Difference-in-Differences Summary ===\n")

# Before period (after_switch = 0)
before_switched = df[(df['after_switch'] == 0) & (df['switched_branch'] == 1)]['service_score'].mean()
before_control = df[(df['after_switch'] == 0) & (df['switched_branch'] == 0)]['service_score'].mean()

# After period (after_switch = 1)
after_switched = df[(df['after_switch'] == 1) & (df['switched_branch'] == 1)]['service_score'].mean()
after_control = df[(df['after_switch'] == 1) & (df['switched_branch'] == 0)]['service_score'].mean()

print("Service Score Averages:")
print(f"  Switched branches - Before: {before_switched:.2f}, After: {after_switched:.2f}, Change: {after_switched - before_switched:.2f}")
print(f"  Control branches  - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")
print(f"\n  DiD Estimate (treatment effect): {(after_switched - before_switched) - (after_control - before_control):.2f}")

# Run regression with standard errors clustered by branch
print("\n=== Regression Analysis ===\n")

# Simple OLS first
model = ols('service_score ~ treated + post + DiD', data=df).fit()
print("Simple OLS (no clustering):")
print(model.summary().tables[1])

# Clustered standard errors by branch
model_clustered = ols('service_score ~ treated + post + DiD', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nWith Clustered Standard Errors (by branch):")
print(model_clustered.summary().tables[1])

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

print("=== CRITICAL CHECKS ===\n")

# 1. Check balance - how many branches in each group?
print("1. GROUP SIZES:")
n_switched = df['branch_id'][df['switched_branch'] == 1].nunique()
n_control = df['branch_id'][df['switched_branch'] == 0].nunique()
print(f"   Switched branches: {n_switched}")
print(f"   Control branches: {n_control}")
print(f"   Total: {n_switched + n_control}")

# 2. Check for any branches that switched mid-period (should be none based on design)
print("\n2. SWITCHING CONSISTENCY:")
branch_switch_status = df.groupby('branch_id')['switched_branch'].first()
print(f"   All branches have consistent switch status: {len(branch_switch_status.unique()) == 1 or set(branch_switch_status.unique()) <= {0, 1}}")

# 3. Pre-treatment trend test - regress service_score on treated, week, and treated*week interaction
print("\n3. PRE-TREATMENT PARALLEL TRENDS TEST:")
pre_data = df[df['week'] < 15].copy()
pre_data['treated_week_interaction'] = pre_data['treated'] * pre_data['week']
pre_model = ols('service_score ~ treated + week + treated_week_interaction', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print(f"   Treated*Week interaction coefficient: {pre_model.params['treated_week_interaction']:.3f}")
print(f"   P-value: {pre_model.pvalues['treated_week_interaction']:.3f}")
print(f"   Interpretation: {'Parallel trends NOT rejected (good)' if pre_model.pvalues['treated_week_interaction'] > 0.05 else 'WARNING: Parallel trends may be violated'}")

# 4. Check if effect is driven by outliers
print("\n4. OUTLIER CHECK:")
print(f"   Service score range: {df['service_score'].min():.1f} to {df['service_score'].max():.1f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Event study - estimate effect for each week relative to baseline
print("=== EVENT STUDY ANALYSIS ===\n")

# Create week dummies interacted with treated
for week in range(6, 23):
    df[f'week_{week}'] = (df['week'] == week).astype(int)
    df[f'treated_week_{week}'] = df['treated'] * df[f'week_{week}']

# Use week 6 as baseline, estimate coefficients for weeks 7-22
# This shows the treated-control difference for each week
print("Weekly treated-control differences (relative to week 6 baseline):")
results = []
for week in range(6, 23):
    week_data = df[df['week'] == week]
    diff = week_data[week_data['switched_branch'] == 1]['service_score'].mean() - \
           week_data[week_data['switched_branch'] == 0]['service_score'].mean()
    results.append((week, diff))
    marker = " <-- SWITCH" if week == 14 else ""
    print(f"   Week {week}: {diff:+.2f}{marker}")

# Calculate average pre and post with standard errors
print("\n=== CONFIDENCE INTERVALS ===")
pre_data = df[df['week'] < 15]
post_data = df[df['week'] >= 15]

# Bootstrap for CI
np.random.seed(42)
n_boot = 1000
did_boot = []
branches = df['branch_id'].unique()

for _ in range(n_boot):
    # Sample branches with replacement
    sampled_branches = np.random.choice(branches, size=len(branches), replace=True)
    boot_df = df[df['branch_id'].isin(sampled_branches)]
    
    if len(boot_df) > 100:
        before_switched = boot_df[(boot_df['post'] == 0) & (boot_df['treated'] == 1)]['service_score'].mean()
        before_control = boot_df[(boot_df['post'] == 0) & (boot_df['treated'] == 0)]['service_score'].mean()
        after_switched = boot_df[(boot_df['post'] == 1) & (boot_df['treated'] == 1)]['service_score'].mean()
        after_control = boot_df[(boot_df['post'] == 1) & (boot_df['treated'] == 0)]['service_score'].mean()
        
        did = (after_switched - before_switched) - (after_control - before_control)
        did_boot.append(did)

did_boot = np.array(did_boot)
print(f"Bootstrap DiD estimate: {np.mean(did_boot):.2f}")
print(f"Bootstrap 95% CI: {np.percentile(did_boot, 2.5):.2f} to {np.percentile(did_boot, 97.5):.2f}")
print(f"Bootstrap std error: {np.std(did_boot):.2f}")

# Check if post-treatment effects are significantly different from pre-treatment
print("\n=== PRE vs POST COMPARISON ===")
pre_avg = np.mean([r[1] for r in results if r[0] < 15])
post_avg = np.mean([r[1] for r in results if r[0] >= 15])
print(f"Average pre-treatment difference: {pre_avg:.2f}")
print(f"Average post-treatment difference: {post_avg:.2f}")
print(f"Estimated treatment effect: {post_avg - pre_avg:.2f}")
