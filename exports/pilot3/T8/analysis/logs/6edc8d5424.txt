import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study / parallel trends check
# Create week dummies interacted with treatment
df['week_centered'] = df['week'] - 14  # center around switch week

# Run regression with week fixed effects to check for differential pre-trends
print("=== Event Study Regression (checking parallel trends) ===")
# Create week dummies (using week 14 as reference)
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=False)
df = pd.concat([df, week_dummies], axis=1)

# Get week columns
week_cols = [c for c in df.columns if c.startswith('week_')]

# Build formula for event study - interact each week with treatment
# Using week 14 as baseline
treatment_weeks = [f'week_{w}:switched_branch' for w in range(6, 23) if w != 14]
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'week_{w}' for w in range(6, 23) if w != 14]) + ' + ' + ' + '.join(treatment_weeks)

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract treatment-week interaction coefficients
print("\nTreatment-week interaction coefficients (relative to week 14):")
pre_trend_coefs = []
post_trend_coefs = []
for w in range(6, 23):
    if w == 14:
        continue
    coef_name = f'switched_branch:week_{w}'
    if coef_name in model_event.params.index:
        coef = model_event.params[coef_name]
        se = model_event.bse[coef_name]
        pval = model_event.pvalues[coef_name]
        period = "PRE" if w < 14 else "POST"
        print(f"Week {w} ({period}): coef={coef:.2f}, SE={se:.2f}, p={pval:.3f}")
        if w < 14:
            pre_trend_coefs.append((w, coef, se))
        else:
            post_trend_coefs.append((w, coef, se))

# Test if pre-trends are jointly zero
print("\n=== Pre-trend Joint Significance Test ===")
pre_trend_vars = [f'switched_branch:week_{w}' for w in range(6, 14)]
pre_trend_vars = [v for v in pre_trend_vars if v in model_event.params.index]
if len(pre_trend_vars) > 0:
    from statsmodels.stats.anova import anova_lm
    # Wald test for joint significance
    hypotheses = ', '.join([f'{v} = 0' for v in pre_trend_vars])
    wald_test = model_event.wald_test(hypotheses)
    print(f"Wald test for pre-trends: F={wald_test.statistic[0,0]:.2f}, p={wald_test.pvalue:.3f}")

# Create visualization data
plt.figure(figsize=(12, 6))

# Calculate weekly averages by group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

weeks = weekly_avg.index
control = weekly_avg[0].values
treatment = weekly_avg[1].values

plt.plot(weeks, control, 'o-', label='Control (No Switch)', linewidth=2, markersize=8)
plt.plot(weeks, treatment, 's-', label='Treatment (Switched)', linewidth=2, markersize=8)

plt.axvspan(14.5, 22, alpha=0.2, color='green', label='Post-Switch Period')
plt.axvspan(6, 14.5, alpha=0.2, color='orange', label='Pre-Switch Period')

plt.xlabel('Week', fontsize=12)
plt.ylabel('Service Score', fontsize=12)

plt.legend(loc='best')
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
