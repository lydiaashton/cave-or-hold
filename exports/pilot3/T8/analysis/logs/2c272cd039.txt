import pandas as pd
import numpy as np
import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch*after_switch) + error
# beta3 is the DiD estimator

model = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Simple DiD Regression:")
print(model.summary())

# Now let's cluster standard errors at the branch level (more appropriate for panel data)
model_clustered = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with Clustered SE (by branch):")
print(model_clustered.summary())

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Add branch fixed effects for a more robust specification
# This controls for time-invariant branch characteristics
df = df.copy()
df['branch_id'] = df['branch_id'].astype(str)

# Model with branch fixed effects
model_fe = smf.ols('service_score ~ after_switch + treatment + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("DiD with Branch Fixed Effects (Clustered SE):")
print(model_fe.summary())

# Also check effect size in context
print("\n\nEffect size context:")
print(f"Baseline service score (control, pre): {df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean():.2f}")
print(f"Treatment effect: {model_fe.params['treatment']:.2f} points")
print(f"Effect as % of baseline: {model_fe.params['treatment'] / df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean() * 100:.1f}%")

# Confidence interval
print(f"\n95% CI for treatment effect: [{model_fe.conf_int().loc['treatment'][0]:.2f}, {model_fe.conf_int().loc['treatment'][1]:.2f}]")
