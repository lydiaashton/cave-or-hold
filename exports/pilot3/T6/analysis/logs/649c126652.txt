import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

# Calculate means by group and period
print("=== Mean Service Scores by Group and Period ===")
pivot = df.pivot_table(values='service_score', 
                       index='switched_branch', 
                       columns='after_switch', 
                       aggfunc='mean')
print(pivot)
print()

# Calculate DiD manually
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
before_treated = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_treated = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print("Control group (no switch):")
print(f"  Before: {before_control:.2f}")
print(f"  After: {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")
print()
print("Treated group (switched):")
print(f"  Before: {before_treated:.2f}")
print(f"  After: {after_treated:.2f}")
print(f"  Change: {after_treated - before_treated:.2f}")
print()

did_estimate = (after_treated - before_treated) - (after_control - before_control)
print(f"Difference-in-Differences Estimate: {did_estimate:.2f}")
print()

# Run regression with standard errors clustered by branch
print("=== Regression Analysis ===")
model = ols('service_score ~ treated + post + treated_post', data=df).fit()
print(model.summary())
print()

# Now with cluster-robust standard errors
model_clustered = ols('service_score ~ treated + post + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Regression with Cluster-Robust SE (by branch) ===")
print(model_clustered.summary())

pre_data = df[df['week'].isin(pre_weeks)]

box_data = [before_control, after_control, before_treated, after_treated]
