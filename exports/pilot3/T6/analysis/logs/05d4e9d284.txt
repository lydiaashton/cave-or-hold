import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimator

df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression with cluster-robust standard errors at branch level
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Basic DiD Regression (without cluster-robust SE):")
print(model.summary())

# Now with cluster-robust standard errors
model_cluster = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with Cluster-Robust Standard Errors (clustered by branch):")
print(model_cluster.summary())

# Also add controls for staffing_index and visits_index
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with Controls (clustered by branch):")
print(model_controls.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check pre-trends statistically (weeks 6-14 only)
pre_period = df[df['week'] < 15]
pre_model = ols('service_score ~ week + switched_branch + week:switched_branch', data=pre_period).fit(cov_type='cluster', cov_kwds={'groups': pre_period['branch_id']})
print("Pre-trends test (interaction of week and switched_branch):")
print(pre_model.summary())

# Also run an event study style analysis
# Create relative week indicators
df['relative_week'] = df['week'] - 15  # week 15 is 0, week 14 is -1, etc.

# Create dummies for each week relative to switch (excluding week 14 as reference)
for w in range(-9, 8):  # weeks -9 to 7 relative to switch
    if w != -1:  # skip week -1 (week 14) as reference
        df[f'week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
event_study_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_study_vars)
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nEvent Study Regression:")
print(event_model.summary())

# Extract coefficients for the plot
coeffs = []
weeks_rel = []
for w in range(-9, 8):
    if w == -1:
        coeffs.append(0)  # reference period
        weeks_rel.append(w)
    else:
        var_name = f'week_{w}'
        if var_name in event_model.params:
            coeffs.append(event_model.params[var_name])
            weeks_rel.append(w)

print("\n\nEvent Study Coefficients:")
for w, c in zip(weeks_rel, coeffs):
    print(f"Week {w+15} (relative {w}): {c:.3f}")

# Create dummies for each week relative to switch (excluding week 14 as reference)
for w in range(-9, 8):  # weeks -9 to 7 relative to switch
    if w != -1:  # skip week -1 (week 14) as reference
        df[f'w{abs(w)}_neg' if w < 0 else f'w{w}_pos'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Build formula
event_study_vars = []
for w in range(-9, 8):
    if w == -1:
        continue  # reference period
    elif w < 0:
        event_study_vars.append(f'w{abs(w)}_neg')
    else:
        event_study_vars.append(f'w{w}_pos')

formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_study_vars)
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients and confidence intervals for plotting
coeffs = []
ci_lower = []
ci_upper = []
weeks_rel = []

    weeks_rel.append(w)

        ci_lower.append(0)
        ci_upper.append(0)
    else:
        if w < 0:
            var_name = f'w{abs(w)}_neg'
        else:
            var_name = f'w{w}_pos'
        
        coeffs.append(event_model.params[var_name])
        ci_lower.append(event_model.conf_int().loc[var_name, 0])
        ci_upper.append(event_model.conf_int().loc[var_name, 1])
