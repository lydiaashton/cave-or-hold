import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*treatment + error
# beta3 is the treatment effect

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== DIFFERENCE-IN-DIFFERENCES REGRESSION ===")
print(model.summary())

# Also run with cluster-robust standard errors at branch level
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== WITH CLUSTER-ROBUST STANDARD ERRORS (by branch) ===")
print(model_clustered.summary())

import pandas as pd
import numpy as np

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Check parallel trends: compare trends before week 15
before_data = df[df['week'] < 15].copy()
before_data['week_centered'] = before_data['week'] - before_data['week'].mean()
parallel_model = ols('service_score ~ switched_branch * week_centered', data=before_data).fit(cov_type='cluster', cov_kwds={'groups': before_data['branch_id']})
print("=== PARALLEL TRENDS TEST (pre-switch period only) ===")
print(parallel_model.summary())

# Now run DiD with covariates (staffing_index and visits_index)
df['treatment'] = df['switched_branch'] * df['after_switch']
model_with_covariates = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
                            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD WITH COVARIATES ===")
print(model_with_covariates.summary())

# Also check branch fixed effects model
# This controls for time-invariant branch characteristics
df_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_fe = pd.concat([df, df_dummies], axis=1)

# Build formula with branch fixed effects
branch_cols = [col for col in df_fe.columns if col.startswith('branch_')]
formula = 'service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index + ' + ' + '.join(branch_cols)
model_fe = ols(formula, data=df_fe).fit(cov_type='cluster', cov_kwds={'groups': df_fe['branch_id']})
print("\n=== DiD WITH BRANCH FIXED EFFECTS ===")
# Just print the key coefficients
print("\nKey coefficients:")
print(model_fe.params[['treatment', 'after_switch', 'switched_branch']])

    cov_type='cluster', cov_kwds={'groups': df['branch_id']})
