import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Calculate means by group and period
print("=== Mean Service Scores by Group and Period ===")
means = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(means)

# Calculate the DiD estimate manually
print("\n=== Manual DiD Calculation ===")
# Control group (switched_branch=0)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_change = control_after - control_before

# Treatment group (switched_branch=1)
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_change = treat_after - treat_before

did_estimate = treat_change - control_change

print(f"Control group - Before: {control_before:.2f}, After: {control_after:.2f}, Change: {control_change:.2f}")
print(f"Treatment group - Before: {treat_before:.2f}, After: {treat_after:.2f}, Change: {treat_change:.2f}")
print(f"\nDiD Estimate (Treatment Effect): {did_estimate:.3f}")

# Run regression with branch fixed effects
print("\n=== Regression with Branch Fixed Effects ===")
df['branch_id_str'] = df['branch_id'].astype(str)
model = ols('service_score ~ treatment + after_switch + C(branch_id_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model.summary().tables[1])

df['branch_id_str'] = df['branch_id'].astype(str)

# Create event study with better variable names
df['week_relative'] = df['week'] - 15  # Week 15 is period 0

# Create dummies for each week relative to treatment
for w in range(-9, 8):  # weeks -9 to +7 relative to treatment
    var_name = f'w{w}' if w >= 0 else f'w_neg{-w}'
    df[var_name] = ((df['week_relative'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression - omitting week -1 (w_neg1) as reference
event_vars = [f'w{w}' if w >= 0 else f'w_neg{-w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ ' + ' + '.join(event_vars) + ' + C(branch_id_str)'
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Coefficients (relative to week -1) ===")
event_results = []
for w in range(-9, 8):

    else:
        var_name = f'w{w}' if w >= 0 else f'w_neg{-w}'
        coef = model_event.params[var_name]
        se = model_event.bse[var_name]
        ci_low = model_event.conf_int().loc[var_name][0]
        ci_high = model_event.conf_int().loc[var_name][1]

df = pd.read_csv('branches.csv')

df['branch_id_str'] = df['branch_id'].astype(str)

# Main DiD regression with branch fixed effects
model = ols('service_score ~ treatment + after_switch + C(branch_id_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get key statistics
treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_ci = model.conf_int().loc['treatment']
treatment_pval = model.pvalues['treatment']

# Calculate effect size relative to baseline
baseline_mean = df['service_score'].mean()
effect_size_pct = (treatment_coef / baseline_mean) * 100

# Get sample sizes
n_branches = df['branch_id'].nunique()
n_switched = df[df['switched_branch']==1]['branch_id'].nunique()
n_control = df[df['switched_branch']==0]['branch_id'].nunique()
n_observations = len(df)

# Pre vs post means
pre_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
post_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

post_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print(f"=== SUMMARY STATISTICS FOR BOARD REPORT ===")
