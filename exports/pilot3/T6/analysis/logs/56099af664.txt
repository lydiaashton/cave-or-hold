import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Count branches by switch status
switch_counts = df.groupby('branch_id')['switched_branch'].first().value_counts()
print("Branches by switch status:")
print(switch_counts)
print(f"\nTotal: {switch_counts.sum()} branches")
print(f"Switched: {switch_counts.get(1, 0)} ({100*switch_counts.get(1, 0)/switch_counts.sum():.1f}%)")
print(f"Not switched: {switch_counts.get(0, 0)} ({100*switch_counts.get(0, 0)/switch_counts.sum():.1f}%)")

# Calculate mean service scores by group and period
print("\n\nMean service scores by group and period:")
grouped = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(grouped)

# Calculate the difference-in-differences manually
print("\n\nDifference-in-Differences Calculation:")
# Treatment group (switched_branch=1)
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treatment_diff = treatment_after - treatment_before

# Control group (switched_branch=0)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_diff = control_after - control_before

did_effect = treatment_diff - control_diff

print(f"Treatment group - Before: {treatment_before:.2f}, After: {treatment_after:.2f}, Change: {treatment_diff:.2f}")
print(f"Control group - Before: {control_before:.2f}, After: {control_after:.2f}, Change: {control_diff:.2f}")
print(f"\nDifference-in-Differences estimate: {did_effect:.2f}")

# Run a regression for DiD with standard errors
print("\n\nRegression Analysis (Difference-in-Differences):")
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model.summary())

# Add branch fixed effects
print("\n\nWith Branch Fixed Effects:")
df['branch_id'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit()
print(model_fe.summary())

# Cluster-robust standard errors by branch
print("\n\nWith Branch Fixed Effects and Cluster-Robust SEs:")
model_fe_cluster = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id'].astype(int)})
print(model_fe_cluster.summary())

import numpy as np

from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Clean DiD with branch fixed effects (switched_branch is absorbed by FE)
print("Difference-in-Differences with Branch Fixed Effects:")
model_fe = ols('service_score ~ after_switch + treatment + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id'].astype(int)})

# Check parallel trends - look at trends before the switch
print("\n\nParallel Trends Check - Mean service scores by week and group:")

import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# One-sided p-value for the treatment effect
from statsmodels.formula.api import ols
model = ols('service_score ~ after_switch + treatment + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id'].astype(int)})

coef = model.params['treatment']
se = model.bse['treatment']
t_stat = coef / se
# One-sided p-value (testing if effect > 0)
one_sided_p = 1 - stats.norm.cdf(t_stat)

print(f"Treatment coefficient: {coef:.2f}")
print(f"Standard error: {se:.2f}")
print(f"T-statistic: {t_stat:.2f}")
print(f"Two-sided p-value: {model.pvalues['treatment']:.3f}")
print(f"One-sided p-value: {one_sided_p:.3f}")

# Power analysis approximation
print(f"\n90% CI: [{model.conf_int(alpha=0.10).loc['treatment'][0]:.2f}, {model.conf_int(alpha=0.10).loc['treatment'][1]:.2f}]")

# Effect size in practical terms
baseline = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
print(f"\nBaseline (control, before): {baseline:.1f}")
print(f"Effect as % of baseline: {100*coef/baseline:.1f}%")

# What would we need for significance?
print(f"\nFor p<0.05 (two-sided), we'd need t-stat > 1.96")
print(f"Current t-stat: {t_stat:.2f}")
print(f"Required coefficient for significance (same SE): {1.96*se:.2f}")
