import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression (no fixed effects)
model1 = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
print("Model 1: Basic DiD (no fixed effects)")
print(model1.summary().tables[1])
print()

# DiD with branch fixed effects
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_branch = pd.concat([df, branch_dummies], axis=1)

# Build formula with branch fixed effects
branch_cols = [c for c in df_with_branch.columns if c.startswith('branch_')]
formula = f'service_score ~ switched_branch + after_switch + did + {" + ".join(branch_cols)}'
model2 = ols(formula, data=df_with_branch).fit()
print("Model 2: DiD with Branch Fixed Effects")
print(f"R-squared: {model2.rsquared:.4f}")
print(f"DiD coefficient: {model2.params['did']:.4f}")
print(f"DiD std err: {model2.bse['did']:.4f}")
print(f"DiD t-stat: {model2.tvalues['did']:.4f}")
print(f"DiD p-value: {model2.pvalues['did']:.6f}")
ci = model2.conf_int().loc['did']
print(f"DiD 95% CI: [{ci[0]:.4f}, {ci[1]:.4f}]")
print()

# DiD with branch and week fixed effects
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df_full = pd.concat([df, branch_dummies, week_dummies], axis=1)

week_cols = [c for c in df_full.columns if c.startswith('week_')]
formula_full = f'service_score ~ switched_branch + after_switch + did + {" + ".join(branch_cols)} + {" + ".join(week_cols)}'

print(f"DiD 95% CI: [{ci[0]:.4f}, {ci[1]:.4f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Run a simple regression to test if pre-trends are flat
pre_data = df[df['week'] <= 14].copy()
pre_data['week_centered'] = pre_data['week'] - 10  # Center around middle of pre-period
pre_data['interaction'] = pre_data['switched_branch'] * pre_data['week_centered']

trend_test = ols('service_score ~ switched_branch + week_centered + interaction', data=pre_data).fit()
print("Pre-trend test (interaction term should be ~0 if parallel trends hold):")
print(f"Interaction coefficient: {trend_test.params['interaction']:.4f}")
print(f"Interaction p-value: {trend_test.pvalues['interaction']:.4f}")
print()

# Event study: estimate effect for each week relative to switch
# Create relative week variable (week 15 is the first post-switch week)
df['relative_week'] = df['week'] - 15  # Week 15 becomes 0, week 14 becomes -1, etc.

# Create dummies for each week relative to switch (excluding week 14 as reference)
# We'll estimate coefficients for each week
event_study_data = df.copy()

# Create week dummies
for w in range(-9, 8):  # weeks -9 to 7 (week 6 to 22, relative to week 15)
    if w != -1:  # Skip week -1 (week 14) as reference
        event_study_data[f'week_{w}'] = ((event_study_data['relative_week'] == w) & 
                                          (event_study_data['switched_branch'] == 1)).astype(int)

# Build formula
week_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = f"service_score ~ switched_branch + after_switch + {' + '.join(week_vars)}"

# Add branch and week fixed effects
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
week_dummies = pd.get_dummies(df['week'], prefix='wk', drop_first=True)
event_study_full = pd.concat([event_study_data, branch_dummies, week_dummies], axis=1)

branch_cols = [c for c in event_study_full.columns if c.startswith('branch_')]
wk_cols = [c for c in event_study_full.columns if c.startswith('wk_')]

formula_full = f"service_score ~ switched_branch + after_switch + {' + '.join(week_vars)} + {' + '.join(branch_cols)} + {' + '.join(wk_cols)}"

event_model = ols(formula_full, data=event_study_full).fit()

print("Event Study Results (coefficients for treatment branches by week relative to switch):")
print("(Week -1 / Week 14 is the reference category)")
print()

results = []
for w in range(-9, 8):
    if w != -1:
        var_name = f'week_{w}'
        coef = event_model.params[var_name]
        se = event_model.bse[var_name]
        pval = event_model.pvalues[var_name]
        ci_low = event_model.conf_int().loc[var_name][0]
        ci_high = event_model.conf_int().loc[var_name][1]
        results.append({'relative_week': w, 'coef': coef, 'se': se, 'pval': pval, 'ci_low': ci_low, 'ci_high': ci_high})
        week_num = w + 15
        print(f"Week {week_num:2d} (rel {w:2d}): coef={coef:6.3f}, SE={se:.3f}, 95% CI=[{ci_low:.3f}, {ci_high:.3f}], p={pval:.4f}")

print()
print("Pre-switch weeks (should be close to 0 if parallel trends hold):")
pre_results = [r for r in results if r['relative_week'] < -1]
print(f"Average pre-switch coefficient: {np.mean([r['coef'] for r in pre_results]):.3f}")
print(f"Pre-switch coefficients are jointly insignificant: {all(r['pval'] > 0.05 for r in pre_results)}")

import pandas as pd

# Load the data
df = pd.read_csv('branches.csv')

# Create relative week (week 15 = 0)
df['relative_week'] = df['week'] - 15

# Create event study dummies for treatment branches only (relative to week 14 as baseline)
# Week 14 is relative_week = -1, which will be our reference
for w in range(-9, 8):
    if w != -1:  # Skip reference week
        df[f'treat_week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Prepare data for regression
# Create branch and week fixed effects
branch_dummies = pd.get_dummies(df['branch_id'], prefix='br', drop_first=True)
week_dummies = pd.get_dummies(df['week'], prefix='wk', drop_first=True)

# Combine all variables
event_cols = [c for c in df.columns if c.startswith('treat_week_')]
X = pd.concat([df[['switched_branch', 'after_switch'] + event_cols], branch_dummies, week_dummies], axis=1)
y = df['service_score']

# Drop one of the treatment week dummies to avoid collinearity (already done by skipping -1)
# But we may still have issues, so let's check
X = X.dropna(axis=1)

# Run regression
model = sm.OLS(y, sm.add_constant(X)).fit()

print("Event Study Results:")
print("(Week 14 / relative_week -1 is the reference category)")
print()

results = []
for w in range(-9, 8):
    if w != -1:
        var_name = f'treat_week_{w}'
        if var_name in model.params.index:
            coef = model.params[var_name]
            se = model.bse[var_name]
            pval = model.pvalues[var_name]
            ci = model.conf_int().loc[var_name]
            results.append({'relative_week': w, 'coef': coef, 'se': se, 'pval': pval, 'ci_low': ci[0], 'ci_high': ci[1]})
            week_num = w + 15
