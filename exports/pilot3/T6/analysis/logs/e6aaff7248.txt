import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['switched_branch'] * df['after_switch']

# Summary table: mean service scores by group and period
summary_table = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).unstack()
print("Service Score by Group and Period:")
print(summary_table)

# Calculate the DiD estimate manually
# Control group (switched_branch=0)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_change = control_after - control_before

# Treatment group (switched_branch=1)
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_change = treat_after - treat_before

did_estimate = treat_change - control_change

print(f"\n=== Manual DiD Calculation ===")
print(f"Control (non-switched) before: {control_before:.2f}")
print(f"Control (non-switched) after: {control_after:.2f}")
print(f"Control change: {control_change:.2f}")
print(f"\nTreatment (switched) before: {treat_before:.2f}")
print(f"Treatment (switched) after: {treat_after:.2f}")
print(f"Treatment change: {treat_change:.2f}")
print(f"\nDiD Estimate (Treatment change - Control change): {did_estimate:.2f}")

# Run DiD regression
model = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
print(f"\n=== DiD Regression Results ===")
print(model.summary())

# Cluster-robust standard errors by branch
model_clustered = ols('service_score ~ switched_branch + after_switch + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"\n=== DiD Regression with Clustered SE (by branch) ===")
print(model_clustered.summary())

import pandas as pd

df = pd.read_csv('branches.csv')

# Create weekly averages for visualization
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()

# Plot
fig, ax = plt.subplots(figsize=(10, 6))

for group in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == group]
    label = 'Treatment (New Procedure)' if group == 1 else 'Control (Old Procedure)'
    color = 'blue' if group == 1 else 'gray'
    ax.plot(subset['week'], subset['service_score'], marker='o', label=label, color=color, linewidth=2, markersize=6)

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch implemented (Week 15)')

ax.set_title('Service Score Trends: Treatment vs Control Branches', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=11)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']

# Main DiD model
model = ols('service_score ~ switched_branch + after_switch + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Key Statistics ===")
print(f"DiD Point Estimate: {model.params['did']:.2f} points")
print(f"Standard Error: {model.bse['did']:.2f}")
print(f"t-statistic: {model.tvalues['did']:.2f}")
print(f"p-value: {model.pvalues['did']:.3f}")
print(f"90% CI: [{model.params['did'] - 1.645*model.bse['did']:.2f}, {model.params['did'] + 1.645*model.bse['did']:.2f}]")
print(f"95% CI: [{model.params['did'] - 1.96*model.bse['did']:.2f}, {model.params['did'] + 1.96*model.bse['did']:.2f}]")

# Context
avg_score = df['service_score'].mean()
print(f"\nAverage service score: {avg_score:.1f}")
print(f"Effect as % of average: {model.params['did']/avg_score*100:.1f}%")

# What's the probability the effect is positive?
# Using t-distribution
from scipy import stats
t_stat = model.tvalues['did']
df_residual = model.df_resid
# One-tailed p-value (probability effect > 0)
prob_positive = 1 - stats.t.cdf(0, df_residual, loc=model.params['did'], scale=model.bse['did'])
# Actually simpler: 
prob_positive = stats.t.sf(0, df_residual, loc=model.params['did'], scale=model.bse['did'])
print(f"\nProbability effect is positive: {prob_positive:.0%}")

# Lower bound at 80% CI
lower_80 = model.params['did'] - 1.28 * model.bse['did']
print(f"80% CI lower bound: {lower_80:.2f}")
