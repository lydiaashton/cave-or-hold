df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with cluster-robust standard errors
print("=== DiD Regression (Basic) ===")
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model1.summary())

# Run DiD with controls
print("\n=== DiD Regression (With Controls) ===")
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()

# Cluster-robust standard errors by branch
print("\n=== DiD with Cluster-Robust SEs (by branch) ===")
model3 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
cov_clusters = sm.stats.sandwich_covariance.cov_cluster(model3, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clusters))
print(f"Treatment coefficient: {model3.params['treatment']:.4f}")
print(f"Clustered SE: {se_clustered[3]:.4f}")
t_stat = model3.params['treatment'] / se_clustered[3]

pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()

ci_high = treatment_effect + 1.96 * se
