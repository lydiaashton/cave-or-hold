import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Average service scores by group and period
print("=== Average Service Scores ===")
print("\nBefore switch (weeks 6-14):")
before = df[df['after_switch'] == 0]
print("  Switched branches:", before[before['switched_branch'] == 1]['service_score'].mean())
print("  Non-switched branches:", before[before['switched_branch'] == 0]['service_score'].mean())

print("\nAfter switch (weeks 15-22):")
after = df[df['after_switch'] == 1]
print("  Switched branches:", after[after['switched_branch'] == 1]['service_score'].mean())
print("  Non-switched branches:", after[after['switched_branch'] == 0]['service_score'].mean())

# Calculate the difference-in-differences
before_switched = before[before['switched_branch'] == 1]['service_score'].mean()
before_control = before[before['switched_branch'] == 0]['service_score'].mean()
after_switched = after[after['switched_branch'] == 1]['service_score'].mean()
after_control = after[after['switched_branch'] == 0]['service_score'].mean()

diff_switched = after_switched - before_switched
diff_control = after_control - before_control
did_estimate = diff_switched - diff_control

print(f"\n=== Difference-in-Differences ===")
print(f"Change in switched branches: {diff_switched:.3f}")
print(f"Change in control branches: {diff_control:.3f}")
print(f"DiD estimate (effect of new procedure): {did_estimate:.3f}")

# Week-by-week trends
print("\n=== Week-by-Week Average Service Scores ===")
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly.columns = ['Control', 'Switched']
weekly['Diff'] = weekly['Switched'] - weekly['Control']
print(weekly)

import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# More detailed pre-trends analysis
print("=== Pre-Trends Analysis (Week-by-Week Before Switch) ===")
pre_data = df[df['week'] < 15].copy()
weekly_pre = pre_data.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_pre.columns = ['Control', 'Switched']
weekly_pre['Diff'] = weekly_pre['Switched'] - weekly_pre['Control']
print(weekly_pre)

# Check the trend in the difference
print("\nTrend in pre-switch difference (linear regression):")
from scipy import stats
weeks = weekly_pre.index.values
diffs = weekly_pre['Diff'].values
slope, intercept, r_value, p_value, std_err = stats.linregress(weeks, diffs)
print(f"Slope: {slope:.3f}, P-value: {p_value:.4f}")

# Post-switch analysis
print("\n=== Post-Switch Analysis (Week-by-Week After Switch) ===")
post_data = df[df['week'] >= 15].copy()
weekly_post = post_data.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_post.columns = ['Control', 'Switched']
weekly_post['Diff'] = weekly_post['Switched'] - weekly_post['Control']
print(weekly_post)

# Average post-switch difference
print(f"\nAverage post-switch difference: {weekly_post['Diff'].mean():.3f}")
print(f"Average pre-switch difference: {weekly_pre['Diff'].mean():.3f}")
print(f"Change in difference (DiD): {weekly_post['Diff'].mean() - weekly_pre['Diff'].mean():.3f}")

# Create a visualization
fig, ax = plt.subplots(figsize=(12, 6))

# Plot average service scores by week and group
weekly_all = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_all.columns = ['Control', 'Switched']

ax.plot(weekly_all.index, weekly_all['Control'], 'o-', label='Control (Old Procedure)', color='blue', linewidth=2)
ax.plot(weekly_all.index, weekly_all['Switched'], 's-', label='Switched (New Procedure)', color='green', linewidth=2)

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Point (Week 15)')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Scores by Week: Control vs. Switched Branches', fontsize=14)
ax.legend(loc='best')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("\nPlot saved as service_scores_trend.png")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Final DiD model with robust standard errors clustered by branch
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()

# Get the key statistics
treatment_effect = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pval = model.pvalues['treatment']
treatment_ci_low = model.conf_int().loc['treatment'][0]
treatment_ci_high = model.conf_int().loc['treatment'][1]

print("=== FINAL ANALYSIS SUMMARY ===")
print(f"\nTreatment Effect (New Procedure): {treatment_effect:.2f} points")
print(f"95% Confidence Interval: [{treatment_ci_low:.2f}, {treatment_ci_high:.2f}]")
print(f"P-value: {treatment_pval:.6f}")
print(f"Statistically significant: {'Yes' if treatment_pval < 0.05 else 'No'}")

# Calculate practical significance
mean_score = df['service_score'].mean()
std_score = df['service_score'].std()
print(f"\nPractical Significance:")
print(f"  Mean service score: {mean_score:.2f}")
print(f"  Effect as % of mean: {treatment_effect/mean_score*100:.1f}%")
print(f"  Effect size (Cohen's d): {treatment_effect/std_score:.2f}")

# Group summaries
print("\n=== Group Statistics ===")
before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

print("\nBefore Switch (Weeks 6-14):")
print(f"  Switched branches mean: {before[before['switched_branch']==1]['service_score'].mean():.2f}")
print(f"  Control branches mean: {before[before['switched_branch']==0]['service_score'].mean():.2f}")

print("\nAfter Switch (Weeks 15-22):")
print(f"  Switched branches mean: {after[after['switched_branch']==1]['service_score'].mean():.2f}")
print(f"  Control branches mean: {after[after['switched_branch']==0]['service_score'].mean():.2f}")

print("\n  Switched branches change: {:.2f} points".format(
    after[after['switched_branch']==1]['service_score'].mean() - before[before['switched_branch']==1]['service_score'].mean()))
print("  Control branches change: {:.2f} points".format(
    after[after['switched_branch']==0]['service_score'].mean() - before[before['switched_branch']==0]['service_score'].mean()))

pre_data['week_centered'] = pre_data['week'] - pre_data['week'].mean()
pre_data['treatment_group'] = pre_data['switched_branch']
pre_data['trend_interaction'] = pre_data['week_centered'] * pre_data['treatment_group']

pre_model = ols('service_score ~ week_centered + treatment_group + trend_interaction', data=pre_data).fit()
print(f"   Pre-switch trend difference slope: {pre_model.params['trend_interaction']:.3f}")
print(f"   P-value: {pre_model.pvalues['trend_interaction']:.4f}")
print(f"   → Parallel trends assumption: {'VIOLATED (p<0.05)' if pre_model.pvalues['trend_interaction'] < 0.05 else 'NOT REJECTED'}")

# 2. Event study - look at the pattern
print("\n2. EVENT STUDY PATTERN")
df['rel_week'] = df['week'] - 14
event_cols = []
for w in range(-8, 9):
    if w != 0:
        col = f'rw_{abs(w)}_{"neg" if w < 0 else "pos"}'
        df[col] = ((df['rel_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_cols.append(col)

event_formula = 'service_score ~ switched_branch + C(week) + ' + ' + '.join(event_cols)

for w in range(-8, 0):

    pre_coefs.append(coef)
    sig = "*" if pval < 0.1 else ""
