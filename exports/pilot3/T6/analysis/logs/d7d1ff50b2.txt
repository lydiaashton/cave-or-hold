import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Check how many branches switched vs didn't
print("Branches by treatment status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Check weeks per branch
print("\nWeeks per branch:")
print(df.groupby('branch_id')['week'].count().describe())

# Check if all branches have data for all weeks
print("\nMissing weeks per branch:")
weeks_per_branch = df.groupby('branch_id')['week'].apply(lambda x: sorted(x))
for branch_id in df['branch_id'].unique()[:5]:
    print(f"Branch {branch_id}: {weeks_per_branch[branch_id]}")

# Average service scores by group and period
print("\n\n=== Service Scores by Group and Period ===")
df['period'] = df['after_switch'].map({0: 'Before', 1: 'After'})
df['group'] = df['switched_branch'].map({0: 'Control', 1: 'Treatment'})

summary = df.groupby(['group', 'period'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Calculate the difference-in-differences manually
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
before_treatment = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_treatment = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"\nBefore - Control: {before_control:.2f}")
print(f"After - Control: {after_control:.2f}")
print(f"Control change: {after_control - before_control:.2f}")
print(f"\nBefore - Treatment: {before_treatment:.2f}")
print(f"After - Treatment: {after_treatment:.2f}")
print(f"Treatment change: {after_treatment - before_treatment:.2f}")

did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\n=== Difference-in-Differences Estimate: {did_estimate:.2f} ===")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['did'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
print("=== Basic Difference-in-Differences Regression ===")
model1 = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
print(model1.summary())

# DiD with branch fixed effects
print("\n=== DiD with Branch Fixed Effects ===")
df['branch_id_str'] = df['branch_id'].astype(str)
model2 = ols('service_score ~ switched_branch + after_switch + did + C(branch_id_str)', data=df).fit()
print(model2.summary())

# DiD with branch fixed effects and control variables
print("\n=== DiD with Branch FE and Controls ===")
model3 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index + C(branch_id_str)', data=df).fit()
print(model3.summary())

# Cluster-robust standard errors by branch
print("\n=== DiD with Cluster-Robust SE (by branch) ===")
model4 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index + C(branch_id_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model4.summary())

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']
df['branch_id_str'] = df['branch_id'].astype(str)
df['week_str'] = df['week'].astype(str)

# TWFE with cluster-robust SE
model = ols('service_score ~ did + C(branch_id_str) + C(week_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Key Statistics for Board Decision ===")
print(f"Treatment Effect: {model.params['did']:.2f}")
print(f"Cluster-Robust SE: {model.bse['did']:.2f}")
print(f"95% CI: [{model.params['did'] - 1.96*model.bse['did']:.2f}, {model.params['did'] + 1.96*model.bse['did']:.2f}]")
print(f"P-value: {model.pvalues['did']:.3f}")

# What's the probability the effect is actually negative?
# Using normal approximation
from scipy import stats
z = model.params['did'] / model.bse['did']
prob_negative = stats.norm.cdf(-z)
print(f"\nProbability effect is actually negative: {prob_negative:.1%}")

# Minimum detectable effect
print(f"\nEffect would need to be > {1.96*model.bse['did']:.2f} points to be significant at p<0.05")

# Compare to typical variation
branch_std = df.groupby('branch_id')['service_score'].std().mean()
week_std = df.groupby('week')['service_score'].std().mean()
print(f"\nTypical branch-to-branch variation: {branch_std:.1f} points")
print(f"Typical week-to-week variation: {week_std:.1f} points")
print(f"Estimated treatment effect: {model.params['did']:.2f} points")
