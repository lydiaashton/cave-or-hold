import pandas as pd

# Weekly averages by treatment group
weekly_avg = df.groupby(['treatment', 'week'])['service_score'].mean().unstack(level=0)
weekly_avg.columns = ['Control (No Switch)', 'Treatment (Switched)']
weekly_avg.index.name = 'Week'

# Pre-treatment period (weeks 6-14)
pre_period = weekly_avg.loc[6:14]

print("=== Pre-Treatment Weekly Averages ===")
print(pre_period.round(2))

print("\n=== Pre-Treatment Trend Test ===")
# Test if trends were parallel before treatment
pre_df = df[df['week'] <= 14].copy()
pre_df['week_centered'] = pre_df['week'] - 10  # Center around middle of pre-period
pre_df['treatment_trend'] = pre_df['treatment'] * pre_df['week_centered']

# Regression to test parallel trends
import statsmodels.formula.api as smf
trend_model = smf.ols('service_score ~ treatment + week_centered + treatment_trend', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print(trend_model.summary().tables[1])
print(f"\nTreatment trend interaction coefficient: {trend_model.params['treatment_trend']:.3f}, p-value: {trend_model.pvalues['treatment_trend']:.3f}")

treatment = weekly_avg['Treatment (Switched)'].values

plt.plot(weeks, treatment, 's-', label='Treatment (Switched)', linewidth=2, markersize=8, color='#A23B72')

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create treatment group indicator
branch_switched = df.groupby('branch_id')['switched_branch'].first().to_dict()
df['treatment'] = df['branch_id'].map(branch_switched)

# Event study - create relative time indicators
# Week 14 is the last pre-treatment week, so we use it as reference
df['relative_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create event study dummies (excluding reference period week 14)
event_study_vars = []
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != 0:  # Skip reference week
        var_name = f'rel_w{w}'
        df[var_name] = ((df['relative_week'] == w) & (df['treatment'] == 1)).astype(int)
        event_study_vars.append(var_name)

# Run event study regression
event_study_formula = 'service_score ~ treatment + ' + ' + '.join(event_study_vars)
event_model = smf.ols(event_study_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Coefficients (Treatment x Relative Week) ===")
coeffs = []
for var in event_study_vars:
    coef = event_model.params[var]
    pval = event_model.pvalues[var]
    ci_low = event_model.conf_int().loc[var, 0]
    ci_high = event_model.conf_int().loc[var, 1]
    week = int(var.replace('rel_w', '')) + 14
    coeffs.append({'week': week, 'relative_week': int(var.replace('rel_w', '')), 'coef': coef, 'pval': pval, 'ci_low': ci_low, 'ci_high': ci_high})

coeffs_df = pd.DataFrame(coeffs)
print(coeffs_df.sort_values('relative_week').to_string(index=False))

# Plot event study
plt.figure(figsize=(12, 6))
pre = coeffs_df[coeffs_df['relative_week'] < 0]
post = coeffs_df[coeffs_df['relative_week'] >= 0]

plt.errorbar(pre['relative_week'], pre['coef'], 
             yerr=[pre['coef'] - pre['ci_low'], pre['ci_high'] - pre['coef']],
             fmt='o', capsize=5, label='Pre-Treatment', color='#2E86AB', markersize=8, linewidth=2)
plt.errorbar(post['relative_week'], post['coef'], 
             yerr=[post['coef'] - post['ci_low'], post['ci_high'] - post['coef']],
             fmt='s', capsize=5, label='Post-Treatment', color='#A23B72', markersize=8, linewidth=2)

plt.axhline(y=0, color='gray', linestyle='-', linewidth=1)
plt.axvline(x=-0.5, color='gray', linestyle='--', linewidth=2, label='Switch Point')
plt.xlabel('Relative Week (Week 14 = Reference)', fontsize=12)
plt.ylabel('Coefficient (Difference from Control)', fontsize=12)
plt.title('Event Study: Treatment Effect Over Time', fontsize=14, fontweight='bold')
plt.legend(loc='upper left', fontsize=10)
plt.grid(True, alpha=0.3)
plt.xticks(range(-8, 9))
plt.tight_layout()

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create treatment group indicator
branch_switched = df.groupby('branch_id')['switched_branch'].first().to_dict()
df['treatment'] = df['branch_id'].map(branch_switched)

# Event study - create relative time indicators
# Week 14 is the last pre-treatment week, so we use it as reference
df['relative_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create event study dummies with safe names (avoiding negative signs in variable names)
event_study_vars = []
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != 0:  # Skip reference week
        var_name = f'rel_week_{w:+d}'  # e.g., rel_week_-8, rel_week_+1
        df[var_name] = ((df['relative_week'] == w) & (df['treatment'] == 1)).astype(int)
        event_study_vars.append(var_name)

# Run event study regression
event_study_formula = 'service_score ~ treatment + ' + ' + '.join(event_study_vars)
event_model = smf.ols(event_study_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Coefficients (Treatment x Relative Week) ===")
coeffs = []
for var in event_study_vars:
    coef = event_model.params[var]
    pval = event_model.pvalues[var]
    ci_low = event_model.conf_int().loc[var, 0]
    ci_high = event_model.conf_int().loc[var, 1]
    # Extract the week number from variable name
    week_num = int(var.split('_')[-1])
    coeffs.append({'week': week_num + 14, 'relative_week': week_num, 'coef': coef, 'pval': pval, 'ci_low': ci_low, 'ci_high': ci_high})

coeffs_df = pd.DataFrame(coeffs)
print(coeffs_df.sort_values('relative_week').to_string(index=False))

# Plot event study
plt.figure(figsize=(12, 6))
pre = coeffs_df[coeffs_df['relative_week'] < 0]
post = coeffs_df[coeffs_df['relative_week'] >= 0]

plt.errorbar(pre['relative_week'], pre['coef'], 
             yerr=[pre['coef'] - pre['ci_low'], pre['ci_high'] - pre['coef']],
             fmt='o', capsize=5, label='Pre-Treatment', color='#2E86AB', markersize=8, linewidth=2)
plt.errorbar(post['relative_week'], post['coef'], 
             yerr=[post['coef'] - post['ci_low'], post['ci_high'] - post['coef']],
             fmt='s', capsize=5, label='Post-Treatment', color='#A23B72', markersize=8, linewidth=2)

plt.axhline(y=0, color='gray', linestyle='-', linewidth=1)
plt.axvline(x=-0.5, color='gray', linestyle='--', linewidth=2, label='Switch Point')
plt.xlabel('Relative Week (Week 14 = Reference)', fontsize=12)
plt.ylabel('Coefficient (Difference from Control)', fontsize=12)
plt.title('Event Study: Treatment Effect Over Time', fontsize=14, fontweight='bold')
plt.legend(loc='upper left', fontsize=10)
plt.grid(True, alpha=0.3)
plt.xticks(range(-8, 9))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")

# Check pre-treatment pattern
print("\n=== Pre-Treatment Pattern ===")
print(f"Pre-treatment coefficients range: {pre['coef'].min():.2f} to {pre['coef'].max():.2f}")
print(f"Pre-treatment coefficients mean: {pre['coef'].mean():.2f}")
print(f"Post-treatment coefficients range: {post['coef'].min():.2f} to {post['coef'].max():.2f}")
print(f"Post-treatment coefficients mean: {post['coef'].mean():.2f}")

import pandas as pd

df = pd.read_csv('branches.csv')

# Create treatment group indicator
branch_switched = df.groupby('branch_id')['switched_branch'].first().to_dict()
df['treatment'] = df['branch_id'].map(branch_switched)

# Event study - create relative time indicators
df['relative_week'] = df['week'] - 14  # Week 14 = 0

# Create event study dummies with simple names (prefix + number)
event_study_vars = []
for w in range(-8, 9):
    if w != 0:
        var_name = f'rw{abs(w)}_{"n" if w < 0 else "p"}'  # e.g., rw8_n for -8, rw1_p for +1
        df[var_name] = ((df['relative_week'] == w) & (df['treatment'] == 1)).astype(int)
        event_study_vars.append(var_name)

# Build design matrix manually
X = pd.DataFrame()
X['treatment'] = df['treatment']
for var in event_study_vars:
    X[var] = df[var]
X = sm.add_constant(X)
y = df['service_score']

# Fit model with cluster-robust SE
model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Coefficients ===")
coeffs = []
for var in event_study_vars:
    coef = model.params[var]
    pval = model.pvalues[var]
    ci_low, ci_high = model.conf_int().loc[var]
    # Decode week from variable name
    week_num = int(var.split('_')[0].replace('rw', ''))
    if 'n' in var:
        week_num = -week_num
    coeffs.append({'week': week_num + 14, 'relative_week': week_num, 'coef': coef, 'pval': pval, 'ci_low': ci_low, 'ci_high': ci_high})

coeffs_df = pd.DataFrame(coeffs)
print(coeffs_df.sort_values('relative_week').to_string(index=False))
