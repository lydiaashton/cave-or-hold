import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Calculate mean service scores by group and period
print("=== Mean Service Scores by Group and Period ===")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
pivot.columns = ['Before (weeks 6-14)', 'After (weeks 15-22)']
pivot.index = ['Control (no switch)', 'Treatment (switched)']
print(pivot)

# Calculate differences
print("\n=== Difference-in-Differences Calculation ===")
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

control_diff = control_after - control_before
treatment_diff = treatment_after - treatment_before
did_effect = treatment_diff - control_diff

print(f"Control group (no switch):")
print(f"  Before: {control_before:.2f}")
print(f"  After: {control_after:.2f}")
print(f"  Change: {control_diff:.2f}")
print(f"\nTreatment group (switched):")
print(f"  Before: {treatment_before:.2f}")
print(f"  After: {treatment_after:.2f}")
print(f"  Change: {treatment_diff:.2f}")
print(f"\nDifference-in-Differences Effect: {did_effect:.2f}")

# Run DiD regression
print("\n=== DiD Regression (Simple) ===")
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

model = ols('service_score ~ treatment + post + did', data=df).fit()
print(model.summary())

# Run DiD regression with branch fixed effects
print("\n=== DiD Regression with Branch Fixed Effects ===")
df['branch_id'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ treatment + post + did + C(branch_id)', data=df).fit()
print(model_fe.summary())

# Run DiD regression with controls
print("\n=== DiD Regression with Branch FE and Controls ===")
model_controls = ols('service_score ~ treatment + post + did + C(branch_id) + staffing_index + visits_index', data=df).fit()
print(model_controls.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.stats.weightstats import DescrStatsW

df = pd.read_csv('branches.csv')

# Check parallel trends assumption - look at pre-treatment trends
print("=== Parallel Trends Check: Pre-Treatment Service Scores by Week ===")
pre_data = df[df['week'] < 15].copy()
pre_pivot = pre_data.groupby(['switched_branch', 'week'])['service_score'].mean().unstack()
print(pre_pivot)

# Calculate week-by-week differences before treatment
print("\n=== Pre-Treatment Weekly Differences (Treatment - Control) ===")
pre_means = pre_data.groupby(['switched_branch', 'week'])['service_score'].mean().unstack()
diffs = pre_means.loc[1] - pre_means.loc[0]
print(diffs)
print(f"\nAverage pre-treatment difference: {diffs.mean():.2f}")
print(f"Std dev of pre-treatment differences: {diffs.std():.2f}")

# Run DiD with clustered standard errors at branch level
print("\n=== DiD Regression with Clustered SE (Branch Level) ===")
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Use simple DiD first
model = sm.OLS(df['service_score'], sm.add_constant(df[['treatment', 'post', 'did']])).fit()
print(model.summary())

# Now with controls
print("\n=== DiD with Controls and Clustered SE ===")
X = df[['treatment', 'post', 'did', 'staffing_index', 'visits_index']]
X = sm.add_constant(X)
model_ctrl = sm.OLS(df['service_score'], X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_ctrl.summary())

# Check effect size relative to baseline
baseline = df[df['after_switch'] == 0]['service_score'].mean()
effect = model_ctrl.params['did']
print(f"\n=== Effect Size ===")
print(f"Baseline service score (pre-period): {baseline:.2f}")
print(f"Estimated treatment effect: {effect:.2f} points")
print(f"Effect as % of baseline: {effect/baseline*100:.1f}%")

# Confidence interval
ci = model_ctrl.conf_int().loc['did']
print(f"95% CI: [{ci[0]:.2f}, {ci[1]:.2f}]")
