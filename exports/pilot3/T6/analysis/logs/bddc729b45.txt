import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from statsmodels.stats.multicomp import pairwise_tukeyhsd

df = pd.read_csv('branches.csv')

# Run DiD with cluster-robust standard errors
# First, the basic model
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()

# Get cluster-robust standard errors
from statsmodels.stats.sandwich_covariance import cov_cluster
cluster_robust_se = np.sqrt(np.diag(cov_cluster(model, df['branch_id'])))

print("DiD Results with Cluster-Robust Standard Errors (clustered by branch):")
print("="*70)
print(f"{'Variable':<35} {'Coefficient':>12} {'Std Error':>12} {'t-stat':>10} {'P-value':>10}")
print("="*70)

params = model.params
for i, (name, coef) in enumerate(params.items()):
    se = cluster_robust_se[i]
    t_stat = coef / se
    # Approximate p-value using normal distribution
    from scipy import stats
    p_val = 2 * (1 - stats.norm.cdf(abs(t_stat)))
    print(f"{name:<35} {coef:>12.4f} {se:>12.4f} {t_stat:>10.3f} {p_val:>10.4f}")

print("="*70)
print(f"\nKey finding: The interaction term (switched_branch:after_switch) represents")
print(f"the DiD estimate: +{params['switched_branch:after_switch']:.2f} points")
print(f"with cluster-robust p-value: {2 * (1 - stats.norm.cdf(abs(params['switched_branch:after_switch'] / cluster_robust_se[3]))):.4f}")

# Also calculate confidence interval
from scipy import stats
ci_lower = params['switched_branch:after_switch'] - 1.96 * cluster_robust_se[3]
ci_upper = params['switched_branch:after_switch'] + 1.96 * cluster_robust_se[3]
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
