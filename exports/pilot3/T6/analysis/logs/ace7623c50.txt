import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# 1. Calculate mean service scores by group and period
print("=" * 60)
print("MEAN SERVICE SCORES BY GROUP AND PERIOD")
print("=" * 60)

# Before period (after_switch = 0)
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()

# After period (after_switch = 1)
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nControl Group (didn't switch):")
print(f"  Before (weeks 6-14): {before_control:.2f}")
print(f"  After (weeks 15-22): {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")

print(f"\nTreatment Group (switched):")
print(f"  Before (weeks 6-14): {before_treatment:.2f}")
print(f"  After (weeks 15-22): {after_treatment:.2f}")
print(f"  Change: {after_treatment - before_treatment:.2f}")

# DiD estimate
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

# 2. Run DiD regression
print("\n" + "=" * 60)
print("DIFFERENCE-IN-DIFFERENCES REGRESSION")
print("=" * 60)

# Basic DiD model
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Cluster-robust standard errors by branch
print("\n" + "=" * 60)
print("CLUSTER-ROBUST STANDARD ERRORS (by branch)")
print("=" * 60)

model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model3.summary().tables[1])

print("=" * 70)
print("PLACEBO TEST: Fake switch at week 12")
print("=" * 70)

model_placebo = ols('service_score ~ switched_branch + after_switch_placebo + treatment_placebo + staffing_index + visits_index', 
                    data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print(f"\nPlacebo treatment effect: {model_placebo.params['treatment_placebo']:.2f}")
print(f"Standard error: {model_placebo.bse['treatment_placebo']:.2f}")

model_het = ols('service_score ~ switched_branch + post_early + post_late + treat_early + treat_late + staffing_index + visits_index',

# Check if they're different
print(f"\nDifference (late - early): {model_het.params['treat_late'] - model_het.params['treat_early']:.2f}")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Key concern: Parallel trends assumption
print("=" * 70)
print("CRITICAL ASSESSMENT: PARALLEL TRENDS")
print("=" * 70)

# Weekly gaps between treatment and control
weekly_means = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_gap = weekly_means[1] - weekly_means[0]

print("\nWeekly gap (Treatment - Control):")
for week, gap in weekly_gap.items():
    period = "PRE" if week < 15 else "POST"
    print(f"  Week {week}: {gap:.2f} [{period}]")

# Pre-trend slope
pre_gaps = weekly_gap[weekly_gap.index < 15]
post_gaps = weekly_gap[weekly_gap.index >= 15]

import statsmodels.api as sm
pre_X = sm.add_constant(pre_gaps.index.values)
pre_model = sm.OLS(pre_gaps.values, pre_X).fit()

post_X = sm.add_constant(post_gaps.index.values)
post_model = sm.OLS(post_gaps.values, post_X).fit()

print(f"\nPre-treatment trend in gap: {pre_model.params[1]:.3f} points/week (p={pre_model.pvalues[1]:.4f})")
print(f"Post-treatment trend in gap: {post_model.params[1]:.3f} points/week (p={post_model.pvalues[1]:.4f})")

# The gap was ALREADY widening before the switch
print(f"\n⚠️  The gap was widening by {pre_model.params[1]:.2f} points/week BEFORE the switch")
print(f"   This suggests treatment branches were already improving faster")

# Placebo test results
print("\n" + "=" * 70)
print("PLACEBO TEST RESULTS")
print("=" * 70)

df['after_switch_placebo'] = (df['week'] >= 12).astype(int)
df['treatment_placebo'] = df['switched_branch'] * df['after_switch_placebo']

model_placebo = ols('service_score ~ switched_branch + after_switch_placebo + treatment_placebo + staffing_index + visits_index', 
                    data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print(f"\nPlacebo effect (fake switch at week 12): {model_placebo.params['treatment_placebo']:.2f} (p={model_placebo.pvalues['treatment_placebo']:.4f})")
print(f"Actual effect (real switch at week 15): {df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()}")

# Calculate actual DiD again
model_actual = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
                   data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"\nActual DiD estimate: {model_actual.params['treatment']:.2f} (p={model_actual.pvalues['treatment']:.4f})")

# The problem: placebo effect ≈ actual effect
print(f"\n⚠️  Placebo effect ({model_placebo.params['treatment_placebo']:.2f}) ≈ Actual effect ({model_actual.params['treatment']:.2f})")
print(f"   This suggests the 'effect' may reflect pre-existing trends, not the procedure change")

# What would happen if we account for pre-trends?
print("\n" + "=" * 70)
print("ADJUSTED ESTIMATE ACCOUNTING FOR PRE-TRENDS")
print("=" * 70)

# If pre-trend was 0.26 points/week, over 8 weeks that's ~2.1 points of the gap change
pre_trend_adjustment = pre_model.params[1] * 8  # 8 post-treatment weeks
adjusted_effect = model_actual.params['treatment'] - pre_trend_adjustment

print(f"\nPre-treatment trend: {pre_model.params[1]:.3f} points/week")
print(f"Over 8 post-treatment weeks: {pre_trend_adjustment:.2f} points")
print(f"Raw DiD estimate: {model_actual.params['treatment']:.2f} points")
print(f"Adjusted estimate (subtracting pre-trend): {adjusted_effect:.2f} points")

if adjusted_effect < 0:
    print(f"\n⚠️  After accounting for pre-trends, the effect could be ZERO or NEGATIVE")
else:
    print(f"\n⚠️  After accounting for pre-trends, the effect is much smaller: {adjusted_effect:.2f} points")
