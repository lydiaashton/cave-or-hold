import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Basic DiD regression
model_basic = ols('service_score ~ treatment + post + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Basic DiD Regression (with clustered SEs):")
print(model_basic.summary())

# DiD with branch fixed effects
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ treatment + post + did + C(branch_id_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with Branch Fixed Effects:")
# Just show the key coefficients
print(f"DID coefficient: {model_fe.params['did']:.4f}")
print(f"DID std err: {model_fe.bse['did']:.4f}")
print(f"DID p-value: {model_fe.pvalues['did']:.4f}")
print(f"DID 95% CI: [{model_fe.conf_int().loc['did'][0]:.4f}, {model_fe.conf_int().loc['did'][1]:.4f}]")

# DiD with branch and week fixed effects
model_full = ols('service_score ~ treatment + did + C(branch_id_str) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with Branch and Week Fixed Effects:")
print(f"DID coefficient: {model_full.params['did']:.4f}")
print(f"DID std err: {model_full.bse['did']:.4f}")
print(f"DID p-value: {model_full.pvalues['did']:.4f}")
print(f"DID 95% CI: [{model_full.conf_int().loc['did'][0]:.4f}, {model_full.conf_int().loc['did'][1]:.4f}]")

# Check pre-trends with event study approach
print("\n\nEvent Study - Pre-treatment coefficients:")
# Create week dummies interacted with treatment
df['week_6'] = (df['week'] == 6).astype(int)
# Use week 14 as reference (last pre-treatment week)
pre_weeks = [6, 7, 8, 9, 10, 11, 12, 13]
post_weeks = [15, 16, 17, 18, 19, 20, 21, 22]

# Create interaction terms
for w in pre_weeks + post_weeks:
    df[f'week_{w}'] = (df['week'] == w).astype(int)
    df[f'did_week_{w}'] = df['treatment'] * df[f'week_{w}']

# Event study regression (week 14 as reference)
event_study_vars = ' + '.join([f'did_week_{w}' for w in pre_weeks + post_weeks])
model_event = ols(f'service_score ~ {event_study_vars} + C(branch_id_str) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nPre-treatment interaction coefficients (should be ~0 if parallel trends hold):")
for w in pre_weeks:
    coef = model_event.params[f'did_week_{w}']
    se = model_event.bse[f'did_week_{w}']
    pval = model_event.pvalues[f'did_week_{w}']
    print(f"Week {w}: coef={coef:.3f}, se={se:.3f}, p={pval:.3f}")

print("\nPost-treatment interaction coefficients:")
for w in post_weeks:
    coef = model_event.params[f'did_week_{w}']
    se = model_event.bse[f'did_week_{w}']
    pval = model_event.pvalues[f'did_week_{w}']
    print(f"Week {w}: coef={coef:.3f}, se={se:.3f}, p={pval:.3f}")
