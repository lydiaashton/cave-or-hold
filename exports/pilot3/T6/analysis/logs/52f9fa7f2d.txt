import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with robust standard errors
# Model 1: Simple DiD
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='HC1')
print("=== Model 1: Simple DiD Regression ===")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='HC1')
print("\n=== Model 2: DiD with Controls ===")
print(model2.summary().tables[1])

# Model 3: DiD with branch fixed effects
# Create dummy variables for branches
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df_with_dummies['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Build formula with all branch dummies
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_')]
formula = 'service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index + ' + ' + '.join(branch_cols)
model3 = ols(formula, data=df_with_dummies).fit(cov_type='cluster', cov_kwds={'groups': df_with_dummies['branch_id']})
print("\n=== Model 3: DiD with Branch Fixed Effects (clustered SEs) ===")
print(f"Treatment coefficient: {model3.params['treatment']:.4f}")
print(f"Treatment SE: {model3.bse['treatment']:.4f}")
print(f"Treatment p-value: {model3.pvalues['treatment']:.4f}")
print(f"95% CI: [{model3.conf_int().loc['treatment'][0]:.4f}, {model3.conf_int().loc['treatment'][1]:.4f}]")

# Pre-trends visualization
print("\n=== Pre-Trends Analysis ===")
pre_period = df[df['week'] < 15].copy()
pre_trends = pre_period.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Calculate the gap between treatment and control in pre-period
control_pre = pre_trends[pre_trends['switched_branch'] == 0].set_index('week')['service_score']
treat_pre = pre_trends[pre_trends['switched_branch'] == 1].set_index('week')['service_score']
gap_pre = treat_pre - control_pre

print("Treatment-Control gap in pre-period (weeks 6-14):")
for week in range(6, 15):
    if week in gap_pre.index:
        print(f"Week {week}: {gap_pre[week]:.2f}")

print(f"\nAverage pre-period gap: {gap_pre.mean():.2f}")
print(f"Std of pre-period gap: {gap_pre.std():.2f}")

    # Sample branches with replacement
    branches = df['branch_id'].unique()
    sampled_branches = np.random.choice(branches, size=len(branches), replace=True)
    df_sample = df[df['branch_id'].isin(sampled_branches)]

# Final model with branch fixed effects and clustered SEs

df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_')]
formula = 'service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index + ' + ' + '.join(branch_cols)
model_fe = ols(formula, data=df_with_dummies).fit(cov_type='cluster', cov_kwds={'groups': df_with_dummies['branch_id']})

print("=== Final Model (Branch Fixed Effects, Clustered SEs) ===")

print(f"Standard error: {model_fe.bse['treatment']:.2f}")
print(f"p-value: {model_fe.pvalues['treatment']:.3f}")

print(f"\n=== Practical Significance ===")

import pandas as pd

import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Key question: Did treatment branches actually improve, or did control branches just get worse?
print("=== Critical Question: What Drove the Effect? ===")
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print(f"Treatment change: {treat_after - treat_before:.2f} points")
print(f"Control change: {control_after - control_before:.2f} points")
print(f"\nThe DiD effect ({(treat_after - treat_before) - (control_after - control_before):.2f}) is driven by:")
print(f"  - Treatment branches: {'improved' if treat_after > treat_before else 'declined' if treat_after < treat_before else 'stayed flat'}")
print(f"  - Control branches: {'improved' if control_after > control_before else 'declined' if control_after < control_before else 'stayed flat'}")

# Check if something unusual happened to control group at week 15
print("\n=== Control Group Week-by-Week ===")
control_weekly = df[df['switched_branch']==0].groupby('week')['service_score'].mean()
for week in range(6, 23):
    if week in control_weekly.index:
        change = control_weekly[week] - control_weekly.get(week-1, control_weekly[week]) if week > 6 else 0
        print(f"Week {week}: {control_weekly[week]:.1f} (change: {change:+.1f})")

# Event study visualization
print("\n=== Event Study - Treatment vs Control Weekly Gaps ===")
weekly_gap = df.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
