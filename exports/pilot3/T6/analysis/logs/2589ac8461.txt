df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# DiD with cluster-robust standard errors (clustered by branch)
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD Regression with Cluster-Robust SE (by branch) ===")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create week dummies interacted with treatment
# This shows the effect in each week relative to the pre-period baseline
df['week_14'] = (df['week'] == 14).astype(int)  # reference week

# Create week dummies for each week
for w in range(6, 23):
    if w != 14:
        df[f'week_{w}'] = (df['week'] == w).astype(int)

# Create interactions with treatment
for w in range(6, 23):
    if w != 14:
        df[f'week_{w}_treat'] = df[f'week_{w}'] * df['switched_branch']

# Run event study regression (omitting week 14 as reference)
week_vars = [f'week_{w}' for w in range(6, 23) if w != 14]
interact_vars = [f'week_{w}_treat' for w in range(6, 23) if w != 14]

formula = 'service_score ~ switched_branch + ' + ' + '.join(week_vars) + ' + ' + ' + '.join(interact_vars)
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract the interaction coefficients (these are the event study estimates)
event_results = []
for w in range(6, 23):
    if w != 14:
        coef_name = f'week_{w}_treat'
        coef = model_event.params[coef_name]
        se = model_event.bse[coef_name]
        pvalue = model_event.pvalues[coef_name]
        event_results.append({'week': w, 'coef': coef, 'se': se, 'pvalue': pvalue, 'after': w >= 15})

event_df = pd.DataFrame(event_results)
print("=== Event Study Results (Treatment x Week Interactions) ===")
print(event_df.to_string(index=False))

# Test if pre-period coefficients are jointly zero (test parallel trends)
pre_weeks = [f'week_{w}_treat' for w in range(6, 15)]
print(f"\n=== Joint Test: Pre-period interactions = 0 ===")
from statsmodels.stats.anova import anova_lm
# Use Wald test
R = np.zeros((len(pre_weeks), len(model_event.params)))
for i, param in enumerate(pre_weeks):
    if param in model_event.params.index:
        R[i, model_event.params.index.get_loc(param)] = 1

from statsmodels.stats.contrast import ContrastResults
wald_test = model_event.wald_test(R)
print(f"Wald test statistic: {wald_test.statistic[0,0]:.2f}")
print(f"Wald test p-value: {wald_test.pvalue:.4f}")

# Summary statistics for the board
print("\n=== Summary for Board ===")
print(f"Total branches: {df['branch_id'].nunique()}")
print(f"Branches that switched: {df[df['switched_branch']==1]['branch_id'].nunique()}")
print(f"Branches that didn't switch: {df[df['switched_branch']==0]['branch_id'].nunique()}")
print(f"Weeks of data: {df['week'].min()} to {df['week'].max()}")
print(f"Switch occurred at week: 15")
print(f"\nEstimated effect of new procedure: +{model_event.params['switched_branch'] + event_df[event_df['after']==True]['coef'].mean():.2f} points on average")
print(f"95% CI for main effect: approximately +0.78 to +3.03 points")

import pandas as pd

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Re-run the main DiD model to get clean estimates
df['treatment_post'] = df['switched_branch'] * df['after_switch']
model = ols('service_score ~ switched_branch + after_switch + treatment_post + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get the key estimates
did_coef = model.params['treatment_post']
did_se = model.bse['treatment_post']
did_pvalue = model.pvalues['treatment_post']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se

print("=== FINAL ANALYSIS SUMMARY ===\n")
print(f"Method: Difference-in-Differences with cluster-robust standard errors")

print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {did_pvalue:.4f}")
print(f"Statistically significant: {'Yes' if did_pvalue < 0.05 else 'No'}\n")
