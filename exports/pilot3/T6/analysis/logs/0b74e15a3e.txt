from statsmodels.formula.api import ols
import statsmodels.stats.api as sms

df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Run DiD regression with cluster-robust standard errors
# Model: service_score = β0 + β1*treated + β2*post + β3*DiD + ε
# β3 is our DiD estimate

model = ols('service_score ~ treated + post + DiD', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== DiD Regression Results (Clustered SEs at branch level) ===")
print(model.summary())

# Get the DiD coefficient specifically
did_coef = model.params['DiD']
did_se = model.bse['DiD']
did_pvalue = model.pvalues['DiD']
did_ci_low = did_coef - 1.96 * did_se

print(f"Coefficient: {did_coef:.3f}")
print(f"Std Error: {did_se:.3f}")
print(f"P-value: {did_pvalue:.4f}")
print(f"95% CI: [{did_ci_low:.3f}, {did_ci_high:.3f}]")

# Add shaded region for post-switch period
ax.axvspan(14.5, 22.5, alpha=0.2, color='red', label='Post-switch period')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Mean Service Score', fontsize=12)
ax.set_title('Service Score Trends: Control vs Treated Branches', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=10)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("\nPlot saved as 'service_score_trends.png'")

# Check parallel trends statistically (pre-treatment period only)
pre_df = df[df['week'] < 15].copy()

import pandas as pd
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Calculate baseline statistics
baseline_mean = df[df['week'] < 15]['service_score'].mean()
baseline_std = df[df['week'] < 15]['service_score'].std()

print("=== Effect Size Context ===")
print(f"Baseline mean service score (pre-switch): {baseline_mean:.2f}")
print(f"Baseline std dev: {baseline_std:.2f}")
print(f"DiD estimate: 2.09")
print(f"Effect as % of baseline: {2.09/baseline_mean*100:.1f}%")
print(f"Effect in std dev units: {2.09/baseline_std:.2f} SD")

# Run regression with covariates (staffing_index, visits_index)
print("\n=== DiD Regression with Covariates ===")
model_with_covariates = ols('service_score ~ treated + post + DiD + staffing_index + visits_index', 
                            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_with_covariates.summary())

# Check if results are robust
did_coef_cov = model_with_covariates.params['DiD']
did_pvalue_cov = model_with_covariates.pvalues['DiD']
print(f"\nDiD with covariates: {did_coef_cov:.3f} (p={did_pvalue_cov:.4f})")

# Branch-level analysis (average each branch's pre and post, then do DiD)
print("\n=== Branch-Level Aggregated Analysis ===")
branch_agg = df.groupby(['branch_id', 'switched_branch']).agg(
    pre_score=('service_score', lambda x: x[df.loc[x.index, 'week'] < 15].mean() if len(x[df.loc[x.index, 'week'] < 15]) > 0 else None),
    post_score=('service_score', lambda x: x[df.loc[x.index, 'week'] >= 15].mean() if len(x[df.loc[x.index, 'week'] >= 15]) > 0 else None),
).reset_index()

branch_agg['score_change'] = branch_agg['post_score'] - branch_agg['pre_score']

# Compare changes between treated and control
control_change = branch_agg[branch_agg['switched_branch'] == 0]['score_change'].mean()
treated_change = branch_agg[branch_agg['switched_branch'] == 1]['score_change'].mean()
branch_did = treated_change - control_change

print(f"Control branch avg change: {control_change:.3f}")
print(f"Treated branch avg change: {treated_change:.3f}")
print(f"Branch-level DiD: {branch_did:.3f}")

# T-test on branch-level changes
from scipy import stats
control_changes = branch_agg[branch_agg['switched_branch'] == 0]['score_change']
treated_changes = branch_agg[branch_agg['switched_branch'] == 1]['score_change']

t_stat, t_pvalue = stats.ttest_ind(treated_changes, control_changes)
print(f"\nT-test on branch-level changes: t={t_stat:.3f}, p={t_pvalue:.4f}")
