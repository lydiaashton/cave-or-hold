import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD groups
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['treatment_post'] = df['treatment'] * df['post']

# Summary table: average service scores by group and period
summary_table = df.groupby(['treatment', 'post'])['service_score'].agg(['mean', 'std', 'count']).unstack()
print("Service Score Summary by Group and Period:")
print(summary_table)
print()

# Calculate the DiD manually
before_control = df[(df['treatment'] == 0) & (df['post'] == 0)]['service_score'].mean()
after_control = df[(df['treatment'] == 0) & (df['post'] == 1)]['service_score'].mean()
before_treatment = df[(df['treatment'] == 1) & (df['post'] == 0)]['service_score'].mean()
after_treatment = df[(df['treatment'] == 1) & (df['post'] == 1)]['service_score'].mean()

print("Before/After Means:")
print(f"Control (not switched): Before={before_control:.2f}, After={after_control:.2f}, Change={after_control - before_control:.2f}")
print(f"Treatment (switched): Before={before_treatment:.2f}, After={after_treatment:.2f}, Change={after_treatment - before_treatment:.2f}")
print()

# Difference-in-Differences estimate
did_estimate = (after_treatment - before_treatment) - (after_control - before_control)
print(f"DiD Estimate (treatment effect): {did_estimate:.2f}")
print()

# Run regression with DiD specification
# Model 1: Basic DiD
model1 = ols('service_score ~ C(treatment) + C(post) + treatment_post', data=df).fit()
print("Model 1: Basic DiD Regression")
print(model1.summary())
print()

# Model 2: DiD with controls (staffing_index, visits_index)
model2 = ols('service_score ~ C(treatment) + C(post) + treatment_post + staffing_index + visits_index', data=df).fit()
print("Model 2: DiD with Controls")
print(model2.summary())
print()

# Model 3: Branch fixed effects
model3 = ols('service_score ~ C(treatment) + C(post) + treatment_post + C(branch_id)', data=df).fit()
print("Model 3: DiD with Branch Fixed Effects")
print(model3.summary())
