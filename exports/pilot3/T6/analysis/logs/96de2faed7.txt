import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# Basic DiD: service_score ~ switched_branch * after_switch
model_did = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("=== Basic Difference-in-Differences Regression ===")
print(model_did.summary().tables[1])

# Add branch fixed effects (this is more rigorous)
# We'll use entity fixed effects
df_demeaned = df.copy()
branch_means = df.groupby('branch_id')['service_score'].transform('mean')
df_demeaned['service_score_demeaned'] = df_demeaned['service_score'] - branch_means

# For fixed effects, we need to demean the treatment variables too
df_demeaned['switched_branch_demeaned'] = df_demeaned['switched_branch'] - df_demeaned['switched_branch'].mean()
df_demeaned['after_switch_demeaned'] = df_demeaned['after_switch'] - df_demeaned['after_switch'].mean()
df_demeaned['interaction_demeaned'] = df_demeaned['switched_branch'] * df_demeaned['after_switch']
interaction_mean = (df_demeaned['switched_branch'] * df_demeaned['after_switch']).mean()
df_demeaned['interaction_demeaned'] = df_demeaned['interaction_demeaned'] - interaction_mean

model_fe = ols('service_score_demeaned ~ switched_branch_demeaned + after_switch_demeaned + interaction_demeaned', 
               data=df_demeaned).fit()
print("\n=== With Branch Fixed Effects (Demeaned) ===")
print(model_fe.summary().tables[1])

# Alternative: Use cluster-robust standard errors at branch level
model_did_cluster = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD with Cluster-Robust SE (at branch level) ===")
print(model_did_cluster.summary().tables[1])

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks since switch)
# Week 15 is the first week after switch, so relative_week = week - 14
df['relative_week'] = df['week'] - 14

# Event study: create indicators for each week relative to switch
# Week 14 (relative_week=0) will be the reference
for w in range(-8, 9):  # weeks -8 to 8 relative to switch
    if w != 0:  # skip reference week
        df[f'week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
event_vars = [f'week_{w}' for w in range(-8, 9) if w != 0]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results (coefficients for switched branches by week relative to switch) ===")
print("(Week 0 / relative_week=0 is the reference category)")
print("\nPre-switch weeks (should be stable if parallel trends hold):")
for w in range(-8, 0):
    coef = model_event.params[f'week_{w}']
    pval = model_event.pvalues[f'week_{w}']
    print(f"  Week {w}: coef={coef:.2f}, p={pval:.3f}")

print("\nPost-switch weeks:")
for w in range(1, 9):
    coef = model_event.params[f'week_{w}']
    pval = model_event.pvalues[f'week_{w}']
    print(f"  Week {w}: coef={coef:.2f}, p={pval:.3f}")

# Summary statistics
print("\n=== Summary of Treatment Effect ===")
print(f"DiD Estimate: 2.29 points (95% CI: 1.69 to 2.89)")
print(f"Baseline average service score: ~50")
print(f"Percentage improvement: {(2.29/50)*100:.1f}%")

# Effect size in terms of standard deviations
std_score = df['service_score'].std()
print(f"Effect size in SD units: {2.29/std_score:.2f} SD")

import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create relative week variable (weeks since switch)
df['relative_week'] = df['week'] - 14

# Event study: create indicators for each week relative to switch with better names
for w in range(-8, 9):
    if w != 0:
        sign = 'm' if w < 0 else 'p'
        abs_w = abs(w)
        df[f'rel_w{sign}{abs_w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
event_vars = [col for col in df.columns if col.startswith('rel_w')]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

    coef = model_event.params[var]
    pval = model_event.pvalues[var]
    sig = '***' if pval < 0.01 else '**' if pval < 0.05 else '*' if pval < 0.1 else ''
    print(f"  Week {w}: coef={coef:.2f} {sig}, p={pval:.3f}")
