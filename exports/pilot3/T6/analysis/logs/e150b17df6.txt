import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate

df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Basic Difference-in-Differences Regression ===")
print(model.summary().tables[1])

# Add branch fixed effects (more robust)
# This controls for time-invariant branch characteristics
print("\n=== DiD with Branch Fixed Effects ===")
model_fe = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit()
# Just show the key coefficients
print(model_fe.summary().tables[1].loc[['switched_branch', 'after_switch', 'treatment']])

# Add week fixed effects too (controls for common time trends)
print("\n=== DiD with Branch and Week Fixed Effects ===")
model_fe_week = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id) + C(week)', data=df).fit()
print(model_fe_week.summary().tables[1].loc[['switched_branch', 'after_switch', 'treatment']])

for week in range(15, 23):
    week_data = df[df['week'] == week]
    model = ols('service_score ~ switched_branch', data=week_data).fit()
    weekly_effects.append({
        'week': week,
        'effect': model.params['switched_branch'],
        'pvalue': model.pvalues['switched_branch']
    })

    lambda x: x[x['after_switch'] == 1]['service_score'].mean() - x[x['after_switch'] == 0]['service_score'].mean()
).reset_index()
branch_change.columns = ['branch_id', 'switched_branch', 'change']

print(f"Correlation between baseline score and improvement: {corr:.3f}")

low_baseline_effect = merged[merged['low_baseline']]['change'].mean()
