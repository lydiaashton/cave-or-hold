from statsmodels.formula.api import ols

# Test for pre-trends: interact switched_branch with week in the pre-period only
# If there's a differential trend before the switch, that would violate parallel trends assumption
pre_df = df[df['week'] < 15].copy()
pre_df['week_centered'] = pre_df['week'] - pre_df['week'].mean()

# Test if treatment and control had different trends before the switch
pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_df).fit(cov_type='cluster', cov_kwds={'groups': pre_df['branch_id']})
print("=== Pre-Trend Test (Interaction in Pre-Period Only) ===")
print(pre_model.summary())

# The coefficient on switched_branch:week_centered tests whether the trends were different
# If not significant, parallel trends assumption is more plausible

print("\n=== Effect Size Calculation ===")
# Calculate the DiD estimate and express it relative to the baseline
did_estimate = 2.1876  # from earlier
baseline_avg = df[df['after_switch'] == 0]['service_score'].mean()
print(f"DiD estimate: {did_estimate:.2f} points")
print(f"Baseline average service score: {baseline_avg:.2f}")

# Also calculate as % of standard deviation
baseline_std = df[df['after_switch'] == 0]['service_score'].std()

import matplotlib.pyplot as plt
import pandas as pd

df = pd.read_csv('branches.csv')

# Calculate average service score by week and group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_score']

# Pivot for easier plotting
pivot = weekly_avg.pivot(index='week', columns='switched_branch', values='avg_score')
pivot.columns = ['Control (No Switch)', 'Treatment (Switched)']

# Create the plot
fig, ax = plt.subplots(figsize=(12, 7))

ax.plot(pivot.index, pivot['Control (No Switch)'], 'o-', label='Control Branches (n=14)', 
        linewidth=2.5, markersize=8, color='#2E86AB')
ax.plot(pivot.index, pivot['Treatment (Switched)'], 's-', label='Switched Branches (n=20)', 
        linewidth=2.5, markersize=8, color='#A23B72')

# Add vertical line at switch point
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2.5, label='Procedure Change (Week 15)')

# Add shaded region for post-period
ax.axvspan(14.5, 22.5, alpha=0.15, color='red', label='Post-Switch Period')

ax.set_xlabel('Week of Year', fontsize=13, fontweight='bold')
ax.set_ylabel('Average Service Score', fontsize=13, fontweight='bold')
ax.set_title('Library Service Scores: Impact of New Shelving Procedure', fontsize=15, fontweight='bold')
ax.legend(loc='upper left', fontsize=11, framealpha=0.9)
ax.grid(True, alpha=0.3, linestyle='--')
ax.set_xticks(range(6, 23))
ax.set_ylim(40, 65)

# Add annotations
ax.annotate('New procedure\nimplemented', xy=(15, 62), fontsize=10, color='red', fontweight='bold')

plt.tight_layout()
plt.savefig('service_scores_over_time.png', dpi=150, bbox_inches='tight')
print("Plot saved as 'service_scores_over_time.png'")

# Also create a summary statistics table
print("\n=== Summary Statistics Table ===")
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
summary.columns = ['Mean', 'Std Dev', 'N']
print(summary)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Recalculate the key statistics
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main DiD model with cluster-robust SE
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Key Regression Results ===")
print(f"\nTreatment effect (DiD coefficient): {model.params['treatment']:.3f}")
print(f"Standard error: {model.bse['treatment']:.3f}")
print(f"t-statistic: {model.tvalues['treatment']:.3f}")
print(f"p-value: {model.pvalues['treatment']:.3f}")

# 95% CI
coef = model.params['treatment']
se = model.bse['treatment']
ci_lower = coef - 1.96 * se
ci_upper = coef + 1.96 * se
print(f"\n95% Confidence Interval: [{ci_lower:.3f}, {ci_upper:.3f}]")

# What's the probability the effect is positive? (assuming normal distribution)
from scipy import stats
z_score = coef / se
prob_positive = 1 - stats.norm.cdf(0, loc=coef, scale=se)
print(f"\nProbability effect is positive: {prob_positive*100:.1f}%")

# Also calculate one-sided p-value (testing if effect > 0)
one_sided_p = 1 - stats.norm.cdf(z_score)
print(f"One-sided p-value (testing effect > 0): {one_sided_p:.3f}")

# Baseline for context
baseline = df[df['after_switch'] == 0]['service_score'].mean()
print(f"\nBaseline average service score: {baseline:.2f}")
print(f"Effect as % of baseline: {(coef/baseline)*100:.1f}%")
