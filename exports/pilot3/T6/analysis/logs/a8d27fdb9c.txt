import numpy as np

df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']

# Calculate pre and post averages by treatment status
pre_treatment = df[(df['post'] == 0) & (df['treatment'] == 1)]['service_score'].mean()
pre_control = df[(df['post'] == 0) & (df['treatment'] == 0)]['service_score'].mean()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Difference-in-differences regression
# Basic DiD model
model = ols('service_score ~ treatment * post', data=df).fit()
print("Basic DiD Regression:")
print(model.summary())

# DiD with branch fixed effects
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = df.copy()
for col in branch_dummies.columns:
    df_with_fe[col] = branch_dummies[col]

# Regression with branch fixed effects
fe_cols = ['treatment', 'post', 'treatment:post'] + branch_dummies.columns.tolist()
X = df_with_fe[fe_cols]
X = sm.add_constant(X)
y = df_with_fe['service_score']

model_fe = sm.OLS(y, X).fit()
print("\n\nDiD with Branch Fixed Effects:")
print(model_fe.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# Create treatment and post indicators
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Basic DiD regression
X = sm.add_constant(df[['treatment', 'post', 'did']])
y = df['service_score']
model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Basic DiD Regression (clustered SEs by branch):")
print(model.summary())

# DiD with branch fixed effects
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
X_fe = pd.concat([df[['post', 'did']], branch_dummies], axis=1)
X_fe = sm.add_constant(X_fe)
model_fe = sm.OLS(y, X_fe).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD with Branch Fixed Effects (clustered SEs):")
print(model_fe.summary())

# DiD with branch and week fixed effects
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
X_full = pd.concat([df[['did']], branch_dummies, week_dummies], axis=1)
X_full = sm.add_constant(X_full)
model_full = sm.OLS(y, X_full).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD with Branch and Week Fixed Effects (clustered SEs):")
print(model_full.summary())

df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Get the key coefficient and confidence interval
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)

X = pd.concat([df[['did']], branch_dummies, week_dummies], axis=1)
X = sm.add_constant(X)
y = df['service_score']

model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

did_coef = model.params['did']
did_se = model.bse['did']
did_pvalue = model.pvalues['did']
ci_lower = model.conf_int().loc['did', 0]
ci_upper = model.conf_int().loc['did', 1]

print(f"DiD Coefficient: {did_coef:.2f}")
print(f"Standard Error: {did_se:.2f}")
print(f"P-value: {did_pvalue:.3f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")

# What's the probability the effect is positive?
from scipy import stats
t_stat = did_coef / did_se
# Approximate probability effect > 0
prob_positive = 1 - stats.t.cdf(0, df=len(df) - X.shape[1], loc=t_stat, scale=1)
print(f"\nApproximate probability effect is positive: {prob_positive:.1%}")

# Effect size relative to baseline
baseline = df['service_score'].mean()
print(f"\nBaseline average score: {baseline:.1f}")
print(f"Effect as % of baseline: {did_coef/baseline*100:.1f}%")

# Compare to typical variation
score_std = df['service_score'].std()
print(f"\nTypical score variation (std dev): {score_std:.2f}")
