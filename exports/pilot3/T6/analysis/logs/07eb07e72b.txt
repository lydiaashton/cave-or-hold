branch_changes = branch_changes.join(branch_info)

print(f"Treated After: {treated_after:.4f}")

print(f"\nControl change: {control_change:.4f}")
print(f"Treated change: {treated_change:.4f}")
print(f"DiD estimate: {did:.4f}")

# Event study - create week dummies interacted with treatment
df['post_week'] = df['week'].apply(lambda x: x - 14 if x >= 15 else None)

# Create dummy variables for each week
for w in range(6, 23):
    df[f'week_{w}'] = (df['week'] == w).astype(int)

# Event study regression (week 14 as reference)
event_cols = [f'week_{w}' for w in list(range(6, 15)) + list(range(15, 23))]

formula = 'service_score ~ switched_branch + ' + ' + '.join(event_cols) + ' + ' + ' + '.join([f'switched_branch * week_{w}' for w in list(range(6, 15)) + list(range(15, 23))])

# Calculate weekly DiD estimates
for week in range(15, 23):
    week_data = df[df['week'] == week]
    control_mean = week_data[week_data['switched_branch']==0]['service_score'].mean()
    treated_mean = week_data[week_data['switched_branch']==1]['service_score'].mean()
    
    # Get baseline (week 14) for comparison
    baseline = df[df['week']==14]
    control_base = baseline[baseline['switched_branch']==0]['service_score'].mean()
    treated_base = baseline[baseline['switched_branch']==1]['service_score'].mean()
    
    # DiD for this specific week vs week 14
    control_change = control_mean - control_base
    treated_change = treated_mean - treated_base
    did = treated_change - control_change
    
    print(f"Week {week}: DiD = {did:.2f}")

# Final verification of main result
print("\n" + "=" * 60)
print("FINAL VERIFICATION")
