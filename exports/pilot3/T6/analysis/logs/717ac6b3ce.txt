import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD table: average service scores by group and period
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES SUMMARY TABLE")
print("=" * 60)

# Before period (weeks 6-14)
before = df[df['after_switch'] == 0]
before_switched = before[before['switched_branch'] == 1]['service_score'].mean()
before_control = before[before['switched_branch'] == 0]['service_score'].mean()

# After period (weeks 15-22)
after = df[df['after_switch'] == 1]
after_switched = after[after['switched_branch'] == 1]['service_score'].mean()
after_control = after[after['switched_branch'] == 0]['service_score'].mean()

print(f"\n{'Group':<20} {'Before (W6-14)':<18} {'After (W15-22)':<18} {'Change':<10}")
print("-" * 60)
print(f"{'Switched Branches':<20} {before_switched:<18.2f} {after_switched:<18.2f} {after_switched - before_switched:<10.2f}")
print(f"{'Control Branches':<20} {before_control:<18.2f} {after_control:<18.2f} {after_control - before_control:<10.2f}")
print("-" * 60)
did_effect = (after_switched - before_switched) - (after_control - before_control)
print(f"{'DiD Effect':<20} {'':<18} {'':<18} {did_effect:<10.2f}")

# Run regression with cluster-robust standard errors
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD model
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary())

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary())

# Cluster-robust standard errors by branch
model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 3: DiD with controls (cluster-robust SE by branch)")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Check parallel trends: average service score by week for each group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_score']

# Pivot for easier viewing
pivot = weekly_avg.pivot(index='week', columns='switched_branch', values='avg_score')
pivot.columns = ['Control (No Switch)', 'Treatment (Switched)']
pivot.index.name = 'Week'

print("=" * 60)
print("PARALLEL TRENDS CHECK: Weekly Average Service Scores")
print("=" * 60)
print(pivot.to_string())

# Calculate the difference between groups each week
pivot['Difference'] = pivot['Treatment (Switched)'] - pivot['Control (No Switch)']
print("\n" + "=" * 60)
print("Difference (Treatment - Control) by Week")
print("=" * 60)
print(pivot['Difference'].to_string())

# Event study: estimate effect for each week relative to switch
# Create relative week variable (week 15 = 0, before is negative, after is positive)
df['relative_week'] = df['week'] - 15

# Calculate average treatment effect by relative week
event_study = df.groupby(['relative_week', 'switched_branch'])['service_score'].mean().unstack()
event_study['diff'] = event_study[1] - event_study[0]
event_study.index = event_study.index.map(lambda x: f"Week {x+15}")

print("\n" + "=" * 60)
print("EVENT STUDY: Treatment Effect by Relative Week")
print("=" * 60)
print(event_study[['diff']].to_string())

# Create visualization data
plt.figure(figsize=(12, 6))

# Plot trends
switched = weekly_avg[weekly_avg['switched_branch'] == 1]
control = weekly_avg[weekly_avg['switched_branch'] == 0]

plt.plot(switched['week'], switched['avg_score'], 'b-o', label='Switched Branches', linewidth=2, markersize=6)

plt.xlabel('Week', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Score Trends: Switched vs Control Branches', fontsize=14, fontweight='bold')

plt.grid(True, alpha=0.3)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study regression with week fixed effects
# Create dummy variables for each week relative to switch (omitting week -1 as reference)
df['relative_week'] = df['week'] - 15

# Create week dummies
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    if w != -1:  # omit week -1 as reference
        df[f'week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Event study regression
event_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = f"service_score ~ switched_branch + after_switch + {' + '.join(event_vars)} + staffing_index + visits_index"

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=" * 70)
print("EVENT STUDY REGRESSION: Weekly Treatment Effects")
print("(Reference: Week 14, one week before switch)")
print("=" * 70)

# Extract coefficients for the interaction terms
results = []
for w in range(-9, 8):
    if w == -1:
        results.append({'relative_week': w, 'coef': 0.0, 'std_err': 0.0, 'p_value': 1.0, 'ci_lower': 0.0, 'ci_upper': 0.0})
    else:
        var_name = f'week_{w}'
        if var_name in model_event.params.index:
            coef = model_event.params[var_name]
            std_err = model_event.bse[var_name]
            p_value = model_event.pvalues[var_name]
            ci_lower = coef - 1.96 * std_err
            ci_upper = coef + 1.96 * std_err
            results.append({'relative_week': w, 'coef': coef, 'std_err': std_err, 
                          'p_value': p_value, 'ci_lower': ci_lower, 'ci_upper': ci_upper})

event_df = pd.DataFrame(results)
event_df['week_label'] = event_df['relative_week'].apply(lambda x: f"Week {x+15}")
event_df['significant'] = event_df['p_value'] < 0.05

print(event_df[['week_label', 'coef', 'std_err', 'ci_lower', 'ci_upper', 'p_value', 'significant']].to_string(index=False))

# Summary statistics for pre and post periods
print("\n" + "=" * 70)
print("PRE-POST SUMMARY STATISTICS")
print("=" * 70)

pre_treatment = df[(df['week'] < 15) & (df['switched_branch'] == 1)]
pre_control = df[(df['week'] < 15) & (df['switched_branch'] == 0)]
post_treatment = df[(df['week'] >= 15) & (df['switched_branch'] == 1)]
post_control = df[(df['week'] >= 15) & (df['switched_branch'] == 0)]

print(f"\nPre-switch Treatment: n={len(pre_treatment)}, mean={pre_treatment['service_score'].mean():.2f}, std={pre_treatment['service_score'].std():.2f}")
print(f"Pre-switch Control:   n={len(pre_control)}, mean={pre_control['service_score'].mean():.2f}, std={pre_control['service_score'].std():.2f}")
print(f"Post-switch Treatment: n={len(post_treatment)}, mean={post_treatment['service_score'].mean():.2f}, std={post_treatment['service_score'].std():.2f}")
print(f"Post-switch Control:   n={len(post_control)}, mean={post_control['service_score'].mean():.2f}, std={post_control['service_score'].std():.2f}")

# Check balance on covariates
print("\n" + "=" * 70)
print("COVARIATE BALANCE CHECK (Pre-switch period)")
print("=" * 70)
print(f"Staffing Index - Treatment: {pre_treatment['staffing_index'].mean():.3f}, Control: {pre_control['staffing_index'].mean():.3f}")
print(f"Visits Index - Treatment: {pre_treatment['visits_index'].mean():.3f}, Control: {pre_control['visits_index'].mean():.3f}")

expected_diff = a + b * post_weeks
