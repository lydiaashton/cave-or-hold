import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Count branches that switched vs didn't
branches = df.groupby('branch_id')['switched_branch'].first()
print("Branches that switched:", branches.sum())
print("Branches that didn't switch:", (branches == 0).sum())
print("Total branches:", len(branches))

# Calculate mean service scores by group and period
print("\n=== Service Score Analysis ===")

# Before switch (weeks 6-14), After switch (weeks 15-22)
before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

# Switched branches
switched_before = before[before['switched_branch'] == 1]['service_score'].mean()
switched_after = after[after['switched_branch'] == 1]['service_score'].mean()

# Non-switched branches
control_before = before[before['switched_branch'] == 0]['service_score'].mean()
control_after = after[after['switched_branch'] == 0]['service_score'].mean()

print(f"\nSwitched branches:")
print(f"  Before (weeks 6-14): {switched_before:.2f}")
print(f"  After (weeks 15-22): {switched_after:.2f}")
print(f"  Change: {switched_after - switched_before:.2f}")

print(f"\nNon-switched branches (control):")
print(f"  Before (weeks 6-14): {control_before:.2f}")
print(f"  After (weeks 15-22): {control_after:.2f}")
print(f"  Change: {control_after - control_before:.2f}")

# Difference-in-differences
did_estimate = (switched_after - switched_before) - (control_after - control_before)
print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate

df['treatment'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Also run with cluster-robust standard errors by branch
model_cluster = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== With Cluster-Robust SE (by branch) ===")
print(model_cluster.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate weekly gap between treatment and control
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_avg.columns = ['Control', 'Treatment']
weekly_avg['Gap'] = weekly_avg['Treatment'] - weekly_avg['Control']

# Fit a linear trend to the pre-period gap (weeks 6-14)
pre_period = weekly_avg[weekly_avg.index <= 14].copy()
pre_period = pre_period.reset_index()
pre_period['week_num'] = pre_period['week']

# Linear regression on pre-period gap using statsmodels
X = sm.add_constant(pre_period['week_num'])
y = pre_period['Gap']
pre_model = sm.OLS(y, X).fit()

print("=== Pre-Trend Analysis (Weeks 6-14) ===")
print(pre_model.summary())
print(f"\nPre-period gap trend: {pre_model.params['week_num']:.3f} points per week")

# Predict what the gap would have been post-switch if pre-trend continued
post_weeks = weekly_avg[weekly_avg.index >= 15].copy()
post_weeks = post_weeks.reset_index()
post_weeks['week_num'] = post_weeks['week']

X_post = sm.add_constant(post_weeks['week_num'])
predicted_gap = pre_model.predict(X_post)
actual_gap = post_weeks['Gap']

print(f"\n=== Post-Switch Comparison ===")
print(f"Average actual gap (weeks 15-22): {actual_gap.mean():.2f}")
print(f"Average predicted gap (if pre-trend continued): {predicted_gap.mean():.2f}")
print(f"Difference: {actual_gap.mean() - predicted_gap.mean():.2f}")

# Calculate the "true" treatment effect accounting for pre-trend
# DiD estimate was 2.02, but pre-trend suggests gap was already widening
did_estimate = 2.02
pre_trend_adjustment = pre_model.params['week_num'] * 8  # 8 weeks post-period
print(f"\n=== Adjusted Effect Estimate ===")
print(f"Raw DiD estimate: {did_estimate:.2f}")
print(f"Pre-trend would predict: {pre_trend_adjustment:.2f} points of gap widening")
print(f"Adjusted estimate: {did_estimate - pre_trend_adjustment:.2f}")

# Plot
plt.figure(figsize=(10, 6))
plt.plot(weekly_avg.index, weekly_avg['Gap'], 'o-', label='Actual Gap', linewidth=2, markersize=8)
plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (Week 15)')

# Extend pre-trend line
all_weeks = np.arange(6, 23)
X_all = sm.add_constant(all_weeks)
predicted_all = pre_model.predict(X_all)
plt.plot(all_weeks, predicted_all, 'gray', linestyle=':', linewidth=2, label='Pre-Trend Projection')

plt.xlabel('Week')
plt.ylabel('Treatment - Control Gap')
plt.title('Gap Analysis: Was the Switch Responsible for Improvement?')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('pretrend_analysis.png', dpi=150)
print("\nPlot saved as pretrend_analysis.png")
