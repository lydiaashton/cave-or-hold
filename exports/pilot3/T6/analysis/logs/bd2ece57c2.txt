import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

print("=" * 70)
print("CRITICAL VERIFICATION CHECKS")
print("=" * 70)

# 1. Check parallel trends assumption more carefully
print("\n1. PARALLEL TRENDS CHECK")
print("-" * 50)

pre_df = df[df['week'] < 15].copy()

# Weekly averages for both groups pre-treatment
pre_weekly = pre_df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
pre_weekly.columns = ['Control', 'Treatment']
pre_weekly['Gap'] = pre_weekly['Treatment'] - pre_weekly['Control']

print("Pre-treatment weekly gaps (Treatment - Control):")
print(pre_weekly)

# Test if the gap is changing over time (slope of gap)
weeks = pre_weekly.index.values
gaps = pre_weekly['Gap'].values
slope, intercept = np.polyfit(weeks, gaps, 1)
print(f"\nTrend in pre-treatment gap: {slope:.3f} points/week")
print(f"If slope ≈ 0, parallel trends assumption holds")

# Statistical test: regress gap on week
gap_df = pre_weekly.reset_index()
gap_df['week_centered'] = gap_df['week'] - gap_df['week'].mean()
trend_test = ols('Gap ~ week_centered', data=gap_df).fit()
print(f"Trend test p-value: {trend_test.pvalues['week_centered']:.3f}")
if trend_test.pvalues['week_centered'] > 0.05:
    print("→ No statistically significant trend in pre-treatment gap (GOOD)")
else:
    print("→ Significant trend in pre-treatment gap (CONCERN)")

# 2. Check for selection bias
print("\n" + "=" * 70)
print("2. SELECTION BIAS CHECK")
print("-" * 50)

# Compare treatment and control branches on observable characteristics
