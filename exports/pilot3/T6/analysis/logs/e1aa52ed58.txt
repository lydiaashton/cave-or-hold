import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Create treatment groups
df['group'] = df['switched_branch'].map({1: 'Switched', 0: 'Control'})
df['period'] = df['after_switch'].map({0: 'Before', 1: 'After'})

# Summary table: average service score by group and period
summary = df.groupby(['group', 'period'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("Service Score by Group and Period:")
print(summary)
print()

# Calculate the difference-in-differences
before_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print("Mean Service Scores:")
print(f"  Switched branches - Before: {before_switched:.2f}, After: {after_switched:.2f}, Change: {after_switched - before_switched:.2f}")
print(f"  Control branches  - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")
print()

# Difference-in-differences
did = (after_switched - before_switched) - (after_control - before_control)
print(f"Difference-in-Differences estimate: {did:.2f}")
print()

# Weekly trends to check parallel trends assumption
weekly_avg = df.groupby(['group', 'week'])['service_score'].mean().unstack('group')
print("Weekly averages by group:")
print(weekly_avg.round(2))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Difference-in-differences regression
# Basic DiD model
model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("Basic DiD Regression:")
print(model1.summary().tables[1])
print()

# DiD with branch fixed effects
model2 = ols('service_score ~ switched_branch * after_switch + C(branch_id)', data=df).fit()
print("DiD with Branch Fixed Effects:")
print(model2.summary().tables[1])
print()

# DiD with controls (staffing and visits)
model3 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit()
print("DiD with Controls:")
