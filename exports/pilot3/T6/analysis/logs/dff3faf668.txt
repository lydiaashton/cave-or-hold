import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD: average service scores by group and period
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES ANALYSIS")
print("=" * 60)

# Before period (weeks 6-14)
before = df[df['after_switch'] == 0]
after = df[df['after_switch'] == 1]

print("\n1. AVERAGE SERVICE SCORES BY GROUP AND PERIOD")
print("-" * 60)

# Switched branches
switched_before = before[before['switched_branch'] == 1]['service_score'].mean()
switched_after = after[after['switched_branch'] == 1]['service_score'].mean()

# Non-switched branches
control_before = before[before['switched_branch'] == 0]['service_score'].mean()
control_after = after[after['switched_branch'] == 0]['service_score'].mean()

print(f"\nSwitched Branches (Treatment Group):")
print(f"  Before (weeks 6-14): {switched_before:.2f}")
print(f"  After (weeks 15-22): {switched_after:.2f}")
print(f"  Change: {switched_after - switched_before:.2f}")

print(f"\nNon-Switched Branches (Control Group):")
print(f"  Before (weeks 6-14): {control_before:.2f}")
print(f"  After (weeks 15-22): {control_after:.2f}")
print(f"  Change: {control_after - control_before:.2f}")

# DiD estimate
did_estimate = (switched_after - switched_before) - (control_after - control_before)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")
print("(This is the estimated effect of the new procedure)")

# Number of branches in each group
print("\n2. NUMBER OF BRANCHES")

# Run DiD regression
print("\n3. REGRESSION ANALYSIS")

model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

import pandas as pd

treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_ci_lower = treatment_coef - 1.96 * treatment_se
treatment_ci_upper = treatment_coef + 1.96 * treatment_se

print("TREATMENT EFFECT ESTIMATE (with 95% CI)")

print(f"95% Confidence Interval: [{treatment_ci_lower:.2f}, {treatment_ci_upper:.2f}]")
print(f"Standard Error: {treatment_se:.2f}")
print(f"P-value: {model.pvalues['treatment']:.3f}")

# Calculate effect size relative to baseline
baseline_avg = df[df['after_switch'] == 0]['service_score'].mean()
effect_size_pct = (treatment_coef / baseline_avg) * 100
print(f"\nBaseline average service score: {baseline_avg:.2f}")
print(f"Effect size as % of baseline: {effect_size_pct:.1f}%")

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
from statsmodels.stats.sandwich_covariance import cov_cluster
from scipy import stats

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Full model with clustered SE
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
cov_clustered = cov_cluster(model, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))

treatment_coef = model.params['treatment']
treatment_se_robust = se_clustered[3]
t_stat = treatment_coef / treatment_se_robust
df_clusters = df['branch_id'].nunique() - 1
p_value_clustered = 2 * (1 - stats.t.cdf(abs(t_stat), df=df_clusters))

print("RECONSIDERING THE EVIDENCE")
print("=" * 60)
print(f"Treatment effect: {treatment_coef:.2f} points")
print(f"Clustered SE: {treatment_se_robust:.2f}")
print(f"t-statistic: {t_stat:.2f}")
print(f"Degrees of freedom (clusters - 1): {df_clusters}")
print(f"Clustered p-value: {p_value_clustered:.3f}")
print(f"\n90% CI: [{treatment_coef - 1.645*treatment_se_robust:.2f}, {treatment_coef + 1.645*treatment_se_robust:.2f}]")
print(f"95% CI: [{treatment_coef - 1.96*treatment_se_robust:.2f}, {treatment_coef + 1.96*treatment_se_robust:.2f}]")

# Probability that effect is positive (one-tailed)
prob_positive = stats.t.cdf(t_stat, df=df_clusters)
print(f"\nProbability effect is positive (one-tailed): {prob_positive:.1%}")

# Power analysis - what effect size could we detect?
print("\n\nDETECTABLE EFFECT SIZES")
print("=" * 60)
# With 34 clusters, 8 post periods, what's the MDE?
n_clusters = df['branch_id'].nunique()
n_post = 8
n_pre = 9
print(f"Number of clusters: {n_clusters}")
print(f"Pre-period weeks: {n_pre}")
print(f"Post-period weeks: {n_post}")

# Rough MDE calculation
se_approx = treatment_se_robust
mde_80 = 2.8 * se_approx  # Approximate for 80% power
mde_90 = 3.2 * se_approx
print(f"\nApproximate Minimum Detectable Effect (80% power): {mde_80:.2f} points")
print(f"Approximate Minimum Detectable Effect (90% power): {mde_90:.2f} points")
print(f"\nOur estimate ({treatment_coef:.2f}) is {treatment_coef/mde_80*100:.0f}% of the 80% MDE")
