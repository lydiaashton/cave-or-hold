import pandas as pd
import numpy as np
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# Create the treatment interaction
df['treatment'] = df['switched_branch'] * df['after_switch']

# Two-way fixed effects using demeaning approach
# First, demean by branch
branch_means = df.groupby('branch_id')[['service_score', 'switched_branch', 'after_switch', 'treatment', 'staffing_index', 'visits_index']].transform('mean')
df_demeaned = df[['service_score', 'switched_branch', 'after_switch', 'treatment', 'staffing_index', 'visits_index']] - branch_means

# Then demean by week
week_means = df_demeaned.groupby(df['week']).transform('mean')
df_twfe = df_demeaned - week_means

# Run regression on demeaned data
X = sm.add_constant(df_twfe[['treatment', 'staffing_index', 'visits_index']])
y = df_twfe['service_score']

twfe_model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Two-Way Fixed Effects (Branch + Week):")
print(twfe_model.summary())
print()

# Effect size calculation
control_mean = df[df['switched_branch'] == 0]['service_score'].mean()
treatment_effect = twfe_model.params['treatment']
effect_size_pct = (treatment_effect / control_mean) * 100
print(f"Control group mean service score: {control_mean:.2f}")
print(f"Estimated treatment effect: {treatment_effect:.2f} points")
print(f"Effect size as % of control mean: {effect_size_pct:.1f}%")
print()

# Power analysis - what's the minimum detectable effect?
n_treatment = df[df['switched_branch'] == 1]['branch_id'].nunique()
n_control = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"Number of treatment branches: {n_treatment}")
print(f"Number of control branches: {n_control}")
print(f"Total observations: {len(df)}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Key statistics for decision-making
df['treatment'] = df['switched_branch'] * df['after_switch']

# Two-way fixed effects
branch_means = df.groupby('branch_id')[['service_score', 'treatment', 'staffing_index', 'visits_index']].transform('mean')
df_demeaned = df[['service_score', 'treatment', 'staffing_index', 'visits_index']] - branch_means
week_means = df_demeaned.groupby(df['week']).transform('mean')
df_twfe = df_demeaned - week_means

X = sm.add_constant(df_twfe[['treatment', 'staffing_index', 'visits_index']])
y = df_twfe['service_score']
twfe_model = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== KEY STATISTICS FOR DECISION ===")
print(f"\nTreatment Effect: {twfe_model.params['treatment']:.2f} points")
print(f"Standard Error: {twfe_model.bse['treatment']:.2f}")
print(f"P-value: {twfe_model.pvalues['treatment']:.3f}")
print(f"95% CI: [{twfe_model.conf_int().loc['treatment', 0]:.2f}, {twfe_model.conf_int().loc['treatment', 1]:.2f}]")
print(f"90% CI: [{twfe_model.conf_int(alpha=0.10).loc['treatment', 0]:.2f}, {twfe_model.conf_int(alpha=0.10).loc['treatment', 1]:.2f}]")

# Probability effect is positive (one-sided test)
from scipy import stats
t_stat = twfe_model.params['treatment'] / twfe_model.bse['treatment']
df_residual = twfe_model.df_resid
# One-sided p-value (testing if effect > 0)
one_sided_p = 1 - stats.t.cdf(t_stat, df_residual)
print(f"\nOne-sided p-value (testing effect > 0): {one_sided_p:.3f}")

# Effect size
control_mean = df[df['switched_branch'] == 0]['service_score'].mean()
effect_pct = (twfe_model.params['treatment'] / control_mean) * 100
print(f"\nEffect as % of baseline: {effect_pct:.1f}%")

# How many branches improved vs declined
pre_scores = df[df['week'] < 15].groupby('branch_id')['service_score'].mean()
post_scores = df[df['week'] >= 15].groupby('branch_id')['service_score'].mean()
changes = post_scores - pre_scores
changes = changes.merge(df[['branch_id', 'switched_branch']].drop_duplicates(), on='branch_id')

treatment_improved = (changes[changes['switched_branch']==1]['change'] > 0).sum()
treatment_total = (changes['switched_branch']==1).sum()
control_improved = (changes[changes['switched_branch']==0]['change'] > 0).sum()
control_total = (changes['switched_branch']==0).sum()

print(f"\nBranches with improved scores:")
print(f"  Treatment: {treatment_improved}/{treatment_total} ({100*treatment_improved/treatment_total:.0f}%)")
print(f"  Control: {control_improved}/{control_total} ({100*control_improved/control_total:.0f}%)")
