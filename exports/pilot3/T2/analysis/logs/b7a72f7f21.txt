import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from scipy import stats
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch'] * df['after_switch']

# Summary table: mean service scores by group and period
summary_table = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print("Service Score by Group and Period:")
print(summary_table)

# Calculate the DiD estimate manually
# Control group (switched_branch=0)
control_pre = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_post = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
control_change = control_post - control_pre

# Treatment group (switched_branch=1)
treat_pre = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treat_post = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
treat_change = treat_post - treat_pre

# DiD estimate
did_estimate = treat_change - control_change

print("\n--- Difference-in-Differences Calculation ---")
print(f"Control group (no switch): Pre={control_pre:.2f}, Post={control_post:.2f}, Change={control_change:.2f}")
print(f"Treatment group (switch): Pre={treat_pre:.2f}, Post={treat_post:.2f}, Change={treat_change:.2f}")
print(f"\nDiD Estimate (Treatment effect): {did_estimate:.2f}")

# Load the data

# Create the DiD interaction term
df['treated'] = df['switched_branch'] * df['after_switch']

# Run OLS regression with cluster-robust standard errors
# Model: service_score ~ switched_branch + after_switch + treated
model = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# Check if staffing and visits changed differently for treatment vs control
print("=== Checking if staffing/visits changed differently ===")

# Staffing index DiD
model_staffing = ols('staffing_index ~ switched_branch + after_switch + treated', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nStaffing Index DiD:")
print(f"Treatment effect: {model_staffing.params['treated']:.3f} (p={model_staffing.pvalues['treated']:.3f})")

# Visits index DiD
model_visits = ols('visits_index ~ switched_branch + after_switch + treated', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nVisits Index DiD:")
print(f"Treatment effect: {model_visits.params['treated']:.3f} (p={model_visits.pvalues['treated']:.3f})")

# Add controls to main model
print("\n=== DiD with controls ===")
model_controls = ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', 
                     data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_controls.summary().tables[1])

# Check for heterogeneous effects by branch size or other factors
# Let's see if effect varies by week (dynamic effects)
print("\n=== Dynamic Effects (Event Study) ===")
df['week_relative'] = df['week'] - 15  # weeks relative to switch (0 = first post week)

# Create week dummies interacted with treatment
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    if w != -1:  # omit week -1 as reference
        df[f'week_{w}'] = ((df['week_relative'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run regression with all week dummies
week_cols = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(week_cols)
model_dynamic = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Dynamic effects coefficients:")
for w in range(-9, 8):
    if w != -1:
        col = f'week_{w}'
        coef = model_dynamic.params[col]
        pval = model_dynamic.pvalues[col]
        ci_low, ci_high = model_dynamic.conf_int().loc[col]
        print(f"Week {w}: {coef:.2f} (95% CI: [{ci_low:.2f}, {ci_high:.2f}], p={pval:.3f})")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# Check if staffing and visits changed differently for treatment vs control
print("=== Checking if staffing/visits changed differently ===")

# Staffing index DiD
model_staffing = ols('staffing_index ~ switched_branch + after_switch + treated', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nStaffing Index DiD:")
print(f"Treatment effect: {model_staffing.params['treated']:.3f} (p={model_staffing.pvalues['treated']:.3f})")

# Visits index DiD
model_visits = ols('visits_index ~ switched_branch + after_switch + treated', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nVisits Index DiD:")
print(f"Treatment effect: {model_visits.params['treated']:.3f} (p={model_visits.pvalues['treated']:.3f})")

# Add controls to main model
print("\n=== DiD with controls ===")
model_controls = ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', 
                     data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_controls.summary().tables[1])

# Check for dynamic effects (event study) - simpler approach
print("\n=== Dynamic Effects (Event Study) ===")
df['week_relative'] = df['week'] - 15  # weeks relative to switch (0 = first post week)

# Create week dummies interacted with treatment (rename to avoid formula issues)
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    if w != -1:  # omit week -1 as reference
        col_name = f'w{w:+d}'  # e.g., w-9, w+0, w+7
        df[col_name] = ((df['week_relative'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run regression with all week dummies
week_cols = [f'w{w:+d}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(week_cols)
model_dynamic = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Dynamic effects coefficients (relative to week -1):")
dynamic_results = []
for w in range(-9, 8):
    if w != -1:
        col = f'w{w:+d}'
        coef = model_dynamic.params[col]
        pval = model_dynamic.pvalues[col]
        ci_low, ci_high = model_dynamic.conf_int().loc[col]
        dynamic_results.append((w, coef, ci_low, ci_high, pval))
        print(f"Week {w:+d}: {coef:.2f} (95% CI: [{ci_low:.2f}, {ci_high:.2f}], p={pval:.3f})")

# Plot the event study
fig, ax = plt.subplots(figsize=(12, 6))
weeks = [r[0] for r in dynamic_results]
coefs = [r[1] for r in dynamic_results]
ci_low = [r[2] for r in dynamic_results]
ci_high = [r[3] for r in dynamic_results]

ax.errorbar(weeks, coefs, yerr=[np.array(coefs) - np.array(ci_low), np.array(ci_high) - np.array(coefs)], 
            fmt='o', capsize=5, markersize=8, linewidth=2, color='blue')
ax.axhline(y=0, color='red', linestyle='--', linewidth=2)
ax.axvline(x=-0.5, color='gray', linestyle='-', linewidth=2, label='Switch Point')
ax.set_xlabel('Weeks Relative to Switch (0 = first post-switch week)', fontsize=12)
ax.set_ylabel('Treatment Effect (vs week -1)', fontsize=12)
ax.set_title('Event Study: Dynamic Treatment Effects on Service Score', fontsize=14, fontweight='bold')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(-9, 8))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150, bbox_inches='tight')
print("\nEvent study plot saved as 'event_study.png'")
plt.close()

import pandas as pd

import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')
df['treated'] = df['switched_branch'] * df['after_switch']

# Check if staffing and visits changed differently for treatment vs control
print("=== Checking if staffing/visits changed differently ===")

# Staffing index DiD
model_staffing = ols('staffing_index ~ switched_branch + after_switch + treated', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nStaffing Index DiD:")
print(f"Treatment effect: {model_staffing.params['treated']:.3f} (p={model_staffing.pvalues['treated']:.3f})")

# Visits index DiD
model_visits = ols('visits_index ~ switched_branch + after_switch + treated', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nVisits Index DiD:")
print(f"Treatment effect: {model_visits.params['treated']:.3f} (p={model_visits.pvalues['treated']:.3f})")

# Add controls to main model
print("\n=== DiD with controls ===")
model_controls = ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', 
                     data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_controls.summary().tables[1])

# Check for dynamic effects (event study) - simpler approach
print("\n=== Dynamic Effects (Event Study) ===")
df['week_relative'] = df['week'] - 15  # weeks relative to switch (0 = first post week)

# Create week dummies interacted with treatment (use safe column names)
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    if w != -1:  # omit week -1 as reference
        col_name = f'd{w+10:02d}'  # e.g., d01, d09, d10, d17
        df[col_name] = ((df['week_relative'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run regression with all week dummies
week_cols = [f'd{w+10:02d}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(week_cols)
model_dynamic = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Dynamic effects coefficients (relative to week -1):")
dynamic_results = []
for i, w in enumerate(range(-9, 8)):
    if w != -1:
        col = f'd{w+10:02d}'
        coef = model_dynamic.params[col]
        pval = model_dynamic.pvalues[col]
        ci_low, ci_high = model_dynamic.conf_int().loc[col]
        dynamic_results.append((w, coef, ci_low, ci_high, pval))
        print(f"Week {w:+d}: {coef:.2f} (95% CI: [{ci_low:.2f}, {ci_high:.2f}], p={pval:.3f})")

# Plot the event study
fig, ax = plt.subplots(figsize=(12, 6))
weeks = [r[0] for r in dynamic_results]
coefs = [r[1] for r in dynamic_results]
ci_low = [r[2] for r in dynamic_results]
ci_high = [r[3] for r in dynamic_results]

ax.errorbar(weeks, coefs, yerr=[np.array(coefs) - np.array(ci_low), np.array(ci_high) - np.array(coefs)],

ax.legend()
plt.tight_layout()
