import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Calculate mean service scores by group and time period
print("Mean service scores by group and time period:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(pivot)

# Calculate the differences
print("\n" + "="*60)
print("DIFFERENCE-IN-DIFFERENCES CALCULATION:")
print("="*60)

# Control group (switched_branch=0)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_diff = control_after - control_before

# Treatment group (switched_branch=1)
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_diff = treat_after - treat_before

# Difference-in-differences
did = treat_diff - control_diff

print(f"\nControl group (no switch):")
print(f"  Before: {control_before:.2f}")
print(f"  After:  {control_after:.2f}")
print(f"  Change: {control_diff:.2f}")

print(f"  Before: {treat_before:.2f}")
print(f"  After:  {treat_after:.2f}")
print(f"  Change: {treat_diff:.2f}")

print(f"\nDifference-in-Differences estimate: {did:.2f}")
print(f"  (Treatment change) - (Control change) = {treat_diff:.2f} - {control_diff:.2f} = {did:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*treated + beta2*post + beta3*(treated*post) + error
# beta3 is the DID estimate

df['treated_post'] = df['switched_branch'] * df['after_switch']

# Basic OLS regression
model = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()
print("Basic OLS Regression:")
print(model.summary())

# Now with cluster-robust standard errors (clustered by branch)
model_clustered = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n" + "="*60)
print("Regression with Cluster-Robust Standard Errors (by branch):")
print("="*60)
print(model_clustered.summary())

# Full regression with controls
df['treated_post'] = df['switched_branch'] * df['after_switch']

print("="*60)
print("REGRESSION WITH CONTROLS (staffing_index, visits_index):")
print("="*60)

model_with_controls = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', 
                          data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_with_controls.summary())

               data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study: effect by weeks since switch
df['weeks_since_switch'] = df['week'] - 15  # 0 = first week after switch, negative = before

# Create bins for event study
def week_bin(w):
    if w < -4:
        return "-5 or earlier"
    elif w == -4:
        return "-4"
    elif w == -3:
        return "-3"
    elif w == -2:
        return "-2"
    elif w == -1:
        return "-1 (last week before)"
    elif w == 0:
        return "0 (first week after)"
    elif w == 1:
        return "1"
    elif w == 2:
        return "2"
    elif w == 3:
        return "3"
    elif w >= 4:
        return "4 or later"
    else:
        return "other"

df['week_bin'] = df['weeks_since_switch'].apply(week_bin)

# Calculate DID for each time bin
event_study = []
for bin_name in sorted(df['week_bin'].unique()):
    bin_data = df[df['week_bin'] == bin_name]
    if len(bin_data) == 0:
        continue
    
    control_mean = bin_data[bin_data['switched_branch']==0]['service_score'].mean()
    treat_mean = bin_data[bin_data['switched_branch']==1]['service_score'].mean()
    diff = treat_mean - control_mean
    n_control = len(bin_data[bin_data['switched_branch']==0])
    n_treat = len(bin_data[bin_data['switched_branch']==1])
    
    event_study.append({
        'week_bin': bin_name,
        'control_mean': control_mean,
        'treat_mean': treat_mean,
        'difference': diff,
        'n_control': n_control,
        'n_treat': n_treat
    })

event_df = pd.DataFrame(event_study)
print("Event Study Results (Treatment - Control difference by week):")
print(event_df.to_string(index=False))

# Create a plot
plt.figure(figsize=(14, 6))

# Plot the differences
bins_ordered = ["-5 or earlier", "-4", "-3", "-2", "-1 (last week before)", 
                "0 (first week after)", "1", "2", "3", "4 or later"]
bins_present = [b for b in bins_ordered if b in event_df['week_bin'].values]

x_pos = range(len(bins_present))
diffs = [event_df[event_df['week_bin']==b]['difference'].values[0] for b in bins_present]

plt.bar(x_pos, diffs, color=['gray' if 'before' in b or b.startswith('-') or b[0]=='-' else 'steelblue' for b in bins_present], alpha=0.7)
plt.axhline(y=0, color='black', linestyle='-', linewidth=1)
plt.axvline(x=bins_present.index("0 (first week after)")-0.5, color='red', linestyle='--', linewidth=2, label='Switch Point')

plt.xlabel('Weeks Relative to Switch', fontsize=12)
plt.ylabel('Service Score Difference (Treatment - Control)', fontsize=12)
plt.title('Event Study: Service Score Gap Between Treatment and Control Groups', fontsize=14)
plt.xticks(x_pos, bins_present, rotation=45, ha='right')
plt.legend()
plt.grid(True, alpha=0.3, axis='y')
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as event_study.png")
