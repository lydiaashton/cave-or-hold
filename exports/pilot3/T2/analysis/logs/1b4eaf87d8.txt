import pandas as pd
import numpy as np
import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create the interaction term for difference-in-differences
df['treatment'] = df['switched_branch'] * df['after_switch']

# Difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the treatment effect

model = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Cluster-robust standard errors by branch
model_clustered = smf.ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Clustered Standard Errors (by branch) ===")
print(model_clustered.summary())

# Main DiD model with branch fixed effects and clustered SEs
model = smf.ols('service_score ~ after_switch + treatment + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Final DiD Model (with branch fixed effects) ===")
print(model.summary())

# Extract key results
treatment_coef = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pval = model.pvalues['treatment']
treatment_ci_lower = model.conf_int().loc['treatment', 0]
treatment_ci_upper = model.conf_int().loc['treatment', 1]

print(f"\n=== Key Results ===")
print(f"Treatment effect estimate: {treatment_coef:.2f} points")
print(f"95% Confidence interval: [{treatment_ci_lower:.2f}, {treatment_ci_upper:.2f}]")
print(f"P-value: {treatment_pval:.3f}")
print(f"Statistically significant at 5% level: {treatment_pval < 0.05}")
