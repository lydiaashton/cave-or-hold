import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Calculate average service scores by group and period
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).reset_index()
print("Service scores by group and period:")
print(summary)

# Pivot for easier viewing
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
print("\nPivot table (mean service scores):")
print(pivot)

# Calculate difference-in-differences
# Treatment group change
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treatment_change = treatment_after - treatment_before

# Control group change
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_change = control_after - control_before

# Difference-in-differences
did = treatment_change - control_change

print("\n--- Difference-in-Differences Analysis ---")
print(f"Treatment group (switched): Before={treatment_before:.2f}, After={treatment_after:.2f}, Change={treatment_change:.2f}")
print(f"Control group (not switched): Before={control_before:.2f}, After={control_after:.2f}, Change={control_change:.2f}")
print(f"\nDifference-in-Differences estimate: {did:.2f}")
print(f"This represents the estimated effect of the new procedure on service scores.")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with robust standard errors
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimator

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='HC1')
print("Difference-in-Differences Regression Results:")
print(model.summary())

# Extract the treatment effect
treatment_effect = model.params['treatment']
treatment_pvalue = model.pvalues['treatment']
treatment_ci = model.conf_int().loc['treatment']

print(f"\n--- Key Results ---")

print(f"95% Confidence Interval: [{treatment_ci[0]:.3f}, {treatment_ci[1]:.3f}]")
print(f"P-value: {treatment_pvalue:.6f}")
print(f"Statistically significant: {'Yes' if treatment_pvalue < 0.05 else 'No'}")

import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Calculate the before and after average for each branch
branch_before = df[df['after_switch']==0].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_before.columns = ['branch_id', 'switched_branch', 'score_before']

branch_after = df[df['after_switch']==1].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_after.columns = ['branch_id', 'switched_branch', 'score_after']

# Merge
branch_change = branch_before.merge(branch_after, on=['branch_id', 'switched_branch'])
branch_change['change'] = branch_change['score_after'] - branch_change['score_before']

# Compare changes between treatment and control
treatment_changes = branch_change[branch_change['switched_branch']==1]['change']
control_changes = branch_change[branch_change['switched_branch']==0]['change']

print("--- Distribution of Score Changes ---")
print("\nTreatment group (switched branches):")
print(treatment_changes.describe())

print("\nControl group (non-switched branches):")
print(control_changes.describe())

# Test if the difference in changes is significant
from scipy import stats
t_stat, p_val = stats.ttest_ind(treatment_changes, control_changes)
print(f"\nTwo-sample t-test comparing changes:")
print(f"  t-statistic: {t_stat:.3f}")
print(f"  p-value: {p_val:.6f}")

# What percentage of treatment branches improved?
print(f"\nPercentage of treatment branches with improved scores: {(treatment_changes > 0).mean()*100:.1f}%")
print(f"Percentage of control branches with improved scores: {(control_changes > 0).mean()*100:.1f}%")

# Average improvement for treatment branches that improved
print(f"\nAverage improvement for treatment branches that improved: {treatment_changes[treatment_changes > 0].mean():.2f} points")

branch_after = df[df['after_switch']==1].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_change = branch_before.merge(branch_after, on=['branch_id', 'switched_branch'])
branch_change['change'] = branch_change['score_after'] - branch_change['score_before']

ax2 = axes[1]
treatment_changes = branch_change[branch_change['switched_branch']==1]['change']
