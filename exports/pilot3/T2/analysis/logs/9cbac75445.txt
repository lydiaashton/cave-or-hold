import pandas as pd
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model 1: Simple DiD regression
print("=" * 60)
print("MODEL 1: SIMPLE DIFFERENCE-IN-DIFFERENCES")
print("=" * 60)
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model1.summary().tables[1])

# Model 2: DiD with controls
print("\n" + "=" * 60)
print("MODEL 2: DiD WITH CONTROLS (staffing_index, visits_index)")
print("=" * 60)
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model2.summary().tables[1])

print("MODEL 3: DiD WITH BRANCH FIXED EFFECTS")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run the main model for the summary (simple DiD with controls, cluster-robust SE)
model = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Key statistics
treatment_effect = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pval = model.pvalues['treatment']
treatment_ci_low = model.conf_int().loc['treatment'][0]
treatment_ci_high = model.conf_int().loc['treatment'][1]

# Sample sizes
n_branches = df['branch_id'].nunique()
n_treatment_branches = df[df['switched_branch'] == 1]['branch_id'].nunique()
n_control_branches = df[df['switched_branch'] == 0]['branch_id'].nunique()
n_observations = len(df)

# Pre-period comparison (check baseline differences)
pre_df = df[df['after_switch'] == 0]
pre_control_mean = pre_df[pre_df['switched_branch'] == 0]['service_score'].mean()
pre_treatment_mean = pre_df[pre_df['switched_branch'] == 1]['service_score'].mean()
baseline_diff = pre_treatment_mean - pre_control_mean

# Post-period comparison
post_df = df[df['after_switch'] == 1]
post_control_mean = post_df[post_df['switched_branch'] == 0]['service_score'].mean()
post_treatment_mean = post_df[post_df['switched_branch'] == 1]['service_score'].mean()

# Calculate effect size (Cohen's d approximation)
pooled_std = df['service_score'].std()
effect_size = treatment_effect / pooled_std

print("=" * 70)
print("ANALYSIS SUMMARY FOR BOARD")
print("=" * 70)
