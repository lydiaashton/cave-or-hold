import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Count branches by switch status
branches_switched = df[df['switched_branch'] == 1]['branch_id'].nunique()
branches_not_switched = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"Branches that switched: {branches_switched}")
print(f"Branches that didn't switch: {branches_not_switched}")

# Calculate mean service scores for each group/time combination
group_means = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print("\nService scores by group and period:")
print(group_means)

# Calculate the difference-in-differences
# Treatment group (switched=1): before and after
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_diff = treatment_after - treatment_before

# Control group (switched=0): before and after
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
control_diff = control_after - control_before

# Difference-in-differences
did_estimate = treatment_diff - control_diff

print(f"\nTreatment group (switched branches):")
print(f"  Before switch: {treatment_before:.2f}")
print(f"  After switch: {treatment_after:.2f}")
print(f"  Change: {treatment_diff:.2f}")

print(f"\nControl group (non-switched branches):")
print(f"  Before switch: {control_before:.2f}")
print(f"  After switch: {control_after:.2f}")
print(f"  Change: {control_diff:.2f}")

print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# DiD regression with controls
model_controls = ols('service_score ~ treatment + post + did + staffing_index + visits_index', 
                     data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("DiD regression with controls (clustered SE at branch level):")
print(model_controls.summary())

# Try branch fixed effects using a different approach - demeaning
# First, let's see which branches have data in both periods
branch_periods = df.groupby('branch_id')['after_switch'].agg(['min', 'max', 'count'])
print("\n\nBranch period coverage:")
print(branch_periods)

# Let's try a model with branch FE but dropping the treatment term (it's collinear with branch FE)
# The post term and did term should still be estimable
df_with_dummies = pd.get_dummies(df, columns=['branch_id'], drop_first=True)
fe_cols = [c for c in df_with_dummies.columns if c.startswith('branch_id_')]

# Build model formula
formula = 'service_score ~ post + did + staffing_index + visits_index + ' + ' + '.join(fe_cols)
model_fe = ols(formula, data=df_with_dummies).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nBranch FE model (treatment absorbed into FE):")
print(model_fe.summary())

# Also let's get confidence intervals for the DiD estimate
print(f"\n\nDiD estimate: {model_fe.params['did']:.2f}")
print(f"95% CI: [{model_fe.conf_int().loc['did', 0]:.2f}, {model_fe.conf_int().loc['did', 1]:.2f}]")
print(f"p-value: {model_fe.pvalues['did']:.3f}")

import pandas as pd

from scipy import stats

# Check if staffing or visits changed differently for treatment vs control
print("Staffing index changes:")
staffing_diff = df.groupby(['switched_branch', 'after_switch'])['staffing_index'].mean().unstack()
staffing_diff['change'] = staffing_diff[1] - staffing_diff[0]
print(staffing_diff)

print("\nVisits index changes:")
visits_diff = df.groupby(['switched_branch', 'after_switch'])['visits_index'].mean().unstack()
visits_diff['change'] = visits_diff[1] - visits_diff[0]
print(visits_diff)

# Power analysis - what effect size could we detect?
# We have 34 branches, 17 weeks each
# With clustered SE at branch level, effective sample size is closer to number of branches
n_treatment = 20
n_control = 14

# Standard deviation of service score
sd = df['service_score'].std()
print(f"\nOverall SD of service score: {sd:.2f}")

# SD of the difference (before-after) for each branch
branch_data = df.groupby(['branch_id', 'after_switch'])['service_score'].mean().unstack()
branch_data['diff'] = branch_data[1] - branch_data[0]
sd_diff = branch_data['diff'].std()
print(f"SD of branch-level before-after differences: {sd_diff:.2f}")

# For a two-sample t-test with these sample sizes, what's the minimum detectable effect?
# Using 80% power, alpha=0.05
from statsmodels.stats.power import TTestIndPower
power_analysis = TTestIndPower()
# Effect size (Cohen's d) we can detect with 80% power
effect_size_80 = power_analysis.solve_power(effect_size=None, nobs1=n_treatment, ratio=n_control/n_treatment, alpha=0.05, power=0.80)
print(f"\nMinimum detectable effect size (Cohen's d) with 80% power: {effect_size_80:.2f}")
print(f"In service score points (using SD={sd:.2f}): {effect_size_80 * sd:.2f}")

# What's our observed effect size?
observed_effect = 1.90  # DiD estimate
observed_cohens_d = observed_effect / sd_diff
print(f"\nObserved effect: {observed_effect:.2f} points")
print(f"Observed Cohen's d (using SD of differences): {observed_cohens_d:.2f}")
