import pandas as pd

switched_branches = df[df['switched_branch'] == 1]['branch_id'].nunique()
control_branches = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"  Treatment (switched): {switched_branches} branches")
print(f"  Control (not switched): {control_branches} branches")

# Check pre-trends: average service score by week and group

pre_trends.columns = ['Control_Mean', 'Control_Std', 'Control_N', 'Treated_Mean', 'Treated_Std', 'Treated_N']

pre_avg = pre_switch.groupby('switched_branch')['service_score'].mean()

import pandas as pd
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with cluster-robust standard errors
print("=== Difference-in-Differences Regression ===")
print("\nModel 1: Basic DiD (no controls)")
model1 = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()
print(model1.summary())

print("\n\nModel 2: DiD with controls (staffing_index, visits_index)")
model2 = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', data=df).fit()
print(model2.summary())

# Cluster-robust standard errors by branch
print("\n\nModel 3: DiD with cluster-robust SEs (clustered by branch)")
model3 = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model3.summary())

print("\n\nModel 4: DiD with controls and cluster-robust SEs")
model4 = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

import pandas as pd
from statsmodels.formula.api import ols

df['treated_post'] = df['switched_branch'] * df['after_switch']

# Final model with controls and cluster-robust SEs
model = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', 
            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Calculate effect size relative to baseline
