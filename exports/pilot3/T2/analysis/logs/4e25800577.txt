import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
branches_switched = df[df['switched_branch'] == 1]['branch_id'].nunique()
branches_not_switched = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"Branches that switched: {branches_switched}")
print(f"Branches that didn't switch: {branches_not_switched}")
print(f"Total branches: {df['branch_id'].nunique()}")

# Check weeks per branch
weeks_per_branch = df.groupby('branch_id')['week'].nunique()
print(f"\nWeeks per branch (should be 17 for all):")
print(weeks_per_branch.describe())

# Check if all branches have data for all weeks
print(f"\nAny missing weeks? {len(weeks_per_branch[weeks_per_branch < 17])} branches with incomplete data")

# Create treatment group indicator (interaction term for DiD)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary stats by group and period
print("\n=== Service Score by Group and Period ===")
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Calculate the DiD manually
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
pre_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
post_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nPre-period control mean: {pre_control:.2f}")
print(f"Post-period control mean: {post_control:.2f}")
print(f"Control change: {post_control - pre_control:.2f}")
print(f"\nPre-period treatment mean: {pre_treatment:.2f}")
print(f"Post-period treatment mean: {post_treatment:.2f}")
print(f"Treatment change: {post_treatment - pre_treatment:.2f}")

did_effect = (post_treatment - pre_treatment) - (post_control - pre_control)
print(f"\n=== DiD Estimate: {did_effect:.2f} ===")

import numpy as np

from statsmodels.formula.api import ols

df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
print("=== Basic DiD Regression ===")
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model1.summary())

# DiD with branch fixed effects
print("\n=== DiD with Branch Fixed Effects ===")
df_cat = df.copy()
df_cat['branch_id'] = df_cat['branch_id'].astype('category')
model2 = ols('service_score ~ after_switch + treatment + C(branch_id)', data=df_cat).fit()
print(model2.summary())

# DiD with branch and week fixed effects
print("\n=== DiD with Branch and Week Fixed Effects ===")
df_cat['week'] = df_cat['week'].astype('category')
model3 = ols('service_score ~ treatment + C(branch_id) + C(week)', data=df_cat).fit()
