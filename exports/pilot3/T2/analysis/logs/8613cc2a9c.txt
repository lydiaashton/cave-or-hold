import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Basic Difference-in-Differences regression
# service_score = β0 + β1*switched_branch + β2*after_switch + β3*(switched_branch*after_switch) + ε
# β3 is the DiD estimator - the effect of the new procedure

model = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit()
print("Basic DiD Regression Results:")
print(model.summary())

# DiD with controls
model_controls = ols('service_score ~ switched_branch + after_switch + treated_post + staffing_index + visits_index', data=df).fit()
print("\n\nDiD Regression with Controls:")
print(model_controls.summary())

# Calculate the DiD estimate manually
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

did_estimate = (treatment_after - treatment_before) - (control_after - control_before)
print(f"\n\nManual DiD calculation:")
print(f"Control before: {control_before:.2f}, Control after: {control_after:.2f}, Change: {control_after - control_before:.2f}")
print(f"Treatment before: {treatment_before:.2f}, Treatment after: {treatment_after:.2f}, Change: {treatment_after - treatment_before:.2f}")
print(f"DiD Estimate (treatment effect): {did_estimate:.2f}")
