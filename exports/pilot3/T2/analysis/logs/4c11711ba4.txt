import pandas as pd

df = pd.read_csv('branches.csv')

print(pivot)

# Calculate the difference-in-differences manually

before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
# After period (after_switch=1)
after_switched = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression
print("=" * 60)
print("MODEL 1: Basic Difference-in-Differences")
print("=" * 60)
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model1.summary().tables[1])

# DiD with week fixed effects (controls for common time trends)
print("\n" + "=" * 60)
print("MODEL 2: DiD with Week Fixed Effects")
print("=" * 60)
df_week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=True)
df_with_weeks = pd.concat([df, df_week_dummies], axis=1)
week_cols = [c for c in df_with_weeks.columns if c.startswith('week_')]
formula = 'service_score ~ switched_branch + after_switch + treatment + ' + ' + '.join(week_cols)
model2 = ols(formula, data=df_with_weeks).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model2.summary().tables[1])

# Check pre-trends: compare trends before week 15 between groups
print("\n" + "=" * 60)
print("PRE-TREND CHECK: Weeks 6-14 (Before Switch)")
print("=" * 60)
pre_period = df[df['week'] < 15].copy()
pre_period['week_centered'] = pre_period['week'] - pre_period['week'].mean()
model_pre = ols('service_score ~ switched_branch * week_centered', data=pre_period).fit(cov_type='cluster', cov_kwds={'groups': pre_period['branch_id']})
print(model_pre.summary().tables[1])
print("\nInterpretation: The interaction term shows if switched branches had different trends BEFORE the switch.")
print("If significant, this suggests pre-existing differences (violates DiD assumption).")

# Check post-period effects by week
print("\n" + "=" * 60)
print("EFFECT BY WEEK (Post-Switch)")
print("=" * 60)
post_period = df[df['week'] >= 15].copy()
post_period['week_factor'] = post_period['week'].astype(str)
# Create interaction of treatment with each post week
for week in range(15, 23):
    post_period[f'post_week_{week}'] = ((post_period['week'] == week) & (post_period['switched_branch'] == 1)).astype(int)

# Run regression with week-specific treatment effects
week_vars = [f'post_week_{w}' for w in range(15, 23)]
formula_week = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(week_vars)
# Add week dummies for control
week_dummies_post = pd.get_dummies(post_period['week'], prefix='w', drop_first=True)
post_with_weeks = pd.concat([post_period, week_dummies_post], axis=1)
week_dummy_cols = [c for c in post_with_weeks.columns if c.startswith('w_')]
formula_week_full = 'service_score ~ switched_branch + ' + ' + '.join(week_vars) + ' + ' + ' + '.join(week_dummy_cols)
model_week = ols(formula_week_full, data=post_with_weeks).fit(cov_type='cluster', cov_kwds={'groups': post_with_weeks['branch_id']})
print("Week-specific treatment effects (relative to control branches in same week):")
for w in range(15, 23):
    coef = model_week.params[f'post_week_{w}']
    se = model_week.bse[f'post_week_{w}']
    pval = model_week.pvalues[f'post_week_{w}']
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"  Week {w}: {coef:.2f} (SE: {se:.2f}, p={pval:.3f}) {sig}")

import pandas as pd

model2 = ols('service_score ~ switched_branch + treatment + C(week_cat)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
