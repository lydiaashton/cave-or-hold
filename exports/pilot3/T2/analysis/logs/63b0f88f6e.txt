import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# How many branches switched vs didn't?
print("Branches by switch status:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Create treatment interaction term (DiD estimator)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Summary of service scores by group and period
print("\n\nService Score Summary by Group and Period:")
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

# Calculate the DiD manually
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
pre_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
post_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("\n\nManual DiD Calculation:")
print(f"Control group - Pre: {pre_control:.2f}, Post: {post_control:.2f}, Change: {post_control - pre_control:.2f}")
print(f"Treatment group - Pre: {pre_treatment:.2f}, Post: {post_treatment:.2f}, Change: {post_treatment - pre_treatment:.2f}")
did_estimate = (post_treatment - pre_treatment) - (post_control - pre_control)
print(f"DiD Estimate (Treatment Effect): {did_estimate:.2f}")

# Run regression with robust standard errors clustered by branch
print("\n\nRegression Analysis:")
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print(model.summary())

# Try with cluster-robust standard errors
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nClustered Standard Errors (by branch):")
print(model_clustered.summary())

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Check if staffing or visits changed differently for treatment vs control
print("Did staffing or visits change differently for treatment vs control?")

# DiD for staffing_index
model_staffing = ols('staffing_index ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nStaffing Index DiD:")
print(f"Treatment effect: {model_staffing.params['treatment']:.4f}, p-value: {model_staffing.pvalues['treatment']:.4f}")

# DiD for visits_index
model_visits = ols('visits_index ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nVisits Index DiD:")
print(f"Treatment effect: {model_visits.params['treatment']:.4f}, p-value: {model_visits.pvalues['treatment']:.4f}")

# Control for staffing and visits in the main model
print("\n\nMain model controlling for staffing and visits:")
model_controlled = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_controlled.summary())

# Check effect size relative to standard deviation
print("\n\nEffect Size Interpretation:")
print(f"Overall service_score std: {df['service_score'].std():.2f}")
print(f"Treatment effect: {model_controlled.params['treatment']:.2f}")
print(f"Effect as % of one std dev: {model_controlled.params['treatment'] / df['service_score'].std() * 100:.1f}%")

# Confidence interval
print(f"\n95% CI for treatment effect: [{model_controlled.conf_int().loc['treatment'][0]:.2f}, {model_controlled.conf_int().loc['treatment'][1]:.2f}]")
