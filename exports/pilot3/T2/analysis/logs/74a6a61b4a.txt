import pandas as pd
import numpy as np
import statsmodels.api as sm
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression
# Model: service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
X = df[['switched_branch', 'after_switch', 'treatment']]
X = sm.add_constant(X)
y = df['service_score']

model = sm.OLS(y, X).fit()
print("DiD Regression Results:")
print(model.summary())

# Also run with cluster-robust standard errors by branch
model_clustered = sm.OLS(y, X).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with Clustered Standard Errors (by branch):")
print(model_clustered.summary())

# Add controls for staffing and visits
X_controls = df[['switched_branch', 'after_switch', 'treatment', 'staffing_index', 'visits_index']]
X_controls = sm.add_constant(X_controls)
model_controls = sm.OLS(y, X_controls).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with Controls (clustered SE):")
print(model_controls.summary())
