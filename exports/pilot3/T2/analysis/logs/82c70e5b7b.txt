import pandas as pd
import numpy as np
from scipy import stats

# Load the data
df = pd.read_csv('branches.csv')

# Branch-level analysis (averaging within branch before/after)
branch_avg = df.groupby(['branch_id', 'switched_branch', 'after_switch'])['service_score'].mean().reset_index()

# Pivot to get before and after for each branch
branch_pivot = branch_avg.pivot_table(index=['branch_id', 'switched_branch'], columns='after_switch', values='service_score')
branch_pivot.columns = ['Before', 'After']
branch_pivot['Change'] = branch_pivot['After'] - branch_pivot['Before']
branch_pivot = branch_pivot.reset_index()

print("Average change by group:")
print(branch_pivot.groupby('switched_branch')['Change'].agg(['mean', 'std', 'count']))

# T-test on the changes
control_changes = branch_pivot[branch_pivot['switched_branch'] == 0]['Change']
treatment_changes = branch_pivot[branch_pivot['switched_branch'] == 1]['Change']

t_stat, p_val = stats.ttest_ind(treatment_changes, control_changes, equal_var=False)  # Welch's t-test
print(f"\nTwo-sample t-test on changes (Welch's):")
print(f"t-statistic: {t_stat:.3f}, p-value: {p_val:.3f}")

# Confidence interval for the DiD estimate
did_estimate = treatment_changes.mean() - control_changes.mean()
# Standard error of the difference
se_diff = np.sqrt(treatment_changes.var()/len(treatment_changes) + control_changes.var()/len(control_changes))
ci_lower = did_estimate - 1.96 * se_diff
ci_upper = did_estimate + 1.96 * se_diff

print(f"\nDiD Estimate: {did_estimate:.2f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")

# Also check if the treatment group had higher baseline scores
print("\n\nBaseline comparison (before switch):")
print(f"Control mean (before): {control_before:.2f}")
print(f"Treatment mean (before): {treatment_before:.2f}")
baseline_diff = treatment_before - control_before
print(f"Difference: {baseline_diff:.2f}")

import pandas as pd
import numpy as np
from scipy import stats

# Load the data
df = pd.read_csv('branches.csv')

# Calculate the key metrics
control_before = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
control_after = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
treatment_before = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
treatment_after = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print("BASELINE COMPARISON (Before Switch):")
print(f"Control mean (before): {control_before:.2f}")
print(f"Treatment mean (before): {treatment_before:.2f}")
print(f"Difference: {treatment_before - control_before:.2f}")

print("\n\nAFTER SWITCH:")
print(f"Control mean (after): {control_after:.2f}")
print(f"Treatment mean (after): {treatment_after:.2f}")
print(f"Difference: {treatment_after - control_after:.2f}")

# DiD calculation
control_change = control_after - control_before
treatment_change = treatment_after - treatment_before
did_estimate = treatment_change - control_change

print(f"\n\nDIFFERENCE-IN-DIFFERENCES:")
print(f"Control change: {control_change:.2f}")
print(f"Treatment change: {treatment_change:.2f}")
print(f"DiD Estimate (Treatment Effect): {did_estimate:.2f} points")

# Branch-level analysis
branch_avg = df.groupby(['branch_id', 'switched_branch', 'after_switch'])['service_score'].mean().reset_index()
branch_pivot = branch_avg.pivot_table(index=['branch_id', 'switched_branch'], columns='after_switch', values='service_score')
branch_pivot.columns = ['Before', 'After']
branch_pivot['Change'] = branch_pivot['After'] - branch_pivot['Before']
branch_pivot = branch_pivot.reset_index()

control_changes = branch_pivot[branch_pivot['switched_branch'] == 0]['Change']
treatment_changes = branch_pivot[branch_pivot['switched_branch'] == 1]['Change']

# Welch's t-test
t_stat, p_val = stats.ttest_ind(treatment_changes, control_changes, equal_var=False)

# Standard error and CI
se_diff = np.sqrt(treatment_changes.var()/len(treatment_changes) + control_changes.var()/len(control_changes))
ci_lower = did_estimate - 1.96 * se_diff
ci_upper = did_estimate + 1.96 * se_diff

print(f"\nStatistical Significance:")
print(f"t-statistic: {t_stat:.3f}")
print(f"p-value: {p_val:.3f}")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")

# Effect size (Cohen's d)
pooled_std = np.sqrt(((len(treatment_changes)-1)*treatment_changes.var() + (len(control_changes)-1)*control_changes.var()) / (len(treatment_changes) + len(control_changes) - 2))
cohens_d = did_estimate / pooled_std
print(f"Cohen's d (effect size): {cohens_d:.3f}")

# Percentage improvement relative to baseline
pct_improvement = (did_estimate / treatment_before) * 100
print(f"\nPercentage improvement relative to treatment baseline: {pct_improvement:.1f}%")
