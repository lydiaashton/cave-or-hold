import pandas as pd
import numpy as np
from statsmodels.formula.api import ols
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# Create treatment group indicator (switched_branch) and time period indicator (after_switch)
# DiD estimator: interaction between switched_branch and after_switch

# First, let's see the basic averages
print("=== Average Service Scores ===")
print("\nBefore switch (weeks 6-14):")
before = df[df['after_switch'] == 0]
print(f"  Switched branches: {before[before['switched_branch']==1]['service_score'].mean():.2f}")
print(f"  Non-switched branches: {before[before['switched_branch']==0]['service_score'].mean():.2f}")

print("\nAfter switch (weeks 15-22):")
after = df[df['after_switch'] == 1]
print(f"  Switched branches: {after[after['switched_branch']==1]['service_score'].mean():.2f}")
print(f"  Non-switched branches: {after[after['switched_branch']==0]['service_score'].mean():.2f}")

# Calculate the DiD manually
before_switched = before[before['switched_branch']==1]['service_score'].mean()
before_control = before[before['switched_branch']==0]['service_score'].mean()
after_switched = after[after['switched_branch']==1]['service_score'].mean()
after_control = after[after['switched_branch']==0]['service_score'].mean()

diff_switched = after_switched - before_switched
diff_control = after_control - before_control
did_estimate = diff_switched - diff_control

print(f"\n=== Difference-in-Differences Calculation ===")
print(f"Change for switched branches: {diff_switched:.2f}")
print(f"Change for non-switched branches: {diff_control:.2f}")
print(f"DiD Estimate (treatment effect): {did_estimate:.2f}")

# Now run a regression with standard errors
print("\n=== Regression Analysis ===")
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print(model.summary())

# Cluster-robust standard errors by branch
model_clustered = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== With Clustered Standard Errors (by branch) ===")
print(model_clustered.summary())

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Test if the groups had similar trends before the switch
# Run regression on pre-period only with week trend interaction
pre_period = df[df['week'] < 15].copy()
pre_period['week_centered'] = pre_period['week'] - pre_period['week'].mean()
model_pre = ols('service_score ~ switched_branch * week_centered', data=pre_period).fit(cov_type='cluster', cov_kwds={'groups': pre_period['branch_id']})
print("=== Pre-period trend test ===")
print(model_pre.summary())

# The coefficient on switched_branch:week_centered tells us if the trends were different
# If not significant, parallel trends assumption holds

print("\n=== Additional Robustness Checks ===")

# Check balance on covariates
print("\nAverage staffing_index and visits_index by group:")
print(df.groupby('switched_branch')[['staffing_index', 'visits_index']].mean())

# Add covariates to the main model
model_with_covariates = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Model with Covariates ===")
print(model_with_covariates.summary())

# Check effect heterogeneity - is the effect consistent across branches?
# Get branch-level effects
branch_effects = df.groupby(['branch_id', 'switched_branch']).agg({
    'service_score': ['mean'],
    'after_switch': 'mean'  # This tells us how many post-period observations
}).reset_index()

# Actually, let's compute branch-level pre and post means
branch_pre = df[df['after_switch']==0].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_pre.columns = ['branch_id', 'switched_branch', 'pre_score']
branch_post = df[df['after_switch']==1].groupby(['branch_id', 'switched_branch'])['service_score'].mean().reset_index()
branch_post.columns = ['branch_id', 'switched_branch', 'post_score']

branch_level = branch_pre.merge(branch_post, on=['branch_id', 'switched_branch'])
branch_level['change'] = branch_level['post_score'] - branch_level['pre_score']

print("\n=== Branch-level changes ===")
print("Switched branches (n={}):".format(len(branch_level[branch_level['switched_branch']==1])))
print(f"  Mean change: {branch_level[branch_level['switched_branch']==1]['change'].mean():.2f}")
print(f"  Std dev: {branch_level[branch_level['switched_branch']==1]['change'].std():.2f}")

print("\nNon-switched branches (n={}):".format(len(branch_level[branch_level['switched_branch']==0])))
print(f"  Mean change: {branch_level[branch_level['switched_branch']==0]['change'].mean():.2f}")
print(f"  Std dev: {branch_level[branch_level['switched_branch']==0]['change'].std():.2f}")

# Confidence interval for the DiD estimate
switched_change = branch_level[branch_level['switched_branch']==1]['change']
control_change = branch_level[branch_level['switched_branch']==0]['change']

did = switched_change.mean() - control_change.mean()
# Standard error using bootstrap or simple formula
se_did = np.sqrt(switched_change.var()/len(switched_change) + control_change.var()/len(control_change))
ci_lower = did - 1.96 * se_did
ci_upper = did + 1.96 * se_did

print(f"\n=== DiD Estimate with 95% CI ===")
print(f"Estimate: {did:.2f}")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
