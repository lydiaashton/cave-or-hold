import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run Difference-in-Differences regression
# Basic DiD model
model_basic = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Basic DiD Model (with cluster-robust SEs):")
print(model_basic.summary())

# DiD with controls
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Model with Controls (with cluster-robust SEs):")
print(model_controls.summary())

# Check parallel trends - event study style
# Create week dummies interacted with treatment
df['week_centered'] = df['week'] - 14  # Center around switch week

# Create pre/post indicators for each week relative to switch
for w in range(6, 23):
    df[f'week_{w}'] = (df['week'] == w).astype(int)
    
# Event study regression (omit week 14 as reference)
event_study_vars = '+'.join([f'week_{w}*switched_branch' for w in range(6, 23) if w != 14])
event_study_formula = f'service_score ~ switched_branch + after_switch + {event_study_vars}'

# This might be too many variables, let's do a simpler approach
print("\n\nParallel Trends Check - Pre-period trends:")
pre_data = df[df['week'] < 15]
pre_trend = ols('service_score ~ week * switched_branch', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print(pre_trend.summary())

    label = 'Control (Old Procedure)' if group == 0 else 'Treatment (New Procedure)'
    color = 'blue' if group == 0 else 'red'
    plt.plot(group_data['week'], group_data['service_score'], 
             marker='o', linewidth=2, markersize=6, label=label, color=color)

plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Score Trends: Treatment vs Control Groups\n(Library Branch Procedure Change)', fontsize=14, fontweight='bold')
plt.legend(loc='best', fontsize=10)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.ylim(40, 65)

# Add annotations
plt.annotate('Pre-Switch\nPeriod', xy=(10, 62), fontsize=10, alpha=0.7)
plt.annotate('Post-Switch\nPeriod', xy=(18, 62), fontsize=10, alpha=0.7)

print("\n" + "="*70)
print("SUMMARY STATISTICS")
print("="*70)

summary_table = df.groupby(['switched_branch', 'after_switch']).agg({
    'service_score': ['mean', 'std', 'count'],
    'staffing_index': 'mean',
    'visits_index': 'mean'
}).round(2)
print(summary_table)

# Calculate the DiD estimate manually
control_pre = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_post = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treatment_pre = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_post = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

did_estimate = (treatment_post - treatment_pre) - (control_post - control_pre)

print(f"\nDifference-in-Differences Calculation:")
print(f"  Control group change: {control_post - control_pre:.2f}")
print(f"  Treatment group change: {treatment_post - treatment_pre:.2f}")
print(f"  DiD Estimate (Treatment Effect): {did_estimate:.2f} points")

# Effect size relative to baseline
baseline = df['service_score'].mean()
print(f"\nBaseline average service score: {baseline:.2f}")
print(f"Estimated effect as % of baseline: {(did_estimate/baseline)*100:.1f}%")

import pandas as pd

    axes[0].fill_between(group_data['week'], 
                         group_data['mean'] - 1.96*group_data['se'],
                         group_data['mean'] + 1.96*group_data['se'],
                         alpha=0.2, color=color)

axes[0].axvline(x=14.5, color='gray', linestyle='--', linewidth=2)

pre_data = df[df['after_switch']==0]
post_data = df[df['after_switch']==1]

labels_for_box = []
for group in [0, 1]:
    for period in [0, 1]:

        data_for_box.append(subset['service_score'].values)
        period_label = 'Pre' if period == 0 else 'Post'
        group_label = 'Control' if group == 0 else 'Treatment'
        labels_for_box.append(f'{group_label}\n{period_label}')

bp = axes[1].boxplot(data_for_box, labels=labels_for_box, patch_artist=True)
colors = ['lightblue', 'darkblue', 'lightcoral', 'darkred']
for patch, color in zip(bp['boxes'], colors):
    patch.set_facecolor(color)

axes[1].set_ylabel('Service Score', fontsize=11)
axes[1].set_title('Distribution of Service Scores\nby Group and Period', fontsize=12, fontweight='bold')
axes[1].grid(True, alpha=0.3, axis='y')
axes[1].set_ylim(30, 75)

# Add mean markers
for i, data in enumerate(data_for_box):
    axes[1].plot(i+1, np.mean(data), 'D', color='black', markersize=10, label='Mean' if i==0 else '')
