import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Calculate averages for the 2x2 table
print("=== Difference-in-Differences Summary ===\n")

# Before period (after_switch = 0)
before_switched = df[(df['after_switch'] == 0) & (df['switched_branch'] == 1)]['service_score'].mean()
before_control = df[(df['after_switch'] == 0) & (df['switched_branch'] == 0)]['service_score'].mean()

# After period (after_switch = 1)
after_switched = df[(df['after_switch'] == 1) & (df['switched_branch'] == 1)]['service_score'].mean()
after_control = df[(df['after_switch'] == 1) & (df['switched_branch'] == 0)]['service_score'].mean()

print("Service Score Averages:")
print(f"  Switched branches - Before: {before_switched:.2f}, After: {after_switched:.2f}, Change: {after_switched - before_switched:.2f}")
print(f"  Control branches  - Before: {before_control:.2f}, After: {after_control:.2f}, Change: {after_control - before_control:.2f}")
print(f"\n  DiD Estimate (treatment effect): {(after_switched - before_switched) - (after_control - before_control):.2f}")

# Run regression with standard errors clustered by branch
print("\n=== Regression Analysis ===\n")

# Simple OLS first
model = ols('service_score ~ treated + post + DiD', data=df).fit()
print("Simple OLS (no clustering):")
print(model.summary().tables[1])

# Clustered standard errors by branch
model_clustered = ols('service_score ~ treated + post + DiD', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nWith Clustered Standard Errors (by branch):")
print(model_clustered.summary().tables[1])

import pandas as pd
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate weekly averages and differences
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_avg.columns = ['Control', 'Switched']

# Pre-post difference for each group
before_switched = df[(df['after_switch'] == 0) & (df['switched_branch'] == 1)]['service_score'].mean()
after_switched = df[(df['after_switch'] == 1) & (df['switched_branch'] == 1)]['service_score'].mean()
before_control = df[(df['after_switch'] == 0) & (df['switched_branch'] == 0)]['service_score'].mean()
after_control = df[(df['after_switch'] == 1) & (df['switched_branch'] == 0)]['service_score'].mean()

# Create figure with two panels
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 5))

# Panel 1: Time series
ax1.plot(weekly_avg.index, weekly_avg['Control'], 'o-', label='Control (old procedure)', linewidth=2, markersize=6, color='#1f77b4')
ax1.plot(weekly_avg.index, weekly_avg['Switched'], 's-', label='Switched (new procedure)', linewidth=2, markersize=6, color='#ff7f0e')
ax1.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch (week 15)')

ax1.set_ylabel('Average Service Score', fontsize=12)
ax1.set_title('Service Scores Over Time', fontsize=14, fontweight='bold')
ax1.legend(loc='lower right')
ax1.grid(True, alpha=0.3)
ax1.set_xticks(range(6, 23))

# Panel 2: DiD visualization
groups = ['Before', 'After']
control_scores = [before_control, after_control]
switched_scores = [before_switched, after_switched]

x = np.arange(len(groups))
width = 0.35

bars1 = ax2.bar(x - width/2, control_scores, width, label='Control (old)', color='#1f77b4')
bars2 = ax2.bar(x + width/2, switched_scores, width, label='Switched (new)', color='#ff7f0e')

ax2.set_ylabel('Average Service Score', fontsize=12)
ax2.set_title('Difference-in-Differences Summary', fontsize=14, fontweight='bold')
ax2.set_xticks(x)
ax2.set_xticklabels(groups)
ax2.legend()
ax2.grid(True, alpha=0.3, axis='y')

# Add value labels
for bars in [bars1, bars2]:
    for bar in bars:
