import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study - check parallel trends
# Create relative week indicators (weeks before and after switch)
df['relative_week'] = df['week'] - 15  # Week 15 is the switch, so this is 0
df['relative_week'] = df['relative_week'].clip(-9, 7)  # Cap at -9 (week 6) and 7 (week 22)

# Create dummies for each relative week (omit week -1 as reference)
for rw in range(-9, 8):
    if rw != -1:  # Omit -1 as reference
        df[f'rel_week_{rw}'] = (df['relative_week'] == rw).astype(int)

# Run event study regression
event_vars = [f'rel_week_{rw}' for rw in range(-9, 8) if rw != -1]
event_formula = 'service_score ~ switched_branch + ' + ' + '.join(event_vars) + ' + switched_branch * (' + ' + '.join(event_vars) + ')'

# Simpler approach: interact treatment with each relative week
df['treated'] = df['switched_branch']
for rw in range(-9, 8):
    if rw != -1:
        df[f'treated_rel_{rw}'] = df['treated'] * (df['relative_week'] == rw).astype(int)

event_vars_interact = [f'treated_rel_{rw}' for rw in range(-9, 8) if rw != -1]
event_formula = 'service_score ~ switched_branch + C(week) + ' + ' + '.join(event_vars_interact)

model_event = ols(event_formula, data=df).fit()
print("Event Study Regression (cluster-robust SE):")
clustered_event = model_event.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Extract coefficients for the interaction terms
coef_df = pd.DataFrame({
    'relative_week': [rw for rw in range(-9, 8) if rw != -1],
    'coef': [clustered_event.params[f'treated_rel_{rw}'] for rw in range(-9, 8) if rw != -1],
    'se': [clustered_event.bse[f'treated_rel_{rw}'] for rw in range(-9, 8) if rw != -1],
    'pvalue': [clustered_event.pvalues[f'treated_rel_{rw}'] for rw in range(-9, 8) if rw != -1]
})
coef_df['ci_lower'] = coef_df['coef'] - 1.96 * coef_df['se']
coef_df['ci_upper'] = coef_df['coef'] + 1.96 * coef_df['se']
print("\nEvent Study Coefficients:")
print(coef_df)

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
ax.errorbar(coef_df['relative_week'], coef_df['coef'], 
            yerr=[coef_df['coef'] - coef_df['ci_lower'], coef_df['ci_upper'] - coef_df['coef']],
            fmt='o', capsize=5, markersize=8, linewidth=2)
ax.axhline(y=0, color='red', linestyle='--', linewidth=2)
ax.axvline(x=-0.5, color='green', linestyle=':', linewidth=2, label='Switch Point')
ax.set_xlabel('Relative Week (0 = Week 15, switch week)', fontsize=12)
ax.set_ylabel('Effect on Service Score (vs Control)', fontsize=12)
ax.set_title('Event Study: Treatment Effect by Week', fontsize=14)
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as event_study.png")

clustered_controls = model_controls.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(clustered_controls.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Simpler event study approach
# Create relative week and rename to avoid negative signs
df['rel_week'] = df['week'] - 14  # Week 14 is the last pre-switch week
# Now week 14 = 0 (reference), week 15 = 1, etc.

# Create interaction terms manually
pre_weeks = [6, 7, 8, 9, 10, 11, 12, 13]  # Weeks before switch
post_weeks = [15, 16, 17, 18, 19, 20, 21, 22]  # Weeks after switch

# Build regression matrix manually
X_cols = ['switched_branch']
for w in pre_weeks + post_weeks:
    X_cols.append(f'week_{w}')
    X_cols.append(f'switched_week_{w}')

# Create the variables
for w in pre_weeks + post_weeks:
    df[f'week_{w}'] = (df['week'] == w).astype(int)
    df[f'switched_week_{w}'] = ((df['switched_branch'] == 1) & (df['week'] == w)).astype(int)

# Reference: week 14 (last pre-switch week)
# Run regression
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'week_{w}' for w in pre_weeks + post_weeks]) + ' + ' + ' + '.join([f'switched_week_{w}' for w in pre_weeks + post_weeks])
model = ols(formula, data=df).fit()
clustered = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Extract the interaction coefficients (treatment effects by week)
results = []
for w in pre_weeks + post_weeks:
    coef = clustered.params[f'switched_week_{w}']
    se = clustered.bse[f'switched_week_{w}']
    pval = clustered.pvalues[f'switched_week_{w}']
    results.append({
        'week': w,
        'relative_week': w - 14,  # Week 14 = 0
        'coef': coef,
        'se': se,
        'pvalue': pval,
        'ci_lower': coef - 1.96 * se,
        'ci_upper': coef + 1.96 * se,
        'period': 'Pre' if w < 15 else 'Post'
    })

results_df = pd.DataFrame(results)
print("Event Study Results:")
print(results_df)

# Plot event study
fig, ax = plt.subplots(figsize=(12, 6))
pre = results_df[results_df['period'] == 'Pre']
post = results_df[results_df['period'] == 'Post']

ax.errorbar(pre['relative_week'], pre['coef'], 
            yerr=[pre['coef'] - pre['ci_lower'], pre['ci_upper'] - pre['coef']],
            fmt='o', color='blue', capsize=5, markersize=8, linewidth=2, label='Pre-switch')
ax.errorbar(post['relative_week'], post['coef'], 
            yerr=[post['coef'] - post['ci_lower'], post['ci_upper'] - post['coef']],
            fmt='s', color='green', capsize=5, markersize=8, linewidth=2, label='Post-switch')
ax.axhline(y=0, color='red', linestyle='--', linewidth=2)
ax.axvline(x=0.5, color='gray', linestyle=':', linewidth=2, label='Switch Point')
ax.set_xlabel('Relative Week (0 = Week 14, last pre-switch week)', fontsize=12)
ax.set_ylabel('Treatment Effect on Service Score', fontsize=12)
ax.set_title('Event Study: Treatment Effect by Week (vs Control, Week 14 reference)', fontsize=14)
ax.legend(fontsize=11)
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as event_study.png")

# Check pre-trends significance
pre_trend_test = results_df[results_df['period'] == 'Pre']
print(f"\nPre-trend coefficients (should be ~0 if parallel trends hold):")

print(f"\nAre any pre-trends significant at 5% level? {(pre_trend_test['pvalue'] < 0.05).sum()}")
