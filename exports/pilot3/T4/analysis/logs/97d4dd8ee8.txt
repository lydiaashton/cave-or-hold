import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run the DiD regression
# service_score = β0 + β1*(switched_branch) + β2*(after_switch) + β3*(treatment) + ε
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()

print("Difference-in-Differences Regression Results:")
print("="*60)
print(model.summary())

# Also run with cluster-robust standard errors at branch level
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n\nRegression with Cluster-Robust Standard Errors (by branch):")

pre_model = ols('service_score ~ switched_branch + C(week)', data=pre_data).fit()

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Let's visualize the trends
import matplotlib
matplotlib.use('Agg')

# Create a plot of service scores over time by group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

fig, ax = plt.subplots(figsize=(10, 6))

for group in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == group]
    label = 'Control (Old Procedure)' if group == 0 else 'Treatment (New Procedure)'
    ax.plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2)

ax.axvline(x=14.5, color='red', linestyle='--', label='Switch Point (Week 15)')
ax.set_xlabel('Week')
ax.set_ylabel('Average Service Score')
ax.set_title('Service Scores Over Time by Group')
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("Plot saved as service_scores_trend.png")

# Let's also look at the distribution of changes
print("\n\nService score changes by branch (after vs before):")
before = df[df['week'] < 15].groupby('branch_id')['service_score'].mean()
after = df[df['week'] >= 15].groupby('branch_id')['service_score'].mean()
changes = after - before

# Merge with switched_branch info
branch_info = df.groupby('branch_id')['switched_branch'].first()
changes_df = pd.DataFrame({'change': changes, 'switched': branch_info})

print("\nTreatment branches (switched=1):")
print(changes_df[changes_df['switched'] == 1]['change'].describe())

print("\nControl branches (switched=0):")
print(changes_df[changes_df['switched'] == 0]['change'].describe())

# T-test on the changes
from scipy import stats
treatment_changes = changes_df[changes_df['switched'] == 1]['change']
control_changes = changes_df[changes_df['switched'] == 0]['change']

t_stat, p_val = stats.ttest_ind(treatment_changes, control_changes)
print(f"\n\nTwo-sample t-test on branch-level changes:")
print(f"t-statistic: {t_stat:.3f}, p-value: {p_val:.3f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create week-specific treatment indicators for post-period
df['post_week'] = df.apply(lambda row: row['week'] if row['after_switch'] == 1 else np.nan, axis=1)

# Let's do an event study - treatment effect by week relative to switch
# Week 14 is the last pre-period week, so we'll use that as reference
df['relative_week'] = df['week'] - 14  # Week 14 = 0, Week 15 = 1, etc.

# Create indicators for each post-period week interacted with treatment
for w in range(1, 9):  # weeks 15-22 relative to week 14
    df[f'treat_week{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (controlling for week fixed effects and branch group)
# We'll use week 14 (relative_week=0) as the baseline
event_study_vars = ' + '.join([f'treat_week{w}' for w in range(1, 9)])
formula = f'service_score ~ switched_branch + C(week) + {event_study_vars}'

event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Regression (Treatment Effect by Week):")
print("="*60)
print(event_model.summary())

# Extract the treatment effects by week
print("\n\nTreatment Effects by Post-Switch Week:")
for w in range(1, 9):
    coef = event_model.params[f'treat_week{w}']
    se = event_model.bse[f'treat_week{w}']
    pval = event_model.pvalues[f'treat_week{w}']
    print(f"Week {14+w}: coef={coef:.2f}, SE={se:.2f}, p={pval:.3f}")
