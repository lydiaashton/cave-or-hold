pre_period = df[df['week'] < 15]
print("\n\nPre-period (weeks 6-14) average service scores:")
pre_avg = pre_period.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print(pre_avg)

# Post-period average service scores (weeks 15-22)
post_period = df[df['week'] >= 15]
print("\n\nPost-period (weeks 15-22) average service scores:")
post_avg = post_period.groupby('switched_branch')['service_score'].agg(['mean', 'std', 'count'])
print(post_avg)

# Simple difference-in-differences calculation
switched_pre = pre_period[pre_period['switched_branch']==1]['service_score'].mean()
control_pre = pre_period[pre_period['switched_branch']==0]['service_score'].mean()
switched_post = post_period[post_period['switched_branch']==1]['service_score'].mean()
control_post = post_period[post_period['switched_branch']==0]['service_score'].mean()

print("\n\n=== Simple Difference-in-Differences ===")
print(f"Switched branches: Pre={switched_pre:.2f}, Post={switched_post:.2f}, Change={switched_post-switched_pre:.2f}")
print(f"Control branches:  Pre={control_pre:.2f}, Post={control_post:.2f}, Change={control_post-control_pre:.2f}")
did_estimate = (switched_post - switched_pre) - (control_post - control_pre)
print(f"\nDiD Estimate (treatment effect): {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch'] * df['after_switch']

# Main DiD regression with cluster-robust standard errors
did_model = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
print("=== Basic DiD Regression ===")
print(did_model.summary())

# Now with cluster-robust standard errors (clustered by branch)
did_model_robust = did_model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print("\n=== DiD with Cluster-Robust SE (by branch) ===")
print(did_model_robust.summary())

# Add controls for staffing and visits
did_model_controls = ols('service_score ~ switched_branch + after_switch + treated + staffing_index + visits_index', data=df).fit()
did_model_controls_robust = did_model_controls.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print("\n=== DiD with Controls (Cluster-Robust SE) ===")
print(did_model_controls_robust.summary())

# Extract key coefficient
treatment_effect = did_model_robust.params['treated']
treatment_se = did_model_robust.bse['treated']
treatment_p = did_model_robust.pvalues['treated']
treatment_ci_low = did_model_robust.conf_int().loc['treated', 0]
treatment_ci_high = did_model_robust.conf_int().loc['treated', 1]

print(f"\n\n=== KEY RESULT ===")
print(f"Treatment Effect (DiD estimate): {treatment_effect:.3f}")
print(f"Standard Error: {treatment_se:.3f}")
print(f"95% CI: [{treatment_ci_low:.3f}, {treatment_ci_high:.3f}]")
print(f"P-value: {treatment_p:.4f}")
print(f"Statistically significant at 5% level: {treatment_p < 0.05}")
