import pandas as pd
import numpy as np
import statsmodels.api as sm

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['switched_after'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimator - the effect of the new procedure

X = df[['switched_branch', 'after_switch', 'switched_after']]
X = sm.add_constant(X)
y = df['service_score']

model = sm.OLS(y, X).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Calculate the DiD effect manually to verify
before_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

did_effect = (after_switched - before_switched) - (after_control - before_control)
print(f"\n=== Manual DiD Calculation ===")
print(f"Switched branches: before={before_switched:.2f}, after={after_switched:.2f}, change={after_switched - before_switched:.2f}")
print(f"Control branches: before={before_control:.2f}, after={after_control:.2f}, change={after_control - before_control:.2f}")
print(f"DiD effect (new procedure impact): {did_effect:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['switched_after'] = df['switched_branch'] * df['after_switch']

# Add branch fixed effects
# Create dummy variables for each branch
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df, branch_dummies], axis=1)

# Regression with branch fixed effects
fe_cols = [col for col in df_with_fe.columns if col.startswith('branch')]
X_fe = df_with_fe[['switched_branch', 'after_switch', 'switched_after'] + fe_cols]
X_fe = sm.add_constant(X_fe, has_constant='add')
y_fe = df_with_fe['service_score']

model_fe = sm.OLS(y_fe, X_fe).fit()
print("=== Regression with Branch Fixed Effects ===")
print(f"R-squared: {model_fe.rsquared:.4f}")
print(f"\nCoefficients of interest:")
print(f"  switched_branch: {model_fe.params['switched_branch']:.4f} (p={model_fe.pvalues['switched_branch']:.4f})")
print(f"  after_switch: {model_fe.params['after_switch']:.4f} (p={model_fe.pvalues['after_switch']:.4f})")
print(f"  switched_after (DiD effect): {model_fe.params['switched_after']:.4f} (p={model_fe.pvalues['switched_after']:.4f})")

# Cluster-robust standard errors by branch
model_fe_clustered = model_fe.get_robustcov_results(cov_type='cluster', groups=df_with_fe['branch_id'])
print(f"\n=== With Cluster-Robust SE (by branch) ===")
print(f"  switched_after (DiD effect): {model_fe_clustered.params['switched_after']:.4f}")
print(f"  Std Error: {model_fe_clustered.bse['switched_after']:.4f}")
print(f"  p-value: {model_fe_clustered.pvalues['switched_after']:.4f}")
print(f"  95% CI: [{model_fe_clustered.conf_int().loc['switched_after'][0]:.4f}, {model_fe_clustered.conf_int().loc['switched_after'][1]:.4f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['switched_after'] = df['switched_branch'] * df['after_switch']

# Model with covariates
X = df[['switched_branch', 'after_switch', 'switched_after', 'staffing_index', 'visits_index']]
X = sm.add_constant(X)
y = df['service_score']

model_cov = sm.OLS(y, X).fit()
model_cov_clustered = model_cov.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("=== DiD with Covariates (Cluster-Robust SE) ===")
coef = float(model_cov_clustered.params[4])  # switched_after is now at position 4
se = float(model_cov_clustered.bse[4])
pval = float(model_cov_clustered.pvalues[4])
ci_lower = coef - 1.96 * se
ci_upper = coef + 1.96 * se

print(f"  switched_after (DiD effect): {coef:.4f}")
print(f"  Std Error: {se:.4f}")
print(f"  p-value: {pval:.4f}")
print(f"  95% CI: [{ci_lower:.4f}, {ci_upper:.4f}]")

# Create visualization
fig, ax = plt.subplots(figsize=(10, 6))

# Plot weekly averages for both groups
weekly = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
control = weekly[weekly['switched_branch'] == 0]
switched = weekly[weekly['switched_branch'] == 1]

ax.plot(control['week'], control['service_score'], 'o-', label='Control (Old Procedure)', linewidth=2, markersize=8)
