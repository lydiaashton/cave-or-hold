import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

# Summary table: average service scores by group and period
summary_table = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count']).unstack(level=0)
print("Average Service Scores by Group and Period:")
print(summary_table)

# Calculate DiD manually
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treated_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treated_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print("\n=== Difference-in-Differences Calculation ===")
print(f"Control (non-switched) before: {control_before:.2f}")
print(f"Control (non-switched) after: {control_after:.2f}")
print(f"Control change: {control_after - control_before:.2f}")
print()
print(f"Treated (switched) before: {treated_before:.2f}")
print(f"Treated (switched) after: {treated_after:.2f}")
print(f"Treated change: {treated_after - treated_before:.2f}")
print()
did_effect = (treated_after - treated_before) - (control_after - control_before)
print(f"DiD Effect (Treated change - Control change): {did_effect:.2f}")

# Run regression with branch fixed effects
print("\n=== Regression Analysis ===")

# Model 1: Simple DiD without fixed effects
model1 = ols('service_score ~ treated + post + treated_post', data=df).fit()
print("\nModel 1: Simple DiD (no fixed effects)")
print(model1.summary().tables[1])

# Model 2: With branch fixed effects
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Build formula for model with branch fixed effects
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_')]
formula = 'service_score ~ treated + post + treated_post + ' + ' + '.join(branch_cols)
model2 = ols(formula, data=df_with_dummies).fit()
print("\nModel 2: DiD with Branch Fixed Effects")
print(model2.summary().tables[1].loc[['treated', 'post', 'treated_post']])

# Model 3: With controls (staffing and visits)
model3 = ols('service_score ~ treated + post + treated_post + staffing_index + visits_index', data=df).fit()
print("\nModel 3: DiD with Controls")
print(model3.summary().tables[1].loc[['treated', 'post', 'treated_post', 'staffing_index', 'visits_index']])

# Model 4: With branch fixed effects AND controls
formula4 = 'service_score ~ treated + post + treated_post + staffing_index + visits_index + ' + ' + '.join(branch_cols)
model4 = ols(formula4, data=df_with_dummies).fit()
print("\nModel 4: DiD with Branch Fixed Effects and Controls")
print(model4.summary().tables[1].loc[['treated', 'post', 'treated_post', 'staffing_index', 'visits_index']])

import numpy as np

from statsmodels.formula.api import ols
import warnings

df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

# Model 1: Simple DiD without fixed effects
model1 = ols('service_score ~ treated + post + treated_post', data=df).fit()
print("Model 1: Simple DiD (no fixed effects)")
print(model1.summary())

# Model 2: With controls (staffing and visits)
model2 = ols('service_score ~ treated + post + treated_post + staffing_index + visits_index', data=df).fit()
print("\n\nModel 2: DiD with Controls")
print(model2.summary())

# Model 3: With branch fixed effects (using C() notation)
model3 = ols('service_score ~ treated + post + treated_post + C(branch_id)', data=df).fit()
print("\n\nModel 3: DiD with Branch Fixed Effects")
print(model3.summary())

# Model 4: With branch fixed effects AND controls
model4 = ols('service_score ~ treated + post + treated_post + staffing_index + visits_index + C(branch_id)', data=df).fit()

df['post'] = df['after_switch']
df['treated_post'] = df['treated'] * df['post']

print("=== Simple DiD Model (no fixed effects) ===")
model1 = ols('service_score ~ treated + post + treated_post', data=df).fit()
print(model1.summary())

print("\n=== DiD with Controls (no fixed effects) ===")
model2 = ols('service_score ~ treated + post + treated_post + staffing_index + visits_index', data=df).fit()
print(model2.summary())

# Try clustering by branch without fixed effects
print("\n=== DiD with Clustered SE (no fixed effects) ===")
model3 = ols('service_score ~ treated + post + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model3.summary())

print("\n=== DiD with Controls and Clustered SE ===")
model4 = ols('service_score ~ treated + post + treated_post + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model4.summary())
