import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Run Difference-in-Differences regression
# service_score = β0 + β1*treated + β2*post + β3*(treated*post) + ε
# β3 is the DiD estimator - the effect of the new procedure

model = ols('service_score ~ treated + post + DiD', data=df).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Also run with branch fixed effects to control for time-invariant branch characteristics
df['branch_id'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ treated + post + DiD + C(branch_id)', data=df).fit()
print("\n=== DiD with Branch Fixed Effects ===")
print(model_fe.summary())

# Let's also check week-by-week trends
print("\n=== Weekly Average Service Scores by Group ===")
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print(weekly)

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Run DiD with branch fixed effects (drop 'treated' since it's collinear with branch FE)
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ post + DiD + C(branch_id_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== DiD with Branch Fixed Effects (Clustered SEs) ===")
print(model_fe.summary())

# Also run with week fixed effects to control for time trends
df['week_str'] = df['week'].astype(str)
model_twfe = ols('service_score ~ DiD + C(branch_id_str) + C(week_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Two-Way Fixed Effects (Branch + Week) ===")
print(model_twfe.summary())

# Calculate pre-post changes for each group
print("\n=== Pre-Post Changes by Group ===")
pre_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
post_switched = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()
pre_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
post_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"Switched branches: {pre_switched:.2f} -> {post_switched:.2f} (change: {post_switched - pre_switched:.2f})")
print(f"Control branches: {pre_control:.2f} -> {post_control:.2f} (change: {post_control - pre_control:.2f})")
