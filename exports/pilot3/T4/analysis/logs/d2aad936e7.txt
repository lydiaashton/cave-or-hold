import pandas as pd
import numpy as np
import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model with branch and week fixed effects, clustered SEs at branch level
# Using C() for categorical variables
model3 = smf.ols('service_score ~ treatment + staffing_index + visits_index + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== DiD with Branch & Week Fixed Effects (Clustered SEs) ===")
print(model3.summary())

# Extract just the treatment coefficient
print("\n=== Treatment Effect Summary ===")

treatment_se = model3.bse['treatment']
treatment_p = model3.pvalues['treatment']

print(f"Standard error: {treatment_se:.3f}")
print(f"P-value: {treatment_p:.4f}")
print(f"95% CI: [{treatment_coef - 1.96*treatment_se:.3f}, {treatment_coef + 1.96*treatment_se:.3f}]")

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Robustness check: exclude extreme outliers
# Check if results hold without top/bottom 5% of service scores
q05 = df['service_score'].quantile(0.05)
q95 = df['service_score'].quantile(0.95)
df_trimmed = df[(df['service_score'] >= q05) & (df['service_score'] <= q95)].copy()

model_trimmed = smf.ols('service_score ~ treatment + staffing_index + visits_index + C(branch_id) + C(week)', 
                        data=df_trimmed).fit(cov_type='cluster', cov_kwds={'groups': df_trimmed['branch_id']})
print("=== Robustness Check (trimmed 5% tails) ===")
print(f"Treatment effect: {model_trimmed.params['treatment']:.3f}")
print(f"P-value: {model_trimmed.pvalues['treatment']:.4f}")

# Also check with just the main DiD variables (no controls)
model_no_controls = smf.ols('service_score ~ treatment + C(branch_id) + C(week)', 
                            data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== Without Control Variables ===")
print(f"Treatment effect: {model_no_controls.params['treatment']:.3f}")
print(f"P-value: {model_no_controls.pvalues['treatment']:.4f}")
