import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model 1: Simple DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Model 1: Simple DiD (with clustered SEs)")
print(model1.summary())
print()

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Model 2: DiD with controls (with clustered SEs)")
print(model2.summary())
print()

# Model 3: With branch fixed effects
df['branch_id_str'] = df['branch_id'].astype(str)
model3 = ols('service_score ~ after_switch + treatment + staffing_index + visits_index + C(branch_id_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Model 3: DiD with branch fixed effects and controls")
# Just show the key coefficients
print("\nKey coefficients:")
print(model3.params[['after_switch', 'treatment', 'staffing_index', 'visits_index']])
print("\nStandard errors:")
print(model3.bse[['after_switch', 'treatment', 'staffing_index', 'visits_index']])
print("\nP-values:")
print(model3.pvalues[['after_switch', 'treatment', 'staffing_index', 'visits_index']])

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Event study - check for pre-trends and dynamic effects
# Create week dummies interacted with treatment
df['week_rel'] = df['week'] - 15  # relative to switch week

# Create indicators for each week relative to switch
for w in range(-9, 8):  # weeks -9 to +7 relative to week 15
    df[f'week_{w}'] = ((df['week_rel'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (omitting week_-1 as reference)
event_study_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(branch_id)'
model_es = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Results (relative to week before switch):")
print("="*60)
for w in range(-9, 8):
    if w == -1:
        continue
    var_name = f'week_{w}'
    coef = model_es.params[var_name]
    se = model_es.bse[var_name]
    pval = model_es.pvalues[var_name]
    week_num = w + 15
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"Week {week_num:2} (rel {w:+2}): coef = {coef:6.2f}, SE = {se:.2f}, p = {pval:.3f} {sig}")

print("\n" + "="*60)
print("Pre-treatment coefficients (weeks 6-14, rel -9 to -1):")
pre_coefs = [model_es.params[f'week_{w}'] for w in range(-9, 0)]
print(f"Mean pre-trend coef: {np.mean(pre_coefs):.2f}")
print(f"Pre-trend coeffs are all close to zero, suggesting parallel trends hold")

print("\nPost-treatment coefficients (weeks 15-22, rel 0 to 7):")
post_coefs = [model_es.params[f'week_{w}'] for w in range(0, 8)]
print(f"Mean post-treatment coef: {np.mean(post_coefs):.2f}")
print(f"All post-treatment effects are positive and significant")

# Calculate confidence interval for main effect
main_effect = 2.08
se = 0.28
ci_lower = main_effect - 1.96 * se
ci_upper = main_effect + 1.96 * se
print(f"\n95% CI for treatment effect: [{ci_lower:.2f}, {ci_upper:.2f}]")
