import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated_post'] = df['switched_branch'] * df['after_switch']

# Summary by group and period
print("=== Service Score by Group and Period ===")
summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(summary)

print("\n=== Difference-in-Differences Calculation ===")
# Get means for each cell
means = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean()
print(means)

# DiD estimate
control_pre = means.loc[(0, 0)]
control_post = means.loc[(0, 1)]
treated_pre = means.loc[(1, 0)]
treated_post = means.loc[(1, 1)]

diff_control = control_post - control_pre
diff_treated = treated_post - treated_pre
did_estimate = diff_treated - diff_control

print(f"\nControl group (non-switched): Pre={control_pre:.2f}, Post={control_post:.2f}, Change={diff_control:.2f}")
print(f"Treated group (switched): Pre={treated_pre:.2f}, Post={treated_post:.2f}, Change={diff_treated:.2f}")
print(f"\nDiD Estimate (treatment effect): {did_estimate:.3f}")

# Run regression with standard errors clustered by branch
print("\n=== DiD Regression ===")
model = ols('service_score ~ switched_branch + after_switch + treated_post', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model.summary())
