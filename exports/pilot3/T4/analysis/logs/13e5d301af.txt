import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch*after_switch) + error
# beta3 is the DiD estimator

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Difference-in-Differences Regression Results:")
print(model.summary())

# Also run with robust standard errors clustered by branch
print("\n\nWith clustered standard errors by branch:")
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model_clustered.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate average service score by week and group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_avg.columns = ['week', 'switched_branch', 'avg_score']

# Pivot for easier plotting
pivot = weekly_avg.pivot(index='week', columns='switched_branch', values='avg_score')
pivot.columns = ['Control (No Switch)', 'Treatment (Switched)']

print("Weekly averages by group:")
print(pivot)

# Create the plot
fig, ax = plt.subplots(figsize=(10, 6))

ax.plot(pivot.index, pivot['Control (No Switch)'], 'o-', label='Control (No Switch)', linewidth=2, markersize=8)
ax.plot(pivot.index, pivot['Treatment (Switched)'], 's-', label='Treatment (Switched)', linewidth=2, markersize=8)

# Add vertical line at switch point (week 15)
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch implemented')

ax.set_xlabel('Week', fontsize=12)
ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Scores by Week: Treatment vs Control Groups', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=11)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_scores_trend.png', dpi=150)
print("\nPlot saved as service_scores_trend.png")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Main DiD model
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get the treatment effect and confidence interval
treatment_effect = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pvalue = model.pvalues['treatment']
ci_lower = treatment_effect - 1.96 * treatment_se
ci_upper = treatment_effect + 1.96 * treatment_se

print("=== MAIN RESULTS ===")
print(f"Treatment Effect (DiD estimate): {treatment_effect:.2f} points")
print(f"Standard Error: {treatment_se:.2f}")
print(f"95% Confidence Interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {treatment_pvalue:.3f}")

# Calculate effect size relative to baseline
baseline_avg = df[df['after_switch']==0]['service_score'].mean()
print(f"\nBaseline average service score: {baseline_avg:.2f}")
print(f"Effect as % of baseline: {(treatment_effect/baseline_avg)*100:.1f}%")

# Also check the raw difference-in-differences calculation
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

control_change = control_after - control_before
treatment_change = treatment_after - treatment_before
did_raw = treatment_change - control_change

print("\n=== RAW CALCULATIONS ===")
print(f"Control group: {control_before:.2f} -> {control_after:.2f} (change: {control_change:.2f})")
print(f"Treatment group: {treatment_before:.2f} -> {treatment_after:.2f} (change: {treatment_change:.2f})")
print(f"Difference-in-Differences: {did_raw:.2f}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate average service score by week and group
weekly_avg = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()

# Create the plot
fig, ax = plt.subplots(figsize=(10, 6))

control = weekly_avg[weekly_avg['switched_branch']==0]
treatment = weekly_avg[weekly_avg['switched_branch']==1]

ax.plot(control['week'], control['avg_score'], 'o-', label='Control (No Switch)', linewidth=2, markersize=8, color='blue')
ax.plot(treatment['week'], treatment['avg_score'], 's-', label='Treatment (Switched)', linewidth=2, markersize=8, color='green')

ax.set_ylabel('Average Service Score', fontsize=12)
ax.set_title('Service Scores by Week: Treatment vs Control Groups\n(Difference-in-Differences Analysis)', fontsize=14, fontweight='bold')
ax.legend(loc='best', fontsize=11)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))
ax.set_ylim(44, 58)

plt.tight_layout()
