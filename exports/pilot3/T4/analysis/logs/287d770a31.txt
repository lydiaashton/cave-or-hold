print("=== Model 1: Simple DiD (no controls) ===")

# This is more robust as it controls for time-invariant branch characteristics
print("\n=== Model 3: DiD with branch fixed effects ===")
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
X_with_fe = pd.concat([df[['switched_branch', 'after_switch', 'treatment', 'staffing_index', 'visits_index']], branch_dummies], axis=1)
X_with_fe = sm.add_constant(X_with_fe)
y = df['service_score']

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create relative week variable (weeks relative to switch at week 15)
df['relative_week'] = df['week'] - 15

# Event study: create dummies for each week relative to switch
# Use week -1 as reference (the week before switch)
event_study_df = df.copy()

# Create week dummies, excluding week -1 as reference
weeks = sorted(df['relative_week'].unique())
weeks_to_include = [w for w in weeks if w != -1]

for w in weeks_to_include:
    event_study_df[f'week_{w}'] = (event_study_df['relative_week'] == w).astype(int)

# Interact with treatment
for w in weeks_to_include:
    event_study_df[f'treatment_week_{w}'] = event_study_df['switched_branch'] * event_study_df[f'week_{w}']

# Run event study regression
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'week_{w}' for w in weeks_to_include]) + ' + ' + ' + '.join([f'treatment_week_{w}' for w in weeks_to_include]) + ' + staffing_index + visits_index'

event_model = ols(formula, data=event_study_df).fit()

# Extract treatment effects for each week
treatment_coefs = {}
treatment_ci = {}
for w in weeks_to_include:
    coef_name = f'treatment_week_{w}'
    if coef_name in event_model.params:
        treatment_coefs[w] = event_model.params[coef_name]
        ci = event_model.conf_int().loc[coef_name]
        treatment_ci[w] = (ci[0], ci[1])

# Add reference week
treatment_coefs[-1] = 0
treatment_ci[-1] = (0, 0)
