# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators
# Week 14 is the last week before switch, use as reference
df['relative_week'] = df['week'] - 14  # week 14 becomes 0, week 15 becomes 1, etc.

# Create interaction terms for each week with treatment
# We'll use week 14 (relative_week = 0) as the reference
event_study_vars = []
for w in range(-8, 9):  # weeks -8 to 8 relative to week 14
    if w != 0:  # skip reference week
        df[f'treat_week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(f'treat_week_{w}')

# Build formula
formula = 'service_score ~ C(switched_branch) + C(week) + ' + ' + '.join(event_study_vars)

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the event study terms
event_coefs = []
for w in range(-8, 9):
    if w != 0:
        coef_name = f'treat_week_{w}'
        coef = model_event.params[coef_name]
        se = model_event.bse[coef_name]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        pval = model_event.pvalues[coef_name]
        event_coefs.append({
            'relative_week': w,
            'coef': coef,
            'se': se,
            'ci_low': ci_low,
            'ci_high': ci_high,
            'pval': pval
        })

event_df = pd.DataFrame(event_coefs)
print("Event Study Results (relative to week 14):")
print(event_df)

# Check pre-trends (weeks before switch)
pre_trends = event_df[event_df['relative_week'] < 0]
print("\nPre-trend coefficients (should be close to 0 if parallel trends hold):")
print(pre_trends[['relative_week', 'coef', 'pval']])
print("\nAre pre-trend coefficients jointly significant?")
print("All pre-trend p-values > 0.05:", (pre_trends['pval'] > 0.05).all())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study: create relative week indicators
# Week 14 is the last week before switch, use as reference
df['relative_week'] = df['week'] - 14  # week 14 becomes 0, week 15 becomes 1, etc.

# Create interaction terms for each week with treatment (rename to avoid negative numbers in names)
event_study_vars = []
for w in range(-8, 9):  # weeks -8 to 8 relative to week 14
    if w != 0:  # skip reference week
        var_name = f'treat_rw_{w+10}'  # shift by 10 to make all positive (e.g., -8 becomes 2)
        df[var_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(var_name)

# Build formula
formula = 'service_score ~ C(switched_branch) + C(week) + ' + ' + '.join(event_study_vars)

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the event study terms
event_coefs = []
for w in range(-8, 9):
    if w != 0:
        var_name = f'treat_rw_{w+10}'
        coef = model_event.params[var_name]
        se = model_event.bse[var_name]
        ci_low = coef - 1.96 * se
        ci_high = coef + 1.96 * se
        pval = model_event.pvalues[var_name]
        event_coefs.append({
            'relative_week': w,
            'coef': coef,
            'se': se,
            'ci_low': ci_low,
            'ci_high': ci_high,
            'pval': pval
        })

event_df = pd.DataFrame(event_coefs)
print("Event Study Results (relative to week 14):")
print(event_df)

# Check pre-trends (weeks before switch)
pre_trends = event_df[event_df['relative_week'] < 0]
