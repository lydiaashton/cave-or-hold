import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treated'] = df['switched_branch']
df['post'] = df['after_switch']
df['DiD'] = df['treated'] * df['post']

# Basic DiD calculation - average service scores by group and period
pivot = df.groupby(['treated', 'post'])['service_score'].mean().unstack()
print("Average Service Scores by Group and Period:")
print(pivot)
print()

# Calculate DiD manually
before_control = df[(df['treated'] == 0) & (df['post'] == 0)]['service_score'].mean()
after_control = df[(df['treated'] == 0) & (df['post'] == 1)]['service_score'].mean()
before_treated = df[(df['treated'] == 1) & (df['post'] == 0)]['service_score'].mean()
after_treated = df[(df['treated'] == 1) & (df['post'] == 1)]['service_score'].mean()

print("Before period (weeks 6-14):")
print(f"  Control group (not switched): {before_control:.2f}")
print(f"  Treated group (switched): {before_treated:.2f}")
print(f"  Difference: {before_treated - before_control:.2f}")
print()
print("After period (weeks 15-22):")
print(f"  Control group (not switched): {after_control:.2f}")
print(f"  Treated group (switched): {after_treated:.2f}")
print(f"  Difference: {after_treated - after_control:.2f}")
print()

# DiD estimate
did_estimate = (after_treated - before_treated) - (after_control - before_control)
print(f"Difference-in-Differences Estimate: {did_estimate:.2f}")
print()

# Now run regression with standard errors
model = ols('service_score ~ treated + post + DiD', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Regression Results (clustered by branch):")

import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# Distribution of branch-level effects
branch_data = df.groupby('branch_id').agg({
    'switched_branch': 'first',
    'service_score': lambda x: x[x <= 14].mean() if len(x[x <= 14]) > 0 else np.nan
}).reset_index()
branch_data.columns = ['branch_id', 'switched', 'pre_avg']

branch_post = df.groupby('branch_id').apply(
    lambda x: x[x['week'] >= 15]['service_score'].mean() if len(x[x['week'] >= 15]) > 0 else np.nan
).reset_index()
branch_post.columns = ['branch_id', 'post_avg']

branch_data = branch_data.merge(branch_post, on='branch_id')
branch_data['change'] = branch_data['post_avg'] - branch_data['pre_avg']

# How many branches improved vs declined?
switched_branches = branch_data[branch_data['switched'] == 1]
not_switched_branches = branch_data[branch_data['switched'] == 0]

print("Distribution of branch-level changes:")
print(f"\nSwitched branches (n={len(switched_branches)}):")
print(f"  Mean change: {switched_branches['change'].mean():.2f}")
print(f"  Std dev: {switched_branches['change'].std():.2f}")
print(f"  % with positive change: {(switched_branches['change'] > 0).mean()*100:.1f}%")
print(f"  % with negative change: {(switched_branches['change'] < 0).mean()*100:.1f}%")

print(f"\nNon-switched branches (n={len(not_switched_branches)}):")
print(f"  Mean change: {not_switched_branches['change'].mean():.2f}")
print(f"  Std dev: {not_switched_branches['change'].std():.2f}")
print(f"  % with positive change: {(not_switched_branches['change'] > 0).mean()*100:.1f}%")
print(f"  % with negative change: {(not_switched_branches['change'] < 0).mean()*100:.1f}%")

# Statistical test on branch-level changes
t_stat, p_val = stats.ttest_ind(switched_branches['change'].dropna(), not_switched_branches['change'].dropna())
print(f"\nTwo-sample t-test on branch-level changes:")
print(f"  t-statistic: {t_stat:.2f}")
print(f"  p-value: {p_val:.2e}")

# Confidence interval for the effect
switched_mean = switched_branches['change'].mean()
switched_std = switched_branches['change'].std()
switched_n = len(switched_branches)
not_switched_mean = not_switched_branches['change'].mean()
not_switched_std = not_switched_branches['change'].std()
not_switched_n = len(not_switched_branches)

# Standard error of difference
se_diff = np.sqrt(switched_std**2/switched_n + not_switched_std**2/not_switched_n)
ci_lower = (switched_mean - not_switched_mean) - 1.96 * se_diff
ci_upper = (switched_mean - not_switched_mean) + 1.96 * se_diff
print(f"\n95% CI for difference in changes: [{ci_lower:.2f}, {ci_upper:.2f}]")

import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('branches.csv')

# Calculate pre and post averages properly for each branch
def calc_pre_post(group):
    pre = group[group['week'] <= 14]['service_score'].mean()
    post = group[group['week'] >= 15]['service_score'].mean()
    switched = group['switched_branch'].iloc[0]
    return pd.Series({
        'pre_avg': pre,
        'post_avg': post,
        'switched': switched,
        'change': post - pre
    })

branch_stats = df.groupby('branch_id').apply(calc_pre_post).reset_index()

print("Branch-level statistics:")
print(branch_stats.describe())
print()

switched_branches = branch_stats[branch_stats['switched'] == 1]
not_switched_branches = branch_stats[branch_stats['switched'] == 0]

print("Distribution of branch-level changes:")
print(f"\nSwitched branches (n={len(switched_branches)}):")
print(f"  Mean change: {switched_branches['change'].mean():.2f}")
print(f"  Std dev: {switched_branches['change'].std():.2f}")
print(f"  % with positive change: {(switched_branches['change'] > 0).mean()*100:.1f}%")
print(f"  % with negative change: {(switched_branches['change'] < 0).mean()*100:.1f}%")

print(f"\nNon-switched branches (n={len(not_switched_branches)}):")
print(f"  Mean change: {not_switched_branches['change'].mean():.2f}")
print(f"  Std dev: {not_switched_branches['change'].std():.2f}")
print(f"  % with positive change: {(not_switched_branches['change'] > 0).mean()*100:.1f}%")
print(f"  % with negative change: {(not_switched_branches['change'] < 0).mean()*100:.1f}%")

# Statistical test on branch-level changes
t_stat, p_val = stats.ttest_ind(switched_branches['change'].dropna(), not_switched_branches['change'].dropna())
print(f"\nTwo-sample t-test on branch-level changes:")

print(f"\n95% CI for difference in changes: [{ci_lower:.2f}, {ci_upper:.2f}]")
