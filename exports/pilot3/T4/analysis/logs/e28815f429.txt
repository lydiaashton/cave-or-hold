import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treated'] = df['switched_branch'] * df['after_switch']

# Summary: How many branches switched vs didn't?
print("=== Branch Composition ===")
branch_summary = df.groupby('branch_id')['switched_branch'].first()
print(f"Total branches: {len(branch_summary)}")
print(f"Switched branches (treatment): {branch_summary.sum()}")
print(f"Non-switched branches (control): {len(branch_summary) - branch_summary.sum()}")

# Average service scores by group and period
print("\n=== Average Service Scores ===")
score_summary = df.groupby(['switched_branch', 'after_switch'])['service_score'].agg(['mean', 'std', 'count'])
print(score_summary)

# Calculate the DiD estimate manually
print("\n=== Manual DiD Calculation ===")
# Control group (switched_branch=0)
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
control_change = control_after - control_before

# Treatment group (switched_branch=1)
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
treat_change = treat_after - treat_before

did_estimate = treat_change - control_change

print(f"Control group - Before: {control_before:.2f}, After: {control_after:.2f}, Change: {control_change:.2f}")
print(f"Treatment group - Before: {treat_before:.2f}, After: {treat_after:.2f}, Change: {treat_change:.2f}")
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

# Run regression with standard errors
print("\n=== Regression Analysis ===")
# Basic DiD regression
model = ols('service_score ~ switched_branch + after_switch + treated', data=df).fit()
print(model.summary())

# Regression with branch fixed effects
print("\n=== Regression with Branch Fixed Effects ===")
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ switched_branch + after_switch + treated + C(branch_id_str)', data=df).fit()
print(model_fe.summary())

# Regression with branch and week fixed effects
print("\n=== Regression with Branch and Week Fixed Effects ===")
model_fefe = ols('service_score ~ switched_branch + after_switch + treated + C(branch_id_str) + C(week)', data=df).fit()
print(model_fefe.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create interaction term
df['treated'] = df['switched_branch'] * df['after_switch']

# Run regression with branch and week fixed effects, cluster SE by branch
print("=== Difference-in-Differences Regression ===")
print("With Branch and Week Fixed Effects")
print("Standard Errors Clustered by Branch\n")

# Use cluster-robust standard errors
model = ols('service_score ~ switched_branch + after_switch + treated + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
