import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load data
df = pd.read_csv('branches.csv')

# Check how many branches switched vs didn't
branches = df[['branch_id', 'switched_branch']].drop_duplicates()
print("Branches by group:")
print(branches['switched_branch'].value_counts())
print(f"\nTotal branches: {len(branches)}")
print(f"Switched: {branches['switched_branch'].sum()} ({branches['switched_branch'].mean()*100:.1f}%)")
print(f"Control: {(branches['switched_branch']==0).sum()} ({(1-branches['switched_branch'].mean())*100:.1f}%)")

# Average service scores by group and period
print("\n\nAverage Service Scores:")
df['period'] = df['after_switch'].map({0: 'Before', 1: 'After'})
pivot = df.groupby(['switched_branch', 'period'])['service_score'].agg(['mean', 'std', 'count']).round(2)
print(pivot)

# Calculate the difference-in-differences manually
before_treated = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
after_treated = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()
before_control = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
after_control = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()

print(f"\nTreated group: Before={before_treated:.2f}, After={after_treated:.2f}, Change={after_treated-before_treated:.2f}")
print(f"Control group: Before={before_control:.2f}, After={after_control:.2f}, Change={after_control-before_control:.2f}")
did_estimate = (after_treated - before_treated) - (after_control - before_control)
print(f"\nDifference-in-Differences estimate: {did_estimate:.2f}")

# Check pre-trends: average scores by week for each group
print("\n\nPre-trends check (average service score by week):")
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
print(weekly_avg.pivot(index='week', columns='switched_branch', values='service_score').round(2))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load data
df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model 1: Basic DiD regression
print("="*60)
print("MODEL 1: Basic Difference-in-Differences Regression")
print("="*60)
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model1.summary().tables[1])

# Model 2: With branch fixed effects
print("\n" + "="*60)
print("MODEL 2: DiD with Branch Fixed Effects")
print("="*60)
df['branch_id_str'] = df['branch_id'].astype(str)
model2 = ols('service_score ~ after_switch + treatment + C(branch_id_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model2.summary().tables[1])
print(f"\nTreatment coefficient: {model2.params['treatment']:.3f}")
print(f"Standard error: {model2.bse['treatment']:.3f}")
print(f"95% CI: [{model2.conf_int().loc['treatment'][0]:.3f}, {model2.conf_int().loc['treatment'][1]:.3f}]")
print(f"P-value: {model2.pvalues['treatment']:.4f}")

# Model 3: With branch and week fixed effects
print("\n" + "="*60)
print("MODEL 3: DiD with Branch AND Week Fixed Effects")
print("="*60)
model3 = ols('service_score ~ treatment + C(branch_id_str) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model3.summary().tables[1])
print(f"\nTreatment coefficient: {model3.params['treatment']:.3f}")
print(f"Standard error: {model3.bse['treatment']:.3f}")
print(f"95% CI: [{model3.conf_int().loc['treatment'][0]:.3f}, {model3.conf_int().loc['treatment'][1]:.3f}]")
print(f"P-value: {model3.pvalues['treatment']:.4f}")

# Model 4: With controls (staffing and visits)
print("\n" + "="*60)
print("MODEL 4: DiD with Branch/Week FE + Controls")
print("="*60)
model4 = ols('service_score ~ treatment + staffing_index + visits_index + C(branch_id_str) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model4.summary().tables[1])
print(f"\nTreatment coefficient: {model4.params['treatment']:.3f}")
print(f"Standard error: {model4.bse['treatment']:.3f}")
print(f"95% CI: [{model4.conf_int().loc['treatment'][0]:.3f}, {model4.conf_int().loc['treatment'][1]:.3f}]")
print(f"P-value: {model4.pvalues['treatment']:.4f}")
