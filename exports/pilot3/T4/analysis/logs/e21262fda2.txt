import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run the DiD regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch*after_switch) + error
# beta3 is our DiD estimate

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("Difference-in-Differences Regression Results:")
print("=" * 60)
print(model.summary())

# Also run with cluster-robust standard errors at branch level
model_clustered = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nWith Cluster-Robust Standard Errors (clustered by branch):")
print("=" * 60)
print(model_clustered.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Check for heterogeneous effects - does the effect vary by branch characteristics?
# Since we don't have branch-level characteristics, let's check if there's variation
# in the treatment effect over time (dynamic effects)

print("Dynamic Effects - Treatment effect by week post-switch:")
print("=" * 60)

# Create week-specific treatment indicators
for week in range(15, 23):
    df[f'post_week_{week}'] = ((df['week'] == week) & (df['switched_branch'] == 1)).astype(int)

# Run regression with week-specific effects (excluding week 14 as reference)
week_vars = [f'post_week_{w}' for w in range(15, 23)]
formula = 'service_score ~ switched_branch + week + ' + ' + '.join(week_vars)
dynamic_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nWeek-specific treatment effects (relative to pre-switch baseline):")
for week in range(15, 23):
    coef = dynamic_model.params[f'post_week_{week}']
    pval = dynamic_model.pvalues[f'post_week_{week}']
    print(f"  Week {week}: {coef:.2f} points (p={pval:.3f})")

# Check if treatment branches had systematically different baseline scores
print("\n\nBaseline Comparison (Pre-switch period, weeks 6-14):")
pre_data = df[df['after_switch'] == 0]
print(f"  Control group average: {pre_data[pre_data['switched_branch']==0]['service_score'].mean():.2f}")
print(f"  Treatment group average: {pre_data[pre_data['switched_branch']==1]['service_score'].mean():.2f}")
print(f"  Difference: {pre_data[pre_data['switched_branch']==1]['service_score'].mean() - pre_data[pre_data['switched_branch']==0]['service_score'].mean():.2f} points")

# Check balance on control variables
print("\n\nControl Variable Balance (Pre-period averages):")
for var in ['staffing_index', 'visits_index']:
    ctrl_avg = pre_data[pre_data['switched_branch']==0][var].mean()
    treat_avg = pre_data[pre_data['switched_branch']==1][var].mean()
    print(f"  {var}: Control={ctrl_avg:.3f}, Treatment={treat_avg:.3f}, Diff={treat_avg-ctrl_avg:.3f}")
