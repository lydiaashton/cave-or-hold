import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run the main DiD model with clustered SE
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Get the treatment effect and confidence interval
treatment_effect = model.params['treatment']
treatment_se = model.bse['treatment']
treatment_pvalue = model.pvalues['treatment']

# 95% confidence interval
ci_lower = treatment_effect - 1.96 * treatment_se
ci_upper = treatment_effect + 1.96 * treatment_se

print("Treatment Effect Summary:")
print(f"  Point estimate: {treatment_effect:.2f} points")
print(f"  Standard error: {treatment_se:.2f}")
print(f"  95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"  P-value: {treatment_pvalue:.3f}")

# Calculate the effect size relative to baseline
baseline_mean = df[df['after_switch']==0]['service_score'].mean()
print(f"\n  Baseline mean service score: {baseline_mean:.2f}")
print(f"  Effect as % of baseline: {(treatment_effect/baseline_mean)*100:.1f}%")

# Pre-trends check - compare trends before week 15
pre_data = df[df['week'] < 15]
pre_model = ols('service_score ~ switched_branch * week', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})

import pandas as pd

import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Calculate mean service score by week and group
weekly_means = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_std = df.groupby(['week', 'switched_branch'])['service_score'].std().reset_index()
weekly_means = weekly_means.merge(weekly_std, on=['week', 'switched_branch'], suffixes=('_mean', '_std'))

# Create the plot
fig, ax = plt.subplots(figsize=(14, 7))

# Plot lines for each group
for group in [0, 1]:
    subset = weekly_means[weekly_means['switched_branch'] == group]
    label = 'Switched Branches (n=20)' if group == 1 else 'Non-Switched Branches (n=14)'
    color = '#2E86AB' if group == 1 else '#A23B72'
    ax.plot(subset['week'], subset['service_score_mean'], marker='o', label=label, 
            linewidth=2.5, color=color, markersize=8)
    
    # Add confidence intervals
    ci = 1.96 * subset['service_score_std'] / np.sqrt(len(df[df['switched_branch']==group]) / 17)
    ax.fill_between(subset['week'], 
                    subset['service_score_mean'] - ci, 
                    subset['service_score_mean'] + ci, 
                    alpha=0.2, color=color)

# Add vertical line at week 15 (switch point)
ax.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Switch Implemented (Week 15)')
