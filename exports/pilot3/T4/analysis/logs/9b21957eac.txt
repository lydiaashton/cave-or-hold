import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run a difference-in-differences regression
# service_score = beta0 + beta1*switched_branch + beta2*after_switch + beta3*(switched_branch * after_switch) + error
# beta3 is the DiD estimate

# Basic DiD regression
model = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# Now let's add branch fixed effects to control for time-invariant branch characteristics
# We'll use week fixed effects too to control for common time trends
df['switched_after'] = df['switched_branch'] * df['after_switch']

# Regression with branch and week fixed effects
model_fe = ols('service_score ~ switched_branch + after_switch + switched_after + C(branch_id) + C(week)', data=df).fit()
print("\n=== DiD Regression with Branch and Week Fixed Effects ===")
print(model_fe.summary())

# Cluster-robust standard errors at branch level
model_fe_clustered = model_fe.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print("\n=== DiD with Clustered Standard Errors (at branch level) ===")
print(model_fe_clustered.summary())

from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# The key coefficient is on switched_after
# We need to be careful about collinearity - switched_branch is absorbed by branch FE

model_fe = ols('service_score ~ switched_after + C(branch_id) + C(week)', data=df).fit()
