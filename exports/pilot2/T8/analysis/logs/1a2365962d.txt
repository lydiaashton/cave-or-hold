import pandas as pd

import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Run DiD regression with cluster-robust standard errors
# Basic DiD: service_score ~ switched_branch * after_switch
model = smf.ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("Basic DiD Regression (no clustering):")
print(model.summary().tables[1])

# Now with cluster-robust SEs at branch level
model_cluster = smf.ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression (clustered by branch):")
print(model_cluster.summary().tables[1])

# Also control for staffing and visits
model_full = smf.ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with controls (clustered by branch):")
print(model_full.summary().tables[1])

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create week dummies and interact with treatment for pre-trends test
# Use week 6 as baseline
df['week_cat'] = df['week'].astype('category')

# Event study specification: interact treatment with each week (relative to week 15)
df['relative_week'] = df['week'] - 15  # week 15 is period 0

# Create dummies for each relative week
for rw in range(-9, 8):  # weeks -9 to +7 relative to switch
    df[f'rel_week_{rw}'] = (df['relative_week'] == rw).astype(int)

# Run event study regression (excluding week -1 as reference, or week 6)
# Using week -9 (week 6) as baseline
pre_weeks = [-9, -8, -7, -6, -5, -4, -3, -2, -1]  # weeks 6-14
post_weeks = [0, 1, 2, 3, 4, 5, 6, 7]  # weeks 15-22

# Build formula - use week -1 as reference (common practice)
pre_terms = ' + '.join([f'switched_branch:rel_week_{w}' for w in pre_weeks[:-1]])  # exclude -1
post_terms = ' + '.join([f'switched_branch:rel_week_{w}' for w in post_weeks])

formula = f'service_score ~ switched_branch + after_switch + {pre_terms} + {post_terms}'

model_event = smf.ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Coefficients (interaction of switched_branch with relative week):")
print("\nPre-trend coefficients (should be ~0 if parallel trends hold):")
for w in pre_weeks[:-1]:
    coef = model_event.params[f'switched_branch:rel_week_{w}']
    se = model_event.bse[f'switched_branch:rel_week_{w}']
    pval = model_event.pvalues[f'switched_branch:rel_week_{w}']
    print(f"  Week {w+15} (rel {w}): coef={coef:.2f}, SE={se:.2f}, p={pval:.3f}")

print("\nPost-treatment coefficients:")
for w in post_weeks:
    coef = model_event.params[f'switched_branch:rel_week_{w}']
    se = model_event.bse[f'switched_branch:rel_week_{w}']
    pval = model_event.pvalues[f'switched_branch:rel_week_{w}']
    print(f"  Week {w+15} (rel {w}): coef={coef:.2f}, SE={se:.2f}, p={pval:.3f}")

# Test joint significance of pre-trends
pre_terms_list = [f'switched_branch:rel_week_{w}' for w in pre_weeks[:-1]]
from statsmodels.stats.anova import anova_lm

# Use Wald test for joint significance
hypotheses = ', '.join([f'{t} = 0' for t in pre_terms_list])
wald_test = model_event.wald_test(hypotheses)
print(f"\nJoint test of pre-trends (all pre-period coefficients = 0):")
print(f"  F-statistic: {wald_test.statistic[0][0]:.2f}, p-value: {wald_test.pvalue:.3f}")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create relative week
df['relative_week'] = df['week'] - 15

# Create dummies with safe names
for rw in range(-9, 8):
    safe_name = f'rw_{rw}'.replace('-', 'm')
    df[safe_name] = (df['relative_week'] == rw).astype(int)

pre_weeks = [-9, -8, -7, -6, -5, -4, -3, -2]
post_weeks = [0, 1, 2, 3, 4, 5, 6, 7]

pre_terms = ' + '.join([f'switched_branch:rw_{w}'.replace('-', 'm') for w in pre_weeks])
post_terms = ' + '.join([f'switched_branch:rw_{w}'.replace('-', 'm') for w in post_weeks])

formula = f'service_score ~ switched_branch + after_switch + {pre_terms} + {post_terms}'

model_event = smf.ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients and confidence intervals for event study plot
coefficients = []
weeks = []
lower_ci = []
upper_ci = []

for w in pre_weeks + [-1] + post_weeks:  # -1 is reference
    if w == -1:
        coefficients.append(0)
        lower_ci.append(0)
        upper_ci.append(0)
    else:
        safe_name = f'switched_branch:rw_{w}'.replace('-', 'm')
        coef = model_event.params[safe_name]
        se = model_event.bse[safe_name]
        coefficients.append(coef)
        lower_ci.append(coef - 1.96 * se)
        upper_ci.append(coef + 1.96 * se)
    weeks.append(w)

# Create event study plot
fig, ax = plt.subplots(figsize=(10, 6))

ax.bar(weeks, coefficients, color=['gray' if w < 0 else 'steelblue' for w in weeks], alpha=0.7, edgecolor='black')
ax.errorbar(weeks, coefficients, yerr=[np.array(coefficients) - np.array(lower_ci), np.array(upper_ci) - np.array(coefficients)], 
            fmt='none', ecolor='black', capsize=5, linewidth=1.5)

ax.axhline(y=0, color='black', linestyle='-', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='--', linewidth=2, label='Switch')

ax.set_xlabel('Relative Week (0 = Week 15, Switch)', fontsize=12)
ax.set_ylabel('Effect on Service Score (vs Control)', fontsize=12)
ax.set_title('Event Study: Effect of New Shelving Procedure on Service Scores', fontsize=14, fontweight='bold')
ax.set_xticks(weeks)
ax.set_xticklabels([f'{w+15}' if w >= 0 else f'{w+15}' for w in weeks])
ax.legend(loc='upper left', fontsize=11)
ax.grid(True, alpha=0.3, axis='y')

plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("Event study plot saved as event_study.png")

post_coefs = []
for w in post_weeks:

    post_coefs.append(model_event.params[safe_name])

import pandas as pd
import numpy as np
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Re-examine the key evidence
print("=" * 70)
print("RECONSIDERING THE EVIDENCE")
print("=" * 70)

# 1. Simple DiD
model = smf.ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
did_effect = model.params['switched_branch:after_switch']
did_pval = model.pvalues['switched_branch:after_switch']
did_ci = model.conf_int().loc['switched_branch:after_switch']

print(f"\n1. Main DiD Estimate:")
print(f"   Effect: {did_effect:.2f} points")
print(f"   p-value: {did_pval:.3f}")
print(f"   95% CI: [{did_ci[0]:.2f}, {did_ci[1]:.2f}]")
print(f"   Interpretation: There is a {did_pval*100:.0f}% chance of seeing this result if there were truly no effect")

# 2. One-sided test (since we had a directional hypothesis)
from scipy import stats
t_stat = did_effect / model.bse['switched_branch:after_switch']
one_sided_pval = 1 - stats.norm.cdf(t_stat)
print(f"\n2. One-sided test (we expected improvement):")
print(f"   p-value: {one_sided_pval:.3f}")
print(f"   Interpretation: There is a {one_sided_pval*100:.1f}% chance of seeing this result if there were truly no effect")

# 3. Event study post-period effects
print(f"\n3. Event Study - Post-Period Week-by-Week:")
print(f"   (Relative to week 14 as baseline)")

# Recreate event study
df['relative_week'] = df['week'] - 15
for rw in range(-9, 8):
    safe_name = f'rw_{rw}'.replace('-', 'm')
    df[safe_name] = (df['relative_week'] == rw).astype(int)

pre_weeks = [-9, -8, -7, -6, -5, -4, -3, -2]
post_weeks = [0, 1, 2, 3, 4, 5, 6, 7]

pre_terms = ' + '.join([f'switched_branch:rw_{w}'.replace('-', 'm') for w in pre_weeks])
post_terms = ' + '.join([f'switched_branch:rw_{w}'.replace('-', 'm') for w in post_weeks])
formula = f'service_score ~ switched_branch + after_switch + {pre_terms} + {post_terms}'

model_event = smf.ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

significant_weeks = 0
for w in post_weeks:
    safe_name = f'switched_branch:rw_{w}'.replace('-', 'm')
    coef = model_event.params[safe_name]
    pval = model_event.pvalues[safe_name]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.10 else ""
    if pval < 0.05:
        significant_weeks += 1
    print(f"   Week {w+15}: {coef:.2f} points (p={pval:.3f}) {sig}")

print(f"\n   Weeks with p < 0.05: {significant_weeks} of 8")
print(f"   Average post-period effect: {np.mean([model_event.params[f'switched_branch:rw_{w}'.replace('-', 'm')] for w in post_weeks]):.2f} points")

# 4. Probability the effect is positive
print(f"\n4. Probability the true effect is positive:")
prob_positive = 1 - stats.norm.cdf(0, did_effect, model.bse['switched_branch:after_switch'])
print(f"   Based on the estimate and SE: {prob_positive*100:.1f}%")

# 5. Effect size context
print(f"\n5. Effect Size Context:")
avg_score = df['service_score'].mean()
print(f"   Average service score: {avg_score:.1f}")
print(f"   DiD effect as % of average: {did_effect/avg_score*100:.1f}%")
print(f"   DiD effect in SD units: {did_effect/df['service_score'].std():.2f} standard deviations")

print("\n" + "=" * 70)
