import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch and week fixed effects
print("=== DiD Regression with Branch and Week Fixed Effects ===")
model = ols('service_score ~ treatment + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model.summary().tables[1])

# Also run with controls
print("\n=== DiD Regression with Controls (staffing, visits) ===")
model2 = ols('service_score ~ treatment + staffing_index + visits_index + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model2.summary().tables[1])

# Check parallel trends - event study
print("\n=== Event Study (Parallel Trends Check) ===")
# Create week dummies interacted with treatment
df['relative_week'] = df['week'] - 14  # week 14 is the last pre-period week
df['relative_week'] = df['relative_week'].astype(int)

# Create dummies for each week relative to switch, interacted with treatment
event_study_vars = []
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != -1:  # omit week -1 (week 13) as reference
        df[f'treat_week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(f'treat_week_{w}')

# Run event study regression
event_formula = f'service_score ~ {" + ".join(event_study_vars)} + C(branch_id) + C(week)'
event_model = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the interaction terms
event_results = []
for w in range(-8, 9):
    if w != -1:
        coef = event_model.params[f'treat_week_{w}']
        se = event_model.bse[f'treat_week_{w}']
        event_results.append({'relative_week': w, 'coef': coef, 'se': se, 'pvalue': event_model.pvalues[f'treat_week_{w}']})

event_df = pd.DataFrame(event_results)
print(event_df.round(3))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with branch and week fixed effects
print("=== DiD Regression (Key Results) ===")
model = ols('service_score ~ treatment + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"Treatment effect: {model.params['treatment']:.3f}")
print(f"Standard error: {model.bse['treatment']:.3f}")
print(f"P-value: {model.pvalues['treatment']:.4f}")
print(f"95% CI: [{model.conf_int().loc['treatment'][0]:.3f}, {model.conf_int().loc['treatment'][1]:.3f}]")

# Also run with controls
print("\n=== DiD Regression with Controls ===")
model2 = ols('service_score ~ treatment + staffing_index + visits_index + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"Treatment effect: {model2.params['treatment']:.3f}")
print(f"Standard error: {model2.bse['treatment']:.3f}")
print(f"P-value: {model2.pvalues['treatment']:.4f}")

# Event study - parallel trends check
print("\n=== Event Study (Parallel Trends Check) ===")
df['relative_week'] = df['week'] - 14  # week 14 is last pre-period week

# Create dummies for each week relative to switch, interacted with treatment
event_study_vars = []
for w in range(-8, 9):  # weeks -8 to +8 relative to week 14
    if w != -1:  # omit week -1 (week 13) as reference
        df[f'treat_week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
        event_study_vars.append(f'treat_week_{w}')

# Run event study regression
event_formula = f'service_score ~ {" + ".join(event_study_vars)} + C(branch_id) + C(week)'
event_model = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for the interaction terms
event_results = []
for w in range(-8, 9):
    if w != -1:
        coef = event_model.params[f'treat_week_{w}']
        se = event_model.bse[f'treat_week_{w}']
        pval = event_model.pvalues[f'treat_week_{w}']
        significant = '*' if pval < 0.05 else ''
        event_results.append({'relative_week': w, 'coef': round(coef, 2), 'se': round(se, 2), 'pvalue': round(pval, 3), 'sig': significant})

event_df = pd.DataFrame(event_results)
print(event_df.to_string(index=False))

# Check pre-trends (weeks -8 to -2 should be close to 0 and not significant)
pre_trends = event_df[event_df['relative_week'] < -1]
print(f"\nPre-trend coefficients (should be near 0):")
print(f"Mean: {pre_trends['coef'].mean():.2f}")
print(f"Range: [{pre_trends['coef'].min():.2f}, {pre_trends['coef'].max():.2f}]")
print(f"All pre-trends insignificant: {(pre_trends['pvalue'] > 0.05).all()}")

# Post-treatment effects
post_trends = event_df[event_df['relative_week'] > -1]
print(f"\nPost-treatment coefficients:")
print(post_trends[['relative_week', 'coef', 'sig']].to_string(index=False))
