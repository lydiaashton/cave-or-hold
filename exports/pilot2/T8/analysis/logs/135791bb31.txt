import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['did'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with controls
print("=== Difference-in-Differences Regression ===")
model = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit()
print(model.summary())

# Also run with branch fixed effects
print("\n=== DiD with Branch Fixed Effects ===")
# Create branch dummies
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_fe = pd.concat([df, branch_dummies], axis=1)

# Build formula with fixed effects
fe_cols = [c for c in df_with_fe.columns if c.startswith('branch')]
formula = 'service_score ~ switched_branch + after_switch + did + staffing_index + visits_index + ' + ' + '.join(fe_cols)
model_fe = ols(formula, data=df_with_fe).fit()
print(model_fe.summary())

# Check parallel trends - compare pre-treatment trends
print("\n=== Parallel Trends Check (Pre-Period Only) ===")
pre_df = df[df['after_switch'] == 0]
pre_model = ols('service_score ~ week * switched_branch + staffing_index + visits_index', data=pre_df).fit()
print(pre_model.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']

# Run DiD with branch fixed effects using a cleaner approach
# Use C() for categorical variables
print("=== DiD with Branch Fixed Effects (Clean) ===")
model_fe = ols('service_score ~ C(branch_id) + switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
# Note: switched_branch is collinear with branch FE, so it will be dropped
print(model_fe.summary())

# Let's run it without the collinear variable
model_fe2 = ols('service_score ~ C(branch_id) + after_switch + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD with Branch FE (without switched_branch) ===")
print(model_fe2.summary())

# Event study / parallel trends visualization
print("\n=== Weekly Differences (Treatment - Control) ===")
weekly_diff = df.groupby('week').apply(
    lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()
)
print(weekly_diff)

# Create visualization
fig, ax = plt.subplots(figsize=(10, 6))

# Plot weekly averages for both groups
weekly = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly.columns = ['Control (No Switch)', 'Treatment (Switched)']

ax.plot(weekly.index, weekly['Control (No Switch)'], 'o-', label='Control (No Switch)', linewidth=2, markersize=8)
ax.plot(weekly.index, weekly['Treatment (Switched)'], 's-', label='Treatment (Switched)', linewidth=2, markersize=8)

ax.set_ylabel('Average Service Score', fontsize=12)

ax.legend(loc='best')
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150)
print("\nPlot saved as service_score_trends.png")

# Calculate confidence intervals for the DiD estimate
print("\n=== DiD Effect Summary ===")
did_coef = model_fe2.params['did']

did_pvalue = model_fe2.pvalues['did']
print(f"DiD Coefficient: {did_coef:.2f}")
