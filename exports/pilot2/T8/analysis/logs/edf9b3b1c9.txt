import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# ============ 1. DESCRIPTIVE STATS ============
print("=" * 60)
print("DESCRIPTIVE STATISTICS")
print("=" * 60)

# Treatment vs Control group sizes
print("\nBranches by group:")
print(df.groupby('switched_branch')['branch_id'].nunique())

# Mean service scores by group and period
print("\nMean Service Scores:")
pivot = df.groupby(['switched_branch', 'after_switch'])['service_score'].mean().unstack()
pivot.columns = ['Before (Weeks 6-14)', 'After (Weeks 15-22)']
pivot.index = ['Control (No Switch)', 'Treatment (Switched)']
print(pivot)

# Calculate DiD manually
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

control_change = control_after - control_before
treatment_change = treatment_after - treatment_before
did_estimate = treatment_change - control_change

print(f"\nControl group change: {control_change:.3f}")
print(f"Treatment group change: {treatment_change:.3f}")
print(f"DiD Estimate: {did_estimate:.3f}")

# ============ 2. REGRESSION ANALYSIS ============
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Simple DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
print("\nModel 1: Simple DiD (no controls)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with controls")
print(model2.summary().tables[1])

# Clustered standard errors by branch
print("\n" + "=" * 60)
print("CLUSTERED STANDARD ERRORS (by branch)")
print("=" * 60)

from statsmodels.stats.sandwich_covariance import cov_cluster
cov_clustered = cov_cluster(model2, df['branch_id'])
se_clustered = np.sqrt(np.diag(cov_clustered))
t_clustered = model2.params / se_clustered
from scipy import stats
p_clustered = 2 * (1 - stats.t.cdf(np.abs(t_clustered), df=len(df) - len(df['branch_id'].unique())))

clustered_results = pd.DataFrame({
    'coef': model2.params,
    'se_clustered': se_clustered,
    't': t_clustered,
    'p_value': p_clustered
})
print(clustered_results)
