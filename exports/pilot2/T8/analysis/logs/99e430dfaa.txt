import pandas as pd

df = pd.read_csv('branches.csv')

# Check parallel trends - compare trends before the switch
print("=== Pre-Trend Check (Weeks 6-14) ===")
pre_data = df[df['week'] < 15]
pre_means = pre_data.groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
print(pre_means)

# Calculate pre-trend slopes
for group in [0, 1]:
    group_pre = pre_data[pre_data['switched_branch'] == group]
    corr = np.corrcoef(group_pre['week'], group_pre['service_score'])[0,1]
    print(f"Group {group} pre-trend correlation with week: {corr:.3f}")

# Create visualization
fig, ax = plt.subplots(figsize=(10, 6))

for group in [0, 1]:
    group_data = df[df['switched_branch'] == group]
    weekly_means = group_data.groupby('week')['service_score'].mean()
    label = 'Switched branches' if group == 1 else 'Non-switched branches'
    ax.plot(weekly_means.index, weekly_means.values, marker='o', label=label, linewidth=2)

# Create dummies for each week relative to switch (excluding the reference period)
# Use w+10 to avoid negative numbers in column names
for w in range(-9, 8):  # weeks -9 to -1 (before) and 0 to 7 (after)
    if w != -1:  # use week -1 as reference
        df_event[f'rw_{w+10}'] = ((df_event['weeks_relative'] == w) & (df_event['switched_branch'] == 1)).astype(int)

model5 = smf.ols(formula3, data=df_event).fit(cov_type='cluster', cov_kwds={'groups': df_event['branch_id']})

print("Event study coefficients (relative to week -1, switched branches):")
print("Week -1 is the reference period (coefficient = 0)")
for w in range(-9, 8):
    if w != -1:
        col = f'rw_{w+10}'
        coef = model5.params[col]
        se = model5.bse[col]
        pval = model5.pvalues[col]
        period = "After" if w >= 0 else "Before"
        print(f"  Week {w:2d} ({period}): {coef:6.2f} (SE: {se:.2f}, p={pval:.3f})")
