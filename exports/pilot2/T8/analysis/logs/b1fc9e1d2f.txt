import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

# Load data
df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# ============================================
# 1. Simple Difference-in-Differences Calculation
# ============================================

# Mean service scores by group and period
pre_treatment = df[df['after_switch'] == 0]
post_treatment = df[df['after_switch'] == 1]

# Control group (switched_branch = 0)
control_pre = pre_treatment[pre_treatment['switched_branch'] == 0]['service_score'].mean()
control_post = post_treatment[post_treatment['switched_branch'] == 0]['service_score'].mean()

# Treatment group (switched_branch = 1)
treatment_pre = pre_treatment[pre_treatment['switched_branch'] == 1]['service_score'].mean()
treatment_post = post_treatment[post_treatment['switched_branch'] == 1]['service_score'].mean()

print("=" * 60)
print("SIMPLE DIFFERENCE-IN-DIFFERENCES")
print("=" * 60)
print(f"\nControl Group (non-switched branches):")
print(f"  Pre-switch mean: {control_pre:.2f}")
print(f"  Post-switch mean: {control_post:.2f}")
print(f"  Change: {control_post - control_pre:.2f}")

print(f"\nTreatment Group (switched branches):")
print(f"  Pre-switch mean: {treatment_pre:.2f}")
print(f"  Post-switch mean: {treatment_post:.2f}")
print(f"  Change: {treatment_post - treatment_pre:.2f}")

# DiD estimate
did_estimate = (treatment_post - treatment_pre) - (control_post - control_pre)
print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")

# ============================================
# 2. Regression-based DiD with controls
# ============================================

print("\n" + "=" * 60)
print("REGRESSION-BASED DiD ANALYSIS")
print("=" * 60)

# Basic DiD model
model_basic = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nBasic DiD Model (with cluster-robust SEs):")
print(model_basic.summary().tables[1])

# DiD model with controls
model_controls = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nDiD Model with Controls (with cluster-robust SEs):")
print(model_controls.summary().tables[1])

# ============================================
# 3. Check Parallel Trends - Pre-treatment comparison
# ============================================

print("\n" + "=" * 60)
print("PARALLEL TRENDS CHECK (Pre-treatment weeks 6-14)")

weekly_means['se'] = weekly_means['std'] / np.sqrt(weekly_means['count'])

# Pivot for plotting
pivot_means = weekly_means.pivot(index='week', columns='switched_branch', values='mean')
pivot_se = weekly_means.pivot(index='week', columns='switched_branch', values='se')

# Create the plot
fig, ax = plt.subplots(figsize=(12, 7))

                pivot_means[1] - 1.96 * pivot_se[1], 
                pivot_means[1] + 1.96 * pivot_se[1], 
                alpha=0.2, color='blue')

# Plot control group (switched = 0)
ax.plot(pivot_means.index, pivot_means[0], 'r-s', linewidth=2, label='Non-Switched Branches (Control)', markersize=6)
ax.fill_between(pivot_means.index, 
                pivot_means[0] - 1.96 * pivot_se[0], 
                pivot_means[0] + 1.96 * pivot_se[0],
