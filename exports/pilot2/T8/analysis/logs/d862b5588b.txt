import numpy as np

from statsmodels.formula.api import ols

# Create the interaction term for DiD
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit()
print("Simple DiD Regression:")
print(model.summary().tables[1])

# Now with branch fixed effects
print("\n\nDiD with Branch Fixed Effects:")
df['branch_id_str'] = df['branch_id'].astype(str)
model_fe = ols('service_score ~ switched_branch + after_switch + treatment_post + C(branch_id_str)', data=df).fit()

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Simple DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment_post', data=df).fit()
print("=== Simple DiD Regression ===")
print(f"Treatment effect (treatment_post): {model.params['treatment_post']:.3f}")
print(f"Std Error: {model.bse['treatment_post']:.3f}")
print(f"P-value: {model.pvalues['treatment_post']:.4f}")
print(f"95% CI: [{model.conf_int().loc['treatment_post'][0]:.3f}, {model.conf_int().loc['treatment_post'][1]:.3f}]")
