import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment_x_after'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with and without controls
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES REGRESSION ANALYSIS")
print("=" * 60)

# Model 1: Basic DiD (no controls)
model1 = ols('service_score ~ switched_branch + after_switch + treatment_x_after', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# Model 2: DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment_x_after + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with staffing and visits controls")
print(model2.summary().tables[1])

# Model 3: DiD with branch fixed effects
# This controls for time-invariant branch characteristics
model3 = ols('service_score ~ after_switch + treatment_x_after + staffing_index + visits_index + C(branch_id)', data=df).fit()
print("\nModel 3: DiD with branch fixed effects")
# Just show the key coefficients
print("\nKey coefficients:")
print(model3.params[['after_switch', 'treatment_x_after']].round(4))
print("\nStandard errors:")
print(model3.bse[['after_switch', 'treatment_x_after']].round(4))

# Check pre-trends (parallel trends assumption)
# Compare trends in weeks 6-14 (before switch) between treatment and control
print("\n" + "=" * 60)
print("PARALLEL TRENDS CHECK (Pre-switch weeks 6-14)")
print("=" * 60)

pre_switch = df[df['week'] < 15].copy()
pre_switch['week_centered'] = pre_switch['week'] - pre_switch['week'].mean()

# Regress service_score on week, switched_branch, and their interaction (in pre-period only)
pre_trend_model = ols('service_score ~ week_centered * switched_branch', data=pre_switch).fit()
print("\nPre-trend test (interaction should be non-significant if parallel trends hold):")
print(pre_trend_model.summary().tables[1])

# Also check week-by-week differences
print("\nAverage service score difference (Treatment - Control) by week:")
weekly_diff = pre_switch.groupby('week').apply(
    lambda x: x[x['switched_branch']==1]['service_score'].mean() - x[x['switched_branch']==0]['service_score'].mean()
)
print(weekly_diff.round(2))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')
df['treatment_x_after'] = df['switched_branch'] * df['after_switch']

# Event study: Create week dummies interacted with treatment
print("=" * 60)
print("EVENT STUDY / DYNAMIC DiD ANALYSIS")
print("=" * 60)

# Create week dummies (use week 14 as reference - last pre-switch week)
df = df.copy()
for w in range(6, 23):
    if w != 14:  # reference category
        df[f'week_{w}'] = (df['week'] == w).astype(int)
        df[f'week_{w}_treated'] = df[f'week_{w}'] * df['switched_branch']

# Run event study regression
event_study_vars = [f'week_{w}_treated' for w in range(6, 23) if w != 14]
formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(week) + C(branch_id)'
event_model = ols(formula, data=df).fit()

print("\nEvent Study Coefficients (Treatment effect by week, relative to week 14):")
print("-" * 60)
results = []
for w in range(6, 23):
    if w == 14:
        results.append((w, 0.0, 0.0, 0.0, 1.0))  # reference
    else:
        coef = event_model.params[f'week_{w}_treated']
        se = event_model.bse[f'week_{w}_treated']
        t_stat = coef / se
        p_val = 2 * (1 - sm.stats.norm.cdf(abs(t_stat)))
        results.append((w, coef, se, t_stat, p_val))

results_df = pd.DataFrame(results, columns=['week', 'coef', 'se', 't_stat', 'p_value'])
print(results_df.to_string(index=False, float_format='%.3f'))

# Check significance
print("\n" + "-" * 60)
print("Pre-switch weeks (6-14) - should be non-significant for valid DiD:")
pre_switch_results = results_df[results_df['week'] < 15]
print(pre_switch_results[['week', 'coef', 'p_value']].to_string(index=False, float_format='%.3f'))
print(f"  Any p < 0.05? {(pre_switch_results['p_value'] < 0.05).any()}")

print("\nPost-switch weeks (15-22) - treatment effect:")
post_switch_results = results_df[results_df['week'] >= 15]
print(post_switch_results[['week', 'coef', 'p_value']].to_string(index=False, float_format='%.3f'))
print(f"  Average post-switch effect: {post_switch_results['coef'].mean():.2f}")

# Create event study plot
fig, ax = plt.subplots(figsize=(10, 6))
ax.errorbar(results_df['week'], results_df['coef'], yerr=1.96*results_df['se'], 
            fmt='o', capsize=4, label='Treatment Effect')
ax.axhline(y=0, color='black', linestyle='--', alpha=0.5)
ax.axvline(x=14.5, color='red', linestyle='-', alpha=0.5, label='Switch occurred')
ax.set_xlabel('Week')
ax.set_ylabel('Treatment Effect (vs Week 14)')
ax.set_title('Event Study: Treatment Effect by Week\n(Reference: Week 14, last pre-switch week)')
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved to event_study.png")

# Additional robustness: Check if branches that switched were different
print("\n" + "=" * 60)
print("BRANCH CHARACTERISTICS COMPARISON")
print("=" * 60)

branch_stats = df.groupby('branch_id').agg({
    'switched_branch': 'first',
    'service_score': 'mean',
    'staffing_index': 'mean',
    'visits_index': 'mean'
}).reset_index()

print("\nAverage characteristics by group:")
comp = branch_stats.groupby('switched_branch')[['service_score', 'staffing_index', 'visits_index']].mean()
print(comp.round(3))

# T-test for differences
from scipy import stats
treated = branch_stats[branch_stats['switched_branch'] == 1]
control = branch_stats[branch_stats['switched_branch'] == 0]

for col in ['service_score', 'staffing_index', 'visits_index']:
    t_stat, p_val = stats.ttest_ind(treated[col], control[col])
    print(f"\n{col}: t={t_stat:.3f}, p={p_val:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from scipy import stats as scipy_stats

df = pd.read_csv('branches.csv')
df['treatment_x_after'] = df['switched_branch'] * df['after_switch']

print("=" * 60)
print("ROBUSTNESS CHECKS")
print("=" * 60)

# Robustness 1: Use only weeks 11-14 as pre-period (when trends were more parallel)
print("\n1. DiD using only weeks 11-14 as pre-period:")
df_restricted = df[df['week'] >= 11].copy()
model_restricted = ols('service_score ~ switched_branch + after_switch + treatment_x_after + staffing_index + visits_index', 
                       data=df_restricted).fit()
print(model_restricted.summary().tables[1][['coef', 'P>|t|']].loc[['switched_branch', 'after_switch', 'treatment_x_after']])

# Robustness 2: Include branch-specific time trends
print("\n2. DiD with branch-specific linear time trends:")
df['week_trend'] = df['week'] - df['week'].min()
df['branch_trend'] = df['switched_branch'] * df['week_trend']
model_trends = ols('service_score ~ switched_branch + after_switch + treatment_x_after + staffing_index + visits_index + branch_trend + C(branch_id)', 
                   data=df).fit()
print("Treatment effect (treatment_x_after):", model_trends.params['treatment_x_after'].round(4))
print("Std Error:", model_trends.bse['treatment_x_after'].round(4))
t_stat = model_trends.params['treatment_x_after'] / model_trends.bse['treatment_x_after']
p_val = 2 * (1 - scipy_stats.norm.cdf(abs(t_stat)))
print(f"t-stat: {t_stat:.3f}, p-value: {p_val:.4f}")

# Robustness 3: Post-switch only analysis (weeks 18-22 when effect is clear)
print("\n3. Post-switch period only (weeks 18-22) - Treatment vs Control comparison:")
df_post = df[df['week'] >= 18].copy()
