import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Create a treatment variable (interaction of switched_branch and after_switch)
df['treatment'] = df['switched_branch'] * df['after_switch']

# Difference-in-differences summary table
print("=" * 60)
print("DIFFERENCE-IN-DIFFERENCES SUMMARY")
print("=" * 60)

# Calculate means for each group/period combination
before_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
before_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()
after_treatment = df[(df['switched_branch'] == 1) & (df['after_switch'] == 1)]['service_score'].mean()

print(f"\nControl Group (did not switch):")
print(f"  Before switch (weeks 6-14): {before_control:.2f}")
print(f"  After switch (weeks 15-22): {after_control:.2f}")
print(f"  Change: {after_control - before_control:.2f}")

print(f"\nTreatment Group (switched):")
print(f"  Before switch (weeks 6-14): {before_treatment:.2f}")
print(f"  After switch (weeks 15-22): {after_treatment:.2f}")
print(f"  Change: {after_treatment - before_treatment:.2f}")

# Difference-in-differences estimate
diff_treatment = after_treatment - before_treatment
diff_control = after_control - before_control
did_estimate = diff_treatment - diff_control

print(f"\nDifference-in-Differences Estimate: {did_estimate:.2f}")
print(f"  (Treatment change) - (Control change) = {diff_treatment:.2f} - {diff_control:.2f}")

# Run regression with robust standard errors
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='HC3')
print(model.summary())

# Also check with controls (staffing and visits)
print("\n" + "=" * 60)
print("REGRESSION WITH CONTROLS")
print("=" * 60)

print(model_controls.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run regression on pre-period only to test parallel trends
pre_data = df[df['week'] < 15].copy()
pre_data['time_trend'] = pre_data['week'] - 6  # weeks since start
pre_data['interaction'] = pre_data['switched_branch'] * pre_data['time_trend']

pre_model = ols('service_score ~ switched_branch + time_trend + interaction', data=pre_data).fit(cov_type='HC3')
print("Pre-trend parallel trends test:")
print(pre_model.summary())

# Event study - estimate effect for each week relative to switch
print("\n" + "=" * 60)
print("EVENT STUDY ANALYSIS")
print("=" * 60)

# Create week dummies relative to switch (week 15 is the switch)
df['relative_week'] = df['week'] - 15

# Run event study regression (omitting week -1 as reference)
event_data = df.copy()
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    if w != -1:  # omit reference category
        event_data[f'week_{w}'] = ((event_data['relative_week'] == w) & (event_data['switched_branch'] == 1)).astype(int)

# Build formula
event_vars = [f'week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)

event_model = ols(formula, data=event_data).fit(cov_type='HC3')

# Extract coefficients for the event study
print("\nEvent Study Coefficients (relative to week -1):")
coef_dict = {}
for w in range(-9, 8):
    if w != -1:
        var_name = f'week_{w}'
        if var_name in event_model.params.index:
            coef = event_model.params[var_name]
            se = event_model.bse[var_name]
            pval = event_model.pvalues[var_name]
            coef_dict[w] = {'coef': coef, 'se': se, 'pval': pval}
            sig = '*' if pval < 0.1 else ''
            print(f"Week {w:+d}: {coef:.2f} (SE: {se:.2f}, p={pval:.3f}){sig}")

# Create event study plot
fig, ax = plt.subplots(figsize=(10, 6))

weeks = sorted(coef_dict.keys())
coefs = [coef_dict[w]['coef'] for w in weeks]
ses = [coef_dict[w]['se'] for w in weeks]

# Calculate confidence intervals
ci_lower = [coef_dict[w]['coef'] - 1.96 * coef_dict[w]['se'] for w in weeks]
ci_upper = [coef_dict[w]['coef'] + 1.96 * coef_dict[w]['se'] for w in weeks]

ax.plot(weeks, coefs, marker='o', linestyle='-', linewidth=2, color='blue', label='Effect Estimate')
ax.fill_between(weeks, ci_lower, ci_upper, alpha=0.3, color='blue', label='95% CI')
ax.axhline(y=0, color='black', linestyle='--', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='-', linewidth=2, label='Switch Week (15)')
ax.set_xlabel('Weeks Relative to Switch')
ax.set_ylabel('Service Score Effect')
ax.set_title('Event Study: Effect of New Procedure on Service Scores')
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(weeks)

plt.tight_layout()
plt.savefig('event_study.png', dpi=150, bbox_inches='tight')
print("\nEvent study plot saved to event_study.png")

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study - estimate effect for each week relative to switch
print("=" * 60)
print("EVENT STUDY ANALYSIS")
print("=" * 60)

# Create week dummies relative to switch (week 15 is the switch)
df['relative_week'] = df['week'] - 15

# Create properly named dummy variables
event_data = df.copy()
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    if w != -1:  # omit reference category (week -1, i.e., week 14)
        col_name = f'w{abs(w)}_{"neg" if w < 0 else "pos"}' if w != 0 else 'w0'
        event_data[col_name] = ((event_data['relative_week'] == w) & (event_data['switched_branch'] == 1)).astype(int)

# Build formula with proper variable names
event_vars = []
week_mapping = {}
for w in range(-9, 8):
    if w != -1:
        col_name = f'w{abs(w)}_{"neg" if w < 0 else "pos"}' if w != 0 else 'w0'
        event_vars.append(col_name)
        week_mapping[col_name] = w

formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)

event_model = ols(formula, data=event_data).fit(cov_type='HC3')

# Extract coefficients for the event study
print("\nEvent Study Coefficients (relative to week -1, i.e., week 14):")
coef_dict = {}
for var_name, w in week_mapping.items():
    coef = event_model.params[var_name]
    se = event_model.bse[var_name]
    pval = event_model.pvalues[var_name]
    coef_dict[w] = {'coef': coef, 'se': se, 'pval': pval}
    sig = '*' if pval < 0.1 else ''
    print(f"Week {w:+d} (actual week {15+w}): {coef:.2f} (SE: {se:.2f}, p={pval:.3f}){sig}")

# Create event study plot
fig, ax = plt.subplots(figsize=(10, 6))

weeks = sorted(coef_dict.keys())
coefs = [coef_dict[w]['coef'] for w in weeks]
ses = [coef_dict[w]['se'] for w in weeks]

# Calculate confidence intervals
ci_lower = [coef_dict[w]['coef'] - 1.96 * coef_dict[w]['se'] for w in weeks]
ci_upper = [coef_dict[w]['coef'] + 1.96 * coef_dict[w]['se'] for w in weeks]

ax.plot(weeks, coefs, marker='o', linestyle='-', linewidth=2, color='blue', label='Effect Estimate')

ax.axhline(y=0, color='black', linestyle='--', linewidth=1)
ax.axvline(x=-0.5, color='red', linestyle='-', linewidth=2, label='Switch Week (15)')
ax.set_xlabel('Weeks Relative to Switch')
ax.set_ylabel('Service Score Effect')
ax.set_title('Event Study: Effect of New Procedure on Service Scores\n(Reference: Week 14, one week before switch)')
ax.legend()
ax.grid(True, alpha=0.3)
ax.set_xticks(weeks)

plt.tight_layout()
plt.savefig('event_study.png', dpi=150, bbox_inches='tight')

df = pd.read_csv('branches.csv')

print("=" * 60)
print("BRANCH FIXED EFFECTS MODEL")
print("=" * 60)

# Branch fixed effects model - this controls for time-invariant branch characteristics
# We use the treatment interaction as the key variable
df['treatment'] = df['switched_branch'] * df['after_switch']

# Add branch as categorical
df['branch_cat'] = df['branch_id'].astype('category')

# Run with branch fixed effects
fe_model = ols('service_score ~ after_switch + treatment + C(branch_cat)', data=df).fit(cov_type='HC3')
print(fe_model.summary())

# Extract the treatment effect
treatment_coef = fe_model.params['treatment']
treatment_se = fe_model.bse['treatment']
treatment_pval = fe_model.pvalues['treatment']

print(f"\nTreatment effect (with branch FE): {treatment_coef:.2f} (SE: {treatment_se:.2f}, p={treatment_pval:.3f})")

# Also run a simpler model with branch and week fixed effects
print("\n" + "=" * 60)
print("BRANCH + WEEK FIXED EFFECTS MODEL")
print("=" * 60)

df['week_cat'] = df['week'].astype('category')
fe_week_model = ols('service_score ~ treatment + C(branch_cat) + C(week_cat)', data=df).fit(cov_type='HC3')
print(fe_week_model.summary())

treatment_coef2 = fe_week_model.params['treatment']
treatment_se2 = fe_week_model.bse['treatment']
treatment_pval2 = fe_week_model.pvalues['treatment']

print(f"\nTreatment effect (with branch + week FE): {treatment_coef2:.2f} (SE: {treatment_se2:.2f}, p={treatment_pval2:.3f})")

# Power analysis - what effect size could we detect?
print("\n" + "=" * 60)

n_control = n_branches - n_treated

print(f"Total observations: {len(df)}")

main_se = treatment_se2

print("\n1. PARALLEL TRENDS ASSUMPTION")

df['relative_week'] = df['week'] - 15
event_data = df.copy()

for w in range(-9, 8):
    if w != -1:
        col_name = f'w{abs(w)}_{"neg" if w < 0 else "pos"}' if w != 0 else 'w0'
        event_data[col_name] = ((event_data['relative_week'] == w) & (event_data['switched_branch'] == 1)).astype(int)

event_vars = []
week_mapping = {}
for w in range(-9, 8):
    if w != -1:
        col_name = f'w{abs(w)}_{"neg" if w < 0 else "pos"}' if w != 0 else 'w0'
        event_vars.append(col_name)
        week_mapping[col_name] = w

formula = 'service_score ~ switched_branch + after_switch + ' + ' + '.join(event_vars)
event_model = ols(formula, data=event_data).fit(cov_type='HC3')

pre_weeks = [w for w in week_mapping.values() if w < 0]
pre_significant = []
for var_name, w in week_mapping.items():
    if w < 0:
        pval = event_model.pvalues[var_name]
        if pval < 0.1:
            pre_significant.append((w, pval))

print(f"Pre-period weeks with significant differences (p<0.1): {len(pre_significant)}")
for w, p in pre_significant:
    print(f"  Week {w}: p={p:.3f}")

if len(pre_significant) >= 2:
    print("\n⚠️  CONCERN: Multiple pre-trends are significant.")
    print("   This violates the parallel trends assumption required for DiD.")
