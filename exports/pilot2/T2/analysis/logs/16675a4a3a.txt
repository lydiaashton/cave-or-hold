import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Run DiD regression with controls
# Model 1: Basic DiD
model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Model 1: Basic DiD (with cluster-robust SEs) ===")
print(model1.summary().tables[1])
print()

# Model 2: DiD with control variables
model2 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Model 2: DiD + Controls (with cluster-robust SEs) ===")
print(model2.summary().tables[1])
print()

# Model 3: DiD with branch and week fixed effects
df['branch_id_str'] = df['branch_id'].astype(str)
df['week_str'] = df['week'].astype(str)
model3 = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index + C(branch_id_str) + C(week_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== Model 3: DiD + Controls + Branch & Week FE (with cluster-robust SEs) ===")
# Just show the key coefficients
coef_df = model3.params.to_frame()
coef_df['std_err'] = model3.bse
coef_df['p_value'] = model3.pvalues
print(coef_df.loc[['switched_branch', 'after_switch', 'switched_branch:after_switch']])
print()

# Check parallel trends - event study approach
# Create relative week indicators
df['relative_week'] = df['week'] - 14  # week 14 is the last pre-switch week
print("=== Event Study: Service Scores by Relative Week ===")
event_study = df.groupby(['switched_branch', 'relative_week'])['service_score'].mean().unstack(level=0)
print(event_study.round(2))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Run the preferred model (with controls and fixed effects)
df['branch_id_str'] = df['branch_id'].astype(str)
df['week_str'] = df['week'].astype(str)
model = ols('service_score ~ switched_branch * after_switch + staffing_index + visits_index + C(branch_id_str) + C(week_str)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract key results
did_effect = model.params['switched_branch:after_switch']
did_se = model.bse['switched_branch:after_switch']
did_pvalue = model.pvalues['switched_branch:after_switch']
did_ci_lower = did_effect - 1.96 * did_se
did_ci_upper = did_effect + 1.96 * did_se

print("=== Key Results ===")
print(f"DiD Effect Estimate: {did_effect:.2f} points")
print(f"95% Confidence Interval: [{did_ci_lower:.2f}, {did_ci_upper:.2f}]")
print(f"P-value: {did_pvalue:.6f}")
print()

# Calculate percentage improvement relative to baseline
baseline_score = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
pct_improvement = (did_effect / baseline_score) * 100
print(f"Baseline (treatment pre-switch): {baseline_score:.2f}")
print(f"Percentage improvement: {pct_improvement:.1f}%")
print()

# Check how many branches benefited
branch_effects = df.groupby('branch_id').apply(
    lambda x: x[x['after_switch'] == 1]['service_score'].mean() - x[x['after_switch'] == 0]['service_score'].mean()
).reset_index()
branch_effects.columns = ['branch_id', 'score_change']
branch_effects['switched'] = df.groupby('branch_id')['switched_branch'].first().values

switched_branches = branch_effects[branch_effects['switched'] == 1]
print(f"Switched branches that improved: {(switched_branches['score_change'] > 0).sum()} / {len(switched_branches)} ({(switched_branches['score_change'] > 0).mean()*100:.1f}%)")
print(f"Average change for switched branches: {switched_branches['score_change'].mean():.2f}")
print(f"Average change for non-switched branches: {branch_effects[branch_effects['switched'] == 0]['score_change'].mean():.2f}")
print()

# Create visualization data
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
weekly_avg['period'] = weekly_avg['week'].apply(lambda x: 'Pre-Switch' if x < 15 else 'Post-Switch')

print("=== Weekly Averages for Visualization ===")
print(weekly_avg.pivot(index='week', columns='switched_branch', values='service_score').round(2))

import matplotlib.pyplot as plt
import pandas as pd

df = pd.read_csv('branches.csv')

# Create the visualization
fig, ax = plt.subplots(figsize=(12, 6))

# Weekly averages by group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Plot treatment group
treatment = weekly_avg[weekly_avg['switched_branch'] == 1]
control = weekly_avg[weekly_avg['switched_branch'] == 0]

ax.plot(treatment['week'], treatment['service_score'], 'b-o', linewidth=2, markersize=6, label='Switched Branches (New Procedure)')
ax.plot(control['week'], control['service_score'], 'r-s', linewidth=2, markersize=6, label='Non-Switched Branches (Old Procedure)')

# Add vertical line at switch point
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch Point (Week 15)')

# Add shaded region for post-switch period
ax.axvspan(14.5, 22, alpha=0.2, color='green', label='Post-Switch Period')

ax.set_xlabel('Week of Year', fontsize=12)
ax.set_ylabel('Service Score', fontsize=12)
ax.set_title('Library Service Scores: New vs. Old Procedure\nDifference-in-Differences Analysis', fontsize=14, fontweight='bold')
ax.legend(loc='upper left', fontsize=10)
ax.grid(True, alpha=0.3)
ax.set_xticks(range(6, 23))

# Add annotation for the effect
ax.annotate('Effect: +1.82 points\n(95% CI: 1.25 to 2.39)', 
            xy=(18, 52), xytext=(16, 54),
            fontsize=11, fontweight='bold',
            bbox=dict(boxstyle='round', facecolor='wheat', alpha=0.8),
            arrowprops=dict(arrowstyle='->', connectionstyle='arc3,rad=0'))

plt.tight_layout()
plt.savefig('service_score_analysis.png', dpi=150, bbox_inches='tight')
print("Chart saved as 'service_score_analysis.png'")
plt.show()
