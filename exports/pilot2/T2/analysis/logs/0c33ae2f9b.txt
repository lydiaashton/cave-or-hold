import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from scipy import stats
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['treatment'] = df['switched_branch'] * df['after_switch']

# ===== 1. Summary Statistics =====
print("=" * 60)
print("SUMMARY STATISTICS")
print("=" * 60)

# Pre-period (weeks 6-14) and Post-period (weeks 15-22)
pre_df = df[df['after_switch'] == 0]
post_df = df[df['after_switch'] == 1]

# Treatment and Control groups
treatment_pre = pre_df[pre_df['switched_branch'] == 1]['service_score']
treatment_post = post_df[post_df['switched_branch'] == 1]['service_score']
control_pre = pre_df[pre_df['switched_branch'] == 0]['service_score']
control_post = post_df[post_df['switched_branch'] == 0]['service_score']

print("\nService Score Means:")
print(f"  Treatment (switched) - Pre:  {treatment_pre.mean():.2f} (n={len(treatment_pre)})")
print(f"  Treatment (switched) - Post: {treatment_post.mean():.2f} (n={len(treatment_post)})")
print(f"  Control (not switched) - Pre:  {control_pre.mean():.2f} (n={len(control_pre)})")
print(f"  Control (not switched) - Post: {control_post.mean():.2f} (n={len(control_post)})")

# Calculate DiD estimate manually
treatment_change = treatment_post.mean() - treatment_pre.mean()
control_change = control_post.mean() - control_pre.mean()
did_estimate = treatment_change - control_change

print(f"\n  Treatment change (Post - Pre): {treatment_change:.2f}")
print(f"  Control change (Post - Pre):   {control_change:.2f}")
print(f"  DiD Estimate:                  {did_estimate:.2f}")

# ===== 2. Regression Analysis =====
print("\n" + "=" * 60)
print("REGRESSION ANALYSIS")
print("=" * 60)

# Basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("\nModel 1: Basic DiD (no controls)")
print(model1.summary().tables[1])

# DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("\nModel 2: DiD with controls (staffing_index, visits_index)")
print(model2.summary().tables[1])

# Cluster-robust standard errors by branch
model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 3: DiD with controls + Cluster-robust SE (by branch)")
print(model3.summary().tables[1])

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# ===== 3. Parallel Trends Check =====
print("=" * 60)
print("PARALLEL TRENDS CHECK (Pre-Period Only)")
print("=" * 60)

pre_df = df[df['after_switch'] == 0]

# Weekly averages by group
weekly_avg = pre_df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
print("\nWeekly averages (pre-period, weeks 6-14):")
print(weekly_avg)

# Test for differential pre-trends
pre_df_copy = pre_df.copy()
pre_df_copy['week_centered'] = pre_df_copy['week'] - pre_df_copy['week'].mean()
pre_df_copy['switched_x_week'] = pre_df_copy['switched_branch'] * pre_df_copy['week_centered']

import statsmodels.formula.api as smf
trend_model = smf.ols('service_score ~ switched_branch + week_centered + switched_x_week', data=pre_df_copy).fit(cov_type='cluster', cov_kwds={'groups': pre_df_copy['branch_id']})
print("\nPre-trend test (interaction of switched_branch × week):")
print(trend_model.summary().tables[1])

# ===== 4. Event Study / Weekly Effects =====
print("\n" + "=" * 60)
print("WEEKLY EFFECTS (Event Study)")
print("=" * 60)

# Create week dummies interacted with treatment
df_copy = df.copy()
for w in range(6, 23):
    df_copy[f'week_{w}'] = (df_copy['week'] == w).astype(int)
    df_copy[f'treat_week_{w}'] = df_copy[f'week_{w}'] * df_copy['switched_branch']

# Use week 14 as reference (last pre-period week)
week_vars = [f'treat_week_{w}' for w in range(6, 23) if w != 14]
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'week_{w}' for w in range(6, 23) if w != 14]) + ' + ' + ' + '.join(week_vars) + ' + staffing_index + visits_index'

event_study = smf.ols(formula, data=df_copy).fit(cov_type='cluster', cov_kwds={'groups': df_copy['branch_id']})

print("\nEvent study coefficients (treatment × week interactions, ref=week 14):")
for w in range(6, 23):
    if w == 14:
        print(f"  Week {w}: 0.00 (reference)")
    else:
        coef = event_study.params[f'treat_week_{w}']
        se = event_study.bse[f'treat_week_{w}']
        pval = event_study.pvalues[f'treat_week_{w}']
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"  Week {w}: {coef:.2f} (SE={se:.2f}, p={pval:.3f}) {sig}")

# ===== 5. Visualization Data =====
print("\n" + "=" * 60)
print("VISUALIZATION DATA")
print("=" * 60)

# Weekly averages by group for plotting
weekly_data = df.groupby(['week', 'switched_branch'])['service_score'].mean().reset_index()
weekly_data.columns = ['week', 'switched_branch', 'avg_score']
weekly_data['group'] = weekly_data['switched_branch'].map({0: 'Control (Old Procedure)', 1: 'Treatment (New Procedure)'})

print("\nWeekly averages by group:")
print(weekly_data)

# Save for plotting
weekly_data.to_csv('weekly_averages.csv', index=False)
