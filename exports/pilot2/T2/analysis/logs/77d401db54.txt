from statsmodels.formula.api import ols

# Check parallel trends - plot average service score by week for each group

print(weekly_avg.round(2))

# Run DiD regression
print("\n--- Difference-in-Differences Regression ---")
# Basic DiD model
model1 = ols('service_score ~ switched_branch * after_switch', data=df).fit()
print(model1.summary())

# DiD with branch fixed effects
print("\n--- DiD with Branch Fixed Effects ---")

# Create dummies manually for better control
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)

X = pd.DataFrame({
    'switched_branch': df['switched_branch'],
    'after_switch': df['after_switch'],
    'interaction': df['switched_branch'] * df['after_switch'],
    'staffing_index': df['staffing_index'],
    'visits_index': df['visits_index']
})

y = df['service_score']

model = OLS(y, X).fit()

# Cluster-robust SE at branch level
cov_cluster = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])
print(f"\nWith cluster-robust SE:")
