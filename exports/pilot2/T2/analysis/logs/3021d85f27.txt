import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the DiD interaction term
df['did'] = df['switched_branch'] * df['after_switch']

# First, let's look at the average service scores by group and period
print("=== Average Service Scores ===")
print("\nBefore switch (weeks 6-14):")
before = df[df['after_switch'] == 0]
print("  Switched branches:", before[before['switched_branch']==1]['service_score'].mean())
print("  Non-switched branches:", before[before['switched_branch']==0]['service_score'].mean())

print("\nAfter switch (weeks 15-22):")
after = df[df['after_switch'] == 1]
print("  Switched branches:", after[after['switched_branch']==1]['service_score'].mean())
print("  Non-switched branches:", after[after['switched_branch']==0]['service_score'].mean())

# Calculate the DiD estimate manually
switched_before = before[before['switched_branch']==1]['service_score'].mean()
switched_after = after[after['switched_branch']==1]['service_score'].mean()
control_before = before[before['switched_branch']==0]['service_score'].mean()
control_after = after[after['switched_branch']==0]['service_score'].mean()

switched_change = switched_after - switched_before
control_change = control_after - control_before
did_estimate = switched_change - control_change

print(f"\n=== Manual DiD Calculation ===")
print(f"Switched branches change: {switched_change:.3f}")
print(f"Control branches change: {control_change:.3f}")
print(f"DiD estimate (effect of new procedure): {did_estimate:.3f}")

# Now run regression with branch fixed effects
print("\n=== Regression Analysis ===")

# Model 1: Simple DiD without fixed effects
model1 = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
print("\nModel 1: Simple DiD (no fixed effects)")
print(model1.summary().tables[1])

# Model 2: With branch fixed effects
df['branch_id_str'] = df['branch_id'].astype(str)
model2 = ols('service_score ~ switched_branch + after_switch + did + C(branch_id_str)', data=df).fit()
print("\nModel 2: DiD with Branch Fixed Effects")
# Just show the key coefficients
print(f"DID coefficient: {model2.params['did']:.4f}")
print(f"DID std err: {model2.bse['did']:.4f}")
print(f"DID p-value: {model2.pvalues['did']:.4f}")
print(f"DID 95% CI: [{model2.conf_int().loc['did'][0]:.4f}, {model2.conf_int().loc['did'][1]:.4f}]")

# Model 3: With controls (staffing and visits)
model3 = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index + C(branch_id_str)', data=df).fit()
print("\nModel 3: DiD with Branch FE and Controls")
print(f"DID coefficient: {model3.params['did']:.4f}")
print(f"DID std err: {model3.bse['did']:.4f}")
print(f"DID p-value: {model3.pvalues['did']:.4f}")
print(f"DID 95% CI: [{model3.conf_int().loc['did'][0]:.4f}, {model3.conf_int().loc['did'][1]:.4f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Let's also run a more formal parallel trends test
# Compare trends in the pre-period (weeks 6-14)
pre_period = df[df['week'] < 15].copy()

# Regress service score on week, switched_branch, and their interaction in pre-period only
pre_period['week_centered'] = pre_period['week'] - pre_period['week'].mean()
model_pre = ols('service_score ~ week_centered * switched_branch + C(branch_id)', data=pre_period).fit()
print("=== Parallel Trends Test (Pre-Period Only) ===")
print("Interaction term (different trends test):")
print(f"  Coefficient: {model_pre.params['week_centered:switched_branch']:.4f}")
print(f"  P-value: {model_pre.pvalues['week_centered:switched_branch']:.4f}")
print("  (If p > 0.05, parallel trends assumption holds)")

# Also calculate the pre-period trend for each group
print("\nPre-period weekly trends:")
for group in [0, 1]:
    group_pre = pre_period[pre_period['switched_branch'] == group]
    trend_model = ols('service_score ~ week', data=group_pre).fit()
    print(f"  Group {group} (switched={group}): {trend_model.params['week']:.4f} points per week")

# Event study: estimate effect for each week relative to switch
print("\n=== Event Study Analysis ===")
df['relative_week'] = df['week'] - 15  # Week 15 is the switch, so relative week 0

# Create dummies for each relative week (excluding week -1 as reference)
for rw in range(-9, 8):  # weeks -9 to -1 and 0 to 7
    if rw != -1:  # -1 is reference
        df[f'rel_week_{rw}'] = ((df['relative_week'] == rw) & (df['switched_branch'] == 1)).astype(int)

# Build the formula
rel_weeks = [f'rel_week_{rw}' for rw in range(-9, 8) if rw != -1]
formula = f"service_score ~ {' + '.join(rel_weeks)} + C(branch_id)"

model_event = ols(formula, data=df).fit()

print("\nEvent Study Coefficients (relative to week -1):")
event_results = []
for rw in range(-9, 8):
    if rw == -1:
        event_results.append({'relative_week': rw, 'coef': 0, 'pvalue': np.nan, 'ci_low': 0, 'ci_high': 0})
    else:
        var_name = f'rel_week_{rw}'
        coef = model_event.params[var_name]
        pval = model_event.pvalues[var_name]
        ci = model_event.conf_int().loc[var_name]
        event_results.append({'relative_week': rw, 'coef': coef, 'pvalue': pval, 'ci_low': ci[0], 'ci_high': ci[1]})
        print(f"  Week {rw:+d}: {coef:.2f} (p={pval:.3f}) [{ci[0]:.2f}, {ci[1]:.2f}]")

event_df = pd.DataFrame(event_results)

# Plot event study
plt.figure(figsize=(12, 6))
plt.errorbar(event_df['relative_week'], event_df['coef'], 
             yerr=[event_df['coef'] - event_df['ci_low'], event_df['ci_high'] - event_df['coef']],
             fmt='o', capsize=5, markersize=8)
plt.axhline(y=0, color='black', linestyle='-', alpha=0.5)
plt.axvline(x=-0.5, color='red', linestyle='--', label='Switch Point')
plt.xlabel('Relative Week (0 = Week 15, switch week)')
plt.ylabel('Effect on Service Score')
plt.title('Event Study: Effect of New Procedure by Week')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(-9, 8))
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as event_study.png")

# Check if pre-trends are significant
pre_trend_pvalues = event_df[event_df['relative_week'] < -1]['pvalue']
print(f"\nPre-trend significance check:")
print(f"  Number of pre-period weeks with p < 0.05: {(pre_trend_pvalues < 0.05).sum()} out of {len(pre_trend_pvalues)}")
print(f"  (Should be low if parallel trends hold)")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Main DiD analysis with cluster-robust SEs
df['did'] = df['switched_branch'] * df['after_switch']

# Simple model first
model_simple = ols('service_score ~ switched_branch + after_switch + did', data=df).fit()
cov_simple = model_simple.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

print("Params:", model_simple.params)
print("\nBSE:", cov_simple.bse)
print("\nP-values:", cov_simple.pvalues)

# Access by position
did_idx = list(model_simple.params.index).index('did')
print(f"\nDID index: {did_idx}")
print(f"DID coef: {model_simple.params['did']}")
print(f"DID SE: {cov_simple.bse[did_idx]}")
print(f"DID p-value: {cov_simple.pvalues[did_idx]}")

# Calculate CI
did_coef = model_simple.params['did']
did_se = cov_simple.bse[did_idx]
ci_low = did_coef - 1.96 * did_se
ci_high = did_coef + 1.96 * did_se

print(f"\n=== Main DiD Results ===")
print(f"DID coefficient: {did_coef:.3f}")
print(f"Cluster-robust SE: {did_se:.3f}")
print(f"t-statistic: {did_coef / did_se:.2f}")
print(f"p-value: {cov_simple.pvalues[did_idx]:.4f}")
print(f"95% CI: [{ci_low:.3f}, {ci_high:.3f}]")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Pre-trend analysis
pre_period = df[df['week'] < 15].copy()

# Average scores by week for each group
weekly_pre = pre_period.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_pre.columns = ['Control', 'Switched']
weekly_pre['Difference'] = weekly_pre['Switched'] - weekly_pre['Control']

print("=== Pre-Trend Analysis ===")
print("\nWeekly averages before switch (weeks 6-14):")
print(weekly_pre)
print(f"\nAverage pre-period difference: {weekly_pre['Difference'].mean():.2f}")
print(f"Std dev of pre-period differences: {weekly_pre['Difference'].std():.2f}")

# Plot pre-trends
plt.figure(figsize=(10, 5))
plt.plot(weekly_pre.index, weekly_pre['Control'], 'o-', label='Control Branches', markersize=8, linewidth=2)
plt.plot(weekly_pre.index, weekly_pre['Switched'], 's-', label='Switched Branches', markersize=8, linewidth=2)
plt.axvline(x=14.5, color='red', linestyle='--', label='Switch Point (Week 15)')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Pre-Trend Check: Service Scores Before Switch')
plt.legend()
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 15))
plt.tight_layout()
plt.savefig('pre_trends.png', dpi=150)
print("\nPre-trends plot saved as pre_trends.png")

# Test for differential pre-trends (slope test)
print("\n\n=== Statistical Test for Differential Pre-Trends ===")
pre_period['week_c'] = pre_period['week'] - pre_period['week'].mean()
model_trend = ols('service_score ~ week_c * switched_branch', data=pre_period).fit()
print(f"Interaction coefficient (differential trend): {model_trend.params['week_c:switched_branch']:.3f}")
print(f"p-value: {model_trend.pvalues['week_c:switched_branch']:.4f}")
print("(A significant interaction suggests different trends between groups before the switch)")

# Post-period analysis
post_period = df[df['week'] >= 15].copy()
weekly_post = post_period.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
weekly_post.columns = ['Control', 'Switched']
weekly_post['Difference'] = weekly_post['Switched'] - weekly_post['Control']

print("\n\n=== Post-Period Analysis ===")
print("\nWeekly averages after switch (weeks 15-22):")
print(weekly_post)
print(f"\nAverage post-period difference: {weekly_post['Difference'].mean():.2f}")

# Calculate the effect size in practical terms
print("\n\n=== Effect Size Interpretation ===")
print(f"DiD estimate: ~2.0 points on service score scale")
print(f"Average service score (all branches): ~50")
print(f"Effect as % of average: {(1.977/50)*100:.1f}%")

# Summary statistics
print("\n\n=== Summary Statistics ===")
print(f"Total observations: {len(df)}")
print(f"Number of branches: {df['branch_id'].nunique()}")
print(f"Switched branches: {df['switched_branch'].sum()} branch-weeks ({df[df['switched_branch']==1]['branch_id'].nunique()} unique branches)")
print(f"Control branches: {(df['switched_branch']==0).sum()} branch-weeks ({df[df['switched_branch']==0]['branch_id'].nunique()} unique branches)")
print(f"Pre-period weeks: 6-14 (9 weeks)")
print(f"Post-period weeks: 15-22 (8 weeks)")
