import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

df = pd.read_csv('branches.csv')

# Event study: interact switched_branch with each week dummy
# Use week 14 as reference (last pre-treatment week)
df['week_cat'] = df['week'].astype('category')

# Create week dummies and interact with switched_branch
week_dummies = pd.get_dummies(df['week'], prefix='week', drop_first=False)
df = pd.concat([df, week_dummies], axis=1)

# For event study, we'll use week 14 as baseline
# Create interaction terms: switched_branch * week_dummies (excluding week 14)
for w in range(6, 23):
    if w != 14:
        df[f'switched_x_week{w}'] = df['switched_branch'] * df[f'week_{w}']

# Build formula for event study
pre_weeks = [f'week_{w}' for w in range(6, 14)]  # weeks 6-13
post_weeks = [f'week_{w}' for w in range(15, 23)]  # weeks 15-22
interactions = [f'switched_x_week{w}' for w in range(6, 23) if w != 14]

# Event study regression (without controls first)
formula = f"service_score ~ switched_branch + {' + '.join(pre_weeks + post_weeks)} + {' + '.join(interactions)}"
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

# Extract coefficients for interaction terms (these are the event study estimates)
event_results = {}
for w in range(6, 23):
    if w == 14:
        event_results[w] = {'coef': 0, 'se': 0, 'ci_low': 0, 'ci_high': 0}  # reference
    else:
        coef_name = f'switched_x_week{w}'
        coef = event_model.params[coef_name]
        se = event_model.bse[coef_name]
        event_results[w] = {
            'coef': coef,
            'se': se,
            'ci_low': coef - 1.96*se,
            'ci_high': coef + 1.96*se
        }

print("Event Study Results (coefficients relative to week 14):")
print(f"{'Week':<6} {'Coef':>8} {'SE':>8} {'95% CI Low':>12} {'95% CI High':>12}")
print("-" * 50)
for w in range(6, 23):
    r = event_results[w]
    print(f"{w:<6} {r['coef']:>8.3f} {r['se']:>8.3f} {r['ci_low']:>12.3f} {r['ci_high']:>12.3f}")

# Create event study plot
weeks = list(event_results.keys())
coefs = [event_results[w]['coef'] for w in weeks]
ci_low = [event_results[w]['ci_low'] for w in weeks]
ci_high = [event_results[w]['ci_high'] for w in weeks]

fig, ax = plt.subplots(figsize=(10, 6))
ax.plot(weeks, coefs, 'o-', label='DiD Estimate', color='blue')
ax.fill_between(weeks, ci_low, ci_high, alpha=0.3, color='blue', label='95% CI')

ax.axhline(y=0, color='gray', linestyle='-', alpha=0.5)
ax.set_xlabel('Week')
ax.set_ylabel('Service Score Effect (relative to week 14)')
ax.set_title('Event Study: Effect of New Procedure on Service Score')
ax.legend()
ax.set_xticks(weeks)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved to event_study.png")

# Test weeks 6-13 interactions

hypotheses = [f"{coef} = 0" for coef in pre_treatment_interactions]
