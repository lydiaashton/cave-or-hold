df['treatment_post'] = df['switched_branch'] * df['after_switch']

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# DiD with covariates
model_cov = ols('service_score ~ treatment_post + C(branch_id) + after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== DiD with Covariates (Clustered SE) ===")
print(f"Treatment effect: {model_cov.params['treatment_post']:.2f}")
print(f"Std Error: {model_cov.bse['treatment_post']:.2f}")
print(f"P-value: {model_cov.pvalues['treatment_post']:.3f}")
print(f"95% CI: [{model_cov.conf_int().loc['treatment_post'][0]:.2f}, {model_cov.conf_int().loc['treatment_post'][1]:.2f}]")

# Event study - check for pre-trends
# Create week dummies interacted with treatment
df['relative_week'] = df['week'] - 15  # week 15 is the switch week (0)

# Create indicators for each week relative to switch, interacted with treatment
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    df[f'treat_week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (omitting week -1 as reference)
event_study_vars = [f'treat_week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(branch_id) + C(week)'
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\n=== Event Study Results (Pre-trends Check) ===")
print("Coefficients for treatment interacted with week indicators (reference: week -1):")
for w in range(-9, 8):
    if w != -1:
        var = f'treat_week_{w}'
        if var in model_event.params:
            coef = model_event.params[var]
            se = model_event.bse[var]
            pval = model_event.pvalues[var]
            period = "Pre" if w < 0 else "Post"
            sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
            print(f"Week {w:2d} ({period:4s}): coef = {coef:6.2f}, SE = {se:.2f}, p = {pval:.3f} {sig}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create interaction term
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Event study - check for pre-trends
# Create week dummies interacted with treatment, using clean names
df['relative_week'] = df['week'] - 15  # week 15 is the switch week (0)

# Create indicators for each week relative to switch, interacted with treatment
# Use clean variable names without negative signs
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    clean_name = f"treat_w{w:02d}"  # e.g., treat_w-09, treat_w00, treat_w07
    df[clean_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression (omitting week -1 as reference)
event_study_vars = [f'treat_w{w:02d}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ ' + ' + '.join(event_study_vars) + ' + C(branch_id) + C(week)'
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results (Pre-trends Check) ===")
print("Coefficients for treatment interacted with week indicators (reference: week -1):")
pre_trend_pvals = []
post_trend_pvals = []
for w in range(-9, 8):
    if w != -1:
        var = f'treat_w{w:02d}'
        if var in model_event.params:
            coef = model_event.params[var]
            se = model_event.bse[var]
            pval = model_event.pvalues[var]
            period = "Pre" if w < 0 else "Post"
            sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
            print(f"Week {w:2d} ({period:4s}): coef = {coef:6.2f}, SE = {se:.2f}, p = {pval:.3f} {sig}")
            if w < 0:
                pre_trend_pvals.append(pval)
            else:
                post_trend_pvals.append(pval)

print(f"\nPre-trend p-values: {pre_trend_pvals}")
print(f"Any pre-trends significant at 10%? {any(p < 0.1 for p in pre_trend_pvals)}")

# Create event study plot
import matplotlib.pyplot as plt

weeks = [w for w in range(-9, 8) if w != -1]
coefs = [model_event.params[f'treat_w{w:02d}'] for w in weeks]

fig, ax = plt.subplots(figsize=(12, 6))

ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('event_study.png', dpi=150)
print("\nEvent study plot saved as 'event_study.png'")

# Create indicators with clean names that don't start with numbers
# Use prefix "pre" for negative weeks and "post" for non-negative
event_vars = []
for w in range(-9, 8):
    if w == -1:
        continue  # reference category
    if w < -1:
        clean_name = f"pre_{abs(w)}"  # e.g., pre_9, pre_8, ..., pre_2
    else:
        clean_name = f"post_{w}"  # e.g., post_0, post_1, ..., post_7
    df[clean_name] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)
    event_vars.append(clean_name)

formula = 'service_score ~ ' + ' + '.join(event_vars) + ' + C(branch_id) + C(week)'
model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

pre_pvals = []

post_ses = []

    if w < -1:
        var = f"pre_{abs(w)}"
        period = "Pre"
    else:
        var = f"post_{w}"
        period = "Post"
    
    coef = model_event.params[var]
    se = model_event.bse[var]
    pval = model_event.pvalues[var]
    sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
    print(f"Week {w:2d} ({period:4s}): coef = {coef:6.2f}, SE = {se:.2f}, p = {pval:.3f} {sig}")
    
    if w < -1:
        pre_weeks.append(w)
        pre_coefs.append(coef)
        pre_ses.append(se)
        pre_pvals.append(pval)
    else:

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment_post'] = df['switched_branch'] * df['after_switch']

# Multiple model specifications to assess robustness
print("=== Comparing Model Specifications ===\n")

# Model 1: Simple DiD (no fixed effects, no clustering)
m1 = ols('service_score ~ treatment_post + switched_branch + after_switch', data=df).fit()
print("Model 1: Simple DiD (no FE, no clustering)")
print(f"  Coef: {m1.params['treatment_post']:.2f}, SE: {m1.bse['treatment_post']:.2f}, p: {m1.pvalues['treatment_post']:.3f}")

# Model 2: With branch FE, no clustering
m2 = ols('service_score ~ treatment_post + C(branch_id) + after_switch', data=df).fit()
print("\nModel 2: Branch FE, no clustering")
print(f"  Coef: {m2.params['treatment_post']:.2f}, SE: {m2.bse['treatment_post']:.2f}, p: {m2.pvalues['treatment_post']:.3f}")

# Model 3: With branch FE, clustered SE
m3 = ols('service_score ~ treatment_post + C(branch_id) + after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 3: Branch FE, clustered SE (preferred)")
print(f"  Coef: {m3.params['treatment_post']:.2f}, SE: {m3.bse['treatment_post']:.2f}, p: {m3.pvalues['treatment_post']:.3f}")

# Model 4: With covariates, clustered SE
m4 = ols('service_score ~ treatment_post + C(branch_id) + after_switch + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\nModel 4: Branch FE + covariates, clustered SE")
print(f"  Coef: {m4.params['treatment_post']:.2f}, SE: {m4.bse['treatment_post']:.2f}, p: {m4.pvalues['treatment_post']:.3f}")

# Model 5: Branch-level aggregation (one obs per branch)
branch_means = df.groupby(['branch_id', 'switched_branch']).apply(
    lambda x: pd.Series({
        'pre_score': x[x['after_switch']==0]['service_score'].mean(),
        'post_score': x[x['after_switch']==1]['service_score'].mean()
    })
).reset_index()
branch_means['change'] = branch_means['post_score'] - branch_means['pre_score']

from scipy import stats
treat_change = branch_means[branch_means['switched_branch']==1]['change']
control_change = branch_means[branch_means['switched_branch']==0]['change']

t_stat, p_val = stats.ttest_ind(treat_change, control_change, equal_var=False)
print("\nModel 5: Branch-level change (t-test)")
print(f"  Treatment mean change: {treat_change.mean():.2f}")
print(f"  Control mean change: {control_change.mean():.2f}")
print(f"  Diff: {treat_change.mean() - control_change.mean():.2f}")
print(f"  p-value (Welch's t-test): {p_val:.3f}")

# Bootstrap for more robust inference
print("\n=== Bootstrap Analysis (10,000 iterations) ===")
np.random.seed(42)
bootstrap_diffs = []
for i in range(10000):
    # Resample branches with replacement
    sampled_branches = branch_means.sample(n=len(branch_means), replace=True)
    treat = sampled_branches[sampled_branches['switched_branch']==1]['change'].mean()
    control = sampled_branches[sampled_branches['switched_branch']==0]['change'].mean()
    bootstrap_diffs.append(treat - control)

bootstrap_diffs = np.array(bootstrap_diffs)
ci_lower = np.percentile(bootstrap_diffs, 2.5)
ci_upper = np.percentile(bootstrap_diffs, 97.5)
p_boot = np.mean(bootstrap_diffs <= 0)  # one-sided p-value for positive effect

print(f"Bootstrap mean effect: {np.mean(bootstrap_diffs):.2f}")
print(f"Bootstrap 95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"Bootstrap one-sided p-value (effect > 0): {p_boot:.3f}")
print(f"Probability effect is positive: {(bootstrap_diffs > 0).mean() * 100:.1f}%")
