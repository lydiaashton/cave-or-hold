import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression
model = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit()
print("=== Basic DiD Regression ===")
print(model.summary())

# Now with branch fixed effects and week fixed effects
# This is more robust
df_with_dummies = df.copy()

# Create branch and week dummies
branch_dummies = pd.get_dummies(df_with_dummies['branch_id'], prefix='branch', drop_first=True)
week_dummies = pd.get_dummies(df_with_dummies['week'], prefix='week', drop_first=True)

# Combine
X = pd.concat([df_with_dummies[['switched_branch', 'after_switch', 'treatment']], 
               branch_dummies, week_dummies], axis=1)
y = df_with_dummies['service_score']

# Add constant
X = sm.add_constant(X)

# Fit the model
fe_model = sm.OLS(y, X).fit()
print("\n=== DiD Regression with Branch and Week Fixed Effects ===")
print("Treatment effect:", fe_model.params['treatment'])
print("Standard error:", fe_model.bse['treatment'])
print("t-statistic:", fe_model.tvalues['treatment'])
print("p-value:", fe_model.pvalues['treatment'])
print("95% CI:", fe_model.conf_int().loc['treatment'])

    t_stat, p_val = stats.ttest_ind(
