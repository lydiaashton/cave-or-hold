treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print(f"Treatment group (new procedure): Before={treatment_before:.2f}, After={treatment_after:.2f}")
print(f"  -> Treatment change: {treatment_after - treatment_before:.2f}")

print(f"\nDifference-in-Differences Estimate: {did_effect:.2f}")

# Try different model specifications
print("=== COMPARING MODEL SPECIFICATIONS ===\n")

# Model 1: Simple DiD (no FE, no controls)
m1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Model 1: Simple DiD (no FE, no controls)")
print(f"  Treatment coef: {m1.params['treatment']:.2f}, SE: {m1.bse['treatment']:.2f}, p: {m1.pvalues['treatment']:.3f}")

# Model 2: DiD with branch FE (no controls)
m2 = ols('service_score ~ switched_branch + after_switch + treatment + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Model 2: DiD + Branch FE (no controls)")
print(f"  Treatment coef: {m2.params['treatment']:.2f}, SE: {m2.bse['treatment']:.2f}, p: {m2.pvalues['treatment']:.3f}")

# Model 3: DiD with branch FE + controls
m3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index + C(branch_id)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Model 3: DiD + Branch FE + Controls")
print(f"  Treatment coef: {m3.params['treatment']:.2f}, SE: {m3.bse['treatment']:.2f}, p: {m3.pvalues['treatment']:.3f}")

# Model 4: Just treatment and branch FE (most parsimonious)
m4 = ols('service_score ~ treatment + C(branch_id) + C(week)', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Model 4: Treatment + Branch FE + Week FE")
print(f"  Treatment coef: {m4.params['treatment']:.2f}, SE: {m4.bse['treatment']:.2f}, p: {m4.pvalues['treatment']:.3f}")

print("\n=== RAW DIFFERENCE-IN-DIFFERENCES CALCULATION ===")
# Simple 2x2 DiD
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treatment_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treatment_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

control_change = control_after - control_before
treatment_change = treatment_after - treatment_before

print(f"Treatment: {treatment_before:.2f} -> {treatment_after:.2f} (change: {treatment_change:.2f})")

np.random.seed(42)
n_bootstrap = 1000
did_boot = []

for i in range(n_bootstrap):
    # Sample branches with replacement
    sampled_branches = np.random.choice(df['branch_id'].unique(), size=df['branch_id'].nunique(), replace=True)
    boot_df = df[df['branch_id'].isin(sampled_branches)]
    
    cb = boot_df[(boot_df['switched_branch']==0) & (boot_df['after_switch']==0)]['service_score'].mean()
    ca = boot_df[(boot_df['switched_branch']==0) & (boot_df['after_switch']==1)]['service_score'].mean()

    ta = boot_df[(boot_df['switched_branch']==1) & (boot_df['after_switch']==1)]['service_score'].mean()
    
    did_boot.append((ta - tb) - (ca - cb))
