import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Pre-trends test: interact switched_branch with week dummies for pre-period only
pre_data = df[df['week'] <= 14].copy()
pre_data['week_cat'] = pre_data['week'].astype('category')

# Test if treatment and control had different trends before the switch
# Using week 6 as reference
model_pre = ols('service_score ~ switched_branch * week_cat', data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})

print("=== Pre-Trends Test (Interaction with Week Dummies, Pre-Period Only) ===")
# Look at the interaction terms - they should not be significant if parallel trends hold
interaction_terms = [c for c in model_pre.params.index if 'switched_branch:week_cat' in c]
print("\nInteraction coefficients (should be non-significant for parallel trends):")
for term in interaction_terms:
    print(f"  {term}: coef={model_pre.params[term]:.3f}, p={model_pre.pvalues[term]:.3f}")

# Get confidence interval for main DiD effect
model_main = ols('service_score ~ switched_branch * after_switch', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
did_coef = model_main.params['switched_branch:after_switch']
did_ci = model_main.conf_int().loc['switched_branch:after_switch']
print(f"\n=== DiD Effect Size ===")
print(f"Point estimate: {did_coef:.3f}")
print(f"95% CI: [{did_ci[0]:.3f}, {did_ci[1]:.3f}]")

# Calculate effect relative to baseline
baseline = df[(df['switched_branch'] == 0) & (df['after_switch'] == 0)]['service_score'].mean()
print(f"\nEffect as % of baseline control group mean: {(did_coef/baseline)*100:.2f}%")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study specification - relative time to switch
# Week 14 is the last pre-switch week, so relative_time = week - 14
df['relative_time'] = df['week'] - 14

# Create dummies for each relative time period (excluding period -1 as reference)
# relative_time: -8 to -1 (pre), 1 to 8 (post)
for rt in range(-8, 9):
    if rt != 0:  # Skip 0 as it doesn't exist (week 14 is the reference)
        df[f'rel_time_{rt}'] = (df['relative_time'] == rt).astype(int)

# Actually, let's use week 14 (relative_time=0) as the reference
# Create dummies for all other periods
event_study_vars = []
for rt in range(-8, 9):
    if rt != 0:
        event_study_vars.append(f'rel_time_{rt}')

# Build formula
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'switched_branch:rel_time_{rt}' for rt in range(-8, 9) if rt != 0])
# Also need the main effects for relative time
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'rel_time_{rt}' for rt in range(-8, 9) if rt != 0]) + ' + ' + ' + '.join([f'switched_branch:rel_time_{rt}' for rt in range(-8, 9) if rt != 0])

model_es = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results ===")
print("Coefficients for switched_branch:rel_time_X (effect relative to week 14):")
for rt in range(-8, 9):
    if rt != 0:
        term = f'switched_branch:rel_time_{rt}'
        if term in model_es.params:
            coef = model_es.params[term]
            pval = model_es.pvalues[term]
            sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
            period = "Pre" if rt < 0 else "Post"
            print(f"  Week {14+rt:2d} (rel {rt:+2d}, {period:4s}): {coef:6.3f} (p={pval:.3f}) {sig}")

df['relative_time'] = df['week'] - 14  # Week 14 is reference (0)

# Using C() to treat relative_time as categorical
model_es = sm.OLS.from_formula(
    'service_score ~ switched_branch + C(relative_time) + switched_branch:C(relative_time)',
    data=df
).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
