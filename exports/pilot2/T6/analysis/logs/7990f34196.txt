from statsmodels.formula.api import ols

# Create interaction term for DiD
df['treatment_interaction'] = df['switched_branch'] * df['after_switch']

# Basic DiD regression (no controls)
print("=== Model 1: Basic DiD (no controls) ===")
model1 = ols('service_score ~ switched_branch + after_switch + treatment_interaction', data=df).fit()
print(model1.summary().tables[1])

# DiD regression with controls
print("\n=== Model 2: DiD with staffing and visits controls ===")
model2 = ols('service_score ~ switched_branch + after_switch + treatment_interaction + staffing_index + visits_index', data=df).fit()
