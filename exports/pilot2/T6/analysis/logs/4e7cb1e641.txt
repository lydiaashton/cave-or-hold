import pandas as pd
import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run DiD regression with controls
model = smf.ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Also run a version with branch fixed effects
print("\n=== DiD with Branch Fixed Effects ===")
# Create dummy variables for branches
df_with_dummies = df.copy()
branch_dummies = pd.get_dummies(df['branch_id'], prefix='branch', drop_first=True)
df_with_dummies = pd.concat([df_with_dummies, branch_dummies], axis=1)

# Get all column names for the regression
control_cols = ['switched_branch', 'after_switch', 'treatment', 'staffing_index', 'visits_index']
branch_cols = [c for c in df_with_dummies.columns if c.startswith('branch_')]
all_cols = control_cols + branch_cols

X = df_with_dummies[all_cols]
X = sm.add_constant(X)
y = df_with_dummies['service_score']

model_fe = sm.OLS(y, X).fit()
print(f"R-squared: {model_fe.rsquared:.4f}")
print(f"Treatment coefficient: {model_fe.params['treatment']:.4f}")
print(f"Treatment p-value: {model_fe.pvalues['treatment']:.6f}")
print(f"Treatment 95% CI: [{model_fe.conf_int().loc['treatment'][0]:.4f}, {model_fe.conf_int().loc['treatment'][1]:.4f}]")

import pandas as pd
import statsmodels.api as sm
import statsmodels.formula.api as smf

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Run regression with clustered standard errors at branch level
model = smf.ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("=== DiD Regression with Clustered Standard Errors (by branch) ===")
print(model.summary())

# Check balance - are the groups similar before the switch?
print("\n=== Pre-Switch Balance Check (Weeks 6-14) ===")
pre_switch = df[df['after_switch'] == 0]
for col in ['service_score', 'staffing_index', 'visits_index']:
    control_mean = pre_switch[pre_switch['switched_branch'] == 0][col].mean()
    treatment_mean = pre_switch[pre_switch['switched_branch'] == 1][col].mean()
    diff = treatment_mean - control_mean
    print(f"{col}: Control={control_mean:.2f}, Treatment={treatment_mean:.2f}, Diff={diff:.2f}")

# Check if the effect is consistent across post-switch weeks
print("\n=== Treatment Effect by Post-Switch Week ===")
post_switch = df[df['after_switch'] == 1]
for week in sorted(post_switch['week'].unique()):
    week_data = df[df['week'] == week]
    control_mean = week_data[week_data['switched_branch'] == 0]['service_score'].mean()
    treatment_mean = week_data[week_data['switched_branch'] == 1]['service_score'].mean()
    diff = treatment_mean - control_mean
    print(f"Week {week}: Control={control_mean:.2f}, Treatment={treatment_mean:.2f}, Diff={diff:.2f}")
