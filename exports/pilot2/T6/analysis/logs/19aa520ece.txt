import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load data
df = pd.read_csv('branches.csv')

# Create interaction term for DiD
df['treatment'] = df['switched_branch']
df['post'] = df['after_switch']
df['did'] = df['treatment'] * df['post']

# Run DiD regression
model = ols('service_score ~ treatment + post + did', data=df).fit()
print("=== Difference-in-Differences Regression ===")
print(model.summary())

# Calculate the DiD estimate manually to verify
control_before = df[(df['switched_branch']==0) & (df['after_switch']==0)]['service_score'].mean()
control_after = df[(df['switched_branch']==0) & (df['after_switch']==1)]['service_score'].mean()
treat_before = df[(df['switched_branch']==1) & (df['after_switch']==0)]['service_score'].mean()
treat_after = df[(df['switched_branch']==1) & (df['after_switch']==1)]['service_score'].mean()

print("\n=== Manual DiD Calculation ===")
print(f"Control before: {control_before:.2f}")
print(f"Control after: {control_after:.2f}")
print(f"Control change: {control_after - control_before:.2f}")
print(f"Treatment before: {treat_before:.2f}")
print(f"Treatment after: {treat_after:.2f}")
print(f"Treatment change: {treat_after - treat_before:.2f}")
print(f"DiD estimate: {(treat_after - treat_before) - (control_after - control_before):.2f}")

# Cluster-robust standard errors at branch level
model_clustered = ols('service_score ~ treatment + post + did', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD with Cluster-Robust SE (by branch) ===")
print(model_clustered.summary())

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load data
df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']

# Check parallel trends - plot average service scores by week for each group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

# Create the plot
plt.figure(figsize=(10, 6))
for switched in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == switched]
    label = 'Switched to New Procedure' if switched == 1 else 'Kept Old Procedure'
    plt.plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2)

plt.axvline(x=14.5, color='red', linestyle='--', label='Switch Week (15)')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Scores Over Time by Group')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('service_scores_trend.png', dpi=150)
print("Plot saved as service_scores_trend.png")

# Check pre-trend statistically (weeks 6-14 only)
pre_switch = df[df['week'] < 15].copy()
pre_switch['week_centered'] = pre_switch['week'] - 10  # center around middle of pre-period

# Test if trends were parallel before the switch
pre_model = ols('service_score ~ switched_branch * week_centered', data=pre_switch).fit(cov_type='cluster', cov_kwds={'groups': pre_switch['branch_id']})
print("\n=== Pre-Trend Test (weeks 6-14) ===")
print(pre_model.summary())

# Add control variables to main model
df['did'] = df['switched_branch'] * df['after_switch']
model_controls = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n=== DiD with Control Variables ===")
print(model_controls.summary())

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load data
df = pd.read_csv('branches.csv')
df['did'] = df['switched_branch'] * df['after_switch']

# Event study / dynamic DiD - check effect by week relative to switch
df['relative_week'] = df['week'] - 15  # week 15 is the switch (0), before is negative, after is positive

# Create week dummies relative to switch (using week 14 as reference)
for w in range(-9, 8):  # weeks -9 to +7 relative to switch
    if w != -1:  # skip week -1 (week 14) as reference
        df[f'rel_week_{w}'] = (df['relative_week'] == w).astype(int)

# Run event study regression (only treatment group interactions)
event_vars = [f'rel_week_{w}' for w in range(-9, 8) if w != -1]
formula = 'service_score ~ switched_branch + ' + ' + '.join([f'switched_branch * {v}' for v in event_vars])
event_model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Coefficients (Treatment x Relative Week) ===")
for w in range(-9, 8):
    if w != -1:
        coef_name = f'switched_branch:rel_week_{w}'
        if coef_name in event_model.params.index:
            coef = event_model.params[coef_name]
            se = event_model.bse[coef_name]
            pval = event_model.pvalues[coef_name]
            print(f"Week {15+w}: coef={coef:.2f}, SE={se:.2f}, p={pval:.3f}")

# Confidence intervals for the main DiD estimate
main_model = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
did_coef = main_model.params['did']
did_se = main_model.bse['did']
did_pval = main_model.pvalues['did']
ci_lower = did_coef - 1.96 * did_se
ci_upper = did_coef + 1.96 * did_se

print(f"\n=== Main DiD Estimate Summary ===")
print(f"Estimate: {did_coef:.2f} points")
print(f"95% CI: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {did_pval:.3f}")
print(f"Standard Error: {did_se:.2f}")

# Effect size relative to baseline
baseline = df[df['after_switch']==0]['service_score'].mean()
print(f"\nBaseline average service score: {baseline:.2f}")
print(f"Effect as % of baseline: {did_coef/baseline*100:.1f}%")

print(f"Standard error: {did_se:.3f}")

ci90_upper = did_coef + 1.645 * did_se
print(f"\n90% CI: [{ci90_lower:.2f}, {ci90_upper:.2f}]")
