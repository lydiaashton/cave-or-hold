import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create DiD term
df['did'] = df['switched_branch'] * df['after_switch']

# Run the main model with controls and clustered SE
model = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit()
model_clustered = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Extract key results
did_coef = model_clustered.params['did']
did_se = model_clustered.bse['did']
did_pvalue = model_clustered.pvalues['did']
did_ci_lower = model_clustered.conf_int().loc['did', 0]
did_ci_upper = model_clustered.conf_int().loc['did', 1]

print("=== KEY RESULTS ===")

print(f"Standard Error: {did_se:.3f}")
print(f"P-value: {did_pvalue:.6f}")
print(f"95% CI: [{did_ci_lower:.3f}, {did_ci_upper:.3f}]")

# Calculate effect size relative to baseline
baseline_avg = df[df['after_switch'] == 0]['service_score'].mean()
effect_pct = (did_coef / baseline_avg) * 100
print(f"\nBaseline average service score: {baseline_avg:.2f}")
print(f"Effect as % of baseline: {effect_pct:.2f}%")

# Check how many branches are in each group
n_treatment = df[df['switched_branch'] == 1]['branch_id'].nunique()
n_control = df[df['switched_branch'] == 0]['branch_id'].nunique()
print(f"\nNumber of treatment branches: {n_treatment}")
print(f"Number of control branches: {n_control}")

# Check post-period only effect
post_period = df[df['after_switch'] == 1].copy()
post_avg_treatment = post_period[post_period['switched_branch'] == 1]['service_score'].mean()
post_avg_control = post_period[post_period['switched_branch'] == 0]['service_score'].mean()

import pandas as pd
import numpy as np
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create DiD term
df['did'] = df['switched_branch'] * df['after_switch']

# Run the main model with controls and clustered SE
model = ols('service_score ~ switched_branch + after_switch + did + staffing_index + visits_index', data=df).fit()
model_clustered = model.get_robustcov_results(cov_type='cluster', groups=df['branch_id'])

# Print the summary to see all results
print(model_clustered.summary())

# Extract key results directly
params = model_clustered.params
bse = model_clustered.bse
pvalues = model_clustered.pvalues

did_coef = params['did']
did_se = bse['did']
did_pvalue = pvalues['did']

# Calculate CI manually
from scipy import stats
n = len(df)
k = len(params)
t_crit = stats.t.ppf(0.975, n - k)
did_ci_lower = did_coef - t_crit * did_se
did_ci_upper = did_coef + t_crit * did_se

print("\n=== KEY RESULTS ===")
print(f"DiD Coefficient: {did_coef:.3f}")
print(f"Standard Error: {did_se:.3f}")
print(f"P-value: {did_pvalue:.6f}")
print(f"95% CI: [{did_ci_lower:.3f}, {did_ci_upper:.3f}]")

# Calculate effect size relative to baseline
baseline_avg = df[df['after_switch'] == 0]['service_score'].mean()
