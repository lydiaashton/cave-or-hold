import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.api as sm
import statsmodels.formula.api as smf

# Load the data
df = pd.read_csv('branches.csv')

# Create a visualization of trends over time
plt.figure(figsize=(12, 6))

# Calculate weekly averages by group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

for switch_status in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == switch_status]
    label = 'Switched to New Procedure' if switch_status == 1 else 'Kept Old Procedure'
    plt.plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2)

plt.axvline(x=14.5, color='red', linestyle='--', label='Switch Week (15)')
plt.xlabel('Week')
plt.ylabel('Average Service Score')
plt.title('Service Score Trends: Treatment vs Control Groups')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('trends.png', dpi=150)
print("Saved trends.png")

# Run difference-in-differences regression
# Basic DiD: service_score ~ switched_branch * after_switch
model = smf.ols('service_score ~ switched_branch * after_switch', data=df).fit()
print("\n=== Basic Difference-in-Differences Regression ===")
print(model.summary())

# Add branch fixed effects
model_fe = smf.ols('service_score ~ switched_branch * after_switch + C(branch_id)', data=df).fit()
print("\n=== DiD with Branch Fixed Effects ===")
print(model_fe.summary())

# Add controls (staffing_index, visits_index) and branch fixed effects
model_full = smf.ols('service_score ~ switched_branch * after_switch + C(branch_id) + staffing_index + visits_index', data=df).fit()
print("\n=== DiD with Branch FE and Controls ===")
print(model_full.summary())

import pandas as pd
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create a visualization of trends over time
plt.figure(figsize=(12, 6))

# Calculate weekly averages by group
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()

for switch_status in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == switch_status]
    label = 'Switched to New Procedure (n=20)' if switch_status == 1 else 'Kept Old Procedure (n=14)'
    plt.plot(subset['week'], subset['service_score'], marker='o', label=label, linewidth=2, markersize=6)

plt.axvline(x=14.5, color='red', linestyle='--', linewidth=2, label='Procedure Switch (Week 15)')
plt.xlabel('Week of Year', fontsize=12)
plt.ylabel('Average Service Score', fontsize=12)
plt.title('Service Score Trends: Treatment vs Control Groups\n(Difference-in-Differences Analysis)', fontsize=14)
plt.legend(fontsize=11)
plt.grid(True, alpha=0.3)
plt.xticks(range(6, 23))
plt.tight_layout()
plt.savefig('trends.png', dpi=150)
print("Saved trends.png")

# Also check pre-trends (weeks 6-14) to assess parallel trends assumption
pre_trends = df[df['week'] <= 14].groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
print("\nPre-switch weekly averages (weeks 6-14):")
print(pre_trends.pivot(index='week', columns='switched_branch', values='service_score'))

# Calculate pre-trend differences
for week in range(6, 15):
    control = pre_trends[(pre_trends['week']==week) & (pre_trends['switched_branch']==0)]['service_score'].values[0]
