import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Check parallel trends: average service score by week and treatment group
print("=== Parallel Trends Check: Service Score by Week and Group ===")
trend = df.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()
trend.columns = ['Non-Switched', 'Switched']
print(trend.round(2))

# Calculate pre-treatment difference
pre_trend = trend.loc[6:14]
print(f"\nPre-treatment period (weeks 6-14):")
print(f"  Non-switched avg: {pre_trend['Non-Switched'].mean():.2f}")
print(f"  Switched avg: {pre_trend['Switched'].mean():.2f}")
print(f"  Difference (Switched - Non-switched): {pre_trend['Switched'].mean() - pre_trend['Non-Switched'].mean():.2f}")

# Post-treatment difference
post_trend = trend.loc[15:22]
print(f"\nPost-treatment period (weeks 15-22):")
print(f"  Non-switched avg: {post_trend['Non-Switched'].mean():.2f}")
print(f"  Switched avg: {post_trend['Switched'].mean():.2f}")
print(f"  Difference (Switched - Non-switched): {post_trend['Switched'].mean() - post_trend['Non-Switched'].mean():.2f}")

# Simple DiD estimate
pre_diff = pre_trend['Switched'].mean() - pre_trend['Non-Switched'].mean()
post_diff = post_trend['Switched'].mean() - post_trend['Non-Switched'].mean()
did_estimate = post_diff - pre_diff
print(f"\n=== Simple Difference-in-Differences Estimate ===")
print(f"Pre-treatment gap: {pre_diff:.2f}")
print(f"Post-treatment gap: {post_diff:.2f}")
print(f"DiD Estimate (effect of switch): {did_estimate:.2f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model 1: Simple DiD (no controls)
print("=== Model 1: Simple Difference-in-Differences ===")
model1 = ols('service_score ~ switched_branch + after_switch + treatment', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model1.summary())

# Model 2: DiD with controls
print("\n=== Model 2: DiD with Controls (staffing_index, visits_index) ===")
model2 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(model2.summary())

# Model 3: With branch fixed effects
print("\n=== Model 3: DiD with Branch Fixed Effects ===")
df_with_dummies = pd.get_dummies(df, columns=['branch_id'], drop_first=True)
control_vars = ['switched_branch', 'after_switch', 'treatment', 'staffing_index', 'visits_index']
branch_dummies = [col for col in df_with_dummies.columns if col.startswith('branch_id_')]
formula = 'service_score ~ ' + ' + '.join(control_vars + branch_dummies)
model3 = ols(formula, data=df_with_dummies).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"Treatment coefficient: {model3.params['treatment']:.4f}")
print(f"Standard error: {model3.bse['treatment']:.4f}")
print(f"95% CI: [{model3.conf_int().loc['treatment'][0]:.4f}, {model3.conf_int().loc['treatment'][1]:.4f}]")
print(f"P-value: {model3.pvalues['treatment']:.4f}")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')
df['treatment'] = df['switched_branch'] * df['after_switch']

# Model 3: With branch fixed effects using C() notation
print("=== Model 3: DiD with Branch Fixed Effects ===")
model3 = ols('service_score ~ switched_branch + after_switch + treatment + staffing_index + visits_index + C(branch_id)', 
             data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print(f"Treatment coefficient: {model3.params['treatment']:.4f}")
print(f"Standard error: {model3.bse['treatment']:.4f}")
print(f"95% CI: [{model3.conf_int().loc['treatment'][0]:.4f}, {model3.conf_int().loc['treatment'][1]:.4f}]")
print(f"P-value: {model3.pvalues['treatment']:.6f}")

# Let's also check pre-trends more formally - test if trends were parallel before week 15
print("\n=== Pre-Trend Analysis (Weeks 6-14 only) ===")
pre_data = df[df['week'] <= 14].copy()
pre_data['post_fake'] = (pre_data['week'] >= 12).astype(int)  # Fake treatment at week 12
pre_data['fake_treatment'] = pre_data['switched_branch'] * pre_data['post_fake']

# Test parallel trends with a fake treatment date
pre_model = ols('service_score ~ switched_branch + post_fake + fake_treatment + staffing_index + visits_index + C(branch_id)',
                data=pre_data).fit(cov_type='cluster', cov_kwds={'groups': pre_data['branch_id']})
print(f"Fake treatment coefficient (weeks 12-14): {pre_model.params['fake_treatment']:.4f}")
print(f"P-value: {pre_model.pvalues['fake_treatment']:.4f}")
print("(If not significant, supports parallel trends assumption)")

# Event study - effect by week relative to switch
print("\n=== Event Study: Treatment Effect by Week ===")
df['relative_week'] = df['week'] - 14  # Week 14 is last pre-treatment week
df['treatment_week'] = df['treatment']

# Create week-specific treatment effects (interactions)
for w in range(6, 23):
    df[f'week_{w}'] = (df['week'] == w).astype(int)
    df[f'treat_week_{w}'] = (df['switched_branch'] == 1).astype(int) * (df['week'] == w).astype(int)

# Run event study regression (excluding week 14 as reference)
treat_vars = [f'treat_week_{w}' for w in range(6, 23) if w != 14]
week_vars = [f'week_{w}' for w in range(6, 23) if w != 14]
event_formula = 'service_score ~ ' + ' + '.join(treat_vars + week_vars + ['staffing_index', 'visits_index', 'C(branch_id)'])
event_model = ols(event_formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("\nTreatment effects by week (reference: week 14):")
for w in range(6, 23):
    if w != 14:
        coef = event_model.params.get(f'treat_week_{w}', 0)
        pval = event_model.pvalues.get(f'treat_week_{w}', 1)
        marker = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        period = "PRE" if w < 15 else "POST"
        print(f"  Week {w} ({period}): {coef:.2f} (p={pval:.3f}) {marker}")
