import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Create relative week indicators (event study)
# Week 14 is the last pre-switch week, so we'll use that as reference
df['relative_week'] = df['week'] - 14

# Create indicators for each week relative to switch (excluding week 0 as reference)
# We'll create dummies for weeks -8 to -1 (pre) and 1 to 8 (post)
pre_weeks = [-8, -7, -6, -5, -4, -3, -2, -1]
post_weeks = [1, 2, 3, 4, 5, 6, 7, 8]

# Create the event study variables
for w in pre_weeks + post_weeks:
    df[f'week_{w}'] = ((df['relative_week'] == w) & (df['switched_branch'] == 1)).astype(int)

# Run event study regression
pre_vars = [f'week_{w}' for w in pre_weeks]
post_vars = [f'week_{w}' for w in post_weeks]
all_vars = pre_vars + post_vars

formula = 'service_score ~ ' + ' + '.join(all_vars) + ' + switched_branch + staffing_index + visits_index'
model = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("Event Study Regression (week 0/reference is week 14):")
print("Pre-treatment coefficients (should be near zero if parallel trends hold):")
for w in pre_weeks:
    coef = model.params[f'week_{w}']
    se = model.bse[f'week_{w}']
    pval = model.pvalues[f'week_{w}']
    print(f"  Week {14+w}: coef={coef:.3f}, SE={se:.3f}, p={pval:.3f}")

print("\nPost-treatment coefficients (treatment effect):")
for w in post_weeks:
    coef = model.params[f'week_{w}']
    se = model.bse[f'week_{w}']
    pval = model.pvalues[f'week_{w}']
    print(f"  Week {14+w}: coef={coef:.3f}, SE={se:.3f}, p={pval:.3f}")

# Test joint significance of pre-treatment coefficients
from statsmodels.stats.contrast import ContrastResults

# Create a joint test for all pre-treatment coefficients
pre_test = model.f_test([f'week_{w} = 0' for w in pre_weeks])
print(f"\nJoint test of pre-treatment coefficients (parallel trends test):")
print(f"  F-statistic: {pre_test.fvalue:.3f}, p-value: {pre_test.pvalue:.4f}")
