import pandas as pd
import numpy as np

df = pd.read_csv('branches.csv')

# Create the visualization
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Plot 1: Service score trends by group
ax1 = axes[0]
for switched in [0, 1]:
    subset = weekly_avg[weekly_avg['switched_branch'] == switched]
    label = 'Switched to New Procedure' if switched == 1 else 'Kept Old Procedure'
    ax1.plot(subset['week'], subset['avg_score'], marker='o', linewidth=2, label=label, markersize=6)

ax1.grid(True, alpha=0.3)
ax1.set_xticks(range(6, 23))

# Plot 2: Distribution of scores before and after
ax2 = axes[1]
colors = ['#2E86AB', '#A23B72']
labels = ['Switched', 'Control']

before_switched = df[(df['after_switch'] == 0) & (df['switched_branch'] == 1)]['service_score']

box_data = [before_switched, after_switched, before_control, after_control]

ax2.set_title('Service Score Distribution: Before vs After', fontsize=14, fontweight='bold')
ax2.grid(True, alpha=0.3, axis='y')

plt.tight_layout()
plt.savefig('service_score_analysis.png', dpi=150, bbox_inches='tight')
print("Chart saved as service_score_analysis.png")

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

df = pd.read_csv('branches.csv')

# Event study regression - interaction of switched_branch with each week
# Create week dummies and interact with switched_branch
df = df.copy()
for w in range(6, 23):
    df[f'week_{w}'] = (df['week'] == w).astype(int)

# Use week 6 as baseline, interact weeks 7-22 with switched_branch
# This shows the differential effect for each week relative to week 6
event_study_vars = ' + '.join([f'switched_branch:week_{w}' for w in range(7, 23)])
formula = f'service_score ~ switched_branch + {" + ".join([f"week_{w}" for w in range(7, 23)])} + {event_study_vars}'

model_event = ols(formula, data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

print("=== Event Study Results (relative to Week 6) ===")
print("Coefficients for switched_branch:week interactions:")
print("-" * 60)

for w in range(7, 23):
    coef_name = f'switched_branch:week_{w}'
    if coef_name in model_event.params:
        coef = model_event.params[coef_name]
        se = model_event.bse[coef_name]
        pval = model_event.pvalues[coef_name]
        period = "Before" if w < 15 else "AFTER"
        sig = "***" if pval < 0.01 else "**" if pval < 0.05 else "*" if pval < 0.1 else ""
        print(f"Week {w} ({period}): coef={coef:6.2f}, SE={se:.2f}, p={pval:.3f} {sig}")

# Calculate average pre and post effects
print("\n" + "=" * 60)
print("Average differential (switched - control) by period:")
pre_weeks = list(range(6, 15))
post_weeks = list(range(15, 23))

pre_diffs = []
post_diffs = []
for w in pre_weeks:
    week_data = df[df['week'] == w]
    diff = week_data[week_data['switched_branch'] == 1]['service_score'].mean() - \
           week_data[week_data['switched_branch'] == 0]['service_score'].mean()
    pre_diffs.append(diff)

for w in post_weeks:
    week_data = df[df['week'] == w]
    diff = week_data[week_data['switched_branch'] == 1]['service_score'].mean() - \
           week_data[week_data['switched_branch'] == 0]['service_score'].mean()
    post_diffs.append(diff)

print(f"Before (weeks 6-14): avg diff = {np.mean(pre_diffs):.2f}")
print(f"After (weeks 15-22): avg diff = {np.mean(post_diffs):.2f}")
print(f"Increase in differential: {np.mean(post_diffs) - np.mean(pre_diffs):.2f}")
