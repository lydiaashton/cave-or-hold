import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment_interaction'] = df['switched_branch'] * df['after_switch']

# Run basic DiD regression
model1 = ols('service_score ~ switched_branch + after_switch + treatment_interaction', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("Basic DiD Regression (clustered SEs by branch):")
print(model1.summary().tables[1])

# Run DiD with controls
model2 = ols('service_score ~ switched_branch + after_switch + treatment_interaction + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})
print("\n\nDiD Regression with controls (clustered SEs by branch):")
print(model2.summary().tables[1])

# Check post-treatment trends (weeks 15-22)
print("\nWeekly mean service scores by group (post-treatment weeks 15-22):")
post_switch = df[df['week'] >= 15].groupby(['switched_branch', 'week'])['service_score'].mean().unstack(level=0)
print(post_switch.round(2))

# Event study - check effects by week relative to switch
print("\nEvent study - DiD estimates by week:")
df['weeks_since_switch'] = df['week'] - 14  # week 14 is the last pre-switch week
event_study = []
for w in df['weeks_since_switch'].unique():
    week_data = df[df['weeks_since_switch'] == w]
    if len(week_data) > 0:
        treat_mean = week_data[week_data['switched_branch'] == 1]['service_score'].mean()
        control_mean = week_data[week_data['switched_branch'] == 0]['service_score'].mean()
        diff = treat_mean - control_mean
        event_study.append({'week': w, 'weeks_since_switch': int(w), 'treatment_mean': treat_mean, 'control_mean': control_mean, 'diff': diff})

event_df = pd.DataFrame(event_study).sort_values('weeks_since_switch')
print(event_df.round(2))

import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols

# Load the data
df = pd.read_csv('branches.csv')

# Create the interaction term for DiD
df['treatment_interaction'] = df['switched_branch'] * df['after_switch']

# Run DiD with controls and get confidence interval
model = ols('service_score ~ switched_branch + after_switch + treatment_interaction + staffing_index + visits_index', data=df).fit(cov_type='cluster', cov_kwds={'groups': df['branch_id']})

treatment_effect = model.params['treatment_interaction']
treatment_se = model.bse['treatment_interaction']
ci_lower = treatment_effect - 1.96 * treatment_se
ci_upper = treatment_effect + 1.96 * treatment_se

print("=== MAIN RESULTS ===")
print(f"Treatment effect (DiD estimate): {treatment_effect:.2f} points")
print(f"Standard error: {treatment_se:.2f}")
print(f"95% Confidence interval: [{ci_lower:.2f}, {ci_upper:.2f}]")
print(f"P-value: {model.pvalues['treatment_interaction']:.4f}")

# Calculate percentage improvement relative to baseline
baseline_score = df[(df['switched_branch'] == 1) & (df['after_switch'] == 0)]['service_score'].mean()
pct_improvement = (treatment_effect / baseline_score) * 100
print(f"\nBaseline (treatment group, pre-switch): {baseline_score:.2f}")
print(f"Percentage improvement: {pct_improvement:.1f}%")

# Check if effect is consistent across branches
print("\n=== EFFECT BY BRANCH SIZE (quartiles of observations) ===")
branch_counts = df.groupby('branch_id').size()
quartiles = branch_counts.quantile([0.25, 0.50, 0.75])
print(f"Branch observation counts - Q1: {quartiles[0.25]}, Median: {quartiles[0.50]}, Q3: {quartiles[0.75]}")

# Check distribution of service scores
print("\n=== SERVICE SCORE DISTRIBUTION ===")
print(f"Overall mean: {df['service_score'].mean():.2f}")
print(f"Overall std: {df['service_score'].std():.2f}")
print(f"Min: {df['service_score'].min():.2f}, Max: {df['service_score'].max():.2f}")

# Check balance of controls between treatment and control
print("\n=== CONTROL VARIABLE BALANCE ===")
control_stats = df.groupby('switched_branch')[['staffing_index', 'visits_index']].agg(['mean', 'std'])
print(control_stats.round(3))

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('branches.csv')

# Create weekly averages for plotting
weekly_avg = df.groupby(['switched_branch', 'week'])['service_score'].mean().reset_index()
weekly_avg['group'] = weekly_avg['switched_branch'].map({0: 'Control (Old Procedure)', 1: 'Treatment (New Procedure)'})

# Create the plot
fig, ax = plt.subplots(figsize=(12, 6))

# Plot lines for each group
for group_name, group_data in weekly_avg.groupby('group'):
    ax.plot(group_data['week'], group_data['service_score'], marker='o', linewidth=2.5, label=group_name)

# Add vertical line at switch point
ax.axvline(x=14.5, color='gray', linestyle='--', linewidth=2, label='Switch implemented (Week 15)')

# Add shading for pre/post periods
ax.axvspan(6, 14.5, alpha=0.1, color='blue', label='Pre-switch period')
ax.axvspan(14.5, 22, alpha=0.1, color='green', label='Post-switch period')

# Labels and title
ax.set_xlabel('Week of Year', fontsize=12)
ax.set_ylabel('Mean Service Score', fontsize=12)
ax.set_title('Library Service Scores: New vs. Old Shelving Procedure\nDifference-in-Differences Analysis', fontsize=14, fontweight='bold')
ax.legend(loc='upper left', fontsize=10)
ax.set_xticks(range(6, 23))
ax.grid(True, alpha=0.3)

# Add annotation for the treatment effect
ax.annotate('Treatment Effect: +1.96 points\n(3.9% improvement)', 
            xy=(18, 52), xytext=(16, 48),
            arrowprops=dict(arrowstyle='->', color='red', lw=2),
            fontsize=11, fontweight='bold', color='darkred',
            bbox=dict(boxstyle='round', facecolor='white', edgecolor='red', alpha=0.8))

plt.tight_layout()
plt.savefig('service_score_trends.png', dpi=150, bbox_inches='tight')
print("Plot saved as 'service_score_trends.png'")
plt.show()
