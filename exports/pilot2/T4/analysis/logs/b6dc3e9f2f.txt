after_control = df[(df['switched_branch'] == 0) & (df['after_switch'] == 1)]['service_score'].mean()

pre_data = df[df['week'] <= 14].copy()
weekly_avg = pre_data.groupby(['week', 'switched_branch'])['service_score'].mean().unstack()

print(weekly_avg)

# Calculate the week-by-week difference
weekly_avg['diff'] = weekly_avg[1] - weekly_avg[0]

pre_data['did_trend'] = pre_data['switched_branch'] * pre_data['week_centered']

trend_model = ols('service_score ~ switched_branch + week_centered + did_trend', data=pre_data).fit()
print("\nPre-trend test (interaction of treatment x week trend):")

event_study_df = df.copy()

event_formula = 'service_score ~ C(relative_week) * switched_branch + staffing_index + visits_index + C(branch_id)'
